    31→
    32→/// Concrete implementation of [OfflineRepository] that coordinates
    33→/// between local and remote data sources with an offline-first strategy.
    34→///
    35→/// When the device is online:
    36→///   - Read operations try the remote source first and cache the result.
    37→///   - Write operations go to the remote source and update the local cache.
    38→///
    39→/// When the device is offline:
    40→///   - Read operations fall back to the local data source.
    41→///   - Write operations are persisted locally and enqueued in the sync
    42→///     queue for automatic replay when connectivity is restored.
    43→class OfflineRepositoryImpl implements OfflineRepository {
    44→  OfflineRepositoryImpl({
    45→    required OfflineLocalDataSource localDataSource,
    46→    required OfflineRemoteDataSource remoteDataSource,
    47→    required NetworkInfo networkInfo,
    48→    required CacheManager cacheManager,
    49→    required SyncEngine syncEngine,
    50→  })  : _localDataSource = localDataSource,
    51→        _remoteDataSource = remoteDataSource,
    52→        _networkInfo = networkInfo,
    53→        _cacheManager = cacheManager,
    54→        _syncEngine = syncEngine;
    55→
    56→  final OfflineLocalDataSource _localDataSource;
    57→  final OfflineRemoteDataSource _remoteDataSource;
    58→  final NetworkInfo _networkInfo;
    59→  final CacheManager _cacheManager;
    60→  final SyncEngine _syncEngine;

    62→  // ═══════════════════════════════════════════════════════════════════════
    63→  // Helper: Convert exceptions to Failures
    64→  // ═══════════════════════════════════════════════════════════════════════
    65→
    66→  Failure _mapExceptionToFailure(Object e) {
    67→    if (e is AuthException) {
    68→      return Failure.auth(message: e.message, code: e.code);
    69→    } else if (e is ServerException) {
    70→      return Failure.server(
    71→        message: e.message,
    72→        statusCode: e.statusCode,
    73→        data: e.data,
    74→      );
    75→    } else if (e is CacheException) {
    76→      return Failure.cache(message: e.message);
    77→    } else if (e is NetworkException) {
    78→      return Failure.network(message: e.message);
    79→    } else if (e is ValidationException) {
    80→      return Failure.validation(
    81→        message: e.message,
    82→        fieldErrors: e.fieldErrors,
    83→      );
    84→    } else if (e is NotFoundException) {
    85→      return Failure.notFound(message: e.message);
    86→    } else if (e is UnauthorizedException) {
    87→      return Failure.unauthorized(message: e.message);
    88→    } else if (e is ForbiddenException) {
    89→      return Failure.forbidden(message: e.message);
    90→    } else {
    91→      AppLogger.error(
    92→        'Unexpected exception in OfflineRepositoryImpl',
    93→        error: e,
    94→      );
    95→      return Failure.server(
    96→        message: 'An unexpected error occurred: $e',
    97→        statusCode: 500,
    98→      );
    99→    }
   100→  }

   106→  @override
   107→  Future<Result<SyncStatusInfo>> getSyncStatus(String userId) async {
   108→    try {
   109→      // Sync status is always derived from local data (sync queue state).
   110→      final status = _localDataSource.getSyncStatus(userId);
   111→      return Success(status);
   112→    } on CacheException catch (e) {
   113→      return FailureResult(Failure.cache(message: e.message));
   114→    } catch (e) {
   115→      return FailureResult(_mapExceptionToFailure(e));
   116→    }
   117→  }

   119→  @override
   120→  Future<Result<bool>> triggerSync(String userId) async {
   121→    try {
   122→      final isOnline = await _networkInfo.isConnected;
   123→      if (!isOnline) {
   124→        return const FailureResult(
   125→          Failure.network(message: 'No network connection available'),
   126→        );
   127→      }
   128→
   129→      // Process the sync queue via SyncEngine.
   130→      await _syncEngine.processQueue();
   131→      return const Success(true);
   132→    } on NetworkException catch (e) {
   133→      return FailureResult(Failure.network(message: e.message));
   134→    } on ServerException catch (e) {
   135→      return FailureResult(Failure.server(
   136→        message: e.message,
   137→        statusCode: e.statusCode,
   138→        data: e.data,
   139→      ),);
   140→    } catch (e) {
   141→      return FailureResult(_mapExceptionToFailure(e));
   142→    }
   143→  }

   168→  @override
   169→  Future<Result<bool>> downloadResourceForOffline(
   170→    String userId,
   171→    String resourceType,
   172→    String resourceId,
   173→  ) async {
   174→    try {
   175→      final isOnline = await _networkInfo.isConnected;
   176→
   177→      if (isOnline) {
   178→        // Get the download URL from the server.
   179→        final downloadUrl = await _remoteDataSource.getFileDownloadUrl(
   180→          resourceType,
   181→          resourceId,
   182→        );
   183→
   184→        // Save the resource metadata locally.
   185→        await _localDataSource.saveOfflineResource({
   186→          'user_id': userId,
   187→          'resource_type': resourceType,
   188→          'resource_id': resourceId,
   189→          'title': '$resourceType - $resourceId',
   190→          'file_path': downloadUrl,
   191→          'file_size_bytes': 0, // Updated when download completes
   192→          'mime_type': 'application/octet-stream',
   193→        });
   194→
   195→        // Enqueue the actual download in the sync queue.
   196→        await _cacheManager.enqueueSync(
   197→          userId: userId,
   198→          tableName: 'resources',
   199→          operation: 'download',
   200→          payload: {
   201→            'resource_type': resourceType,
   202→            'resource_id': resourceId,
   203→            'download_url': downloadUrl,
   204→          },
   205→          priority: 3,
   206→        );
   207→
   208→        return const Success(true);
   209→      } else {
   210→        // Offline: queue the download for later.
   211→        await _cacheManager.enqueueSync(
   212→          userId: userId,
   213→          tableName: 'resources',
   214→          operation: 'download',
   215→          payload: {
   216→            'resource_type': resourceType,
   217→            'resource_id': resourceId,
   218→          },
   219→          priority: 3,
   220→        );
   221→
   222→        return const Success(true);
   223→      }
   224→    } on ServerException catch (e) {
   225→      return FailureResult(Failure.server(
   226→        message: e.message,
   227→        statusCode: e.statusCode,
   228→        data: e.data,
   229→      ),);
   230→    } on CacheException catch (e) {
   231→      return FailureResult(Failure.cache(message: e.message));
   232→    } catch (e) {
   233→      return FailureResult(_mapExceptionToFailure(e));
   234→    }
   235→  }

   271→  @override
   272→  Future<Result<List<OfflineExamConfig>>> getOfflineExamConfigs(
   273→    String schoolId,
   274→  ) async {
   275→    try {
   276→      final isOnline = await _networkInfo.isConnected;
   277→
   278→      if (isOnline) {
   279→        // Try remote first.
   280→        try {
   281→          final remoteConfigs =
   282→              await _remoteDataSource.getOfflineExamConfigs(schoolId);
   283→
   284→          final configs = remoteConfigs
   285→              .map((data) => OfflineExamConfig(
   286→                    id: data['id'] as String,
   287→                    schoolId: data['school_id'] as String,
   288→                    examId: data['exam_id'] as String,
   289→                    allowsOffline: data['allows_offline'] as bool? ?? false,
   290→                    offlineType: OfflineType.fromString(
   291→                            data['offline_type'] as String?,) ??
   292→                        OfflineType.none,
   293→                    maxOfflineAttempts:
   294→                        data['max_offline_attempts'] as int? ?? 0,
   295→                    requiresOnlineSubmission:
   296→                        data['requires_online_submission'] as bool? ?? true,
   297→                    autoSubmitOnReconnect:
   298→                        data['auto_submit_on_reconnect'] as bool? ?? false,
   299→                    createdAt: data['created_at'] != null
   300→                        ? DateTime.parse(data['created_at'] as String)
   301→                        : DateTime.now(),
   302→                  ),)
   303→              .toList();
   304→
   305→          // Cache locally.
   306→          await _cacheManager.cacheData(
   307→            key: 'offline_exam_configs_$schoolId',
   308→            userId: schoolId,
   309→            resourceType: 'offline_exam_config',
   310→            resourceId: schoolId,
   311→            data: {'configs': remoteConfigs},
   312→          );
   313→
   314→          return Success(configs);
   315→        } on ServerException {
   316→          // Fall back to local cache.
   317→          AppLogger.warning(
   318→            '[OfflineRepo] Remote configs fetch failed, falling back to cache',
   319→          );
   320→        }
   321→      }
   322→
   323→      // Read from local cache.
   324→      final cached = await _cacheManager.getCachedData(
   325→        key: 'offline_exam_configs_$schoolId',
   326→        userId: schoolId,
   327→      );
   328→
   329→      if (cached != null && cached['configs'] != null) {
   330→        final configsList =
   331→            List<Map<String, dynamic>>.from(cached['configs'] as List);
   332→        final configs = configsList
   333→            .map((data) => OfflineExamConfig(
   334→                  id: data['id'] as String,
   335→                  schoolId: data['school_id'] as String,
   336→                  examId: data['exam_id'] as String,
   337→                  allowsOffline: data['allows_offline'] as bool? ?? false,
   338→                  offlineType:
   339→                      OfflineType.fromString(data['offline_type'] as String?) ??
   340→                          OfflineType.none,
   341→                  maxOfflineAttempts: data['max_offline_attempts'] as int? ?? 0,
   342→                  requiresOnlineSubmission:
   343→                      data['requires_online_submission'] as bool? ?? true,
   344→                  autoSubmitOnReconnect:
   345→                      data['auto_submit_on_reconnect'] as bool? ?? false,
   346→                  createdAt: data['created_at'] != null
   347→                      ? DateTime.parse(data['created_at'] as String)
   348→                      : DateTime.now(),
   349→                ),)
   350→            .toList();
   351→        return Success(configs);
   352→      }
   353→
   354→      return const Success([]);
   355→    } on CacheException catch (e) {
   356→      return FailureResult(Failure.cache(message: e.message));
   357→    } catch (e) {
   358→      return FailureResult(_mapExceptionToFailure(e));
   359→    }
   360→  }

   362→  @override
   363→  Future<Result<OfflineExamConfig>> updateOfflineExamConfig(
   364→    OfflineExamConfig config,
   365→  ) async {
   366→    try {
   367→      final isOnline = await _networkInfo.isConnected;
   368→
   369→      if (isOnline) {
   370→        // Update on server (would use a proper Supabase call).
   371→        // Cache the updated config locally.
   372→        await _cacheManager.cacheData(
   373→          key: 'offline_exam_config_${config.examId}',
   374→          userId: config.schoolId,
   375→          resourceType: 'offline_exam_config',
   376→          resourceId: config.examId,
   377→          data: {
   378→            'id': config.id,
   379→            'school_id': config.schoolId,
   380→            'exam_id': config.examId,
   381→            'allows_offline': config.allowsOffline,
   382→            'offline_type': config.offlineType.value,
   383→            'max_offline_attempts': config.maxOfflineAttempts,
   384→            'requires_online_submission': config.requiresOnlineSubmission,
   385→            'auto_submit_on_reconnect': config.autoSubmitOnReconnect,
   386→          },
   387→        );
   388→      } else {
   389→        // Queue for sync.
   390→        await _cacheManager.enqueueSync(
   391→          userId: config.schoolId,
   392→          tableName: 'offline_exam_configs',
   393→          operation: 'update',
   394→          payload: {
   395→            'id': config.id,
   396→            'school_id': config.schoolId,
   397→            'exam_id': config.examId,
   398→            'allows_offline': config.allowsOffline,

   403→          },
   404→          recordId: config.id,
   405→        );

   422→  @override
   423→  Future<Result<OfflineExamAttempt>> saveOfflineExamAttempt(
   424→    OfflineExamAttempt attempt,
   425→  ) async {
   426→    try {
   427→      // Always save locally first.
   428→      final savedAttempt = await _localDataSource.saveExamAttempt({
   429→        'id': attempt.id,
   430→        'exam_id': attempt.examId,
   431→        'student_id': attempt.studentId,
   432→        'school_id': attempt.schoolId,
   433→        'attempt_data': attempt.attemptData,
   434→        'answers': attempt.answers,
   435→        'started_at': attempt.startedAt.toIso8601String(),
   436→        'completed_at': attempt.completedAt?.toIso8601String(),
   437→        'time_taken_seconds': attempt.timeTakenSeconds,
   438→        'integrity_hash': attempt.integrityHash,
   439→      });
   440→
   441→      // Try to sync immediately if online.
   442→      final isOnline = await _networkInfo.isConnected;
   443→      if (isOnline && attempt.completedAt != null) {
   444→        try {
   445→          await _remoteDataSource.syncExamAttempt(
   446→            attempt.id,
   447→            {
   448→              'exam_id': attempt.examId,
   449→              'student_id': attempt.studentId,
   450→              'school_id': attempt.schoolId,
   451→              'attempt_data': attempt.attemptData,
   452→              'answers': attempt.answers,
   453→              'started_at': attempt.startedAt.toIso8601String(),
   454→              'completed_at': attempt.completedAt?.toIso8601String(),
   455→              'time_taken_seconds': attempt.timeTakenSeconds,
   456→              'integrity_hash': attempt.integrityHash,
   457→            },
   458→          );
   459→
   460→          // Mark as synced locally.
   461→          await _localDataSource.markExamAttemptSynced(attempt.id);
   462→
   463→          return Success(savedAttempt.copyWith(
   464→            syncStatus: AttemptSyncStatus.synced,
   465→            syncedAt: DateTime.now(),
   466→          ),);
   467→        } on ServerException catch (e) {
   468→          // Sync failed — keep as pending, queue for later.
   469→          AppLogger.warning(
   470→            '[OfflineRepo] Exam attempt sync failed, queued for later: ${e.message}',
   471→          );
   472→
   473→          await _cacheManager.enqueueSync(
   474→            userId: attempt.studentId,
   475→            tableName: 'offline_exam_attempts',
   476→            operation: 'sync',
   477→            payload: {
   478→              'attempt_id': attempt.id,
   479→              'exam_id': attempt.examId,
   480→              'student_id': attempt.studentId,
   481→              'school_id': attempt.schoolId,
   482→              'attempt_data': attempt.attemptData,
   483→              'answers': attempt.answers,
   484→              'integrity_hash': attempt.integrityHash,
   485→            },
   486→            recordId: attempt.id,
   487→            priority: 2, // High priority
   488→          );
   489→        }
   490→      } else if (attempt.completedAt != null) {
   491→        // Offline — queue for sync.
   492→        await _cacheManager.enqueueSync(
   493→          userId: attempt.studentId,
   494→          tableName: 'offline_exam_attempts',
   495→          operation: 'sync',
   496→          payload: {
   497→            'attempt_id': attempt.id,
   498→            'exam_id': attempt.examId,
   499→            'student_id': attempt.studentId,
   500→            'school_id': attempt.schoolId,
   501→            'attempt_data': attempt.attemptData,
   502→            'answers': attempt.answers,
   503→            'integrity_hash': attempt.integrityHash,
   504→          },
   505→          recordId: attempt.id,
   506→          priority: 2,
   507→        );
   508→      }
   509→
   510→      return Success(savedAttempt);
   511→    } on CacheException catch (e) {
   512→      return FailureResult(Failure.cache(message: e.message));
   513→    } catch (e) {
   514→      return FailureResult(_mapExceptionToFailure(e));
   515→    }
   516→  }

   533→  @override
   534→  Future<Result<bool>> syncExamAttempt(String attemptId) async {
   535→    try {
   536→      final isOnline = await _networkInfo.isConnected;
   537→      if (!isOnline) {
   538→        return const FailureResult(
   539→          Failure.network(message: 'Cannot sync while offline'),
   540→        );
   541→      }
   542→
   543→      // Get the attempt from local storage.
   544→      final pendingAttempts =
   545→          await _localDataSource.getPendingExamAttempts('');
   546→      final attempt = pendingAttempts.where((a) => a.id == attemptId).firstOrNull;
   547→
   548→      if (attempt == null) {
   549→        return FailureResult(
   550→          Failure.notFound(message: 'Exam attempt not found: $attemptId'),
   551→        );
   552→      }
   553→
   554→      // Sync with server.
   555→      await _remoteDataSource.syncExamAttempt(
   556→        attemptId,
   557→        {
   558→          'exam_id': attempt.examId,
   559→          'student_id': attempt.studentId,
   560→          'school_id': attempt.schoolId,
   561→          'attempt_data': attempt.attemptData,
   562→          'answers': attempt.answers,
   563→          'started_at': attempt.startedAt.toIso8601String(),
   564→          'completed_at': attempt.completedAt?.toIso8601String(),
   565→          'time_taken_seconds': attempt.timeTakenSeconds,
   566→          'integrity_hash': attempt.integrityHash,
   567→        },
   568→      );
   569→
   570→      // Mark as synced locally.
   571→      await _localDataSource.markExamAttemptSynced(attemptId);
   572→
   573→      return const Success(true);
   574→    } on ServerException catch (e) {
   575→      return FailureResult(Failure.server(
   576→        message: e.message,
   577→        statusCode: e.statusCode,
   578→        data: e.data,
   579→      ),);
   580→    } on CacheException catch (e) {
   581→      return FailureResult(Failure.cache(message: e.message));
   582→    } catch (e) {
   583→      return FailureResult(_mapExceptionToFailure(e));
   584→    }
   585→  }

   979→  Future<Result<bool>> syncAnalyticsEvents(String userId) async {
   980→    try {
   981→      final isOnline = await _networkInfo.isConnected;
   982→      if (!isOnline) {
   983→        return const FailureResult(
   984→          Failure.network(message: 'Cannot sync analytics while offline'),
   985→        );
   986→      }
   987→
   988→      // Get pending analytics sync items.
   989→      final pendingItems = await _cacheManager.getPendingSyncItems(
   990→        userId: userId,
   991→      );
   992→
   993→      final analyticsItems = pendingItems
   994→          .where((item) => item['tableName'] == 'app_usage_events')
   995→          .map((item) => item['payload'] as Map<String, dynamic>)
   996→          .toList();
   997→
   998→      if (analyticsItems.isEmpty) {
   999→        return const Success(true);
  1000→      }
  1001→
  1002→      final success =
  1003→          await _remoteDataSource.logUsageEvents(analyticsItems);
  1004→
  1005→      // Mark synced items as completed.
  1006→      for (final item in pendingItems
  1007→          .where((item) => item['tableName'] == 'app_usage_events')) {
  1008→        await _cacheManager.markSyncItemCompleted(id: item['id'] as String);
  1009→      }
  1010→
  1011→      return Success(success);
  1012→    } on ServerException catch (e) {
  1013→      return FailureResult(Failure.server(
  1014→        message: e.message,
  1015→        statusCode: e.statusCode,
  1016→        data: e.data,
  1017→      ),);
  1018→    } on CacheException catch (e) {
  1019→      return FailureResult(Failure.cache(message: e.message));
  1020→    } catch (e) {
  1021→      return FailureResult(_mapExceptionToFailure(e));
  1022→    }
  1023→  }
