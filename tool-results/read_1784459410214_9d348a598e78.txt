    18→/// Operations Intelligence Center with AI — the "game-changer" feature.
    19→///
    20→/// Provides AI-driven operational insights through four tabs:
    21→/// - **Alerts**: Intelligence alerts with filtering and acknowledge/resolve actions
    22→/// - **Churn Prediction**: Schools at risk of churning with probability scores
    23→/// - **Revenue Forecast**: 6-month revenue prediction with confidence intervals
    24→/// - **Cost Optimization**: AI provider cost analysis and optimization suggestions
    25→class IntelligenceCenterPage extends ConsumerStatefulWidget {
    26→  const IntelligenceCenterPage({super.key});
    27→
    28→  @override
    29→  ConsumerState<IntelligenceCenterPage> createState() =>
    30→      _IntelligenceCenterPageState();
    31→}
    32→
    33→class _IntelligenceCenterPageState
    34→    extends ConsumerState<IntelligenceCenterPage>
    35→    with SingleTickerProviderStateMixin {
    36→  late final TabController _tabController;
    37→
    38→  // ─── Alert Filters ──────────────────────────────────────────────────────
    39→
    40→  IntelligenceAlertType? _filterAlertType;
    41→  IntelligenceSeverity? _filterSeverity;
    42→
    43→  static const _tabs = [
    44→    Tab(icon: Icon(Icons.notification_important), text: 'Alerts'),
    45→    Tab(icon: Icon(Icons.person_off), text: 'Churn Prediction'),
    46→    Tab(icon: Icon(Icons.show_chart), text: 'Revenue Forecast'),
    47→    Tab(icon: Icon(Icons.savings), text: 'Cost Optimization'),
    48→  ];
    49→
    50→  @override
    51→  void initState() {
    52→    super.initState();
    53→    _tabController = TabController(length: _tabs.length, vsync: this);
    54→    _tabController.addListener(_onTabChanged);
    55→    WidgetsBinding.instance.addPostFrameCallback((_) {
    56→      _loadData();
    57→    });
    58→  }
    59→
    60→  @override
    61→  void dispose() {
    62→    _tabController.removeListener(_onTabChanged);
    63→    _tabController.dispose();
    64→    super.dispose();
    65→  }
    66→
    67→  void _onTabChanged() {
    68→    if (!_tabController.indexIsChanging) {
    69→      _loadTabData(_tabController.index);
    70→    }
    71→  }
    72→
    73→  void _loadData() {
    74→    _loadTabData(_tabController.index);
    75→  }
    76→
    77→  void _loadTabData(int index) {
    78→    final notifier = ref.read(intelligenceProvider.notifier);
    79→    switch (index) {
    80→      case 0:
    81→        notifier.loadAlerts(
    82→          type: _filterAlertType,
    83→          severity: _filterSeverity,
    84→          unresolvedOnly: false,
    85→        );
    86→        break;
    87→      case 1:
    88→        notifier.loadChurnPredictions();
    89→        break;
    90→      case 2:
    91→        notifier.loadRevenueForecast();
    92→        break;
    93→      case 3:
    94→        notifier.loadCostOptimizations();
    95→        break;
    96→    }
    97→  }
    98→

   604→class _RevenueForecastTab extends StatelessWidget {
   605→  const _RevenueForecastTab({required this.revenueForecast});
   606→
   607→  final Map<String, dynamic>? revenueForecast;
   608→
   609→  @override
   610→  Widget build(BuildContext context) {
   611→    final cs = Theme.of(context).colorScheme;
   612→
   613→    if (revenueForecast == null) {
   614→      return const AdminEmptyState(
   615→        message: 'No revenue forecast data available. Generate insights first.',
   616→        icon: Icons.show_chart,
   617→      );
   618→    }
   619→
   620→    final forecastMonths =
   621→        revenueForecast!['months'] as List<dynamic>? ?? [];
   622→    final overallConfidence =
   623→        (revenueForecast!['overallConfidence'] as num?)?.toDouble() ?? 0.0;
   624→
   625→    if (forecastMonths.isEmpty) {
   626→      return const AdminEmptyState(
   627→        message: 'No forecast data available.',
   628→        icon: Icons.show_chart,
   629→      );
   630→    }
   631→
   632→    // Find max value for scaling
   633→    double maxValue = 0;
   634→    for (final m in forecastMonths) {
   635→      final data = m as Map<String, dynamic>;
   636→      final upper = (data['upperBound'] as num?)?.toDouble() ?? 0;
   637→      if (upper > maxValue) maxValue = upper;
