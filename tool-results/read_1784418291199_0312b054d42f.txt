   223→   223→                  gradient: AppColors.brandGradient,
   224→   224→                  borderRadius: BorderRadius.only(
   225→   225→                    bottomLeft: Radius.circular(Spacings.mdRadius),

   234→   234→                ),
   235→   235→              ),
   236→   236→            ),
   237→   237→        ],
   238→   238→      ),

   240→   240→  }

   242→   242→  String _formatCurrency(double amount, String currency) {
   243→   243→    final symbol = currency == 'NGN' ? '\u20A6' : '\$';
   244→   244→    return '$symbol${amount.toStringAsFixed(0)}';
   245→   245→  }

   251→   251→
   252→   252→/// Shows AI credit balance with a circular progress ring.
   253→   253→///
   254→   254→/// Displays remaining / total credits, a circular progress indicator
   255→   255→/// representing usage percentage, and a "Purchase More" button when credits
   256→   256→/// are running low.
   257→   257→class CreditBalanceCard extends StatelessWidget {
   258→   258→  const CreditBalanceCard({
   259→   259→    super.key,
   260→   260→    required this.balance,
   261→   261→    this.onPurchaseMore,
   262→   262→  });
   263→   263→
   264→   264→  final AiCreditBalanceEntity balance;
   265→   265→  final VoidCallback? onPurchaseMore;
   266→   266→
   267→   267→  @override
   268→   268→  Widget build(BuildContext context) {
   269→   269→    final cs = context.colorScheme;
   270→   270→    final tt = context.textTheme;
   271→   271→    final isDark = context.isDarkMode;
   272→   272→    final usagePercent = balance.usagePercent;
   273→   273→    final remainingPercent = 100 - usagePercent;
   274→   274→
   275→   275→    // Choose colour based on credit health
   276→   276→    final Color progressColor;
   277→   277→    if (balance.isExhausted) {
   278→   278→      progressColor = AppColors.error;
   279→   279→    } else if (balance.isLow) {
   280→   280→      progressColor = AppColors.warning;
   281→   281→    } else {
   282→   282→      progressColor = AppColors.success;
   283→   283→    }
   284→   284→
   285→   285→    return Card(
   286→   286→      elevation: Spacings.elevationSm,
   287→   287→      shadowColor: cs.shadow.withValues(alpha: 0.06),
   288→   288→      shape: RoundedRectangleBorder(
   289→   289→        borderRadius: Spacings.borderRadiusLg,
   290→   290→      ),
   291→   291→      child: Padding(
   292→   292→        padding: const EdgeInsets.all(Spacings.lg),
   293→   293→        child: Column(
   294→   294→          children: [
   295→   295→            // ── Top Row: Progress Ring + Info ────────────────────────

   307→   307→                            ? remainingPercent / 100
   308→   308→                            : 0,
   309→   309→                        strokeWidth: 6,
   310→   310→                        backgroundColor:
   311→   311→                            progressColor.withValues(alpha: isDark ? 0.20 : 0.12),
   312→   312→                        valueColor: AlwaysStoppedAnimation(progressColor),
   313→   313→                        strokeCap: StrokeCap.round,
   314→   314→                      ),
   315→   315→                      Column(
   316→   316→                        mainAxisSize: MainAxisSize.min,
   317→   317→                        children: [
   318→   318→                          Text(
   319→   319→                            '${remainingPercent.toStringAsFixed(0)}%',
   320→   320→                            style: tt.titleSmall?.copyWith(
   321→   321→                              fontWeight: AppTypography.wBold,
   322→   322→                              color: cs.onSurface,
   323→   323→                              height: 1.0,
   324→   324→                            ),
   325→   325→                          ),
   326→   326→                          Text(
   327→   327→                            'remaining',
   328→   328→                            style: tt.labelSmall?.copyWith(
   329→   329→                              color: cs.onSurfaceVariant,
   330→   330→                              fontSize: 9,
   331→   331→                            ),
   332→   332→                          ),
   333→   333→                        ],
   334→   334→                      ),
   335→   335→                    ],
   336→   336→                  ),
   337→   337→                ),
   338→   338→                const SizedBox(width: Spacings.lg),
   339→   339→
   340→   340→                // Credit numbers
   341→   341→                Expanded(
   342→   342→                  child: Column(
   343→   343→                    crossAxisAlignment: CrossAxisAlignment.start,
   344→   344→                    children: [
   345→   345→                      Text(
   346→   346→                        'AI Credits',
   347→   347→                        style: tt.titleMedium?.copyWith(
   348→   348→                          fontWeight: AppTypography.wSemiBold,
   349→   349→                          color: cs.onSurface,
   350→   350→                        ),
   351→   351→                      ),
   352→   352→                      const SizedBox(height: Spacings.sm),
   353→   353→                      Text(
   354→   354→                        '${balance.remainingCredits} / ${balance.totalCredits}',
   355→   355→                        style: tt.headlineSmall?.copyWith(
   356→   356→                          fontWeight: AppTypography.wBold,
   357→   357→                          color: cs.onSurface,
   358→   358→                        ),
   359→   359→                      ),
   360→   360→                      const SizedBox(height: Spacings.xs),
   361→   361→                      Text(
   362→   362→                        '${balance.usedCredits} used this cycle',
   363→   363→                        style: tt.bodySmall?.copyWith(
   364→   364→                          color: cs.onSurfaceVariant,
   365→   365→                        ),
   366→   366→                      ),
   367→   367→                    ],
   368→   368→                  ),
   369→   369→                ),
   370→   370→              ],
   371→   371→            ),
   372→   372→
   373→   373→            // ── Warning / Exhausted indicator ─────────────────────────
   374→   374→            if (balance.isExhausted || balance.isLow) ...[
   375→   375→              const SizedBox(height: Spacings.md),

   443→   443→/// Displays a transaction in a list.
   444→   444→///

   446→   446→/// payment method, a channel icon, and an arrow indicator.

   464→   464→    final channelIcon = _channelIcon(transaction.channel);
   465→   465→    final isSuccessful = transaction.isSuccessful;

   470→   470→      onTap: onTap,

   489→   489→          size: Spacings.mdIcon - 4,
   490→   490→        ),

   492→   492→      title: Row(
   493→   493→        children: [
   494→   494→          Text(
   495→   495→            _formatCurrency(
   496→   496→              isRefund ? transaction.refundAmount : transaction.amount,
   497→   497→              transaction.currency,
   498→   498→            ),
   499→   499→            style: tt.titleSmall?.copyWith(
   500→   500→              fontWeight: AppTypography.wSemiBold,
   501→   501→              color: cs.onSurface,
   502→   502→            ),
   503→   503→          ),
   504→   504→          const SizedBox(width: Spacings.sm),
   505→   505→          _TransactionStatusBadge(status: transaction.status),
   506→   506→        ],
   507→   507→      ),
   508→   508→      subtitle: Text(
   509→   509→        _formatDate(transaction.initiatedAt),
   510→   510→        style: tt.bodySmall?.copyWith(
   511→   511→          color: cs.onSurfaceVariant,
   512→   512→        ),
   513→   513→      ),
   514→   514→      trailing: Row(
   515→   515→        mainAxisSize: MainAxisSize.min,
   516→   516→        children: [
   517→   517→          if (transaction.paymentMethodSummary != null)
   518→   518→            Flexible(
   519→   519→              child: Text(
   520→   520→                transaction.paymentMethodSummary!,

   522→   522→                  color: cs.onSurfaceVariant,
   523→   523→                ),
   524→   524→                overflow: TextOverflow.ellipsis,

   526→   526→            ),

  1971→  1971→/// Returns a darker variant of [color] for text on light backgrounds.
  1972→  1972→Color _darkenColor(Color color) {
  1973→  1973→  final hsl = HSLColor.fromColor(color);

  1975→  1975→}

  1983→  1983→/// Formats a date as 'dd MMM yyyy' (e.g. '12 Mar 2025').

  1985→  1985→  const months = [
  1986→  1986→    '', 'Jan', 'Feb', 'Mar', 'Apr', 'May', 'Jun',
  1987→  1987→    'Jul', 'Aug', 'Sep', 'Oct', 'Nov', 'Dec',
  1988→  1988→  ];
  1989→  1989→  return '${date.day} ${months[date.month]} ${date.year}';
  1990→  1990→}

  1992→  1992→/// Formats an amount with the appropriate currency symbol.
  1993→  1993→String _formatCurrency(double amount, String currency) {
  1994→  1994→  final symbol = currency == 'NGN' ? '\u20A6' : '\$';
  1995→  1995→  if (amount == amount.roundToDouble()) {
  1996→  1996→    return '$symbol${amount.toStringAsFixed(0)}';
  1997→  1997→  }
  1998→  1998→  return '$symbol${amount.toStringAsFixed(2)}';
  1999→  1999→}
