    72→  void _listenForMessages() {
    73→    final state = ref.read(teachingResourceProvider);
    74→    if (state.successMessage != null) {
    75→      _showSnackBar(state.successMessage!, isError: false);
    76→      ref.read(teachingResourceProvider.notifier).clearSuccessMessage();
    77→    }
    78→    if (state.error != null) {
    79→      _showSnackBar(state.error!, isError: true);
    80→      ref.read(teachingResourceProvider.notifier).clearError();
    81→    }
    82→  }

    84→  void _showSnackBar(String message, {required bool isError}) {
    85→    if (!mounted) return;
    86→    final cs = context.colorScheme;
    87→    context.scaffoldMessenger.showSnackBar(
    88→      SnackBar(
    89→        content: Text(message),
    90→        backgroundColor: isError ? cs.error : cs.primary,
    91→        behavior: SnackBarBehavior.floating,
    92→        shape: RoundedRectangleBorder(
    93→          borderRadius: BorderRadius.circular(Spacings.smRadius),
    94→        ),
    95→        duration: const Duration(seconds: 3),
    96→      ),
    97→    );
    98→  }

   100→  List<TeachingResourceEntity> _applyLocalFilters(
   101→    List<TeachingResourceEntity> resources,
   102→  ) {
   103→    var filtered = resources;
   104→
   105→    // Search filter
   106→    final query = _searchCtrl.text.trim().toLowerCase();
   107→    if (query.isNotEmpty) {
   108→      filtered = filtered
   109→          .where((r) =>
   110→              r.title.toLowerCase().contains(query) ||
   111→              (r.subject?.toLowerCase().contains(query) ?? false) ||
   112→              (r.description?.toLowerCase().contains(query) ?? false) ||
   113→              (r.topic?.toLowerCase().contains(query) ?? false))
   114→          .toList();
   115→    }
   116→
   117→    // Resource type filter
   118→    if (_filterResourceType != null) {
   119→      filtered = filtered
   120→          .where((r) => r.resourceType == _filterResourceType)
   121→          .toList();
   122→    }
   123→
   124→    // Favorites only
   125→    if (_favoritesOnly) {
   126→      filtered = filtered.where((r) => r.isFavorite).toList();
   127→    }
   128→
   129→    return filtered;
   130→  }

   446→  Widget _buildFilterDropdown<T>({
   447→    required String label,
   448→    required T? value,
   449→    required List<T> items,
   450→    required String Function(T) itemLabel,
   451→    required ValueChanged<T?> onChanged,
   452→  }) {
   453→    final cs = context.colorScheme;
   454→    final isSelected = value != null;
   455→
   456→    return InputChip(
   457→      selected: isSelected,
   458→      label: Text(
   459→        isSelected ? itemLabel(value as T) : label,
   460→        style: context.textTheme.labelMedium?.copyWith(
   461→          color: isSelected ? cs.onPrimary : cs.onSurfaceVariant,
   462→        ),
   463→      ),
   464→      avatar: Icon(
   465→        Icons.filter_list_rounded,
   466→        size: 16,
   467→        color: isSelected ? cs.onPrimary : cs.onSurfaceVariant,
   468→      ),
   469→      onPressed: () => _showFilterSheet<T>(
   470→        label: label,
   471→        items: items,
   472→        itemLabel: itemLabel,
   473→        selectedValue: value,
   474→        onSelected: onChanged,
   475→      ),
   476→      onDeleted: isSelected ? () => onChanged(null) : null,
   477→      deleteIconColor: isSelected ? cs.onPrimary : null,
   478→      selectedColor: cs.primary,
   479→      backgroundColor: cs.surfaceContainerLow,
   480→      side: BorderSide(
   481→        color: isSelected ? cs.primary : cs.outlineVariant,
   482→      ),
   483→    );
   484→  }
   485→
   486→  void _showFilterSheet<T>({
   487→    required String label,
   488→    required List<T> items,
   489→    required String Function(T) itemLabel,
   490→    required T? selectedValue,
   491→    required ValueChanged<T?> onSelected,
   492→  }) {
   493→    showModalBottomSheet(
   494→      context: context,
   495→      shape: const RoundedRectangleBorder(
   496→        borderRadius: BorderRadius.vertical(
   497→          top: Radius.circular(Spacings.lgRadius),
   498→        ),
   499→      ),
   500→      builder: (ctx) => SafeArea(
   501→        child: Column(
   502→          mainAxisSize: MainAxisSize.min,
   503→          children: [
   504→            Padding(
   505→              padding: const EdgeInsets.all(Spacings.lg),
   506→              child: Row(
   507→                children: [
   508→                  Text(
   509→                    'Filter by $label',
   510→                    style: context.textTheme.titleMedium?.copyWith(
   511→                      fontWeight: AppTypography.wSemiBold,
   512→                    ),
   513→                  ),
   514→                  const Spacer(),
   515→                  AppIconButton(
   516→                    icon: Icons.close,
   517→                    onPressed: () => Navigator.pop(ctx),
   518→                    variant: AppIconButtonVariant.standard,
   519→                  ),
   520→                ],
   521→              ),
   522→            ),
   523→            const Divider(height: 1),
   524→            Flexible(
   525→              child: ListView.builder(
   526→                shrinkWrap: true,
   527→                itemCount: items.length,
   528→                itemBuilder: (_, i) {
   529→                  final item = items[i];
   530→                  final isSelected = item == selectedValue;
   531→                  return ListTile(
   532→                    title: Text(itemLabel(item)),
   533→                    trailing: isSelected
   534→                        ? Icon(Icons.check_rounded,
   535→                            color: context.colorScheme.primary)
   536→                        : null,
   537→                    onTap: () {
   538→                      onSelected(isSelected ? null : item);
   539→                      Navigator.pop(ctx);
   540→                    },
   541→                  );
   542→                },
   543→              ),
   544→            ),
   545→          ],
   546→        ),
   547→      ),
   548→    );
   549→  }

   933→  Future<void> _handleCreateResource(
   934→    String title,
   935→    ResourceType type,
   936→    String subject,
   937→    String className,
   938→    String topic,
   939→    String content,
   940→  ) async {
   941→    if (title.isEmpty) {
   942→      _showSnackBar('Title is required', isError: true);
   943→      return;
   944→    }
   945→
   946→    Navigator.pop(context); // Close dialog
   947→
   948→    final now = DateTime.now();
   949→    await ref.read(teachingResourceProvider.notifier).createResource(
   950→          CreateResourceParams(
   951→            resource: TeachingResourceEntity(
   952→              id: '',
   953→              teacherId: '',
   954→              title: title,
   955→              resourceType: type,
   956→              subject: subject.isEmpty ? null : subject,
   957→              className: className.isEmpty ? null : className,
   958→              topic: topic.isEmpty ? null : topic,
   959→              content: content.isEmpty ? null : content,
   960→              createdAt: now,
   961→              updatedAt: now,
   962→            ),
   963→          ),
   964→        );
   965→
   966→    _listenForMessages();
   967→  }

   974→    String topic,
   975→  ) async {

   982→            topic: topic.isEmpty ? null : topic,

   985→        );
   986→
   987→    _listenForMessages();
   988→  }
