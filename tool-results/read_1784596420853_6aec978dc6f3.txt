   366→  Widget _buildConnectionIndicator(BuildContext context, String status) {

   368→    final color = switch (status) {

   405→              const SizedBox(width: Spacings.xs),

   637→  // ── Time Warnings ─────────────────────────────────────────────────────
   638→
   639→  void _checkTimeWarnings(ExamTakerState state) {
   640→    if (state.timeRemaining.inMinutes == 5 &&
   641→        !_hasShownTimeWarning &&

   652→        }
   653→      });
   654→    }
   655→
   656→    if (state.timeRemaining.inMinutes == 0 &&
   657→        state.timeRemaining.inSeconds > 30 &&
