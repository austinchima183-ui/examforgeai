     1→// ===========================================================================
     2→// ExamForge AI — Smart Synchronization Engine
     3→// ===========================================================================
     4→//
     5→// The heart of the offline-first architecture. Manages a persistent queue
     6→// of pending operations, processes them in priority order respecting
     7→// connectivity quality, detects and resolves conflicts, and provides
     8→// an [OfflineAwareRepository] mixin so any repository can transparently
     9→// fall back to cached data when offline.
    10→//
    11→// Usage:
    12→//   // Watch sync state:
    13→//   final syncState = ref.watch(syncEngineProvider);
    14→//
    15→//   // Enqueue an operation:
    16→//   ref.read(syncEngineProvider.notifier).enqueueOperation(
    17→//     tableName: 'questions',
    18→//     operation: SyncOperationType.insert,
    19→//     payload: {'title': 'New Question', ...},
    20→//     priority: SyncPriority.high,
    21→//   );
    22→//
    23→//   // Use offline-aware execution:
    24→//   final result = await OfflineAwareRepository(ref).executeOfflineAware(
    25→//     remoteCall: () => apiClient.get('/questions'),
    26→//     localFallback: () => cacheManager.getCachedData(...),
    27→//     tableName: 'questions',
    28→//     operation: SyncOperationType.insert,
    29→//     syncPayload: data,
    30→//   );

    53→// SYNC OPERATION TYPE — Enhanced Enum
    54→// ═══════════════════════════════════════════════════════════════════════════════
    55→
    56→/// Represents the type of database operation being synced.
    57→enum SyncOperationType {
    58→  insert(value: 'insert', label: 'Insert'),
    59→  update(value: 'update', label: 'Update'),
    60→  delete(value: 'delete', label: 'Delete');
    61→
    62→  const SyncOperationType({required this.value, required this.label});
    63→
    64→  /// String identifier used for serialisation and database storage.
    65→  final String value;
    66→
    67→  /// Human-readable label for UI display and logging.
    68→  final String label;
    69→
    70→  /// Constructs a [SyncOperationType] from its serialised [value] string.
    71→  ///
    72→  /// Returns [SyncOperationType.update] as a safe default if the string
    73→  /// does not match any known operation type.
    74→  static SyncOperationType fromString(String value) {
    75→    return SyncOperationType.values.firstWhere(
    76→      (t) => t.value == value,
    77→      orElse: () => SyncOperationType.update,
    78→    );
    79→  }
    80→}

   119→/// Strategy for resolving a sync conflict between local and server data.
   120→enum ConflictResolution {
   121→  localWins(value: 'local_wins', label: 'Local Wins'),
   122→  serverWins(value: 'server_wins', label: 'Server Wins'),
   123→  merge(value: 'merge', label: 'Merge'),
   124→  manual(value: 'manual', label: 'Manual');
   125→
   126→  const ConflictResolution({required this.value, required this.label});
   127→
   128→  /// String identifier used for serialisation and database storage.
   129→  final String value;
   130→
   131→  /// Human-readable label for UI display and logging.
   132→  final String label;
   133→
   134→  /// Constructs a [ConflictResolution] from its serialised [value] string.
   135→  ///
   136→  /// Returns [ConflictResolution.manual] as a safe default so that
   137→  /// unrecognised strategies never silently discard data.
   138→  static ConflictResolution fromString(String value) {
   139→    return ConflictResolution.values.firstWhere(
   140→      (r) => r.value == value,
   141→      orElse: () => ConflictResolution.manual,
   142→    );
   143→  }
   144→}

   150→/// Priority level for a sync queue item. Lower values are processed first.
   151→///
   152→/// The [level] integer is used for sorting; the semantic names are for
   153→/// human consumption and UI rendering.
   154→enum SyncPriority {
   155→  critical(value: 'critical', label: 'Critical', level: 1),
   156→  high(value: 'high', label: 'High', level: 3),
   157→  normal(value: 'normal', label: 'Normal', level: 5),
   158→  low(value: 'low', label: 'Low', level: 7),
   159→  background(value: 'background', label: 'Background', level: 10);
   160→
   161→  const SyncPriority({
   162→    required this.value,
   163→    required this.label,
   164→    required this.level,
   165→  });
   166→
   167→  /// String identifier used for serialisation and database storage.
   168→  final String value;
   169→
   170→  /// Human-readable label for UI display and logging.
   171→  final String label;
   172→
   173→  /// Numeric level used for priority sorting (lower = higher priority).
   174→  final int level;
   175→
   176→  /// Constructs a [SyncPriority] from its serialised [value] string.
   177→  ///
   178→  /// Returns [SyncPriority.normal] as a safe default.
   179→  static SyncPriority fromString(String value) {
   180→    return SyncPriority.values.firstWhere(
   181→      (p) => p.value == value,
   182→      orElse: () => SyncPriority.normal,
   183→    );
   184→  }
   185→
   186→  /// Constructs a [SyncPriority] from its numeric [level].
   187→  ///
   188→  /// Returns [SyncPriority.normal] if no priority matches exactly.
   189→  static SyncPriority fromLevel(int level) {
   190→    return SyncPriority.values.firstWhere(
   191→      (p) => p.level == level,
   192→      orElse: () => SyncPriority.normal,
   193→    );
   194→  }
   195→}

   199→// ═══════════════════════════════════════════════════════════════════════════════

   201→/// Represents a single pending sync operation in the persistent queue.
   202→///
   203→/// Instances are stored in the local [LocalSyncQueueTable] via
   204→/// [CacheManager] and processed by [SyncEngine.processQueue].
   205→class SyncQueueItem extends Equatable {
   206→  const SyncQueueItem({
   207→    required this.id,
   208→    required this.userId,
   209→    required this.tableName,
   210→    required this.recordId,
   211→    required this.operationType,
   212→    required this.payload,
   213→    required this.priority,
   214→    required this.attempts,
   215→    required this.maxAttempts,
   216→    required this.status,
   217→    required this.errorMessage,
   218→    required this.createdAt,
   219→    required this.updatedAt,
   220→    required this.nextRetryAt,
   221→  });
   222→
   223→  /// Unique identifier for this queue entry.
   224→  final String id;
   225→
   226→  /// Owner of the sync entry.
   227→  final String userId;
   228→
   229→  /// Target table name on the remote (e.g. `questions`, `exams`).
   230→  final String tableName;
   231→
   232→  /// Remote record ID. Null for inserts (server assigns the ID).
   233→  final String? recordId;
   234→
   235→  /// The type of operation to perform.
   236→  final SyncOperationType operationType;
   237→
   238→  /// JSON-serialisable payload for the operation.
   239→  final Map<String, dynamic> payload;
   240→
   241→  /// Priority level (lower = higher priority).
   242→  final SyncPriority priority;
   243→
   244→  /// Number of attempts made so far.
   245→  final int attempts;
   246→
   247→  /// Maximum number of retry attempts before marking dead.
   248→  final int maxAttempts;
   249→
   250→  /// Current lifecycle state.
   251→  final SyncStatus status;
   252→
   253→  /// Error message from the last failed attempt, if any.
   254→  final String? errorMessage;
   255→
   256→  /// When the entry was created locally.
   257→  final DateTime createdAt;
   258→
   259→  /// When the entry was last updated.
   260→  final DateTime updatedAt;
   261→
   262→  /// When the next retry should be attempted (null = immediate).
   263→  final DateTime? nextRetryAt;
   264→
   265→  // ── Computed Getters ──────────────────────────────────────────────────
   266→
   267→  /// Whether this item can be retried (has not exceeded max attempts
   268→  /// and is not in a terminal dead state).
   269→  bool get isRetryable => attempts < maxAttempts && status != SyncStatus.dead;
   270→
   271→  /// Whether this item is critical or high priority (level <= 3).
   272→  bool get isCritical => priority.level <= 3;
   273→
   274→  // ── Factory ───────────────────────────────────────────────────────────
   275→
   276→  /// Constructs a [SyncQueueItem] from a raw database row map
   277→  /// (as returned by [CacheManager.getPendingSyncItems]).
   278→  factory SyncQueueItem.fromMap(Map<String, dynamic> map) {
   279→    return SyncQueueItem(
   280→      id: map['id'] as String,
   281→      userId: map['userId'] as String,
   282→      tableName: map['tableName'] as String,
   283→      recordId: map['recordId'] as String?,
   284→      operationType: SyncOperationType.fromString(map['operation'] as String),
   285→      payload: map['payload'] is String
   286→          ? jsonDecode(map['payload'] as String) as Map<String, dynamic>
   287→          : Map<String, dynamic>.from(map['payload'] as Map),
   288→      priority: map['priority'] is int
   289→          ? SyncPriority.fromLevel(map['priority'] as int)
   290→          : SyncPriority.fromString(map['priority'] as String),
   291→      attempts: map['attempts'] as int,
   292→      maxAttempts: map['maxAttempts'] as int,
   293→      status: SyncStatus.fromString(map['status'] as String),
   294→      errorMessage: map['errorMessage'] as String?,
   295→      createdAt: map['createdAt'] is DateTime
   296→          ? map['createdAt'] as DateTime
   297→          : DateTime.parse(map['createdAt'] as String),
   298→      updatedAt: map['updatedAt'] is DateTime
   299→          ? map['updatedAt'] as DateTime
   300→          : DateTime.parse(map['updatedAt'] as String),
   301→      nextRetryAt: map['nextRetryAt'] == null
   302→          ? null
   303→          : map['nextRetryAt'] is DateTime
   304→              ? map['nextRetryAt'] as DateTime
   305→              : DateTime.parse(map['nextRetryAt'] as String),
   306→    );
   307→  }
   308→
   309→  /// Serialises this item to a JSON-compatible map.
   310→  Map<String, dynamic> toMap() => {
   311→        'id': id,
   312→        'userId': userId,
   313→        'tableName': tableName,
   314→        'recordId': recordId,
   315→        'operation': operationType.value,
   316→        'payload': jsonEncode(payload),
   317→        'priority': priority.level,
   318→        'attempts': attempts,
   319→        'maxAttempts': maxAttempts,
   320→        'status': status.value,
   321→        'errorMessage': errorMessage,
   322→        'createdAt': createdAt.toIso8601String(),
   323→        'updatedAt': updatedAt.toIso8601String(),
   324→        'nextRetryAt': nextRetryAt?.toIso8601String(),
   325→      };
   326→
   327→  /// Returns a copy with the specified fields replaced.
   328→  SyncQueueItem copyWith({
   329→    String? id,
   330→    String? userId,
   331→    String? tableName,
   332→    String? recordId,
   333→    SyncOperationType? operationType,
   334→    Map<String, dynamic>? payload,
   335→    SyncPriority? priority,
   336→    int? attempts,
   337→    int? maxAttempts,
   338→    SyncStatus? status,
   339→    String? errorMessage,
   340→    DateTime? createdAt,
   341→    DateTime? updatedAt,
   342→    DateTime? nextRetryAt,
   343→    bool clearErrorMessage = false,
   344→    bool clearNextRetryAt = false,
   345→  }) {
   346→    return SyncQueueItem(
   347→      id: id ?? this.id,
   348→      userId: userId ?? this.userId,
   349→      tableName: tableName ?? this.tableName,
   350→      recordId: recordId ?? this.recordId,
   351→      operationType: operationType ?? this.operationType,
   352→      payload: payload ?? this.payload,
   353→      priority: priority ?? this.priority,
   354→      attempts: attempts ?? this.attempts,
   355→      maxAttempts: maxAttempts ?? this.maxAttempts,
   356→      status: status ?? this.status,
   357→      errorMessage:
   358→          clearErrorMessage ? null : (errorMessage ?? this.errorMessage),
   359→      createdAt: createdAt ?? this.createdAt,
   360→      updatedAt: updatedAt ?? this.updatedAt,
   361→      nextRetryAt: clearNextRetryAt ? null : (nextRetryAt ?? this.nextRetryAt),
   362→    );
   363→  }
   364→
   365→  @override
   366→  List<Object?> get props => [
   367→        id,
   368→        userId,
   369→        tableName,
   370→        recordId,
   371→        operationType,
   372→        payload,
   373→        priority,
   374→        attempts,
   375→        maxAttempts,
   376→        status,
   377→        errorMessage,
   378→        createdAt,
   379→        updatedAt,
   380→        nextRetryAt,
   381→      ];
   382→
   383→  @override
   384→  String toString() => 'SyncQueueItem('
   385→      'id: $id, '
   386→      'table: $tableName, '
   387→      'operation: ${operationType.value}, '
   388→      'priority: ${priority.value}, '
   389→      'status: ${status.value}, '
   390→      'attempts: $attempts/$maxAttempts'
   391→      ')';
   392→}

   398→/// Represents a detected conflict between local and server data.
   399→///
   400→/// Created when [SyncEngine.detectConflict] determines that the local
   401→/// version of a record diverges from the server version. The conflict
   402→/// must be resolved before the sync can proceed.
   403→class SyncConflict extends Equatable {
   404→  const SyncConflict({
   405→    required this.id,
   406→    required this.userId,
   407→    required this.tableName,
   408→    required this.recordId,
   409→    required this.localData,
   410→    required this.serverData,
   411→    required this.resolution,
   412→    required this.resolvedData,
   413→    required this.createdAt,
   414→  });
   415→
   416→  /// Unique identifier for this conflict entry.
   417→  final String id;
   418→
   419→  /// Owner of the conflicting record.
   420→  final String userId;
   421→
   422→  /// Table where the conflict occurred.
   423→  final String tableName;
   424→
   425→  /// ID of the conflicting record.
   426→  final String recordId;
   427→
   428→  /// The local version of the data at the time of conflict detection.
   429→  final Map<String, dynamic> localData;
   430→
   431→  /// The server version of the data at the time of conflict detection.
   432→  final Map<String, dynamic> serverData;
   433→
   434→  /// The resolution strategy that was (or will be) applied.
   435→  final ConflictResolution resolution;
   436→
   437→  /// The merged / resolved data, or null if not yet resolved.
   438→  final Map<String, dynamic>? resolvedData;
   439→
   440→  /// When the conflict was first detected.
   441→  final DateTime createdAt;
   442→
   443→  /// Whether the conflict has been resolved.
   444→  bool get isResolved => resolution != ConflictResolution.manual ||
   445→      resolvedData != null;
   446→
   447→  /// Factory for creating a new unresolved conflict.
   448→  factory SyncConflict.create({
   449→    required String userId,
   450→    required String tableName,
   451→    required String recordId,
   452→    required Map<String, dynamic> localData,
   453→    required Map<String, dynamic> serverData,
   454→  }) {
   455→    return SyncConflict(
   456→      id: const Uuid().v4(),
   457→      userId: userId,
   458→      tableName: tableName,
   459→      recordId: recordId,
   460→      localData: localData,
   461→      serverData: serverData,
   462→      resolution: ConflictResolution.manual,
   463→      resolvedData: null,
   464→      createdAt: DateTime.now(),
   465→    );
   466→  }
   467→
   468→  /// Returns a copy with the conflict resolved.
   469→  SyncConflict resolve(
   470→    ConflictResolution resolution, {
   471→    Map<String, dynamic>? mergedData,
   472→  }) {
   473→    return SyncConflict(
   474→      id: id,
   475→      userId: userId,
   476→      tableName: tableName,
   477→      recordId: recordId,
   478→      localData: localData,
   479→      serverData: serverData,
   480→      resolution: resolution,
   481→      resolvedData: mergedData,
   482→      createdAt: createdAt,
   483→    );
   484→  }
   485→
   486→  @override
   487→  List<Object?> get props => [
   488→        id,
   489→        userId,
   490→        tableName,
   491→        recordId,
   492→        localData,
   493→        serverData,
   494→        resolution,
   495→        resolvedData,
   496→        createdAt,
   497→      ];
   498→
   499→  @override
   500→  String toString() => 'SyncConflict('
   501→      'id: $id, '
   502→      'table: $tableName, '
   503→      'record: $recordId, '
   504→      'resolution: ${resolution.value}'
   505→      ')';
   506→}

   509→// SYNC RESULT
   510→// ═══════════════════════════════════════════════════════════════════════════════
   511→
   512→/// Aggregated result of a sync batch operation.
   513→///
   514→/// Returned by [SyncEngine.processQueue] to summarise what happened
   515→/// during a single sync pass.
   516→class SyncResult extends Equatable {
   517→  const SyncResult({
   518→    required this.itemsSynced,
   519→    required this.itemsFailed,
   520→    required this.conflicts,
   521→    required this.duration,
   522→    required this.errors,
   523→  });
   524→
   525→  /// Number of items successfully synced to the server.
   526→  final int itemsSynced;
   527→
   528→  /// Number of items that failed to sync.
   529→  final int itemsFailed;
   530→
   531→  /// List of conflicts detected during this sync pass.
   532→  final List<SyncConflict> conflicts;
   533→
   534→  /// Wall-clock duration of the sync operation.
   535→  final Duration duration;
   536→
   537→  /// Human-readable error messages from failed items.
   538→  final List<String> errors;
   539→
   540→  // ── Computed Getters ──────────────────────────────────────────────────
   541→
   542→  /// Whether all items synced successfully (zero failures).
   543→  bool get isSuccess => itemsFailed == 0;
   544→
   545→  /// Whether any conflicts were detected.
   546→  bool get hasConflicts => conflicts.isNotEmpty;
   547→
   548→  /// Total number of items processed.
   549→  int get totalProcessed => itemsSynced + itemsFailed;
   550→
   551→  /// Success rate as a value between 0.0 and 1.0.
   552→  double get successRate => totalProcessed == 0
   553→      ? 0.0
   554→      : itemsSynced / totalProcessed;
   555→
   556→  /// An empty result indicating no work was done.
   557→  static const empty = SyncResult(
   558→    itemsSynced: 0,
   559→    itemsFailed: 0,
   560→    conflicts: [],
   561→    duration: Duration.zero,
   562→    errors: [],
   563→  );
   564→
   565→  @override
   566→  List<Object?> get props => [
   567→        itemsSynced,
   568→        itemsFailed,
   569→        conflicts,
   570→        duration,
   571→        errors,
   572→      ];
   573→
   574→  @override
   575→  String toString() => 'SyncResult('
   576→      'synced: $itemsSynced, '
   577→      'failed: $itemsFailed, '
   578→      'conflicts: ${conflicts.length}, '
   579→      'duration: ${duration.inMilliseconds}ms'
   580→      ')';
   581→}

   600→    this.errors = const [],
   601→    this.syncProgress = 0.0,
   602→    this.isAutoSyncEnabled = true,
   603→    this.conflictItems = const [],
   604→  });
   605→
   606→  /// Whether a sync operation is currently in progress.
   607→  final bool isSyncing;
   608→
   609→  /// Timestamp of the last successful sync completion.
   610→  final DateTime? lastSyncAt;
   611→
   612→  /// Number of items currently in the pending state.
   613→  final int pendingCount;
   614→
   615→  /// Number of items currently in the failed state.
   616→  final int failedCount;
   617→
   618→  /// Number of items being processed in the current batch.
   619→  final int currentBatch;
   620→
   621→  /// Result of the most recent sync pass, if any.
   622→  final SyncResult? lastResult;
   623→
   624→  /// Transient error messages for UI display.
   625→  final List<String> errors;
   626→
   627→  /// Sync progress as a value between 0.0 and 1.0.
   628→  final double syncProgress;
   629→
   630→  /// Whether automatic periodic sync is enabled.
   631→  final bool isAutoSyncEnabled;
   632→
   633→  /// Currently unresolved conflicts awaiting manual resolution.
   634→  final List<SyncConflict> conflictItems;
   635→
   636→  // ── Computed Getters ──────────────────────────────────────────────────
   637→
   638→  /// Whether there are any items waiting to be synced.
   639→  bool get hasPendingItems => pendingCount > 0;
   640→
   641→  /// Whether there are any failed items that may be retryable.
   642→  bool get hasFailedItems => failedCount > 0;
   643→
   644→  /// Whether there are any unresolved conflicts.
   645→  bool get hasConflicts => conflictItems.isNotEmpty;
   646→
   647→  /// Overall sync health assessment.
   648→  ///
   649→  ///   - **good**: no failed items and no conflicts.
   650→  ///   - **warning**: some failures but < 20 % of total, or conflicts exist.
   651→  ///   - **critical**: failures >= 20 % of total, or more than 10 dead items.
   652→  String get syncHealth {
   653→    final total = pendingCount + failedCount;
   654→    if (total == 0 && !hasConflicts) return 'good';
   655→
   656→    final failureRatio = total > 0 ? failedCount / total : 0.0;
   657→
   658→    if (failureRatio >= 0.2 || failedCount > 10) return 'critical';
   659→    if (failedCount > 0 || hasConflicts) return 'warning';
   660→    return 'good';
   661→  }

   726→// SYNC ENGINE
   727→// ═══════════════════════════════════════════════════════════════════════════════
   728→
   729→/// Core synchronisation engine for ExamForge AI's offline-first architecture.
   730→///
   731→/// Extends [StateNotifier]<[SyncEngineState]> so that any widget or
   732→/// provider watching [syncEngineProvider] rebuilds automatically when
   733→/// sync state changes.
   734→///
   735→/// The engine:
   736→///   1. Maintains a persistent queue of pending operations in the local DB.
   737→///   2. Processes them in priority order, respecting batch size limits
   738→///      derived from the current connectivity quality.
   739→///   3. Detects conflicts by comparing local and server timestamps.
   740→///   4. Auto-resolves conflicts where possible (server wins for non-drafts,
   741→///      merge for drafts) and escalates the rest to manual resolution.
   742→///   5. Applies exponential backoff for failed items.
   743→///   6. Runs a periodic auto-sync timer when enabled and online.
   744→class SyncEngine extends StateNotifier<SyncEngineState> {
   745→  SyncEngine({
   746→    required CacheManager cacheManager,
   747→    required dynamic supabaseClient,
   748→    required ConnectivityEngine connectivityEngine,
   749→    this.maxRetryAttempts = 5,
   750→    this.defaultSyncInterval = const Duration(seconds: 30),
   751→  })  : _cacheManager = cacheManager,
   752→        _supabaseClient = supabaseClient,
   753→        _connectivityEngine = connectivityEngine,
   754→        super(const SyncEngineState());
   755→
   756→  // ── Dependencies ──────────────────────────────────────────────────────
   757→
   758→  /// Persistent cache / sync-queue manager.
   759→  final CacheManager _cacheManager;
   760→
   761→  /// Supabase client for remote operations. Typed as `dynamic` to
   762→  /// decouple from the concrete Supabase version at compile time.
   763→  final dynamic _supabaseClient;
   764→
   765→  /// Reference to the connectivity engine for quality-aware decisions.
   766→  final ConnectivityEngine _connectivityEngine;
   767→
   768→  // ── Configuration ─────────────────────────────────────────────────────
   769→
   770→  /// Maximum retry attempts before marking an item as dead.
   771→  final int maxRetryAttempts;
   772→
   773→  /// Default interval between automatic sync passes.
   774→  final Duration defaultSyncInterval;
   775→
   776→  /// UUID generator for new records.
   777→  static const _uuid = Uuid();
   778→
   779→  // ── Internal State ────────────────────────────────────────────────────
   780→
   781→  /// Periodic timer for auto-sync.
   782→  Timer? _autoSyncTimer;
   783→
   784→  /// Subscription to connectivity state changes.
   785→  StreamSubscription<ConnectivityState>? _connectivitySubscription;
   786→
   787→  /// In-memory list of pending items loaded from the DB.
   788→  List<SyncQueueItem> _pendingItems = [];
   789→
   790→  /// In-memory list of detected conflicts.
   791→  List<SyncConflict> _conflicts = [];
   792→
   793→  /// Whether the engine has been initialised.
   794→  bool _isInitialized = false;
   795→
   796→  /// Whether the engine has been disposed.
   797→  bool _isDisposed = false;
   798→
   799→  // ═══════════════════════════════════════════════════════════════════════
   800→  // LIFECYCLE
   801→  // ═══════════════════════════════════════════════════════════════════════
   802→
   803→  /// Initialises the engine by loading pending items from the local DB
   804→  /// and starting the auto-sync timer if enabled.
   805→  ///
   806→  /// Should be called once after the engine is created (typically from
   807→  /// the Riverpod provider's constructor).
   808→  Future<void> initialize() async {
   809→    if (_isInitialized) return;
   810→
   811→    AppLogger.info('SyncEngine: initializing...');
   812→
   813→    try {
   814→      // Load pending items from the local database.
   815→      await _loadPendingItems();
   816→
   817→      // Load any persisted conflicts (could be stored in cache).
   818→      await _loadConflicts();
   819→
   820→      // Subscribe to connectivity changes to trigger sync when coming online.
   821→      _connectivitySubscription =
   822→          _connectivityEngine.stream.listen(_onConnectivityChanged);
   823→
   824→      // Start auto-sync if enabled.
   825→      if (state.isAutoSyncEnabled) {
   826→        _startAutoSyncTimer();
   827→      }
   828→
   829→      _isInitialized = true;
   830→
   831→      // Update state with loaded counts.
   832→      state = state.copyWith(
   833→        pendingCount: _pendingItems
   834→            .where((i) => i.status == SyncStatus.pending)
   835→            .length,
   836→        failedCount: _pendingItems
   837→            .where((i) => i.status == SyncStatus.failed)
   838→            .length,
   839→        conflictItems: _conflicts,
   840→      );
   841→
   842→      AppLogger.info(
   843→        'SyncEngine: initialized '
   844→        '(pending: ${state.pendingCount}, failed: ${state.failedCount}, '
   845→        'conflicts: ${_conflicts.length})',
   846→      );
   847→    } catch (e, st) {
   848→      AppLogger.error(
   849→        'SyncEngine: initialization failed',
   850→        error: e,
   851→        stackTrace: st,
   852→      );
   853→      state = state.copyWith(
   854→        errors: ['Initialization failed: $e'],
   855→      );
   856→    }
   857→  }

   859→  @override
   860→  void dispose() {
   861→    _isDisposed = true;
   862→    _autoSyncTimer?.cancel();
   863→    _connectivitySubscription?.cancel();
   864→    _pendingItems.clear();
   865→    _conflicts.clear();
   866→    super.dispose();
   867→  }

   873→  /// Adds an operation to the sync queue.
   874→  ///
   875→  /// If the device is online and the engine is not currently syncing,
   876→  /// the operation may be processed immediately on the next sync tick.
   877→  Future<void> enqueueOperation({
   878→    required String tableName,
   879→    required SyncOperationType operation,
   880→    required Map<String, dynamic> payload,
   881→    String? recordId,
   882→    SyncPriority priority = SyncPriority.normal,
   883→  }) async {
   884→    try {
   885→      // Get current user ID from Supabase auth.
   886→      final userId = _currentUserId;
   887→      if (userId == null) {
   888→        AppLogger.warning('SyncEngine: cannot enqueue — no authenticated user');
   889→        return;
   890→      }
   891→
   892→      // Persist to the local database via CacheManager.
   893→      await _cacheManager.enqueueSync(
   894→        userId: userId,
   895→        tableName: tableName,
   896→        operation: operation.value,
   897→        payload: payload,
   898→        recordId: recordId,
   899→        priority: priority.level,
   900→      );
   901→
   902→      // Create an in-memory representation.
   903→      final now = DateTime.now();
   904→      final item = SyncQueueItem(
   905→        id: _uuid.v4(),
   906→        userId: userId,
   907→        tableName: tableName,
   908→        recordId: recordId,
   909→        operationType: operation,
   910→        payload: payload,
   911→        priority: priority,
   912→        attempts: 0,
   913→        maxAttempts: maxRetryAttempts,
   914→        status: SyncStatus.pending,
   915→        errorMessage: null,
   916→        createdAt: now,
   917→        updatedAt: now,
   918→        nextRetryAt: null,
   919→      );
   920→
   921→      _pendingItems.add(item);
   922→      _sortPendingItems();
   923→
   924→      state = state.copyWith(
   925→        pendingCount: _pendingItems
   926→            .where((i) => i.status == SyncStatus.pending)
   927→            .length,
   928→      );
   929→
   930→      AppLogger.info(
   931→        'SyncEngine: enqueued ${operation.value} on $tableName '
   932→        '(priority: ${priority.value}, queue depth: ${_pendingItems.length})',
   933→      );
   934→    } catch (e, st) {
   935→      AppLogger.error(
   936→        'SyncEngine: failed to enqueue operation',
   937→        error: e,
   938→        stackTrace: st,
   939→      );
   940→      state = state.copyWith(
   941→        errors: ['Failed to enqueue operation: $e'],
   942→      );
   943→    }
   944→  }
   945→
   946→  /// Cancels a pending operation by its [id].
   947→  ///
   948→  /// Completed, dead, or in-progress items cannot be cancelled.
   949→  Future<void> cancelOperation(String id) async {
   950→    try {
   951→      final index = _pendingItems.indexWhere((i) => i.id == id);
   952→      if (index == -1) {
   953→        AppLogger.warning('SyncEngine: cancel — item $id not found in memory');
   954→        return;
   955→      }
   956→
   957→      final item = _pendingItems[index];
   958→      if (item.status == SyncStatus.completed ||
   959→          item.status == SyncStatus.dead ||
   960→          item.status == SyncStatus.inProgress) {
   961→        AppLogger.warning(
   962→          'SyncEngine: cannot cancel item $id in state ${item.status.value}',
   963→        );
   964→        return;
   965→      }
   966→
   967→      // Remove from in-memory list.
   968→      _pendingItems.removeAt(index);
   969→
   970→      // Remove from persistent store.
   971→      await _cacheManager.markSyncItemCompleted(id: id);
   972→
   973→      state = state.copyWith(
   974→        pendingCount: _pendingItems
   975→            .where((i) => i.status == SyncStatus.pending)
   976→            .length,
   977→        failedCount: _pendingItems
   978→            .where((i) => i.status == SyncStatus.failed)
   979→            .length,
   980→      );
   981→
   982→      AppLogger.info('SyncEngine: cancelled operation $id');
   983→    } catch (e, st) {
   984→      AppLogger.error(
   985→        'SyncEngine: failed to cancel operation $id',
   986→        error: e,
   987→        stackTrace: st,
   988→      );
   989→    }
   990→  }
   991→
   992→  /// Removes all completed items from the queue (both in-memory and DB).
   993→  Future<void> clearCompleted() async {
   994→    try {
   995→      final completedIds = _pendingItems
   996→          .where((i) => i.status == SyncStatus.completed)
   997→          .map((i) => i.id)
   998→          .toList();
   999→
  1000→      _pendingItems.removeWhere((i) => i.status == SyncStatus.completed);
  1001→
  1002→      // Clean up dead items while we are at it.
  1003→      await _cacheManager.cleanupDeadSyncItems();
  1004→
  1005→      state = state.copyWith(
  1006→        pendingCount: _pendingItems
  1007→            .where((i) => i.status == SyncStatus.pending)
  1008→            .length,
  1009→        failedCount: _pendingItems
  1010→            .where((i) => i.status == SyncStatus.failed)
  1011→            .length,
  1012→      );
  1013→
  1014→      AppLogger.info(
  1015→        'SyncEngine: cleared ${completedIds.length} completed items',
  1016→      );
  1017→    } catch (e, st) {
  1018→      AppLogger.error(
  1019→        'SyncEngine: failed to clear completed items',
  1020→        error: e,
  1021→        stackTrace: st,
  1022→      );
  1023→    }
  1024→  }

  1026→  /// Returns pending items for a specific [tableName].
  1027→  List<SyncQueueItem> getPendingForTable(String tableName) {
  1028→    return _pendingItems
  1029→        .where(
  1030→          (i) =>
  1031→              i.tableName == tableName &&
  1032→              (i.status == SyncStatus.pending ||
  1033→                  i.status == SyncStatus.failed),
  1034→        )
  1035→        .toList();
  1036→  }
  1037→
  1038→  // ═══════════════════════════════════════════════════════════════════════
  1039→  // PUBLIC API — PROCESS QUEUE
  1040→  // ═══════════════════════════════════════════════════════════════════════
  1041→
  1042→  /// Processes pending sync items in priority order, respecting the
  1043→  /// batch size dictated by the current connectivity quality.
  1044→  ///
  1045→  /// Returns a [SyncResult] summarising the outcome of this sync pass.
  1046→  Future<SyncResult> processQueue() async {
  1047→    if (state.isSyncing) {
  1048→      AppLogger.debug('SyncEngine: processQueue skipped — already syncing');
  1049→      return SyncResult.empty;
  1050→    }
  1051→
  1052→    // Check if we should sync at all given current connectivity.
  1053→    if (!_connectivityEngine.shouldPerformSync()) {
  1054→      AppLogger.info(
  1055→        'SyncEngine: processQueue skipped — '
  1056→        'connectivity quality is ${_connectivityEngine.state.connectionQuality.value}',
  1057→      );
  1058→      return SyncResult.empty;
  1059→    }
  1060→
  1061→    // Filter items that are eligible for processing.
  1062→    final eligibleItems = _pendingItems.where((i) {
  1063→      if (i.status == SyncStatus.completed || i.status == SyncStatus.dead) {
  1064→        return false;
  1065→      }
  1066→      if (i.status == SyncStatus.inProgress) return false;
  1067→      // Respect scheduled retry time.
  1068→      if (i.nextRetryAt != null &&
  1069→          DateTime.now().isBefore(i.nextRetryAt!)) {
  1070→        return false;
  1071→      }
  1072→      return true;
  1073→    }).toList();
  1074→
  1075→    if (eligibleItems.isEmpty) {
  1076→      AppLogger.debug('SyncEngine: no eligible items to process');
  1077→      return SyncResult.empty;
  1078→    }
  1079→
  1080→    // Determine batch size from adaptive behaviour.
  1081→    final behavior = AdaptiveBehavior(_connectivityEngine.state);
  1082→    final batchSize = behavior.batchSize;
  1083→    final batch = eligibleItems.take(batchSize).toList();
  1084→
  1085→    AppLogger.info(
  1086→      'SyncEngine: processing ${batch.length} items '
  1087→      '(batch size: $batchSize, quality: ${_connectivityEngine.state.connectionQuality.value})',
  1088→    );
  1089→
  1090→    state = state.copyWith(
  1091→      isSyncing: true,
  1092→      currentBatch: batch.length,
  1093→      syncProgress: 0.0,
  1094→    );
  1095→
  1096→    final stopwatch = Stopwatch()..start();
  1097→    var itemsSynced = 0;
  1098→    var itemsFailed = 0;
  1099→    final detectedConflicts = <SyncConflict>[];
  1100→    final errors = <String>[];
  1101→
  1102→    for (var i = 0; i < batch.length; i++) {
  1103→      if (_isDisposed) break;
  1104→
  1105→      final item = batch[i];
  1106→
  1107→      // Update progress.
  1108→      state = state.copyWith(
  1109→        syncProgress: (i + 1) / batch.length,
  1110→      );
  1111→
  1112→      try {
  1113→        final result = await processItem(item);
  1114→        result.fold(
  1115→          onSuccess: (_) {
  1116→            itemsSynced++;
  1117→          },
  1118→          onFailure: (failure) {
  1119→            // Check if this is a conflict.
  1120→            if (failure is ServerFailure && failure.statusCode == 409) {
  1121→              // Conflict detected — try auto-resolution.
  1122→              final conflictResult = _handleConflictForItem(item, failure);
  1123→              if (conflictResult != null) {
  1124→                detectedConflicts.add(conflictResult);
  1125→                itemsFailed++;
  1126→              } else {
  1127→                itemsSynced++; // Auto-resolved
  1128→              }
  1129→            } else if (failure is ValidationFailure) {
  1130→              // Validation errors are permanent — mark as dead.
  1131→              _markItemDead(item, failure.message);
  1132→              itemsFailed++;
  1133→              errors.add('${item.tableName}/${item.recordId}: ${failure.message}');
  1134→            } else if (failure is NetworkFailure) {
  1135→              // Network errors are transient — mark for retry.
  1136→              _markItemFailed(item, failure.message);
  1137→              itemsFailed++;
  1138→              errors.add('${item.tableName}/${item.recordId}: ${failure.message}');
  1139→            } else {
  1140→              _markItemFailed(item, failure.message);
  1141→              itemsFailed++;
  1142→              errors.add('${item.tableName}/${item.recordId}: ${failure.message}');
  1143→            }
  1144→          },
  1145→        );
  1146→      } catch (e, st) {
  1147→        AppLogger.error(
  1148→          'SyncEngine: unexpected error processing item ${item.id}',
  1149→          error: e,
  1150→          stackTrace: st,
  1151→        );
  1152→        _markItemFailed(item, e.toString());
  1153→        itemsFailed++;
  1154→        errors.add('${item.tableName}/${item.recordId}: $e');
  1155→      }
  1156→    }
  1157→
  1158→    stopwatch.stop();
  1159→
  1160→    final result = SyncResult(
  1161→      itemsSynced: itemsSynced,
  1162→      itemsFailed: itemsFailed,
  1163→      conflicts: detectedConflicts,
  1164→      duration: stopwatch.elapsed,
  1165→      errors: errors,
  1166→    );
  1167→
  1168→    // Refresh pending items from DB to get accurate counts.
  1169→    await _loadPendingItems();
  1170→
  1171→    state = state.copyWith(
  1172→      isSyncing: false,
  1173→      lastSyncAt: DateTime.now(),
  1174→      currentBatch: 0,
  1175→      syncProgress: 1.0,
  1176→      lastResult: result,
  1177→      pendingCount: _pendingItems
  1178→          .where((i) => i.status == SyncStatus.pending)
  1179→          .length,
  1180→      failedCount: _pendingItems
  1181→          .where((i) => i.status == SyncStatus.failed)
  1182→          .length,
  1183→    );
  1184→
  1185→    AppLogger.info(
  1186→      'SyncEngine: sync completed — '
  1187→      'synced: $itemsSynced, failed: $itemsFailed, '
  1188→      'conflicts: ${detectedConflicts.length}, '
  1189→      'duration: ${stopwatch.elapsedMilliseconds}ms',
  1190→    );
  1191→
  1192→    return result;
  1193→  }

  1195→  /// Processes a single sync item against the Supabase backend.
  1196→  ///
  1197→  /// Returns a [Result] indicating success or failure with a [Failure].
  1198→  Future<Result<void>> processItem(SyncQueueItem item) async {
  1199→    AppLogger.debug(
  1200→      'SyncEngine: processing ${item.operationType.value} '
  1201→      'on ${item.tableName}/${item.recordId ?? "new"}',
  1202→    );
  1203→
  1204→    // Mark as in-progress in memory.
  1205→    _updateItemInMemory(item.id, status: SyncStatus.inProgress);
  1206→
  1207→    try {
  1208→      final table = _supabaseClient.from(item.tableName);
  1209→
  1210→      switch (item.operationType) {
  1211→        case SyncOperationType.insert:
  1212→          await table.insert(item.payload);
  1213→          AppLogger.debug(
  1214→            'SyncEngine: inserted into ${item.tableName}',
  1215→          );
  1216→
  1217→        case SyncOperationType.update:
  1218→          if (item.recordId == null) {
  1219→            return FailureResult(Failure.validation(
  1220→              message: 'Cannot update without a recordId',
  1221→            ));
  1222→          }
  1223→          await table.update(item.payload).eq('id', item.recordId!);
  1224→          AppLogger.debug(
  1225→            'SyncEngine: updated ${item.tableName}/${item.recordId}',
  1226→          );
  1227→
  1228→        case SyncOperationType.delete:
  1229→          if (item.recordId == null) {
  1230→            return FailureResult(Failure.validation(
  1231→              message: 'Cannot delete without a recordId',
  1232→            ));
  1233→          }
  1234→          await table.delete().eq('id', item.recordId!);
  1235→          AppLogger.debug(
  1236→            'SyncEngine: deleted from ${item.tableName}/${item.recordId}',
  1237→          );
  1238→      }
  1239→
  1240→      // Mark as completed in the persistent store.
  1241→      await _cacheManager.markSyncItemCompleted(id: item.id);
  1242→      _updateItemInMemory(item.id, status: SyncStatus.completed);
  1243→
  1244→      // Update local cache to reflect the synced state.
  1245→      await _updateLocalCache(item);
  1246→
  1247→      return Success(null);
  1248→    } on sb.PostgrestException catch (e) {
  1249→      AppLogger.warning(
  1250→        'SyncEngine: Supabase error for ${item.id}: ${e.message} '
  1251→        '(code: ${e.code})',
  1252→      );
  1253→
  1254→      // Map Postgrest error codes to failure types.
  1255→      if (e.code == '409' || e.code == '23505') {
  1256→        return FailureResult(Failure.server(
  1257→          message: e.message,
  1258→          statusCode: 409,
  1259→          data: e.details,
  1260→        ));
  1261→      }
  1262→      if (e.code == '422' || e.code == '400') {
  1263→        return FailureResult(Failure.validation(
  1264→          message: e.message,
  1265→        ));
  1266→      }
  1267→      if (e.code == '401') {
  1268→        return FailureResult(Failure.unauthorized(
  1269→          message: e.message,
  1270→        ));
  1271→      }
  1272→      if (e.code == '403') {
  1273→        return FailureResult(Failure.forbidden(
  1274→          message: e.message,
  1275→        ));
  1276→      }
  1277→      if (e.code == '404') {
  1278→        return FailureResult(Failure.notFound(
  1279→          message: e.message,
  1280→        ));
  1281→      }
  1282→      return FailureResult(Failure.server(
  1283→        message: e.message,
  1284→        statusCode: int.tryParse(e.code ?? '') ?? 500,
  1285→      ));
  1286→    } on sb.AuthException catch (e) {
  1287→      return FailureResult(Failure.auth(
  1288→        message: e.message,
  1289→        code: 'auth_error',
  1290→      ));
  1291→    } catch (e) {
  1292→      // Assume network-level error for any other exception.
  1293→      return FailureResult(Failure.network(
  1294→        message: e.toString(),
  1295→      ));
  1296→    }
  1297→  }

  1303→  /// Detects a conflict by comparing local and server data using
  1304→  /// timestamps and checksums.
  1305→  ///
  1306→  /// A conflict exists when:
  1307→  ///   - The server's `updated_at` is newer than the local `updated_at`
  1308→  ///     embedded in the payload, **and**
  1309→  ///   - The payload content differs from the server version.
  1310→  ///
  1311→  /// Returns a [SyncConflict] if a conflict is detected, or `null`
  1312→  /// if the data is compatible.
  1313→  SyncConflict? detectConflict(
  1314→    SyncQueueItem item,
  1315→    Map<String, dynamic> serverData,
  1316→  ) {
  1317→    final localUpdatedAt = item.payload['updated_at'];
  1318→    final serverUpdatedAt = serverData['updated_at'];
  1319→
  1320→    // Parse timestamps for comparison.
  1321→    DateTime? localTime;
  1322→    DateTime? serverTime;
  1323→
  1324→    if (localUpdatedAt is String) {
  1325→      localTime = DateTime.tryParse(localUpdatedAt);
  1326→    } else if (localUpdatedAt is DateTime) {
  1327→      localTime = localUpdatedAt;
  1328→    }
  1329→
  1330→    if (serverUpdatedAt is String) {
  1331→      serverTime = DateTime.tryParse(serverUpdatedAt);
  1332→    } else if (serverUpdatedAt is DateTime) {
  1333→      serverTime = serverUpdatedAt;
  1334→    }
  1335→
  1336→    // If we can't parse timestamps, assume conflict to be safe.
  1337→    if (localTime == null || serverTime == null) {
  1338→      AppLogger.warning(
  1339→        'SyncEngine: conflict detection — unparseable timestamps for '
  1340→        '${item.tableName}/${item.recordId}',
  1341→      );
  1342→      return SyncConflict.create(
  1343→        userId: item.userId,
  1344→        tableName: item.tableName,
  1345→        recordId: item.recordId ?? item.id,
  1346→        localData: item.payload,
  1347→        serverData: serverData,
  1348→      );
  1349→    }
  1350→
  1351→    // Server is newer → potential conflict.
  1352→    if (serverTime.isAfter(localTime)) {
  1353→      // Check if the actual content differs by comparing checksums
  1354→      // or by a key-by-key comparison of non-metadata fields.
  1355→      if (_dataDiffers(item.payload, serverData)) {
  1356→        AppLogger.info(
  1357→          'SyncEngine: conflict detected for '
  1358→          '${item.tableName}/${item.recordId} '
  1359→          '(local: $localTime, server: $serverTime)',
  1360→        );
  1361→        return SyncConflict.create(
  1362→          userId: item.userId,
  1363→          tableName: item.tableName,
  1364→          recordId: item.recordId ?? item.id,
  1365→          localData: item.payload,
  1366→          serverData: serverData,
  1367→        );
  1368→      }
  1369→      // Server is newer but data is the same — no real conflict.
  1370→      return null;
  1371→    }
  1372→
  1373→    // Local is newer or same — no conflict (local wins).
  1374→    return null;
  1375→  }
  1376→
  1377→  /// Resolves a conflict according to the specified [resolution] strategy.
  1378→  ///
  1379→  /// For [ConflictResolution.merge], [mergedData] must be provided.
  1380→  /// For other strategies, the resolution is deterministic.
  1381→  Future<void> resolveConflict(
  1382→    SyncQueueItem item,
  1383→    ConflictResolution resolution, {
  1384→    Map<String, dynamic>? mergedData,
  1385→  }) async {
  1386→    try {
  1387→      final conflictIndex = _conflicts.indexWhere(
  1388→        (c) => c.recordId == (item.recordId ?? item.id),
  1389→      );
  1390→
  1391→      if (conflictIndex == -1) {
  1392→        AppLogger.warning(
  1393→          'SyncEngine: resolveConflict — no matching conflict found for '
  1394→          '${item.tableName}/${item.recordId}',
  1395→        );
  1396→        return;
  1397→      }
  1398→
  1399→      final conflict = _conflicts[conflictIndex];
  1400→      Map<String, dynamic> resolvedData;
  1401→
  1402→      switch (resolution) {
  1403→        case ConflictResolution.localWins:
  1404→          resolvedData = conflict.localData;
  1405→          AppLogger.info(
  1406→            'SyncEngine: conflict resolved — local wins for '
  1407→            '${item.tableName}/${conflict.recordId}',
  1408→          );
  1409→
  1410→        case ConflictResolution.serverWins:
  1411→          resolvedData = conflict.serverData;
  1412→          AppLogger.info(
  1413→            'SyncEngine: conflict resolved — server wins for '
  1414→            '${item.tableName}/${conflict.recordId}',
  1415→          );
  1416→
  1417→        case ConflictResolution.merge:
  1418→          if (mergedData == null) {
  1419→            AppLogger.error(
  1420→              'SyncEngine: merge resolution requires mergedData',
  1421→            );
  1422→            return;
  1423→          }
  1424→          resolvedData = mergedData;
  1425→          AppLogger.info(
  1426→            'SyncEngine: conflict resolved — merge for '
  1427→            '${item.tableName}/${conflict.recordId}',
  1428→          );
  1429→
  1430→        case ConflictResolution.manual:
  1431→          AppLogger.info(
  1432→            'SyncEngine: conflict resolution deferred to manual for '
  1433→            '${item.tableName}/${conflict.recordId}',
  1434→          );
  1435→          return;
  1436→      }
  1437→
  1438→      // Update the conflict record.
  1439→      _conflicts[conflictIndex] = conflict.resolve(
  1440→        resolution,
  1441→        mergedData: resolvedData,
  1442→      );
  1443→
  1444→      // Update the queue item's payload and re-enqueue.
  1445→      _updateItemInMemory(
  1446→        item.id,
  1447→        payload: resolvedData,
  1448→        status: SyncStatus.pending,
  1449→      );
  1450→
  1451→      // Update the server with resolved data.
  1452→      try {
  1453→        final table = _supabaseClient.from(item.tableName);
  1454→        if (item.recordId != null) {
  1455→          await table.update(resolvedData).eq('id', item.recordId!);
  1456→        }
  1457→        await _cacheManager.markSyncItemCompleted(id: item.id);
  1458→        _updateItemInMemory(item.id, status: SyncStatus.completed);
  1459→      } catch (e) {
  1460→        AppLogger.error(
  1461→          'SyncEngine: failed to apply resolved data to server',
  1462→          error: e,
  1463→        );
  1464→        _markItemFailed(item, 'Failed to apply resolution: $e');
  1465→      }
  1466→
  1467→      // Remove the resolved conflict.
  1468→      _conflicts.removeAt(conflictIndex);
  1469→
  1470→      state = state.copyWith(
  1471→        conflictItems: List.from(_conflicts),
  1472→        pendingCount: _pendingItems
  1473→            .where((i) => i.status == SyncStatus.pending)
  1474→            .length,
  1475→      );
  1476→    } catch (e, st) {
  1477→      AppLogger.error(
  1478→        'SyncEngine: failed to resolve conflict',
  1479→        error: e,
  1480→        stackTrace: st,
  1481→      );
  1482→    }
  1483→  }
  1484→
  1485→  /// Attempts automatic conflict resolution.
  1486→  ///
  1487→  /// Strategy:
  1488→  ///   - **Non-draft data**: Server wins (the server is the source of truth
  1489→  ///     for committed records).
  1490→  ///   - **Draft data** (payload contains `is_draft: true`): Merge by
  1491→  ///     taking the union of local and server fields, with server fields
  1492→  ///     winning for overlapping keys — except `content` and `title`,
  1493→  ///     which prefer the local version.
  1494→  ///
  1495→  /// Returns the resolution used, or `null` if auto-resolution is not
  1496→  /// possible and manual intervention is required.
  1497→  ConflictResolution? autoResolveConflict(
  1498→    SyncQueueItem item,
  1499→    Map<String, dynamic> serverData,
  1500→  ) {
  1501→    final isDraft = item.payload['is_draft'] == true;
  1502→
  1503→    if (!isDraft) {
  1504→      // For non-draft data, server is authoritative.
  1505→      AppLogger.info(
  1506→        'SyncEngine: auto-resolving conflict — server wins '
  1507→        '(non-draft ${item.tableName}/${item.recordId})',
  1508→      );
  1509→      return ConflictResolution.serverWins;
  1510→    }
  1511→
  1512→    // For drafts, attempt a merge.
  1513→    // Merge strategy: start with server data, overlay local changes for
  1514→    // user-editable fields.
  1515→    final mergedData = Map<String, dynamic>.from(serverData);
  1516→    const localPreferredFields = {'content', 'title', 'description', 'body'};
  1517→
  1518→    for (final field in localPreferredFields) {
  1519→      if (item.payload.containsKey(field) &&
  1520→          item.payload[field] != null) {
  1521→        mergedData[field] = item.payload[field];
  1522→      }
  1523→    }
  1524→
  1525→    // Check if the merge actually differs from both sides.
  1526→    // If it does, we can auto-resolve; otherwise, manual is needed.
  1527→    if (_dataDiffers(mergedData, serverData) ||
  1528→        _dataDiffers(mergedData, item.payload)) {
  1529→      AppLogger.info(
  1530→        'SyncEngine: auto-resolving conflict — merge '
  1531→        '(draft ${item.tableName}/${item.recordId})',
  1532→      );
  1533→      return ConflictResolution.merge;
  1534→    }
  1535→
  1536→    // Data is identical after merge — no real conflict.
  1537→    return ConflictResolution.serverWins;
  1538→  }

  1544→  /// Retries all failed items that are retryable.

  1548→    final retryable = _pendingItems
  1549→        .where((i) =>
  1550→            i.status == SyncStatus.failed &&
  1551→            i.isRetryable &&
  1552→            (i.nextRetryAt == null ||
  1553→                DateTime.now().isAfter(i.nextRetryAt!)))
  1554→        .toList();
  1555→
  1556→    if (retryable.isEmpty) {
  1557→      AppLogger.debug('SyncEngine: no retryable failed items');
  1558→      return SyncResult.empty;
  1559→    }
  1560→
  1561→    AppLogger.info(
  1562→      'SyncEngine: retrying ${retryable.length} failed items',
  1563→    );
  1564→
  1565→    // Reset their status to pending so processQueue picks them up.
  1566→    for (final item in retryable) {
  1567→      _updateItemInMemory(item.id, status: SyncStatus.pending);
  1568→    }
  1569→
  1570→    return processQueue();

  1573→  /// Forces a full sync of a specific table — pulls from server and
  1574→  /// pushes local changes.
  1575→  ///
  1576→  /// 1. Push any pending local mutations for [tableName].
  1577→  /// 2. Pull latest data from server and update the local cache.
  1578→  Future<SyncResult> syncTable(String tableName) async {
  1579→    AppLogger.info('SyncEngine: forcing full sync for table $tableName');
  1580→
  1581→    // Step 1: Push local changes first.
  1582→    final pushResult = await processQueue();
  1583→
  1584→    // Step 2: Pull latest from server.
  1585→    try {
  1586→      final userId = _currentUserId;
  1587→      if (userId == null) {
  1588→        AppLogger.warning('SyncEngine: cannot sync table — no user');
  1589→        return pushResult;
  1590→      }
  1591→
  1592→      final response = await _supabaseClient
  1593→          .from(tableName)
  1594→          .select()
  1595→          .eq('user_id', userId);
  1596→
  1597→      if (response is List) {
  1598→        // Cache each record locally.
  1599→        for (final record in response) {
  1600→          if (record is Map<String, dynamic>) {
  1601→            final recordId = record['id'] as String?;
  1602→            if (recordId != null) {
  1603→              await _cacheManager.cacheData(
  1604→                key: '$tableName:$recordId',
  1605→                userId: userId,
  1606→                resourceType: tableName,
  1607→                resourceId: recordId,
  1608→                data: record,
  1609→              );
  1610→            }
  1611→          }
  1612→        }
  1613→        AppLogger.info(
  1614→          'SyncEngine: pulled ${response.length} records from $tableName',
  1615→        );
  1616→      }
  1617→    } catch (e, st) {
  1618→      AppLogger.error(
  1619→        'SyncEngine: failed to pull data from $tableName',
  1620→        error: e,
  1621→        stackTrace: st,
  1622→      );
  1623→    }
  1624→
  1625→    return pushResult;
  1626→  }

  1628→  /// Forces a complete synchronisation of all tables.
  1629→  ///
  1630→  /// Processes all pending operations and then pulls the latest data
  1631→  /// for each known table.
  1632→  Future<SyncResult> forceFullSync() async {

  1649→            .select()
  1650→            .eq('user_id', userId);
  1651→
  1652→        if (response is List) {
  1653→          for (final record in response) {
  1654→            if (record is Map<String, dynamic>) {
  1655→              final recordId = record['id'] as String?;
  1656→              if (recordId != null) {
  1657→                await _cacheManager.cacheData(
  1658→                  key: '$table:$recordId',
  1659→                  userId: userId,

  1829→  /// Updates an item's in-memory state by [id].
  1830→  void _updateItemInMemory(
  1831→    String id, {
  1832→    SyncStatus? status,
  1833→    Map<String, dynamic>? payload,
  1834→    int? attempts,
  1835→    String? errorMessage,
  1836→    bool clearErrorMessage = false,
