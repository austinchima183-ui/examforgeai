   130→/// Immutable state for the main Exam Ecosystem feature.
   131→class ExamEcosystemState {
   132→  const ExamEcosystemState({
   133→    this.bodies = const [],
   134→    this.products = const [],
   135→    this.mockExams = const [],
   136→    this.isLoading = false,
   137→    this.error,
   138→    this.selectedBodyId,
   139→    this.selectedProductId,
   140→  });
   141→
   142→  final List<ExaminationBody> bodies;
   143→  final List<ExaminationProduct> products;
   144→  final List<MockExam> mockExams;
   145→  final bool isLoading;
   146→  final String? error;
   147→  final String? selectedBodyId;
   148→  final String? selectedProductId;
   149→
   150→  /// Get the currently selected body.
   151→  ExaminationBody? get selectedBody => selectedBodyId != null
   152→      ? bodies.where((b) => b.id == selectedBodyId).firstOrNull
   153→      : null;
   154→
   155→  /// Get products filtered by selected body.
   156→  List<ExaminationProduct> get filteredProducts => selectedBodyId != null
   157→      ? products.where((p) => p.examBodyId == selectedBodyId).toList()
   158→      : products;
   159→
   160→  /// Get mock exams filtered by selected product.
   161→  List<MockExam> get filteredMockExams => selectedProductId != null
   162→      ? mockExams
   163→          .where((m) => m.examinationProductId == selectedProductId)
   164→          .toList()
   165→      : selectedBodyId != null
   166→          ? mockExams
   167→              .where((m) => m.examBodyType == selectedBody?.examBodyType)
   168→              .toList()
   169→          : mockExams;
   170→
   171→  ExamEcosystemState copyWith({
   172→    List<ExaminationBody>? bodies,
   173→    List<ExaminationProduct>? products,
   174→    List<MockExam>? mockExams,
   175→    bool? isLoading,
   176→    String? error,
   177→    String? selectedBodyId,
   178→    String? selectedProductId,
   179→    bool clearSelectedBodyId = false,
   180→    bool clearSelectedProductId = false,
   181→    bool clearError = false,
   182→  }) {
   183→    return ExamEcosystemState(
   184→      bodies: bodies ?? this.bodies,
   185→      products: products ?? this.products,
   186→      mockExams: mockExams ?? this.mockExams,

   191→          : (selectedBodyId ?? this.selectedBodyId),

   199→/// Notifier for the Exam Ecosystem feature.
   200→class ExamEcosystemNotifier extends StateNotifier<ExamEcosystemState> {
   201→  ExamEcosystemNotifier({
   202→    required GetExaminationBodiesUseCase getExaminationBodies,
   203→    required GetExaminationProductsUseCase getExaminationProducts,
   204→    required GetMockExamsUseCase getMockExams,
   205→    required ExamEcosystemRepository repository,
   206→  })  : _getExaminationBodies = getExaminationBodies,
   207→        _getExaminationProducts = getExaminationProducts,
   208→        _getMockExams = getMockExams,
   209→        _repository = repository,
   210→        super(const ExamEcosystemState());
   211→
   212→  final GetExaminationBodiesUseCase _getExaminationBodies;
   213→  final GetExaminationProductsUseCase _getExaminationProducts;
   214→  final GetMockExamsUseCase _getMockExams;
   215→  final ExamEcosystemRepository _repository;
   216→
   217→  /// Loads all initial data (bodies, products, mock exams).
   218→  Future<void> loadAll() async {
   219→    state = state.copyWith(isLoading: true, clearError: true);
   220→
   221→    final results = await Future.wait([
   222→      _getExaminationBodies(const GetExaminationBodiesParams()),
   223→      _getExaminationProducts(const GetExaminationProductsParams()),
   224→      _getMockExams(const GetMockExamsParams()),
   225→    ]);
   226→
   227→    final bodiesResult = results[0] as Result<List<ExaminationBody>>;
   228→    final productsResult = results[1] as Result<List<ExaminationProduct>>;
   229→    final mockExamsResult = results[2] as Result<List<MockExam>>;
   230→
   231→    bodiesResult.fold(
   232→      onSuccess: (bodies) {
   233→        state = state.copyWith(bodies: bodies);
   234→      },
   235→      onFailure: (failure) {
   236→        state = state.copyWith(error: _mapFailure(failure));
   237→      },
   238→    );
   239→
   240→    productsResult.fold(
   241→      onSuccess: (products) {
   242→        state = state.copyWith(products: products);
   243→      },
   244→      onFailure: (failure) {
   245→        AppLogger.warning('Failed to load products: $failure');
   246→      },
   247→    );
   248→
   249→    mockExamsResult.fold(
   250→      onSuccess: (mockExams) {
   251→        state = state.copyWith(mockExams: mockExams);
   252→      },
   253→      onFailure: (failure) {
   254→        AppLogger.warning('Failed to load mock exams: $failure');
   255→      },
   256→    );
   257→
   258→    state = state.copyWith(isLoading: false);
   259→  }
   260→
   261→  /// Select an exam body to filter products.
   262→  Future<void> selectBody(String? bodyId) async {
   263→    state = state.copyWith(
   264→      selectedBodyId: bodyId,
   265→      clearSelectedProductId: bodyId == null,
   266→    );
   267→  }
   268→
   269→  /// Select a product to filter mock exams.
   270→  void selectProduct(String? productId) {
   271→    state = state.copyWith(selectedProductId: productId);
   272→  }
   273→
   274→  /// Refresh all data.
   275→  Future<void> refresh() async {
   276→    await loadAll();
   277→  }
   278→
   279→  /// Clear any error.
   280→  void clearError() {
   281→    state = state.copyWith(clearError: true);
   282→  }
   283→
   284→  String _mapFailure(Failure failure) {
   285→    return failure.when(
   286→      server: (message, _, __) => message,
   287→      cache: (message) => message,
   288→      auth: (message, _) => message,
   289→      network: (message) => message,
   290→      validation: (message, _) => message,
   291→      notFound: (message) => message,
   292→      unauthorized: (message) => message,
   293→      forbidden: (message) => message,
   294→    );
   295→  }
   296→}

   313→/// Immutable state for the Readiness feature.
   314→class ReadinessState {
   315→  const ReadinessState({
   316→    this.assessments = const [],
   317→    this.currentAssessment,
   318→    this.isLoading = false,
   319→    this.error,
   320→  });
   321→
   322→  final List<ReadinessAssessment> assessments;
   323→  final ReadinessAssessment? currentAssessment;
   324→  final bool isLoading;
   325→  final String? error;
   326→
   327→  /// Overall readiness score (average of all assessments).
   328→  double get overallReadinessScore {
   329→    if (assessments.isEmpty) return 0;
   330→    return assessments
   331→            .map((a) => a.readinessScore)
   332→            .reduce((a, b) => a + b) /
   333→        assessments.length;
   334→  }
   335→
   336→  /// Average readiness level.
   337→  ReadinessLevel get overallReadinessLevel {
   338→    if (assessments.isEmpty) return ReadinessLevel.notStarted;
   339→    final score = overallReadinessScore;
   340→    if (score >= 90) return ReadinessLevel.examReady;
   341→    if (score >= 75) return ReadinessLevel.advanced;
   342→    if (score >= 60) return ReadinessLevel.proficient;
   343→    if (score >= 40) return ReadinessLevel.developing;
   344→    if (score >= 20) return ReadinessLevel.beginning;
   345→    return ReadinessLevel.notStarted;
   346→  }
   347→
   348→  /// All weak topics across assessments.
   349→  List<String> get allWeakTopics =>
   350→      assessments.expand((a) => a.weakTopics).toSet().toList();
   351→
   352→  /// All strong topics across assessments.
   353→  List<String> get allStrongTopics =>
   354→      assessments.expand((a) => a.strongTopics).toSet().toList();
   355→
   356→  /// All recommendations across assessments.
   357→  List<String> get allRecommendations =>
   358→      assessments.expand((a) => a.recommendations).toSet().toList();
   359→
   360→  ReadinessState copyWith({
   361→    List<ReadinessAssessment>? assessments,
   362→    ReadinessAssessment? currentAssessment,
   363→    bool? isLoading,
   364→    String? error,
   365→    bool clearCurrentAssessment = false,
   366→    bool clearError = false,
   367→  }) {
   368→    return ReadinessState(
   369→      assessments: assessments ?? this.assessments,
   370→      currentAssessment: clearCurrentAssessment
   371→          ? null
   372→          : (currentAssessment ?? this.currentAssessment),
   373→      isLoading: isLoading ?? this.isLoading,
   374→      error: clearError ? null : (error ?? this.error),
   375→    );
   376→  }
   377→}

   380→class ReadinessNotifier extends StateNotifier<ReadinessState> {
   381→  ReadinessNotifier({
   382→    required GetReadinessAssessmentUseCase getReadinessAssessment,
   383→    required CalculateReadinessUseCase calculateReadiness,
   384→    required GetExamReadinessUseCase getExamReadiness,
   385→    required ExamEcosystemRepository repository,

   388→        _calculateReadiness = calculateReadiness,
   389→        _getExamReadiness = getExamReadiness,
   390→        _repository = repository,
   391→        _userId = userId,
   392→        super(const ReadinessState());
   393→
   394→  final GetReadinessAssessmentUseCase _getReadinessAssessment;
   395→  final CalculateReadinessUseCase _calculateReadiness;
   396→  final GetExamReadinessUseCase _getExamReadiness;
   397→  final ExamEcosystemRepository _repository;
   398→  final String? _userId;
   399→
   400→  /// Load readiness assessments for a user.
   401→  Future<void> loadReadiness({String? examBodyId}) async {
   402→    if (_userId == null) return;
   403→
   404→    state = state.copyWith(isLoading: true, clearError: true);
   405→
   406→    final result = await _repository.getUserReadiness(
   407→      userId: _userId,
   408→      examBodyId: examBodyId,
   409→    );
   410→
   411→    result.fold(
   412→      onSuccess: (assessments) {
   413→        state = state.copyWith(
   414→          isLoading: false,
   415→          assessments: assessments,
   416→          currentAssessment: assessments.isNotEmpty ? assessments.first : null,
   417→        );
   418→      },
   419→      onFailure: (failure) {
   420→        state = state.copyWith(
   421→          isLoading: false,
   422→          error: _mapFailure(failure),
   423→        );
   424→      },
   425→    );
   426→  }
   427→
   428→  /// Calculate readiness for a specific exam.
   429→  Future<void> calculateReadiness({
   430→    required String examBodyId,
   431→    String? subjectId,
   432→  }) async {
   433→    if (_userId == null) return;
   434→
   435→    state = state.copyWith(isLoading: true, clearError: true);
   436→
   437→    final result = await _calculateReadiness(
   438→      CalculateReadinessParams(
   439→        userId: _userId,
   440→        examBodyId: examBodyId,
   441→        subjectId: subjectId,
   442→      ),
   443→    );
   444→
   445→    result.fold(
   446→      onSuccess: (assessment) {
   447→        state = state.copyWith(
   448→          isLoading: false,
   449→          currentAssessment: assessment,
   450→          assessments: [assessment, ...state.assessments],
   451→        );
   452→      },
   453→      onFailure: (failure) {
   454→        state = state.copyWith(
   455→          isLoading: false,
   456→          error: _mapFailure(failure),
   457→        );
   458→      },
   459→    );
   460→  }
   461→
   462→  /// Get exam-level readiness summary.
   463→  Future<void> loadExamReadiness({required String examBodyId}) async {
   464→    if (_userId == null) return;
   465→
   466→    state = state.copyWith(isLoading: true, clearError: true);
   467→
   468→    final result = await _getExamReadiness(
   469→      GetExamReadinessParams(userId: _userId, examBodyId: examBodyId),
   470→    );
   471→
   472→    result.fold(
   473→      onSuccess: (assessments) {
   474→        state = state.copyWith(
   475→          isLoading: false,
   476→          assessments: assessments,
   477→        );
   478→      },
   479→      onFailure: (failure) {
   480→        state = state.copyWith(
   481→          isLoading: false,
   482→          error: _mapFailure(failure),
   483→        );
   484→      },
   485→    );
   486→  }

   524→/// Immutable state for the Study Plan feature.

   527→    this.plans = const [],
   528→    this.currentPlan,
   529→    this.activities = const [],
   530→    this.isLoading = false,
   531→    this.isGenerating = false,

   535→  final List<StudyPlan> plans;
   536→  final StudyPlan? currentPlan;
   537→  final List<StudyPlanActivity> activities;
   538→  final bool isLoading;
   539→  final bool isGenerating;
   540→  final String? error;
   541→
   542→  /// Active plans.
   543→  List<StudyPlan> get activePlans =>
   544→      plans.where((p) => p.isActive).toList();
   545→
   546→  /// Current streak from active plan.
   547→  int get currentStreak => currentPlan?.currentStreakDays ?? 0;
   548→
   549→  /// Longest streak across all plans.
   550→  int get longestStreak =>
   551→      plans.fold(0, (max, p) => p.longestStreakDays > max ? p.longestStreakDays : max);
   552→
   553→  /// Today's activities.
   554→  List<StudyPlanActivity> get todayActivities => activities
   555→      .where((a) => a.isScheduledToday && !a.isCompleted)
   556→      .toList();
   557→
   558→  /// Overdue activities.
   559→  List<StudyPlanActivity> get overdueActivities =>
   560→      activities.where((a) => a.isOverdue).toList();
   561→
   562→  /// Completed activities.
   563→  List<StudyPlanActivity> get completedActivities =>
   564→      activities.where((a) => a.isCompleted).toList();
   565→
   566→  StudyPlanState copyWith({
   567→    List<StudyPlan>? plans,
   568→    StudyPlan? currentPlan,

   587→}

   606→        super(const StudyPlanState());
   607→
   608→  final GetStudyPlansUseCase _getStudyPlans;
   609→  final CreateStudyPlanUseCase _createStudyPlan;
   610→  final GenerateAiStudyPlanUseCase _generateAiStudyPlan;
   611→  final GetStudyPlanActivitiesUseCase _getStudyPlanActivities;
   612→  final CompleteStudyPlanActivityUseCase _completeStudyPlanActivity;
   613→  final ExamEcosystemRepository _repository;
   614→  final String? _userId;
   615→
   616→  /// Load all study plans for the current user.
   617→  Future<void> loadPlans() async {
   618→    if (_userId == null) return;
   619→
   620→    state = state.copyWith(isLoading: true, clearError: true);
   621→
   622→    final result = await _getStudyPlans(
   623→      GetStudyPlansParams(userId: _userId),
   624→    );
   625→
   626→    result.fold(
   627→      onSuccess: (plans) {
   628→        state = state.copyWith(
   629→          isLoading: false,
   630→          plans: plans,
   631→          currentPlan: plans.isNotEmpty ? plans.first : null,
   632→        );
   633→        if (plans.isNotEmpty) {
   634→          loadActivities(plans.first.id);
   635→        }
   636→      },
   637→      onFailure: (failure) {
   638→        state = state.copyWith(
   639→          isLoading: false,
   640→          error: _mapFailure(failure),
   641→        );
   642→      },
   643→    );
   644→  }
   645→
   646→  /// Load activities for a specific plan.
   647→  Future<void> loadActivities(String planId) async {
   648→    state = state.copyWith(isLoading: true);
   649→
   650→    final result = await _getStudyPlanActivities(
   651→      GetStudyPlanActivitiesParams(studyPlanId: planId),
   652→    );
   653→
   654→    result.fold(
   655→      onSuccess: (activities) {
   656→        state = state.copyWith(
   657→          isLoading: false,
   658→          activities: activities,
   659→        );
   660→      },
   661→      onFailure: (failure) {
   662→        state = state.copyWith(
   663→          isLoading: false,
   664→          error: _mapFailure(failure),
   665→        );
   666→      },
   667→    );
   668→  }
   669→
   670→  /// Select a plan as the current plan.
   671→  Future<void> selectPlan(StudyPlan plan) async {
   672→    state = state.copyWith(currentPlan: plan);
   673→    await loadActivities(plan.id);
   674→  }
   675→
   676→  /// Create a new study plan.
   677→  Future<void> createPlan(StudyPlan plan) async {
   678→    state = state.copyWith(isLoading: true, clearError: true);
   679→
   680→    final result = await _createStudyPlan(
   681→      CreateStudyPlanParams(studyPlan: plan),
   682→    );
   683→
   684→    result.fold(
   685→      onSuccess: (created) {
   686→        state = state.copyWith(
   687→          isLoading: false,
   688→          plans: [created, ...state.plans],
   689→          currentPlan: created,
   690→        );
   691→        loadActivities(created.id);
   692→      },
   693→      onFailure: (failure) {
   694→        state = state.copyWith(
   695→          isLoading: false,
   696→          error: _mapFailure(failure),
   697→        );
   698→      },
   699→    );
   700→  }

   702→  /// Generate an AI study plan.
   703→  Future<void> generateAiPlan({
   704→    required String examBodyId,
   705→    String? subjectId,
   706→    String? educationalLevelId,
   707→    DateTime? targetDate,
   708→    int dailyStudyMinutes = 60,
   709→  }) async {
   710→    if (_userId == null) return;
   711→
   712→    state = state.copyWith(isGenerating: true, clearError: true);
   713→
   714→    final result = await _generateAiStudyPlan(
   715→      GenerateAiStudyPlanParams(
   716→        userId: _userId,
   717→        examBodyId: examBodyId,
   718→        subjectId: subjectId,
   719→        educationalLevelId: educationalLevelId,
   720→        targetDate: targetDate,
   721→        dailyStudyMinutes: dailyStudyMinutes,
   722→      ),
   723→    );
   724→
   725→    result.fold(
   726→      onSuccess: (generated) {
   727→        state = state.copyWith(
   728→          isGenerating: false,
   729→          plans: [generated, ...state.plans],
   730→          currentPlan: generated,
   731→        );
   732→        loadActivities(generated.id);
   733→      },
   734→      onFailure: (failure) {
   735→        state = state.copyWith(
   736→          isGenerating: false,
   737→          error: _mapFailure(failure),
   738→        );
   739→      },
   740→    );
   741→  }
   742→
   743→  /// Complete a study plan activity.
   744→  Future<void> completeActivity({
   745→    required String activityId,
   746→    double? performanceScore,
   747→    Map<String, dynamic>? metadata,
   748→  }) async {
   749→    final result = await _completeStudyPlanActivity(
   750→      CompleteStudyPlanActivityParams(
   751→        activityId: activityId,
   752→        performanceScore: performanceScore,
   753→        metadata: metadata,
   754→      ),
   755→    );
   756→
   757→    result.fold(
   758→      onSuccess: (completed) {
   759→        state = state.copyWith(
   760→          activities: state.activities
   761→              .map((a) => a.id == completed.id ? completed : a)
   762→              .toList(),
   763→        );
   764→      },
   765→      onFailure: (failure) {
   766→        AppLogger.warning('Failed to complete activity: $failure');
   767→      },
   768→    );
   769→  }
   770→
   771→  /// Delete a study plan.
   772→  Future<void> deletePlan(String planId) async {
   773→    final result = await _repository.deleteStudyPlan(planId);
   774→
   775→    result.fold(
   776→      onSuccess: (_) {
   777→        final updatedPlans =
   778→            state.plans.where((p) => p.id != planId).toList();
   779→        state = state.copyWith(
   780→          plans: updatedPlans,
   781→          currentPlan: state.currentPlan?.id == planId
   782→              ? (updatedPlans.isNotEmpty ? updatedPlans.first : null)
   783→              : state.currentPlan,
   784→        );
   785→      },
   786→      onFailure: (failure) {
   787→        state = state.copyWith(error: _mapFailure(failure));
   788→      },
   789→    );
   790→  }

   792→  /// Clear the current error.
   793→  void clearError() {
   794→    state = state.copyWith(clearError: true);
   795→  }

   797→  String _mapFailure(Failure failure) {
   798→    return failure.when(
   799→      server: (message, _, __) => message,
   800→      cache: (message) => message,
   801→      auth: (message, _) => message,
   802→      network: (message) => message,
   803→      validation: (message, _) => message,
   804→      notFound: (message) => message,
   805→      unauthorized: (message) => message,
   806→      forbidden: (message) => message,
   807→    );
   808→  }
