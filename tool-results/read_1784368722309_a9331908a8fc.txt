    99→/// Riverpod provider for the application's [GoRouter].
   100→///
   101→/// Watches [authStateProvider] so that the router automatically
   102→/// re-evaluates its [redirect] callback whenever the authentication
   103→/// state changes (sign-in, sign-out, token refresh).
   104→///
   105→/// Usage in `main.dart`:
   106→/// ```dart
   107→/// final router = ref.read(appRouterProvider);
   108→/// MaterialApp.router(routerConfig: router);
   109→/// ```
   110→final appRouterProvider = Provider<GoRouter>((ref) {
   111→  // Watch auth state so the router re-evaluates redirects on change.
   112→  final authState = ref.watch(authStateProvider);
   113→
   114→  // Also watch onboarding state and role so those redirects stay live.
   115→  ref.watch(onboardingCompleteProvider);
   116→  ref.watch(userRoleProvider);
   117→
   118→  return GoRouter(
   119→    navigatorKey: _rootNavigatorKey,
   120→    initialLocation: RouteNames.splash,
   121→    debugLogDiagnostics: true,
   122→
   123→    // ─── Global Redirect ──────────────────────────────────────────
   124→    redirect: (context, state) {
   125→      final currentPath = state.matchedLocation;
   126→      final isAuthenticated = authState.when(
   127→        data: (authData) => authData.session != null,
   128→        loading: () {
   129→          // While loading, only allow splash. Everything else redirects
   130→          // to splash so we don't flash the login page briefly.
   131→          if (currentPath != RouteNames.splash) {
   132→            return RouteNames.splash;
   133→          }
   134→          return null;
   135→        },
   136→        error: (_, __) => false,
   137→      );
   138→
   139→      // ── Auth Guard ────────────────────────────────────────────
   140→      final authRedirect = AuthGuard.evaluate(
   141→        isAuthenticated: isAuthenticated,
   142→        currentPath: currentPath,
   143→      );
   144→      if (authRedirect != null) return authRedirect;
   145→
   146→      // ── Onboarding Guard ─────────────────────────────────────
   147→      final onboardingAsync = ref.read(onboardingCompleteProvider);
   148→      final isOnboardingComplete = onboardingAsync.when(
   149→        data: (complete) => complete,
   150→        loading: () => true, // Assume complete while loading to avoid flicker.
   151→        error: (_, __) => true,
   152→      );
   153→
   154→      final onboardingRedirect = OnboardingGuard.evaluate(
   155→        isAuthenticated: isAuthenticated,
   156→        isOnboardingComplete: isOnboardingComplete,
   157→        currentPath: currentPath,
   158→      );
   159→      if (onboardingRedirect != null) return onboardingRedirect;
   160→
   161→      // ── Role-Based Guard ─────────────────────────────────────
   162→      final roleAsync = ref.read(userRoleProvider);
   163→      final userRole = roleAsync.when(
   164→        data: UserRole.fromString,
   165→        loading: () => null,
   166→        error: (_, __) => null,
   167→      );
   168→
   169→      final roleRedirect = RoleBasedGuard.evaluate(
   170→        userRole: userRole,
   171→        currentPath: currentPath,
   172→      );
   173→      if (roleRedirect != null) return roleRedirect;
   174→
   175→      // No redirect needed — proceed as requested.
   176→      return null;
   177→    },
   178→

   180→    routes: [
   181→      // ── Splash ────────────────────────────────────────────────

   694→// NOT FOUND PAGE (404)
   695→// ═══════════════════════════════════════════════════════════════════════
   696→
   697→/// Shown when the user navigates to a route that does not exist in
   698→/// the route tree.
   699→///
   700→/// Provides a friendly message and a button to navigate back to the
   701→/// dashboard (or login if unauthenticated).
   702→class NotFoundPage extends ConsumerWidget {
   703→  const NotFoundPage({
   704→    required this.error,
   705→    this.path,
   706→    super.key,
   707→  });
   708→
   709→  /// The [GoException] from GoRouter describing the error.
   710→  final GoException? error;
   711→
   712→  /// The unmatched path the user attempted to visit.
   713→  final String? path;
   714→
   715→  @override
   716→  Widget build(BuildContext context, WidgetRef ref) {
   717→    final theme = Theme.of(context);
   718→    final isAuthenticated = ref.watch(isAuthenticatedProvider);
   719→
   720→    return Scaffold(
   721→      body: Center(
   722→        child: Padding(
   723→          padding: const EdgeInsets.all(32),
   724→          child: Column(
   725→            mainAxisSize: MainAxisSize.min,
   726→            children: [
   727→              Icon(
   728→                Icons.search_off_rounded,
   729→                size: 80,
   730→                color: theme.colorScheme.error,
   731→              ),
   732→              const SizedBox(height: 24),
   733→              Text(
   734→                'Page Not Found',
   735→                style: theme.textTheme.headlineMedium?.copyWith(
   736→                  fontWeight: FontWeight.bold,
   737→                ),
   738→                textAlign: TextAlign.center,
   739→              ),
   740→              const SizedBox(height: 12),
   741→              if (path != null)
   742→                Text(
   743→                  'The page "$path" does not exist.',
   744→                  style: theme.textTheme.bodyLarge?.copyWith(
   745→                    color: theme.colorScheme.onSurfaceVariant,
   746→                  ),
   747→                  textAlign: TextAlign.center,
   748→                ),
   749→              if (error != null) ...[
   750→                const SizedBox(height: 8),
   751→                Text(
   752→                  error!.message,
   753→                  style: theme.textTheme.bodySmall?.copyWith(
   754→                    color: theme.colorScheme.outline,
   755→                  ),
   756→                  textAlign: TextAlign.center,
   757→                ),
   758→              ],
   759→              const SizedBox(height: 32),
   760→              FilledButton.icon(
   761→                onPressed: () => context.go(
   762→                  isAuthenticated
   763→                      ? RouteNames.dashboard
   764→                      : RouteNames.login,
   765→                ),
   766→                icon: const Icon(Icons.home),
   767→                label: Text(
   768→                  isAuthenticated ? 'Go to Dashboard' : 'Go to Login',
   769→                ),
   770→              ),
   771→            ],
   772→          ),
   773→        ),
   774→      ),
   775→    );
   776→  }
   777→}
