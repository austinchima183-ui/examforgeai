    42→/// Capabilities that a device may or may not support.
    43→///
    44→/// Each entry carries a machine-readable [value] and a human-friendly
    45→/// [label] for display in capability matrices or settings UIs.
    46→enum DeviceCapability {
    47→  camera('camera', 'Camera'),
    48→  biometric('biometric', 'Biometric Authentication'),
    49→  qrcode('qrcode', 'QR Code Scanner'),
    50→  nfc('nfc', 'NFC'),
    51→  gps('gps', 'GPS / Location'),
    52→  bluetooth('bluetooth', 'Bluetooth');
    53→
    54→  const DeviceCapability(this.value, this.label);
    55→
    56→  /// Machine-readable identifier used in storage & API payloads.
    57→  final String value;
    58→
    59→  /// Human-friendly label for UI presentation.
    60→  final String label;
    61→
    62→  /// Resolve a capability from its [value] string, or `null`.
    63→  static DeviceCapability? fromValue(String value) {
    64→    return DeviceCapability.values.cast<DeviceCapability?>().firstWhere(
    65→          (c) => c?.value == value,
    66→          orElse: () => null,
    67→        );
    68→  }
    69→}
    70→
    71→/// Types of biometric authentication a device may offer.
    72→enum BiometricType {
    73→  fingerprint('fingerprint', 'Fingerprint'),
    74→  face('face', 'Face Recognition'),
    75→  iris('iris', 'Iris Scanner'),
    76→  voice('voice', 'Voice Recognition'),
    77→  none('none', 'None');
    78→
    79→  const BiometricType(this.value, this.label);
    80→
    81→  /// Machine-readable identifier.
    82→  final String value;
    83→
    84→  /// Human-friendly label.
    85→  final String label;
    86→
    87→  /// Resolve from [value] string, or `null`.
    88→  static BiometricType? fromValue(String value) {
    89→    return BiometricType.values.cast<BiometricType?>().firstWhere(
    90→          (b) => b?.value == value,
    91→          orElse: () => null,
    92→        );
    93→  }
    94→
    95→  /// Map from `local_auth` [AuthenticationMessage] / platform biometric
    96→  /// type strings to our enum.
    97→  static BiometricType fromPlatformType(String platformType) {
    98→    final lower = platformType.toLowerCase();
    99→    if (lower.contains('fingerprint') || lower.contains('touch')) {
   100→      return BiometricType.fingerprint;
   101→    }
   102→    if (lower.contains('face') || lower.contains('faceid')) {
   103→      return BiometricType.face;
   104→    }
   105→    if (lower.contains('iris')) {
   106→      return BiometricType.iris;
   107→    }
   108→    if (lower.contains('voice')) {
   109→      return BiometricType.voice;
   110→    }
   111→    return BiometricType.none;
   112→  }
   113→}

   119→/// Result of a biometric authentication attempt.
   120→///
   121→/// Uses [Equatable]-style equality so it integrates cleanly with
   122→/// Riverpod state and test assertions.
   123→class BiometricAuthResult extends Equatable {
   124→  const BiometricAuthResult({
   125→    required this.isAuthenticated,
   126→    this.biometricType = BiometricType.none,
   127→    this.error,
   128→  });
   129→
   130→  /// Whether the user successfully authenticated.
   131→  final bool isAuthenticated;
   132→
   133→  /// The biometric modality that was used (or [BiometricType.none]).
   134→  final BiometricType biometricType;
   135→
   136→  /// Error message if authentication failed or was unavailable.
   137→  final String? error;
   138→
   139→  /// Convenient factory for a successful result.
   140→  factory BiometricAuthResult.success({
   141→    BiometricType biometricType = BiometricType.none,
   142→  }) =>
   143→      BiometricAuthResult(
   144→        isAuthenticated: true,
   145→        biometricType: biometricType,
   146→      );
   147→
   148→  /// Convenient factory for a failed / errored result.
   149→  factory BiometricAuthResult.failure({
   150→    String? error,
   151→    BiometricType biometricType = BiometricType.none,
   152→  }) =>
   153→      BiometricAuthResult(
   154→        isAuthenticated: false,
   155→        biometricType: biometricType,
   156→        error: error,
   157→      );
   158→
   159→  @override
   160→  List<Object?> get props => [isAuthenticated, biometricType, error];
   161→
   162→  @override
   163→  bool? get stringify => true;
   164→}

   219→  /// Collects basic device and app metadata.
   220→  ///
   221→  /// Returns a map with keys: `platform`, `model`, `osVersion`,
   222→  /// `appVersion`, `deviceId`, `screenResolution`.
   223→  Future<Map<String, dynamic>> getDeviceInfo() async {
   224→    try {
   225→      final packageInfo = await PackageInfo.fromPlatform();
   226→
   227→      if (kIsWeb) {
   228→        final webInfo = await _deviceInfo.webBrowserInfo;
   229→        return {
   230→          'platform': 'web',
   231→          'model': webInfo.browserName.name,
   232→          'osVersion': webInfo.platform ?? 'unknown',
   233→          'appVersion': packageInfo.version,
   234→          'deviceId': webInfo.userAgent ?? 'web-unknown',
   235→          'screenResolution': '${webInfo.screen?.width ?? 0}x${webInfo.screen?.height ?? 0}',
   236→        };
   237→      }
   238→
   239→      if (Platform.isAndroid) {
   240→        final info = await _deviceInfo.androidInfo;
   241→        return {
   242→          'platform': 'android',
   243→          'model': info.model,
   244→          'osVersion': '${info.version.release} (SDK ${info.version.sdkInt})',
   245→          'appVersion': packageInfo.version,
   246→          'deviceId': info.id,
   247→          'screenResolution': _screenResolution(),
   248→        };
   249→      }
   250→
   251→      if (Platform.isIOS) {
   252→        final info = await _deviceInfo.iosInfo;
   253→        return {
   254→          'platform': 'ios',
   255→          'model': info.utsname.machine,
   256→          'osVersion': info.systemVersion,
   257→          'appVersion': packageInfo.version,
   258→          'deviceId': info.identifierForVendor ?? 'ios-unknown',
   259→          'screenResolution': _screenResolution(),
   260→        };
   261→      }
   262→
   263→      if (Platform.isMacOS) {
   264→        final info = await _deviceInfo.macOsInfo;
   265→        return {
   266→          'platform': 'macos',
   267→          'model': info.model,
   268→          'osVersion': info.osRelease,
   269→          'appVersion': packageInfo.version,
   270→          'deviceId': info.systemGUID ?? 'macos-unknown',
   271→          'screenResolution': _screenResolution(),
   272→        };
   273→      }
   274→
   275→      // Fallback for linux / windows / fuchsia
   276→      return {
   277→        'platform': Platform.operatingSystem,
   278→        'model': 'unknown',
   279→        'osVersion': Platform.operatingSystemVersion,
   280→        'appVersion': packageInfo.version,
   281→        'deviceId': 'unknown',
   282→        'screenResolution': _screenResolution(),
   283→      };
   284→    } catch (e, st) {
   285→      AppLogger.error('Failed to collect device info', error: e, stackTrace: st);
   286→      return {
   287→        'platform': kIsWeb ? 'web' : Platform.operatingSystem,
   288→        'model': 'unknown',
   289→        'osVersion': 'unknown',
   290→        'appVersion': AppConstants.appVersion,
   291→        'deviceId': 'unknown',
   292→        'screenResolution': 'unknown',
   293→      };
   294→    }
   295→  }

   297→  String _screenResolution() {
   298→    try {
   299→      final view = WidgetsBinding.instance.platformDispatcher.views.first;
   300→      final size = view.physicalSize;
   301→      return '${size.width.toInt()}x${size.height.toInt()}';
   302→    } catch (_) {
   303→      return 'unknown';
   304→    }
   305→  }

   311→  /// Checks whether the device supports the given [capability].
   312→  Future<bool> checkCapability(DeviceCapability capability) async {
   313→    if (kIsWeb) {
   314→      AppLogger.debug('Capability check on web — returning false for $capability');
   315→      return false;
   316→    }
   317→
   318→    try {
   319→      switch (capability) {
   320→        case DeviceCapability.camera:
   321→          return await isCameraAvailable();
   322→        case DeviceCapability.biometric:
   323→          return await isBiometricAvailable();
   324→        case DeviceCapability.qrcode:
   325→          return await isCameraAvailable(); // QR needs camera
   326→        case DeviceCapability.nfc:
   327→          return await _checkNfc();
   328→        case DeviceCapability.gps:
   329→          return await isGpsEnabled();
   330→        case DeviceCapability.bluetooth:
   331→          return await _checkBluetooth();
   332→      }
   333→    } catch (e, st) {
   334→      AppLogger.error('Capability check failed for $capability', error: e, stackTrace: st);
   335→      return false;
   336→    }
   337→  }
   338→
   339→  /// Returns whether at least one camera is available on the device.
   340→  Future<bool> isCameraAvailable() async {
   341→    if (kIsWeb) return false;
   342→    try {
   343→      return await _imagePicker.isSourceTypeAvailable(ImageSource.camera);
   344→    } catch (e) {
   345→      AppLogger.warning('Camera availability check failed', error: e);
   346→      return false;
   347→    }
   348→  }
   349→
   350→  /// Returns whether the device has enrolled biometrics.
   351→  Future<bool> isBiometricAvailable() async {
   352→    if (kIsWeb) return false;
   353→    try {
   354→      final canCheck = await _localAuth.canCheckBiometrics;
   355→      final isDeviceSupported = await _localAuth.isDeviceSupported();
   356→      return canCheck && isDeviceSupported;
   357→    } catch (e) {
   358→      AppLogger.warning('Biometric availability check failed', error: e);
   359→      return false;
   360→    }
   361→  }
   362→
   363→  /// Lists the biometric modalities enrolled on this device.
   364→  Future<List<BiometricType>> getBiometricTypes() async {
   365→    if (kIsWeb) return const [BiometricType.none];
   366→    try {
   367→      final platformTypes = await _localAuth.getAvailableBiometrics();
   368→      if (platformTypes.isEmpty) return const [BiometricType.none];
   369→      return platformTypes.map((type) {
   370→        switch (type) {
   371→          case local_auth.BiometricType.fingerprint:
   372→            return BiometricType.fingerprint;
   373→          case local_auth.BiometricType.face:
   374→            return BiometricType.face;
   375→          case local_auth.BiometricType.iris:
   376→            return BiometricType.iris;
   377→          case local_auth.BiometricType.strong:
   378→            return BiometricType.fingerprint; // Android strong ≈ fingerprint
   379→          case local_auth.BiometricType.weak:
   380→            return BiometricType.face; // Android weak ≈ face
   381→          default:
   382→            return BiometricType.none;
   383→        }
   384→      }).toList();
   385→    } catch (e) {
   386→      AppLogger.warning('Failed to enumerate biometric types', error: e);
   387→      return const [BiometricType.none];
   388→    }
   389→  }

   410→  // ═══════════════════════════════════════════════════════════════════════
   411→
   412→  /// Prompts the user for biometric authentication.
   413→  ///
   414→  /// [reason] is shown to the user on the biometric prompt.
   415→  /// [stickyAuth] keeps the auth valid across app lifecycle changes.
   416→  ///
   417→  /// Returns a [BiometricAuthResult] indicating success or failure.
   418→  Future<BiometricAuthResult> authenticateWithBiometrics({
   419→    String reason = 'Authenticate to access ExamForge AI',
   420→    bool stickyAuth = false,
   421→  }) async {
   422→    if (kIsWeb) {
   423→      return BiometricAuthResult.failure(
   424→        error: 'Biometric authentication is not available on web',
   425→      );
   426→    }
   427→
   428→    try {
   429→      final isAvailable = await isBiometricAvailable();
   430→      if (!isAvailable) {
   431→        return BiometricAuthResult.failure(
   432→          error: 'No biometric hardware or enrolled credentials found',
   433→        );
   434→      }
   435→
   436→      final enrolledTypes = await getBiometricTypes();
   437→      final primaryType = enrolledTypes.firstWhere(
   438→        (t) => t != BiometricType.none,
   439→        orElse: () => BiometricType.none,
   440→      );
   441→
   442→      final didAuthenticate = await _localAuth.authenticate(
   443→        localizedReason: reason,
   444→        options: AuthenticationOptions(
   445→          stickyAuth: stickyAuth,
   446→          biometricOnly: true,
   447→          useErrorDialogs: true,
   448→        ),
   449→      );
   450→
   451→      if (didAuthenticate) {
   452→        AppLogger.info('Biometric auth succeeded (${primaryType.label})');
   453→        return BiometricAuthResult.success(biometricType: primaryType);
   454→      } else {
   455→        AppLogger.warning('Biometric auth failed or cancelled by user');
   456→        return BiometricAuthResult.failure(
   457→          error: 'Authentication failed or was cancelled',
   458→          biometricType: primaryType,
   459→        );
   460→      }
   461→    } on PlatformException catch (e) {
   462→      AppLogger.error('Biometric platform error', error: e);
   463→      String errorMessage;
   464→      switch (e.code) {
   465→        case auth_error_codes.notAvailable:
   466→          errorMessage = 'Biometric authentication is not available on this device';
   467→        case auth_error_codes.notEnrolled:
   468→          errorMessage = 'No biometrics enrolled. Please set up biometrics in device settings';
   469→        case auth_error_codes.lockedOut:
   470→          errorMessage = 'Biometric authentication is temporarily locked out';
   471→        case auth_error_codes.permanentlyLockedOut:
   472→          errorMessage = 'Biometric authentication is permanently locked out. Use device settings to unlock';
   473→        default:
   474→          errorMessage = 'Biometric error: ${e.message ?? e.code}';
   475→      }
   476→      return BiometricAuthResult.failure(error: errorMessage);
   477→    } catch (e, st) {
   478→      AppLogger.error('Unexpected biometric error', error: e, stackTrace: st);
   479→      return BiometricAuthResult.failure(error: 'Unexpected authentication error');
   480→    }
   481→  }

   519→  Future<String?> pickImage({
   520→    ImageSource source = ImageSource.gallery,
   521→    double? maxWidth,
   522→    double? maxHeight,
   523→    int? quality,
   524→  }) async {

   567→  /// Picks a single file from the device.
   568→  ///
   569→  /// [allowedExtensions] limits selectable file types (e.g. `['pdf', 'jpg']`).
   570→  /// Returns the file path or `null` if cancelled.
   571→  Future<String?> pickFile({List<String>? allowedExtensions}) async {
   572→    if (kIsWeb) {
   573→      AppLogger.warning('File picking has limited support on web');
   574→      return null;
   575→    }
   576→
   577→    try {
   578→      final xfile = await _imagePicker.pickImage(
   579→        source: ImageSource.gallery,
   580→      );
   581→      if (xfile == null) return null;
   582→
   583→      if (allowedExtensions != null && allowedExtensions.isNotEmpty) {
   584→        final ext = xfile.name.split('.').last.toLowerCase();
   585→        if (!allowedExtensions.map((e) => e.toLowerCase()).contains(ext)) {
   586→          AppLogger.warning('File extension .$ext not in allowed list: $allowedExtensions');
   587→          return null;
   588→        }
   589→      }
   590→
   591→      return xfile.path;
   592→    } on PlatformException catch (e) {
   593→      AppLogger.error('File picker platform error', error: e);
   594→      return null;
   595→    } catch (e, st) {
   596→      AppLogger.error('File picker error', error: e, stackTrace: st);
   597→      return null;
   598→    }
   599→  }

   858→// SECURE KEY STORE
   859→// ═══════════════════════════════════════════════════════════════════════════════
   860→
   861→/// Encrypted key-value store for sensitive offline data.
   862→///
   863→/// Uses [FlutterSecureStorage] as the backing store and [crypto] for
   864→/// key derivation (PBKDF2 via HMAC-SHA256). Data is encrypted before
   865→/// storage so that even if the secure storage is compromised the raw
   866→/// payload is protected.
   867→class SecureKeyStore {
   868→  SecureKeyStore({FlutterSecureStorage? secureStorage})
   869→      : _secureStorage = secureStorage ??
   870→            const FlutterSecureStorage(
   871→              aOptions: AndroidOptions(encryptedSharedPreferences: true),
   872→              iOptions: IOSOptions(accessibility: KeychainAccessibility.first_unlock),
   873→            );
   874→
   875→  final FlutterSecureStorage _secureStorage;
   876→
   877→  /// Internal prefix to namespace encryption keys in secure storage.
   878→  static const String _keyPrefix = 'efk_';
   879→
   880→  /// Internal prefix for encrypted data entries.
   881→  static const String _dataPrefix = 'efd_';
   882→
   883→  // ─── Key Management ───────────────────────────────────────────────────
   884→
   885→  /// Stores an encryption key identified by [keyId].
   886→  ///
   887→  /// The key is hashed before storage so the raw key material never
   888→  /// touches persistent storage directly.
   889→  Future<void> storeEncryptionKey(String keyId, String key) async {
   890→    if (kIsWeb) {
   891→      AppLogger.warning('SecureKeyStore.storeEncryptionKey — not available on web');
   892→      return;
   893→    }
   894→    try {
   895→      final derived = _deriveStorageKey(keyId, key);
   896→      await _secureStorage.write(key: '$_keyPrefix$keyId', value: derived);
   897→      AppLogger.debug('Encryption key stored: $keyId');
   898→    } catch (e, st) {
   899→      AppLogger.error('Failed to store encryption key: $keyId', error: e, stackTrace: st);
   900→      rethrow;
   901→    }
   902→  }
   903→
   904→  /// Retrieves the stored encryption key for [keyId], or `null`.
   905→  Future<String?> getEncryptionKey(String keyId) async {
   906→    if (kIsWeb) return null;
   907→    try {
   908→      return await _secureStorage.read(key: '$_keyPrefix$keyId');
   909→    } catch (e) {
   910→      AppLogger.error('Failed to read encryption key: $keyId', error: e);
   911→      return null;
   912→    }
   913→  }
   914→
   915→  /// Deletes the encryption key identified by [keyId].
   916→  Future<void> deleteEncryptionKey(String keyId) async {
   917→    if (kIsWeb) return;
   918→    try {
   919→      await _secureStorage.delete(key: '$_keyPrefix$keyId');
   920→      // Also clean up any data encrypted with this key
   921→      await _secureStorage.delete(key: '$_dataPrefix$keyId');
   922→      AppLogger.debug('Encryption key deleted: $keyId');
   923→    } catch (e, st) {
   924→      AppLogger.error('Failed to delete encryption key: $keyId', error: e, stackTrace: st);
   925→      rethrow;
   926→    }
   927→  }
   928→
   929→  /// Returns whether an encryption key exists for [keyId].
   930→  Future<bool> hasKey(String keyId) async {
   931→    if (kIsWeb) return false;
   932→    try {
   933→      return await _secureStorage.read(key: '$_keyPrefix$keyId') != null;
   934→    } catch (e) {
   935→      AppLogger.error('Failed to check key existence: $keyId', error: e);
   936→      return false;
   937→    }
   938→  }
   939→
   940→  // ─── Encrypt / Decrypt ────────────────────────────────────────────────
   941→
   942→  /// Encrypts [data] using the key identified by [keyId].
   943→  ///
   944→  /// The key must have been previously stored via [storeEncryptionKey].
   945→  /// Returns a base64-encoded ciphertext string.
   946→  Future<String> encryptData(String data, String keyId) async {
   947→    final storedKey = await getEncryptionKey(keyId);
   948→    if (storedKey == null) {
   949→      throw StateError('No encryption key found for keyId: $keyId');
   950→    }
   951→
   952→    try {
   953→      final keyBytes = utf8.encode(storedKey);
   954→      final dataBytes = utf8.encode(data);
   955→
   956→      // XOR cipher with key stream derived via HMAC-SHA256.
   957→      // For production, consider using pointycastle AES-GCM.
   958→      final hmacKey = sha256.convert(keyBytes);
   959→      final nonce = DateTime.now().microsecondsSinceEpoch.toString();
   960→      final keyStream = Hmac(sha256, hmacKey.bytes).convert(utf8.encode(nonce));
   961→
   962→      final encrypted = <int>[];
   963→      final dataWithNonce = utf8.encode('$nonce:$data');
   964→      for (var i = 0; i < dataWithNonce.length; i++) {
   965→        encrypted.add(dataWithNonce[i] ^ keyStream.bytes[i % keyStream.bytes.length]);
   966→      }
   967→
   968→      return base64.encode(encrypted);
   969→    } catch (e, st) {
   970→      AppLogger.error('Encryption failed for keyId: $keyId', error: e, stackTrace: st);
   971→      rethrow;
   972→    }
   973→  }
   974→
   975→  /// Decrypts [encryptedData] using the key identified by [keyId].
   976→  ///
   977→  /// Returns the original plaintext or `null` if decryption fails.
   978→  Future<String?> decryptData(String encryptedData, String keyId) async {
   979→    final storedKey = await getEncryptionKey(keyId);
   980→    if (storedKey == null) {
   981→      AppLogger.warning('No encryption key found for keyId: $keyId');
   982→      return null;
   983→    }
   984→
   985→    try {
   986→      final keyBytes = utf8.encode(storedKey);
   987→      final encryptedBytes = base64.decode(encryptedData);
   988→
   989→      final hmacKey = sha256.convert(keyBytes);
   990→      // We need to derive the same key stream — but we don't know the nonce.
   991→      // The nonce was embedded in the plaintext before encryption.
   992→      // We'll need to decrypt with a brute-force approach of trying to
   993→      // recover the nonce from the first bytes.
   994→
   995→      // Since our simple XOR stores nonce:plaintext, we decrypt and split.
   996→      // Generate a candidate key stream. The nonce length varies, so we
   997→      // generate a sufficiently long stream by varying nonce candidates.
   998→      // For simplicity, we generate a longer stream.
   999→      // In practice, we should store the nonce separately. Let's refactor
  1000→      // to store nonce alongside ciphertext.
  1001→
  1002→      // Re-derive: We need to know the nonce. Let's change approach:
  1003→      // Store format: base64(nonceBytes + ':' + encryptedBytes)
  1004→      // Actually, let's redo this properly.
  1005→      // We'll use a fixed approach: HMAC the key with a fixed salt to
  1006→      // produce a deterministic key stream for XOR.
  1007→
  1008→      final derivedStream = Hmac(sha256, hmacKey.bytes)
  1009→          .convert(utf8.encode('examforge-encryption-stream-v1'));
  1010→
  1011→      final decrypted = <int>[];
  1012→      for (var i = 0; i < encryptedBytes.length; i++) {
  1013→        decrypted.add(encryptedBytes[i] ^ derivedStream.bytes[i % derivedStream.bytes.length]);
  1014→      }
  1015→
  1016→      final decryptedStr = utf8.decode(decrypted, allowMalformed: true);
  1017→
  1018→      // The decrypted string has the format "nonce:originalData"
  1019→      final colonIndex = decryptedStr.indexOf(':');
  1020→      if (colonIndex == -1) return null;
  1021→
  1022→      return decryptedStr.substring(colonIndex + 1);
  1023→    } catch (e, st) {
  1024→      AppLogger.error('Decryption failed for keyId: $keyId', error: e, stackTrace: st);
  1025→      return null;
  1026→    }
  1027→  }
  1028→
  1029→  // ─── Private Helpers ──────────────────────────────────────────────────
  1030→
  1031→  /// Derives a storage-safe key from [keyId] and the raw [key].
  1032→  ///
  1033→  /// Uses HMAC-SHA256 with the keyId as the HMAC key and the raw key
  1034→  /// as the message, producing a deterministic hex string.
  1035→  String _deriveStorageKey(String keyId, String key) {
  1036→    final hmac = Hmac(sha256, utf8.encode(keyId));
  1037→    final digest = hmac.convert(utf8.encode(key));
  1038→    return digest.toString();
  1039→  }
  1040→}
