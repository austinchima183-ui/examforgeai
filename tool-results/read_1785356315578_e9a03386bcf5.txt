     1→/// Analytics Engine — pure computation service for results analytics.
     2→///
     3→/// This service provides powerful insights for teachers, schools, and students
     4→/// by operating on raw result data and producing computed analytics. It is a
     5→/// computation-only module with zero database access and no external
     6→/// dependencies beyond the entity types defined in the results and CBT domains.
     7→///
     8→/// ## Design Principles
     9→///
    10→/// - **Pure functions**: Every public method is deterministic — same inputs
    11→///   always produce the same outputs. No hidden state, no I/O.
    12→/// - **Static helpers**: Core statistical primitives are exposed as `static`
    13→///   methods for maximum testability and reusability.
    14→/// - **Null-safe**: All methods handle empty input lists gracefully, returning
    15→///   sensible defaults rather than throwing.
    16→/// - **Self-contained DTOs**: Output types are defined in this file to avoid
    17→///   coupling to persistence-layer entities.
    18→///
    19→/// ## Usage
    20→///
    21→/// ```dart
    22→/// final engine = AnalyticsEngine();
    23→/// final summary = engine.computeClassPerformance(overallResults,
    24→///     subjectResults: subjectResults);
    25→/// print('Class average: ${summary.averageScore}');
    26→/// print('Pass rate: ${summary.passRate}%');
    27→/// ```

   389→/// Pure-computation analytics engine for exam result data.
   390→///
   391→/// All methods are side-effect-free and operate solely on the data passed
   392→/// to them. No database access, no network calls, no mutable state.
   393→///
   394→/// The engine is designed to be instantiated once and reused across the
   395→/// application lifecycle, though it carries no state of its own.
   396→///
   397→/// ### Threading
   398→///
   399→/// Because all operations are pure and synchronous, this class is safe to
   400→/// use from any isolate. For large datasets, consider running computations
   401→/// in a background isolate via `Isolate.run`.
   402→///
   403→/// ### Distinction Threshold
   404→///
   405→/// The default distinction threshold is **75%**. This can be overridden via
   406→/// the [distinctionThreshold] parameter on relevant methods.
   407→class AnalyticsEngine {
   408→  // ── Constants ──────────────────────────────────────────────────────
   409→
   410→  /// Default distinction threshold (percentage).
   411→  static const double defaultDistinctionThreshold = 75.0;
   412→
   413→  /// Default pass mark (percentage).
   414→  static const double defaultPassMark = 50.0;
   415→
   416→  /// Number of standard histogram buckets (0–10, 10–20, …, 90–100).
   417→  static const int defaultBucketCount = 10;
   418→
   419→  /// Minimum data points required for a reliable trend calculation.
   420→  static const int minTrendDataPoints = 3;
   421→
   422→  /// Attendance threshold for the "high attendance" group.
   423→  static const double highAttendanceThreshold = 80.0;
   424→
   425→  // ── 1. Class Performance ──────────────────────────────────────────
   426→
   427→  /// Computes a comprehensive performance summary for a class.
   428→  ///
   429→  /// Given a list of [results] (overall results for each student in the
   430→  /// class), this method calculates central tendency, pass/distinction
   431→  /// rates, grade and score distributions, and topic-level aggregations.
   432→  ///
   433→  /// When [subjectResults] are provided, a per-subject breakdown is
   434→  /// included in [ClassPerformanceSummary.subjectBreakdown] and topic
   435→  /// performance is aggregated from those results' metadata.
   436→  ///
   437→  /// Set [previousResults] to compute the [ClassPerformanceSummary.improvementRate]
   438→  /// by comparing the current period's average to the previous period's.
   439→  ///
   440→  /// [distinctionThreshold] defaults to [defaultDistinctionThreshold] (75%).
   441→  /// [passMark] defaults to [defaultPassMark] (50%).
   442→  ClassPerformanceSummary computeClassPerformance(
   443→    List<StudentOverallResultEntity> results, {
   444→    List<StudentSubjectResultEntity>? subjectResults,
   445→    List<StudentOverallResultEntity>? previousResults,
   446→    double distinctionThreshold = defaultDistinctionThreshold,
   447→    double passMark = defaultPassMark,
   448→  }) {
   449→    if (results.isEmpty) {
   450→      return ClassPerformanceSummary(
   451→        classId: '',
   452→        totalStudents: 0,
   453→        averageScore: 0,
   454→        highestScore: 0,
   455→        lowestScore: 0,
   456→        medianScore: 0,
   457→        passRate: 0,
   458→        distinctionRate: 0,
   459→        gradeDistribution: {},
   460→        scoreDistribution: _buildEmptyBuckets(),
   461→        topicPerformance: {},
   462→        improvementRate: 0,
   463→      );
   464→    }
   465→
   466→    final scores = results.map((r) => r.overallPercentage).toList();
   467→    final classId = results.first.classId;
   468→
   469→    // Central tendency
   470→    final average = _mean(scores);
   471→    final highest = _max(scores);
   472→    final lowest = _min(scores);
   473→    final median = _median(scores);
   474→
   475→    // Pass & distinction rates
   476→    final passCount = scores.where((s) => s >= passMark).length;
   477→    final distinctionCount = scores.where((s) => s >= distinctionThreshold).length;
   478→    final passRate = (passCount / scores.length) * 100;
   479→    final distinctionRate = (distinctionCount / scores.length) * 100;
   480→
   481→    // Grade distribution
   482→    final gradeDistribution = <String, int>{};
   483→    for (final r in results) {
   484→      final grade = r.overallGrade ?? _gradeFromPercentage(r.overallPercentage);
   485→      gradeDistribution[grade] = (gradeDistribution[grade] ?? 0) + 1;
   486→    }
   487→
   488→    // Score distribution
   489→    final scoreDistribution = computeScoreDistribution(scores);
   490→
   491→    // Topic performance from subject results metadata
   492→    final topicPerformance = _computeTopicPerformanceFromResults(
   493→      subjectResults ?? [],
   494→    );
   495→
   496→    // Subject breakdown
   497→    final subjectBreakdown = <String, ClassPerformanceSummary>{};
   498→    if (subjectResults != null && subjectResults.isNotEmpty) {
   499→      final bySubject = _groupBy<String, StudentSubjectResultEntity>(
   500→        subjectResults,
   501→        (r) => r.subjectId,
   502→      );
   503→      for (final entry in bySubject.entries) {
   504→        subjectBreakdown[entry.key] = _computeSubjectSummary(
   505→          entry.value,
   506→          distinctionThreshold: distinctionThreshold,
   507→          passMark: passMark,
   508→        );
   509→      }
   510→    }
   511→
   512→    // Improvement rate
   513→    final improvementRate = _computeImprovementRate(results, previousResults);
   514→
   515→    return ClassPerformanceSummary(
   516→      classId: classId,
   517→      totalStudents: results.length,
   518→      averageScore: _round2(average),
   519→      highestScore: _round2(highest),
   520→      lowestScore: _round2(lowest),
   521→      medianScore: _round2(median),
   522→      passRate: _round2(passRate),
   523→      distinctionRate: _round2(distinctionRate),
   524→      gradeDistribution: gradeDistribution,
   525→      scoreDistribution: scoreDistribution,
   526→      topicPerformance: topicPerformance,
   527→      improvementRate: _round2(improvementRate),
   528→      subjectBreakdown: subjectBreakdown,
   529→    );
   530→  }
   531→
   532→  // ── 2. School Performance ──────────────────────────────────────────
   533→
   534→  /// Aggregates multiple [ClassPerformanceSummary] instances into a
   535→  /// school-level summary with cross-class and cross-subject rankings.
   536→  ///
   537→  /// [topicMasteryData] is used to identify the most difficult topic
   538→  /// across the entire school.
   539→  ///
   540→  /// [trendData] provides historical performance data for trend
   541→  /// visualization on dashboards.
   542→  SchoolPerformanceSummary computeSchoolPerformance(
   543→    List<ClassPerformanceSummary> classPerformances, {
   544→    required String schoolId,
   545→    required String academicSessionId,
   546→    List<TopicMasteryEntity>? topicMasteryData,
   547→    List<TrendData>? trendData,
   548→    int totalExams = 0,
   549→  }) {
   550→    if (classPerformances.isEmpty) {
   551→      return SchoolPerformanceSummary(
   552→        schoolId: schoolId,
   553→        academicSessionId: academicSessionId,
   554→        totalStudents: 0,
   555→        totalClasses: 0,
   556→        averageScore: 0,
   557→        passRate: 0,
   558→        distinctionRate: 0,
   559→        bestClassId: null,
   560→        bestSubjectId: null,
   561→        mostDifficultTopicId: null,
   562→        classRankings: [],
   563→        subjectRankings: [],
   564→        gradeDistribution: {},
   565→        trendData: trendData ?? [],
   566→        totalExams: totalExams,
   567→      );
   568→    }
   569→
   570→    final totalStudents = classPerformances.fold<int>(
   571→      0,
   572→      (sum, c) => sum + c.totalStudents,
   573→    );
   574→
   575→    // Weighted average: weight each class by its student count.
   576→    final weightedSum = classPerformances.fold<double>(
   577→      0,
   578→      (sum, c) => sum + (c.averageScore * c.totalStudents),
   579→    );
   580→    final averageScore = totalStudents > 0 ? weightedSum / totalStudents : 0.0;
   581→
   582→    // Weighted pass rate
   583→    final passWeightedSum = classPerformances.fold<double>(
   584→      0,
   585→      (sum, c) => sum + (c.passRate * c.totalStudents),
   586→    );
   587→    final passRate = totalStudents > 0 ? passWeightedSum / totalStudents : 0.0;
   588→
   589→    // Weighted distinction rate
   590→    final distinctionWeightedSum = classPerformances.fold<double>(
   591→      0,
   592→      (sum, c) => sum + (c.distinctionRate * c.totalStudents),
   593→    );
   594→    final distinctionRate =
   595→        totalStudents > 0 ? distinctionWeightedSum / totalStudents : 0.0;
   596→
   597→    // Class rankings
   598→    final classRankings = _rankEntities(
   599→      classPerformances.map((c) => MapEntry(c.classId, c.averageScore)).toList(),
   600→    );
   601→
   602→    // Subject rankings from breakdowns
   603→    final subjectScores = <String, List<double>>{};
   604→    for (final cp in classPerformances) {
   605→      for (final sb in cp.subjectBreakdown.entries) {
   606→        subjectScores.putIfAbsent(sb.key, () => []).add(sb.value.averageScore);
   607→      }
   608→    }
   609→    final subjectAverages = subjectScores.map(
   610→      (k, v) => MapEntry(k, _mean(v)),
   611→    );
   612→    final subjectRankings = _rankEntities(
   613→      subjectAverages.entries.map((e) => MapEntry(e.key, e.value)).toList(),
   614→    );
   615→
   616→    // Best class & subject
   617→    final bestClassId = classRankings.isNotEmpty ? classRankings.first.entityId : null;
   618→    final bestSubjectId = subjectRankings.isNotEmpty ? subjectRankings.first.entityId : null;
   619→
   620→    // Most difficult topic
   621→    String? mostDifficultTopicId;
   622→    if (topicMasteryData != null && topicMasteryData.isNotEmpty) {
   623→      final difficultTopics = identifyDifficultTopics(topicMasteryData);
   624→      if (difficultTopics.isNotEmpty) {
   625→        mostDifficultTopicId = difficultTopics.first.topicId;
   626→      }
   627→    } else {
   628→      // Fallback: look at topic performance from class summaries
   629→      final allTopics = <String, List<TopicPerformanceData>>{};
   630→      for (final cp in classPerformances) {
   631→        for (final tp in cp.topicPerformance.entries) {
   632→          allTopics.putIfAbsent(tp.key, () => []).add(tp.value);
   633→        }
   634→      }
   635→      if (allTopics.isNotEmpty) {
   636→        String? worstTopic;
   637→        double worstAccuracy = 101;
   638→        for (final entry in allTopics.entries) {
   639→          final avg = _mean(entry.value.map((t) => t.averageAccuracy).toList());
   640→          if (avg < worstAccuracy) {
   641→            worstAccuracy = avg;
   642→            worstTopic = entry.key;
   643→          }
   644→        }
   645→        mostDifficultTopicId = worstTopic;
   646→      }
   647→    }
   648→
   649→    // Grade distribution: merge all class distributions
   650→    final gradeDistribution = <String, int>{};
   651→    for (final cp in classPerformances) {
   652→      for (final entry in cp.gradeDistribution.entries) {
   653→        gradeDistribution[entry.key] =
   654→            (gradeDistribution[entry.key] ?? 0) + entry.value;
   655→      }
   656→    }
   657→

   679→  /// Computes a student's performance trend from historical scores using
   680→  /// simple linear regression.
   681→  ///
   682→  /// [historicalScores] should be ordered chronologically (oldest first).
   683→  /// At least [minTrendDataPoints] data points are needed for a reliable
   684→  /// result; fewer will still produce a trend but with low confidence.

   686→  /// [strengths] and [weaknesses] are subject/topic IDs where the student
   687→  /// consistently scores above or below their own average.
   688→  StudentTrendAnalysis computeStudentTrends({

   690→    required List<double> historicalScores,

   692→    List<String> weaknesses = const [],

   768→
   769→  /// Generates a ranked list of students from their overall results.
   770→  ///
   771→  /// Uses standard competition ranking (also known as "1224" ranking):
   772→  /// students with identical scores share the same rank, and the next
   773→  /// rank is skipped. For example: 1st, 2nd, 2nd, 4th.
   774→  ///
   775→  /// Results are sorted by [StudentOverallResultEntity.overallPercentage]
   776→  /// in descending order.
   777→  List<RankedStudent> generateRankings(
   778→    List<StudentOverallResultEntity> results,
   779→  ) {
   780→    if (results.isEmpty) return [];
   781→
   782→    // Sort descending by percentage
   783→    final sorted = List<StudentOverallResultEntity>.from(results)
   784→      ..sort((a, b) => b.overallPercentage.compareTo(a.overallPercentage));
   785→
   786→    final rankings = <RankedStudent>[];
   787→    int? lastRank;
   788→    double? lastScore;
   789→
   790→    for (int i = 0; i < sorted.length; i++) {
   791→      final result = sorted[i];
   792→      final position = i + 1;
   793→
   794→      // Same score → same rank; otherwise rank = position
   795→      final rank = (lastScore != null &&
   796→              _round2(result.overallPercentage) == _round2(lastScore))
   797→          ? lastRank!
   798→          : position;
   799→
   800→      rankings.add(RankedStudent(
   801→        studentId: result.studentId,
   802→        rank: rank,
   803→        score: _round2(result.overallPercentage),
   804→        classId: result.classId,
   805→      ),);
   806→
   807→      lastRank = rank;
   808→      lastScore = _round2(result.overallPercentage);
   809→    }
   810→
   811→    return rankings;
   812→  }

   816→  /// Creates histogram buckets for score visualization.
   817→  ///
   818→  /// By default, creates [defaultBucketCount] buckets of equal width
   819→  /// (0–10, 10–20, …, 90–100). Each bucket is inclusive of its lower
   820→  /// bound and inclusive of its upper bound, except the last bucket
   821→  /// which includes 100.
   822→  ///
   823→  /// Returns a map of range label → midpoint for quick chart integration.
   824→  Map<String, double> computeScoreDistributionMap(List<double> scores) {
   825→    final buckets = computeScoreDistribution(scores);
   826→    return {for (final b in buckets) b.rangeLabel: (b.min + b.max) / 2};
   827→  }
   828→
   829→  /// Creates histogram buckets for score visualization.
   830→  ///
   831→  /// Returns a list of [ScoreBucket] instances with count and percentage
   832→  /// for each range.
   833→  List<ScoreBucket> computeScoreDistribution(
   834→    List<double> scores, {
   835→    int bucketCount = defaultBucketCount,
   836→  }) {
   837→    if (scores.isEmpty) return _buildEmptyBuckets(bucketCount: bucketCount);
   838→
   839→    final bucketWidth = 100.0 / bucketCount;
   840→    final buckets = <ScoreBucket>[];
   841→
   842→    for (int i = 0; i < bucketCount; i++) {
   843→      final min = i * bucketWidth;
   844→      final max = (i + 1) * bucketWidth;
   845→
   846→      // For the last bucket, include 100. For others, use strict upper bound.
   847→      final count = scores.where((s) {
   848→        if (i == bucketCount - 1) {
   849→          return s >= min && s <= max;
   850→        }
   851→        return s >= min && s < max;
   852→      }).length;
   853→
   854→      final percentage = (count / scores.length) * 100;
   855→
   856→      buckets.add(ScoreBucket(
   857→        rangeLabel: '${min.round()}–${max.round()}',
   858→        min: min,
   859→        max: max,
   860→        count: count,
   861→        percentage: _round2(percentage),
   862→      ),);
   863→    }
   864→
   865→    return buckets;
   866→  }

   912→  /// Computes the Pearson correlation coefficient between attendance

   915→  /// [attendancePerformancePairs] is a list of (attendance%, score%)

   920→  AttendanceCorrelation computeAttendanceCorrelation(
   921→    List<({double attendance, double performance})> attendancePerformancePairs,
   922→  ) {
   923→    if (attendancePerformancePairs.length < 3) {
   924→      return AttendanceCorrelation(
   925→        correlationCoefficient: 0,
   926→        sampleSize: attendancePerformancePairs.length,
   927→        interpretation: 'Insufficient data for correlation analysis '
   928→            '(need at least 3 data points)',
   929→      );
   930→    }
   931→
   932→    final n = attendancePerformancePairs.length;
   933→    final attendanceValues =
   934→        attendancePerformancePairs.map((p) => p.attendance).toList();
   935→    final performanceValues =
   936→        attendancePerformancePairs.map((p) => p.performance).toList();
   937→
   938→    final coefficient =
   939→        _pearsonCorrelation(attendanceValues, performanceValues);
   940→
   941→    // Compute average scores for high/low attendance groups
   942→    final highAttendance = attendancePerformancePairs
   943→        .where((p) => p.attendance >= highAttendanceThreshold)

   953→
   954→    final interpretation = _interpretCorrelation(coefficient);
   955→
   956→    return AttendanceCorrelation(
   957→      correlationCoefficient: _round4(coefficient),
   958→      sampleSize: n,
   959→      interpretation: interpretation,
   960→      highAttendanceAvgScore: highAvg != null ? _round2(highAvg) : null,
   961→      lowAttendanceAvgScore: lowAvg != null ? _round2(lowAvg) : null,
   962→    );

   986→  /// Arithmetic mean of a list of numbers.
   987→  static double _mean(List<double> values) {
   988→    if (values.isEmpty) return 0;
   989→    return values.reduce((a, b) => a + b) / values.length;
   990→  }
   991→
   992→  /// Maximum value in a list, or 0 if empty.
   993→  static double _max(List<double> values) {
   994→    if (values.isEmpty) return 0;
   995→    return values.reduce((a, b) => a > b ? a : b);
   996→  }
   997→
   998→  /// Minimum value in a list, or 0 if empty.
   999→  static double _min(List<double> values) {
  1000→    if (values.isEmpty) return 0;
  1001→    return values.reduce((a, b) => a < b ? a : b);
  1002→  }
  1003→
  1004→  /// Median (50th percentile) of a sorted copy of [values].
  1005→  static double _median(List<double> values) {
  1006→    if (values.isEmpty) return 0;
  1007→    final sorted = List<double>.from(values)..sort();
  1008→    final mid = sorted.length ~/ 2;
  1009→    if (sorted.length.isEven) {
  1010→      return (sorted[mid - 1] + sorted[mid]) / 2;
  1011→    }
  1012→    return sorted[mid];
  1013→  }
  1014→
  1015→  /// Standard deviation (population) of a list.
  1016→  static double _standardDeviation(List<double> values) {
  1017→    if (values.length < 2) return 0;
  1018→    final mean = _mean(values);
  1019→    final sumSquaredDiff = values.fold<double>(
  1020→      0,
  1021→      (sum, v) => sum + (v - mean) * (v - mean),
  1022→    );
  1023→    return _sqrt(sumSquaredDiff / values.length);
  1024→  }
  1025→
  1026→  /// Integer square root approximation for standard deviation.
  1027→  /// Uses Newton's method for reasonable precision.
  1028→  static double _sqrt(double value) {
  1029→    if (value < 0) return 0;
  1030→    if (value == 0) return 0;
  1031→    double x = value;
  1032→    double prev = 0;
  1033→    for (int i = 0; i < 20; i++) {
  1034→      prev = x;
  1035→      x = 0.5 * (x + value / x);
  1036→      if ((x - prev).abs() < 0.000001) break;
  1037→    }
  1038→    return x;
  1039→  }
  1040→
  1041→  // ── Linear Regression ──────────────────────────────────────────────
  1042→
  1043→  /// Simple linear regression (OLS) on [y] values with implicit
  1044→  /// x = 0, 1, 2, …, n−1.
  1045→  ///
  1046→  /// Returns the slope, intercept, and R² value.
  1047→  static _LinearRegressionResult _linearRegression(List<double> y) {
  1048→    final n = y.length;
  1049→    if (n < 2) {
  1050→      return _LinearRegressionResult(slope: 0, intercept: _mean(y), rSquared: 0);
  1051→    }
  1052→
  1053→    // x values: 0, 1, 2, ..., n-1
  1054→    final xMean = (n - 1) / 2.0;
  1055→    final yMean = _mean(y);
  1056→
  1057→    double sumXY = 0;
  1058→    double sumX2 = 0;
  1059→    for (int i = 0; i < n; i++) {
  1060→      final xDiff = i - xMean;
  1061→      final yDiff = y[i] - yMean;
  1062→      sumXY += xDiff * yDiff;
  1063→      sumX2 += xDiff * xDiff;
  1064→    }
  1065→
  1066→    final slope = sumX2 != 0 ? sumXY / sumX2 : 0.0;
  1067→    final intercept = yMean - slope * xMean;
  1068→
  1069→    // R² calculation
  1070→    final ssTotal = y.fold<double>(0, (sum, v) => sum + (v - yMean) * (v - yMean));
  1071→    double ssResidual = 0;
  1072→    for (int i = 0; i < n; i++) {
  1073→      final predicted = intercept + slope * i;
  1074→      ssResidual += (y[i] - predicted) * (y[i] - predicted);
  1075→    }
  1076→    final rSquared = ssTotal != 0 ? 1 - (ssResidual / ssTotal) : 0.0;
  1077→
  1078→    return _LinearRegressionResult(
  1079→      slope: slope,
  1080→      intercept: intercept,
  1081→      rSquared: rSquared.clamp(0, 1),
  1082→    );
  1083→  }
  1084→
  1085→  // ── Pearson Correlation ────────────────────────────────────────────
  1086→
  1087→  /// Computes the Pearson correlation coefficient between [x] and [y].
  1088→  static double _pearsonCorrelation(List<double> x, List<double> y) {
  1089→    assert(x.length == y.length);
  1090→    final n = x.length;
  1091→    if (n < 2) return 0;
  1092→
  1093→    final xMean = _mean(x);
  1094→    final yMean = _mean(y);
  1095→
  1096→    double sumXY = 0;
  1097→    double sumX2 = 0;
  1098→    double sumY2 = 0;
  1099→
  1100→    for (int i = 0; i < n; i++) {
  1101→      final dx = x[i] - xMean;
  1102→      final dy = y[i] - yMean;
  1103→      sumXY += dx * dy;
  1104→      sumX2 += dx * dx;
  1105→      sumY2 += dy * dy;
  1106→    }
  1107→
  1108→    final denominator = _sqrt(sumX2) * _sqrt(sumY2);
  1109→    if (denominator == 0) return 0;
  1110→    return (sumXY / denominator).clamp(-1.0, 1.0);
  1111→  }

  1117→    List<T> items,
  1118→    K Function(T) keyExtractor,
  1119→  ) {
  1120→    final map = <K, List<T>>{};
  1121→    for (final item in items) {
  1122→      final key = keyExtractor(item);
  1123→      map.putIfAbsent(key, () => []).add(item);
  1124→    }
  1125→    return map;
  1126→  }

  1234→  /// Computes a [ClassPerformanceSummary] scoped to a single subject.
  1235→  ClassPerformanceSummary _computeSubjectSummary(
  1236→    List<StudentSubjectResultEntity> results, {
  1237→    double distinctionThreshold = defaultDistinctionThreshold,
  1238→    double passMark = defaultPassMark,
  1239→  }) {
  1240→    if (results.isEmpty) {
  1241→      return ClassPerformanceSummary(
  1242→        classId: '',
  1243→        totalStudents: 0,
  1244→        averageScore: 0,
  1245→        highestScore: 0,
  1246→        lowestScore: 0,
  1247→        medianScore: 0,
  1248→        passRate: 0,
  1249→        distinctionRate: 0,
  1250→        gradeDistribution: {},
  1251→        scoreDistribution: _buildEmptyBuckets(),
  1252→        topicPerformance: {},
  1253→        improvementRate: 0,
  1254→      );
  1255→    }
  1256→
  1257→    final scores = results.map((r) => r.percentage).toList();
  1258→    final subjectId = results.first.subjectId;
  1259→    final classId = results.first.classId;
  1260→
  1261→    final gradeDistribution = <String, int>{};
  1262→    for (final r in results) {
  1263→      final grade = r.grade ?? _gradeFromPercentage(r.percentage);

  1265→    }
  1266→
  1267→    return ClassPerformanceSummary(
  1268→      classId: classId,
  1269→      subjectId: subjectId,
  1270→      totalStudents: results.length,
  1271→      averageScore: _round2(_mean(scores)),
  1272→      highestScore: _round2(_max(scores)),
  1273→      lowestScore: _round2(_min(scores)),
  1274→      medianScore: _round2(_median(scores)),
  1275→      passRate: _round2(
  1276→        (scores.where((s) => s >= passMark).length / scores.length) * 100,
  1277→      ),
  1278→      distinctionRate: _round2(
  1279→        (scores.where((s) => s >= distinctionThreshold).length / scores.length) *
  1280→            100,
  1281→      ),
  1282→      gradeDistribution: gradeDistribution,
  1283→      scoreDistribution: computeScoreDistribution(scores),
  1284→      topicPerformance: _computeTopicPerformanceFromResults(results),
  1285→      improvementRate: 0,
  1286→    );
  1287→  }

  1291→  /// Computes the improvement rate between current and previous results.
  1292→  ///
  1293→  /// Returns the average delta in percentage points (positive = improvement).
  1294→  /// Matches students by ID. Students not in both lists are excluded.
  1295→  static double _computeImprovementRate(
  1296→    List<StudentOverallResultEntity> current,
  1297→    List<StudentOverallResultEntity>? previous,
  1298→  ) {
  1299→    if (previous == null || previous.isEmpty) return 0;
  1300→
  1301→    final previousMap = <String, double>{};
  1302→    for (final r in previous) {
  1303→      previousMap[r.studentId] = r.overallPercentage;
  1304→    }
  1305→
  1306→    double totalDelta = 0;
  1307→    int matchedCount = 0;
  1308→    for (final r in current) {
  1309→      final prevScore = previousMap[r.studentId];
  1310→      if (prevScore != null) {
  1311→        totalDelta += r.overallPercentage - prevScore;
  1312→        matchedCount++;
  1313→      }
  1314→    }
  1315→
  1316→    return matchedCount > 0 ? totalDelta / matchedCount : 0;
  1317→  }

  1384→  /// Returns a human-readable interpretation of a correlation coefficient.
  1385→  static String _interpretCorrelation(double r) {
  1386→    final absR = r.abs();
  1387→    final direction = r >= 0 ? 'positive' : 'negative';
  1388→
  1389→    String strength;
  1390→    if (absR >= 0.8) {
  1391→      strength = 'very strong';
  1392→    } else if (absR >= 0.6) {
  1393→      strength = 'strong';
  1394→    } else if (absR >= 0.4) {
  1395→      strength = 'moderate';
  1396→    } else if (absR >= 0.2) {
  1397→      strength = 'weak';
  1398→    } else {
  1399→      strength = 'very weak';
  1400→    }
  1401→
  1402→    return 'There is a $strength $direction correlation between attendance '
  1403→        'and performance (r = ${r.toStringAsFixed(3)}).';
  1404→  }
