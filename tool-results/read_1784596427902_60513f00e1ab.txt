    27→    27→  const StudentPortalDashboardPage({super.key});
    28→    28→

    30→    30→  ConsumerState<StudentPortalDashboardPage> createState() =>
    31→    31→      _StudentPortalDashboardPageState();
    32→    32→}
    33→    33→
    34→    34→class _StudentPortalDashboardPageState
    35→    35→    extends ConsumerState<StudentPortalDashboardPage> {
    36→    36→  @override
    37→    37→  void initState() {
    38→    38→    super.initState();
    39→    39→    WidgetsBinding.instance.addPostFrameCallback((_) {
    40→    40→      ref.read(studentDashboardProvider.notifier).loadDashboard();
    41→    41→    });
    42→    42→  }
    43→    43→
    44→    44→  @override
    45→    45→  Widget build(BuildContext context) {
    46→    46→    final dashboardState = ref.watch(studentDashboardProvider);
    47→    47→
    48→    48→    if (dashboardState.isLoading && dashboardState.stats.upcomingExams == 0) {
    49→    49→      return const _DashboardLoadingSkeleton();
    50→    50→    }
    51→    51→
    52→    52→    if (dashboardState.error != null &&
    53→    53→        dashboardState.stats.upcomingExams == 0) {
    54→    54→      return AppErrorState(
    55→    55→        icon: Icons.error_outline_rounded,
    56→    56→        title: 'Failed to Load Dashboard',
    57→    57→        message: dashboardState.error,
    58→    58→        onRetry: () =>
    59→    59→            ref.read(studentDashboardProvider.notifier).refresh(),
    60→    60→      );
    61→    61→    }
    62→    62→
    63→    63→    final stats = dashboardState.stats;
    64→    64→
    65→    65→    return RefreshIndicator(
    66→    66→      onRefresh: () =>
    67→    67→          ref.read(studentDashboardProvider.notifier).refresh(),
    68→    68→      child: SingleChildScrollView(
    69→    69→        physics: const AlwaysScrollableScrollPhysics(),
    70→    70→        padding: Spacings.paddingScreen,
    71→    71→        child: context.isDesktop
    72→    72→            ? _buildDesktopLayout(context, ref, dashboardState, stats)
    73→    73→            : _buildMobileLayout(context, ref, dashboardState, stats),
    74→    74→      ),
    75→    75→    );
    76→    76→  }

   696→   696→  String _formatDate(DateTime? date) {
   697→   697→    if (date == null) return 'N/A';
   698→   698→    final now = DateTime.now();
   699→   699→    final diff = date.difference(now).inDays;
   700→   700→    if (diff == 0) return 'Today';
   701→   701→    if (diff == 1) return 'Tomorrow';
   702→   702→    if (diff > 0 && diff <= 7) return 'In $diff days';
   703→   703→    return '${date.day}/${date.month}/${date.year}';
   704→   704→  }
