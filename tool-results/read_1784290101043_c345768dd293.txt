    86→/// Riverpod provider for the application's [GoRouter].
    87→///
    88→/// Watches [authStateProvider] so that the router automatically
    89→/// re-evaluates its [redirect] callback whenever the authentication
    90→/// state changes (sign-in, sign-out, token refresh).
    91→///
    92→/// Usage in `main.dart`:
    93→/// ```dart
    94→/// final router = ref.read(appRouterProvider);
    95→/// MaterialApp.router(routerConfig: router);
    96→/// ```
    97→final appRouterProvider = Provider<GoRouter>((ref) {
    98→  // Watch auth state so the router re-evaluates redirects on change.
    99→  final authState = ref.watch(authStateProvider);
   100→
   101→  // Also watch onboarding state and role so those redirects stay live.
   102→  ref.watch(onboardingCompleteProvider);
   103→  ref.watch(userRoleProvider);
   104→
   105→  return GoRouter(
   106→    navigatorKey: _rootNavigatorKey,
   107→    initialLocation: RouteNames.splash,
   108→    debugLogDiagnostics: true,
   109→
   110→    // ─── Global Redirect ──────────────────────────────────────────
   111→    redirect: (context, state) {
   112→      final currentPath = state.matchedLocation;
   113→      final isAuthenticated = authState.when(
   114→        data: (authData) => authData.session != null,
   115→        loading: () {
   116→          // While loading, only allow splash. Everything else redirects
   117→          // to splash so we don't flash the login page briefly.
   118→          if (currentPath != RouteNames.splash) {
   119→            return RouteNames.splash;
   120→          }
   121→          return null;
   122→        },
   123→        error: (_, __) => false,
   124→      );
   125→
   126→      // ── Auth Guard ────────────────────────────────────────────
   127→      final authRedirect = AuthGuard.evaluate(
   128→        isAuthenticated: isAuthenticated,
   129→        currentPath: currentPath,
   130→      );
   131→      if (authRedirect != null) return authRedirect;
   132→
   133→      // ── Onboarding Guard ─────────────────────────────────────
   134→      final onboardingAsync = ref.read(onboardingCompleteProvider);
   135→      final isOnboardingComplete = onboardingAsync.when(
   136→        data: (complete) => complete,
   137→        loading: () => true, // Assume complete while loading to avoid flicker.
   138→        error: (_, __) => true,
   139→      );
   140→
   141→      final onboardingRedirect = OnboardingGuard.evaluate(
   142→        isAuthenticated: isAuthenticated,
   143→        isOnboardingComplete: isOnboardingComplete,
   144→        currentPath: currentPath,
   145→      );
   146→      if (onboardingRedirect != null) return onboardingRedirect;
   147→
   148→      // ── Role-Based Guard ─────────────────────────────────────
   149→      final roleAsync = ref.read(userRoleProvider);
   150→      final userRole = roleAsync.when(
   151→        data: UserRole.fromString,
   152→        loading: () => null,
   153→        error: (_, __) => null,
   154→      );
   155→
   156→      final roleRedirect = RoleBasedGuard.evaluate(
   157→        userRole: userRole,
   158→        currentPath: currentPath,
   159→      );
   160→      if (roleRedirect != null) return roleRedirect;
   161→
   162→      // No redirect needed — proceed as requested.
   163→      return null;
   164→    },
   165→

   167→    routes: [
   168→      // ── Splash ────────────────────────────────────────────────

   609→/// Shown when the user navigates to a route that does not exist in
   610→/// the route tree.
   611→///
   612→/// Provides a friendly message and a button to navigate back to the
   613→/// dashboard (or login if unauthenticated).
   614→class NotFoundPage extends ConsumerWidget {
   615→  const NotFoundPage({
   616→    required this.error,
   617→    this.path,
   618→    super.key,
   619→  });
   620→
   621→  /// The [GoException] from GoRouter describing the error.
   622→  final GoException? error;
   623→
   624→  /// The unmatched path the user attempted to visit.
   625→  final String? path;
   626→
   627→  @override
   628→  Widget build(BuildContext context, WidgetRef ref) {
   629→    final theme = Theme.of(context);
   630→    final isAuthenticated = ref.watch(isAuthenticatedProvider);
   631→
   632→    return Scaffold(
   633→      body: Center(
   634→        child: Padding(
   635→          padding: const EdgeInsets.all(32),
   636→          child: Column(
   637→            mainAxisSize: MainAxisSize.min,
   638→            children: [
   639→              Icon(
   640→                Icons.search_off_rounded,
   641→                size: 80,
   642→                color: theme.colorScheme.error,
   643→              ),
   644→              const SizedBox(height: 24),
   645→              Text(
   646→                'Page Not Found',
   647→                style: theme.textTheme.headlineMedium?.copyWith(
   648→                  fontWeight: FontWeight.bold,
   649→                ),
   650→                textAlign: TextAlign.center,
   651→              ),
   652→              const SizedBox(height: 12),
   653→              if (path != null)
   654→                Text(
   655→                  'The page "$path" does not exist.',
   656→                  style: theme.textTheme.bodyLarge?.copyWith(
   657→                    color: theme.colorScheme.onSurfaceVariant,
   658→                  ),
   659→                  textAlign: TextAlign.center,
   660→                ),
   661→              if (error != null) ...[
   662→                const SizedBox(height: 8),
   663→                Text(
   664→                  error!.message,
   665→                  style: theme.textTheme.bodySmall?.copyWith(
   666→                    color: theme.colorScheme.outline,
   667→                  ),
   668→                  textAlign: TextAlign.center,
   669→                ),
   670→              ],
   671→              const SizedBox(height: 32),
   672→              FilledButton.icon(
   673→                onPressed: () => context.go(
   674→                  isAuthenticated
   675→                      ? RouteNames.dashboard
   676→                      : RouteNames.login,
   677→                ),
   678→                icon: const Icon(Icons.home),
   679→                label: Text(
   680→                  isAuthenticated ? 'Go to Dashboard' : 'Go to Login',
   681→                ),
   682→              ),
   683→            ],
   684→          ),
   685→        ),
   686→      ),
   687→    );
   688→  }
