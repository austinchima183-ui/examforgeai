    32→/// Full-featured question editor supporting create and edit modes.
    33→///
    34→/// The form is divided into sections:
    35→/// 1. Question Type selection (grid of selectable type cards)
    36→/// 2. Question Content (multiline text field)
    37→/// 3. Answer Configuration (dynamic based on selected type)
    38→/// 4. Classification (subject, topic, subtopic, class, category)
    39→/// 5. Settings (difficulty, exam type, marks, negative marks, time)
    40→/// 6. Explanation & Notes
    41→/// 7. Attachments
    42→/// 8. Tags
    43→///
    44→/// Supports:
    45→/// - Preview mode toggle
    46→/// - Save Draft / Publish actions
    47→/// - Form validation
    48→/// - Unsaved changes dialog on back
    49→/// - Loading state while saving
    50→class QuestionEditorPage extends ConsumerStatefulWidget {
    51→  const QuestionEditorPage({
    52→    super.key,
    53→    this.questionId,
    54→  });
    55→
    56→  /// If provided, the editor loads this question for editing.
    57→  final String? questionId;
    58→
    59→  @override
    60→  ConsumerState<QuestionEditorPage> createState() =>
    61→      _QuestionEditorPageState();
    62→}
    63→
    64→class _QuestionEditorPageState extends ConsumerState<QuestionEditorPage> {
    65→  final _formKey = GlobalKey<FormState>();
    66→  final _contentController = TextEditingController();
    67→  final _explanationController = TextEditingController();
    68→  final _teacherNotesController = TextEditingController();
    69→  final _referenceMaterialsController = TextEditingController();
    70→  final _marksController = TextEditingController();
    71→  final _negativeMarksController = TextEditingController();
    72→  final _timeAllowedController = TextEditingController();
    73→  final _tagSearchController = TextEditingController();
    74→
    75→  bool _hasInitialized = false;
    76→  bool _hasUnsavedChanges = false;
    77→
    78→  @override
    79→  void initState() {
    80→    super.initState();
    81→    WidgetsBinding.instance.addPostFrameCallback((_) {
    82→      _initializeEditor();
    83→    });
    84→  }
    85→
    86→  @override
    87→  void dispose() {
    88→    _contentController.dispose();
    89→    _explanationController.dispose();
    90→    _teacherNotesController.dispose();
    91→    _referenceMaterialsController.dispose();
    92→    _marksController.dispose();
    93→    _negativeMarksController.dispose();
    94→    _timeAllowedController.dispose();
    95→    _tagSearchController.dispose();
    96→    super.dispose();
    97→  }
    98→
    99→  Future<void> _initializeEditor() async {
   100→    if (widget.questionId != null) {
   101→      await ref
   102→          .read(questionEditorProvider.notifier)
   103→          .loadQuestionForEdit(widget.questionId!);
   104→    }
   105→    _syncControllersFromState();
   106→    _hasInitialized = true;
   107→  }
   108→
   109→  void _syncControllersFromState() {
   110→    final state = ref.read(questionEditorProvider);
   111→    _contentController.text = state.content;
   112→    _explanationController.text = state.explanation;
   113→    _teacherNotesController.text = state.teacherNotes;
   114→    _referenceMaterialsController.text = state.referenceMaterials;
   115→    _marksController.text = state.marks.toString();
   116→    _negativeMarksController.text = state.negativeMarks.toString();
   117→    if (state.timeAllowedSeconds != null) {
   118→      _timeAllowedController.text = (state.timeAllowedSeconds! ~/ 60).toString();
   119→    }
   120→  }
   121→
   122→  void _markUnsaved() {
   123→    if (!_hasUnsavedChanges) {
   124→      setState(() => _hasUnsavedChanges = true);
   125→    }
   126→  }

   128→  // ─── Build ──────────────────────────────────────────────────────────
   129→
   130→  @override
   131→  Widget build(BuildContext context) {
   132→    final editorState = ref.watch(questionEditorProvider);
   133→    final cs = context.colorScheme;
   134→    final tt = context.textTheme;
   135→
   136→    return PopScope(
   137→      canPop: !_hasUnsavedChanges,
   138→      onPopInvokedWithResult: (didPop, _) async {
   139→        if (didPop) return;
   140→        if (_hasUnsavedChanges) {
   141→          final shouldLeave = await _showUnsavedChangesDialog();
   142→          if (shouldLeave == true && mounted) {
   143→            context.pop();
   144→          }
   145→        }
   146→      },
   147→      child: Scaffold(
   148→        appBar: AppAppBar(
   149→          title: editorState.isEditMode ? 'Edit Question' : 'Create Question',
   150→          actions: [
   151→            // Preview toggle
   152→            AppIconButton(
   153→              icon: editorState.isPreviewMode
   154→                  ? Icons.edit_rounded
   155→                  : Icons.visibility_rounded,
   156→              onPressed: () {
   157→                ref.read(questionEditorProvider.notifier).togglePreviewMode();
   158→              },
   159→              tooltip: editorState.isPreviewMode ? 'Edit' : 'Preview',
   160→            ),
   161→            // Save Draft
   162→            if (!editorState.isPreviewMode)
   163→              AppButton(
   164→                label: 'Save Draft',
   165→                onPressed: editorState.isSaving ? null : _saveDraft,
   166→                variant: AppButtonVariant.outlined,
   167→                size: AppButtonSize.small,
   168→                isLoading: editorState.isSaving,
   169→              ),
   170→            const SizedBox(width: Spacings.sm),
   171→            // Publish
   172→            AppButton(
   173→              label: 'Publish',
   174→              onPressed: editorState.isSaving ? null : _publish,
   175→              variant: AppButtonVariant.elevated,
   176→              size: AppButtonSize.small,
   177→              isLoading: editorState.isSaving,
   178→            ),
   179→            const SizedBox(width: Spacings.sm),
   180→          ],
   181→        ),
   182→        body: editorState.isLoading
   183→            ? const Center(
   184→                child: AppLoadingSpinner(size: AppLoadingSpinnerSize.large),

   336→  // ─── 1. Question Type Grid ──────────────────────────────────────────
   337→
   338→  Widget _buildQuestionTypeGrid(QuestionEditorState editorState) {
   339→    final cs = context.colorScheme;
   340→    final isDesktop = context.isDesktop;
   341→    final crossAxisCount = isDesktop ? 4 : (context.isTablet ? 3 : 2);
   342→
   343→    return GridView.count(
   344→      shrinkWrap: true,
   345→      physics: const NeverScrollableScrollPhysics(),
   346→      crossAxisCount: crossAxisCount,
   347→      childAspectRatio: 2.8,
   348→      crossAxisSpacing: Spacings.sm,
   349→      mainAxisSpacing: Spacings.sm,
   350→      children: QuestionType.values.map((type) {
   351→        final isSelected = editorState.selectedQuestionType == type;
   352→        return InkWell(
   353→          onTap: () {
   354→            ref.read(questionEditorProvider.notifier).setQuestionType(type);
   355→            _markUnsaved();
   356→          },
   357→          borderRadius: BorderRadius.circular(Spacings.mdRadius),
   358→          child: Container(
   359→            padding: const EdgeInsets.symmetric(
   360→              horizontal: Spacings.md,
   361→              vertical: Spacings.sm,
   362→            ),
   363→            decoration: BoxDecoration(
   364→              color: isSelected
   365→                  ? cs.primary.withOpacity(context.isDarkMode ? 0.20 : 0.08)
   366→                  : cs.surfaceContainerLow,
   367→              borderRadius: BorderRadius.circular(Spacings.mdRadius),
   368→              border: Border.all(
   369→                color: isSelected ? cs.primary : cs.outlineVariant.withOpacity(0.5),
   370→                width: isSelected ? 2.0 : 1.0,
   371→              ),
   372→            ),
   373→            child: Row(
   374→              children: [
   375→                QuestionTypeBadge(
   376→                  type: type,
   377→                  variant: QuestionTypeBadgeVariant.iconOnly,
   378→                  size: QuestionTypeBadgeSize.small,
   379→                ),
   380→                const SizedBox(width: Spacings.sm),
   381→                Flexible(
   382→                  child: Text(
   383→                    type.label,
   384→                    style: context.textTheme.bodySmall?.copyWith(
   385→                      color: isSelected ? cs.primary : cs.onSurface,
   386→                      fontWeight: isSelected
   387→                          ? AppTypography.wSemiBold
   388→                          : null,
   389→                    ),
   390→                    maxLines: 1,
   391→                    overflow: TextOverflow.ellipsis,
   392→                  ),
   393→                ),
   394→              ],
   395→            ),
   396→          ),
   397→        );
   398→      }).toList(),
   399→    );
   400→  }

   402→  // ─── 3. Answer Configuration ────────────────────────────────────────
   403→
   404→  Widget _buildAnswerConfiguration(QuestionEditorState editorState) {
   405→    final type = editorState.selectedQuestionType;
   406→    final cs = context.colorScheme;
   407→
   408→    switch (type) {
   409→      case QuestionType.multipleChoice:
   410→      case QuestionType.multipleResponse:
   411→      case QuestionType.imageBased:
   412→      case QuestionType.audioBased:
   413→      case QuestionType.videoBased:
   414→        return AnswerOptionsEditor(
   415→          options: editorState.answerOptions,
   416→          isMultipleChoice: type == QuestionType.multipleChoice,
   417→          onAddOption: () {
   418→            ref.read(questionEditorProvider.notifier).addAnswerOption();
   419→            _markUnsaved();
   420→          },
   421→          onRemoveOption: (index) {
   422→            ref.read(questionEditorProvider.notifier).removeAnswerOption(index);
   423→            _markUnsaved();
   424→          },
   425→          onUpdateOption: (index, option) {
   426→            ref.read(questionEditorProvider.notifier).updateAnswerOption(index, option);
   427→            _markUnsaved();
   428→          },
   429→          onSetCorrectAnswer: (index) {
   430→            ref.read(questionEditorProvider.notifier).setCorrectAnswer(index);
   431→            _markUnsaved();
   432→          },
   433→          onSetCorrectAnswers: (indices) {
   434→            ref.read(questionEditorProvider.notifier).setCorrectAnswers(indices);
   435→            _markUnsaved();
   436→          },
   437→        );
   438→
   439→      case QuestionType.trueFalse:
   440→        return _buildTrueFalseEditor(editorState);
   441→
   442→      case QuestionType.matching:
   443→        return MatchingPairsEditor(
   444→          pairs: editorState.matchingPairs,
   445→          onAddPair: () {
   446→            ref.read(questionEditorProvider.notifier).addMatchingPair();
   447→            _markUnsaved();
   448→          },
   449→          onRemovePair: (index) {
   450→            ref.read(questionEditorProvider.notifier).removeMatchingPair(index);
   451→            _markUnsaved();
   452→          },
   453→          onUpdatePair: (index, pair) {
   454→            ref.read(questionEditorProvider.notifier).updateMatchingPair(index, pair);
   455→            _markUnsaved();
   456→          },
   457→        );
   458→
   459→      case QuestionType.ordering:
   460→        return OrderingItemsEditor(
   461→          items: editorState.orderingItems,
   462→          onAddItem: () {
   463→            ref.read(questionEditorProvider.notifier).addOrderingItem();
   464→            _markUnsaved();
   465→          },
   466→          onRemoveItem: (index) {
   467→            ref.read(questionEditorProvider.notifier).removeOrderingItem(index);
   468→            _markUnsaved();
   469→          },
   470→          onUpdateItem: (index, item) {
   471→            ref.read(questionEditorProvider.notifier).updateOrderingItem(index, item);
   472→            _markUnsaved();
   473→          },
   474→          onReorder: (oldIndex, newIndex) {
   475→            ref.read(questionEditorProvider.notifier).reorderOrderingItems(oldIndex, newIndex);
   476→            _markUnsaved();
   477→          },
   478→        );
   479→
   480→      case QuestionType.fillInBlank:
   481→        return FillInBlankEditor(
   482→          answers: editorState.fillInBlankAnswers,
   483→          onAddBlank: () {
   484→            ref.read(questionEditorProvider.notifier).addFillInBlankAnswer();
   485→            _markUnsaved();
   486→          },
   487→          onRemoveBlank: (index) {
   488→            ref.read(questionEditorProvider.notifier).removeFillInBlankAnswer(index);
   489→            _markUnsaved();
   490→          },
   491→          onUpdateBlank: (index, answer) {
   492→            ref.read(questionEditorProvider.notifier).updateFillInBlankAnswer(index, answer);
   493→            _markUnsaved();
   494→          },
   495→        );
   496→
   497→      case QuestionType.shortAnswer:
   498→      case QuestionType.essay:
   499→      case QuestionType.numerical:
   500→      case QuestionType.practical:
   501→      case QuestionType.caseStudy:
   502→        return AppTextField(
   503→          label: 'Model Answer',
   504→          hint: 'Enter the model answer or expected response…',
   505→          controller: _explanationController, // Reuse for model answer
   506→          maxLines: 8,
   507→          minLines: 3,
   508→          onChanged: (v) {
   509→            _markUnsaved();
   510→          },
   511→        );
   512→    }
   513→  }

   515→  // ─── True/False Editor ──────────────────────────────────────────────
   516→
   517→  Widget _buildTrueFalseEditor(QuestionEditorState editorState) {
   518→    final cs = context.colorScheme;
   519→    final tt = context.textTheme;
   520→
   521→    // Check if "True" is the correct answer
   522→    final isTrueCorrect = editorState.answerOptions.any(
   523→      (o) => o.isCorrect && o.content.toLowerCase() == 'true',
   524→    );
   525→
   526→    return Column(
   527→      crossAxisAlignment: CrossAxisAlignment.start,
   528→      children: [
   529→        Text(
   530→          'Select the correct answer',
   531→          style: tt.bodySmall?.copyWith(color: cs.onSurfaceVariant),
   532→        ),
   533→        const SizedBox(height: Spacings.md),
   534→        SegmentedButton<bool>(
   535→          segments: const [
   536→            ButtonSegment(value: true, label: Text('True'), icon: Icon(Icons.check_rounded)),
   537→            ButtonSegment(value: false, label: Text('False'), icon: Icon(Icons.close_rounded)),
   538→          ],
   539→          selected: {isTrueCorrect},
   540→          onSelectionChanged: (selection) {
   541→            final isTrue = selection.first;
   542→            // Update the state with True/False options
   543→            ref.read(questionEditorProvider.notifier).setCorrectAnswer(isTrue ? 0 : 1);
   544→            _markUnsaved();
   545→          },
   546→        ),
   547→      ],
   548→    );
   549→  }

   952→  Future<void> _saveDraft() async {
   953→    if (!_formKey.currentState!.validate()) return;
   954→
   955→    final notifier = ref.read(questionEditorProvider.notifier);
   956→    await notifier.saveAsDraft();
   957→
   958→    _handleSaveResult();

   961→  // ─── Publish ────────────────────────────────────────────────────────
   962→
   963→  Future<void> _publish() async {
   964→    if (!_formKey.currentState!.validate()) return;
   965→
   966→    final notifier = ref.read(questionEditorProvider.notifier);
   967→    await notifier.saveAndPublish();
   968→
   969→    _handleSaveResult();
   970→  }
   971→
   972→  void _handleSaveResult() {
   973→    final state = ref.read(questionEditorProvider);
   974→    if (state.error != null && mounted) {
   975→      AppDialog.showError(
   976→        context: context,
   977→        title: 'Save Failed',
   978→        message: state.error!,
   979→      );
   980→    } else if (state.successMessage != null && mounted) {
   981→      setState(() => _hasUnsavedChanges = false);
   982→      ScaffoldMessenger.of(context).showSnackBar(
   983→        SnackBar(
   984→          content: Text(state.successMessage!),
   985→          behavior: SnackBarBehavior.floating,
   986→        ),
   987→      );
   988→      if (mounted) context.pop();
   989→    }
   990→  }

  1072→  String _formatFileSize(int bytes) {
  1073→    if (bytes < 1024) return '$bytes B';
  1074→    if (bytes < 1024 * 1024) return '${(bytes / 1024).toStringAsFixed(1)} KB';
  1075→    return '${(bytes / (1024 * 1024)).toStringAsFixed(1)} MB';
