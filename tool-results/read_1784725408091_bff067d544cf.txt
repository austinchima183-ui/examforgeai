    32→    32→/// Full-featured question editor supporting create and edit modes.
    33→    33→///
    34→    34→/// The form is divided into sections:
    35→    35→/// 1. Question Type selection (grid of selectable type cards)
    36→    36→/// 2. Question Content (multiline text field)
    37→    37→/// 3. Answer Configuration (dynamic based on selected type)
    38→    38→/// 4. Classification (subject, topic, subtopic, class, category)
    39→    39→/// 5. Settings (difficulty, exam type, marks, negative marks, time)
    40→    40→/// 6. Explanation & Notes
    41→    41→/// 7. Attachments
    42→    42→/// 8. Tags
    43→    43→///
    44→    44→/// Supports:
    45→    45→/// - Preview mode toggle
    46→    46→/// - Save Draft / Publish actions
    47→    47→/// - Form validation
    48→    48→/// - Unsaved changes dialog on back
    49→    49→/// - Loading state while saving
    50→    50→class QuestionEditorPage extends ConsumerStatefulWidget {
    51→    51→  const QuestionEditorPage({
    52→    52→    super.key,
    53→    53→    this.questionId,
    54→    54→  });

    59→    59→  @override

    93→    93→    _negativeMarksController.dispose();

    98→    98→
    99→    99→  Future<void> _initializeEditor() async {
   100→   100→    if (widget.questionId != null) {
   101→   101→      await ref
   102→   102→          .read(questionEditorProvider.notifier)
   103→   103→          .loadQuestionForEdit(widget.questionId!);
   104→   104→    }
   105→   105→    _syncControllersFromState();
   106→   106→    _hasInitialized = true;
   107→   107→  }
   108→   108→
   109→   109→  void _syncControllersFromState() {
   110→   110→    final state = ref.read(questionEditorProvider);
   111→   111→    _contentController.text = state.content;
   112→   112→    _explanationController.text = state.explanation;
   113→   113→    _teacherNotesController.text = state.teacherNotes;
   114→   114→    _referenceMaterialsController.text = state.referenceMaterials;
   115→   115→    _marksController.text = state.marks.toString();
   116→   116→    _negativeMarksController.text = state.negativeMarks.toString();
   117→   117→    if (state.timeAllowedSeconds != null) {
   118→   118→      _timeAllowedController.text = (state.timeAllowedSeconds! ~/ 60).toString();
   119→   119→    }
   120→   120→  }
   121→   121→
   122→   122→  void _markUnsaved() {
   123→   123→    if (!_hasUnsavedChanges) {
   124→   124→      setState(() => _hasUnsavedChanges = true);
   125→   125→    }
   126→   126→  }

   147→   147→      child: Scaffold(
   148→   148→        appBar: AppAppBar(

   150→   150→          actions: [
   151→   151→            // Preview toggle
   152→   152→            AppIconButton(
   153→   153→              icon: editorState.isPreviewMode
   154→   154→                  ? Icons.edit_rounded

   156→   156→              onPressed: () {
   157→   157→                ref.read(questionEditorProvider.notifier).togglePreviewMode();
   158→   158→              },
   159→   159→              tooltip: editorState.isPreviewMode ? 'Edit' : 'Preview',
   160→   160→            ),
   161→   161→            // Save Draft
   162→   162→            if (!editorState.isPreviewMode)
   163→   163→              AppButton(
   164→   164→                label: 'Save Draft',
   165→   165→                onPressed: editorState.isSaving ? null : _saveDraft,
   166→   166→                variant: AppButtonVariant.outlined,
   167→   167→                size: AppButtonSize.small,
   168→   168→                isLoading: editorState.isSaving,
   169→   169→              ),
   170→   170→            const SizedBox(width: Spacings.sm),
   171→   171→            // Publish

   173→   173→              label: 'Publish',
   174→   174→              onPressed: editorState.isSaving ? null : _publish,

   176→   176→              size: AppButtonSize.small,
   177→   177→              isLoading: editorState.isSaving,
   178→   178→            ),
   179→   179→            const SizedBox(width: Spacings.sm),

   182→   182→        body: editorState.isLoading

   338→   338→  Widget _buildQuestionTypeGrid(QuestionEditorState editorState) {
   339→   339→    final cs = context.colorScheme;
   340→   340→    final isDesktop = context.isDesktop;

   353→   353→          onTap: () {

   402→   402→  // ─── 3. Answer Configuration ────────────────────────────────────────
   403→   403→
   404→   404→  Widget _buildAnswerConfiguration(QuestionEditorState editorState) {
   405→   405→    final type = editorState.selectedQuestionType;
   406→   406→    final cs = context.colorScheme;
   407→   407→
   408→   408→    switch (type) {
   409→   409→      case QuestionType.multipleChoice:
   410→   410→      case QuestionType.multipleResponse:
   411→   411→      case QuestionType.imageBased:
   412→   412→      case QuestionType.audioBased:
   413→   413→      case QuestionType.videoBased:
   414→   414→        return AnswerOptionsEditor(
   415→   415→          options: editorState.answerOptions,
   416→   416→          isMultipleChoice: type == QuestionType.multipleChoice,
   417→   417→          onAddOption: () {
   418→   418→            ref.read(questionEditorProvider.notifier).addAnswerOption();
   419→   419→            _markUnsaved();
   420→   420→          },
   421→   421→          onRemoveOption: (index) {
   422→   422→            ref.read(questionEditorProvider.notifier).removeAnswerOption(index);
   423→   423→            _markUnsaved();
   424→   424→          },
   425→   425→          onUpdateOption: (index, option) {
   426→   426→            ref.read(questionEditorProvider.notifier).updateAnswerOption(index, option);
   427→   427→            _markUnsaved();
   428→   428→          },
   429→   429→          onSetCorrectAnswer: (index) {
   430→   430→            ref.read(questionEditorProvider.notifier).setCorrectAnswer(index);
   431→   431→            _markUnsaved();
   432→   432→          },
   433→   433→          onSetCorrectAnswers: (indices) {
   434→   434→            ref.read(questionEditorProvider.notifier).setCorrectAnswers(indices);
   435→   435→            _markUnsaved();
   436→   436→          },
   437→   437→        );
   438→   438→
   439→   439→      case QuestionType.trueFalse:
   440→   440→        return _buildTrueFalseEditor(editorState);
   441→   441→
   442→   442→      case QuestionType.matching:
   443→   443→        return MatchingPairsEditor(
   444→   444→          pairs: editorState.matchingPairs,
   445→   445→          onAddPair: () {
   446→   446→            ref.read(questionEditorProvider.notifier).addMatchingPair();
   447→   447→            _markUnsaved();
   448→   448→          },
   449→   449→          onRemovePair: (index) {
   450→   450→            ref.read(questionEditorProvider.notifier).removeMatchingPair(index);
   451→   451→            _markUnsaved();
   452→   452→          },
   453→   453→          onUpdatePair: (index, pair) {
   454→   454→            ref.read(questionEditorProvider.notifier).updateMatchingPair(index, pair);
   455→   455→            _markUnsaved();
   456→   456→          },
   457→   457→        );
   458→   458→
   459→   459→      case QuestionType.ordering:
   460→   460→        return OrderingItemsEditor(
   461→   461→          items: editorState.orderingItems,
   462→   462→          onAddItem: () {
   463→   463→            ref.read(questionEditorProvider.notifier).addOrderingItem();
   464→   464→            _markUnsaved();
   465→   465→          },
   466→   466→          onRemoveItem: (index) {
   467→   467→            ref.read(questionEditorProvider.notifier).removeOrderingItem(index);
   468→   468→            _markUnsaved();
   469→   469→          },
   470→   470→          onUpdateItem: (index, item) {
   471→   471→            ref.read(questionEditorProvider.notifier).updateOrderingItem(index, item);
   472→   472→            _markUnsaved();
   473→   473→          },
   474→   474→          onReorder: (oldIndex, newIndex) {
   475→   475→            ref.read(questionEditorProvider.notifier).reorderOrderingItems(oldIndex, newIndex);
   476→   476→            _markUnsaved();
   477→   477→          },
   478→   478→        );
   479→   479→
   480→   480→      case QuestionType.fillInBlank:
   481→   481→        return FillInBlankEditor(
   482→   482→          answers: editorState.fillInBlankAnswers,
   483→   483→          onAddBlank: () {
   484→   484→            ref.read(questionEditorProvider.notifier).addFillInBlankAnswer();
   485→   485→            _markUnsaved();
   486→   486→          },
   487→   487→          onRemoveBlank: (index) {
   488→   488→            ref.read(questionEditorProvider.notifier).removeFillInBlankAnswer(index);
   489→   489→            _markUnsaved();
   490→   490→          },
   491→   491→          onUpdateBlank: (index, answer) {
   492→   492→            ref.read(questionEditorProvider.notifier).updateFillInBlankAnswer(index, answer);
   493→   493→            _markUnsaved();
   494→   494→          },
   495→   495→        );
   496→   496→
   497→   497→      case QuestionType.shortAnswer:
   498→   498→      case QuestionType.essay:
   499→   499→      case QuestionType.numerical:
   500→   500→      case QuestionType.practical:
   501→   501→      case QuestionType.caseStudy:
   502→   502→        return AppTextField(
   503→   503→          label: 'Model Answer',
   504→   504→          hint: 'Enter the model answer or expected response…',
   505→   505→          controller: _explanationController, // Reuse for model answer
   506→   506→          maxLines: 8,
   507→   507→          minLines: 3,
   508→   508→          onChanged: (v) {
   509→   509→            _markUnsaved();
