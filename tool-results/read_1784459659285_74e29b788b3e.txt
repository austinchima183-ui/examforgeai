    19→/// Real-time infrastructure health monitoring page.
    20→///
    21→/// Displays service status cards, health indicators, maintenance windows,
    22→/// and provides actions for running health checks. Auto-refreshes every
    23→/// 30 seconds.
    24→///
    25→/// Follows the standard ExamForge AI page pattern:
    26→/// - [ConsumerStatefulWidget] + private state class
    27→/// - `initState` → `addPostFrameCallback` → `_loadData()` via `ref.read`
    28→/// - `ref.watch(provider)` in `build()` for reactive rebuilds
    29→/// - Loading → Error → Content pattern
    30→/// - Design tokens everywhere
    31→class InfrastructureMonitoringPage extends ConsumerStatefulWidget {
    32→  const InfrastructureMonitoringPage({super.key});
    33→
    34→  @override
    35→  ConsumerState<InfrastructureMonitoringPage> createState() =>
    36→      _InfrastructureMonitoringPageState();
    37→}

    45→  @override

   182→  // ─── Overall System Health ──────────────────────────────────────────────

   184→  Widget _buildOverallHealth(InfrastructureMonitoringState state, ColorScheme cs) {
   185→    final services = state.services;
   186→    if (services.isEmpty) return const SizedBox.shrink();
   187→
   188→    final healthyCount =
   189→        services.where((s) => s.healthStatus == HealthStatus.healthy).length;
   190→    final degradedCount =
   191→        services.where((s) => s.healthStatus == HealthStatus.degraded).length;
   192→    final unhealthyCount = services
   193→        .where((s) =>
   194→            s.healthStatus == HealthStatus.unhealthy ||
   195→            s.healthStatus == HealthStatus.down)
   196→        .length;
   197→    final maintenanceCount = services
   198→        .where((s) => s.healthStatus == HealthStatus.maintenance)
   199→        .length;
   200→
   201→    // Determine overall health
   202→    HealthStatus overallStatus;
   203→    if (unhealthyCount > 0) {
   204→      overallStatus = HealthStatus.unhealthy;
   205→    } else if (degradedCount > 0) {
   206→      overallStatus = HealthStatus.degraded;
   207→    } else if (maintenanceCount > 0) {
   208→      overallStatus = HealthStatus.maintenance;
   209→    } else {
   210→      overallStatus = HealthStatus.healthy;
   211→    }
   212→
   213→    return Card(
   214→      elevation: Spacings.elevationSm,
   215→      shape: RoundedRectangleBorder(borderRadius: Spacings.borderRadiusMd),
   216→      child: Padding(
   217→        padding: Spacings.paddingAll,
   218→        child: Column(
   219→          crossAxisAlignment: CrossAxisAlignment.start,
   220→          children: [
   221→            Row(
   222→              children: [
   223→                Text(
   224→                  'Overall System Health',
   225→                  style: AppTypography.wBold.copyWith(fontSize: 16),
   226→                ),
   227→                const Spacer(),
   228→                HealthIndicator(status: overallStatus),
   229→              ],
   230→            ),
   231→            const SizedBox(height: Spacings.md),
   232→            Row(
   233→              children: [
   234→                _buildHealthStatChip(
   235→                  'Healthy',
   236→                  healthyCount,
   237→                  AppColors.success,
   238→                ),
   239→                const SizedBox(width: Spacings.md),
   240→                _buildHealthStatChip(
   241→                  'Degraded',
   242→                  degradedCount,
   243→                  AppColors.warning,
   244→                ),
   245→                const SizedBox(width: Spacings.md),
   246→                _buildHealthStatChip(
   247→                  'Down',
   248→                  unhealthyCount,
   249→                  AppColors.error,
   250→                ),
   251→                const SizedBox(width: Spacings.md),
   252→                _buildHealthStatChip(
   253→                  'Maintenance',
   254→                  maintenanceCount,
   255→                  AppColors.info,
   256→                ),
   257→              ],
   258→            ),
   259→          ],
   260→        ),
   261→      ),
   262→    );
   263→  }

   589→  // ─── Date/Time Picker Helper ─────────────────────────────────────────────
   590→
   591→  Future<DateTime?> _pickDateTime(
   592→      BuildContext context, DateTime? initial) async {
   593→    final date = await showDatePicker(
   594→      context: context,
   595→      initialDate: initial ?? DateTime.now().add(const Duration(days: 1)),
   596→      firstDate: DateTime.now(),
   597→      lastDate: DateTime.now().add(const Duration(days: 365)),
   598→    );
   599→    if (date == null) return null;
   600→
   601→    final time = await showTimePicker(
   602→      context: context,
   603→      initialTime: TimeOfDay.fromDateTime(initial ?? DateTime.now()),
   604→    );
   605→    if (time == null) return null;
   606→
   607→    return DateTime(
   608→      date.year, date.month, date.day, time.hour, time.minute,
   609→    );
   610→  }
   611→
   612→  // ─── Formatting Helpers ─────────────────────────────────────────────────
   613→
   614→  static String _formatDateTime(DateTime dt) {
   615→    final month = _monthShort(dt.month);
   616→    return '$month ${dt.day}, ${dt.year} at ${dt.hour.toString().padLeft(2, '0')}:${dt.minute.toString().padLeft(2, '0')}';
   617→  }
   618→
   619→  static String _monthShort(int month) {
   620→    const months = [
   621→      '', 'Jan', 'Feb', 'Mar', 'Apr', 'May', 'Jun',
   622→      'Jul', 'Aug', 'Sep', 'Oct', 'Nov', 'Dec'
   623→    ];
   624→    return months[month];
   625→  }
   626→}

   632→class _ServiceStatusCard extends StatelessWidget {

   639→
   640→  final InfrastructureService service;
   641→  final bool isRunningHealthCheck;
   642→  final VoidCallback onRunHealthCheck;

   644→
   645→  Color _responseTimeColor(int? ms) {
   646→    if (ms == null) return cs.onSurface.withValues(alpha: 0.4);
   647→    if (ms < 500) return AppColors.success;
   648→    if (ms < 2000) return AppColors.warning;
   649→    return AppColors.error;
   650→  }
   651→
   652→  Color _errorRateColor(double rate) {

   658→  String _formatLastCheck(DateTime? dt) {
   659→    if (dt == null) return 'Never';
   660→    final diff = DateTime.now().difference(dt);
   661→    if (diff.inSeconds < 60) return '${diff.inSeconds}s ago';
   662→    if (diff.inMinutes < 60) return '${diff.inMinutes}m ago';
   663→    if (diff.inHours < 24) return '${diff.inHours}h ago';
   664→    return '${diff.inDays}d ago';
   665→  }

   866→        '${dt.hour.toString().padLeft(2, '0')}:${dt.minute.toString().padLeft(2, '0')}';
