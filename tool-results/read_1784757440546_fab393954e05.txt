    75→  Future<Result<List<PlatformSetting>>> call(BulkUpdateSettingsParams params) async {
    76→    if (params.settings.isEmpty) {
    77→      return const FailureResult(Failure.validation(fieldErrors: const {}, message: 'No settings provided'));

    79→    final readonlySettings = params.settings.where((s) => s.isReadonly).toList();
    80→    if (readonlySettings.isNotEmpty) {
    81→      return FailureResult(Failure.validation(
    82→        message: 'Cannot modify readonly settings: ${readonlySettings.map((s) => s.key).join(', ')}',
    83→        fieldErrors: const {},
    84→      ));
    85→    }
    86→    return _repository.bulkUpdateSettings(params.settings);
    87→  }
