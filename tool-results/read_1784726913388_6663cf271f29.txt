     8→
     9→/// Helper to read a field from JSON that may be in snake_case (Supabase)
    10→/// or camelCase (internal).
    11→T? _readField<T>(Map<String, dynamic> json, String snakeKey, String camelKey) {
    12→  return json[snakeKey] as T? ?? json[camelKey] as T?;
    13→}
    14→
    15→/// Helper to read a DateTime field from JSON (snake_case or camelCase).
    16→DateTime _readDateTime(Map<String, dynamic> json, String snakeKey, String camelKey) {
    17→  final raw = json[snakeKey] as String? ?? json[camelKey] as String;
    18→  return DateTime.parse(raw);
    19→}
    20→
    21→/// Helper to read a nullable DateTime field from JSON.
    22→DateTime? _readNullableDateTime(Map<String, dynamic> json, String snakeKey, String camelKey) {
    23→  final raw = json[snakeKey] as String? ?? json[camelKey] as String?;
    24→  if (raw == null) return null;
    25→  return DateTime.parse(raw);
    26→}
    27→
    28→/// Helper to read a List<Map<String, dynamic>> from JSON.
    29→List<Map<String, dynamic>> _readListOfMaps(dynamic value) {
    30→  if (value == null) return [];
    31→  return (value as List).map((e) => Map<String, dynamic>.from(e as Map)).toList();
    32→}
    33→
    34→/// Helper to read a nullable List<Map<String, dynamic>> from JSON.
    35→List<Map<String, dynamic>>? _readNullableListOfMaps(dynamic value) {
    36→  if (value == null) return null;
    37→  return (value as List).map((e) => Map<String, dynamic>.from(e as Map)).toList();
    38→}
    39→
    40→/// Helper to read a List<String> from JSON.
    41→List<String> _readListOfStrings(dynamic value) {
    42→  if (value == null) return [];
    43→  return (value as List).cast<String>();
    44→}
    45→
    46→/// Helper to read a nullable List<String> from JSON.
    47→List<String>? _readNullableListOfStrings(dynamic value) {
    48→  if (value == null) return null;
    49→  return (value as List).cast<String>();
    50→}
    51→
    52→/// Helper to read a nullable Map<String, dynamic> from JSON.
    53→Map<String, dynamic>? _readNullableMap(dynamic value) {
    54→  if (value == null) return null;
    55→  return Map<String, dynamic>.from(value as Map);
    56→}
    57→
    58→/// Helper to read a non-null Map<String, dynamic> from JSON.
    59→Map<String, dynamic> _readMap(dynamic value) {
    60→  if (value == null) return {};
    61→  return Map<String, dynamic>.from(value as Map);
    62→}
    63→
    64→/// Helper to read a List<BloomTaxonomy> from JSON (stored as list of strings).
    65→List<BloomTaxonomy>? _readNullableBloomList(dynamic value) {
    66→  if (value == null) return null;
    67→  return (value as List).map((e) {
    68→    final str = e as String;
    69→    return BloomTaxonomy.fromString(str) ?? BloomTaxonomy.remember;
    70→  }).toList();
    71→}
