    20→    required this.label,
    21→  });

    23→  /// The string representation stored in the backend.
    24→  final String value;
    25→
    26→  /// Human-readable label for display in the UI.
    27→  final String label;
    28→
    29→  /// Parses a raw [value] string into a [MessageDirection].

    35→          (direction) => direction?.value == value,
    36→          orElse: () => null,
    37→        );
