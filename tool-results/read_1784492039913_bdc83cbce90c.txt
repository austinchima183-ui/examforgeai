   219→/// Provides the Supabase client singleton.
   220→///
   221→/// Depends on [SupabaseConfig.initialize] having been called in `main()`.
   222→final supabaseClientProvider = Provider<sb.SupabaseClient>((ref) {
   223→  try {
   224→    return SupabaseConfig.client;
   225→  } catch (e) {
   226→    AppLogger.error('Failed to resolve Supabase client', error: e);
   227→    rethrow;
   228→  }
   229→});
   230→
   231→/// Provides a configured [Dio] instance with base options pulled from
   232→/// [AppConfig] and [ApiConstants].
   233→final dioProvider = Provider<Dio>((ref) {
   234→  final dio = Dio(
   235→    BaseOptions(
   236→      baseUrl: AppConfig.baseUrl,
   237→      connectTimeout: AppConfig.connectTimeoutDuration,
   238→      receiveTimeout: AppConfig.receiveTimeoutDuration,
   239→      sendTimeout: AppConfig.sendTimeoutDuration,
   240→      headers: ApiConstants.baseHeaders(),
   241→      responseType: ResponseType.json,
   242→      validateStatus: (status) => status != null && status < 500,
   243→    ),
   244→  );
   245→  return dio;
   246→});
   247→
   248→/// Provides the [ApiClient] wrapper around [Dio].
   249→final apiClientProvider = Provider<ApiClient>((ref) {
   250→  final dio = ref.watch(dioProvider);
   251→  return ApiClient(dio);
   252→});
   253→
   254→/// Re-exports the [NetworkInfo] provider from the core layer so that
   255→/// it is accessible through the DI container alongside other services.
   256→final networkInfoProvider = Provider<NetworkInfo>((ref) {
   257→  // Delegate to the core-layer provider.
   258→  return ref.watch(coreNetworkInfoProvider);
   259→});

   302→/// Stream provider that emits the current Supabase [sb.AuthState].
   303→///
   304→/// Automatically reconnects on transient failures and emits the latest
   305→/// state to any listener.
   306→final authStateProvider = StreamProvider<sb.AuthState>((ref) {
   307→  final supabaseClient = ref.watch(supabaseClientProvider);
   308→  return supabaseClient.auth.onAuthStateChange;
   309→});
   310→
   311→/// Convenience provider that yields the currently authenticated [sb.User],
   312→/// or `null` if not signed in.
   313→final currentUserProvider = Provider<sb.User?>((ref) {
   314→  final authState = ref.watch(authStateProvider);
   315→  return authState.when(
   316→    data: (state) => state.session?.user,
   317→    loading: () => SupabaseConfig.currentUser,
   318→    error: (_, __) => null,
   319→  );
   320→});
   321→
   322→/// Convenience provider that yields `true` when the user is authenticated.
   323→final isAuthenticatedProvider = Provider<bool>((ref) {
   324→  final user = ref.watch(currentUserProvider);
   325→  return user != null;
   326→});

   329→///
   330→/// This is a future provider since it reads from secure storage.
   331→final userRoleProvider = FutureProvider<String>((ref) async {
   332→  final storageService = ref.watch(storageServiceProvider);
   333→  return storageService.getUserRole() ?? 'student';

   352→    data: (results) =>
   353→        results.any((r) => r != ConnectivityResult.none),
   354→    loading: () => true,
   355→    error: (_, __) => false,
