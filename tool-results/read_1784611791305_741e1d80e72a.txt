    13→// ───────────────────────────────────────────────────────────────────────
    14→// Helper: parse DateTime accepting both snake_case and camelCase keys
    15→// ───────────────────────────────────────────────────────────────────────
    16→
    17→DateTime _parseDateTime(dynamic value) {
    18→  if (value is String) return DateTime.parse(value);
    19→  if (value is DateTime) return value;
    20→  return DateTime.now();
    21→}
    22→
    23→DateTime? _parseDateTimeNullable(dynamic value) {
    24→  if (value == null) return null;
    25→  if (value is String) return DateTime.parse(value);
    26→  if (value is DateTime) return value;
    27→  return null;
    28→}
    29→
    30→// ───────────────────────────────────────────────────────────────────────
    31→// Helper: parse list of strings from JSON
    32→// ───────────────────────────────────────────────────────────────────────
    33→
    34→List<String> _parseStringList(dynamic value) {
    35→  if (value == null) return const [];
    36→  if (value is List) {
    37→    return value.whereType<String>().toList();
    38→  }
    39→  return const [];
    40→}
    41→
    42→// ───────────────────────────────────────────────────────────────────────
    43→// Helper: parse nested model lists from JSON
    44→// ───────────────────────────────────────────────────────────────────────
    45→
    46→List<ExamTemplateSectionModel> _parseTemplateSections(dynamic data) {
    47→  if (data == null) return const [];
    48→  if (data is List) {
    49→    return data
    50→        .whereType<Map<String, dynamic>>()
    51→        .map((e) => ExamTemplateSectionModel.fromJson(e))
    52→        .toList();
    53→  }
    54→  return const [];
    55→}
