    13→/// Concrete implementation of [ResultsRepository] that delegates
    14→/// all operations to [ResultsRemoteDataSource].
    15→///
    16→/// Responsibilities:
    17→/// - Converts domain entities to/from data models
    18→/// - Catches data-layer exceptions and converts them to domain [Failure] types
    19→/// - Returns [Result] to force callers to handle both success and failure
    20→class ResultsRepositoryImpl implements ResultsRepository {
    21→  ResultsRepositoryImpl({
    22→    required ResultsRemoteDataSource remoteDataSource,
    23→    required sb.SupabaseClient supabaseClient,
    24→  })  : _remoteDataSource = remoteDataSource,
    25→        _supabaseClient = supabaseClient;
    26→
    27→  final ResultsRemoteDataSource _remoteDataSource;

    30→  // ═══════════════════════════════════════════════════════════════════════
    31→  // Helper: Convert exceptions to Failures
    32→  // ═══════════════════════════════════════════════════════════════════════
    33→
    34→  Failure _mapExceptionToFailure(Object e) {
    35→    if (e is AuthException) {
    36→      return Failure.auth(message: e.message, code: e.code);
    37→    } else if (e is ServerException) {
    38→      return Failure.server(
    39→        message: e.message,
    40→        statusCode: e.statusCode,
    41→        data: e.data,
    42→      );
    43→    } else if (e is CacheException) {
    44→      return Failure.cache(message: e.message);
    45→    } else if (e is NetworkException) {
    46→      return Failure.network(message: e.message);
    47→    } else if (e is ValidationException) {
    48→      return Failure.validation(
    49→        message: e.message,
    50→        fieldErrors: e.fieldErrors,
    51→      );
    52→    } else if (e is NotFoundException) {
    53→      return Failure.notFound(message: e.message);
    54→    } else if (e is UnauthorizedException) {
    55→      return Failure.unauthorized(message: e.message);
    56→    } else if (e is ForbiddenException) {
    57→      return Failure.forbidden(message: e.message);
    58→    } else {
    59→      AppLogger.error('Unexpected exception in ResultsRepositoryImpl', error: e);
    60→      return Failure.server(
    61→        message: 'An unexpected error occurred: $e',
    62→        statusCode: 500,
    63→      );
    64→    }
    65→  }

   171→  @override
   172→  Future<Result<GradeScaleEntryEntity?>> applyGradeScale(
   173→    double percentage,
   174→    String scaleId,
   175→  ) async {
   176→    try {
   177→      final scaleModel = await _remoteDataSource.getGradeScale(scaleId);
   178→      final scale = scaleModel.toEntity();
   179→
   180→      // Find the matching entry for the given percentage
   181→      GradeScaleEntryEntity? match;
   182→      for (final entry in scale.scaleEntries) {
   183→        if (percentage >= entry.minPercentage &&
   184→            percentage <= entry.maxPercentage) {
   185→          match = entry;
   186→          break;
   187→        }
   188→      }
   189→
   190→      return Success(match);
   191→    } on NotFoundException catch (e) {
   192→      return FailureResult(Failure.notFound(message: e.message));
   193→    } on ServerException catch (e) {
   194→      return FailureResult(Failure.server(
   195→        message: e.message,
   196→        statusCode: e.statusCode,
   197→        data: e.data,
   198→      ));
   199→    } catch (e) {

   302→  @override
   303→  Future<Result<AiGradingResultEntity>> reviewAiGrading({
   304→    required String aiGradingId,
   305→    required double finalScore,
   306→    required bool isAccepted,
   307→    String? reviewComment,
   308→  }) async {
   309→    try {
   310→      // Fetch the existing AI grading result to update it
   311→      final existingList = await _supabaseClient
   312→          .from('ai_grading_results')
   313→          .select()
   314→          .eq('id', aiGradingId)
   315→          .single();
   316→
   317→      final existing = AiGradingResultModel.fromJson(existingList);
   318→
   319→      final updated = existing.copyWith(
   320→        finalScore: finalScore,
   321→        isAccepted: isAccepted,
   322→        reviewComment: reviewComment,
   323→        reviewedAt: DateTime.now(),
   324→        status: isAccepted ? 'overridden' : existing.status,
   325→      );
   326→
   327→      final result = await _remoteDataSource.updateAiGradingResult(updated);
   328→      return Success(result.toEntity());
   329→    } on sb.PostgrestException catch (e) {
   330→      if (e.code == 'PGRST116') {
   331→        return FailureResult(
   332→          Failure.notFound(message: 'AI grading result not found: $aiGradingId'),
   333→        );
   334→      }
   335→      return FailureResult(Failure.server(
   336→        message: e.message,
   337→        statusCode: int.tryParse(e.code ?? '') ?? 500,
   338→        data: e.details,
   339→      ));
   340→    } on ServerException catch (e) {
   341→      return FailureResult(Failure.server(
   342→        message: e.message,
   343→        statusCode: e.statusCode,
   344→        data: e.data,
   345→      ));
   346→    } catch (e) {
   347→      return FailureResult(_mapExceptionToFailure(e));
   348→    }
   349→  }

  1212→  @override
  1213→  Future<Result<void>> recomputeClassResults({
  1214→    required String classId,
  1215→    required String academicSessionId,
  1216→  }) async {
  1217→    try {
  1218→      // 1. Get all student/subject pairs for this class and session
  1219→      final subjectResults = await _supabaseClient
  1220→          .from('student_subject_results')
  1221→          .select('student_id, subject_id')
  1222→          .eq('class_id', classId)
  1223→          .eq('academic_session_id', academicSessionId);
  1224→
  1225→      // 2. Call the compute_student_subject_result RPC for each pair
  1226→      for (final row in subjectResults as List) {
  1227→        try {
  1228→          await _supabaseClient.rpc(
  1229→            'compute_student_subject_result',
  1230→            params: {
  1231→              'p_student_id': row['student_id'],
  1232→              'p_subject_id': row['subject_id'],
  1233→              'p_class_id': classId,
  1234→              'p_academic_session_id': academicSessionId,
  1235→            },
  1236→          );
  1237→        } catch (e) {
  1238→          AppLogger.warning(
  1239→            'Failed to recompute result for student ${row['student_id']}, '
  1240→            'subject ${row['subject_id']}: $e',
  1241→          );
  1242→          // Continue with other students even if one fails
  1243→        }
  1244→      }
  1245→
  1246→      // 3. Recompute overall results for all students in the class
  1247→      final studentIds = (subjectResults as List)
  1248→          .map((r) => r['student_id'] as String)
  1249→          .toSet();
  1250→
  1251→      for (final studentId in studentIds) {
  1252→        try {
  1253→          await _supabaseClient.rpc(
  1254→            'compute_student_overall_result',
  1255→            params: {
  1256→              'p_student_id': studentId,
  1257→              'p_class_id': classId,
  1258→              'p_academic_session_id': academicSessionId,
  1259→            },
  1260→          );
  1261→        } catch (e) {
  1262→          AppLogger.warning(
  1263→            'Failed to recompute overall result for student $studentId: $e',
  1264→          );
  1265→          // Continue with other students even if one fails
  1266→        }
  1267→      }
  1268→
  1269→      return const Success(null);
  1270→    } on sb.PostgrestException catch (e) {
  1271→      AppLogger.error('Recompute class results failed', error: e);
  1272→      return FailureResult(Failure.server(
  1273→        message: e.message,
