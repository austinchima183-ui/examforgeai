    21→/// Seller's control center with tabs for Overview, Products, Analytics,
    22→/// and Earnings.
    23→///
    24→/// ```dart
    25→/// Navigator.push(context, MaterialPageRoute(builder: (_) => SellerDashboardPage()));
    26→/// ```
    27→class SellerDashboardPage extends ConsumerStatefulWidget {
    28→  const SellerDashboardPage({super.key});
    29→
    30→  @override
    31→  ConsumerState<SellerDashboardPage> createState() =>
    32→      _SellerDashboardPageState();
    33→}
    34→
    35→class _SellerDashboardPageState extends ConsumerState<SellerDashboardPage>
    36→    with SingleTickerProviderStateMixin {
    37→  late final TabController _tabController;
    38→
    39→  static const _tabs = [
    40→    Tab(icon: Icon(Icons.dashboard_outlined), text: 'Overview'),
    41→    Tab(icon: Icon(Icons.inventory_2_outlined), text: 'Products'),
    42→    Tab(icon: Icon(Icons.analytics_outlined), text: 'Analytics'),
    43→    Tab(icon: Icon(Icons.account_balance_wallet_outlined), text: 'Earnings'),
    44→  ];
    45→
    46→  @override
    47→  void initState() {
    48→    super.initState();
    49→    _tabController = TabController(length: _tabs.length, vsync: this);
    50→    WidgetsBinding.instance.addPostFrameCallback((_) => _loadData());
    51→  }
    52→
    53→  @override
    54→  void dispose() {
    55→    _tabController.dispose();
    56→    super.dispose();
    57→  }
    58→
    59→  Future<void> _loadData() async {
    60→    final sellerNotifier = ref.read(sellerProvider.notifier);
    61→    await Future.wait([
    62→      sellerNotifier.loadSellerProfile(userId: 'current_user'),
    63→      sellerNotifier.loadSellerProducts(sellerId: 'current_seller'),
    64→      sellerNotifier.loadSellerAnalytics(sellerId: 'current_seller'),
    65→      ref
    66→          .read(marketplaceNotificationProvider.notifier)
    67→          .loadNotifications(userId: 'current_user'),
    68→    ]);
    69→  }
    70→
    71→  Future<void> _refresh() async {
    72→    final sellerNotifier = ref.read(sellerProvider.notifier);
    73→    await Future.wait([
    74→      sellerNotifier.loadSellerProfile(userId: 'current_user'),
    75→      sellerNotifier.loadSellerProducts(sellerId: 'current_seller'),
    76→      sellerNotifier.loadSellerAnalytics(sellerId: 'current_seller'),
    77→      ref
    78→          .read(marketplaceNotificationProvider.notifier)
    79→          .loadNotifications(userId: 'current_user'),
    80→    ]);
    81→  }
    82→
    83→  void _showSnackBar(String message, {bool isError = false}) {
    84→    context.scaffoldMessenger.showSnackBar(
    85→      SnackBar(
    86→        content: Text(message),
    87→        backgroundColor: isError ? AppColors.error : null,
    88→        behavior: SnackBarBehavior.floating,
    89→      ),
    90→    );
    91→  }
    92→
    93→  @override
    94→  Widget build(BuildContext context) {
    95→    final sellerState = ref.watch(sellerProvider);
    96→    final notifState = ref.watch(marketplaceNotificationProvider);
    97→
    98→    // Listen for error/success messages
    99→    ref.listen<SellerState>(sellerProvider, (prev, next) {
   100→      if (next.error != null && (prev?.error != next.error)) {
   101→        _showSnackBar(next.error!, isError: true);
   102→        ref.read(sellerProvider.notifier).clearError();
   103→      }
   104→      if (next.successMessage != null &&
   105→          (prev?.successMessage != next.successMessage)) {
   106→        _showSnackBar(next.successMessage!);
   107→        ref.read(sellerProvider.notifier).clearSuccess();
   108→      }
   109→    });
   110→
   111→    return Scaffold(
   112→      appBar: AppAppBar(
   113→        title: 'Seller Dashboard',
   114→        actions: [
   115→          Stack(
   116→            alignment: Alignment.center,
   117→            children: [
   118→              AppIconButton(
   119→                icon: Icons.notifications_outlined,
   120→                onPressed: () {
   121→                  // Navigate to notifications
   122→                },
   123→                tooltip: 'Notifications',
   124→              ),
   125→              if (notifState.hasUnread)
   126→                Positioned(
   127→                  top: 4,
   128→                  right: 4,
   129→                  child: Container(
   130→                    padding: const EdgeInsets.all(2),
   131→                    decoration: const BoxDecoration(
   132→                      color: AppColors.error,
   133→                      shape: BoxShape.circle,
   134→                    ),
   135→                    constraints: const BoxConstraints(
   136→                      minWidth: 16,
   137→                      minHeight: 16,
   138→                    ),
   139→                    child: Text(
   140→                      '${notifState.unreadCount}',
   141→                      style: const TextStyle(
   142→                        color: Colors.white,
   143→                        fontSize: 10,
   144→                        fontWeight: AppTypography.wBold,
   145→                      ),
   146→                      textAlign: TextAlign.center,
   147→                    ),
   148→                  ),
   149→                ),
   150→            ],
   151→          ),
   152→        ],
   153→        bottom: TabBar(
   154→          controller: _tabController,
   155→          tabs: _tabs,
   156→          isScrollable: context.isMobile,
   157→          labelStyle: context.textTheme.labelLarge?.copyWith(
   158→            fontWeight: AppTypography.wSemiBold,
   159→          ),
   160→          unselectedLabelStyle: context.textTheme.labelMedium,
   161→          indicatorSize: TabBarIndicatorSize.label,
   162→        ),
   163→      ),
   164→      body: sellerState.isLoading && sellerState.products.isEmpty
   165→          ? const Center(child: AppLoadingSpinner())
   166→          : sellerState.error != null && sellerState.products.isEmpty
   167→              ? AppErrorState.serverError(onRetry: _loadData)
   168→              : RefreshIndicator(
   169→                  onRefresh: _refresh,
   170→                  child: TabBarView(
   171→                    controller: _tabController,
   172→                    children: [
   173→                      _OverviewTab(state: sellerState),
   174→                      _ProductsTab(state: sellerState),
   175→                      _AnalyticsTab(state: sellerState),
   176→                      _EarningsTab(state: sellerState),
   177→                    ],
   178→                  ),
   179→                ),
   180→    );
   181→  }
   182→}

   637→class _ProductsTab extends ConsumerStatefulWidget {

   645→
   646→class _ProductsTabState extends ConsumerState<_ProductsTab> {
   647→  /// `null` means "All" (no filter); otherwise filter by specific status.
   648→  MarketplaceProductStatus? _filter;
   649→
   650→  static const _filterOptions = <(MarketplaceProductStatus?, String, IconData)>[
   651→    (null, 'All', Icons.list_rounded),
   652→    (MarketplaceProductStatus.draft, 'Draft', Icons.edit_note_rounded),
   653→    (MarketplaceProductStatus.pendingReview, 'Pending', Icons.hourglass_top_rounded),
   654→    (MarketplaceProductStatus.approved, 'Approved', Icons.check_circle_rounded),
   655→    (MarketplaceProductStatus.rejected, 'Rejected', Icons.cancel_rounded),
   656→  ];
   657→
   658→  List<MarketplaceProductEntity> get _filteredProducts {

  1716→// ═══════════════════════════════════════════════════════════════════════════════
  1717→
  1718→/// Formats a numeric value with comma separators for thousands.
  1719→String _formatNumber(double value) {
  1720→  if (value >= 1000000) {
  1721→    return '${(value / 1000000).toStringAsFixed(1)}M';
  1722→  }
  1723→  if (value >= 1000) {
  1724→    return '${(value / 1000).toStringAsFixed(1)}K';
  1725→  }
  1726→  return value.toInt().toString();
  1727→}
  1728→
  1729→/// Formats a [DateTime] as a relative or short date string.
  1730→String _formatDate(DateTime date) {
  1731→  final now = DateTime.now();
  1732→  final diff = now.difference(date);
  1733→
  1734→  if (diff.inDays == 0) {
  1735→    if (diff.inHours == 0) {
  1736→      if (diff.inMinutes == 0) return 'Just now';
  1737→      return '${diff.inMinutes}m ago';
  1738→    }
  1739→    return '${diff.inHours}h ago';
  1740→  }
  1741→  if (diff.inDays < 7) return '${diff.inDays}d ago';
  1742→  if (diff.inDays < 30) {
  1743→    final weeks = (diff.inDays / 7).floor();
  1744→    return '${weeks}w ago';
  1745→  }
  1746→  return '${date.day}/${date.month}/${date.year}';
  1747→}
