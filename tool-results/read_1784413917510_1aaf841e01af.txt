    22→/// The main Parent Portal Dashboard page.
    23→///
    24→/// Presents a modern, information-dense layout with:
    25→/// - Welcome section with parent name and current date
    26→/// - Child summary cards (horizontal scrollable)
    27→/// - Quick stats row (messages, notifications, events, insights)
    28→/// - AI Insights section (severity color-coded)
    29→/// - Upcoming events section
    30→/// - Recent announcements section
    31→/// - Pending assignments section (grouped by child)
    32→/// - Notifications preview section
    33→///
    34→/// Uses [ConsumerStatefulWidget] with a private [_State] class pattern,
    35→/// loads data in [initState] via [WidgetsBinding.instance.addPostFrameCallback],
    36→/// and renders a responsive layout using [CustomScrollView] with slivers.
    37→class ParentDashboardPage extends ConsumerStatefulWidget {
    38→  const ParentDashboardPage({super.key});
    39→
    40→  @override
    41→  ConsumerState<ParentDashboardPage> createState() => _State();
    42→}

   125→  Widget _buildBody(BuildContext context, ParentDashboardState state) {
   126→    // Loading state
   127→    if (state.isLoading && state.dashboard == null) {
   128→      return _buildShimmerLoading(context);
   129→    }
   130→
   131→    // Error state
   132→    if (state.error != null && state.dashboard == null) {
   133→      return AppErrorState.genericError(
   134→        message: state.error,
   135→        onRetry: () =>
   136→            ref.read(parentDashboardProvider.notifier).loadDashboard(),
   137→      );
   138→    }
   139→
   140→    // Empty state
   141→    final dashboard = state.dashboard;
   142→    if (dashboard == null) {
   143→      return AppEmptyState.noData(
   144→        title: 'No Dashboard Data',
   145→        subtitle: 'Your parent portal dashboard will appear here once loaded.',
   146→        actionLabel: 'Refresh',
   147→        onAction: () =>
   148→            ref.read(parentDashboardProvider.notifier).loadDashboard(),
   149→      );
   150→    }
   151→
   152→    // Success — render dashboard
   153→    return RefreshIndicator(
   154→      onRefresh: () =>
   155→          ref.read(parentDashboardProvider.notifier).refreshDashboard(),
   156→      child: CustomScrollView(
   157→        physics: const AlwaysScrollableScrollPhysics(),
   158→        slivers: [
   159→          // ─── Welcome Section ──────────────────────────────────────
   160→          SliverToBoxAdapter(
   161→            child: _buildWelcomeSection(context, dashboard),
   162→          ),
   163→
   164→          // ─── Child Summary Cards ──────────────────────────────────
   165→          SliverToBoxAdapter(
   166→            child: _buildChildSummaryCards(context, dashboard),
   167→          ),
   168→
   169→          // ─── Quick Stats Row ─────────────────────────────────────
   170→          SliverToBoxAdapter(
   171→            child: _buildQuickStatsRow(context, dashboard),
   172→          ),
   173→
   174→          // ─── AI Insights Section ─────────────────────────────────
   175→          SliverToBoxAdapter(
   176→            child: _buildAiInsightsSection(context),
   177→          ),
   178→
   179→          // ─── Upcoming Events Section ─────────────────────────────
   180→          SliverToBoxAdapter(
   181→            child: _buildUpcomingEventsSection(context, dashboard),
   182→          ),
   183→
   184→          // ─── Recent Announcements Section ────────────────────────
   185→          SliverToBoxAdapter(
   186→            child: _buildAnnouncementsSection(context, dashboard),
   187→          ),
   188→
   189→          // ─── Pending Assignments Section ─────────────────────────
   190→          SliverToBoxAdapter(
   191→            child: _buildPendingAssignmentsSection(context, dashboard),
   192→          ),
   193→
   194→          // ─── Notifications Preview ───────────────────────────────
   195→          SliverToBoxAdapter(
   196→            child: _buildNotificationsPreview(context, dashboard),
   197→          ),
   198→
   199→          // Bottom padding
   200→          const SliverPadding(padding: EdgeInsets.only(bottom: Spacings.xxl)),
   201→        ],
   202→      ),
   203→    );
   204→  }

  1282→  /// Formats a [DateTime] as a short time string (e.g., "2:30 PM").
  1283→  String _formatTime(DateTime dt) {
  1284→    final hour = dt.hour > 12 ? dt.hour - 12 : dt.hour;
  1285→    final minute = dt.minute.toString().padLeft(2, '0');
  1286→    final ampm = dt.hour >= 12 ? 'PM' : 'AM';
  1287→    return '$hour:$minute $ampm';
  1288→  }
