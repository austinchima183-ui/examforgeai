   187→/// Optimises image loading based on network connection quality.
   188→///
   189→/// Adapts image URLs, quality factors, and caching behaviour to match the
   190→/// current connectivity situation. On excellent connections the original
   191→/// full-resolution image is served; on limited connections thumbnails and
   192→/// reduced-quality variants are preferred.
   193→class ImageOptimizer {
   194→  ImageOptimizer({
   195→    required this.config,
   196→    required this.cacheManager,
   197→  });
   198→
   199→  final PerformanceConfig config;
   200→  final CacheManager cacheManager;
   201→
   202→  /// Returns an optimised image URL adjusted for [connectionQuality].
   203→  ///
   204→  /// - **Excellent / Good** (≥ 0.6): returns the original [url] unchanged.
   205→  /// - **Limited** (0.3 – 0.59): appends `w` and `q` query parameters if the
   206→  ///   URL appears to support them (e.g. CDN-backed URLs with existing query
   207→  ///   params), otherwise appends a standard `?width=` & `?quality=` suffix.
   208→  /// - **Offline / Very poor** (< 0.3): returns the original URL — the cache
   209→  ///   layer will attempt to serve a previously cached version.
   210→  String getOptimizedUrl(
   211→    String url, {
   212→    required double connectionQuality,
   213→    int? maxWidth,
   214→    int? maxHeight,
   215→  }) {
   216→    try {
   217→      // Excellent / good — full quality
   218→      if (connectionQuality >= 0.6) return url;
   219→
   220→      // Limited connection — request smaller images
   221→      if (connectionQuality >= 0.3) {
   222→        final width = maxWidth ?? _defaultWidthForQuality(connectionQuality);
   223→        final height = maxHeight ?? (width * 1.5).round();
   224→        final quality = (getImageQuality(connectionQuality) * 100).round();
   225→
   226→        final separator = url.contains('?') ? '&' : '?';
   227→        return '$url${separator}w=$width&h=$height&q=$quality';
   228→      }
   229→
   230→      // Offline / very poor — return original, rely on cache
   231→      return url;
   232→    } catch (e, st) {
   233→      AppLogger.warning('ImageOptimizer: failed to optimize URL', error: e, stackTrace: st);
   234→      return url;
   235→    }
   236→  }
   237→
   238→  /// Returns the appropriate image quality factor for [connectionQuality].
   239→  ///
   240→  /// | Quality range | Factor |
   241→  /// |---------------|--------|
   242→  /// | ≥ 0.8         | 1.0    |
   243→  /// | ≥ 0.6         | 0.8    |
   244→  /// | ≥ 0.3         | 0.5    |
   245→  /// | < 0.3         | 0.3    |
   246→  double getImageQuality(double connectionQuality) {
   247→    if (connectionQuality >= 0.8) return 1.0;
   248→    if (connectionQuality >= 0.6) return 0.8;
   249→    if (connectionQuality >= 0.3) return 0.5;
   250→    return 0.3;
   251→  }
   252→
   253→  /// Whether a thumbnail should be loaded instead of a full image.
   254→  ///
   255→  /// Returns `true` for limited and offline connections.
   256→  bool shouldLoadThumbnail(double connectionQuality) => connectionQuality < 0.6;
   257→
   258→  /// Returns a cached image provider for [url], optionally with a custom
   259→  /// [quality] and [cacheKey].
   260→  ///
   261→  /// Uses [CachedNetworkImageProvider] when image caching is enabled in the
   262→  /// config; otherwise falls back to [NetworkImage].
   263→  ImageProvider getCachedImageProvider(
   264→    String url, {
   265→    double? quality,
   266→    String? cacheKey,
   267→  }) {
   268→    if (!config.enableImageCache) {
   269→      return NetworkImage(url);
   270→    }
   271→
   272→    final effectiveQuality = quality ?? config.imageQuality;
   273→
   274→    return CachedNetworkImageProvider(
   275→      url,
   276→      cacheKey: cacheKey,
   277→      maxWidth: (effectiveQuality < 1.0) ? _defaultWidthForQuality(effectiveQuality) : null,
   278→      errorListener: (error) {
   279→        AppLogger.warning(
   280→          'ImageOptimizer: cached image load failed for $url',
   281→          error: error,
   282→        );
   283→      },
   284→    );
   285→  }
   286→
   287→  /// Preloads a list of image [urls] into the cache.
   288→  ///
   289→  /// Images are preloaded with quality adjusted by [connectionQuality].
   290→  /// Errors are caught per-image so that one failure does not block the rest.
   291→  ///
   292→  /// This method triggers [CachedNetworkImageProvider] to download and cache
   293→  /// each image. The returned future completes once all downloads finish
   294→  /// (or fail gracefully).
   295→  Future<void> preloadImages(
   296→    List<String> urls, {
   297→    double connectionQuality = 1.0,
   298→  }) async {
   299→    final quality = getImageQuality(connectionQuality);
   300→
   301→    await Future.wait(
   302→      urls.map((url) async {
   303→        try {
   304→          final optimizedUrl = getOptimizedUrl(
   305→            url,
   306→            connectionQuality: connectionQuality,
   307→          );
   308→          final provider = CachedNetworkImageProvider(
   309→            optimizedUrl,
   310→            maxWidth: quality < 1.0 ? _defaultWidthForQuality(quality) : null,
   311→          );
   312→          // Resolve the provider into an ImageStream and wait for the
   313→          // first frame to be ready — the caching layer stores the file.
   314→          final stream = provider.resolve(const ImageConfiguration());
   315→          final completer = Completer<void>();
   316→          late ImageStreamListener listener;
   317→          listener = ImageStreamListener(
   318→            (ImageInfo info, bool synchronousCall) {
   319→              completer.complete();
   320→              stream.removeListener(listener);
   321→            },
   322→            onError: (dynamic exception, StackTrace? stackTrace) {
   323→              completer.completeError(exception, stackTrace);
   324→              stream.removeListener(listener);
   325→            },
   326→          );
   327→          stream.addListener(listener);
   328→          await completer.future;
   329→        } catch (e, st) {
   330→          AppLogger.warning(
   331→            'ImageOptimizer: failed to preload $url',
   332→            error: e,
   333→            stackTrace: st,
   334→          );
   335→        }
   336→      }),
   337→    );
   338→  }
   339→
   340→  /// Clears the image cache managed by [CachedNetworkImage].
   341→  ///
   342→  /// Clears both the Flutter in-memory image cache and the disk cache
   343→  /// used by the `flutter_cache_manager` under the hood.
   344→  Future<void> clearImageCache() async {
   345→    try {
   346→      // Clear the in-memory Flutter image cache.
   347→      PaintingBinding.instance.imageCache.clear();
   348→      PaintingBinding.instance.imageCache.clearLiveImages();
   349→
   350→      // Clear the DefaultCacheManager disk cache used by CachedNetworkImage.
   351→      await DefaultCacheManager().emptyCache();
   352→
   353→      AppLogger.info('ImageOptimizer: image cache cleared');
   354→    } catch (e, st) {
   355→      AppLogger.error(
   356→        'ImageOptimizer: failed to clear image cache',
   357→        error: e,
   358→        stackTrace: st,
   359→      );
   360→    }
   361→  }
   362→
   363→  /// Returns the estimated size of the image cache in bytes.
   364→  ///
   365→  /// Queries the Flutter in-memory image cache. Disk cache size from
   366→  /// `flutter_cache_manager` is not directly queryable, so only the
   367→  /// in-memory portion is reported.
   368→  Future<int> getImageCacheSize() async {
   369→    try {
   370→      final memoryBytes = PaintingBinding.instance.imageCache.currentSizeBytes;
   371→      return memoryBytes;
   372→    } catch (e, st) {
   373→      AppLogger.warning(
   374→        'ImageOptimizer: failed to get image cache size',
   375→        error: e,
   376→        stackTrace: st,
   377→      );
   378→      return 0;
   379→    }
   380→  }

   385→    if (quality >= 0.8) return 1200;
   386→    if (quality >= 0.5) return 600;
   387→    return 300;

   392→// LAZY LOAD CONTROLLER
   393→// ═══════════════════════════════════════════════════════════════════════════════
   394→
   395→/// Manages lazy loading of list data driven by scroll position.
   396→///
   397→/// Provides a [shouldLoadMore] helper that inspects scroll notifications and
   398→/// a [createSliverLazyLoader] factory that builds a complete paginated list
   399→/// widget with loading, error, and empty states.
   400→class LazyLoadController {
   401→  LazyLoadController({required this.config});
   402→
   403→  final PerformanceConfig config;
   404→
   405→  /// Determines whether more data should be loaded based on a scroll
   406→  /// notification and a [threshold] (0.0 – 1.0 of scroll extent).
   407→  ///
   408→  /// Returns `true` when the current scroll position has reached or exceeded
   409→  /// [threshold] fraction of the maximum scroll extent. Also respects
   410→  /// [PerformanceConfig.enableLazyLoading] — returns `false` if disabled.
   411→  bool shouldLoadMore(
   412→    ScrollNotification notification, {
   413→    double threshold = 0.8,
   414→  }) {
   415→    if (!config.enableLazyLoading) return false;
   416→
   417→    if (notification is ScrollUpdateNotification) {
   418→      final metrics = notification.metrics;
   419→      if (metrics.maxScrollExtent <= 0) return false;
   420→
   421→      final scrollFraction = metrics.pixels / metrics.maxScrollExtent;
   422→      return scrollFraction >= threshold;
   423→    }
   424→
   425→    return false;
   426→  }
   427→
   428→  /// Creates a widget that lazily loads paginated data.
   429→  ///
   430→  /// [loader] fetches a page of items given an [offset] and [limit].
   431→  /// [builder] renders each item at the given [index].
   432→  /// Optional [loadingIndicator], [errorWidget], and [emptyWidget] replace
   433→  /// the default states.
   434→  Widget createSliverLazyLoader<T>({
   435→    required Future<List<T>> Function(int offset, int limit) loader,
   436→    required Widget Function(BuildContext, T, int) builder,
   437→    int pageSize = 20,
   438→    Widget? loadingIndicator,
   439→    Widget? errorWidget,
   440→    Widget? emptyWidget,
   441→  }) {
   442→    return _LazyLoadSliver<T>(
   443→      loader: loader,
   444→      builder: builder,
   445→      pageSize: pageSize,
   446→      loadingIndicator: loadingIndicator,
   447→      errorWidget: errorWidget,
   448→      emptyWidget: emptyWidget,
   449→      debounceDuration: config.debounceDuration,
   450→    );
   451→  }
   452→}
   453→
   454→// ── Private Lazy Load Sliver Widget ──────────────────────────────────────
   455→
   456→/// Stateful widget that manages paginated data loading inside a [CustomScrollView].
   457→///
   458→/// Not meant for direct instantiation — use [LazyLoadController.createSliverLazyLoader].
   459→class _LazyLoadSliver<T> extends StatefulWidget {
   460→  const _LazyLoadSliver({
   461→    required this.loader,
   462→    required this.builder,
   463→    this.pageSize = 20,
   464→    this.loadingIndicator,
   465→    this.errorWidget,
   466→    this.emptyWidget,
   467→    this.debounceDuration = const Duration(milliseconds: 300),
   468→  });
   469→
   470→  final Future<List<T>> Function(int offset, int limit) loader;
   471→  final Widget Function(BuildContext, T, int) builder;
   472→  final int pageSize;
   473→  final Widget? loadingIndicator;
   474→  final Widget? errorWidget;
   475→  final Widget? emptyWidget;
   476→  final Duration debounceDuration;
   477→
   478→  @override
   479→  State<_LazyLoadSliver<T>> createState() => _LazyLoadSliverState<T>();
   480→}
   481→
   482→class _LazyLoadSliverState<T> extends State<_LazyLoadSliver<T>> {
   483→  final List<T> _items = [];
   484→  bool _isLoading = false;
   485→  bool _hasError = false;
   486→  bool _hasReachedEnd = false;
   487→  String? _errorMessage;
   488→  int _currentOffset = 0;
   489→  Timer? _debounceTimer;
   490→
   491→  @override
   492→  void initState() {
   493→    super.initState();
   494→    _loadMore();
   495→  }
   496→
   497→  @override
   498→  void dispose() {
   499→    _debounceTimer?.cancel();
   500→    super.dispose();
   501→  }
   502→
   503→  Future<void> _loadMore() async {
   504→    if (_isLoading || _hasReachedEnd) return;
   505→
   506→    setState(() {
   507→      _isLoading = true;
   508→      _hasError = false;
   509→      _errorMessage = null;
   510→    });
   511→
   512→    try {
   513→      final newItems = await widget.loader(_currentOffset, widget.pageSize);
   514→
   515→      if (!mounted) return;
   516→
   517→      setState(() {
   518→        _isLoading = false;
   519→        if (newItems.isEmpty) {
   520→          _hasReachedEnd = true;
   521→        } else {
   522→          _items.addAll(newItems);
   523→          _currentOffset += newItems.length;
   524→          if (newItems.length < widget.pageSize) {
   525→            _hasReachedEnd = true;
   526→          }
   527→        }
   528→      });
   529→    } catch (e, st) {
   530→      AppLogger.error('LazyLoadController: failed to load page', error: e, stackTrace: st);
   531→      if (!mounted) return;
   532→      setState(() {
   533→        _isLoading = false;
   534→        _hasError = true;
   535→        _errorMessage = e.toString();
   536→      });
   537→    }
   538→  }
   539→
   540→  void _onScroll(ScrollNotification notification) {
   541→    if (_debounceTimer?.isActive ?? false) return;
   542→
   543→    final shouldLoad = notification.metrics.maxScrollExtent > 0 &&
   544→        notification.metrics.pixels /
   545→                notification.metrics.maxScrollExtent >=
   546→            0.8;
   547→
   548→    if (shouldLoad && !_isLoading && !_hasReachedEnd) {
   549→      _debounceTimer = Timer(widget.debounceDuration, _loadMore);
   550→    }
   551→  }
   552→
   553→  @override
   554→  Widget build(BuildContext context) {
   555→    return SliverToBoxAdapter(
   556→      child: NotificationListener<ScrollNotification>(
   557→        onNotification: (notification) {
   558→          _onScroll(notification);
   559→          return false;
   560→        },
   561→        child: Column(
   562→          children: [
   563→            if (_items.isEmpty && _isLoading)
   564→              widget.loadingIndicator ??
   565→                  const Padding(
   566→                    padding: EdgeInsets.all(24),
   567→                    child: Center(child: CircularProgressIndicator()),
   568→                  ),
   569→            if (_items.isEmpty && _hasError)
   570→              widget.errorWidget ??
   571→                  Padding(
   572→                    padding: const EdgeInsets.all(24),
   573→                    child: Column(
   574→                      children: [
   575→                        Text(
   576→                          'Failed to load data${_errorMessage != null ? ': $_errorMessage' : ''}',
   577→                          style: Theme.of(context).textTheme.bodyMedium?.copyWith(
   578→                                color: Theme.of(context).colorScheme.error,
   579→                              ),
   580→                        ),
   581→                        const SizedBox(height: 8),
   582→                        TextButton(
   583→                          onPressed: _loadMore,
   584→                          child: const Text('Retry'),
   585→                        ),
   586→                      ],
   587→                    ),
   588→                  ),
   589→            if (_items.isEmpty && !_isLoading && !_hasError && _hasReachedEnd)
   590→              widget.emptyWidget ??
   591→                  Padding(
   592→                    padding: const EdgeInsets.all(24),
   593→                    child: Text(
   594→                      'No data available',
   595→                      style: Theme.of(context).textTheme.bodyMedium,
   596→                    ),
   597→                  ),
   598→            ..._items.asMap().entries.map(
   599→                  (entry) => widget.builder(context, entry.value, entry.key),
   600→                ),
   601→            if (_isLoading && _items.isNotEmpty)
   602→              widget.loadingIndicator ??
   603→                  const Padding(
   604→                    padding: EdgeInsets.all(16),
   605→                    child: Center(child: CircularProgressIndicator()),
   606→                  ),
   607→            if (_hasError && _items.isNotEmpty)
   608→              Padding(
   609→                padding: const EdgeInsets.all(16),
   610→                child: TextButton(
   611→                  onPressed: _loadMore,
   612→                  child: const Text('Tap to retry'),
   613→                ),
   614→              ),
   615→          ],
   616→        ),
   617→      ),
   618→    );
   619→  }

   623→// REQUEST BATCHER
   624→// ═══════════════════════════════════════════════════════════════════════════════
   625→
   626→/// Batches concurrent requests to reduce network overhead.
   627→///
   628→/// When multiple widgets request the same data within a short [batchWindow],
   629→/// only one actual request is made and its result is shared. Requests are
   630→/// grouped by [groupKey]; requests with different keys run independently.
   631→///
   632→/// This is particularly useful when several widgets on the same screen need
   633→/// overlapping API data.
   634→class RequestBatcher {
   635→  RequestBatcher({this.defaultBatchWindow = const Duration(milliseconds: 50)});
   636→
   637→  /// Default batch window for collecting requests.
   638→  final Duration defaultBatchWindow;
   639→
   640→  /// Active batch groups: key → list of pending completer/request pairs.
   641→  final Map<String, _BatchGroup<dynamic>> _batchGroups = {};
   642→
   643→  /// Timers for each batch group.
   644→  final Map<String, Timer> _timers = {};
   645→
   646→  /// Submits [request] for batched execution.
   647→  ///
   648→  /// If a batch for [groupKey] is already collecting, the request is added
   649→  /// to that batch. If no batch exists, a new one is created and a timer
   650→  /// is started for [batchWindow]. When the timer fires, the first request
   651→  /// in the group is executed and the result is fanned out to all waiters.
   652→  ///
   653→  /// If [groupKey] is `null`, the request is executed immediately without
   654→  /// batching.
   655→  Future<T> batch<T>({
   656→    required Future<T> Function() request,
   657→    String? groupKey,
   658→    Duration batchWindow = const Duration(milliseconds: 50),
   659→  }) async {
   660→    // No grouping → execute immediately
   661→    if (groupKey == null) {
   662→      return request();
   663→    }
   664→
   665→    final completer = Completer<T>();
   666→
   667→    final group = _batchGroups[groupKey];
   668→    if (group != null && group.isActive) {
   669→      // Add to existing batch
   670→      group.add<T>(completer);
   671→      return completer.future;
   672→    }
   673→
   674→    // Start a new batch
   675→    final newGroup = _BatchGroup<T>();
   676→    newGroup.add<T>(completer);
   677→    _batchGroups[groupKey] = newGroup;
   678→
   679→    // Set timer to flush the batch
   680→    _timers[groupKey] = Timer(batchWindow, () => _flushBatch<T>(groupKey, request));
   681→
   682→    return completer.future;
   683→  }
   684→
   685→  /// Flushes a batch group by executing the [request] once and distributing
   686→  /// the result to all waiting completers.
   687→  Future<void> _flushBatch<T>(String key, Future<T> Function() request) async {
   688→    final group = _batchGroups.remove(key) as _BatchGroup<T>?;
   689→    _timers.remove(key)?.cancel();
   690→
   691→    if (group == null || group.completers.isEmpty) return;
   692→
   693→    try {
   694→      final result = await request();
   695→      for (final completer in group.completers) {
   696→        if (!completer.isCompleted) {
   697→          (completer as Completer<T>).complete(result);
   698→        }
   699→      }
   700→    } catch (e, st) {
   701→      AppLogger.warning(
   702→        'RequestBatcher: batch request failed for key=$key',
   703→        error: e,
   704→        stackTrace: st,
   705→      );
   706→      for (final completer in group.completers) {
   707→        if (!completer.isCompleted) {
   708→          (completer as Completer<T>).completeError(e, st);
   709→        }
   710→      }
   711→    }
   712→  }
   713→
   714→  /// Cancels all pending batches and completes them with errors.
   715→  void cancelAll() {
   716→    for (final entry in _timers.entries) {
   717→      entry.value.cancel();
   718→      final group = _batchGroups[entry.key];
   719→      if (group != null) {
   720→        for (final completer in group.completers) {
   721→          if (!completer.isCompleted) {
   722→            completer.completeError(
   723→              StateError('RequestBatcher: batch cancelled for key=${entry.key}'),
   724→            );
   725→          }
   726→        }
   727→      }
   728→    }
   729→    _timers.clear();
   730→    _batchGroups.clear();
   731→  }
   732→}
   733→
   734→/// Internal representation of a batch group.
   735→class _BatchGroup<T> {
   736→  final List<Completer<dynamic>> completers = [];
   737→  bool isActive = true;
   738→
   739→  void add<R>(Completer<R> completer) {
   740→    completers.add(completer);
   741→  }
   742→}

   745→// MEMORY MANAGER
   746→// ═══════════════════════════════════════════════════════════════════════════════
   747→
   748→/// Monitors and manages application memory.
   749→///
   750→/// Provides methods to query current memory usage, detect memory pressure,
   751→/// register disposable resources, and clean up caches when memory is low.
   752→class MemoryManager {
   753→  MemoryManager({required this.cacheManager});
   754→
   755→  final CacheManager cacheManager;
   756→
   757→  /// Callbacks registered for low-memory notifications.
   758→  final List<VoidCallback> _lowMemoryCallbacks = [];
   759→
   760→  /// Disposable resources that can be freed on memory pressure.
   761→  final List<Disposable> _disposables = [];
   762→
   763→  /// Whether a low-memory event has been detected.
   764→  bool _isUnderPressure = false;
   765→
   766→  /// Returns the current memory usage as a map with `usedMB` and `totalMB`.
   767→  ///
   768→  /// On native platforms this reads from [ProcessInfo]; on web a stub is
   769→  /// returned since `dart:io` is unavailable.
   770→  Map<String, dynamic> getCurrentMemoryUsage() {
   771→    try {
   772→      if (kIsWeb) {
   773→        return {'usedMB': 0, 'totalMB': 0, 'platform': 'web'};
   774→      }
   775→      final currentRss = ProcessInfo.currentRss;
   776→      final maxRss = ProcessInfo.maxRss;
   777→      return {
   778→        'usedMB': (currentRss / (1024 * 1024)).round(),
   779→        'totalMB': (maxRss / (1024 * 1024)).round(),
   780→        'platform': Platform.operatingSystem,
   781→      };
   782→    } catch (e, st) {
   783→      AppLogger.warning('MemoryManager: failed to query memory', error: e, stackTrace: st);
   784→      return {'usedMB': 0, 'totalMB': 0, 'error': e.toString()};
   785→    }
   786→  }
   787→
   788→  /// Whether the app is currently under memory pressure.
   789→  ///
   790→  /// Uses a heuristic: if current RSS exceeds 70% of max RSS, or if the
   791→  /// platform has signalled a memory warning, this returns `true`.
   792→  bool isMemoryPressure() {
   793→    try {
   794→      if (kIsWeb) return false;
   795→
   796→      final usage = getCurrentMemoryUsage();
   797→      final usedMB = usage['usedMB'] as int? ?? 0;
   798→      final totalMB = usage['totalMB'] as int? ?? 0;
   799→
   800→      if (totalMB > 0 && usedMB > totalMB * 0.7) {
   801→        _isUnderPressure = true;
   802→      }
   803→
   804→      return _isUnderPressure;
   805→    } catch (e, st) {
   806→      AppLogger.warning('MemoryManager: pressure check failed', error: e, stackTrace: st);
   807→      return false;
   808→    }
   809→  }
   810→
   811→  /// Clears caches and invokes low-memory callbacks.
   812→  ///
   813→  /// Call this when the OS signals a memory warning or when [isMemoryPressure]
   814→  /// returns `true`.
   815→  Future<void> clearCaches() async {
   816→    AppLogger.info('MemoryManager: clearing caches due to memory pressure');
   817→    _isUnderPressure = true;
   818→
   819→    // Notify registered callbacks
   820→    for (final callback in _lowMemoryCallbacks) {
   821→      try {
   822→        callback();
   823→      } catch (e, st) {
   824→        AppLogger.error(
   825→          'MemoryManager: low-memory callback threw',
   826→          error: e,
   827→          stackTrace: st,
   828→        );
   829→      }
   830→    }
   831→
   832→    // Evict images from the Flutter image cache
   833→    try {
   834→      PaintingBinding.instance.imageCache.clear();
   835→      PaintingBinding.instance.imageCache.clearLiveImages();
   836→    } catch (e, st) {
   837→      AppLogger.warning('MemoryManager: image cache clear failed', error: e, stackTrace: st);
   838→    }
   839→
   840→    // Dispose registered resources
   841→    disposeAll();
   842→
   843→    _isUnderPressure = false;
   844→  }
   845→
   846→  /// Registers a [callback] to be invoked when a low-memory event is
   847→  /// detected.
   848→  void onLowMemory(VoidCallback callback) {
   849→    _lowMemoryCallbacks.add(callback);
   850→  }
   851→
   852→  /// Registers a [Disposable] resource that will be freed on memory pressure.
   853→  void registerDisposable(Disposable disposable) {
   854→    _disposables.add(disposable);
   855→  }
   856→
   857→  /// Disposes all registered disposable resources and clears the list.
   858→  void disposeAll() {
   859→    for (final disposable in _disposables) {
   860→      try {
   861→        disposable.dispose();
   862→      } catch (e, st) {
   863→        AppLogger.error(
   864→          'MemoryManager: failed to dispose ${disposable.runtimeType}',
   865→          error: e,
   866→          stackTrace: st,
   867→        );
   868→      }
   869→    }
   870→    _disposables.clear();
   871→  }
   872→}
   873→
   874→/// Interface for resources that can be disposed by [MemoryManager].
   875→abstract class Disposable {
   876→  /// Releases resources held by this object.
   877→  void dispose();
   878→}

   881→// DATA COMPRESSOR
   882→// ═══════════════════════════════════════════════════════════════════════════════
   883→
   884→/// Compresses and decompresses data for efficient storage and transfer.
   885→///
   886→/// Uses gzip encoding via `dart:io` [GZipCodec]. On web platforms (where
   887→/// `dart:io` is unavailable), methods return uncompressed base64 as a
   888→/// graceful fallback.
   889→class DataCompressor {
   890→  /// Compresses [data] using gzip and returns a base64-encoded string.
   891→  Future<String> compressString(String data) async {
   892→    try {
   893→      if (kIsWeb) {
   894→        // No dart:io on web — return base64 of raw UTF-8 bytes.
   895→        return base64Encode(utf8.encode(data));
   896→      }
   897→      final bytes = utf8.encode(data);
   898→      final compressed = gzip.encode(bytes);
   899→      return base64Encode(compressed);
   900→    } catch (e, st) {
   901→      AppLogger.error('DataCompressor: compressString failed', error: e, stackTrace: st);
   902→      rethrow;
   903→    }
   904→  }
   905→
   906→  /// Decompresses a base64-encoded gzip string back to the original string.
   907→  Future<String> decompressString(String compressed) async {
   908→    try {
   909→      final bytes = base64Decode(compressed);
   910→      if (kIsWeb) {
   911→        return utf8.decode(bytes);
   912→      }
   913→      final decompressed = gzip.decode(bytes);
   914→      return utf8.decode(decompressed);
   915→    } catch (e, st) {
   916→      AppLogger.error('DataCompressor: decompressString failed', error: e, stackTrace: st);
   917→      rethrow;
   918→    }
   919→  }
   920→
   921→  /// Compresses a JSON map and returns a base64-encoded string.
   922→  Future<String> compressJson(Map<String, dynamic> data) async {
   923→    return compressString(jsonEncode(data));
   924→  }
   925→
   926→  /// Decompresses a base64-encoded gzip string back to a JSON map.
   927→  Future<Map<String, dynamic>> decompressJson(String compressed) async {
   928→    final jsonString = await decompressString(compressed);
   929→    return jsonDecode(jsonString) as Map<String, dynamic>;
   930→  }
   931→
   932→  /// Estimates the compression ratio for [data] without actually compressing.
   933→  ///
   934→  /// Returns a value between 0.0 and 1.0 where lower values mean better
   935→  /// compression. Uses a quick heuristic based on character diversity.
   936→  double estimateCompressionRatio(String data) {
   937→    if (data.isEmpty) return 1.0;
   938→
   939→    // Quick heuristic: measure unique character ratio.
   940→    // Highly repetitive strings compress well (low ratio).
   941→    final uniqueChars = <int>{};
   942→    for (final codeUnit in data.codeUnits) {
   943→      uniqueChars.add(codeUnit);
   944→      // Stop sampling after 1024 characters for speed.
   945→      if (uniqueChars.length >= 1024) break;
   946→    }
   947→
   948→    final diversityRatio = uniqueChars.length / data.length.clamp(1, 1024);
   949→
   950→    // JSON / structured text typically compresses to ~15-30% of original.
   951→    // High diversity → poor compression (~80-95%), low diversity → good (~10-30%).
   952→    final estimatedRatio = 0.1 + (diversityRatio * 0.85);
   953→    return estimatedRatio.clamp(0.1, 1.0);
   954→  }
   955→}

   958→// CACHE STRATEGY
   959→// ═══════════════════════════════════════════════════════════════════════════════

   961→/// Defines the caching strategy for a resource type.
   962→///
   963→/// Each strategy determines the order of preference between network and
   964→/// cache sources when fetching data.
   965→enum CacheStrategy {
   966→  /// Always fetch from the network; never use cache.
   967→  networkOnly('Network Only — Always fetch from the network, bypass cache.'),
   968→
   969→  /// Try the network first; fall back to cache on failure.
   970→  networkFirst('Network First — Try network, fall back to cache on error.'),
   971→
   972→  /// Try the cache first; fetch from network only on cache miss.
   973→  cacheFirst('Cache First — Serve from cache if available, else network.'),
   974→
   975→  /// Only use cache; never hit the network.
   976→  cacheOnly('Cache Only — Only use cached data, never fetch from network.'),
   977→
   978→  /// Return stale cache immediately while refreshing in the background.
   979→  staleWhileRevalidate(
   980→    'Stale While Revalidate — Serve stale cache immediately, '
   981→    'then refresh from network in the background.',
   982→  );
   983→
   984→  const CacheStrategy(this.description);
   985→
   986→  /// Human-readable description of this strategy.
   987→  final String description;
   988→}

   991→// CACHE POLICY
   992→// ═══════════════════════════════════════════════════════════════════════════════
   993→
   994→/// Defines the caching behaviour for a specific resource type.
   995→///
   996→/// Combines a [CacheStrategy] with time-to-live, entry limits, and stale
   997→/// time to give fine-grained control over how each type of data is cached.
   998→class CachePolicy {
   999→  const CachePolicy({
  1000→    required this.strategy,
  1001→    required this.ttl,
  1002→    required this.maxEntries,
  1003→    required this.staleTime,
  1004→  });
  1005→
  1006→  /// The caching strategy to use.
  1007→  final CacheStrategy strategy;
  1008→
  1009→  /// How long cached data is considered fresh.
  1010→  final Duration ttl;
  1011→
  1012→  /// Maximum number of entries to keep in cache for this policy.
  1013→  final int maxEntries;
  1014→
  1015→  /// Duration after which cached data is considered stale (but still usable
  1016→  /// for [CacheStrategy.staleWhileRevalidate]).
  1017→  final Duration staleTime;
  1018→
  1019→  // ── Predefined Policies ────────────────────────────────────────────────
  1020→
  1021→  /// Policy for API data: network first, 5-minute TTL, 100 entries.
  1022→  static const apiData = CachePolicy(
  1023→    strategy: CacheStrategy.networkFirst,
  1024→    ttl: Duration(minutes: 5),
  1025→    maxEntries: 100,
  1026→    staleTime: Duration(minutes: 3),
  1027→  );
  1028→
  1029→  /// Policy for images: cache first, 7-day TTL, 500 entries.
  1030→  static const images = CachePolicy(
  1031→    strategy: CacheStrategy.cacheFirst,
  1032→    ttl: Duration(days: 7),
  1033→    maxEntries: 500,
  1034→    staleTime: Duration(days: 5),
  1035→  );
  1036→
  1037→  /// Policy for user data: network first, 1-minute TTL, 50 entries.
  1038→  static const userData = CachePolicy(
  1039→    strategy: CacheStrategy.networkFirst,
  1040→    ttl: Duration(minutes: 1),
  1041→    maxEntries: 50,
  1042→    staleTime: Duration(seconds: 30),
  1043→  );
  1044→
  1045→  /// Policy for questions: stale-while-revalidate, 1-hour TTL, 1000 entries.
  1046→  static const questions = CachePolicy(
  1047→    strategy: CacheStrategy.staleWhileRevalidate,
  1048→    ttl: Duration(hours: 1),
  1049→    maxEntries: 1000,
  1050→    staleTime: Duration(minutes: 30),
  1051→  );
  1052→
  1053→  /// Policy for resources: cache first, 24-hour TTL, 200 entries.
  1054→  static const resources = CachePolicy(
  1055→    strategy: CacheStrategy.cacheFirst,
  1056→    ttl: Duration(hours: 24),
  1057→    maxEntries: 200,
  1058→    staleTime: Duration(hours: 12),
  1059→  );
  1060→
  1061→  /// Whether a cache entry created at [cachedAt] is still fresh.
  1062→  bool isFresh(DateTime cachedAt) {
  1063→    return DateTime.now().difference(cachedAt) < ttl;
  1064→  }
  1065→
  1066→  /// Whether a cache entry created at [cachedAt] is stale but still usable.
  1067→  bool isStale(DateTime cachedAt) {
  1068→    final age = DateTime.now().difference(cachedAt);
  1069→    return age >= staleTime && age < ttl;
  1070→  }
  1071→
  1072→  /// Whether a cache entry created at [cachedAt] has expired.
  1073→  bool isExpired(DateTime cachedAt) {
  1074→    return DateTime.now().difference(cachedAt) >= ttl;
  1075→  }
  1076→
  1077→  @override
  1078→  String toString() => 'CachePolicy('
  1079→      'strategy: $strategy, '
  1080→      'ttl: $ttl, '
  1081→      'maxEntries: $maxEntries, '
  1082→      'staleTime: $staleTime'
  1083→      ')';
  1084→}

  1090→/// Simple performance monitoring utility.
  1091→///
  1092→/// Tracks operation durations and arbitrary metrics. Designed for development
  1093→/// and production diagnostics — all timing data is logged via [AppLogger].
  1094→class PerformanceMonitor {
  1095→  PerformanceMonitor();
  1096→
  1097→  /// Active timers keyed by name.
  1098→  final Map<String, Stopwatch> _timers = {};
  1099→
  1100→  /// Completed durations keyed by operation name.
  1101→  final Map<String, Duration> _durations = {};
  1102→
  1103→  /// Arbitrary metric values keyed by name.
  1104→  final Map<String, dynamic> _metrics = {};
  1105→
  1106→  /// Starts a named timer.
  1107→  ///
  1108→  /// If a timer with [key] is already running it is restarted.
  1109→  void startTimer(String key) {
  1110→    _timers[key] = Stopwatch()..start();
  1111→  }
  1112→
  1113→  /// Stops the timer for [key] and returns the elapsed [Duration].
  1114→  ///
  1115→  /// The duration is stored internally for later retrieval via
  1116→  /// [getSlowOperations] and logged via [AppLogger].
  1117→  Duration endTimer(String key) {
  1118→    final stopwatch = _timers.remove(key);
  1119→    if (stopwatch == null) {
  1120→      AppLogger.warning('PerformanceMonitor: no timer found for key=$key');
  1121→      return Duration.zero;
  1122→    }
  1123→
  1124→    stopwatch.stop();
  1125→    final duration = stopwatch.elapsed;
  1126→    _durations[key] = duration;
  1127→    AppLogger.debug('PerformanceMonitor: $key completed in ${duration.inMilliseconds}ms');
  1128→    return duration;
  1129→  }
  1130→
  1131→  /// Records an arbitrary metric value under [key].
  1132→  void recordMetric(String key, dynamic value) {
  1133→    _metrics[key] = value;
  1134→  }
  1135→
  1136→  /// Returns a snapshot of all recorded metrics and durations.
  1137→  Map<String, dynamic> getMetrics() {
  1138→    return {
  1139→      'durations': Map<String, int>.fromEntries(
  1140→        _durations.entries.map((e) => MapEntry(e.key, e.value.inMilliseconds)),
  1141→      ),
  1142→      'metrics': Map<String, dynamic>.from(_metrics),
  1143→      'activeTimers': _timers.keys.toList(),
  1144→    };
  1145→  }
  1146→
  1147→  /// Returns operations that took longer than [threshold].
  1148→  Map<String, Duration> getSlowOperations({
  1149→    Duration threshold = const Duration(seconds: 1),
  1150→  }) {
  1151→    return Map.fromEntries(
  1152→      _durations.entries.where((entry) => entry.value > threshold),
  1153→    );
  1154→  }
  1155→
  1156→  /// Clears all recorded metrics and durations.
  1157→  void clear() {
  1158→    _timers.clear();
  1159→    _durations.clear();
  1160→    _metrics.clear();
  1161→  }
  1162→}
