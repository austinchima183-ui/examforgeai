    30→String _roleLabel(String role) {
    31→  for (final entry in _kRoleOptions) {
    32→    if (entry.key == role) return entry.value;
    33→  }
    34→  return role;
    35→}
    36→
    37→Color _roleColor(String role) {
    38→  switch (role) {
    39→    case 'super_admin':
    40→      return AppColors.error;
    41→    case 'school_admin':
    42→      return AppColors.info;
    43→    case 'teacher':
    44→      return AppColors.success;
    45→    case 'student':
    46→      return AppColors.seed;
    47→    case 'parent':
    48→      return AppColors.warning;
    49→    default:
    50→      return Colors.grey;
    51→  }
    52→}

    55→  switch (role) {
    56→    case 'super_admin':
    57→      return Icons.admin_panel_settings_outlined;
    58→    case 'school_admin':
    59→      return Icons.domain_outlined;
    60→    case 'teacher':

    71→String _formatLastLogin(DateTime? lastLoginAt) {
    72→  if (lastLoginAt == null) return 'Never';
    73→  final now = DateTime.now();
    74→  final diff = now.difference(lastLoginAt);
    75→  if (diff.inMinutes < 1) return 'Just now';
    76→  if (diff.inHours < 1) return '${diff.inMinutes}m ago';
    77→  if (diff.inDays < 1) return '${diff.inHours}h ago';
    78→  if (diff.inDays < 30) return '${diff.inDays}d ago';
    79→  return '${lastLoginAt.day}/${lastLoginAt.month}/${lastLoginAt.year}';
    80→}

   638→    final cs = Theme.of(context).colorScheme;
   639→    final roleClr = _roleColor(user.role);

   641→    return Card(
   642→      elevation: Spacings.elevationSm,
