   201→   201→  Future<GenerationRequestModel> createGenerationRequest(
   202→   202→    Map<String, dynamic> requestData,
   203→   203→  ) async {
   204→   204→    try {
   205→   205→      final response = await _supabase
   206→   206→          .from(_generationRequestsTable)
   207→   207→          .insert(requestData)
   208→   208→          .select();
   209→   209→
   210→   210→      if (response.isEmpty) {
   211→   211→        throw const ServerException(
   212→   212→          message: 'Generation request creation returned no data.',

   216→   216→
   217→   217→      AppLogger.info('Generation request created: ${response.first['id']}');
   218→   218→      return GenerationRequestModel.fromJson(response.first);
   219→   219→    } on sb.PostgrestException catch (e) {
   220→   220→      throw _mapPostgrestException(e);
   221→   221→    } on ServerException {
   222→   222→      rethrow;
   223→   223→    } catch (e) {
   224→   224→      AppLogger.error('Unexpected createGenerationRequest error', error: e);
   225→   225→      throw const ServerException(
   226→   226→        message: 'Failed to create generation request.',
   227→   227→        statusCode: 500,
   228→   228→      );
   229→   229→    }

   291→   291→  @override
   292→   292→  Future<List<GenerationRequestModel>> getGenerationRequests(
   293→   293→    Map<String, dynamic> filters,
   294→   294→  ) async {
   295→   295→    try {
   296→   296→      var query = _supabase
   297→   297→          .from(_generationRequestsTable)
   298→   298→          .select()
   299→   299→          .order('created_at', ascending: false);
   300→   300→
   301→   301→      // Apply filters
   302→   302→      if (filters.containsKey('school_id')) {
   303→   303→        query = query.eq('school_id', filters['school_id'] as String);
   304→   304→      }
   305→   305→      if (filters.containsKey('status')) {
   306→   306→        query = query.eq('status', filters['status'] as String);
   307→   307→      }
   308→   308→      if (filters.containsKey('requested_by')) {
   309→   309→        query = query.eq('requested_by', filters['requested_by'] as String);
   310→   310→      }
   311→   311→
   312→   312→      // Pagination
   313→   313→      final page = filters['page'] as int? ?? 1;
   314→   314→      final perPage = filters['per_page'] as int? ?? filters['perPage'] as int? ?? 20;
   315→   315→      final from = (page - 1) * perPage;
   316→   316→      query = query.range(from, from + perPage - 1);
   317→   317→
   318→   318→      final response = await query;
   319→   319→      return response
   320→   320→          .map<GenerationRequestModel>(
   321→   321→              (json) => GenerationRequestModel.fromJson(json))
   322→   322→          .toList();
   323→   323→    } on sb.PostgrestException catch (e) {
   324→   324→      throw _mapPostgrestException(e);
   325→   325→    } catch (e) {
   326→   326→      AppLogger.error('Unexpected getGenerationRequests error', error: e);
   327→   327→      throw const ServerException(
   328→   328→        message: 'Failed to get generation requests.',
   329→   329→        statusCode: 500,
   330→   330→      );
   331→   331→    }
