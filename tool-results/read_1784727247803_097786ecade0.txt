   181→  // ── Exception mapping helper ───────────────────────────────────────────
   182→  Never _mapPostgrestException(sb.PostgrestException e) {
   183→    AppLogger.error('Postgrest error: ${e.message}', error: e);
   184→    switch (e.code) {
   185→      case 'PGRST116':
   186→        throw NotFoundException(e.message);
   187→      case '23505':
   188→        throw ServerException(
   189→          message: 'A record with this data already exists.',
   190→          statusCode: 409,
   191→        );
   192→      case '23503':
   193→        throw ServerException(
   194→          message: 'Referenced record not found.',
   195→          statusCode: 404,
   196→        );
   197→      case '42501':
   198→        throw ForbiddenException('You do not have permission for this action.');
   199→      default:
   200→        throw ServerException(
   201→          message: e.message,
   202→          statusCode: int.tryParse(e.code ?? '') ?? 500,
   203→        );
   204→    }
   205→  }

   306→  @override
   307→  Future<List<LessonPlanModel>> getLessonPlans(
   308→    Map<String, dynamic> filters,
   309→  ) async {
   310→    try {
   311→      AppLogger.info('Fetching lesson plans with filters: $filters');
   312→      var query = _supabase.from(_lessonPlansTable).select();
   313→
   314→      // Search filter
   315→      if (filters['searchQuery'] != null) {
   316→        query = query.ilike('title', '%${filters['searchQuery']}%');
   317→      }
   318→
   319→      // Subject filter
   320→      if (filters['subject_id'] != null) {
   321→        query = query.eq('subject_id', filters['subject_id']);
   322→      }
   323→
   324→      // Class filter
   325→      if (filters['class_id'] != null) {
   326→        query = query.eq('class_id', filters['class_id']);
   327→      }
   328→
   329→      // Published filter
   330→      if (filters['is_published'] != null) {
   331→        query = query.eq('is_published', filters['is_published']);
   332→      }
   333→
   334→      // Archived filter
   335→      if (filters['is_archived'] != null) {
   336→        query = query.eq('is_archived', filters['is_archived']);
   337→      }
   338→
   339→      // Tags overlap filter
   340→      if (filters['tags'] != null) {
   341→        query = query.overlaps('tags', filters['tags']);
   342→      }
   343→
   344→      final page = filters['page'] as int? ?? 1;
   345→      final perPage = filters['perPage'] as int? ?? 20;
   346→      final response = await query.order('created_at', ascending: false).range((page - 1) * perPage, page * perPage - 1);
   347→      AppLogger.info('Fetched ${response.length} lesson plans');
   348→      return response.map(LessonPlanModel.fromJson).toList();
   349→    } on sb.PostgrestException catch (e) {
   350→      _mapPostgrestException(e);
   351→    } catch (e) {
   352→      AppLogger.error('Failed to fetch lesson plans', error: e);
   353→      throw ServerException(message: e.toString(), statusCode: 500);
   354→    }
   355→  }
