   207→// NAVIGATOR KEYS

   209→
   210→/// Root navigator key — owns the top-level navigation stack.
   211→final _rootNavigatorKey = GlobalKey<NavigatorState>();
   212→
   213→/// Shell navigator key — owns the nested navigation stack inside the
   214→/// [DashboardShell]. This allows the shell (sidebar / bottom nav) to
   215→/// remain mounted while inner pages transition.
   216→final _shellNavigatorKey = GlobalKey<NavigatorState>();

   218→// ═══════════════════════════════════════════════════════════════════════
   219→// APP ROUTER PROVIDER
   220→// ═══════════════════════════════════════════════════════════════════════
   221→
   222→/// Riverpod provider for the application's [GoRouter].
   223→///
   224→/// Watches [authStateProvider] so that the router automatically
   225→/// re-evaluates its [redirect] callback whenever the authentication
   226→/// state changes (sign-in, sign-out, token refresh).
   227→///
   228→/// Usage in `main.dart`:
   229→/// ```dart
   230→/// final router = ref.read(appRouterProvider);
   231→/// MaterialApp.router(routerConfig: router);
   232→/// ```
   233→final appRouterProvider = Provider<GoRouter>((ref) {
   234→  // Watch auth state so the router re-evaluates redirects on change.
   235→  final authState = ref.watch(authStateProvider);
   236→
   237→  // Also watch onboarding state and role so those redirects stay live.
   238→  ref.watch(onboardingCompleteProvider);
   239→  ref.watch(userRoleProvider);
   240→
   241→  return GoRouter(
   242→    navigatorKey: _rootNavigatorKey,
   243→    initialLocation: RouteNames.splash,
   244→    debugLogDiagnostics: true,
   245→
   246→    // ─── Global Redirect ──────────────────────────────────────────
   247→    redirect: (context, state) {
   248→      final currentPath = state.matchedLocation;
   249→      final isAuthenticated = authState.when(
   250→        data: (authData) => authData.session != null,
   251→        loading: () {
   252→          // While loading, only allow splash. Everything else redirects
   253→          // to splash so we don't flash the login page briefly.
   254→          if (currentPath != RouteNames.splash) {
   255→            return RouteNames.splash;
   256→          }
   257→          return null;
   258→        },
   259→        error: (_, __) => false,
   260→      );
   261→
   262→      // ── Auth Guard ────────────────────────────────────────────
   263→      final authRedirect = AuthGuard.evaluate(
   264→        isAuthenticated: isAuthenticated,
   265→        currentPath: currentPath,
   266→      );
   267→      if (authRedirect != null) return authRedirect;
   268→
   269→      // ── Onboarding Guard ─────────────────────────────────────
   270→      final onboardingAsync = ref.read(onboardingCompleteProvider);
   271→      final isOnboardingComplete = onboardingAsync.when(
   272→        data: (complete) => complete,
   273→        loading: () => true, // Assume complete while loading to avoid flicker.
   274→        error: (_, __) => true,
   275→      );
   276→
   277→      final onboardingRedirect = OnboardingGuard.evaluate(
   278→        isAuthenticated: isAuthenticated,
   279→        isOnboardingComplete: isOnboardingComplete,
   280→        currentPath: currentPath,
   281→      );
   282→      if (onboardingRedirect != null) return onboardingRedirect;
   283→
   284→      // ── Role-Based Guard ─────────────────────────────────────
   285→      final roleAsync = ref.read(userRoleProvider);
   286→      final userRole = roleAsync.when(
   287→        data: UserRole.fromString,
   288→        loading: () => null,
   289→        error: (_, __) => null,
   290→      );
   291→
   292→      final roleRedirect = RoleBasedGuard.evaluate(
   293→        userRole: userRole,
   294→        currentPath: currentPath,
   295→      );
   296→      if (roleRedirect != null) return roleRedirect;
   297→
   298→      // No redirect needed — proceed as requested.
   299→      return null;
   300→    },
   301→
   302→    // ─── Route Tree ───────────────────────────────────────────────
   303→    routes: [
   304→      // ── Splash ────────────────────────────────────────────────
   305→      GoRoute(
   306→        path: RouteNames.splash,
   307→        name: 'splash',
   308→        builder: (context, state) => const SplashPage(),
   309→      ),
   310→
   311→      // ── Onboarding ───────────────────────────────────────────
   312→      GoRoute(
   313→        path: RouteNames.onboarding,
   314→        name: 'onboarding',
   315→        builder: (context, state) => const OnboardingPage(),
   316→      ),
   317→
   318→      // ── Auth Routes ──────────────────────────────────────────
   319→      GoRoute(
   320→        path: RouteNames.login,
   321→        name: 'login',
   322→        builder: (context, state) => const LoginPage(),
   323→      ),
   324→      GoRoute(
   325→        path: RouteNames.register,
   326→        name: 'register',
   327→        builder: (context, state) => const RegisterPage(),
   328→      ),
   329→      GoRoute(
   330→        path: RouteNames.forgotPassword,
   331→        name: 'forgotPassword',
   332→        builder: (context, state) => const ForgotPasswordPage(),
   333→      ),
   334→      GoRoute(
   335→        path: RouteNames.verifyEmail,
   336→        name: 'verifyEmail',
   337→        builder: (context, state) => const VerifyEmailPage(),
   338→      ),
   339→      GoRoute(
   340→        path: RouteNames.resetPassword,
   341→        name: 'resetPassword',
   342→        builder: (context, state) => const ResetPasswordPage(),
   343→      ),
   344→
   345→      // ── Protected Shell Route ─────────────────────────────────
   346→      // All routes inside this shell share the [DashboardShell]
   347→      // wrapper which provides the sidebar / bottom nav and app bar.
   348→      ShellRoute(
   349→        navigatorKey: _shellNavigatorKey,
   350→        builder: (context, state, child) => DashboardShell(child: child),
   351→        routes: [
   352→          // Dashboard with role-based sub-routes.
   353→          GoRoute(
   354→            path: RouteNames.dashboard,
   355→            name: 'dashboard',
   356→            builder: (context, state) => const DashboardRedirector(),
   357→            routes: [
   358→              GoRoute(
   359→                path: 'teacher',
   360→                name: 'teacherDashboard',
   361→                builder: (context, state) => const TeacherDashboardPage(),
   362→              ),
   363→              GoRoute(
   364→                path: 'student',
   365→                name: 'studentDashboard',

  1594→/// Shown when the user navigates to a route that does not exist in
  1595→/// the route tree.
  1596→///
  1597→/// Provides a friendly message and a button to navigate back to the
  1598→/// dashboard (or login if unauthenticated).

  1600→  const NotFoundPage({
  1601→    required this.error,
  1602→    this.path,
  1603→    super.key,
  1604→  });
  1605→
  1606→  /// The [GoException] from GoRouter describing the error.
  1607→  final GoException? error;
