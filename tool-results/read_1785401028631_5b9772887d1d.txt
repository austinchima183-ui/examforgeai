     7→-- Helper function for safe enum creation
     8→CREATE OR REPLACE FUNCTION public.create_enum_if_not_exists(enum_name TEXT, enum_values TEXT[])
     9→RETURNS VOID AS $$
    10→BEGIN
    11→  IF NOT EXISTS (SELECT 1 FROM pg_type WHERE typname = enum_name) THEN
    12→    EXECUTE format('CREATE TYPE public.%I AS ENUM (%s)', enum_name,
    13→      (SELECT string_agg(quote_literal(v), ',') FROM unnest(enum_values) AS v));
    14→  END IF;
    15→END;
    16→$$ LANGUAGE plpgsql;

   979→CREATE OR REPLACE FUNCTION public.get_exam_readiness(p_user_id UUID, p_exam_body_id UUID DEFAULT NULL, p_subject_id UUID DEFAULT NULL)
   980→RETURNS JSONB AS $$
   981→DECLARE
   982→  v_result JSONB;
   983→BEGIN
   984→  SELECT jsonb_build_object(
   985→    'user_id', p_user_id,
   986→    'assessments', (SELECT COALESCE(jsonb_agg(row_to_json(ra.*)), '[]'::jsonb)
   987→      FROM public.readiness_assessments ra
   988→      WHERE ra.user_id = p_user_id
   989→        AND (p_exam_body_id IS NULL OR ra.exam_body_id = p_exam_body_id)
   990→        AND (p_subject_id IS NULL OR ra.subject_id = p_subject_id)
   991→      ORDER BY ra.assessed_at DESC LIMIT 10),
   992→    'overall_readiness', (SELECT ROUND(AVG(readiness_score), 2)
   993→      FROM public.readiness_assessments
   994→      WHERE user_id = p_user_id
   995→        AND (p_exam_body_id IS NULL OR exam_body_id = p_exam_body_id)),
   996→    'weak_areas', (SELECT COALESCE(jsonb_agg(DISTINCT jsonb_array_elements(weak_topics)), '[]'::jsonb)
   997→      FROM public.readiness_assessments
   998→      WHERE user_id = p_user_id AND (p_exam_body_id IS NULL OR exam_body_id = p_exam_body_id)),
   999→    'study_plans', (SELECT COUNT(*) FROM public.study_plans WHERE user_id = p_user_id AND is_active = true),
  1000→    'total_study_minutes', (SELECT COALESCE(SUM(total_study_minutes), 0) FROM public.study_plans WHERE user_id = p_user_id)
  1001→  ) INTO v_result;
  1002→  RETURN v_result;
  1003→END;
  1004→$$ LANGUAGE plpgsql STABLE SECURITY DEFINER;

  1062→CREATE OR REPLACE FUNCTION public.get_analytics_summary(
  1063→  p_school_id UUID DEFAULT NULL,
  1064→  p_start_date DATE DEFAULT NULL,
  1065→  p_end_date DATE DEFAULT NULL
  1066→)
  1067→RETURNS JSONB AS $$
  1068→DECLARE
  1069→  v_result JSONB;
  1070→BEGIN
  1071→  SELECT jsonb_build_object(
  1072→    'total_users', (SELECT COUNT(DISTINCT user_id) FROM public.analytics_events WHERE (p_school_id IS NULL OR school_id = p_school_id) AND (p_start_date IS NULL OR timestamp >= p_start_date) AND (p_end_date IS NULL OR timestamp <= p_end_date)),
  1073→    'total_events', (SELECT COUNT(*) FROM public.analytics_events WHERE (p_school_id IS NULL OR school_id = p_school_id) AND (p_start_date IS NULL OR timestamp >= p_start_date) AND (p_end_date IS NULL OR timestamp <= p_end_date)),
  1074→    'event_types', (SELECT jsonb_object_agg(event_type, cnt) FROM (SELECT event_type, COUNT(*) as cnt FROM public.analytics_events WHERE (p_school_id IS NULL OR school_id = p_school_id) AND (p_start_date IS NULL OR timestamp >= p_start_date) AND (p_end_date IS NULL OR timestamp <= p_end_date) GROUP BY event_type) t),

  1076→    'monthly_active_users', (SELECT COUNT(DISTINCT user_id) FROM public.analytics_events WHERE timestamp >= date_trunc('month', CURRENT_DATE) AND (p_school_id IS NULL OR school_id = p_school_id))
  1077→  ) INTO v_result;
  1078→  RETURN v_result;

  1082→CREATE OR REPLACE FUNCTION public.get_module_subscriptions(p_school_id UUID)
