    14→    14→/// Controls how strictly the AI should grade subjective answers.
    15→    15→///
    16→    16→/// - [lenient]: Favors the student; partial credit is given generously,
    17→    17→///   minor omissions are overlooked.
    18→    18→/// - [standard]: Balanced grading; follows the marking scheme closely
    19→    19→///   while allowing reasonable interpretation.
    20→    20→/// - [strict]: Rigorous grading; demands precise, complete answers that
    21→    21→///   closely match the marking scheme with no leniency for omissions.
    22→    22→enum GradingStrictness {
    23→    23→  lenient(
    24→    24→    value: 'lenient',

    26→    26→    description: 'Favors the student; partial credit is given generously.',
    27→    27→  ),
    28→    28→  standard(
    29→    29→    value: 'standard',
    30→    30→    label: 'Standard',
    31→    31→    description: 'Balanced grading following the marking scheme closely.',
    32→    32→  ),
    33→    33→  strict(
    34→    34→    value: 'strict',
    35→    35→    label: 'Strict',
    36→    36→    description: 'Rigorous grading demanding precise, complete answers.',
    37→    37→  );

    67→    67→    required this.strengths,

    76→    76→

   167→   167→/// Tracks the progress of a batch grading operation.
   168→   168→class BatchGradingProgress {
   169→   169→  const BatchGradingProgress({
   170→   170→    required this.batchId,
   171→   171→    required this.totalAnswers,
   172→   172→    this.completedAnswers = 0,
   173→   173→    this.failedAnswers = 0,
   174→   174→    this.results = const [],
   175→   175→    this.currentQuestionIndex,
   176→   176→    this.status = BatchGradingStatus.pending,
   177→   177→    this.error,
   178→   178→  });
   179→   179→
   180→   180→  /// Unique identifier for this batch operation.
   181→   181→  final String batchId;
   182→   182→
   183→   183→  /// Total number of answers in the batch.
   184→   184→  final int totalAnswers;
   185→   185→
   186→   186→  /// Number of answers that have been graded successfully.
   187→   187→  final int completedAnswers;
   188→   188→
   189→   189→  /// Number of answers that failed to grade.
   190→   190→  final int failedAnswers;
   191→   191→
   192→   192→  /// Results collected so far (populated as each answer is graded).
   193→   193→  final List<EssayGradingResult> results;
   194→   194→
   195→   195→  /// Index of the answer currently being graded (0-based).
   196→   196→  final int? currentQuestionIndex;
   197→   197→
   198→   198→  /// Current status of the batch operation.
   199→   199→  final BatchGradingStatus status;
   200→   200→
   201→   201→  /// Error message if the entire batch failed.
   202→   202→  final String? error;
   203→   203→
   204→   204→  /// Progress as a fraction (0.0–1.0).
   205→   205→  double get progress {
   206→   206→    if (totalAnswers == 0) return 0.0;
   207→   207→    return ((completedAnswers + failedAnswers) / totalAnswers)
   208→   208→        .clamp(0.0, 1.0);
   209→   209→  }
   210→   210→
   211→   211→  /// Whether the batch has finished (completed or failed).
   212→   212→  bool get isDone =>
   213→   213→      status == BatchGradingStatus.completed ||
   214→   214→      status == BatchGradingStatus.failed;
   215→   215→
   216→   216→  /// Total tokens consumed across all graded answers so far.
   217→   217→  int get totalInputTokens =>
   218→   218→      results.fold(0, (sum, r) => sum + r.inputTokens);
   219→   219→
   220→   220→  /// Total output tokens consumed across all graded answers so far.
   221→   221→  int get totalOutputTokens =>
   222→   222→      results.fold(0, (sum, r) => sum + r.outputTokens);
   223→   223→
   224→   224→  BatchGradingProgress copyWith({
   225→   225→    String? batchId,
   226→   226→    int? totalAnswers,
   227→   227→    int? completedAnswers,
   228→   228→    int? failedAnswers,
   229→   229→    List<EssayGradingResult>? results,
   230→   230→    int? currentQuestionIndex,
   231→   231→    BatchGradingStatus? status,
   232→   232→    String? error,
   233→   233→  }) {
   234→   234→    return BatchGradingProgress(
   235→   235→      batchId: batchId ?? this.batchId,
   236→   236→      totalAnswers: totalAnswers ?? this.totalAnswers,
   237→   237→      completedAnswers: completedAnswers ?? this.completedAnswers,
   238→   238→      failedAnswers: failedAnswers ?? this.failedAnswers,
   239→   239→      results: results ?? this.results,
   240→   240→      currentQuestionIndex:
   241→   241→          currentQuestionIndex ?? this.currentQuestionIndex,
   242→   242→      status: status ?? this.status,
   243→   243→      error: error ?? this.error,
   244→   244→    );
   245→   245→  }
   246→   246→
   247→   247→  @override
   248→   248→  String toString() => 'BatchGradingProgress(batch: $batchId, '
   249→   249→      '${completedAnswers + failedAnswers}/$totalAnswers, '
   250→   250→      'progress: ${(progress * 100).toStringAsFixed(1)}%, '
   251→   251→      'status: $status)';
   252→   252→}

   320→   320→/// Orchestrates AI-assisted grading of subjective questions (essays,
   321→   321→/// short answers, case studies, practical responses).
   322→   322→///
   323→   323→/// This service leverages the existing AI infrastructure ([AiService] for
   324→   324→/// provider access and [PromptEngine] for prompt construction) to evaluate
   325→   325→/// student answers against a marking scheme and produce structured grading
   326→   326→/// results with scores, feedback, and rubric breakdowns.
   327→   327→///
   328→   328→/// Usage:
   329→   329→/// ```dart
   330→   330→/// final gradingService = AiGradingService(
   331→   331→///   aiService: aiService,
   332→   332→///   promptEngine: promptEngine,
   333→   333→/// );
   334→   334→///
   335→   335→/// final result = await gradingService.gradeEssayAnswer(
   336→   336→///   questionText: 'Explain the causes of World War I...',
   337→   337→///   studentAnswer: 'The main causes include...',
   338→   338→///   markingScheme: 'Look for: alliance systems (3 marks), ...',
   339→   339→///   maxMarks: 20,
   340→   340→/// );
   341→   341→///
   342→   342→/// if (result.isSuccess) {
   343→   343→///   print('Score: ${result.suggestedScore}/${result.maxPossible}');
   344→   344→///   print('Strengths: ${result.strengths}');
   345→   345→/// }
   346→   346→/// ```
   347→   347→class AiGradingService {
   348→   348→  AiGradingService({
   349→   349→    required AiService aiService,
   350→   350→    required PromptEngine promptEngine,
   351→   351→    this.defaultStrictness = GradingStrictness.standard,
   352→   352→    this.defaultTemperature = 0.3,
   353→   353→    this.defaultMaxTokens = 2048,
   354→   354→    this.rateLimitDelay = const Duration(milliseconds: 500),
   355→   355→  })  : _aiService = aiService,
   356→   356→        _promptEngine = promptEngine;
   357→   357→
   358→   358→  final AiService _aiService;
   359→   359→  final PromptEngine _promptEngine;
   360→   360→
   361→   361→  /// Default grading strictness when not specified per-answer.
   362→   362→  final GradingStrictness defaultStrictness;
   363→   363→
   364→   364→  /// Temperature for grading completions. Lower values produce more
   365→   365→  /// consistent grading. Defaults to 0.3.
   366→   366→  final double defaultTemperature;
   367→   367→
   368→   368→  /// Maximum tokens for the AI response.
   369→   369→  final int defaultMaxTokens;
   370→   370→
   371→   371→  /// Delay between batch grading calls to respect rate limits.
   372→   372→  final Duration rateLimitDelay;
   373→   373→
   374→   374→  // ─── Monotonic ID counter ─────────────────────────────────────────
   375→   375→  static int _counter = 0;
   376→   376→
   377→   377→  // ═══════════════════════════════════════════════════════════════════
   378→   378→  // GradeEssayAnswer
   379→   379→  // ═══════════════════════════════════════════════════════════════════
   380→   380→
   381→   381→  /// Grade a single essay / subjective answer using AI.
   382→   382→  ///
   383→   383→  /// [questionText] – The full question prompt.
   384→   384→  /// [studentAnswer] – The student's written response.
   385→   385→  /// [markingScheme] – The official marking scheme or rubric.
   386→   386→  /// [maxMarks] – Maximum marks for this question.
   387→   387→  /// [strictness] – Grading strictness. Defaults to the service's
   388→   388→  ///   [defaultStrictness] when null.
   389→   389→  /// [providerOverride] – Optional AI provider override.
   390→   390→  /// [questionType] – Type of question (defaults to "essay").
   391→   391→  /// [subject] – Optional subject context.
   392→   392→  /// [gradeLevel] – Optional grade / class level.
   393→   393→  /// [bloomLevel] – Optional Bloom's taxonomy level.
   394→   394→  ///
   395→   395→  /// Returns an [EssayGradingResult] with the suggested score, feedback,
   396→   396→  /// and token usage. On failure the result's [EssayGradingResult.error]
   397→   397→  /// field will be set.
   398→   398→  Future<EssayGradingResult> gradeEssayAnswer({
   399→   399→    required String questionText,
   400→   400→    required String studentAnswer,
   401→   401→    required String markingScheme,
   402→   402→    required double maxMarks,
   403→   403→    GradingStrictness? strictness,
   404→   404→    AiProvider? providerOverride,
   405→   405→    String questionType = 'essay',
   406→   406→    String? subject,
   407→   407→    String? gradeLevel,
   408→   408→    String? bloomLevel,
   409→   409→  }) async {
   410→   410→    final stopwatch = Stopwatch()..start();
   411→   411→
   412→   412→    try {
   413→   413→      // Validate inputs
   414→   414→      if (questionText.trim().isEmpty) {
   415→   415→        stopwatch.stop();
   416→   416→        return EssayGradingResult(
   417→   417→          suggestedScore: 0,
   418→   418→          maxPossible: maxMarks,
   419→   419→          confidenceScore: 0,
   420→   420→          explanation: 'Grading skipped: question text is empty.',
   421→   421→          strengths: [],
   422→   422→          weaknesses: [],
   423→   423→          suggestions: [],
   424→   424→          gradingRubric: {},
   425→   425→          inputTokens: 0,
   426→   426→          outputTokens: 0,
   427→   427→          processingTimeMs: stopwatch.elapsed.inMilliseconds,
   428→   428→          error: 'Question text is empty',
   429→   429→        );
   430→   430→      }
   431→   431→
   432→   432→      if (studentAnswer.trim().isEmpty) {
   433→   433→        stopwatch.stop();
   434→   434→        return EssayGradingResult(
   435→   435→          suggestedScore: 0,
   436→   436→          maxPossible: maxMarks,
   437→   437→          confidenceScore: 1.0,
   438→   438→          explanation: 'Student submitted an empty answer; score is zero.',
   439→   439→          strengths: [],
   440→   440→          weaknesses: ['No answer was provided.'],
   441→   441→          suggestions: ['Attempt all questions even with partial knowledge.'],
   442→   442→          gradingRubric: {},
   443→   443→          inputTokens: 0,
   444→   444→          outputTokens: 0,
   445→   445→          processingTimeMs: stopwatch.elapsed.inMilliseconds,
   446→   446→        );
   447→   447→      }
   448→   448→
   449→   449→      // Resolve strictness: use the provided value or fall back to the
   450→   450→      // service default (Dart does not allow instance fields as default
   451→   451→      // parameter values).
   452→   452→      final effectiveStrictness = strictness ?? defaultStrictness;
   453→   453→
   454→   454→      // 1. Build the grading prompt
   455→   455→      final promptData = buildGradingPrompt(
   456→   456→        questionText: questionText,
   457→   457→        studentAnswer: studentAnswer,
   458→   458→        markingScheme: markingScheme,
   459→   459→        maxMarks: maxMarks,
   460→   460→        strictness: effectiveStrictness,
   461→   461→        questionType: questionType,
   462→   462→        subject: subject,
   463→   463→        gradeLevel: gradeLevel,
   464→   464→        bloomLevel: bloomLevel,
   465→   465→      );
   466→   466→
   467→   467→      // 2. Build completion request
   468→   468→      final completionRequest = AiCompletionRequest(
   469→   469→        systemPrompt: promptData.systemPrompt,
   470→   470→        userPrompt: promptData.userPrompt,
   471→   471→        temperature: defaultTemperature,
   472→   472→        maxTokens: defaultMaxTokens,
   473→   473→        jsonMode: true,
   474→   474→      );
   475→   475→
   476→   476→      // 3. Obtain a provider and call the AI
   477→   477→      final providerType = providerOverride ?? _aiService.defaultProvider;
   478→   478→      final provider = _getProvider(providerType);
   479→   479→
   480→   480→      if (provider == null) {
   481→   481→        stopwatch.stop();
   482→   482→        return EssayGradingResult(
   483→   483→          suggestedScore: 0,
   484→   484→          maxPossible: maxMarks,
   485→   485→          confidenceScore: 0,
   486→   486→          explanation: 'AI provider $providerType is not available.',
   487→   487→          strengths: [],
   488→   488→          weaknesses: [],
   489→   489→          suggestions: [],
   490→   490→          gradingRubric: {},
   491→   491→          inputTokens: 0,
   492→   492→          outputTokens: 0,
   493→   493→          processingTimeMs: stopwatch.elapsed.inMilliseconds,
   494→   494→          error: 'Provider $providerType is not available',
   495→   495→        );
   496→   496→      }
   497→   497→
   498→   498→      AppLogger.info('Starting AI grading for question '
   499→   499→          '(${effectiveStrictness.label}, max=$maxMarks)');
   500→   500→
   501→   501→      final result = await provider.complete(completionRequest);
   502→   502→
   503→   503→      // 4. Parse the AI response
   504→   504→      final parsed = _parseJsonFromResponse(result.content);
   505→   505→
   506→   506→      if (parsed == null) {
   507→   507→        stopwatch.stop();
   508→   508→        AppLogger.warning('Failed to parse AI grading response as JSON');
   509→   509→        return EssayGradingResult(
   510→   510→          suggestedScore: 0,
   511→   511→          maxPossible: maxMarks,
   512→   512→          confidenceScore: 0,
   513→   513→          explanation: 'Failed to parse the AI grading response.',
   514→   514→          strengths: [],

   598→   598→  // ═══════════════════════════════════════════════════════════════════
   599→   599→
   600→   600→  /// Grade multiple essay answers in sequence, respecting rate limits.
   601→   601→  ///
   602→   602→  /// Emits [BatchGradingProgress] updates after each answer is graded,
   603→   603→  /// allowing callers to track progress in real time.
   604→   604→  ///
   605→   605→  /// [answers] – The list of answers to grade.
   606→   606→  /// [providerOverride] – Optional AI provider override for all answers.
   607→   607→  /// [cancelToken] – An optional [Completer] that, when completed, will
   608→   608→  ///   cancel the batch operation after the current answer finishes.
   609→   609→  ///
   610→   610→  /// Returns a [Stream] of progress updates. The final update will have
   611→   611→  /// [BatchGradingProgress.status] set to [BatchGradingStatus.completed]
   612→   612→  /// or [BatchGradingStatus.failed].
   613→   613→  Stream<BatchGradingProgress> batchGradeEssayAnswers({
   614→   614→    required List<EssayAnswerInput> answers,
   615→   615→    AiProvider? providerOverride,
   616→   616→    Completer<void>? cancelToken,
   617→   617→  }) async* {
   618→   618→    final batchId = _generateBatchId();
   619→   619→    var progress = BatchGradingProgress(
   620→   620→      batchId: batchId,
   621→   621→      totalAnswers: answers.length,
   622→   622→      status: BatchGradingStatus.inProgress,
   623→   623→    );
   624→   624→
   625→   625→    yield progress;
   626→   626→
   627→   627→    final results = <EssayGradingResult>[];
   628→   628→
   629→   629→    for (int i = 0; i < answers.length; i++) {
   630→   630→      // Check for cancellation
   631→   631→      if (cancelToken != null && cancelToken.isCompleted) {
   632→   632→        AppLogger.info('Batch grading cancelled at answer $i/$${answers.length}');
   633→   633→        yield progress = progress.copyWith(
   634→   634→          currentQuestionIndex: i,
   635→   635→          status: BatchGradingStatus.cancelled,
   636→   636→          results: List.unmodifiable(results),
   637→   637→        );
   638→   638→        return;
   639→   639→      }
   640→   640→
   641→   641→      final answer = answers[i];
   642→   642→
   643→   643→      yield progress = progress.copyWith(
   644→   644→        currentQuestionIndex: i,
   645→   645→      );
   646→   646→
   647→   647→      AppLogger.info('Batch grading: answer ${i + 1}/${answers.length} '
   648→   648→          '(batch $batchId)');
   649→   649→
   650→   650→      // Grade the individual answer
   651→   651→      final result = await gradeEssayAnswer(
   652→   652→        questionText: answer.questionText,
   653→   653→        studentAnswer: answer.studentAnswer,
   654→   654→        markingScheme: answer.markingScheme,
   655→   655→        maxMarks: answer.maxMarks,
   656→   656→        strictness: answer.strictness,
   657→   657→        providerOverride: providerOverride,
   658→   658→        questionType: answer.questionType,
   659→   659→        subject: answer.subject,
   660→   660→        gradeLevel: answer.gradeLevel,
   661→   661→        bloomLevel: answer.bloomLevel,
   662→   662→      );
   663→   663→
   664→   664→      results.add(result);
   665→   665→
   666→   666→      if (result.isSuccess) {
   667→   667→        yield progress = progress.copyWith(
   668→   668→          completedAnswers: progress.completedAnswers + 1,
   669→   669→          results: List.unmodifiable(results),
   670→   670→          currentQuestionIndex: i,
   671→   671→        );
   672→   672→      } else {
   673→   673→        yield progress = progress.copyWith(
   674→   674→          failedAnswers: progress.failedAnswers + 1,
   675→   675→          results: List.unmodifiable(results),
   676→   676→          currentQuestionIndex: i,
   677→   677→        );
   678→   678→      }
   679→   679→
   680→   680→      // Respect rate limits – wait before the next call (but not after
   681→   681→      // the last one).
   682→   682→      if (i < answers.length - 1) {
   683→   683→        await Future<void>.delayed(rateLimitDelay);
   684→   684→      }
   685→   685→    }
   686→   686→
   687→   687→    // Final progress update
   688→   688→    yield progress = progress.copyWith(
   689→   689→      status: BatchGradingStatus.completed,
   690→   690→      currentQuestionIndex: null,
   691→   691→    );
   692→   692→
   693→   693→    AppLogger.info('Batch grading complete: batch $batchId, '
   694→   694→        '${progress.completedAnswers} succeeded, '
   695→   695→        '${progress.failedAnswers} failed');
   696→   696→  }

   934→   934→      return EssayGradingResult(
   935→   935→        suggestedScore: 0,
   936→   936→        maxPossible: maxPossible,
   937→   937→        confidenceScore: 0,

   939→   939→        strengths: [],
   940→   940→        weaknesses: [],
   941→   941→        suggestions: [],
   942→   942→        gradingRubric: {},
   943→   943→        inputTokens: inputTokens,

   967→   967→  /// Returns an [TokenEstimate] with input, expected output, and total

   969→   969→  TokenEstimate estimateTokens({
   970→   970→    required String questionText,
   971→   971→    required String studentAnswer,

   973→   973→    GradingStrictness strictness = GradingStrictness.standard,

   983→   983→

  1033→  1033→  /// Extract JSON from AI responses.
  1034→  1034→  ///
  1035→  1035→  /// AI models sometimes wrap JSON in markdown code blocks (e.g.
  1036→  1036→  /// ```json ... ```) or prepend conversational text. This helper
  1037→  1037→  /// attempts multiple strategies to extract valid JSON:
  1038→  1038→  ///
  1039→  1039→  /// 1. Direct `jsonDecode` of the entire content
  1040→  1040→  /// 2. Extraction from markdown code blocks
  1041→  1041→  /// 3. Finding the outermost `{ ... }` braces
  1042→  1042→  Map<String, dynamic>? _parseJsonFromResponse(String content) {
  1043→  1043→    // Strategy 1: Direct parse
  1044→  1044→    try {
  1045→  1045→      final decoded = jsonDecode(content);
  1046→  1046→      if (decoded is Map<String, dynamic>) return decoded;
  1047→  1047→      if (decoded is Map) return Map<String, dynamic>.from(decoded);
  1048→  1048→    } on FormatException {
  1049→  1049→      // Fall through to next strategy
  1050→  1050→    }
  1051→  1051→
  1052→  1052→    // Strategy 2: Extract from markdown code blocks
  1053→  1053→    final jsonBlockRegex = RegExp(r'```(?:json)?\s*([\s\S]*?)```');
  1054→  1054→    final blockMatch = jsonBlockRegex.firstMatch(content);
  1055→  1055→    if (blockMatch != null) {
  1056→  1056→      try {
  1057→  1057→        final decoded = jsonDecode(blockMatch.group(1)!.trim());
  1058→  1058→        if (decoded is Map<String, dynamic>) return decoded;
  1059→  1059→        if (decoded is Map) return Map<String, dynamic>.from(decoded);
  1060→  1060→      } on FormatException {
  1061→  1061→        // Fall through to next strategy
  1062→  1062→      }
  1063→  1063→    }
  1064→  1064→
  1065→  1065→    // Strategy 3: Find outermost { ... } braces
  1066→  1066→    final braceStart = content.indexOf('{');
  1067→  1067→    final braceEnd = content.lastIndexOf('}');
  1068→  1068→    if (braceStart != -1 && braceEnd > braceStart) {
  1069→  1069→      try {
  1070→  1070→        final decoded =
  1071→  1071→            jsonDecode(content.substring(braceStart, braceEnd + 1));
  1072→  1072→        if (decoded is Map<String, dynamic>) return decoded;
  1073→  1073→        if (decoded is Map) return Map<String, dynamic>.from(decoded);
  1074→  1074→      } on FormatException {
  1075→  1075→        // All strategies failed
  1076→  1076→      }
  1077→  1077→    }
  1078→  1078→
  1079→  1079→    return null;
  1080→  1080→  }
  1081→  1081→
  1082→  1082→  /// Parse a dynamic value into a `List<String>`.
  1083→  1083→  ///
  1084→  1084→  /// Handles both `List<String>` and `List<dynamic>` from JSON.
  1085→  1085→  List<String> _parseStringList(dynamic value) {

  1087→  1087→      return value.map((e) => e?.toString() ?? '').toList();
  1088→  1088→    }

  1090→  1090→  }
