   368→   368→// ─── 3. StarRating ────────────────────────────────────────────────────────────
   369→   369→
   370→   370→/// Displays a row of 5 stars (filled, half-filled, or outlined) based on
   371→   371→/// [rating] with an optional review count.
   372→   372→///
   373→   373→/// ```dart
   374→   374→/// StarRating(rating: 4.3, totalReviews: 128, showCount: true)
   375→   375→/// ```
   376→   376→class StarRating extends StatelessWidget {
   377→   377→  const StarRating({
   378→   378→    super.key,
   379→   379→    required this.rating,
   380→   380→    this.totalReviews,
   381→   381→    this.starSize = Spacings.mdIcon,
   382→   382→    this.showCount = true,
   383→   383→  });
   384→   384→
   385→   385→  final double rating;
   386→   386→  final int? totalReviews;
   387→   387→  final double starSize;
   388→   388→  final bool showCount;
   389→   389→
   390→   390→  @override
   391→   391→  Widget build(BuildContext context) {
   392→   392→    final cs = context.colorScheme;
   393→   393→    final tt = context.textTheme;
   394→   394→
   395→   395→    return Row(
   396→   396→      mainAxisSize: MainAxisSize.min,
   397→   397→      children: [
   398→   398→        // 5 stars
   399→   399→        for (int i = 1; i <= 5; i++)
   400→   400→          Icon(
   401→   401→            _starIcon(i),
   402→   402→            size: starSize,
   403→   403→            color: AppColors.warning,
   404→   404→          ),
   405→   405→        // Count
   406→   406→        if (showCount && totalReviews != null) ...[
   407→   407→          const SizedBox(width: Spacings.xs),
   408→   408→          Text(
   409→   409→            '($totalReviews)',
   410→   410→            style: tt.labelSmall?.copyWith(
   411→   411→              color: cs.onSurfaceVariant,
   412→   412→            ),
   413→   413→          ),
   414→   414→        ],
   415→   415→      ],
   416→   416→    );
   417→   417→  }
   418→   418→
   419→   419→  IconData _starIcon(int position) {
   420→   420→    if (rating >= position) return Icons.star_rounded;
   421→   421→    if (rating >= position - 0.5) return Icons.star_half_rounded;
   422→   422→    return Icons.star_outline_rounded;
   423→   423→  }
   424→   424→}
