     6→     6→// VALIDATION ENGINE
     7→     7→// ═══════════════════════════════════════════════════════════════════════
     8→     8→
     9→     9→/// Automated validation engine for generated questions.
    10→    10→///
    11→    11→/// Runs a comprehensive suite of checks on each question, covering
    12→    12→/// grammar, answer accuracy, ambiguity, clarity, reading level,
    13→    13→/// curriculum alignment, difficulty consistency, and duplicate detection.
    14→    14→///
    15→    15→/// Each check returns a [ValidationResultEntity] (or null if the check
    16→    16→/// passes). The main [validate] method runs all checks and returns
    17→    17→/// the aggregated list.
    18→    18→class ValidationEngine {
    19→    19→  // ─── Common grammatical error patterns ─────────────────────────────
    20→    20→
    21→    21→  static const List<_GrammarPattern> _grammarPatterns = [
    22→    22→    _GrammarPattern(
    23→    23→      pattern: r'\btheir\s+is\b',
    24→    24→      message: 'Possible grammar error: "their is" — did you mean "there is"?',
    25→    25→    ),
    26→    26→    _GrammarPattern(
    27→    27→      pattern: r'\byour\s+is\b',
    28→    28→      message: 'Possible grammar error: "your is" — did you mean "you\'re is" or "there is"?',
    29→    29→    ),
    30→    30→    _GrammarPattern(
    31→    31→      pattern: r'\bits\s+a\b',
    32→    32→      message: 'Check: "its a" — did you mean "it\'s a"?',
    33→    33→    ),
    34→    34→    _GrammarPattern(
    35→    35→      pattern: r'\bshould\s+of\b',
    36→    36→      message: 'Grammar error: "should of" — should be "should have"',
    37→    37→    ),
    38→    38→    _GrammarPattern(
    39→    39→      pattern: r'\bcould\s+of\b',
    40→    40→      message: 'Grammar error: "could of" — should be "could have"',
    41→    41→    ),
    42→    42→    _GrammarPattern(
    43→    43→      pattern: r'\bwould\s+of\b',
    44→    44→      message: 'Grammar error: "would of" — should be "would have"',
    45→    45→    ),
    46→    46→    _GrammarPattern(
    47→    47→      pattern: r'\beffect\s+on\b',
    48→    48→      message: 'Check: "effect on" — verify if "affect" is the intended word',
    49→    49→    ),
    50→    50→    _GrammarPattern(
    51→    51→      pattern: r'\bless\s+\w+s\b',
    52→    52→      message: 'Check: "less" with plural noun — consider "fewer"',
    53→    53→    ),
    54→    54→    _GrammarPattern(
    55→    55→      pattern: r'\bwhich\s+\w+\s+\w+\s+,\s*which\b',
    56→    56→      message: 'Possible ambiguous "which" reference',
    57→    57→    ),
    58→    58→  ];
    59→    59→
    60→    60→  // ─── Ambiguity indicators ──────────────────────────────────────────
    61→    61→
    62→    62→  static const List<String> _ambiguityIndicators = [
    63→    63→    'it refers to',
    64→    64→    'this could mean',
    65→    65→    'that is to say',
    66→    66→    'in other words',
    67→    67→    'some people say',
    68→    68→    'sometimes',
    69→    69→    'maybe',
    70→    70→    'perhaps',
    71→    71→    'might be considered',
    72→    72→  ];
    73→    73→
    74→    74→  // ─── Main validation entry point ──────────────────────────────────
    75→    75→
    76→    76→  /// Validate a generated question by running all automated checks.
    77→    77→  ///
    78→    78→  /// Returns a list of [ValidationResultEntity] instances. An empty list
    79→    79→  /// means the question passed all checks.
    80→    80→  Future<List<ValidationResultEntity>> validate(
    81→    81→    GeneratedQuestionEntity question,
    82→    82→  ) async {
    83→    83→    final results = <ValidationResultEntity>[];
    84→    84→
    85→    85→    // Run each check and collect non-null results
    86→    86→    final grammar = checkGrammar(question.content);
    87→    87→    if (grammar != null) results.add(grammar);
    88→    88→
    89→    89→    // Check answer options for grammar too
    90→    90→    for (final option in question.answerOptions) {
    91→    91→      final optionText = option['text'] as String? ?? option['content'] as String? ?? '';
    92→    92→      if (optionText.isNotEmpty) {
    93→    93→        final optionGrammar = checkGrammar(optionText);
    94→    94→        if (optionGrammar != null) {
    95→    95→          results.add(optionGrammar.copyWith(
    96→    96→            validationType: 'grammar_option',
    97→    97→            message: 'Option ${option['label'] ?? ""}: ${optionGrammar.message}',
    98→    98→          ));
    99→    99→        }
   100→   100→      }
   101→   101→    }
   102→   102→
   103→   103→    final ambiguity = checkAmbiguity(question.content);
   104→   104→    if (ambiguity != null) results.add(ambiguity);
   105→   105→
   106→   106→    final clarity = checkClarity(question.content);
   107→   107→    if (clarity != null) results.add(clarity);
   108→   108→
   109→   109→    final answerAccuracy = checkAnswerAccuracy(question);
   110→   110→    if (answerAccuracy != null) results.add(answerAccuracy);
   111→   111→
   112→   112→    final difficultyConsistency = checkDifficultyConsistency(question);
   113→   113→    if (difficultyConsistency != null) results.add(difficultyConsistency);
   114→   114→
   115→   115→    final readingLevel = checkReadingLevel(
   116→   116→      question.content,
   117→   117→      question.curriculumAlignment?['class_level'] as String? ?? '',
   118→   118→    );
   119→   119→    if (readingLevel != null) results.add(readingLevel);
   120→   120→
   121→   121→    AppLogger.info('Validation complete for question ${question.id}: '
   122→   122→        '${results.length} issue(s) found');
   123→   123→
   124→   124→    return results;
   125→   125→  }
   126→   126→
   127→   127→  /// Validate with duplicate checking against existing questions.
   128→   128→  Future<List<ValidationResultEntity>> validateWithDuplicates(
   129→   129→    GeneratedQuestionEntity question,
   130→   130→    List<QuestionEntity> existingQuestions,
   131→   131→  ) async {
   132→   132→    final results = await validate(question);
   133→   133→
   134→   134→    final duplicate = await checkDuplicate(question, existingQuestions);
   135→   135→    if (duplicate != null) results.add(duplicate);
   136→   136→
   137→   137→    return results;
   138→   138→  }

   141→   141→  Future<List<ValidationResultEntity>> validateWithCurriculum(

   143→   143→    CurriculumMappingEntity? mapping,
   144→   144→  ) async {
   145→   145→    final results = await validate(question);
   146→   146→
   147→   147→    if (mapping != null) {
   148→   148→      final alignment = checkCurriculumAlignment(question, mapping);
   149→   149→      if (alignment != null) results.add(alignment);

   155→   155→  // ─── Individual Checks ────────────────────────────────────────────
   156→   156→
   157→   157→  /// Check grammar and spelling using rule-based patterns.
   158→   158→  ///
   159→   159→  /// This is a lightweight heuristic check. For production-grade
   160→   160→  /// grammar checking, consider integrating with a language tool API.
   161→   161→  ValidationResultEntity? checkGrammar(String content) {
   162→   162→    if (content.trim().isEmpty) {
   163→   163→      return ValidationResultEntity(
   164→   164→        id: '',
   165→   165→        generatedQuestionId: '',
   166→   166→        validationType: 'grammar',
   167→   167→        severity: ValidationSeverity.critical,
   168→   168→        message: 'Question content is empty',
   169→   169→        suggestion: 'Provide non-empty question content',
   170→   170→        isResolved: false,
   171→   171→        resolvedBy: null,
   172→   172→        resolvedAt: null,
   173→   173→        createdAt: DateTime.now(),
   174→   174→      );
   175→   175→    }
   176→   176→
   177→   177→    final issues = <String>[];
   178→   178→
   179→   179→    for (final pattern in _grammarPatterns) {
   180→   180→      if (RegExp(pattern.pattern, caseSensitive: false).hasMatch(content)) {
   181→   181→        issues.add(pattern.message);
   182→   182→      }
   183→   183→    }
   184→   184→
   185→   185→    // Check for double spaces
   186→   186→    if (RegExp(r'  +').hasMatch(content)) {
   187→   187→      issues.add('Text contains double spaces');
   188→   188→    }
   189→   189→
   190→   190→    // Check for missing question mark at end of direct questions
   191→   191→    final trimmedContent = content.trim();
   192→   192→    if (trimmedContent.contains('?') == false &&
   193→   193→        (trimmedContent.startsWithWh('Wh') ||
   194→   194→            trimmedContent.toLowerCase().startsWith('how') ||
   195→   195→            trimmedContent.toLowerCase().startsWith('what') ||
   196→   196→            trimmedContent.toLowerCase().startsWith('which') ||
   197→   197→            trimmedContent.toLowerCase().startsWith('who') ||
   198→   198→            trimmedContent.toLowerCase().startsWith('where') ||
   199→   199→            trimmedContent.toLowerCase().startsWith('when') ||
   200→   200→            trimmedContent.toLowerCase().startsWith('why'))) {
   201→   201→      issues.add('Question appears to be a direct question but is missing a question mark');
   202→   202→    }
   203→   203→
   204→   204→    // Check for very short content
   205→   205→    if (content.split(' ').length < 4) {
   206→   206→      issues.add('Question content is very short (less than 4 words)');
   207→   207→    }
   208→   208→
   209→   209→    if (issues.isEmpty) return null;
   210→   210→
   211→   211→    return ValidationResultEntity(
   212→   212→      id: '',
   213→   213→      generatedQuestionId: '',
   214→   214→      validationType: 'grammar',
   215→   215→      severity: issues.length > 2 ? ValidationSeverity.warning : ValidationSeverity.info,
   216→   216→      message: 'Grammar/spelling issues found: ${issues.join("; ")}',
   217→   217→      suggestion: 'Review and correct the identified grammar issues',
   218→   218→      isResolved: false,
   219→   219→      resolvedBy: null,
   220→   220→      resolvedAt: null,
   221→   221→      createdAt: DateTime.now(),
   222→   222→    );
   223→   223→  }
   224→   224→
   225→   225→  /// Check for duplicate questions by comparing content similarity.
   226→   226→  ///
   227→   227→  /// Uses a Jaccard similarity measure on word sets. Questions with
   228→   228→  /// >80% word overlap are flagged as potential duplicates.
   229→   229→  Future<ValidationResultEntity?> checkDuplicate(
   230→   230→    GeneratedQuestionEntity question,
   231→   231→    List<QuestionEntity> existingQuestions,
   232→   232→  ) async {
   233→   233→    final questionWords = _normalizeAndTokenize(question.content);
   234→   234→
   235→   235→    if (questionWords.isEmpty) return null;
   236→   236→
   237→   237→    for (final existing in existingQuestions) {
   238→   238→      final existingWords = _normalizeAndTokenize(existing.content);
   239→   239→
   240→   240→      if (existingWords.isEmpty) continue;
   241→   241→
   242→   242→      // Calculate Jaccard similarity
   243→   243→      final intersection = questionWords.intersection(existingWords).length;
   244→   244→      final union = questionWords.union(existingWords).length;
   245→   245→      final similarity = union > 0 ? intersection / union : 0.0;
   246→   246→
   247→   247→      if (similarity > 0.8) {
   248→   248→        return ValidationResultEntity(
   249→   249→          id: '',
   250→   250→          generatedQuestionId: question.id,
   251→   251→          validationType: 'duplicate',
   252→   252→          severity: ValidationSeverity.error,
   253→   253→          message: 'Potential duplicate detected (similarity: '
   254→   254→              '${(similarity * 100).toStringAsFixed(1)}%)',
   255→   255→          suggestion: 'Modify the question to make it distinct, or use the '
   256→   256→              'existing question',
   257→   257→          isResolved: false,
   258→   258→          resolvedBy: null,
   259→   259→          resolvedAt: null,
   260→   260→          createdAt: DateTime.now(),
   261→   261→        );
   262→   262→      }
   263→   263→    }
   264→   264→
   265→   265→    return null;
   266→   266→  }
   267→   267→
   268→   268→  /// Check answer accuracy: correct answer should be clearly correct,
   269→   269→  /// and distractors should be plausible but definitively wrong.
   270→   270→  ValidationResultEntity? checkAnswerAccuracy(
   271→   271→    GeneratedQuestionEntity question,
   272→   272→  ) {
   273→   273→    final options = question.answerOptions;
   274→   274→    if (options.isEmpty) return null; // Not a multiple-choice question
   275→   275→
   276→   276→    // Check that exactly one option is marked correct
   277→   277→    final correctOptions = options.where((o) {
   278→   278→      return (o['is_correct'] as bool?) ?? (o['isCorrect'] as bool?) ?? false;
   279→   279→    }).toList();
   280→   280→
   281→   281→    if (correctOptions.isEmpty) {
   282→   282→      return ValidationResultEntity(
   283→   283→        id: '',
   284→   284→        generatedQuestionId: question.id,
   285→   285→        validationType: 'answer_accuracy',
   286→   286→        severity: ValidationSeverity.critical,
   287→   287→        message: 'No correct answer is marked in the answer options',
   288→   288→        suggestion: 'Mark exactly one option as the correct answer',
   289→   289→        isResolved: false,
   290→   290→        resolvedBy: null,
   291→   291→        resolvedAt: null,
   292→   292→        createdAt: DateTime.now(),
   293→   293→      );
   294→   294→    }
   295→   295→
   296→   296→    // For single-answer MCQs, check that only one is correct
   297→   297→    if (question.questionType == QuestionType.multipleChoice &&
   298→   298→        correctOptions.length > 1) {
   299→   299→      return ValidationResultEntity(
   300→   300→        id: '',
   301→   301→        generatedQuestionId: question.id,
   302→   302→        validationType: 'answer_accuracy',
   303→   303→        severity: ValidationSeverity.error,
   304→   304→        message: 'Multiple-choice question has ${correctOptions.length} '
   305→   305→            'correct answers; expected exactly 1',
   306→   306→        suggestion: 'For MCQs, only one option should be correct. '
   307→   307→            'Use multiple_response type for multiple correct answers.',
   308→   308→        isResolved: false,
   309→   309→        resolvedBy: null,
   310→   310→        resolvedAt: null,
   311→   311→        createdAt: DateTime.now(),
   312→   312→      );
   313→   313→    }
   314→   314→
   315→   315→    // Check that option texts are not empty
   316→   316→    for (int i = 0; i < options.length; i++) {
   317→   317→      final text = options[i]['text'] as String? ??
   318→   318→          options[i]['content'] as String? ?? '';
   319→   319→      if (text.trim().isEmpty) {
   320→   320→        return ValidationResultEntity(
   321→   321→          id: '',
   322→   322→          generatedQuestionId: question.id,
   323→   323→          validationType: 'answer_accuracy',
   324→   324→          severity: ValidationSeverity.error,
   325→   325→          message: 'Answer option ${i + 1} has empty text',
   326→   326→          suggestion: 'Provide text for all answer options',
   327→   327→          isResolved: false,
   328→   328→          resolvedBy: null,
   329→   329→          resolvedAt: null,
   330→   330→          createdAt: DateTime.now(),
   331→   331→        );
   332→   332→      }
   333→   333→    }
   334→   334→
   335→   335→    // Check for duplicate option texts
   336→   336→    final optionTexts = options
   337→   337→        .map((o) => (o['text'] as String? ?? o['content'] as String? ?? '')
   338→   338→            .trim()
   339→   339→            .toLowerCase())
   340→   340→        .toList();
   341→   341→    if (optionTexts.toSet().length != optionTexts.length) {
   342→   342→      return ValidationResultEntity(
   343→   343→        id: '',
   344→   344→        generatedQuestionId: question.id,
   345→   345→        validationType: 'answer_accuracy',
   346→   346→        severity: ValidationSeverity.error,
   347→   347→        message: 'Two or more answer options have identical text',
   348→   348→        suggestion: 'Make each option distinct',
   349→   349→        isResolved: false,
   350→   350→        resolvedBy: null,
   351→   351→        resolvedAt: null,
   352→   352→        createdAt: DateTime.now(),
   353→   353→      );
   354→   354→    }
   355→   355→
   356→   356→    // Check for "All of the above" / "None of the above" patterns
   357→   357→    // These can sometimes indicate poor distractor quality
   358→   358→    for (final option in options) {
   359→   359→      final text = (option['text'] as String? ?? '').toLowerCase();
   360→   360→      final isCorrect = (option['is_correct'] as bool?) ??
   361→   361→          (option['isCorrect'] as bool?) ??
   362→   362→          false;
   363→   363→      if ((text.contains('all of the above') ||
   364→   364→              text.contains('none of the above')) &&
   365→   365→          isCorrect) {
   366→   366→        return ValidationResultEntity(
   367→   367→          id: '',
   368→   368→          generatedQuestionId: question.id,
   369→   369→          validationType: 'answer_accuracy',
   370→   370→          severity: ValidationSeverity.warning,
   371→   371→          message: '"All/None of the above" is marked as correct — '
   372→   372→              'this can be a poor testing strategy',
   373→   373→          suggestion:
   374→   374→              'Consider using specific correct answers instead of catch-all options',
   375→   375→          isResolved: false,
   376→   376→          resolvedBy: null,
   377→   377→          resolvedAt: null,
   378→   378→          createdAt: DateTime.now(),
   379→   379→        );
   380→   380→      }
   381→   381→    }
   382→   382→
   383→   383→    // Check that option lengths are not wildly different
   384→   384→    // (could give away the answer)
   385→   385→    if (options.length >= 3) {
   386→   386→      final lengths = options
   387→   387→          .map((o) =>
   388→   388→              (o['text'] as String? ?? o['content'] as String? ?? '').length)
   389→   389→          .toList();
   390→   390→      final avgLength = lengths.reduce((a, b) => a + b) / lengths.length;
   391→   391→      for (int i = 0; i < lengths.length; i++) {
   392→   392→        if (avgLength > 0 && (lengths[i] / avgLength) > 2.0) {
   393→   393→          return ValidationResultEntity(
   394→   394→            id: '',
   395→   395→            generatedQuestionId: question.id,
   396→   396→            validationType: 'answer_accuracy',
   397→   397→            severity: ValidationSeverity.warning,
   398→   398→            message: 'Option ${i + 1} is significantly longer than other '
   399→   399→                'options, which may give away the answer',
   400→   400→            suggestion:
   401→   401→                'Make all options roughly the same length',
   402→   402→            isResolved: false,
   403→   403→            resolvedBy: null,
   404→   404→            resolvedAt: null,
   405→   405→            createdAt: DateTime.now(),
   406→   406→          );
   407→   407→        }
   408→   408→      }
   409→   409→    }
   410→   410→
   411→   411→    return null;
   412→   412→  }

   414→   414→  /// Check for ambiguous wording in the question content.
   415→   415→  ValidationResultEntity? checkAmbiguity(String content) {
   416→   416→    final lowerContent = content.toLowerCase();
   417→   417→
   418→   418→    // Check for ambiguity indicators
   419→   419→    final foundIndicators = <String>[];
   420→   420→    for (final indicator in _ambiguityIndicators) {
   421→   421→      if (lowerContent.contains(indicator.toLowerCase())) {
   422→   422→        foundIndicators.add(indicator);
   423→   423→      }
   424→   424→    }
   425→   425→
   426→   426→    // Check for pronouns without clear antecedents
   427→   427→    final pronounPattern = RegExp(r'\b(it|they|this|that|these|those)\b',
   428→   428→        caseSensitive: false);
   429→   429→    final pronounMatches = pronounPattern.allMatches(content);
   430→   430→    if (pronounMatches.length > 2) {
   431→   431→      foundIndicators.add(
   432→   432→          'Excessive use of pronouns (it/they/this/that) — '
   433→   433→          'antecedents may be unclear');
   434→   434→    }
   435→   435→
   436→   436→    // Check for "not" and "except" which can create confusion
   437→   437→    if (lowerContent.contains(' not ') && lowerContent.contains(' except ')) {
   438→   438→      foundIndicators.add(
   439→   439→          'Double negative pattern detected (both "not" and "except")');
   440→   440→    }
   441→   441→
   442→   442→    // Check for "which of the following" without a specific criteria
   443→   443→    if (lowerContent.contains('which of the following') &&
   444→   444→        !lowerContent.contains('is true') &&
   445→   445→        !lowerContent.contains('is correct') &&
   446→   446→        !lowerContent.contains('is false') &&
   447→   447→        !lowerContent.contains('are true') &&
   448→   448→        !lowerContent.contains('are correct')) {
   449→   449→      foundIndicators.add(
   450→   450→          '"Which of the following" without clear selection criteria');
   451→   451→    }
   452→   452→
   453→   453→    if (foundIndicators.isEmpty) return null;
   454→   454→
   455→   455→    return ValidationResultEntity(
   456→   456→      id: '',
   457→   457→      generatedQuestionId: '',
   458→   458→      validationType: 'ambiguity',
   459→   459→      severity: ValidationSeverity.warning,
   460→   460→      message: 'Potential ambiguity detected: ${foundIndicators.join("; ")}',
   461→   461→      suggestion: 'Rewrite the question to eliminate ambiguous wording',
   462→   462→      isResolved: false,
   463→   463→      resolvedBy: null,
   464→   464→      resolvedAt: null,
   465→   465→      createdAt: DateTime.now(),
   466→   466→    );
   467→   467→  }

   469→   469→  /// Check clarity of the question content.
   470→   470→  ValidationResultEntity? checkClarity(String content) {
   471→   471→    final issues = <String>[];
   472→   472→
   473→   473→    // Check for overly long sentences
   474→   474→    final sentences = content.split(RegExp(r'[.!?]'));
   475→   475→    for (final sentence in sentences) {
   476→   476→      final wordCount = sentence.trim().split(RegExp(r'\s+')).length;
   477→   477→      if (wordCount > 40) {
   478→   478→        issues.add('Sentence with $wordCount words is too long '
   479→   479→            '(recommended max: 40 words)');
   480→   480→      }
   481→   481→    }
   482→   482→
   483→   483→    // Check for nested clauses (multiple commas might indicate complexity)
   484→   484→    final commaCount = ','.allMatches(content).length;
   485→   485→    if (commaCount > 4) {
   486→   486→      issues.add('High number of commas ($commaCount) suggests '
   487→   487→          'complex sentence structure');
   488→   488→    }
   489→   489→
   490→   490→    // Check for parenthetical expressions that might confuse
   491→   491→    final parenCount = '('.allMatches(content).length;
   492→   492→    if (parenCount > 2) {
   493→   493→      issues.add('Multiple parenthetical expressions may reduce clarity');
   494→   494→    }
   495→   495→
   496→   496→    // Check overall word count
   497→   497→    final totalWords = content.split(RegExp(r'\s+')).length;
   498→   498→    if (totalWords > 100) {
   499→   499→      issues.add('Question is very long ($totalWords words) — '
   500→   500→          'consider simplifying');
   501→   501→    }
   502→   502→
   503→   503→    // Check for double negatives
   504→   504→    if (RegExp(r'\bnot\s+\w+\s+not\b', caseSensitive: false)
   505→   505→        .hasMatch(content)) {
   506→   506→      issues.add('Double negative detected — can confuse students');
   507→   507→    }
   508→   508→
   509→   509→    if (issues.isEmpty) return null;
   510→   510→
   511→   511→    return ValidationResultEntity(
   512→   512→      id: '',
   513→   513→      generatedQuestionId: '',
   514→   514→      validationType: 'clarity',
   515→   515→      severity: ValidationSeverity.info,
   516→   516→      message: 'Clarity concerns: ${issues.join("; ")}',
   517→   517→      suggestion: 'Simplify the question wording for better student comprehension',
   518→   518→      isResolved: false,
   519→   519→      resolvedBy: null,

   521→   521→      createdAt: DateTime.now(),

   525→   525→  /// Estimate reading level using a simplified Flesch-Kincaid approach.
   526→   526→  ValidationResultEntity? checkReadingLevel(
   527→   527→    String content,
   528→   528→    String classLevel,
   529→   529→  ) {
   530→   530→    if (content.trim().isEmpty || classLevel.isEmpty) return null;
   531→   531→
   532→   532→    // Flesch-Kincaid Grade Level estimation
   533→   533→    final words = content.split(RegExp(r'\s+'));
   534→   534→    final wordCount = words.length;
   535→   535→    if (wordCount == 0) return null;
   536→   536→
   537→   537→    // Count syllables (simplified estimation)
   538→   538→    int totalSyllables = 0;
   539→   539→    for (final word in words) {
   540→   540→      totalSyllables += _countSyllables(word);
   541→   541→    }
   542→   542→
   543→   543→    // Count sentences
   544→   544→    final sentences = content.split(RegExp(r'[.!?]'));
   545→   545→    final sentenceCount = sentences.where((s) => s.trim().isNotEmpty).length;
   546→   546→    if (sentenceCount == 0) return null;
   547→   547→
   548→   548→    // Flesch-Kincaid Grade Level formula
   549→   549→    final gradeLevel = (0.39 * (wordCount / sentenceCount)) +
   550→   550→        (11.8 * (totalSyllables / wordCount)) -
   551→   551→        15.59;
   552→   552→
   553→   553→    // Map class level to expected reading level
   554→   554→    final expectedLevel = _classLevelToReadingLevel(classLevel);
   555→   555→
   556→   556→    if (gradeLevel > expectedLevel + 2) {
   557→   557→      return ValidationResultEntity(
   558→   558→        id: '',
   559→   559→        generatedQuestionId: '',
   560→   560→        validationType: 'reading_level',
   561→   561→        severity: ValidationSeverity.warning,
   562→   562→        message: 'Reading level (grade ${gradeLevel.toStringAsFixed(1)}) '
   563→   563→            'is significantly above expected level (grade $expectedLevel) '
   564→   564→            'for class $classLevel',
   565→   565→        suggestion: 'Simplify vocabulary and sentence structure to match '
   566→   566→            'the target reading level',
   567→   567→        isResolved: false,
   568→   568→        resolvedBy: null,
   569→   569→        resolvedAt: null,
   570→   570→        createdAt: DateTime.now(),
   571→   571→      );
   572→   572→    }
   573→   573→
   574→   574→    return null;
   575→   575→  }

   685→   685→  /// Normalize text and split into a set of unique words for comparison.
   686→   686→  Set<String> _normalizeAndTokenize(String text) {
   687→   687→    return text
   688→   688→        .toLowerCase()
   689→   689→        .replaceAll(RegExp(r'[^\w\s]'), '')
   690→   690→        .split(RegExp(r'\s+'))
   691→   691→        .where((w) => w.length > 2) // Ignore very short words
   692→   692→        .toSet();
   693→   693→  }

   695→   695→  /// Count syllables in a word using a simple heuristic.
   696→   696→  int _countSyllables(String word) {
   697→   697→    final w = word.toLowerCase().replaceAll(RegExp(r'[^a-z]'), '');
   698→   698→    if (w.isEmpty) return 0;
   699→   699→    if (w.length <= 3) return 1;
   700→   700→
   701→   701→    int count = 0;
   702→   702→    final vowels = RegExp(r'[aeiouy]');
   703→   703→    bool prevVowel = false;
   704→   704→
   705→   705→    for (int i = 0; i < w.length; i++) {
   706→   706→      final isVowel = vowels.hasMatch(w[i]);
   707→   707→      if (isVowel && !prevVowel) count++;
   708→   708→      prevVowel = isVowel;
   709→   709→    }
   710→   710→
   711→   711→    // Handle silent 'e' at end
   712→   712→    if (w.endsWith('e') && count > 1) count--;
   713→   713→
   714→   714→    // Handle 'le' at end
   715→   715→    if (w.endsWith('le') && w.length > 2 && !vowels.hasMatch(w[w.length - 3])) {
   716→   716→      count++;
   717→   717→    }
   718→   718→
   719→   719→    return count.clamp(1, 20);
   720→   720→  }

   722→   722→  /// Map a class level string to an approximate reading grade level.
   723→   723→  int _classLevelToReadingLevel(String classLevel) {
   724→   724→    // Try to parse numeric class level
   725→   725→    final numericMatch = RegExp(r'(\d+)').firstMatch(classLevel);
   726→   726→    if (numericMatch != null) {
   727→   727→      final grade = int.tryParse(numericMatch.group(1)!) ?? 6;
   728→   728→      return grade + 5; // Class 6 ≈ grade 11 reading level, etc.
   729→   729→    }
   730→   730→
   731→   731→    // Map common class level names
   732→   732→    final lower = classLevel.toLowerCase();
   733→   733→    if (lower.contains('primary') || lower.contains('elementary')) return 5;
   734→   734→    if (lower.contains('junior') || lower.contains('middle')) return 8;
   735→   735→    if (lower.contains('senior') || lower.contains('secondary')) return 11;
   736→   736→    if (lower.contains('university') || lower.contains('college')) return 14;
   737→   737→
   738→   738→    return 10; // Default
   739→   739→  }

   741→   741→  /// Map Bloom's taxonomy level to an expected difficulty level.
   742→   742→  DifficultyLevel _bloomToExpectedDifficulty(BloomTaxonomy bloom) {
   743→   743→    switch (bloom) {
   744→   744→      case BloomTaxonomy.remember:
   745→   745→      case BloomTaxonomy.understand:
   746→   746→        return DifficultyLevel.easy;
   747→   747→      case BloomTaxonomy.apply:
   748→   748→        return DifficultyLevel.medium;
   749→   749→      case BloomTaxonomy.analyze:
   750→   750→        return DifficultyLevel.medium;
   751→   751→      case BloomTaxonomy.evaluate:
   752→   752→        return DifficultyLevel.hard;
   753→   753→      case BloomTaxonomy.create:
   754→   754→        return DifficultyLevel.expert;
   755→   755→    }
   756→   756→  }

   758→   758→  /// Check if two difficulty levels are compatible (adjacent).
   759→   759→  bool _isCompatibleDifficulty(DifficultyLevel actual, DifficultyLevel expected) {
   760→   760→    const order = [
   761→   761→      DifficultyLevel.easy,
   762→   762→      DifficultyLevel.medium,
   763→   763→      DifficultyLevel.hard,
   764→   764→      DifficultyLevel.expert,
   765→   765→    ];
   766→   766→    final actualIndex = order.indexOf(actual);
   767→   767→    final expectedIndex = order.indexOf(expected);
   768→   768→    return (actualIndex - expectedIndex).abs() <= 1;
   769→   769→  }
