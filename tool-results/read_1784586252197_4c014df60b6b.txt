    53→  /// Maps a [sb.PostgrestException] to the appropriate domain exception.
    54→  Never _mapPostgrestException(sb.PostgrestException e) {
    55→    AppLogger.error('Postgrest error: ${e.message}', error: e);
    56→    switch (e.code) {
    57→      case 'PGRST116':
    58→        throw NotFoundException(message: e.message);
    59→      case '23505':
    60→        throw ServerException(
    61→          message: 'A record with this data already exists.',
    62→          statusCode: 409,
    63→        );
    64→      case '23503':
    65→        throw ServerException(
    66→          message: 'Referenced record not found.',
    67→          statusCode: 404,
    68→        );
    69→      case '42501':
    70→        throw ForbiddenException(
    71→          message: 'You do not have permission for this action.',
    72→        );
    73→      default:
    74→        throw ServerException(
    75→          message: e.message,
    76→          statusCode: e.statusCode ?? 500,
    77→        );
    78→    }
    79→  }
    80→
    81→  /// Handles generic catch-all exceptions.
    82→  Never _handleGenericException(Object e, String operation) {
    83→    AppLogger.error('Failed to $operation', error: e);
    84→    if (e is sb.AuthException) {
    85→      throw UnauthorizedException(message: e.message);
    86→    }
    87→    throw ServerException(message: e.toString(), statusCode: 500);
    88→  }

   201→  /// Sends a message in an AI tutor conversation and receives an AI response.
   202→  ///
   203→  /// The flow is:
   204→  /// 1. Insert the user message into [ai_tutor_messages].
   205→  /// 2. Call the `ai-tutor-chat` Edge Function with conversation context.
   206→  /// 3. Insert the AI assistant message into [ai_tutor_messages].
   207→  /// 4. Update the conversation's [last_message] and [message_count].
   208→  /// 5. Return the AI response model.
   209→  Future<AiTutorMessageModel> sendMessage({
   210→    required String conversationId,
   211→    required String content,
   212→    String? subjectId,
   213→    String? topic,
   214→    String? curriculumType,
   215→  }) async {
   216→    try {
   217→      AppLogger.info(
   218→        'Sending message in conversation: $conversationId',
   219→      );
   220→
   221→      // 1. Insert the user message
   222→      final userMessageResponse = await _supabase
   223→          .from(_aiTutorMessagesTable)
   224→          .insert({
   225→            'conversation_id': conversationId,
   226→            'role': 'user',
   227→            'content': content,
   228→          })
   229→          .select()
   230→          .single();
   231→
   232→      AppLogger.info('User message inserted');
   233→
   234→      // 2. Call the AI tutor Edge Function for a response
   235→      final aiResponse = await _supabase.functions.invoke(
   236→        _aiTutorChatFunction,
   237→        body: {
   238→          'conversation_id': conversationId,
   239→          'message': content,
   240→          'subject_id': subjectId,
   241→          'topic': topic,
   242→          'curriculum_type': curriculumType,
   243→        },
   244→      );
   245→
   246→      if (aiResponse.status != 200) {
   247→        AppLogger.error(
   248→          'AI tutor edge function returned status: ${aiResponse.status}',
   249→        );
   250→        throw ServerException(
   251→          message: 'AI tutor service returned an error: ${aiResponse.data}',
   252→          statusCode: aiResponse.status,
   253→        );
   254→      }
   255→
   256→      final aiContent = aiResponse.data['content'] as String? ??
   257→          'I apologize, but I was unable to generate a response. Please try again.';
   258→      final aiMetadata = aiResponse.data['metadata'] as Map<String, dynamic>?;
   259→
   260→      // 3. Insert the AI assistant message
   261→      final assistantMessageResponse = await _supabase
   262→          .from(_aiTutorMessagesTable)
   263→          .insert({
   264→            'conversation_id': conversationId,
   265→            'role': 'assistant',
   266→            'content': aiContent,
   267→            'metadata': aiMetadata ?? {},
   268→          })
   269→          .select()
   270→          .single();
   271→
   272→      // 4. Update the conversation's last_message and message_count
   273→      await _supabase
   274→          .from(_aiTutorConversationsTable)
   275→          .update({
   276→            'last_message': aiContent.length > 100
   277→                ? '${aiContent.substring(0, 100)}...'
   278→                : aiContent,
   279→          })
   280→          .eq('id', conversationId);
   281→
   282→      AppLogger.info('AI response message inserted');
   283→
   284→      return AiTutorMessageModel.fromJson(assistantMessageResponse);
   285→    } on sb.PostgrestException catch (e) {
   286→      _mapPostgrestException(e);
   287→    } catch (e) {
   288→      if (e is ServerException) rethrow;
   289→      _handleGenericException(e, 'send AI tutor message');
   290→    }
   291→  }

   318→  /// Creates a new practice session and fetches random questions from
   319→  /// the question bank matching the given filters.
   320→  ///
   321→  /// The session is created in `in_progress` status. Questions are
   322→  /// selected from the [questions] table using random ordering and
   323→  /// subject/difficulty filters.
   324→  Future<PracticeSessionModel> createPracticeSession({
   325→    required String studentId,
   326→    String? schoolId,
   327→    String? subjectId,
   328→    String? topicId,
   329→    String difficulty = 'medium',
   330→    String mode = 'untimed',
   331→    int? timeLimitSec,
   332→    int questionCount = 10,
   333→  }) async {
   334→    try {
   335→      AppLogger.info(
   336→        'Creating practice session for student: $studentId',
   337→      );
   338→
   339→      // Insert the practice session
   340→      final sessionResponse = await _supabase
   341→          .from(_practiceSessionsTable)
   342→          .insert({
   343→            'student_id': studentId,
   344→            'school_id': schoolId,
   345→            'subject_id': subjectId,
   346→            'topic_id': topicId,
   347→            'difficulty': difficulty,
   348→            'mode': mode,
   349→            'time_limit_sec': timeLimitSec,
   350→            'total_questions': questionCount,
   351→            'status': 'in_progress',
   352→            'started_at': DateTime.now().toUtc().toIso8601String(),
   353→          })
   354→          .select()
   355→          .single();
   356→
   357→      final sessionId = sessionResponse['id'] as String;
   358→
   359→      // Fetch random questions from the question bank
   360→      var query = _supabase
   361→          .from(_questionsTable)
   362→          .select('id')
   363→          .eq('status', 'approved');
   364→
   365→      if (subjectId != null) {
   366→        query = query.eq('subject_id', subjectId);
   367→      }
   368→      if (difficulty != 'mixed') {
   369→        query = query.eq('difficulty', difficulty);
   370→      }
   371→
   372→      // Use limit and random ordering via Supabase
   373→      final questionIds = await query
   374→          .limit(questionCount * 3) // Fetch extra for randomness
   375→          .order('id');
   376→
   377→      // Randomly select the desired number of questions
   378→      final selectedQuestions = <Map<String, dynamic>>[];
   379→      if (questionIds.isNotEmpty) {
   380→        final shuffled = List<Map<String, dynamic>>.from(questionIds)..shuffle();
   381→        selectedQuestions.addAll(
   382→          shuffled.take(min(questionCount, shuffled.length)),
   383→        );
   384→      }
   385→
   386→      // Store the question IDs as the session's question set
   387→      // (practice_answers will be created as students answer)
   388→      // Update the total_questions to match actual available count
   389→      if (selectedQuestions.length != questionCount) {
   390→        await _supabase
   391→            .from(_practiceSessionsTable)
   392→            .update({'total_questions': selectedQuestions.length})
   393→            .eq('id', sessionId);
   394→        sessionResponse['total_questions'] = selectedQuestions.length;
   395→      }
   396→
   397→      AppLogger.info(
   398→        'Practice session created: $sessionId with ${selectedQuestions.length} questions',
   399→      );
   400→
   401→      return PracticeSessionModel.fromJson(sessionResponse);
   402→    } on sb.PostgrestException catch (e) {
   403→      _mapPostgrestException(e);
   404→    } catch (e) {
   405→      _handleGenericException(e, 'create practice session');
   406→    }
   407→  }

   537→  /// Completes a practice session.
   538→  ///
   539→  /// Sets the status to `completed`, calculates the final score
   540→  /// percentage, and records the completion timestamp.
   541→  Future<PracticeSessionModel> completePracticeSession({
   542→    required String sessionId,
   543→  }) async {
   544→    try {
   545→      AppLogger.info('Completing practice session: $sessionId');
   546→
   547→      // Fetch current session to compute score
   548→      final current = await _supabase
   549→          .from(_practiceSessionsTable)
   550→          .select('total_questions, correct_count')
   551→          .eq('id', sessionId)
   552→          .single();
   553→
   554→      final totalQ = current['total_questions'] as int? ?? 0;
   555→      final correctQ = current['correct_count'] as int? ?? 0;
   556→      final scorePct = totalQ > 0 ? (correctQ / totalQ) * 100 : 0.0;
   557→
   558→      final response = await _supabase
   559→          .from(_practiceSessionsTable)
   560→          .update({
   561→            'status': 'completed',
   562→            'score_pct': scorePct,
   563→            'completed_at': DateTime.now().toUtc().toIso8601String(),
   564→          })
   565→          .eq('id', sessionId)
   566→          .select()
   567→          .single();
   568→
   569→      AppLogger.info(
   570→        'Practice session completed: $sessionId (score: ${scorePct.toStringAsFixed(1)}%)',
   571→      );
   572→      return PracticeSessionModel.fromJson(response);
   573→    } on sb.PostgrestException catch (e) {
   574→      _mapPostgrestException(e);
   575→    } catch (e) {
   576→      _handleGenericException(e, 'complete practice session');
   577→    }
   578→  }

   953→  /// Uploads a document for AI chat processing.
   954→  ///
   955→  /// The flow is:
   956→  /// 1. Upload the file to Supabase Storage bucket `student-documents`.
   957→  /// 2. Create a `document_chats` record.
   958→  /// 3. Call the `process-document` Edge Function for text extraction
   959→  ///    (runs asynchronously).
   960→  Future<DocumentChatModel> uploadDocument({
   961→    required String studentId,
   962→    String? schoolId,
   963→    required String fileName,
   964→    required List<int> fileBytes,
   965→    required String fileFormat,
   966→    int? fileSize,
   967→  }) async {
   968→    try {
   969→      AppLogger.info(
   970→        'Uploading document for student: $studentId, file: $fileName',
   971→      );
   972→
   973→      // 1. Upload file to Supabase Storage
   974→      final storagePath =
   975→          '$studentId/${DateTime.now().millisecondsSinceEpoch}_$fileName';
   976→      await _supabase.storage
   977→          .from(_studentDocumentsBucket)
   978→          .uploadBinary(storagePath, fileBytes);
   979→
   980→      // Get the public URL for the uploaded file
   981→      final fileUrl = _supabase.storage
   982→          .from(_studentDocumentsBucket)
   983→          .getPublicUrl(storagePath);
   984→
   985→      // 2. Create a document_chats record
   986→      final response = await _supabase
   987→          .from(_documentChatsTable)
   988→          .insert({
   989→            'student_id': studentId,
   990→            'school_id': schoolId,
   991→            'file_name': fileName,
   992→            'file_url': fileUrl,
   993→            'file_size': fileSize,
   994→            'file_format': fileFormat,
   995→            'status': 'processing',
   996→          })
   997→          .select()
   998→          .single();
   999→
  1000→      final documentId = response['id'] as String;
  1001→
  1002→      // 3. Trigger the process-document edge function (async)
  1003→      _supabase.functions.invoke(
  1004→        _processDocumentFunction,
  1005→        body: {
  1006→          'document_id': documentId,
  1007→          'file_url': fileUrl,
  1008→          'file_format': fileFormat,
  1009→        },
  1010→      ).catchError((error) {
  1011→        AppLogger.error(
  1012→          'Process document edge function failed',
  1013→          error: error,
  1014→        );
  1015→      });
  1016→
  1017→      AppLogger.info('Document uploaded and processing started: $documentId');
  1018→      return DocumentChatModel.fromJson(response);
  1019→    } on sb.PostgrestException catch (e) {
  1020→      _mapPostgrestException(e);
  1021→    } on sb.StorageException catch (e) {
  1022→      AppLogger.error('Storage upload failed', error: e);
  1023→      throw ServerException(
  1024→        message: 'File upload failed: ${e.message}',
  1025→        statusCode: 500,
  1026→      );
  1027→    } catch (e) {
  1028→      if (e is ServerException) rethrow;
  1029→      _handleGenericException(e, 'upload document');
  1030→    }
  1031→  }

  1156→  /// Deletes a document, its chat messages, and the stored file.
  1157→  Future<void> deleteDocument({required String documentId}) async {
  1158→    try {
  1159→      AppLogger.info('Deleting document: $documentId');
  1160→
  1161→      // Fetch the document to get the storage path
  1162→      final doc = await _supabase
  1163→          .from(_documentChatsTable)
  1164→          .select('file_url, student_id')
  1165→          .eq('id', documentId)
  1166→          .single();
  1167→
  1168→      // Delete from database (cascade deletes messages)
  1169→      await _supabase
  1170→          .from(_documentChatsTable)
  1171→          .delete()
  1172→          .eq('id', documentId);
  1173→
  1174→      // Attempt to remove the file from storage (best-effort)
  1175→      final fileUrl = doc['file_url'] as String?;
  1176→      if (fileUrl != null) {
  1177→        try {
  1178→          final studentId = doc['student_id'] as String;
  1179→          // Extract the storage path from the URL
  1180→          final uri = Uri.parse(fileUrl);
  1181→          final pathSegments = uri.pathSegments;
  1182→          final storagePath = pathSegments.skip(2).join('/');
  1183→          await _supabase.storage
  1184→              .from(_studentDocumentsBucket)
  1185→              .remove([storagePath]);
  1186→        } catch (storageError) {
  1187→          AppLogger.warning(
  1188→            'Failed to delete document from storage (non-critical)',
  1189→            error: storageError,
  1190→          );
  1191→        }
  1192→      }
  1193→
  1194→      AppLogger.info('Document deleted successfully');
  1195→    } on sb.PostgrestException catch (e) {
  1196→      _mapPostgrestException(e);
  1197→    } catch (e) {
  1198→      _handleGenericException(e, 'delete document');

  1448→  /// Rates a flashcard using the SM-2 spaced repetition algorithm.
  1449→  ///
  1450→  /// The rating can be one of: `again`, `hard`, `good`, `easy`.
  1451→  ///
  1452→  /// **SM-2 Algorithm Implementation:**
  1453→  /// - `again`: repetitions = 0, interval = 1 min
  1454→  /// - `hard`:  easeFactor -= 0.15, interval = interval * 1.2
  1455→  /// - `good`:  interval = interval * easeFactor
  1456→  /// - `easy`:  easeFactor += 0.15, interval = interval * easeFactor * 1.3
  1457→  /// - easeFactor is clamped between 1.3 and 3.0
  1458→  /// - If repetitions == 0: interval = 1 day
  1459→  /// - If repetitions == 1: interval = 6 days
  1460→  /// - Else: interval = previousInterval * easeFactor
  1461→  Future<FlashcardModel> rateFlashcard({
  1462→    required String cardId,
  1463→    required String rating,
  1464→  }) async {
  1465→    try {
  1466→      AppLogger.info('Rating flashcard: $cardId with rating: $rating');
  1467→
  1468→      // 1. Fetch the current card state
  1469→      final current = await _supabase
  1470→          .from(_flashcardsTable)
  1471→          .select()
  1472→          .eq('id', cardId)
  1473→          .single();
  1474→
  1475→      var repetitions = current['repetitions'] as int? ?? 0;
  1476→      var easeFactor = (current['ease_factor'] as num?)?.toDouble() ?? 2.5;
  1477→      var interval = (current['interval'] as num?)?.toDouble() ?? 0.0;
  1478→      final previousInterval = interval;
  1479→
  1480→      // 2. Apply SM-2 algorithm based on rating
  1481→      switch (rating) {
  1482→        case 'again':
  1483→          repetitions = 0;
  1484→          interval = 1.0 / 1440.0; // 1 minute in days (~0.000694 days)
  1485→          // Alternatively, we can set next_review to now + 1 minute directly
  1486→          break;
  1487→        case 'hard':
  1488→          easeFactor = max(1.3, easeFactor - 0.15);
  1489→          interval = previousInterval * 1.2;
  1490→          if (interval < 1) interval = 1;
  1491→          repetitions = repetitions + 1;
  1492→          break;
  1493→        case 'good':
  1494→          if (repetitions == 0) {
  1495→            interval = 1; // 1 day
  1496→          } else if (repetitions == 1) {
  1497→            interval = 6; // 6 days
  1498→          } else {
  1499→            interval = previousInterval * easeFactor;
  1500→          }
  1501→          repetitions = repetitions + 1;
  1502→          break;
  1503→        case 'easy':
  1504→          easeFactor = min(3.0, easeFactor + 0.15);
  1505→          if (repetitions == 0) {
  1506→            interval = 1;
  1507→          } else if (repetitions == 1) {
  1508→            interval = 6;
  1509→          } else {
  1510→            interval = previousInterval * easeFactor;
  1511→          }
  1512→          interval = interval * 1.3;
  1513→          repetitions = repetitions + 1;
  1514→          break;
  1515→        default:
  1516→          throw ValidationException(
  1517→            message: 'Invalid flashcard rating: $rating',
  1518→            fieldErrors: {'rating': 'Must be one of: again, hard, good, easy'},
  1519→          );
  1520→      }
  1521→
  1522→      // 3. Clamp easeFactor
  1523→      easeFactor = easeFactor.clamp(1.3, 3.0);
  1524→
  1525→      // 4. Compute next review date
  1526→      final nextReview = DateTime.now().add(Duration(
  1527→        milliseconds: (interval * 24 * 60 * 60 * 1000).round(),
  1528→      ));
  1529→
  1530→      // 5. Persist the updated SM-2 state
  1531→      final response = await _supabase
  1532→          .from(_flashcardsTable)
  1533→          .update({
  1534→            'repetitions': repetitions,
  1535→            'ease_factor': easeFactor,
  1536→            'interval': interval,
  1537→            'next_review': nextReview.toUtc().toIso8601String(),
  1538→            'last_reviewed': DateTime.now().toUtc().toIso8601String(),
  1539→          })
  1540→          .eq('id', cardId)
  1541→          .select()
  1542→          .single();
  1543→
  1544→      AppLogger.info(
  1545→        'Flashcard rated: $cardId → next review in ${interval.toStringAsFixed(1)} days',
  1546→      );
  1547→      return FlashcardModel.fromJson(response);
  1548→    } on sb.PostgrestException catch (e) {
  1549→      _mapPostgrestException(e);
  1550→    } catch (e) {
  1551→      if (e is ValidationException) rethrow;
  1552→      _handleGenericException(e, 'rate flashcard');
  1553→    }
  1554→  }
