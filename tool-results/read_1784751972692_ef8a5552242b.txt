    54→  /// Maps a [sb.PostgrestException] to the appropriate domain exception.
    55→  Never _mapPostgrestException(sb.PostgrestException e) {
    56→    AppLogger.error('Postgrest error: ${e.message}', error: e);
    57→    switch (e.code) {
    58→      case 'PGRST116':
    59→        throw NotFoundException(message: e.message);
    60→      case '23505':
    61→        throw ServerException(
    62→          message: 'A record with this data already exists.',
    63→          statusCode: 409,
    64→        );
    65→      case '23503':
    66→        throw ServerException(
    67→          message: 'Referenced record not found.',
    68→          statusCode: 404,
    69→        );
    70→      case '42501':
    71→        throw ForbiddenException(
    72→          message: 'You do not have permission for this action.',
    73→        );
    74→      default:
    75→        throw ServerException(
    76→          message: e.message,
    77→          statusCode: e.statusCode ?? 500,
    78→        );
    79→    }
    80→  }
    81→
    82→  /// Handles generic catch-all exceptions.
    83→  Never _handleGenericException(Object e, String operation) {
    84→    AppLogger.error('Failed to $operation', error: e);
    85→    if (e is sb.AuthException) {
    86→      throw UnauthorizedException(message: e.message);
    87→    }
    88→    throw ServerException(message: e.toString(), statusCode: 500);
    89→  }

   204→  /// Sends a message in an AI tutor conversation and receives an AI response.
   205→  ///
   206→  /// The flow is:
   207→  /// 1. Insert the user message into [ai_tutor_messages].
   208→  /// 2. Call the `ai-tutor-chat` Edge Function with conversation context.
   209→  /// 3. Insert the AI assistant message into [ai_tutor_messages].
   210→  /// 4. Update the conversation's [last_message] and [message_count].
   211→  /// 5. Return the AI response model.
   212→  Future<AiTutorMessageModel> sendMessage({
   213→    required String conversationId,
   214→    required String content,
   215→    String? subjectId,
   216→    String? topic,
   217→    String? curriculumType,
   218→  }) async {
   219→    try {
   220→      AppLogger.info(
   221→        'Sending message in conversation: $conversationId',
   222→      );
   223→
   224→      // 1. Insert the user message
   225→      final userMessageResponse = await _supabase
   226→          .from(_aiTutorMessagesTable)
   227→          .insert({
   228→            'conversation_id': conversationId,
   229→            'role': 'user',
   230→            'content': content,
   231→          })
   232→          .select()
   233→          .single();
   234→
   235→      AppLogger.info('User message inserted');
   236→
   237→      // 2. Call the AI tutor Edge Function for a response
   238→      final aiResponse = await _supabase.functions.invoke(
   239→        _aiTutorChatFunction,
   240→        body: {
   241→          'conversation_id': conversationId,
   242→          'message': content,
   243→          'subject_id': subjectId,
   244→          'topic': topic,
   245→          'curriculum_type': curriculumType,
   246→        },
   247→      );
   248→
   249→      if (aiResponse.status != 200) {
   250→        AppLogger.error(
   251→          'AI tutor edge function returned status: ${aiResponse.status}',
   252→        );
   253→        throw ServerException(
   254→          message: 'AI tutor service returned an error: ${aiResponse.data}',
   255→          statusCode: aiResponse.status,
   256→        );
   257→      }
   258→
   259→      final aiContent = aiResponse.data['content'] as String? ??
   260→          'I apologize, but I was unable to generate a response. Please try again.';
   261→      final aiMetadata = aiResponse.data['metadata'] as Map<String, dynamic>?;
   262→
   263→      // 3. Insert the AI assistant message
   264→      final assistantMessageResponse = await _supabase
   265→          .from(_aiTutorMessagesTable)
   266→          .insert({
   267→            'conversation_id': conversationId,
   268→            'role': 'assistant',
   269→            'content': aiContent,
   270→            'metadata': aiMetadata ?? {},
   271→          })
   272→          .select()
   273→          .single();
   274→
   275→      // 4. Update the conversation's last_message and message_count
   276→      await _supabase
   277→          .from(_aiTutorConversationsTable)
   278→          .update({
   279→            'last_message': aiContent.length > 100
   280→                ? '${aiContent.substring(0, 100)}...'
   281→                : aiContent,
   282→          })
   283→          .eq('id', conversationId);
   284→
   285→      AppLogger.info('AI response message inserted');
   286→
   287→      return AiTutorMessageModel.fromJson(assistantMessageResponse);
   288→    } on sb.PostgrestException catch (e) {
   289→      _mapPostgrestException(e);
   290→    } catch (e) {
   291→      if (e is ServerException) rethrow;
   292→      _handleGenericException(e, 'send AI tutor message');
   293→    }
   294→  }

   321→  /// Creates a new practice session and fetches random questions from
   322→  /// the question bank matching the given filters.
   323→  ///
   324→  /// The session is created in `in_progress` status. Questions are
   325→  /// selected from the [questions] table using random ordering and
   326→  /// subject/difficulty filters.
   327→  Future<PracticeSessionModel> createPracticeSession({
   328→    required String studentId,
   329→    String? schoolId,
   330→    String? subjectId,
   331→    String? topicId,
   332→    String difficulty = 'medium',
   333→    String mode = 'untimed',
   334→    int? timeLimitSec,
   335→    int questionCount = 10,
   336→  }) async {
   337→    try {
   338→      AppLogger.info(
   339→        'Creating practice session for student: $studentId',
   340→      );
   341→
   342→      // Insert the practice session
   343→      final sessionResponse = await _supabase
   344→          .from(_practiceSessionsTable)
   345→          .insert({
   346→            'student_id': studentId,
   347→            'school_id': schoolId,
   348→            'subject_id': subjectId,
   349→            'topic_id': topicId,
   350→            'difficulty': difficulty,
   351→            'mode': mode,
   352→            'time_limit_sec': timeLimitSec,
   353→            'total_questions': questionCount,
   354→            'status': 'in_progress',
   355→            'started_at': DateTime.now().toUtc().toIso8601String(),
   356→          })
   357→          .select()
   358→          .single();
   359→
   360→      final sessionId = sessionResponse['id'] as String;
   361→
   362→      // Fetch random questions from the question bank
   363→      var query = _supabase
   364→          .from(_questionsTable)
   365→          .select('id')
   366→          .eq('status', 'approved');
   367→
   368→      if (subjectId != null) {
   369→        query = query.eq('subject_id', subjectId);
   370→      }
   371→      if (difficulty != 'mixed') {
   372→        query = query.eq('difficulty', difficulty);
   373→      }
   374→
   375→      // Use limit and random ordering via Supabase
   376→      final questionIds = await query
   377→          .limit(questionCount * 3) // Fetch extra for randomness
   378→          .order('id');
   379→
   380→      // Randomly select the desired number of questions
   381→      final selectedQuestions = <Map<String, dynamic>>[];
   382→      if (questionIds.isNotEmpty) {
   383→        final shuffled = List<Map<String, dynamic>>.from(questionIds)..shuffle();
   384→        selectedQuestions.addAll(
   385→          shuffled.take(min(questionCount, shuffled.length)),
   386→        );
   387→      }
   388→
   389→      // Store the question IDs as the session's question set
   390→      // (practice_answers will be created as students answer)
   391→      // Update the total_questions to match actual available count
   392→      if (selectedQuestions.length != questionCount) {
   393→        await _supabase
   394→            .from(_practiceSessionsTable)
   395→            .update({'total_questions': selectedQuestions.length})
   396→            .eq('id', sessionId);
   397→        sessionResponse['total_questions'] = selectedQuestions.length;
   398→      }
   399→
   400→      AppLogger.info(
   401→        'Practice session created: $sessionId with ${selectedQuestions.length} questions',
   402→      );
   403→
   404→      return PracticeSessionModel.fromJson(sessionResponse);
   405→    } on sb.PostgrestException catch (e) {
   406→      _mapPostgrestException(e);
   407→    } catch (e) {
   408→      _handleGenericException(e, 'create practice session');
   409→    }
   410→  }

   542→  /// Completes a practice session.
   543→  ///
   544→  /// Sets the status to `completed`, calculates the final score
   545→  /// percentage, and records the completion timestamp.
   546→  Future<PracticeSessionModel> completePracticeSession({
   547→    required String sessionId,
   548→  }) async {
   549→    try {
   550→      AppLogger.info('Completing practice session: $sessionId');
   551→
   552→      // Fetch current session to compute score
   553→      final current = await _supabase
   554→          .from(_practiceSessionsTable)
   555→          .select('total_questions, correct_count')
   556→          .eq('id', sessionId)
   557→          .single();
   558→
   559→      final totalQ = current['total_questions'] as int? ?? 0;
   560→      final correctQ = current['correct_count'] as int? ?? 0;
   561→      final scorePct = totalQ > 0 ? (correctQ / totalQ) * 100 : 0.0;
   562→
   563→      final response = await _supabase
   564→          .from(_practiceSessionsTable)
   565→          .update({
   566→            'status': 'completed',
   567→            'score_pct': scorePct,
   568→            'completed_at': DateTime.now().toUtc().toIso8601String(),
   569→          })
   570→          .eq('id', sessionId)
   571→          .select()
   572→          .single();
   573→
   574→      AppLogger.info(
   575→        'Practice session completed: $sessionId (score: ${scorePct.toStringAsFixed(1)}%)',
   576→      );
   577→      return PracticeSessionModel.fromJson(response);
   578→    } on sb.PostgrestException catch (e) {
   579→      _mapPostgrestException(e);
   580→    } catch (e) {
   581→      _handleGenericException(e, 'complete practice session');
   582→    }
   583→  }

   958→  /// Uploads a document for AI chat processing.
   959→  ///
   960→  /// The flow is:
   961→  /// 1. Upload the file to Supabase Storage bucket `student-documents`.
   962→  /// 2. Create a `document_chats` record.
   963→  /// 3. Call the `process-document` Edge Function for text extraction
   964→  ///    (runs asynchronously).
   965→  Future<DocumentChatModel> uploadDocument({
   966→    required String studentId,
   967→    String? schoolId,
   968→    required String fileName,
   969→    required List<int> fileBytes,
   970→    required String fileFormat,
   971→    int? fileSize,
   972→  }) async {
   973→    try {
   974→      AppLogger.info(
   975→        'Uploading document for student: $studentId, file: $fileName',
   976→      );
   977→
   978→      // 1. Upload file to Supabase Storage
   979→      final storagePath =
   980→          '$studentId/${DateTime.now().millisecondsSinceEpoch}_$fileName';
   981→      await _supabase.storage
   982→          .from(_studentDocumentsBucket)
   983→          .uploadBinary(storagePath, fileBytes);
   984→
   985→      // Get the public URL for the uploaded file
   986→      final fileUrl = _supabase.storage
   987→          .from(_studentDocumentsBucket)
   988→          .getPublicUrl(storagePath);
   989→
   990→      // 2. Create a document_chats record
   991→      final response = await _supabase
   992→          .from(_documentChatsTable)
   993→          .insert({
   994→            'student_id': studentId,
   995→            'school_id': schoolId,
   996→            'file_name': fileName,
   997→            'file_url': fileUrl,
   998→            'file_size': fileSize,
   999→            'file_format': fileFormat,
  1000→            'status': 'processing',
  1001→          })
  1002→          .select()
  1003→          .single();
  1004→
  1005→      final documentId = response['id'] as String;
  1006→
  1007→      // 3. Trigger the process-document edge function (async)
  1008→      _supabase.functions.invoke(
  1009→        _processDocumentFunction,
  1010→        body: {
  1011→          'document_id': documentId,
  1012→          'file_url': fileUrl,
  1013→          'file_format': fileFormat,
  1014→        },
  1015→      ).catchError((error) {
  1016→        AppLogger.error(
  1017→          'Process document edge function failed',
  1018→          error: error,
  1019→        );
  1020→      });
  1021→
  1022→      AppLogger.info('Document uploaded and processing started: $documentId');
  1023→      return DocumentChatModel.fromJson(response);
  1024→    } on sb.PostgrestException catch (e) {
  1025→      _mapPostgrestException(e);
  1026→    } on sb.StorageException catch (e) {
  1027→      AppLogger.error('Storage upload failed', error: e);
  1028→      throw ServerException(
  1029→        message: 'File upload failed: ${e.message}',
  1030→        statusCode: 500,
  1031→      );
  1032→    } catch (e) {
  1033→      if (e is ServerException) rethrow;
  1034→      _handleGenericException(e, 'upload document');
  1035→    }
  1036→  }

  1163→  /// Deletes a document, its chat messages, and the stored file.
  1164→  Future<void> deleteDocument({required String documentId}) async {
  1165→    try {
  1166→      AppLogger.info('Deleting document: $documentId');
  1167→
  1168→      // Fetch the document to get the storage path
  1169→      final doc = await _supabase
  1170→          .from(_documentChatsTable)
  1171→          .select('file_url, student_id')
  1172→          .eq('id', documentId)
  1173→          .single();
  1174→
  1175→      // Delete from database (cascade deletes messages)
  1176→      await _supabase
  1177→          .from(_documentChatsTable)
  1178→          .delete()
  1179→          .eq('id', documentId);
  1180→
  1181→      // Attempt to remove the file from storage (best-effort)
  1182→      final fileUrl = doc['file_url'] as String?;
  1183→      if (fileUrl != null) {
  1184→        try {
  1185→          final studentId = doc['student_id'] as String;
  1186→          // Extract the storage path from the URL
  1187→          final uri = Uri.parse(fileUrl);
  1188→          final pathSegments = uri.pathSegments;
  1189→          final storagePath = pathSegments.skip(2).join('/');
  1190→          await _supabase.storage
  1191→              .from(_studentDocumentsBucket)
  1192→              .remove([storagePath]);
  1193→        } catch (storageError) {
  1194→          AppLogger.warning(

  1456→  /// Rates a flashcard using the SM-2 spaced repetition algorithm.
  1457→  ///
  1458→  /// The rating can be one of: `again`, `hard`, `good`, `easy`.
  1459→  ///
  1460→  /// **SM-2 Algorithm Implementation:**
  1461→  /// - `again`: repetitions = 0, interval = 1 min
  1462→  /// - `hard`:  easeFactor -= 0.15, interval = interval * 1.2
  1463→  /// - `good`:  interval = interval * easeFactor
  1464→  /// - `easy`:  easeFactor += 0.15, interval = interval * easeFactor * 1.3
  1465→  /// - easeFactor is clamped between 1.3 and 3.0
  1466→  /// - If repetitions == 0: interval = 1 day
  1467→  /// - If repetitions == 1: interval = 6 days
  1468→  /// - Else: interval = previousInterval * easeFactor
  1469→  Future<FlashcardModel> rateFlashcard({
  1470→    required String cardId,
  1471→    required String rating,
  1472→  }) async {
  1473→    try {
  1474→      AppLogger.info('Rating flashcard: $cardId with rating: $rating');
  1475→
  1476→      // 1. Fetch the current card state
  1477→      final current = await _supabase
  1478→          .from(_flashcardsTable)
  1479→          .select()
  1480→          .eq('id', cardId)
  1481→          .single();
  1482→
  1483→      var repetitions = current['repetitions'] as int? ?? 0;
  1484→      var easeFactor = (current['ease_factor'] as num?)?.toDouble() ?? 2.5;
  1485→      var interval = (current['interval'] as num?)?.toDouble() ?? 0.0;
  1486→      final previousInterval = interval;
  1487→
  1488→      // 2. Apply SM-2 algorithm based on rating
  1489→      switch (rating) {
  1490→        case 'again':
  1491→          repetitions = 0;
  1492→          interval = 1.0 / 1440.0; // 1 minute in days (~0.000694 days)
  1493→          // Alternatively, we can set next_review to now + 1 minute directly
  1494→          break;
  1495→        case 'hard':
  1496→          easeFactor = max(1.3, easeFactor - 0.15);
  1497→          interval = previousInterval * 1.2;
  1498→          if (interval < 1) interval = 1;
  1499→          repetitions = repetitions + 1;
  1500→          break;
  1501→        case 'good':
  1502→          if (repetitions == 0) {
  1503→            interval = 1; // 1 day
  1504→          } else if (repetitions == 1) {
  1505→            interval = 6; // 6 days
  1506→          } else {
  1507→            interval = previousInterval * easeFactor;
  1508→          }
  1509→          repetitions = repetitions + 1;
  1510→          break;
  1511→        case 'easy':
  1512→          easeFactor = min(3.0, easeFactor + 0.15);
  1513→          if (repetitions == 0) {
  1514→            interval = 1;
  1515→          } else if (repetitions == 1) {
  1516→            interval = 6;
  1517→          } else {
  1518→            interval = previousInterval * easeFactor;
  1519→          }
  1520→          interval = interval * 1.3;
  1521→          repetitions = repetitions + 1;
  1522→          break;
  1523→        default:
  1524→          throw ValidationException(
  1525→            message: 'Invalid flashcard rating: $rating',
  1526→            fieldErrors: {'rating': 'Must be one of: again, hard, good, easy'},
  1527→          );
  1528→      }
  1529→
  1530→      // 3. Clamp easeFactor
  1531→      easeFactor = easeFactor.clamp(1.3, 3.0);
  1532→
  1533→      // 4. Compute next review date
  1534→      final nextReview = DateTime.now().add(Duration(
  1535→        milliseconds: (interval * 24 * 60 * 60 * 1000).round(),
  1536→      ));
  1537→
  1538→      // 5. Persist the updated SM-2 state
  1539→      final response = await _supabase
  1540→          .from(_flashcardsTable)
  1541→          .update({
  1542→            'repetitions': repetitions,
  1543→            'ease_factor': easeFactor,
  1544→            'interval': interval,
  1545→            'next_review': nextReview.toUtc().toIso8601String(),
  1546→            'last_reviewed': DateTime.now().toUtc().toIso8601String(),
  1547→          })
  1548→          .eq('id', cardId)
  1549→          .select()
  1550→          .single();
  1551→
  1552→      AppLogger.info(
  1553→        'Flashcard rated: $cardId → next review in ${interval.toStringAsFixed(1)} days',
  1554→      );
  1555→      return FlashcardModel.fromJson(response);
  1556→    } on sb.PostgrestException catch (e) {
  1557→      _mapPostgrestException(e);
  1558→    } catch (e) {
  1559→      if (e is ValidationException) rethrow;
  1560→      _handleGenericException(e, 'rate flashcard');
  1561→    }
  1562→  }
