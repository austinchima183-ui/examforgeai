    19→  /// Helper: Convert exceptions to Failures
    20→  Failure _mapExceptionToFailure(Object e) {
    21→    if (e is AuthException) {
    22→      return Failure.auth(message: e.message, code: e.code);
    23→    } else if (e is ServerException) {
    24→      return Failure.server(message: e.message, statusCode: e.statusCode, data: e.data);
    25→    } else if (e is CacheException) {
    26→      return Failure.cache(message: e.message);
    27→    } else if (e is NetworkException) {
    28→      return Failure.network(message: e.message);
    29→    } else if (e is ValidationException) {
    30→      return Failure.validation(message: e.message, fieldErrors: e.fieldErrors);
    31→    } else if (e is NotFoundException) {
    32→      return Failure.notFound(message: e.message);
    33→    } else if (e is UnauthorizedException) {
    34→      return Failure.unauthorized(message: e.message);
    35→    } else if (e is ForbiddenException) {
    36→      return Failure.forbidden(message: e.message);
    37→    } else {
    38→      AppLogger.error('Unexpected exception in TeacherWorkspaceRepositoryImpl', error: e);
    39→      return Failure.server(message: 'An unexpected error occurred: $e', statusCode: 500);
    40→    }
    41→  }
