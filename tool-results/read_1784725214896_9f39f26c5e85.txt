   181→  // ── Exception mapping helper ───────────────────────────────────────────
   182→  Never _mapPostgrestException(sb.PostgrestException e) {
   183→    AppLogger.error('Postgrest error: ${e.message}', error: e);
   184→    switch (e.code) {
   185→      case 'PGRST116':
   186→        throw NotFoundException(message: e.message);
   187→      case '23505':
   188→        throw ServerException(
   189→          message: 'A record with this data already exists.',
   190→          statusCode: 409,
   191→        );
   192→      case '23503':
   193→        throw ServerException(
   194→          message: 'Referenced record not found.',
   195→          statusCode: 404,
   196→        );
   197→      case '42501':
   198→        throw ForbiddenException(
   199→          message: 'You do not have permission for this action.',
   200→        );
   201→      default:
   202→        throw ServerException(
   203→          message: e.message,
   204→          statusCode: e.statusCode ?? 500,
   205→        );
   206→    }
   207→  }

   308→  @override
   309→  Future<List<LessonPlanModel>> getLessonPlans(
   310→    Map<String, dynamic> filters,
   311→  ) async {
   312→    try {
   313→      AppLogger.info('Fetching lesson plans with filters: $filters');
   314→      var query = _supabase.from(_lessonPlansTable).select();
   315→
   316→      // Search filter
   317→      if (filters['searchQuery'] != null) {
   318→        query = query.ilike('title', '%${filters['searchQuery']}%');
   319→      }
   320→
   321→      // Subject filter
   322→      if (filters['subject_id'] != null) {
   323→        query = query.eq('subject_id', filters['subject_id']);
   324→      }
   325→
   326→      // Class filter
   327→      if (filters['class_id'] != null) {
   328→        query = query.eq('class_id', filters['class_id']);
   329→      }
   330→
   331→      // Published filter
   332→      if (filters['is_published'] != null) {
   333→        query = query.eq('is_published', filters['is_published']);
   334→      }
   335→
   336→      // Archived filter
   337→      if (filters['is_archived'] != null) {
   338→        query = query.eq('is_archived', filters['is_archived']);
   339→      }
   340→
   341→      // Tags overlap filter
   342→      if (filters['tags'] != null) {
   343→        query = query.overlaps('tags', filters['tags']);
   344→      }
   345→
   346→      // Ordering
   347→      query = query.order('created_at', ascending: false);
   348→
   349→      // Pagination
   350→      final page = filters['page'] as int? ?? 1;
   351→      final perPage = filters['perPage'] as int? ?? 20;
   352→      query = query.range((page - 1) * perPage, page * perPage - 1);
   353→
   354→      final response = await query;
   355→      AppLogger.info('Fetched ${response.length} lesson plans');
   356→      return response.map(LessonPlanModel.fromJson).toList();
   357→    } on sb.PostgrestException catch (e) {
   358→      _mapPostgrestException(e);
   359→    } catch (e) {
   360→      AppLogger.error('Failed to fetch lesson plans', error: e);
   361→      throw ServerException(message: e.toString(), statusCode: 500);
   362→    }
   363→  }
