   207→  @override
   208→  Future<GradeScaleModel> createGradeScale(GradeScaleModel scale) async {
   209→    try {
   210→      // 1. Insert the grade_scale row (without nested entries)
   211→      final scaleJson = scale.toJson()
   212→        ..remove('scale_entries');
   213→
   214→      final response = await _supabaseClient
   215→          .from(_gradeScalesTable)
   216→          .insert(scaleJson)
   217→          .select()
   218→          .single();
   219→
   220→      final createdScale = GradeScaleModel.fromJson(response);
   221→
   222→      // 2. Insert entries
   223→      if (scale.scaleEntries.isNotEmpty) {
   224→        final entriesRows = scale.scaleEntries.map((e) {
   225→          final json = e.toJson();
   226→          json['grade_scale_id'] = createdScale.id;
   227→          return json;
   228→        }).toList();
   229→
   230→        await _supabaseClient
   231→            .from(_gradeScaleEntriesTable)
   232→            .insert(entriesRows);
   233→      }
   234→
   235→      // 3. Return scale with entries populated
   236→      return createdScale.copyWith(
   237→        scaleEntries: scale.scaleEntries,
   238→      );
   239→    } on sb.PostgrestException catch (e) {
   240→      AppLogger.error('Create grade scale failed', error: e);
   241→      throw ServerException(
   242→        message: e.message,
   243→        statusCode: int.tryParse(e.code ?? '') ?? 500,
   244→        data: e.details,
   245→      );
   246→    } catch (e) {
   247→      AppLogger.error('Create grade scale unexpected error', error: e);
   248→      throw ServerException(
   249→        message: 'Failed to create grade scale: $e',
   250→        statusCode: 500,
   251→      );
   252→    }
   253→  }
   254→
   255→  @override
   256→  Future<GradeScaleModel> updateGradeScale(GradeScaleModel scale) async {
   257→    try {
   258→      // 1. Update the grade_scale row
   259→      final scaleJson = scale.toJson()
   260→        ..remove('scale_entries');
   261→
   262→      final response = await _supabaseClient
   263→          .from(_gradeScalesTable)
   264→          .update(scaleJson)
   265→          .eq('id', scale.id)
   266→          .select()
   267→          .single();
   268→
   269→      final updatedScale = GradeScaleModel.fromJson(response);
   270→
   271→      // 2. Delete existing entries and re-insert
   272→      await _supabaseClient
   273→          .from(_gradeScaleEntriesTable)
   274→          .delete()
   275→          .eq('grade_scale_id', scale.id);
   276→
   277→      if (scale.scaleEntries.isNotEmpty) {
   278→        final entriesRows = scale.scaleEntries.map((e) {
   279→          final json = e.toJson();
   280→          json['grade_scale_id'] = scale.id;
   281→          return json;
   282→        }).toList();
   283→
   284→        await _supabaseClient
   285→            .from(_gradeScaleEntriesTable)
   286→            .insert(entriesRows);
   287→      }
   288→
   289→      // 3. Return scale with entries populated
   290→      return updatedScale.copyWith(
   291→        scaleEntries: scale.scaleEntries,
   292→      );
   293→    } on sb.PostgrestException catch (e) {
   294→      AppLogger.error('Update grade scale failed', error: e);
   295→      if (e.code == 'PGRST116') {
   296→        throw NotFoundException('Grade scale not found: ${scale.id}');
   297→      }
   298→      throw ServerException(
   299→        message: e.message,
   300→        statusCode: int.tryParse(e.code ?? '') ?? 500,
   301→        data: e.details,
   302→      );
   303→    } catch (e) {
   304→      AppLogger.error('Update grade scale unexpected error', error: e);
   305→      throw ServerException(
   306→        message: 'Failed to update grade scale: $e',
   307→        statusCode: 500,
   308→      );
   309→    }
   310→  }

   335→  @override
   336→  Future<GradeScaleModel> getGradeScale(String scaleId) async {
   337→    try {
   338→      final response = await _supabaseClient
   339→          .from(_gradeScalesTable)
   340→          .select('*, grade_scale_entries(*)')
   341→          .eq('id', scaleId)
   342→          .single();
   343→
   344→      // Map nested entries from the join
   345→      final mappedResponse = _mapGradeScaleEntries(response);
   346→
   347→      return GradeScaleModel.fromJson(mappedResponse);
   348→    } on sb.PostgrestException catch (e) {
   349→      AppLogger.error('Get grade scale failed', error: e);
   350→      if (e.code == 'PGRST116') {
   351→        throw NotFoundException('Grade scale not found: $scaleId');
   352→      }
   353→      throw ServerException(
   354→        message: e.message,
   355→        statusCode: int.tryParse(e.code ?? '') ?? 500,
   356→        data: e.details,
   357→      );
   358→    } catch (e) {
   359→      AppLogger.error('Get grade scale unexpected error', error: e);
   360→      throw ServerException(
   361→        message: 'Failed to get grade scale: $e',
   362→        statusCode: 500,
   363→      );
   364→    }
   365→  }

   367→  @override
   368→  Future<List<GradeScaleModel>> getGradeScales(
   369→    String schoolId, {
   370→    bool? isActive,
   371→    String? gradeType,
   372→  }) async {
   373→    try {
   374→      var query = _supabaseClient
   375→          .from(_gradeScalesTable)
   376→          .select('*, grade_scale_entries(*)');
   377→
   378→      query = query.eq('school_id', schoolId);
   379→
   380→      if (isActive != null) {
   381→        query = query.eq('is_active', isActive);
   382→      }
   383→      if (gradeType != null) {
   384→        query = query.eq('grade_type', gradeType);
   385→      }
   386→
   387→      query = query.order('created_at', ascending: false);
   388→
   389→      final response = await query;
   390→
   391→      return response.map<GradeScaleModel>((json) {
   392→        final mappedJson = _mapGradeScaleEntries(json);
   393→        return GradeScaleModel.fromJson(mappedJson);
   394→      }).toList();
   395→    } on sb.PostgrestException catch (e) {
   396→      AppLogger.error('Get grade scales failed', error: e);
   397→      throw ServerException(
   398→        message: e.message,
   399→        statusCode: int.tryParse(e.code ?? '') ?? 500,
   400→        data: e.details,
   401→      );
   402→    } catch (e) {
   403→      AppLogger.error('Get grade scales unexpected error', error: e);
   404→      throw ServerException(
   405→        message: 'Failed to get grade scales: $e',
   406→        statusCode: 500,
   407→      );
   408→    }
   409→  }
   410→
   411→  /// Maps the nested `grade_scale_entries` join result to the model's
   412→  /// expected `scale_entries` key.
   413→  Map<String, dynamic> _mapGradeScaleEntries(Map<String, dynamic> json) {
   414→    final entries = json['grade_scale_entries'];
   415→    if (entries is List) {
   416→      return {
   417→        ...json,
   418→        'scale_entries': entries,
   419→      };
   420→    }
   421→    return json;
   422→  }

  1284→  @override
  1285→  Future<DashboardConfigurationModel> upsertDashboardConfiguration(
  1286→    DashboardConfigurationModel config,
  1287→  ) async {
  1288→    try {
  1289→      // 1. Upsert the dashboard configuration row (without nested widgets)
  1290→      final configJson = config.toJson()
  1291→        ..remove('widgets');
  1292→
  1293→      final response = await _supabaseClient
  1294→          .from(_dashboardConfigurationsTable)
  1295→          .upsert(
  1296→            configJson,
  1297→            onConflict: 'school_id,role',
  1298→          )
  1299→          .select()
  1300→          .single();
  1301→
  1302→      final savedConfig = DashboardConfigurationModel.fromJson(response);
  1303→
  1304→      // 2. Delete existing widget configs and re-insert
  1305→      await _supabaseClient
  1306→          .from(_dashboardWidgetConfigsTable)
  1307→          .delete()
  1308→          .eq('dashboard_id', savedConfig.id);
  1309→
  1310→      if (config.widgets.isNotEmpty) {
  1311→        final widgetRows = config.widgets.map((w) {
  1312→          final json = w.toJson();
  1313→          json['dashboard_id'] = savedConfig.id;
  1314→          return json;
  1315→        }).toList();
  1316→
  1317→        await _supabaseClient
  1318→            .from(_dashboardWidgetConfigsTable)
  1319→            .insert(widgetRows);
  1320→      }
  1321→
  1322→      // 3. Return config with widgets populated
  1323→      return savedConfig.copyWith(
  1324→        widgets: config.widgets,
  1325→      );
  1326→    } on sb.PostgrestException catch (e) {
  1327→      AppLogger.error('Upsert dashboard configuration failed', error: e);
  1328→      throw ServerException(
  1329→        message: e.message,
  1330→        statusCode: int.tryParse(e.code ?? '') ?? 500,
  1331→        data: e.details,
  1332→      );
  1333→    } catch (e) {
  1334→      AppLogger.error(
  1335→        'Upsert dashboard configuration unexpected error',
  1336→        error: e,
  1337→      );
  1338→      throw ServerException(
  1339→        message: 'Failed to upsert dashboard configuration: $e',
  1340→        statusCode: 500,
  1341→      );
  1342→    }
