   397→  @override
   398→  Future<Result<TransactionEntity>> verifyPayment(String txRef) async {
   399→    try {
   400→      // ─── STEP 1: Look up our local transaction FIRST ────────────────
   401→      // This is critical for security: we read the EXPECTED amount from
   402→      // our own database before verifying with Flutterwave. This prevents
   403→      // an attacker from paying a different amount and claiming success.
   404→      final localTx = await _remoteDataSource.getTransaction(txRef);
   405→      final expectedAmount = localTx.amount;
   406→      final expectedCurrency = localTx.currency;
   407→
   408→      // ─── STEP 2: Verify with Flutterwave, passing expected amounts ─
   409→      // The FlutterwaveDataSource now validates that the actual charged
   410→      // amount matches what we recorded in our database.
   411→      final flutterwaveData = await _flutterwaveDataSource.verifyTransaction(
   412→        txRef,
   413→        expectedAmount: expectedAmount,
   414→        expectedCurrency: expectedCurrency,
   415→      );
   416→
   417→      // ─── STEP 3: Update our local transaction record ───────────────
   418→      final status = _mapFlutterwaveStatus(
   419→        flutterwaveData['status'] as String? ?? 'pending',
   420→      );
   421→
   422→      // SECURITY: If the transaction is successful, also verify that
   423→      // the Flutterwave transaction ID hasn't been used before (prevents
   424→      // replay attacks where the same Flw ID is claimed for two tx_refs).
   425→      if (status == TransactionStatus.successful) {
   426→        final flwTxId = flutterwaveData['id']?.toString();
   427→        if (flwTxId != null && flwTxId.isNotEmpty) {
   428→          final isReplay = await _isFlutterwaveTransactionReplay(flwTxId, localTx.id);
   429→          if (isReplay) {
   430→            AppLogger.error(
   431→              'PAYMENT REPLAY ATTACK DETECTED: Flutterwave transaction ID '
   432→              '$flwTxId has already been used for a different transaction!',
   433→            );
   434→            return const FailureResult(Failure.server(
   435→              message: 'Payment verification failed: duplicate transaction.',
   436→              statusCode: 409,
   437→            ),);
   438→          }
   439→        }
   440→      }
   441→
   442→      final updatedTx = await _remoteDataSource.updateTransaction(
   443→        localTx.id,
   444→        {
   445→          'status': status.value,
   446→          'flutterwave_transaction_id': flutterwaveData['id']?.toString(),
   447→          'flutterwave_flw_ref': flutterwaveData['flw_ref'] as String?,
   448→          'flutterwave_fee': flutterwaveData['app_fee'] as double? ?? 0,
   449→          'net_amount': flutterwaveData['amount_settled'] as double? ?? 0,
   450→          'payment_method_summary': flutterwaveData['payment_type'] as String?,
   451→          'processor_response': flutterwaveData,
   452→          'verified_at': DateTime.now().toIso8601String(),
   453→          if (status == TransactionStatus.successful)
   454→            'completed_at': DateTime.now().toIso8601String(),
   455→        },
   456→      );
   457→      return Success(updatedTx.toEntity());
   458→    } on AuthException catch (e) {
   459→      return FailureResult(Failure.auth(message: e.message, code: e.code));
   460→    } on ServerException catch (e) {
   461→      return FailureResult(Failure.server(
   462→        message: e.message, statusCode: e.statusCode, data: e.data,
   463→      ),);
   464→    } on NotFoundException catch (e) {
   465→      return FailureResult(Failure.notFound(message: e.message));
   466→    } catch (e) {
   467→      AppLogger.error('Unexpected verifyPayment error', error: e);
   468→      return const FailureResult(Failure.server(
   469→        message: 'Failed to verify payment.', statusCode: 500,
   470→      ),);
   471→    }
   472→  }

   474→  /// Checks whether a Flutterwave transaction ID has already been
   475→  /// associated with a DIFFERENT local transaction. This prevents
   476→  /// replay attacks where the same successful payment is claimed
   477→  /// for multiple transaction references.
   478→  Future<bool> _isFlutterwaveTransactionReplay(
   479→    String flwTxId,
   480→    String currentLocalTxId,
   481→  ) async {
   482→    try {
   483→      // Query our transactions table for any other transaction with
   484→      // the same Flutterwave transaction ID that's already successful.
   485→      final existingTxs = await _remoteDataSource.getTransactions(
   486→        status: 'successful',
   487→        page: 1,
   488→        perPage: 5,
   489→      );
   490→      return existingTxs.any((tx) =>
   491→          tx.flutterwaveTransactionId == flwTxId &&
   492→          tx.id != currentLocalTxId,);
   493→    } catch (e) {
   494→      // If we can't check, err on the side of caution and allow
   495→      // the transaction through. The amount verification above is
   496→      // the primary defence.
   497→      AppLogger.warning(
   498→        'Could not verify Flutterwave transaction ID uniqueness',
   499→        error: e,
   500→      );
   501→      return false;
   502→    }
   503→  }

   505→  @override
   506→  Future<Result<bool>> processWebhookEvent(Map<String, dynamic> payload) async {
   507→    try {
   508→      final event = payload['event'] as String? ?? '';
   509→      final data = payload['data'] as Map<String, dynamic>? ?? {};
   510→
   511→      // ─── IDEMPOTENCY CHECK (CRITICAL SECURITY FIX) ──────────────────
   512→      // Build an idempotency key from the event type and the unique
   513→      // Flutterwave event/transaction ID. This prevents the same webhook
   514→      // from being processed twice (Flutterwave retries webhooks on
   515→      // failure, so duplicates are expected).
   516→      final flwId = data['id']?.toString() ?? '';
   517→      final idempotencyKey = '${event}_$flwId';
   518→
   519→      final flutterwaveDs = _flutterwaveDataSource;
   520→      if (flutterwaveDs is FlutterwaveDataSourceImpl) {
   521→        if (flutterwaveDs.isWebhookEventProcessed(idempotencyKey)) {
   522→          AppLogger.info(
   523→            'Webhook event already processed (idempotent): $idempotencyKey. '
   524→            'Returning success without re-processing.',
   525→          );
   526→          return const Success(true);
   527→        }
   528→      }
   529→
   530→      AppLogger.info('Processing Flutterwave webhook event: $event');
   531→
   532→      switch (event) {
   533→        case 'charge.completed':
   534→          final txRef = data['tx_ref'] as String?;
   535→          if (txRef != null) {
   536→            // Verify and update the transaction (includes amount
   537→            // verification via the updated verifyPayment method)
   538→            final result = await verifyPayment(txRef);
   539→            if (result is Success) {
   540→              AppLogger.info(
   541→                'Webhook charge.completed processed successfully for $txRef',
   542→              );
   543→            } else {
   544→              AppLogger.warning(
   545→                'Webhook charge.completed verification failed for $txRef',
   546→              );
   547→            }
   548→          }
   549→          break;
   550→        case 'transfer.completed':
   551→        case 'transfer.failed':
   552→          AppLogger.info('Transfer event received: $event');
   553→          break;
   554→        case 'subscription.cancelled':
   555→          final subId = data['subscription_id'] as String?;
   556→          if (subId != null) {
   557→            await _remoteDataSource.updateSubscription(subId, {
   558→              'status': 'cancelled',
   559→              'cancelled_at': DateTime.now().toIso8601String(),
   560→            });
   561→          }
   562→          break;
   563→        default:
   564→          AppLogger.info('Unhandled webhook event: $event');
   565→      }
   566→
   567→      // Mark event as processed AFTER successful handling
   568→      if (flutterwaveDs is FlutterwaveDataSourceImpl) {
   569→        flutterwaveDs.markWebhookEventProcessed(idempotencyKey);
   570→      }
   571→
   572→      return const Success(true);
   573→    } on AuthException catch (e) {
   574→      return FailureResult(Failure.auth(message: e.message, code: e.code));
   575→    } on ServerException catch (e) {
   576→      return FailureResult(Failure.server(
   577→        message: e.message, statusCode: e.statusCode, data: e.data,

   671→        },

   673→      return Success(updatedTx.toEntity());
