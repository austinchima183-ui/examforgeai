    10→/// Concrete implementation of [CcmsRepository] that delegates
    11→/// all operations to [CcmsRemoteDataSource].
    12→///
    13→/// Responsibilities:
    14→/// - Converts domain entities to/from data models
    15→/// - Catches data-layer exceptions and converts them to domain [Failure] types
    16→/// - Returns [Result] to force callers to handle both success and failure
    17→class CcmsRepositoryImpl implements CcmsRepository {
    18→  CcmsRepositoryImpl({
    19→    required CcmsRemoteDataSource remoteDataSource,
    20→  }) : _remoteDataSource = remoteDataSource;
    21→
    22→  final CcmsRemoteDataSource _remoteDataSource;
    23→
    24→  // ═══════════════════════════════════════════════════════════════════════
    25→  // HELPER: Exception → Failure mapping
    26→  // ═══════════════════════════════════════════════════════════════════════
    27→
    28→  Failure _mapException(Object e) {
    29→    if (e is AuthException) {
    30→      return Failure.auth(message: e.message, code: e.code);
    31→    } else if (e is ServerException) {
    32→      return Failure.server(
    33→        message: e.message,
    34→        statusCode: e.statusCode,
    35→        data: e.data,
    36→      );
    37→    } else if (e is NetworkException) {
    38→      return Failure.network(message: e.message);
    39→    } else if (e is ValidationException) {
    40→      return Failure.validation(
    41→        message: e.message,
    42→        fieldErrors: e.fieldErrors,
    43→      );
    44→    } else if (e is NotFoundException) {
    45→      return Failure.notFound(message: e.message);
    46→    } else if (e is UnauthorizedException) {
    47→      return Failure.unauthorized(message: e.message);
    48→    } else if (e is ForbiddenException) {
    49→      return Failure.forbidden(message: e.message);
    50→    } else if (e is FormatException) {
    51→      return Failure.server(
    52→        message: 'Data format error: ${e.message}',
    53→        statusCode: 422,
    54→      );
    55→    } else {
    56→      AppLogger.error('Unexpected CCMS repository error', error: e);
    57→      return Failure.server(
    58→        message: e.toString(),
    59→        statusCode: 500,
    60→      );
    61→    }
    62→  }
