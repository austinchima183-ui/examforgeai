   297→-- ═══════════════════════════════════════════════════════════════════════════════
   298→-- PARENT ENGAGEMENT SUMMARY MATERIALIZED VIEW
   299→-- ═══════════════════════════════════════════════════════════════════════════════
   300→
   301→CREATE MATERIALIZED VIEW IF NOT EXISTS parent_engagement_summary AS
   302→SELECT
   303→  pem.school_id,
   304→  pem.parent_id,
   305→  pem.student_id,
   306→  COUNT(DISTINCT CASE WHEN pem.metric_type = 'report_card_viewed' THEN pem.id END) AS report_card_views,
   307→  COUNT(DISTINCT CASE WHEN pem.metric_type = 'announcement_read' THEN pem.id END) AS announcements_read,
   308→  COUNT(DISTINCT CASE WHEN pem.metric_type = 'message_sent' THEN pem.id END) AS messages_sent,
   309→  COUNT(DISTINCT CASE WHEN pem.metric_type = 'meeting_attended' THEN pem.id END) AS meetings_attended,
   310→  COUNT(DISTINCT CASE WHEN pem.metric_type = 'assignment_checked' THEN pem.id END) AS assignment_checks,
   311→  COUNT(DISTINCT CASE WHEN pem.metric_type = 'attendance_checked' THEN pem.id END) AS attendance_checks,
   312→  COUNT(DISTINCT CASE WHEN pem.metric_type = 'ai_assistant_used' THEN pem.id END) AS ai_assistant_uses,
   313→  COUNT(DISTINCT pem.id) AS total_interactions,
   314→  MAX(pem.recorded_at) AS last_active_at,
   315→  CASE
   316→    WHEN MAX(pem.recorded_at) > now() - INTERVAL '7 days' THEN 'active'
   317→    WHEN MAX(pem.recorded_at) > now() - INTERVAL '30 days' THEN 'moderate'
   318→    ELSE 'inactive'
   319→  END AS engagement_level
   320→FROM parent_engagement_metrics pem
   321→GROUP BY pem.school_id, pem.parent_id, pem.student_id;
   322→
   323→CREATE UNIQUE INDEX idx_engagement_summary_pk ON parent_engagement_summary(school_id, parent_id, student_id);

   326→-- PARENT DASHBOARD VIEW
   327→-- ═══════════════════════════════════════════════════════════════════════════════
   328→
   329→CREATE OR REPLACE VIEW parent_dashboard_view AS
   330→SELECT
   331→  p.id AS parent_id,
   332→  p.school_id,
   333→  p.user_id,
   334→  u.full_name AS parent_name,
   335→  u.email AS parent_email,
   336→  u.phone AS parent_phone,
   337→  u.avatar_url AS parent_avatar,
   338→  -- Child count
   339→  (SELECT COUNT(*) FROM parent_students ps WHERE ps.parent_id = p.id AND ps.is_active = TRUE) AS child_count,
   340→  -- Unread notifications
   341→  (SELECT COUNT(*) FROM parent_notifications pn WHERE pn.parent_id = p.user_id AND pn.is_read = FALSE) AS unread_notifications,
   342→  -- Unread messages
   343→  (SELECT COUNT(*) FROM parent_messages pm WHERE pm.recipient_id = p.user_id AND pm.read_at IS NULL AND pm.is_archived = FALSE) AS unread_messages,
   344→  -- Active insights
   345→  (SELECT COUNT(*) FROM parent_ai_insights pai WHERE pai.parent_id = p.user_id AND pai.is_read = FALSE AND pai.is_dismissed = FALSE) AS active_insights,
   346→  -- Last active
   347→  (SELECT MAX(pal.created_at) FROM parent_activity_logs pal WHERE pal.parent_id = p.user_id) AS last_active_at
   348→FROM parent_profiles p
   349→JOIN users u ON u.id = p.user_id;

   476→-- Get parent dashboard data
   477→CREATE OR REPLACE FUNCTION get_parent_dashboard(p_parent_user_id UUID)
   478→RETURNS JSONB AS $$
   479→DECLARE
   480→  v_result JSONB;
   481→  v_school_id UUID;
   482→  v_parent_id UUID;
   483→BEGIN
   484→  SELECT id, school_id INTO v_parent_id, v_school_id
   485→  FROM parent_profiles WHERE user_id = p_parent_user_id LIMIT 1;
   486→
   487→  SELECT jsonb_build_object(
   488→    'parent_id', v_parent_id,
   489→    'school_id', v_school_id,
   490→    'children', (
   491→      SELECT COALESCE(jsonb_agg(
   492→        jsonb_build_object(
   493→          'student_id', ps.student_id,
   494→          'student_name', ps.student_name,
   495→          'admission_number', ps.student_admission_number,
   496→          'class_name', (SELECT c.name FROM classes c JOIN class_students cs ON cs.class_id = c.id WHERE cs.student_id = ps.student_id LIMIT 1),
   497→          'relationship', ps.relationship,
   498→          'is_primary', ps.is_primary_contact,
   499→          'attendance_summary', jsonb_build_object(
   500→            'present_days', (SELECT COUNT(*) FROM attendance a WHERE a.student_id = ps.student_id AND a.status = 'present' AND a.date >= date_trunc('month', CURRENT_DATE)),
   501→            'absent_days', (SELECT COUNT(*) FROM attendance a WHERE a.student_id = ps.student_id AND a.status = 'absent' AND a.date >= date_trunc('month', CURRENT_DATE)),
   502→            'late_days', (SELECT COUNT(*) FROM attendance a WHERE a.student_id = ps.student_id AND a.status = 'late' AND a.date >= date_trunc('month', CURRENT_DATE))
   503→          ),
   504→          'pending_assignments', (SELECT COUNT(*) FROM homework h JOIN class_students cs ON cs.class_id = h.class_id WHERE cs.student_id = ps.student_id AND h.status = 'published' AND h.due_date > now()),
   505→          'latest_results', (
   506→            SELECT COALESCE(jsonb_agg(
   507→              jsonb_build_object('exam_title', e.title, 'score', er.score, 'total_marks', e.total_marks, 'grade', er.grade)
   508→            ), '[]'::jsonb)
   509→            FROM exam_results er
   510→            JOIN exams e ON e.id = er.exam_id
   511→            WHERE er.student_id = ps.student_id
   512→            ORDER BY er.created_at DESC LIMIT 3
   513→          )
   514→        )
   515→      ), '[]'::jsonb)
   516→      FROM parent_students ps
   517→      WHERE ps.parent_id = v_parent_id AND ps.is_active = TRUE
   518→    ),
   519→    'upcoming_events', (
   520→      SELECT COALESCE(jsonb_agg(
   521→        jsonb_build_object('id', id, 'title', title, 'start_time', start_time, 'end_time', end_time, 'event_type', event_type)
   522→      ), '[]'::jsonb)
   523→      FROM parent_calendar_events
   524→      WHERE school_id = v_school_id
   525→      AND (parent_id = p_parent_user_id OR parent_id IS NULL)

   527→      ORDER BY start_time LIMIT 5

   529→    'recent_announcements', (

   531→        jsonb_build_object('id', a.id, 'title', a.title, 'type', a.announcement_type, 'priority', a.priority, 'created_at', a.created_at)
   532→      ), '[]'::jsonb)
   533→      FROM announcements a
   534→      WHERE a.school_id = v_school_id
   535→      AND (a.target_audience @> ARRAY['parents'] OR a.target_audience = ARRAY['all']::text[])
   536→      ORDER BY a.created_at DESC LIMIT 5
   537→    ),
   538→    'unread_notifications', (SELECT COUNT(*) FROM parent_notifications WHERE parent_id = p_parent_user_id AND is_read = FALSE),
   539→    'unread_messages', (SELECT COUNT(*) FROM parent_messages WHERE recipient_id = p_parent_user_id AND read_at IS NULL AND is_archived = FALSE),
   540→    'active_insights', (SELECT COUNT(*) FROM parent_ai_insights WHERE parent_id = p_parent_user_id AND is_read = FALSE AND is_dismissed = FALSE)
   541→  ) INTO v_result;
   542→
   543→  RETURN v_result;
   544→END;

   547→-- Get child academic performance
   548→CREATE OR REPLACE FUNCTION get_child_performance(p_student_id UUID, p_academic_session_id UUID DEFAULT NULL)
   549→RETURNS JSONB AS $$
   550→DECLARE
   551→  v_result JSONB;
   552→BEGIN
   553→  SELECT jsonb_build_object(
   554→    'student_id', p_student_id,
   555→    'subjects', (
   556→      SELECT COALESCE(jsonb_agg(
   557→        jsonb_build_object(
   558→          'subject_id', cs.subject_id,
   559→          'subject_name', s.name,
   560→          'teacher_name', (SELECT full_name FROM users WHERE id = cs.teacher_id),
   561→          'latest_score', (
   562→            SELECT er.score FROM exam_results er
   563→            JOIN exams e ON e.id = er.exam_id
   564→            JOIN exam_questions eq ON eq.exam_id = e.id
   565→            WHERE er.student_id = p_student_id AND e.subject_id = cs.subject_id
   566→            ORDER BY er.created_at DESC LIMIT 1
   567→          ),
   568→          'average_score', (
   569→            SELECT AVG(er.score) FROM exam_results er
   570→            JOIN exams e ON e.id = er.exam_id
   571→            WHERE er.student_id = p_student_id AND e.subject_id = cs.subject_id
   572→          ),
   573→          'grade', (
   574→            SELECT er.grade FROM exam_results er
   575→            JOIN exams e ON e.id = er.exam_id
   576→            WHERE er.student_id = p_student_id AND e.subject_id = cs.subject_id
   577→            ORDER BY er.created_at DESC LIMIT 1
   578→          )
   579→        )
   580→      ), '[]'::jsonb)
   581→      FROM class_subjects cs
   582→      JOIN subjects s ON s.id = cs.subject_id
   583→      WHERE cs.class_id IN (SELECT cs2.class_id FROM class_students cs2 WHERE cs2.student_id = p_student_id)
   584→    ),
   585→    'overall_average', (
   586→      SELECT AVG(er.score) FROM exam_results er
   587→      WHERE er.student_id = p_student_id
   588→    ),
   589→    'class_average', (
   590→      SELECT AVG(er.score) FROM exam_results er
   591→      WHERE er.student_id IN (
   592→        SELECT cs.student_id FROM class_students cs
   593→        WHERE cs.class_id IN (SELECT cs2.class_id FROM class_students cs2 WHERE cs2.student_id = p_student_id)
   594→      )
   595→    ),
   596→    'attendance_rate', (
   597→      SELECT CASE COUNT(*)
   598→        WHEN 0 THEN NULL
   599→        ELSE ROUND((COUNT(*) FILTER (WHERE a.status = 'present')::NUMERIC / COUNT(*)) * 100, 1)
   600→      END
   601→      FROM attendance a WHERE a.student_id = p_student_id
   602→    ),
   603→    'teacher_remarks', (
   604→      SELECT COALESCE(jsonb_agg(
   605→        jsonb_build_object('teacher_name', u.full_name, 'subject', s.name, 'remark', er.teacher_comment, 'date', er.created_at)
   606→      ), '[]'::jsonb)
   607→      FROM exam_results er
   608→      JOIN users u ON u.id = er.graded_by
   609→      JOIN exams e ON e.id = er.exam_id
   610→      JOIN subjects s ON s.id = e.subject_id
   611→      WHERE er.student_id = p_student_id AND er.teacher_comment IS NOT NULL
   612→      ORDER BY er.created_at DESC LIMIT 5
   613→    )
   614→  ) INTO v_result;
   615→
   616→  RETURN v_result;
   617→END;
   618→$$ LANGUAGE plpgsql SECURITY DEFINER;

   620→-- Get parent engagement analytics (for school admins)
   621→CREATE OR REPLACE FUNCTION get_parent_engagement_analytics(p_school_id UUID)
   622→RETURNS JSONB AS $$
   623→DECLARE
   624→  v_result JSONB;
   625→BEGIN
   626→  SELECT jsonb_build_object(
   627→    'total_parents', (SELECT COUNT(DISTINCT parent_id) FROM parent_engagement_metrics WHERE school_id = p_school_id),
   628→    'active_parents', (SELECT COUNT(DISTINCT parent_id) FROM parent_engagement_metrics WHERE school_id = p_school_id AND recorded_at > now() - INTERVAL '7 days'),
   629→    'moderate_parents', (
   630→      SELECT COUNT(DISTINCT parent_id) FROM parent_engagement_metrics
   631→      WHERE school_id = p_school_id
   632→      AND recorded_at BETWEEN now() - INTERVAL '30 days' AND now() - INTERVAL '7 days'
   633→    ),
   634→    'inactive_parents', (
   635→      SELECT COUNT(DISTINCT pp.user_id) FROM parent_profiles pp
   636→      WHERE pp.school_id = p_school_id
   637→      AND pp.user_id NOT IN (SELECT DISTINCT parent_id FROM parent_engagement_metrics WHERE school_id = p_school_id AND recorded_at > now() - INTERVAL '30 days')
   638→    ),
   639→    'report_card_not_viewed', (
   640→      SELECT COUNT(DISTINCT ps.parent_id) FROM parent_students ps
   641→      WHERE ps.school_id = p_school_id
   642→      AND ps.parent_id NOT IN (
   643→        SELECT DISTINCT parent_id FROM parent_engagement_metrics
   644→        WHERE school_id = p_school_id AND metric_type = 'report_card_viewed'
   645→        AND recorded_at > now() - INTERVAL '30 days'
   646→      )
   647→    ),
   648→    'avg_message_response_hours', (
   649→      SELECT AVG(EXTRACT(EPOCH FROM (pm2.created_at - pm1.created_at)) / 3600)
   650→      FROM parent_messages pm1
   651→      JOIN parent_messages pm2 ON pm2.parent_message_id = pm1.id
   652→      WHERE pm1.school_id = p_school_id
   653→      AND pm1.direction = 'incoming'
   654→      AND pm2.direction = 'outgoing'
   655→      AND pm1.created_at > now() - INTERVAL '30 days'
   656→    ),
   657→    'unread_announcement_count', (
   658→      SELECT COUNT(*) FROM announcements a
   659→      WHERE a.school_id = p_school_id
   660→      AND a.created_at > now() - INTERVAL '7 days'
   661→    ),
   662→    'engagement_by_metric', (
   663→      SELECT jsonb_object_agg(metric_type, cnt)
   664→      FROM (
   665→        SELECT metric_type, COUNT(*) AS cnt
   666→        FROM parent_engagement_metrics
   667→        WHERE school_id = p_school_id
   668→        AND recorded_at > now() - INTERVAL '30 days'
   669→        GROUP BY metric_type
   670→      ) sub
   671→    ),
   672→    'students_needing_support', (
   673→      SELECT COALESCE(jsonb_agg(
   674→        jsonb_build_object(
   675→          'student_id', ps.student_id,
   676→          'student_name', ps.student_name,
   677→          'parent_name', (SELECT full_name FROM users WHERE id = ps.parent_id),
   678→          'engagement_level', pes.engagement_level,
   679→          'last_active', pes.last_active_at
   680→        )
   681→      ), '[]'::jsonb)
   682→      FROM parent_students ps
   683→      LEFT JOIN parent_engagement_summary pes ON pes.parent_id = ps.parent_id AND pes.student_id = ps.student_id
   684→      WHERE ps.school_id = p_school_id
   685→      AND (pes.engagement_level = 'inactive' OR pes.last_active_at < now() - INTERVAL '30 days' OR pes.last_active_at IS NULL)
   686→      LIMIT 20
   687→    ),
   688→    'engagement_trends', (
   689→      SELECT COALESCE(jsonb_agg(
   690→        jsonb_build_object('date', date_trunc('week', recorded_at)::date, 'interactions', cnt)
   691→        ORDER BY date_trunc('week', recorded_at)::date
   692→      ), '[]'::jsonb)
   693→      FROM (
   694→        SELECT date_trunc('week', recorded_at) AS week, COUNT(*) AS cnt
   695→        FROM parent_engagement_metrics
   696→        WHERE school_id = p_school_id AND recorded_at > now() - INTERVAL '12 weeks'
   697→        GROUP BY date_trunc('week', recorded_at)
   698→      ) sub
   699→    )
   700→  ) INTO v_result;
   701→
   702→  RETURN v_result;
   703→END;
   704→$$ LANGUAGE plpgsql SECURITY DEFINER;

   706→-- Generate AI parent insight
   707→CREATE OR REPLACE FUNCTION generate_parent_insight(
   708→  p_parent_id UUID,
   709→  p_student_id UUID,
   710→  p_insight_type parent_insight_type,
   711→  p_data JSONB DEFAULT '{}'::jsonb
   712→)
   713→RETURNS UUID AS $$
   714→DECLARE
   715→  v_insight_id UUID;
   716→  v_title TEXT;
   717→  v_description TEXT;
   718→  v_severity TEXT := 'info';
   719→  v_recommendations TEXT[] := '{}';
   720→BEGIN
   721→  -- Route to specific insight generator based on type
   722→  CASE p_insight_type
   723→    WHEN 'performance_trend' THEN
   724→      v_title := 'Academic Performance Update';
   725→      v_description := COALESCE(p_data->>'summary', 'Your child''s performance has been analyzed.');
   726→      v_severity := COALESCE(p_data->>'severity', 'info');
   727→      v_recommendations := ARRAY[
   728→        'Review the latest report card together',
   729→        'Discuss challenging subjects with the teacher',
   730→        'Set up a regular study schedule'
   731→      ];
   732→
   733→    WHEN 'attendance_alert' THEN
   734→      v_title := 'Attendance Notice';
   735→      v_description := COALESCE(p_data->>'summary', 'Attendance pattern has been detected.');
   736→      v_severity := COALESCE(p_data->>'severity', 'warning');
   737→      v_recommendations := ARRAY[
   738→        'Ensure your child attends school regularly',
   739→        'Contact the school if there are health concerns',
   740→        'Schedule appointments outside school hours when possible'
   741→      ];
   742→
   743→    WHEN 'study_recommendation' THEN
   744→      v_title := 'Study Support Recommendation';
   745→      v_description := COALESCE(p_data->>'summary', 'Based on recent performance, here are study suggestions.');
   746→      v_severity := 'info';
   747→      v_recommendations := ARRAY[
   748→        'Create a quiet study space at home',
   749→        'Set specific study times each day',
   750→        'Use the ExamForge practice tools for revision'
   751→      ];
   752→
   753→    WHEN 'engagement_tip' THEN
   754→      v_title := 'Stay Engaged Tip';
   755→      v_description := COALESCE(p_data->>'summary', 'Tips for supporting your child''s education.');
   756→      v_severity := 'info';
   757→      v_recommendations := ARRAY[
   758→        'Ask about what they learned today',
   759→        'Review homework together',
   760→        'Attend parent-teacher meetings'
   761→      ];
   762→
   763→    WHEN 'milestone' THEN
   764→      v_title := 'Achievement Milestone';
   765→      v_description := COALESCE(p_data->>'summary', 'Your child has reached an academic milestone!');
   766→      v_severity := 'positive';
   767→      v_recommendations := ARRAY['Celebrate this achievement together!'];
   768→
   769→    WHEN 'concern' THEN
   770→      v_title := 'Academic Concern';
   771→      v_description := COALESCE(p_data->>'summary', 'There may be an area requiring attention.');
   772→      v_severity := 'concern';
   773→      v_recommendations := ARRAY[
   774→        'Schedule a meeting with the teacher',
   775→        'Review recent assignments together',
   776→        'Consider additional tutoring support'
   777→      ];
   778→  END CASE;
   779→
   780→  INSERT INTO parent_ai_insights (school_id, parent_id, student_id, insight_type, title, description, severity, recommendations, data_snapshot)
   781→  VALUES (
   782→    (SELECT school_id FROM parent_profiles WHERE user_id = p_parent_id LIMIT 1),
   783→    p_parent_id,
   784→    p_student_id,
   785→    p_insight_type,
   786→    v_title,
   787→    v_description,
   788→    v_severity,
   789→    v_recommendations,
   790→    p_data
   791→  ) RETURNING id INTO v_insight_id;
   792→
   793→  RETURN v_insight_id;
   794→END;
   795→$$ LANGUAGE plpgsql SECURITY DEFINER;
