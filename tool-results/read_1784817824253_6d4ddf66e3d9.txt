    56→  /// Maps a [sb.PostgrestException] to the appropriate domain exception.
    57→  Never _mapPostgrestException(sb.PostgrestException e) {
    58→    AppLogger.error('Postgrest error: ${e.message}', error: e);
    59→    switch (e.code) {
    60→      case 'PGRST116':
    61→        throw NotFoundException(message: e.message);
    62→      case '23505':
    63→        throw const ServerException(
    64→          message: 'A record with this data already exists.',
    65→          statusCode: 409,
    66→        );
    67→      case '23503':
    68→        throw const ServerException(
    69→          message: 'Referenced record not found.',
    70→          statusCode: 404,
    71→        );
    72→      case '42501':
    73→        throw const ForbiddenException(
    74→          message: 'You do not have permission for this action.',
    75→        );
    76→      default:
    77→        throw ServerException(
    78→          message: e.message,
    79→          statusCode: int.tryParse(e.code ?? '') ?? 500,
    80→        );
    81→    }
    82→  }
    83→
    84→  /// Handles generic catch-all exceptions.
    85→  Never _handleGenericException(Object e, String operation) {
    86→    AppLogger.error('Failed to $operation', error: e);
    87→    if (e is sb.AuthException) {
    88→      throw UnauthorizedException(message: e.message);
    89→    }
    90→    throw ServerException(message: e.toString(), statusCode: 500);
    91→  }

    97→  /// Creates a new AI tutor conversation.

   100→  /// fully-hydrated model from the database.
   101→  Future<AiTutorConversationModel> createConversation({
   102→    required String studentId,
   103→    String? schoolId,
   104→    String title = 'New Conversation',
   105→    String? subjectId,
   106→    String? topic,
   107→    String curriculumType = 'nigerian',

   111→      final response = await _supabase
   112→          .from(_aiTutorConversationsTable)
   113→          .insert({
   114→            'student_id': studentId,
   115→            'school_id': schoolId,
   116→            'title': title,
   117→            'subject_id': subjectId,
   118→            'topic': topic,
   119→            'curriculum_type': curriculumType,
   120→          })
   121→          .select()
   122→          .single();
   123→      AppLogger.info('AI tutor conversation created successfully');
   124→      return AiTutorConversationModel.fromJson(response);
   125→    } on sb.PostgrestException catch (e) {
   126→      _mapPostgrestException(e);
   127→    } catch (e) {
   128→      _handleGenericException(e, 'create AI tutor conversation');
   129→    }

   206→  /// Sends a message in an AI tutor conversation and receives an AI response.
   207→  ///
   208→  /// The flow is:
   209→  /// 1. Insert the user message into [ai_tutor_messages].
   210→  /// 2. Call the `ai-tutor-chat` Edge Function with conversation context.
   211→  /// 3. Insert the AI assistant message into [ai_tutor_messages].
   212→  /// 4. Update the conversation's [last_message] and [message_count].
   213→  /// 5. Return the AI response model.
   214→  Future<AiTutorMessageModel> sendMessage({
   215→    required String conversationId,
   216→    required String content,
   217→    String? subjectId,
   218→    String? topic,
   219→    String? curriculumType,
   220→  }) async {
   221→    try {
   222→      AppLogger.info(
   223→        'Sending message in conversation: $conversationId',
   224→      );
   225→
   226→      // 1. Insert the user message
   227→      final userMessageResponse = await _supabase
   228→          .from(_aiTutorMessagesTable)
   229→          .insert({
   230→            'conversation_id': conversationId,
   231→            'role': 'user',
   232→            'content': content,
   233→          })
   234→          .select()
   235→          .single();
   236→
   237→      AppLogger.info('User message inserted');
   238→
   239→      // 2. Call the AI tutor Edge Function for a response
   240→      final aiResponse = await _supabase.functions.invoke(
   241→        _aiTutorChatFunction,
   242→        body: {
   243→          'conversation_id': conversationId,
   244→          'message': content,
   245→          'subject_id': subjectId,
   246→          'topic': topic,
   247→          'curriculum_type': curriculumType,
   248→        },
   249→      );
   250→
   251→      if (aiResponse.status != 200) {
   252→        AppLogger.error(
   253→          'AI tutor edge function returned status: ${aiResponse.status}',
   254→        );
   255→        throw ServerException(
   256→          message: 'AI tutor service returned an error: ${aiResponse.data}',
   257→          statusCode: aiResponse.status,
   258→        );
   259→      }
   260→
   261→      final aiContent = aiResponse.data['content'] as String? ??
   262→          'I apologize, but I was unable to generate a response. Please try again.';
   263→      final aiMetadata = aiResponse.data['metadata'] as Map<String, dynamic>?;
   264→
   265→      // 3. Insert the AI assistant message
   266→      final assistantMessageResponse = await _supabase
   267→          .from(_aiTutorMessagesTable)
   268→          .insert({
   269→            'conversation_id': conversationId,
   270→            'role': 'assistant',
   271→            'content': aiContent,
   272→            'metadata': aiMetadata ?? {},
   273→          })
   274→          .select()
   275→          .single();
   276→
   277→      // 4. Update the conversation's last_message and message_count
   278→      await _supabase
   279→          .from(_aiTutorConversationsTable)
   280→          .update({
   281→            'last_message': aiContent.length > 100
   282→                ? '${aiContent.substring(0, 100)}...'
   283→                : aiContent,
   284→          })
   285→          .eq('id', conversationId);
   286→
   287→      AppLogger.info('AI response message inserted');
   288→
   289→      return AiTutorMessageModel.fromJson(assistantMessageResponse);
   290→    } on sb.PostgrestException catch (e) {
   291→      _mapPostgrestException(e);
   292→    } catch (e) {
   293→      if (e is ServerException) rethrow;
   294→      _handleGenericException(e, 'send AI tutor message');
   295→    }
   296→  }

   323→  /// Creates a new practice session and fetches random questions from
   324→  /// the question bank matching the given filters.
   325→  ///
   326→  /// The session is created in `in_progress` status. Questions are
   327→  /// selected from the [questions] table using random ordering and
   328→  /// subject/difficulty filters.
   329→  Future<PracticeSessionModel> createPracticeSession({
   330→    required String studentId,
   331→    String? schoolId,
   332→    String? subjectId,
   333→    String? topicId,
   334→    String difficulty = 'medium',
   335→    String mode = 'untimed',
   336→    int? timeLimitSec,
   337→    int questionCount = 10,
   338→  }) async {
   339→    try {
   340→      AppLogger.info(
   341→        'Creating practice session for student: $studentId',
   342→      );
   343→
   344→      // Insert the practice session
   345→      final sessionResponse = await _supabase
   346→          .from(_practiceSessionsTable)
   347→          .insert({
   348→            'student_id': studentId,
   349→            'school_id': schoolId,
   350→            'subject_id': subjectId,
   351→            'topic_id': topicId,
   352→            'difficulty': difficulty,
   353→            'mode': mode,
   354→            'time_limit_sec': timeLimitSec,
   355→            'total_questions': questionCount,
   356→            'status': 'in_progress',
   357→            'started_at': DateTime.now().toUtc().toIso8601String(),
   358→          })
   359→          .select()
   360→          .single();
   361→
   362→      final sessionId = sessionResponse['id'] as String;
   363→
   364→      // Fetch random questions from the question bank
   365→      var query = _supabase
   366→          .from(_questionsTable)
   367→          .select('id')
   368→          .eq('status', 'approved');
   369→
   370→      if (subjectId != null) {
   371→        query = query.eq('subject_id', subjectId);
   372→      }
   373→      if (difficulty != 'mixed') {
   374→        query = query.eq('difficulty', difficulty);
   375→      }
   376→
   377→      // Use limit and random ordering via Supabase
   378→      final questionIds = await query
   379→          .limit(questionCount * 3) // Fetch extra for randomness
   380→          .order('id');
   381→
   382→      // Randomly select the desired number of questions
   383→      final selectedQuestions = <Map<String, dynamic>>[];
   384→      if (questionIds.isNotEmpty) {
   385→        final shuffled = List<Map<String, dynamic>>.from(questionIds)..shuffle();
   386→        selectedQuestions.addAll(
   387→          shuffled.take(min(questionCount, shuffled.length)),
   388→        );
   389→      }
   390→
   391→      // Store the question IDs as the session's question set
   392→      // (practice_answers will be created as students answer)
   393→      // Update the total_questions to match actual available count
   394→      if (selectedQuestions.length != questionCount) {
   395→        await _supabase
   396→            .from(_practiceSessionsTable)
   397→            .update({'total_questions': selectedQuestions.length})
   398→            .eq('id', sessionId);
   399→        sessionResponse['total_questions'] = selectedQuestions.length;
   400→      }
   401→
   402→      AppLogger.info(
   403→        'Practice session created: $sessionId with ${selectedQuestions.length} questions',
   404→      );
   405→
   406→      return PracticeSessionModel.fromJson(sessionResponse);
   407→    } on sb.PostgrestException catch (e) {
   408→      _mapPostgrestException(e);
   409→    } catch (e) {
   410→      _handleGenericException(e, 'create practice session');
   411→    }
   412→  }

   544→  /// Completes a practice session.
   545→  ///
   546→  /// Sets the status to `completed`, calculates the final score
   547→  /// percentage, and records the completion timestamp.
   548→  Future<PracticeSessionModel> completePracticeSession({
   549→    required String sessionId,
   550→  }) async {
   551→    try {
   552→      AppLogger.info('Completing practice session: $sessionId');
   553→
   554→      // Fetch current session to compute score
   555→      final current = await _supabase
   556→          .from(_practiceSessionsTable)
   557→          .select('total_questions, correct_count')
   558→          .eq('id', sessionId)
   559→          .single();
   560→
   561→      final totalQ = current['total_questions'] as int? ?? 0;
   562→      final correctQ = current['correct_count'] as int? ?? 0;
   563→      final scorePct = totalQ > 0 ? (correctQ / totalQ) * 100 : 0.0;
   564→
   565→      final response = await _supabase
   566→          .from(_practiceSessionsTable)
   567→          .update({
   568→            'status': 'completed',
   569→            'score_pct': scorePct,
   570→            'completed_at': DateTime.now().toUtc().toIso8601String(),
   571→          })
   572→          .eq('id', sessionId)
   573→          .select()
   574→          .single();
   575→
   576→      AppLogger.info(
   577→        'Practice session completed: $sessionId (score: ${scorePct.toStringAsFixed(1)}%)',
   578→      );
   579→      return PracticeSessionModel.fromJson(response);
   580→    } on sb.PostgrestException catch (e) {
   581→      _mapPostgrestException(e);
   582→    } catch (e) {
   583→      _handleGenericException(e, 'complete practice session');
   584→    }
   585→  }

   960→  /// Uploads a document for AI chat processing.
   961→  ///
   962→  /// The flow is:
   963→  /// 1. Upload the file to Supabase Storage bucket `student-documents`.
   964→  /// 2. Create a `document_chats` record.
   965→  /// 3. Call the `process-document` Edge Function for text extraction
   966→  ///    (runs asynchronously).
   967→  Future<DocumentChatModel> uploadDocument({
   968→    required String studentId,
   969→    String? schoolId,
   970→    required String fileName,
   971→    required Uint8List fileBytes,
   972→    required String fileFormat,
   973→    int? fileSize,
   974→  }) async {
   975→    try {
   976→      AppLogger.info(
   977→        'Uploading document for student: $studentId, file: $fileName',
   978→      );
   979→
   980→      // 1. Upload file to Supabase Storage
   981→      final storagePath =
   982→          '$studentId/${DateTime.now().millisecondsSinceEpoch}_$fileName';
   983→      await _supabase.storage
   984→          .from(_studentDocumentsBucket)
   985→          .uploadBinary(storagePath, fileBytes);
   986→
   987→      // Get the public URL for the uploaded file
   988→      final fileUrl = _supabase.storage
   989→          .from(_studentDocumentsBucket)
   990→          .getPublicUrl(storagePath);
   991→
   992→      // 2. Create a document_chats record
   993→      final response = await _supabase
   994→          .from(_documentChatsTable)
   995→          .insert({
   996→            'student_id': studentId,
   997→            'school_id': schoolId,
   998→            'file_name': fileName,
   999→            'file_url': fileUrl,
  1000→            'file_size': fileSize,
  1001→            'file_format': fileFormat,
  1002→            'status': 'processing',
  1003→          })
  1004→          .select()
  1005→          .single();
  1006→
  1007→      final documentId = response['id'] as String;
  1008→
  1009→      // 3. Trigger the process-document edge function (async)
  1010→      _supabase.functions.invoke(
  1011→        _processDocumentFunction,
  1012→        body: {
  1013→          'document_id': documentId,
  1014→          'file_url': fileUrl,
  1015→          'file_format': fileFormat,
  1016→        },
  1017→      ).catchError((error) {
  1018→        AppLogger.error(
  1019→          'Process document edge function failed',
  1020→          error: error,
  1021→        );
  1022→      });
  1023→
  1024→      AppLogger.info('Document uploaded and processing started: $documentId');
  1025→      return DocumentChatModel.fromJson(response);
  1026→    } on sb.PostgrestException catch (e) {
  1027→      _mapPostgrestException(e);
  1028→    } on sb.StorageException catch (e) {
  1029→      AppLogger.error('Storage upload failed', error: e);
  1030→      throw ServerException(
  1031→        message: 'File upload failed: ${e.message}',
  1032→        statusCode: 500,
  1033→      );
  1034→    } catch (e) {
  1035→      if (e is ServerException) rethrow;
  1036→      _handleGenericException(e, 'upload document');
  1037→    }
  1038→  }

  1165→  /// Deletes a document, its chat messages, and the stored file.
  1166→  Future<void> deleteDocument({required String documentId}) async {
  1167→    try {
  1168→      AppLogger.info('Deleting document: $documentId');
  1169→
  1170→      // Fetch the document to get the storage path
  1171→      final doc = await _supabase
  1172→          .from(_documentChatsTable)
  1173→          .select('file_url, student_id')
  1174→          .eq('id', documentId)
  1175→          .single();
  1176→
  1177→      // Delete from database (cascade deletes messages)
  1178→      await _supabase
  1179→          .from(_documentChatsTable)
  1180→          .delete()
  1181→          .eq('id', documentId);
  1182→
  1183→      // Attempt to remove the file from storage (best-effort)
  1184→      final fileUrl = doc['file_url'] as String?;
  1185→      if (fileUrl != null) {
  1186→        try {
  1187→          final studentId = doc['student_id'] as String;
  1188→          // Extract the storage path from the URL
  1189→          final uri = Uri.parse(fileUrl);
  1190→          final pathSegments = uri.pathSegments;
  1191→          final storagePath = pathSegments.skip(2).join('/');
  1192→          await _supabase.storage
  1193→              .from(_studentDocumentsBucket)
  1194→              .remove([storagePath]);
  1195→        } catch (storageError) {

  1458→  /// Rates a flashcard using the SM-2 spaced repetition algorithm.
  1459→  ///
  1460→  /// The rating can be one of: `again`, `hard`, `good`, `easy`.
  1461→  ///
  1462→  /// **SM-2 Algorithm Implementation:**
  1463→  /// - `again`: repetitions = 0, interval = 1 min
  1464→  /// - `hard`:  easeFactor -= 0.15, interval = interval * 1.2
  1465→  /// - `good`:  interval = interval * easeFactor
  1466→  /// - `easy`:  easeFactor += 0.15, interval = interval * easeFactor * 1.3
  1467→  /// - easeFactor is clamped between 1.3 and 3.0
  1468→  /// - If repetitions == 0: interval = 1 day
  1469→  /// - If repetitions == 1: interval = 6 days
  1470→  /// - Else: interval = previousInterval * easeFactor
  1471→  Future<FlashcardModel> rateFlashcard({
  1472→    required String cardId,
  1473→    required String rating,
  1474→  }) async {
  1475→    try {
  1476→      AppLogger.info('Rating flashcard: $cardId with rating: $rating');
  1477→
  1478→      // 1. Fetch the current card state
  1479→      final current = await _supabase
  1480→          .from(_flashcardsTable)
  1481→          .select()
  1482→          .eq('id', cardId)
  1483→          .single();
  1484→
  1485→      var repetitions = current['repetitions'] as int? ?? 0;
  1486→      var easeFactor = (current['ease_factor'] as num?)?.toDouble() ?? 2.5;
  1487→      var interval = (current['interval'] as num?)?.toDouble() ?? 0.0;
  1488→      final previousInterval = interval;
  1489→
  1490→      // 2. Apply SM-2 algorithm based on rating
  1491→      switch (rating) {
  1492→        case 'again':
  1493→          repetitions = 0;
  1494→          interval = 1.0 / 1440.0; // 1 minute in days (~0.000694 days)
  1495→          // Alternatively, we can set next_review to now + 1 minute directly
  1496→          break;
  1497→        case 'hard':
  1498→          easeFactor = max(1.3, easeFactor - 0.15);
  1499→          interval = previousInterval * 1.2;
  1500→          if (interval < 1) interval = 1;
  1501→          repetitions = repetitions + 1;
  1502→          break;
  1503→        case 'good':
  1504→          if (repetitions == 0) {
  1505→            interval = 1; // 1 day
  1506→          } else if (repetitions == 1) {
  1507→            interval = 6; // 6 days
  1508→          } else {
  1509→            interval = previousInterval * easeFactor;
  1510→          }
  1511→          repetitions = repetitions + 1;
  1512→          break;
  1513→        case 'easy':
  1514→          easeFactor = min(3.0, easeFactor + 0.15);
  1515→          if (repetitions == 0) {
  1516→            interval = 1;
  1517→          } else if (repetitions == 1) {
  1518→            interval = 6;
  1519→          } else {
  1520→            interval = previousInterval * easeFactor;
  1521→          }
  1522→          interval = interval * 1.3;
  1523→          repetitions = repetitions + 1;
  1524→          break;
  1525→        default:
  1526→          throw ValidationException(
  1527→            message: 'Invalid flashcard rating: $rating',
  1528→            fieldErrors: {'rating': 'Must be one of: again, hard, good, easy'},
  1529→          );
  1530→      }
  1531→
  1532→      // 3. Clamp easeFactor
  1533→      easeFactor = easeFactor.clamp(1.3, 3.0);
  1534→
  1535→      // 4. Compute next review date
  1536→      final nextReview = DateTime.now().add(Duration(
  1537→        milliseconds: (interval * 24 * 60 * 60 * 1000).round(),
  1538→      ),);
  1539→
  1540→      // 5. Persist the updated SM-2 state
  1541→      final response = await _supabase
  1542→          .from(_flashcardsTable)
  1543→          .update({
  1544→            'repetitions': repetitions,
  1545→            'ease_factor': easeFactor,
  1546→            'interval': interval,
  1547→            'next_review': nextReview.toUtc().toIso8601String(),
  1548→            'last_reviewed': DateTime.now().toUtc().toIso8601String(),
  1549→          })
  1550→          .eq('id', cardId)
  1551→          .select()
  1552→          .single();
  1553→
  1554→      AppLogger.info(
  1555→        'Flashcard rated: $cardId → next review in ${interval.toStringAsFixed(1)} days',
  1556→      );
  1557→      return FlashcardModel.fromJson(response);
  1558→    } on sb.PostgrestException catch (e) {
  1559→      _mapPostgrestException(e);
  1560→    } catch (e) {
  1561→      if (e is ValidationException) rethrow;
  1562→      _handleGenericException(e, 'rate flashcard');
  1563→    }
  1564→  }
