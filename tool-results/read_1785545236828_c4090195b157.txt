   508→/// Stream provider that emits the current Supabase [sb.AuthState].
   509→///
   510→/// Automatically reconnects on transient failures and emits the latest
   511→/// state to any listener.
   512→///
   513→/// Includes a 10-second timeout: if the stream doesn't emit within 10s
   514→/// (e.g. on Flutter Web where the initial session fetch may silently fail
   515→/// due to CORS or network issues), we emit an empty-auth state so the
   516→/// router can proceed to the login screen instead of hanging forever.
   517→final authStateProvider = StreamProvider<sb.AuthState>((ref) {
   518→  final supabaseClient = ref.watch(supabaseClientProvider);
   519→  final authStream = supabaseClient.auth.onAuthStateChange;
   520→
   521→  // Timeout wrapper: if the stream doesn't emit within 10 seconds,
   522→  // emit a synthetic "no session" state so the app can proceed.
   523→  return authStream.timeout(
   524→    const Duration(seconds: 10),
   525→    onTimeout: (sink) {
   526→      AppLogger.warning('authStateProvider: stream timed out after 10s — '
   527→          'emitting empty auth state');
   528→      sink.add(sb.AuthState(
   529→        event: sb.AuthChangeEvent.initialSession,
   530→        session: null,
   531→      ));
   532→      sink.close();
   533→    },
   534→  );
   535→});
