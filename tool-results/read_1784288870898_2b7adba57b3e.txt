    22→/// The main Teacher Workspace Dashboard page.
    23→///
    24→/// This is the entry point for the workspace module, presenting a modern
    25→/// SaaS-style dashboard with:
    26→/// - Welcome section with teacher name and current date
    27→/// - Stats cards row (Lesson Plans, Worksheets, Assignments, Resources)
    28→/// - Quick AI Actions grid (8 action buttons)
    29→/// - Today's Schedule timeline
    30→/// - Upcoming Events
    31→/// - Recently Generated AI Content
    32→/// - Draft Lesson Plans with "Generate Questions" buttons
    33→///
    34→/// Uses [ConsumerStatefulWidget] with a private [_State] class pattern,
    35→/// loads data in [initState] via [WidgetsBinding.instance.addPostFrameCallback],
    36→/// and renders a responsive layout using [CustomScrollView] with slivers.
    37→class TeacherWorkspaceDashboardPage extends ConsumerStatefulWidget {
    38→  const TeacherWorkspaceDashboardPage({super.key});
    39→
    40→  @override
    41→  ConsumerState<TeacherWorkspaceDashboardPage> createState() => _State();
    42→}
    43→
    44→class _State extends ConsumerState<TeacherWorkspaceDashboardPage> {
    45→  // ─── Lifecycle ──────────────────────────────────────────────────────
    46→
    47→  @override
    48→  void initState() {
    49→    super.initState();
    50→    WidgetsBinding.instance.addPostFrameCallback((_) {
    51→      ref.read(workspaceDashboardProvider.notifier).loadDashboard();
    52→    });
    53→  }
    54→
    55→  // ─── Build ──────────────────────────────────────────────────────────
    56→
    57→  @override
    58→  Widget build(BuildContext context) {
    59→    final dashboardState = ref.watch(workspaceDashboardProvider);
    60→
    61→    return Scaffold(
    62→      appBar: const AppAppBar(title: 'Workspace'),
    63→      body: _buildBody(context, dashboardState),
    64→    );
    65→  }
    66→
    67→  // ─── Body Router ────────────────────────────────────────────────────
    68→
    69→  Widget _buildBody(BuildContext context, WorkspaceDashboardState state) {
    70→    // Loading state
    71→    if (state.isLoading && state.dashboardSummary == null) {
    72→      return const Center(child: AppLoadingSpinner());
    73→    }
    74→
    75→    // Error state
    76→    if (state.error != null && state.dashboardSummary == null) {
    77→      return AppErrorState.genericError(
    78→        message: state.error,
    79→        onRetry: () =>
    80→            ref.read(workspaceDashboardProvider.notifier).loadDashboard(),
    81→      );
    82→    }
    83→
    84→    // Empty state
    85→    final summary = state.dashboardSummary;
    86→    if (summary == null) {
    87→      return AppEmptyState.noData(
    88→        title: 'No Workspace Data',
    89→        subtitle: 'Your workspace dashboard will appear here once loaded.',
    90→        actionLabel: 'Refresh',
    91→        onAction: () =>
    92→            ref.read(workspaceDashboardProvider.notifier).loadDashboard(),
    93→      );
    94→    }
    95→
    96→    // Success — render dashboard
    97→    return RefreshIndicator(
    98→      onRefresh: () =>
    99→          ref.read(workspaceDashboardProvider.notifier).refresh(),
   100→      child: CustomScrollView(
   101→        physics: const AlwaysScrollableScrollPhysics(),
   102→        slivers: [
   103→          // ─── Welcome Section ──────────────────────────────────────
   104→          SliverToBoxAdapter(child: _buildWelcomeSection(context)),
   105→
   106→          // ─── Stats Cards ──────────────────────────────────────────
   107→          SliverToBoxAdapter(child: _buildStatsCards(context, summary)),
   108→
   109→          // ─── Quick AI Actions ─────────────────────────────────────
   110→          SliverToBoxAdapter(child: _buildQuickActionsSection(context)),
   111→
   112→          // ─── Schedule & Events (2-column on desktop) ──────────────
   113→          SliverToBoxAdapter(
   114→            child: _buildScheduleAndEvents(context, summary),
   115→          ),
   116→
   117→          // ─── Recently Generated AI Content ────────────────────────
   118→          SliverToBoxAdapter(
   119→            child: _buildRecentAiContent(context, summary),
   120→          ),
   121→
   122→          // ─── Draft Lesson Plans ───────────────────────────────────
   123→          SliverToBoxAdapter(
   124→            child: _buildDraftLessonPlans(context, summary),
   125→          ),
   126→
   127→          // Bottom padding
   128→          const SliverPadding(padding: EdgeInsets.only(bottom: Spacings.xxl)),
   129→        ],
   130→      ),
   131→    );
   132→  }

   626→  IconData _typeIcon(String type) {
   627→    switch (type) {
   628→      case 'lesson_plan':
   629→        return Icons.auto_stories_rounded;
   630→      case 'worksheet':
   631→        return Icons.assignment_rounded;
   632→      case 'scheme_of_work':
   633→        return Icons.calendar_month_rounded;
   634→      case 'assignment':
   635→        return Icons.task_rounded;
   636→      default:
   637→        return Icons.smart_toy_rounded;
   638→    }

   640→
   641→  String _formatRelativeTime(String isoDate) {
   642→    final dt = DateTime.tryParse(isoDate);
   643→    if (dt == null) return '';
   644→    final diff = DateTime.now().difference(dt);
   645→    if (diff.inMinutes < 1) return 'Just now';
   646→    if (diff.inMinutes < 60) return '${diff.inMinutes}m ago';
   647→    if (diff.inHours < 24) return '${diff.inHours}h ago';
   648→    return '${diff.inDays}d ago';
   649→  }

   680→        ),
   681→      ),
   682→      child: Padding(
   683→        padding: const EdgeInsets.all(Spacings.lg),
   684→        child: Column(
   685→          crossAxisAlignment: CrossAxisAlignment.start,
   686→          children: [
   687→            Row(

   716→                          [subject, topic].whereType<String>().join(' · '),
   717→                          style: tt.bodySmall?.copyWith(
   718→                            color: cs.onSurfaceVariant,
   719→                          ),
   720→                          maxLines: 1,
   721→                          overflow: TextOverflow.ellipsis,
   722→                        ),
   723→                    ],
   724→                  ),
   725→                ),
   726→                // Draft badge
   727→                Container(
   728→                  padding: const EdgeInsets.symmetric(
   729→                    horizontal: Spacings.sm,
   730→                    vertical: Spacings.xs,

   774→  String _formatRelativeTime(String isoDate) {
   775→    final dt = DateTime.tryParse(isoDate);
   776→    if (dt == null) return '';
   777→    final diff = DateTime.now().difference(dt);
   778→    if (diff.inMinutes < 1) return 'just now';
   779→    if (diff.inMinutes < 60) return '${diff.inMinutes}m ago';
   780→    if (diff.inHours < 24) return '${diff.inHours}h ago';
   781→    return '${diff.inDays}d ago';
   782→  }
