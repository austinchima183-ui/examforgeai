    16→Color _priorityColor(TicketPriority priority) {
    17→  switch (priority) {
    18→    case TicketPriority.low:
    19→      return Colors.grey;
    20→    case TicketPriority.medium:
    21→      return AppColors.info;
    22→    case TicketPriority.high:
    23→      return Colors.orange;
    24→    case TicketPriority.urgent:
    25→      return AppColors.error;
    26→    case TicketPriority.critical:
    27→      return AppColors.errorDark;
    28→  }
    29→}
    30→
    31→Color _statusColor(TicketStatus status) {
    32→  switch (status) {
    33→    case TicketStatus.open:
    34→      return AppColors.info;
    35→    case TicketStatus.inProgress:
    36→      return Colors.orange;
    37→    case TicketStatus.waitingOnUser:
    38→      return AppColors.warning;
    39→    case TicketStatus.waitingOnThirdParty:
    40→      return AppColors.warning;
    41→    case TicketStatus.resolved:
    42→      return AppColors.success;
    43→    case TicketStatus.closed:
    44→      return Colors.grey;
    45→    case TicketStatus.reopened:
    46→      return AppColors.info;
    47→  }
    48→}
    49→
    50→String _formatTimeAgo(DateTime dt) {
    51→  final now = DateTime.now();
    52→  final diff = now.difference(dt);
    53→  if (diff.inMinutes < 1) return 'Just now';
    54→  if (diff.inHours < 1) return '${diff.inMinutes}m ago';
    55→  if (diff.inDays < 1) return '${diff.inHours}h ago';
    56→  if (diff.inDays < 7) return '${diff.inDays}d ago';
    57→  return '${dt.day.toString().padLeft(2, '0')}/${dt.month.toString().padLeft(2, '0')}/${dt.year}';
    58→}

    64→enum FeedbackType {

    66→  featureRequest('feature_request', 'Feature Request', Icons.lightbulb_outline),
    67→  complaint('complaint', 'Complaint', Icons.sentiment_dissatisfied_outlined),
    68→  compliment('compliment', 'Compliment', Icons.sentiment_satisfied_outlined),

    71→
    72→  const FeedbackType(this.value, this.label, this.icon);
    73→  final String value;
    74→  final String label;
    75→  final IconData icon;
    76→
    77→  static FeedbackType fromString(String v) => FeedbackType.values.firstWhere(
    78→        (e) => e.value == v,
    79→        orElse: () => FeedbackType.suggestion,
    80→      );
    81→}

   136→    _searchController.dispose();

   140→  void _onTabChanged() {
   141→    if (!_tabController.indexIsChanging) {
   142→      _loadTabData(_tabController.index);
   143→    }
   144→  }
   145→
   146→  void _loadData() {
   147→    _loadTabData(_tabController.index);
   148→  }
   149→
   150→  void _loadTabData(int index) {
   151→    final notifier = ref.read(supportCenterProvider.notifier);
   152→    switch (index) {
   153→      case 0:
   154→        notifier.loadTickets(
   155→          status: _filterStatus,
   156→          priority: _filterPriority,
   157→          search: _searchController.text.trim().isEmpty
   158→              ? null
   159→              : _searchController.text.trim(),
   160→        );
   161→        break;
   162→      case 1:
   163→        // Feedback is loaded alongside tickets; no separate load method
   164→        notifier.loadTickets();
   165→        break;
   166→    }
   167→  }

   215→          ? const Center(
   216→              child: AppLoadingSpinner(size: AppLoadingSpinnerSize.large))

   220→                  children: [
   221→                    // ─── Tab Bar ──────────────────────────────────────
   222→                    Container(
   223→                      decoration: BoxDecoration(

   226→                          bottom: BorderSide(
   227→                            color: cs.outlineVariant,
   228→                            width: 1,
   229→                          ),
   230→                        ),
   231→                      ),
   232→                      child: TabBar(
   233→                        controller: _tabController,
   234→                        tabs: _tabs,
   235→                        labelColor: cs.primary,
   236→                        unselectedLabelColor:
   237→                            cs.onSurface.withValues(alpha: 0.6),
   238→                        indicatorColor: cs.primary,
   239→                        indicatorSize: TabBarIndicatorSize.tab,

   257→                            },

   259→                              setState(() => _filterPriority = priority);
   260→                              _loadData();
   261→                            },
   262→                            onCategoryChanged: (category) {
   263→                              setState(() => _filterCategory = category);
   264→                              _loadData();
   265→                            },
   266→                          ),
   267→                          _FeedbackTab(
   268→                            filterFeedbackType: _filterFeedbackType,
   269→                            onFeedbackTypeChanged: (type) {

   273→                        ],
   274→                      ),
   275→                    ),
   276→                  ],
