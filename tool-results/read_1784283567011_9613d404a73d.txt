   305→   305→  @override
   306→   306→  Future<List<ExamModel>> getExams(Map<String, dynamic> filters) async {
   307→   307→    try {
   308→   308→      var query = _supabaseClient.from(_examsTable).select();
   309→   309→
   310→   310→      if (filters['school_id'] != null) {
   311→   311→        query = query.eq('school_id', filters['school_id'] as String);
   312→   312→      }
   313→   313→      if (filters['subject_id'] != null) {
   314→   314→        query = query.eq('subject_id', filters['subject_id'] as String);
   315→   315→      }
   316→   316→      if (filters['status'] != null) {
   317→   317→        query = query.eq('status', filters['status'] as String);
   318→   318→      }
   319→   319→      if (filters['class_id'] != null) {
   320→   320→        query = query.eq('class_id', filters['class_id'] as String);
   321→   321→      }
   322→   322→
   323→   323→      final page = filters['page'] as int? ?? 1;
   324→   324→      final perPage = filters['per_page'] as int? ?? filters['perPage'] as int? ?? 20;
   325→   325→      final offset = (page - 1) * perPage;
   326→   326→
   327→   327→      query = query.range(offset, offset + perPage - 1);
   328→   328→      query = query.order('created_at', ascending: false);
   329→   329→
   330→   330→      final response = await query;
   331→   331→
   332→   332→      return response
   333→   333→          .map<ExamModel>((json) => ExamModel.fromJson(json))
   334→   334→          .toList();
   335→   335→    } on sb.PostgrestException catch (e) {
   336→   336→      AppLogger.error('Get exams failed', error: e);
   337→   337→      throw ServerException(
   338→   338→        message: e.message,
   339→   339→        statusCode: int.tryParse(e.code ?? '') ?? 500,
   340→   340→        data: e.details,
   341→   341→      );
   342→   342→    } catch (e) {
   343→   343→      AppLogger.error('Get exams unexpected error', error: e);
   344→   344→      throw ServerException(
   345→   345→        message: 'Failed to get exams: $e',
   346→   346→        statusCode: 500,
   347→   347→      );
   348→   348→    }
   349→   349→  }

   413→   413→  Future<ExamModel> cloneExam(String examId) async {
   414→   414→    try {
   415→   415→      // Fetch the original exam with details
   416→   416→      final original = await getExamWithDetails(examId);
   417→   417→
   418→   418→      final now = DateTime.now();
   419→   419→      final cloneData = {
   420→   420→        'school_id': original.schoolId,
   421→   421→        'created_by': original.createdBy,
   422→   422→        'title': '${original.title} (Copy)',
   423→   423→        'description': original.description,
   424→   424→        'subject_id': original.subjectId,
   425→   425→        'class_id': original.classId,
   426→   426→        'academic_session_id': original.academicSessionId,
   427→   427→        'exam_type': original.examType,
   428→   428→        'status': 'draft',
   429→   429→        'start_time': now.add(const Duration(days: 7)).toIso8601String(),
   430→   430→        'end_time': now.add(const Duration(days: 7, hours: 2)).toIso8601String(),
   431→   431→        'time_limit_minutes': original.timeLimitMinutes,
   432→   432→        'total_marks': original.totalMarks,
   433→   433→        'pass_mark': original.passMark,
   434→   434→        'pass_mark_type': original.passMarkType,
   435→   435→        'instructions': original.instructions,
   436→   436→        'allowed_attempts': original.allowedAttempts,
   437→   437→        'negative_marking_enabled': original.negativeMarkingEnabled,
   438→   438→        'negative_mark_value': original.negativeMarkValue,
   439→   439→        'grace_period_minutes': original.gracePeriodMinutes,
   440→   440→        'auto_submit': original.autoSubmit,
   441→   441→        'randomize_questions': original.randomizeQuestions,
   442→   442→        'randomize_options': original.randomizeOptions,
   443→   443→        'show_results': original.showResults,
   444→   444→        'show_correct_answers': original.showCorrectAnswers,
   445→   445→        'show_explanations': original.showExplanations,
   446→   446→        'is_template': original.isTemplate,
   447→   447→        'template_id': original.id,
   448→   448→        'max_students': original.maxStudents,
   449→   449→        'ip_restriction': original.ipRestriction,
   450→   450→        'require_full_screen': original.requireFullScreen,
   451→   451→        'allow_resume': original.allowResume,
   452→   452→        'browser_lockdown': original.browserLockdown,
   453→   453→        'metadata': original.metadata,
   454→   454→      };
   455→   455→
   456→   456→      final clonedExam = await createExam(cloneData);
   457→   457→
   458→   458→      // Clone sections
   459→   459→      for (final section in original.sections) {
   460→   460→        await _supabaseClient.from(_examSectionsTable).insert({
   461→   461→          'exam_id': clonedExam.id,
   462→   462→          'title': section.title,
   463→   463→          'description': section.description,
   464→   464→          'instructions': section.instructions,
   465→   465→          'sort_order': section.sortOrder,
   466→   466→          'time_limit_minutes': section.timeLimitMinutes,
   467→   467→          'randomize_questions': section.randomizeQuestions,
   468→   468→        });
   469→   469→      }
   470→   470→
   471→   471→      // Clone questions
   472→   472→      for (final question in original.questions) {
   473→   473→        await _supabaseClient.from(_examQuestionsTable).insert({
   474→   474→          'exam_id': clonedExam.id,
   475→   475→          'section_id': question.sectionId,
   476→   476→          'question_id': question.questionId,
   477→   477→          'sort_order': question.sortOrder,
   478→   478→          'marks': question.marks,
   479→   479→          'negative_marks': question.negativeMarks,
   480→   480→          'is_compulsory': question.isCompulsory,
   481→   481→        });
   482→   482→      }
   483→   483→
   484→   484→      return clonedExam;
   485→   485→    } on ServerException {
   486→   486→      rethrow;
   487→   487→    } on sb.PostgrestException catch (e) {
   488→   488→      AppLogger.error('Clone exam failed', error: e);

   724→   724→  @override
   725→   725→  Future<ExamResultModel> submitAttempt(
   726→   726→    String attemptId, {
   727→   727→    String submissionType = 'manual',
   728→   728→  }) async {
   729→   729→    try {
   730→   730→      final now = DateTime.now();
   731→   731→
   732→   732→      // Fetch the attempt
   733→   733→      final attemptResponse = await _supabaseClient
   734→   734→          .from(_examAttemptsTable)
   735→   735→          .select()
   736→   736→          .eq('id', attemptId)
   737→   737→          .single();
   738→   738→
   739→   739→      final attempt = ExamAttemptModel.fromJson(attemptResponse);
   740→   740→
   741→   741→      // Fetch all answers for this attempt
   742→   742→      final answersResponse = await _supabaseClient
   743→   743→          .from(_studentAnswersTable)
   744→   744→          .select()
   745→   745→          .eq('attempt_id', attemptId);
   746→   746→
   747→   747→      final answers = answersResponse
   748→   748→          .map<StudentAnswerModel>((a) => StudentAnswerModel.fromJson(a))
   749→   749→          .toList();
   750→   750→
   751→   751→      // Calculate total marks
   752→   752→      double totalMarks = 0;
   753→   753→      for (final answer in answers) {
   754→   754→        totalMarks += answer.marksAwarded - answer.marksDeducted;
   755→   755→      }
   756→   756→
   757→   757→      // Calculate time spent
   758→   758→      final startedAt = attempt.startedAt;
   759→   759→      final timeSpent = now.difference(startedAt).inSeconds;
   760→   760→
   761→   761→      // Determine if all objective questions are graded
   762→   762→      final hasUngradedSubjective = answers.any(
   763→   763→        (a) => a.isCorrect == null && a.marksAwarded == 0,
   764→   764→      );
   765→   765→      final gradingStatus =
   766→   766→          hasUngradedSubjective ? 'partially_graded' : 'auto_graded';
   767→   767→
   768→   768→      // Update the attempt
   769→   769→      await _supabaseClient
   770→   770→          .from(_examAttemptsTable)
   771→   771→          .update({
   772→   772→            'status': submissionType == 'auto_submit' ? 'auto_submitted' : 'submitted',
   773→   773→            'submitted_at': now.toIso8601String(),
   774→   774→            'submission_type': submissionType,
   775→   775→            'time_spent_seconds': timeSpent,
   776→   776→            'total_marks': totalMarks,
   777→   777→            'grading_status': gradingStatus,
   778→   778→            'last_activity_at': now.toIso8601String(),
   779→   779→          })
   780→   780→          .eq('id', attemptId);
   781→   781→
   782→   782→      // Deactivate the session
   783→   783→      await _supabaseClient
   784→   784→          .from(_examSessionsTable)
   785→   785→          .update({
   786→   786→            'is_active': false,
   787→   787→            'connection_status': 'disconnected',
   788→   788→          })
   789→   789→          .eq('attempt_id', attemptId);
   790→   790→
   791→   791→      // Calculate score percentage
   792→   792→      // Fetch exam total marks
   793→   793→      final examResponse = await _supabaseClient
   794→   794→          .from(_examsTable)
   795→   795→          .select('total_marks, pass_mark, pass_mark_type')
   796→   796→          .eq('id', attempt.examId)
   797→   797→          .single();
   798→   798→
   799→   799→      final examTotalMarks =
   800→   800→          (examResponse['total_marks'] as num?)?.toDouble() ?? 0.0;
   801→   801→      final passMark =
   802→   802→          (examResponse['pass_mark'] as num?)?.toDouble() ?? 0.0;
   803→   803→      final passMarkType =
   804→   804→          examResponse['pass_mark_type'] as String? ?? 'percentage';
   805→   805→
   806→   806→      final scorePercentage = examTotalMarks > 0
   807→   807→          ? (totalMarks / examTotalMarks) * 100
   808→   808→          : 0.0;
   809→   809→
   810→   810→      final effectivePassMark = passMarkType == 'percentage'
   811→   811→          ? passMark
   812→   812→          : (examTotalMarks > 0 ? (passMark / examTotalMarks) * 100 : 0.0);
   813→   813→
   814→   814→      final isPassed = scorePercentage >= effectivePassMark;
   815→   815→
   816→   816→      // Create or update result
   817→   817→      final existingResult = await _supabaseClient
   818→   818→          .from(_examResultsTable)
   819→   819→          .select('id')
   820→   820→          .eq('attempt_id', attemptId)
   821→   821→          .maybeSingle();
   822→   822→
   823→   823→      final resultData = {
   824→   824→        'exam_id': attempt.examId,
   825→   825→        'student_id': attempt.studentId,
   826→   826→        'attempt_id': attemptId,
   827→   827→        'total_marks': totalMarks,
   828→   828→        'total_possible': examTotalMarks,
   829→   829→        'score_percentage': scorePercentage,
   830→   830→        'is_passed': isPassed,
   831→   831→        'time_spent_seconds': timeSpent,
   832→   832→        'grading_status': gradingStatus,
   833→   833→        'is_released': false,
   834→   834→      };
   835→   835→
   836→   836→      if (existingResult != null) {
   837→   837→        await _supabaseClient
   838→   838→            .from(_examResultsTable)
   839→   839→            .update(resultData)
   840→   840→            .eq('id', existingResult['id'] as String);
   841→   841→
   842→   842→        final updatedResult = await _supabaseClient
   843→   843→            .from(_examResultsTable)
   844→   844→            .select()
   845→   845→            .eq('id', existingResult['id'] as String)
   846→   846→            .single();
   847→   847→
   848→   848→        return ExamResultModel.fromJson(updatedResult);
   849→   849→      } else {
   850→   850→        final insertResponse = await _supabaseClient
   851→   851→            .from(_examResultsTable)
   852→   852→            .insert(resultData)
   853→   853→            .select()
   854→   854→            .single();
   855→   855→
   856→   856→        return ExamResultModel.fromJson(insertResponse);
   857→   857→      }
   858→   858→    } on sb.PostgrestException catch (e) {
   859→   859→      AppLogger.error('Submit attempt failed', error: e);
   860→   860→      throw ServerException(
   861→   861→        message: e.message,
   862→   862→        statusCode: int.tryParse(e.code ?? '') ?? 500,
   863→   863→        data: e.details,
   864→   864→      );
   865→   865→    } catch (e) {
   866→   866→      if (e is ServerException) rethrow;
   867→   867→      AppLogger.error('Submit attempt unexpected error', error: e);
   868→   868→      throw ServerException(
   869→   869→        message: 'Failed to submit attempt: $e',
   870→   870→        statusCode: 500,
   871→   871→      );
   872→   872→    }
   873→   873→  }
