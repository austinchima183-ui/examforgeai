   306→  @override
   307→  Future<SchoolModel> createSchool(Map<String, dynamic> schoolData) async {
   308→    try {
   309→      final response = await _supabaseClient
   310→          .from(_schoolsTable)
   311→          .insert(schoolData)
   312→          .select()
   313→          .single();
   314→
   315→      return SchoolModel.fromJson(response);
   316→    } on sb.PostgrestException catch (e) {
   317→      AppLogger.error('Create school failed', error: e);
   318→      throw ServerException(
   319→        message: e.message,
   320→        statusCode: int.tryParse(e.code ?? '') ?? 500,
   321→        data: e.details,
   322→      );
   323→    } catch (e) {
   324→      AppLogger.error('Create school unexpected error', error: e);
   325→      throw ServerException(
   326→        message: 'Failed to create school: $e',
   327→        statusCode: 500,
   328→      );
   329→    }
   330→  }

   417→  @override
   418→  Future<List<SchoolModel>> getSchools(Map<String, dynamic> filters) async {
   419→    try {
   420→      // PERF: Explicit columns avoid fetching all school fields for list view
   421→      var query = _supabaseClient.from(_schoolsTable).select('id, name, address, is_active, created_at');
   422→
   423→      if (filters['is_active'] != null) {
   424→        query = query.eq('is_active', filters['is_active'] as bool);
   425→      }
   426→      if (filters['search'] != null) {
   427→        final search = filters['search'] as String;
   428→        query = query.or('name.ilike.%$search%,code.ilike.%$search%');
   429→      }
   430→      if (filters['school_level'] != null) {
   431→        query = query.eq('school_level', filters['school_level'] as String);
   432→      }
   433→      if (filters['subscription_status'] != null) {
   434→        query = query.eq(
   435→          'subscription_status',
   436→          filters['subscription_status'] as String,
   437→        );
   438→      }
   439→
   440→      final page = filters['page'] as int? ?? 1;
   441→      final perPage = filters['per_page'] as int? ?? filters['perPage'] as int? ?? 20;
   442→      final offset = (page - 1) * perPage;
   443→
   444→      final response = await query.range(offset, offset + perPage - 1).order('created_at', ascending: false);
   445→
   446→      return response
   447→          .map<SchoolModel>((json) => SchoolModel.fromJson(json))
   448→          .toList();
   449→    } on sb.PostgrestException catch (e) {
   450→      AppLogger.error('Get schools failed', error: e);
   451→      throw ServerException(
   452→        message: e.message,
   453→        statusCode: int.tryParse(e.code ?? '') ?? 500,
   454→        data: e.details,
   455→      );
   456→    } catch (e) {
   457→      AppLogger.error('Get schools unexpected error', error: e);
   458→      throw ServerException(
   459→        message: 'Failed to get schools: $e',
   460→        statusCode: 500,
   461→      );
   462→    }
   463→  }

   551→  @override

   553→    String schoolId, {
   554→    int limit = PaginatedQueryMixin.defaultPageSize,
   555→    int offset = 0,
   556→  }) async {
   557→    try {
   558→      // PERF: Explicit columns + limit prevent unbounded result sets
   559→      final response = await _supabaseClient
   560→          .from(_schoolBranchesTable)
   561→          .select('id, school_id, name, address, is_active, created_at')
