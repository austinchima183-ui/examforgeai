   307→   307→  @override
   308→   308→  Future<List<ExamModel>> getExams(Map<String, dynamic> filters) async {
   309→   309→    try {
   310→   310→      var filterQuery = _supabaseClient.from(_examsTable).select();
   311→   311→
   312→   312→      if (filters['school_id'] != null) {
   313→   313→        filterQuery = filterQuery.eq('school_id', filters['school_id'] as String);
   314→   314→      }
   315→   315→      if (filters['subject_id'] != null) {
   316→   316→        filterQuery = filterQuery.eq('subject_id', filters['subject_id'] as String);
   317→   317→      }
   318→   318→      if (filters['status'] != null) {
   319→   319→        filterQuery = filterQuery.eq('status', filters['status'] as String);
   320→   320→      }
   321→   321→      if (filters['class_id'] != null) {
   322→   322→        filterQuery = filterQuery.eq('class_id', filters['class_id'] as String);
   323→   323→      }
   324→   324→
   325→   325→      final page = filters['page'] as int? ?? 1;
   326→   326→      final perPage = filters['per_page'] as int? ?? filters['perPage'] as int? ?? 20;
   327→   327→      final offset = (page - 1) * perPage;
   328→   328→
   329→   329→      final transformQuery = filterQuery.order('created_at', ascending: false).range(offset, offset + perPage - 1);
   330→   330→
   331→   331→      final response = await transformQuery;
   332→   332→
   333→   333→      return response
   334→   334→          .map<ExamModel>((json) => ExamModel.fromJson(json))
   335→   335→          .toList();
   336→   336→    } on sb.PostgrestException catch (e) {
   337→   337→      AppLogger.error('Get exams failed', error: e);
   338→   338→      throw ServerException(
   339→   339→        message: e.message,
   340→   340→        statusCode: int.tryParse(e.code ?? '') ?? 500,
   341→   341→        data: e.details,
   342→   342→      );
   343→   343→    } catch (e) {
   344→   344→      AppLogger.error('Get exams unexpected error', error: e);
   345→   345→      throw ServerException(
   346→   346→        message: 'Failed to get exams: $e',
   347→   347→        statusCode: 500,
   348→   348→      );
   349→   349→    }
   350→   350→  }

   413→   413→  @override
   414→   414→  Future<ExamModel> cloneExam(String examId) async {
   415→   415→    try {
   416→   416→      // Fetch the original exam with details
   417→   417→      final original = await getExamWithDetails(examId);
   418→   418→
   419→   419→      final now = DateTime.now();
   420→   420→      final cloneData = {
   421→   421→        'school_id': original.schoolId,
   422→   422→        'created_by': original.createdBy,
   423→   423→        'title': '${original.title} (Copy)',
   424→   424→        'description': original.description,
   425→   425→        'subject_id': original.subjectId,
   426→   426→        'class_id': original.classId,
   427→   427→        'academic_session_id': original.academicSessionId,
   428→   428→        'exam_type': original.examType,
   429→   429→        'status': 'draft',
   430→   430→        'start_time': now.add(const Duration(days: 7)).toIso8601String(),
   431→   431→        'end_time': now.add(const Duration(days: 7, hours: 2)).toIso8601String(),
   432→   432→        'time_limit_minutes': original.timeLimitMinutes,
   433→   433→        'total_marks': original.totalMarks,
   434→   434→        'pass_mark': original.passMark,
   435→   435→        'pass_mark_type': original.passMarkType,
   436→   436→        'instructions': original.instructions,
   437→   437→        'allowed_attempts': original.allowedAttempts,
   438→   438→        'negative_marking_enabled': original.negativeMarkingEnabled,
   439→   439→        'negative_mark_value': original.negativeMarkValue,
   440→   440→        'grace_period_minutes': original.gracePeriodMinutes,
   441→   441→        'auto_submit': original.autoSubmit,
   442→   442→        'randomize_questions': original.randomizeQuestions,
   443→   443→        'randomize_options': original.randomizeOptions,
   444→   444→        'show_results': original.showResults,
   445→   445→        'show_correct_answers': original.showCorrectAnswers,
   446→   446→        'show_explanations': original.showExplanations,
   447→   447→        'is_template': original.isTemplate,
   448→   448→        'template_id': original.id,
   449→   449→        'max_students': original.maxStudents,
   450→   450→        'ip_restriction': original.ipRestriction,
   451→   451→        'require_full_screen': original.requireFullScreen,
   452→   452→        'allow_resume': original.allowResume,
   453→   453→        'browser_lockdown': original.browserLockdown,
   454→   454→        'metadata': original.metadata,
   455→   455→      };
   456→   456→
   457→   457→      final clonedExam = await createExam(cloneData);
   458→   458→
   459→   459→      // Clone sections
   460→   460→      for (final section in original.sections) {
   461→   461→        await _supabaseClient.from(_examSectionsTable).insert({
   462→   462→          'exam_id': clonedExam.id,
   463→   463→          'title': section.title,
   464→   464→          'description': section.description,
   465→   465→          'instructions': section.instructions,
   466→   466→          'sort_order': section.sortOrder,
   467→   467→          'time_limit_minutes': section.timeLimitMinutes,
   468→   468→          'randomize_questions': section.randomizeQuestions,
   469→   469→        });
   470→   470→      }
   471→   471→
   472→   472→      // Clone questions
   473→   473→      for (final question in original.questions) {
   474→   474→        await _supabaseClient.from(_examQuestionsTable).insert({
   475→   475→          'exam_id': clonedExam.id,
   476→   476→          'section_id': question.sectionId,
   477→   477→          'question_id': question.questionId,
   478→   478→          'sort_order': question.sortOrder,
   479→   479→          'marks': question.marks,
   480→   480→          'negative_marks': question.negativeMarks,
   481→   481→          'is_compulsory': question.isCompulsory,
   482→   482→        });
   483→   483→      }
   484→   484→
   485→   485→      return clonedExam;

   655→   655→  @override
   656→   656→  Future<ExamAttemptModel> startAttempt(String examId) async {
   657→   657→    try {
   658→   658→      final userId = _supabaseClient.auth.currentUser?.id;
   659→   659→      if (userId == null) {
   660→   660→        throw const AuthException(
   661→   661→          message: 'Not authenticated',
   662→   662→          code: 'NOT_AUTHENTICATED',
   663→   663→        );
   664→   664→      }
   665→   665→
   666→   666→      // ── Call server-side SQL function which validates eligibility,
   667→   667→      //    creates the attempt AND session atomically ───────────────
   668→   668→      final response = await _supabaseClient.rpc(
   669→   669→        'start_exam_attempt',
   670→   670→        params: {'p_exam_id': examId},
   671→   671→      );
   672→   672→
   673→   673→      // Server returns the created attempt row as JSON
   674→   674→      if (response == null) {
   675→   675→        throw const ServerException(
   676→   676→          message: 'Server returned no data for start_exam_attempt',
   677→   677→          statusCode: 500,
   678→   678→        );
   679→   679→      }
   680→   680→
   681→   681→      final attemptMap = response as Map<String, dynamic>;
   682→   682→
   683→   683→      // Validate the server confirmed the attempt was started
   684→   684→      final attemptStatus = attemptMap['status'] as String?;
   685→   685→      if (attemptStatus != 'in_progress') {
   686→   686→        AppLogger.warning(
   687→   687→          'Server returned unexpected attempt status: $attemptStatus',
   688→   688→        );
   689→   689→        throw ServerException(
   690→   690→          message: 'Server did not start attempt. Status: $attemptStatus',
   691→   691→          statusCode: 409,
   692→   692→          data: attemptMap,
   693→   693→        );
   694→   694→      }
   695→   695→
   696→   696→      return ExamAttemptModel.fromJson(attemptMap);
   697→   697→    } on AuthException {
   698→   698→      rethrow;
   699→   699→    } on sb.PostgrestException catch (e) {
   700→   700→      AppLogger.error('Start attempt RPC failed', error: e);
   701→   701→      // Map known Postgrest error codes to domain-meaningful exceptions
   702→   702→      final msg = e.message;
   703→   703→      if (msg.contains('not assigned') || msg.contains('not eligible')) {
   704→   704→        throw ForbiddenException(message: msg);
   705→   705→      }
   706→   706→      if (msg.contains('max attempts') || msg.contains('already in progress')) {
   707→   707→        throw ValidationException(message: msg);
   708→   708→      }
   709→   709→      throw ServerException(
   710→   710→        message: e.message,
   711→   711→        statusCode: int.tryParse(e.code ?? '') ?? 500,
   712→   712→        data: e.details,
   713→   713→      );
   714→   714→    } on ForbiddenException {
   715→   715→      rethrow;
   716→   716→    } on ValidationException {
   717→   717→      rethrow;
   718→   718→    } catch (e) {
   719→   719→      if (e is ServerException || e is AuthException) rethrow;
   720→   720→      AppLogger.error('Start attempt unexpected error', error: e);
   721→   721→      throw ServerException(
   722→   722→        message: 'Failed to start attempt: $e',
   723→   723→        statusCode: 500,
   724→   724→      );
   725→   725→    }
   726→   726→  }

   898→   898→  @override
   899→   899→  Future<StudentAnswerModel> saveAnswer(
   900→   900→    String attemptId,
   901→   901→    String questionId,
   902→   902→    Map<String, dynamic> answerData,
   903→   903→  ) async {
   904→   904→    try {
   905→   905→      // ── Validate exam timing server-side before accepting the answer ──
   906→   906→      final timingResponse = await _supabaseClient.rpc(
   907→   907→        'validate_exam_timing',
   908→   908→        params: {'p_attempt_id': attemptId},
   909→   909→      );
   910→   910→
   911→   911→      final timingResult = timingResponse as Map<String, dynamic>?;
   912→   912→      final isWithinTime = timingResult?['is_within_time'] as bool? ?? false;
   913→   913→

   922→   922→      }
   923→   923→
   924→   924→      // Use server time for the answer timestamp to avoid client clock skew
   925→   925→      final serverTime = timingResult?['server_time'] as String?;
   926→   926→      final now = serverTime != null
   927→   927→          ? DateTime.parse(serverTime)
   928→   928→          : DateTime.now();
   929→   929→

   945→   945→            })
   946→   946→            .eq('id', existing['id'] as String)
   947→   947→            .select()
   948→   948→            .single();
   949→   949→
   950→   950→        return StudentAnswerModel.fromJson(response);
   951→   951→      } else {
   952→   952→        final response = await _supabaseClient
   953→   953→            .from(_studentAnswersTable)
   954→   954→            .insert({
   955→   955→              'attempt_id': attemptId,
   956→   956→              'question_id': questionId,
   957→   957→              'answer_data': answerData,
   958→   958→              'answered_at': now.toIso8601String(),
   959→   959→            })
   960→   960→            .select()
   961→   961→            .single();
   962→   962→
   963→   963→        return StudentAnswerModel.fromJson(response);
   964→   964→      }

   969→   969→        statusCode: int.tryParse(e.code ?? '') ?? 500,
   970→   970→        data: e.details,
   971→   971→      );
   972→   972→    } on ValidationException {
   973→   973→      rethrow;
   974→   974→    } catch (e) {
   975→   975→      if (e is ServerException || e is ValidationException) rethrow;
   976→   976→      AppLogger.error('Save answer unexpected error', error: e);
   977→   977→      throw ServerException(
   978→   978→        message: 'Failed to save answer: $e',

  1341→  1341→
  1342→  1342→      final totalStudents = students.length;
  1343→  1343→      final startedStudents = attempts

  1348→  1348→              a['status'] == 'submitted' ||
  1349→  1349→              a['status'] == 'auto_submitted' ||
  1350→  1350→              a['status'] == 'timed_out',)

  1352→  1352→      final submittedStudents = completedStudents;
  1353→  1353→
  1354→  1354→      // Calculate score statistics
  1355→  1355→      final scores = results
  1356→  1356→          .map<double>((r) => (r['score_percentage'] as num?)?.toDouble() ?? 0.0)
  1357→  1357→          .toList();

  1361→  1361→          : 0.0;

  1364→  1364→          : 0.0;

  1400→  1400→          .length;

  1403→  1403→          : 0.0;
  1404→  1404→

  1413→  1413→        lowestScore: lowestScore,
  1414→  1414→        medianScore: medianScore,
  1415→  1415→        passRate: passRate,
  1416→  1416→        questionsByCorrectRate: {}, // Requires per-question analysis
  1417→  1417→        averageTimeSpentSeconds: averageTimeSpent,
  1418→  1418→        gradingCompletionPercentage: gradingCompletion,
  1419→  1419→      );
  1420→  1420→    } on sb.PostgrestException catch (e) {
  1421→  1421→      AppLogger.error('Get exam statistics failed', error: e);

  1423→  1423→        message: e.message,

  1440→  1440→      // which is polled every 5-10 seconds during active exams
  1441→  1441→      final activeSessions = await DatabasePoolManager.executeMonitored(
  1442→  1442→        queryType: 'select',

  1451→  1451→      );
  1452→  1452→
  1453→  1453→      // Count students (bounded by exam size)

  1471→  1471→          .eq('exam_id', examId)

  1473→  1473→          .order('submitted_at', ascending: false)
  1474→  1474→          .limit(10);
  1475→  1475→
  1476→  1476→      // Recent monitoring events (last 20)
  1477→  1477→      final recentEvents = await _supabaseClient
  1478→  1478→          .from(_monitoringLogsTable)
  1479→  1479→          .select()
  1480→  1480→          .eq('exam_id', examId)
  1481→  1481→          .order('created_at', ascending: false)
  1482→  1482→          .limit(20);

  1485→  1485→      final activeNow = activeSessions.length;
  1486→  1486→      final completed = submittedAttempts.length;
  1487→  1487→      final notStarted = totalEligible - activeNow - completed;
  1488→  1488→

  1490→  1490→      double averageProgress = 0;

  1492→  1492→        final totalQuestionsAnswered = activeSessions.fold<int>(
  1493→  1493→          0,
  1494→  1494→          (sum, s) =>
  1495→  1495→              sum + ((s['questions_answered'] as int?) ?? 0),
  1496→  1496→        );
  1497→  1497→        // Estimate total questions per exam
  1498→  1498→        averageProgress = totalEligible > 0
  1499→  1499→            ? totalQuestionsAnswered / (activeNow * 10) * 100
  1500→  1500→            : 0;
  1501→  1501→        averageProgress = averageProgress.clamp(0.0, 100.0);
  1502→  1502→      }
  1503→  1503→
  1504→  1504→      return LiveExamStatsModel(
  1505→  1505→        examId: examId,
  1506→  1506→        totalEligible: totalEligible,
  1507→  1507→        activeNow: activeNow,
  1508→  1508→        completed: completed,
  1509→  1509→        notStarted: notStarted < 0 ? 0 : notStarted,
  1510→  1510→        averageProgress: averageProgress,
  1511→  1511→        recentSubmissions: recentSubmissions
  1512→  1512→            .map<ExamAttemptModel>((a) => ExamAttemptModel.fromJson(a))
  1513→  1513→            .toList(),
  1514→  1514→        activeSessions: activeSessions
  1515→  1515→            .map<ExamSessionModel>((s) => ExamSessionModel.fromJson(s))
  1516→  1516→            .toList(),
  1517→  1517→        recentMonitoringEvents: recentEvents
  1518→  1518→            .map<MonitoringLogModel>((e) => MonitoringLogModel.fromJson(e))
  1519→  1519→            .toList(),
  1520→  1520→      );
  1521→  1521→    } on sb.PostgrestException catch (e) {

  1523→  1523→      throw ServerException(
  1524→  1524→        message: e.message,
  1525→  1525→        statusCode: int.tryParse(e.code ?? '') ?? 500,
  1526→  1526→        data: e.details,

  1528→  1528→    } catch (e) {
