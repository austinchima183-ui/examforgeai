   843→-- AUDIT LOG TRIGGER (reuses existing audit_log table)
   844→-- ═══════════════════════════════════════════════════════════════════════════
   845→
   846→CREATE OR REPLACE FUNCTION log_student_portal_audit()
   847→RETURNS TRIGGER AS $$
   848→DECLARE
   849→  table_name TEXT := TG_TABLE_NAME;
   850→  operation  TEXT := TG_OP;
   851→  user_id    UUID;
   852→  record_id  UUID;
   853→BEGIN
   854→  IF operation = 'INSERT' THEN
   855→    record_id := NEW.id;
   856→    user_id := COALESCE(NEW.student_id, NEW.student_id);
   857→  ELSIF operation = 'UPDATE' THEN
   858→    record_id := NEW.id;
   859→    user_id := COALESCE(NEW.student_id, OLD.student_id);
   860→  ELSIF operation = 'DELETE' THEN
   861→    record_id := OLD.id;
   862→    user_id := OLD.student_id;
   863→  END IF;
   864→
   865→  INSERT INTO audit_log (user_id, action, table_name, record_id, new_data, old_data)
   866→  VALUES (
   867→    user_id,
   868→    operation,
   869→    table_name,
   870→    record_id,
   871→    CASE WHEN operation IN ('INSERT', 'UPDATE') THEN to_jsonb(NEW) ELSE NULL END,
   872→    CASE WHEN operation IN ('UPDATE', 'DELETE') THEN to_jsonb(OLD) ELSE NULL END
   873→  );
   874→
   875→  RETURN COALESCE(NEW, OLD);
   876→END;
   877→$$ LANGUAGE plpgsql SECURITY DEFINER;

   892→-- Student dashboard overview (fast read)
   893→CREATE OR REPLACE VIEW v_student_dashboard AS
   894→SELECT
   895→  u.id AS student_id,
   896→  u.school_id,
   897→  u.full_name,
   898→  -- Upcoming exams (next 7 days)
   899→  (SELECT COUNT(*) FROM exam_attempts ea
   900→   JOIN exams e ON ea.exam_id = e.id
   901→   WHERE ea.student_id = u.id
   902→     AND e.status = 'active'
   903→     AND e.start_time <= now() + interval '7 days') AS upcoming_exams,
   904→  -- Pending assignments
   905→  (SELECT COUNT(*) FROM assignment_submissions a
   906→   WHERE a.student_id = u.id
   907→     AND a.status IN ('draft', 'submitted')) AS pending_assignments,
   908→  -- Learning streak
   909→  COALESCE((SELECT learning_streak FROM student_progress_snapshots
   910→            WHERE student_id = u.id
   911→            ORDER BY period_start DESC LIMIT 1), 0) AS learning_streak,
   912→  -- Recent average score
   913→  COALESCE((SELECT avg_score FROM student_progress_snapshots
   914→            WHERE student_id = u.id AND subject_id IS NULL
   915→            ORDER BY period_start DESC LIMIT 1), 0) AS recent_avg_score,
   916→  -- Practice sessions this week
   917→  (SELECT COUNT(*) FROM practice_sessions
   918→   WHERE student_id = u.id
   919→     AND started_at >= date_trunc('week', now())) AS practice_this_week,
   920→  -- Unread notifications
   921→  (SELECT COUNT(*) FROM student_notifications
   922→   WHERE student_id = u.id AND is_read = FALSE) AS unread_notifications
   923→FROM users u
   924→WHERE u.role = 'student';
