   368→// ─── 3. StarRating ────────────────────────────────────────────────────────────
   369→
   370→/// Displays a row of 5 stars (filled, half-filled, or outlined) based on
   371→/// [rating] with an optional review count.
   372→///
   373→/// ```dart
   374→/// StarRating(rating: 4.3, totalReviews: 128, showCount: true)
   375→/// ```
   376→class StarRating extends StatelessWidget {
   377→  const StarRating({
   378→    super.key,
   379→    required this.rating,
   380→    this.totalReviews,
   381→    this.starSize = Spacings.mdIcon,
   382→    this.showCount = true,
   383→  });
   384→
   385→  final double rating;
   386→  final int? totalReviews;
   387→  final double starSize;
   388→  final bool showCount;
   389→
   390→  @override
   391→  Widget build(BuildContext context) {
   392→    final cs = context.colorScheme;
   393→    final tt = context.textTheme;
   394→
   395→    return Row(
   396→      mainAxisSize: MainAxisSize.min,
   397→      children: [
   398→        // 5 stars
   399→        for (int i = 1; i <= 5; i++)
   400→          Icon(
   401→            _starIcon(i),
   402→            size: starSize,
   403→            color: AppColors.warning,
   404→          ),
   405→        // Count
   406→        if (showCount && totalReviews != null) ...[
   407→          const SizedBox(width: Spacings.xs),
   408→          Text(
   409→            '($totalReviews)',
   410→            style: tt.labelSmall?.copyWith(
   411→              color: cs.onSurfaceVariant,
   412→            ),
   413→          ),
   414→        ],
   415→      ],
   416→    );
   417→  }
   418→
   419→  IconData _starIcon(int position) {
   420→    if (rating >= position) return Icons.star_rounded;
   421→    if (rating >= position - 0.5) return Icons.star_half_rounded;
   422→    return Icons.star_outline_rounded;
   423→  }
   424→}

   497→// ─── 5. PriceDisplay ──────────────────────────────────────────────────────────
   498→
   499→/// Shows product price with optional original price strikethrough.
   500→///
   501→/// ```dart
   502→/// PriceDisplay(price: 1500, originalPrice: 2500, currency: 'NGN')
   503→/// ```
   504→class PriceDisplay extends StatelessWidget {
   505→  const PriceDisplay({
   506→    super.key,
   507→    required this.price,
   508→    this.originalPrice,
   509→    this.currency = 'NGN',
   510→    this.fontSize = 14,
   511→    this.isFree = false,
   512→  });
   513→
   514→  final double price;
   515→  final double? originalPrice;
   516→  final String currency;
   517→  final double fontSize;
   518→  final bool isFree;
   519→
   520→  @override
   521→  Widget build(BuildContext context) {
   522→    final cs = context.colorScheme;
   523→
   524→    if (isFree) {
   525→      return Text(
   526→        'Free',
   527→        style: TextStyle(
   528→          fontFamily: AppTypography.fontFamily,
   529→          fontSize: fontSize,
   530→          fontWeight: AppTypography.wBold,
   531→          color: AppColors.success,
   532→          letterSpacing: AppTypography.lsLabel,
   533→        ),
   534→      );
   535→    }
   536→
   537→    final hasDiscount = originalPrice != null && originalPrice! > price;
   538→
   539→    return Row(
   540→      mainAxisSize: MainAxisSize.min,
   541→      crossAxisAlignment: CrossAxisAlignment.baseline,
   542→      textBaseline: TextBaseline.alphabetic,
   543→      children: [
   544→        Text(
   545→          _formatPrice(price, currency),
   546→          style: TextStyle(
   547→            fontFamily: AppTypography.fontFamily,
   548→            fontSize: fontSize,
   549→            fontWeight: AppTypography.wBold,
   550→            color: cs.primary,
   551→            letterSpacing: AppTypography.lsLabel,
   552→          ),
   553→        ),
   554→        if (hasDiscount) ...[
   555→          const SizedBox(width: Spacings.xs),
   556→          Text(
   557→            _formatPrice(originalPrice!, currency),
   558→            style: TextStyle(
   559→              fontFamily: AppTypography.fontFamily,
   560→              fontSize: fontSize - 2,
   561→              fontWeight: AppTypography.wRegular,
   562→              color: cs.onSurfaceVariant,
   563→              decoration: TextDecoration.lineThrough,
   564→              decorationColor: cs.onSurfaceVariant,
   565→              letterSpacing: AppTypography.lsLabel,
   566→            ),
   567→          ),
   568→        ],
   569→      ],
   570→    );
   571→  }
   572→
   573→  String _formatPrice(double amount, String curr) {
   574→    final symbol = switch (curr) {
   575→      'NGN' => '\u20A6',
   576→      'USD' => '\$',
   577→      'GBP' => '\u00A3',
   578→      'EUR' => '\u20AC',
   579→      _ => '$curr ',
   580→    };
   581→    return '$symbol${amount.toStringAsFixed(0)}';
   582→  }
   583→}
