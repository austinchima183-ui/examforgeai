   220→  @override
   221→  Future<GradeScaleModel> createGradeScale(GradeScaleModel scale) async {
   222→    try {
   223→      // 1. Insert the grade_scale row (without nested entries)
   224→      final scaleJson = scale.toJson()
   225→        ..remove('scale_entries');
   226→
   227→      final response = await _supabaseClient
   228→          .from(_gradeScalesTable)
   229→          .insert(scaleJson)
   230→          .select()
   231→          .single();
   232→
   233→      final createdScale = GradeScaleModel.fromJson(response);
   234→
   235→      // 2. Insert entries
   236→      if (scale.scaleEntries.isNotEmpty) {
   237→        final entriesRows = scale.scaleEntries.map((e) {
   238→          final json = e.toJson();
   239→          json['grade_scale_id'] = createdScale.id;
   240→          return json;
   241→        }).toList();
   242→
   243→        await _supabaseClient
   244→            .from(_gradeScaleEntriesTable)
   245→            .insert(entriesRows);
   246→      }
   247→
   248→      // 3. Return scale with entries populated
   249→      return createdScale.copyWith(
   250→        scaleEntries: scale.scaleEntries,
   251→      );
   252→    } on sb.PostgrestException catch (e) {
   253→      AppLogger.error('Create grade scale failed', error: e);
   254→      throw ServerException(
   255→        message: e.message,
   256→        statusCode: int.tryParse(e.code ?? '') ?? 500,
   257→        data: e.details,
   258→      );
   259→    } catch (e) {
   260→      AppLogger.error('Create grade scale unexpected error', error: e);
   261→      throw ServerException(
   262→        message: 'Failed to create grade scale: $e',
   263→        statusCode: 500,
   264→      );
   265→    }
   266→  }
   267→
   268→  @override
   269→  Future<GradeScaleModel> updateGradeScale(GradeScaleModel scale) async {
   270→    try {
   271→      // 1. Update the grade_scale row
   272→      final scaleJson = scale.toJson()
   273→        ..remove('scale_entries');
   274→
   275→      final response = await _supabaseClient
   276→          .from(_gradeScalesTable)
   277→          .update(scaleJson)
   278→          .eq('id', scale.id)
   279→          .select()
   280→          .single();
   281→
   282→      final updatedScale = GradeScaleModel.fromJson(response);
   283→
   284→      // 2. Delete existing entries and re-insert
   285→      await _supabaseClient
   286→          .from(_gradeScaleEntriesTable)
   287→          .delete()
   288→          .eq('grade_scale_id', scale.id);
   289→
   290→      if (scale.scaleEntries.isNotEmpty) {
   291→        final entriesRows = scale.scaleEntries.map((e) {
   292→          final json = e.toJson();
   293→          json['grade_scale_id'] = scale.id;
   294→          return json;
   295→        }).toList();
   296→
   297→        await _supabaseClient
   298→            .from(_gradeScaleEntriesTable)
   299→            .insert(entriesRows);
   300→      }
   301→
   302→      // 3. Return scale with entries populated
   303→      return updatedScale.copyWith(
   304→        scaleEntries: scale.scaleEntries,
   305→      );
   306→    } on sb.PostgrestException catch (e) {
   307→      AppLogger.error('Update grade scale failed', error: e);
   308→      if (e.code == 'PGRST116') {
   309→        throw NotFoundException(message: 'Grade scale not found: ${scale.id}');
   310→      }
   311→      throw ServerException(
   312→        message: e.message,
   313→        statusCode: int.tryParse(e.code ?? '') ?? 500,
   314→        data: e.details,
   315→      );
   316→    } catch (e) {
   317→      AppLogger.error('Update grade scale unexpected error', error: e);
   318→      throw ServerException(
   319→        message: 'Failed to update grade scale: $e',
   320→        statusCode: 500,
   321→      );
   322→    }
   323→  }

   348→  @override
   349→  Future<GradeScaleModel> getGradeScale(String scaleId) async {
   350→    try {
   351→      final response = await _supabaseClient
   352→          .from(_gradeScalesTable)
   353→          .select('*, grade_scale_entries(*)')
   354→          .eq('id', scaleId)
   355→          .single();
   356→
   357→      // Map nested entries from the join
   358→      final mappedResponse = _mapGradeScaleEntries(response);
   359→
   360→      return GradeScaleModel.fromJson(mappedResponse);
   361→    } on sb.PostgrestException catch (e) {
   362→      AppLogger.error('Get grade scale failed', error: e);
   363→      if (e.code == 'PGRST116') {
   364→        throw NotFoundException(message: 'Grade scale not found: $scaleId');
   365→      }
   366→      throw ServerException(
   367→        message: e.message,
   368→        statusCode: int.tryParse(e.code ?? '') ?? 500,
   369→        data: e.details,
   370→      );
   371→    } catch (e) {
   372→      AppLogger.error('Get grade scale unexpected error', error: e);
   373→      throw ServerException(
   374→        message: 'Failed to get grade scale: $e',
   375→        statusCode: 500,
   376→      );
   377→    }

   380→  @override
   381→  Future<List<GradeScaleModel>> getGradeScales(
   382→    String schoolId, {
   383→    bool? isActive,
   384→    String? gradeType,
   385→    int limit = PaginatedQueryMixin.defaultPageSize,
   386→    int offset = 0,
   387→  }) async {
   388→    try {
   389→      // PERF: Added pagination params + limit to grade scales list query
   390→      var filterQuery = _supabaseClient
   391→          .from(_gradeScalesTable)
   392→          .select('*, grade_scale_entries(*)');
   393→
   394→      filterQuery = filterQuery.eq('school_id', schoolId);
   395→
   396→      if (isActive != null) {
   397→        filterQuery = filterQuery.eq('is_active', isActive);
   398→      }
   399→      if (gradeType != null) {
   400→        filterQuery = filterQuery.eq('grade_type', gradeType);
   401→      }
   402→
   403→      final transformQuery = filterQuery.order('created_at', ascending: false).range(offset, offset + limit - 1);
   404→
   405→      final response = await transformQuery;
   406→
   407→      return response.map<GradeScaleModel>((json) {
   408→        final mappedJson = _mapGradeScaleEntries(json);
   409→        return GradeScaleModel.fromJson(mappedJson);
   410→      }).toList();
   411→    } on sb.PostgrestException catch (e) {
   412→      AppLogger.error('Get grade scales failed', error: e);
   413→      throw ServerException(
   414→        message: e.message,
   415→        statusCode: int.tryParse(e.code ?? '') ?? 500,
   416→        data: e.details,
   417→      );
   418→    } catch (e) {
   419→      AppLogger.error('Get grade scales unexpected error', error: e);
   420→      throw ServerException(
   421→        message: 'Failed to get grade scales: $e',
   422→        statusCode: 500,
   423→      );
   424→    }
   425→  }
   426→
   427→  /// Maps the nested `grade_scale_entries` join result to the model's
   428→  /// expected `scale_entries` key.
   429→  Map<String, dynamic> _mapGradeScaleEntries(Map<String, dynamic> json) {
   430→    final entries = json['grade_scale_entries'];
   431→    if (entries is List) {
   432→      return {
   433→        ...json,
   434→        'scale_entries': entries,
   435→      };
   436→    }
   437→    return json;
   438→  }
