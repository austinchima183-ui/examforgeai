     1→     1→/// Report Generator — pure computation service for structured report outputs.
     2→     2→///
     3→     3→/// Transforms domain entities (results, exam, and CBT data) into structured
     4→     4→/// [ReportData] objects ready for export to PDF, Excel, or CSV by the
     5→     5→/// datasource layer (Supabase Edge Functions or local generators).
     6→     6→///
     7→     7→/// ## Design Principles
     8→     8→///
     9→     9→/// - **Pure transformation**: Every public method accepts entities and returns
    10→    10→///   a [ReportData] with no I/O, no database access, no side effects.
    11→    11→/// - **Self-contained DTOs**: Output content types are defined in this file to
    12→    12→///   avoid coupling to persistence-layer models.
    13→    13→/// - **Null-safe**: All methods handle missing or partial data gracefully,
    14→    14→///   providing sensible defaults rather than throwing.
    15→    15→/// - **Format-agnostic**: This service produces structured data only; the
    16→    16→///   actual file generation is handled downstream.
    17→    17→///
    18→    18→/// ## Usage
    19→    19→///
    20→    20→/// ```dart
    21→    21→/// final generator = ReportGenerator();
    22→    22→/// final report = generator.generateStudentReport(
    23→    23→///   overallResult: studentOverall,
    24→    24→///   subjectResults: subjectList,
    25→    25→///   studentName: 'John Doe',
    26→    26→///   className: 'SS2A',
    27→    27→///   sessionName: '2025/2026',
    28→    28→///   trendData: trendPoints,
    29→    29→/// );
    30→    30→/// // Pass report to datasource layer for PDF/Excel/CSV generation
    31→    31→/// ```

   555→   555→        'duration': duration,

   557→   557→        'passMark': passMark,
   558→   558→        'totalStudents': totalStudents,

   573→   573→// REPORT GENERATOR SERVICE

   575→   575→
   576→   576→/// Generates structured report data from domain entities.
   577→   577→///
   578→   578→/// This service is a pure computation layer — it transforms raw entity data
   579→   579→/// into [ReportData] structures that the datasource layer can then render
   580→   580→/// into PDF, Excel, or CSV files. No file I/O is performed here.
   581→   581→class ReportGenerator {
   582→   582→  // ── Student Report ─────────────────────────────────────────────────
   583→   583→
   584→   584→  /// Creates a comprehensive report for a single student.
   585→   585→  ///
   586→   586→  /// Requires the student's [overallResult] entity, their [subjectResults],
   587→   587→  /// and contextual information like [studentName], [className], and
   588→   588→  /// [sessionName]. Optional [trendData] provides historical performance
   589→   589→  /// for chart rendering.
   590→   590→  ///
   591→   591→  /// The optional [gradeScale] is used to derive the overall grade from the
   592→   592→  /// overall percentage if [StudentOverallResultEntity.overallGrade] is null.
   593→   593→  ReportData generateStudentReport({
   594→   594→    required StudentOverallResultEntity overallResult,
   595→   595→    required List<StudentSubjectResultEntity> subjectResults,
   596→   596→    required String studentName,
   597→   597→    required String className,
   598→   598→    required String sessionName,
   599→   599→    List<TrendDataPoint>? trendData,
   600→   600→    GradeScaleEntity? gradeScale,
   601→   601→  }) {
   602→   602→    // Resolve the overall grade — use the entity's grade first, then
   603→   603→    // fall back to the grade scale if available.
   604→   604→    final overallGrade = overallResult.overallGrade ??
   605→   605→        gradeScale?.applyToPercentage(overallResult.overallPercentage)?.grade ??
   606→   606→        _defaultGrade(overallResult.overallPercentage);
   607→   607→
   608→   608→    // Build subject-by-subject breakdown rows.
   609→   609→    final subjectRows = subjectResults.map((sr) {
   610→   610→      final grade = sr.grade ??
   611→   611→          gradeScale?.applyToPercentage(sr.percentage)?.grade ??
   612→   612→          _defaultGrade(sr.percentage);
   613→   613→      return SubjectResultRow(
   614→   614→        subjectName: sr.subjectId, // Caller should map ID → name upstream
   615→   615→        score: sr.totalMarksObtained,
   616→   616→        percentage: sr.percentage,
   617→   617→        grade: grade,
   618→   618→        classAverage: sr.classAverage,
   619→   619→        position: sr.classPosition,
   620→   620→        trend: sr.performanceTrend.label,
   621→   621→      );
   622→   622→    }).toList();
   623→   623→
   624→   624→    // Collect strengths and weaknesses from subject results.
   625→   625→    final strengths = <String>[
   626→   626→      ...overallResult.metadata['strengths'] as List<dynamic>? ?? [],
   627→   627→      ...subjectResults
   628→   628→          .where((sr) => sr.performanceTrend == PerformanceTrend.improving)
   629→   629→          .map((sr) => 'Strong performance in ${sr.subjectId}'),
   630→   630→    ];
   631→   631→
   632→   632→    final weaknesses = <String>[
   633→   633→      ...overallResult.metadata['weaknesses'] as List<dynamic>? ?? [],
   634→   634→      ...subjectResults
   635→   635→          .where((sr) => sr.performanceTrend == PerformanceTrend.declining)
   636→   636→          .map((sr) => 'Needs improvement in ${sr.subjectId}'),
   637→   637→    ];
   638→   638→
   639→   639→    // AI study recommendations from the overall result entity.
   640→   640→    final recommendations = overallResult.aiStudyRecommendations;
   641→   641→
   642→   642→    // Build trend data from supplied points, or default to a single point.
   643→   643→    final trend = trendData ??
   644→   644→        [
   645→   645→          TrendDataPoint(
   646→   646→            period: sessionName,
   647→   647→            score: overallResult.overallPercentage,
   648→   648→          ),
   649→   649→        ];
   650→   650→
   651→   651→    final content = StudentReportContent(
   652→   652→      studentName: studentName,
   653→   653→      className: className,
   654→   654→      sessionName: sessionName,
   655→   655→      overallScore: overallResult.overallPercentage,
   656→   656→      overallGrade: overallGrade,
   657→   657→      overallGpa: overallResult.overallGpa,
   658→   658→      classPosition: overallResult.classPosition,
   659→   659→      classSize: overallResult.classSize,
   660→   660→      subjectResults: subjectRows,
   661→   661→      strengths: strengths,
   662→   662→      weaknesses: weaknesses,
   663→   663→      recommendations: recommendations,
   664→   664→      teacherComment: overallResult.teacherComment,
   665→   665→      trendData: trend,
   666→   666→    );
   667→   667→
   668→   668→    return ReportData(
   669→   669→      type: ReportType.student,
   670→   670→      title: 'Student Report — $studentName',
   671→   671→      content: content.toJson(),
   672→   672→      metadata: _buildMetadata(),
   673→   673→    );
   674→   674→  }
   675→   675→
   676→   676→  // ── Class Report ───────────────────────────────────────────────────
   677→   677→
   678→   678→  /// Creates a class-wide performance report.
   679→   679→  ///
   680→   680→  /// Accepts a [ClassPerformanceEntity] for the class statistics,
   681→   681→  /// [studentResults] for building the rankings table, and
   682→   682→  /// [subjectResults] for subject performance comparisons.
   683→   683→  ///
   684→   684→  /// Optional [gradeScale] is used to derive grades from percentages.
   685→   685→  ReportData generateClassReport({
   686→   686→    required ClassPerformanceEntity classPerformance,
   687→   687→    required List<StudentOverallResultEntity> studentResults,
   688→   688→    required String className,
   689→   689→    required String sessionName,
   690→   690→    List<StudentSubjectResultEntity>? subjectResults,
   691→   691→    GradeScaleEntity? gradeScale,
   692→   692→    Map<String, String>? subjectNameMap,
   693→   693→    Map<String, String>? studentNameMap,
   694→   694→  }) {
   695→   695→    // Build student rankings sorted by overall percentage descending.
   696→   696→    final sorted = List<StudentOverallResultEntity>.from(studentResults)
   697→   697→      ..sort((a, b) => b.overallPercentage.compareTo(a.overallPercentage));
   698→   698→
   699→   699→    final rankings = sorted.asMap().entries.map((entry) {
   700→   700→      final sr = entry.value;
   701→   701→      final grade = sr.overallGrade ??
   702→   702→          gradeScale?.applyToPercentage(sr.overallPercentage)?.grade ??
   703→   703→          _defaultGrade(sr.overallPercentage);
   704→   704→      return StudentRankingRow(
   705→   705→        position: entry.key + 1,
   706→   706→        studentName: studentNameMap?[sr.studentId] ?? sr.studentId,
   707→   707→        totalScore: sr.totalMarksObtained,
   708→   708→        percentage: sr.overallPercentage,
   709→   709→        grade: grade,
   710→   710→        isPassed: sr.subjectsFailed == 0,
   711→   711→      );
   712→   712→    }).toList();
   713→   713→
   714→   714→    // Build subject performance comparison entries.
   715→   715→    final subjectComparisons = <SubjectComparisonEntry>[];
   716→   716→    if (subjectResults != null) {
   717→   717→      // Group by subject and compute aggregated stats.
   718→   718→      final bySubject = <String, List<StudentSubjectResultEntity>>{};
   719→   719→      for (final sr in subjectResults) {
   720→   720→        bySubject.putIfAbsent(sr.subjectId, () => []).add(sr);
   721→   721→      }
   722→   722→      for (final entry in bySubject.entries) {
   723→   723→        final scores = entry.value.map((e) => e.percentage).toList();
   724→   724→        if (scores.isEmpty) continue;
   725→   725→        scores.sort();
   726→   726→        final avg = scores.reduce((a, b) => a + b) / scores.length;
   727→   727→        final passed = entry.value.where((e) => e.isPassed).length;
   728→   728→        subjectComparisons.add(SubjectComparisonEntry(
   729→   729→          subjectName: subjectNameMap?[entry.key] ?? entry.key,
   730→   730→          classAverage: avg,
   731→   731→          highestScore: scores.last,
   732→   732→          lowestScore: scores.first,
   733→   733→          passRate: scores.isNotEmpty
   734→   734→              ? (passed / scores.length) * 100
   735→   735→              : 0,
   736→   736→        ));
   737→   737→      }
   738→   738→    }
   739→   739→
   740→   740→    // Identify difficult topics from class performance metadata.
   741→   741→    final difficultTopics = <String>[];
   742→   742→    final topicPerf = classPerformance.topicPerformance;
   743→   743→    for (final entry in topicPerf.entries) {
   744→   744→      final avgScore = entry.value is num ? (entry.value as num).toDouble() : 0.0;
   745→   745→      if (avgScore < 50) {
   746→   746→        difficultTopics.add(entry.key);
   747→   747→      }
   748→   748→    }
   749→   749→
   750→   750→    // Build grade distribution from class performance entity.
   751→   751→    final gradeDist = <String, int>{};
   752→   752→    for (final entry in classPerformance.gradeDistribution.entries) {
   753→   753→      gradeDist[entry.key] = (entry.value is num
   754→   754→          ? (entry.value as num).toInt()
   755→   755→          : int.tryParse(entry.value.toString()) ?? 0);
   756→   756→    }
   757→   757→
   758→   758→    final content = ClassReportContent(
   759→   759→      className: className,
   760→   760→      sessionName: sessionName,
   761→   761→      totalStudents: classPerformance.totalStudents,
   762→   762→      averageScore: classPerformance.averageScore,
   763→   763→      highestScore: classPerformance.highestScore,
   764→   764→      lowestScore: classPerformance.lowestScore,
   765→   765→      passRate: classPerformance.passRate,
   766→   766→      gradeDistribution: gradeDist,
   767→   767→      studentRankings: rankings,
   768→   768→      subjectComparisons: subjectComparisons,
   769→   769→      difficultTopics: difficultTopics,
   770→   770→    );
   771→   771→
   772→   772→    return ReportData(
   773→   773→      type: ReportType.classReport,
   774→   774→      title: 'Class Report — $className',
   775→   775→      content: content.toJson(),
   776→   776→      metadata: _buildMetadata(),
   777→   777→    );
   778→   778→  }
   779→   779→
   780→   780→  // ── School Report ──────────────────────────────────────────────────
   781→   781→
   782→   782→  /// Creates a school-level performance report.
   783→   783→  ///
   784→   784→  /// Uses [SchoolPerformanceEntity] for school-wide statistics and
   785→   785→  /// optional [classPerformances] and [subjectPerformances] for detailed
   786→   786→  /// rankings.
   787→   787→  ReportData generateSchoolReport({
   788→   788→    required SchoolPerformanceEntity schoolPerformance,
   789→   789→    required String schoolName,
   790→   790→    required String sessionName,
   791→   791→    List<ClassPerformanceEntity>? classPerformances,
   792→   792→    Map<String, String>? classNameMap,
   793→   793→    List<StudentSubjectResultEntity>? subjectResults,
   794→   794→    Map<String, String>? subjectNameMap,
   795→   795→    GradeScaleEntity? gradeScale,
   796→   796→  }) {
   797→   797→    // Build class rankings from supplied class performance data.
   798→   798→    final classRankings = <ClassRankingEntry>[];
   799→   799→    if (classPerformances != null) {
   800→   800→      final sorted = List<ClassPerformanceEntity>.from(classPerformances)
   801→   801→        ..sort((a, b) => b.averageScore.compareTo(a.averageScore));
   802→   802→      for (final cp in sorted) {
   803→   803→        classRankings.add(ClassRankingEntry(
   804→   804→          className: classNameMap?[cp.classId] ?? cp.classId,
   805→   805→          averageScore: cp.averageScore,
   806→   806→          passRate: cp.passRate,
   807→   807→          totalStudents: cp.totalStudents,
   808→   808→        ));
   809→   809→      }
   810→   810→    }
   811→   811→
   812→   812→    // Build subject rankings from supplied subject results.
   813→   813→    final subjectRankings = <SubjectRankingEntry>[];
   814→   814→    if (subjectResults != null) {
   815→   815→      final bySubject = <String, List<StudentSubjectResultEntity>>{};
   816→   816→      for (final sr in subjectResults) {
   817→   817→        bySubject.putIfAbsent(sr.subjectId, () => []).add(sr);
   818→   818→      }
   819→   819→      final subjectStats = <String, ({double avg, double passRate, int count})>{};
   820→   820→      for (final entry in bySubject.entries) {
   821→   821→        final scores = entry.value.map((e) => e.percentage).toList();
   822→   822→        if (scores.isEmpty) continue;
   823→   823→        final avg = scores.reduce((a, b) => a + b) / scores.length;
   824→   824→        final passed = entry.value.where((e) => e.isPassed).length;
   825→   825→        subjectStats[entry.key] = (
   826→   826→          avg: avg,
   827→   827→          passRate: (passed / scores.length) * 100,
   828→   828→          count: scores.length,
   829→   829→        );
   830→   830→      }
   831→   831→      // Sort by average descending.
   832→   832→      final sortedSubjects = subjectStats.entries.toList()
   833→   833→        ..sort((a, b) => b.value.avg.compareTo(a.value.avg));
   834→   834→      for (final entry in sortedSubjects) {
   835→   835→        subjectRankings.add(SubjectRankingEntry(
   836→   836→          subjectName: subjectNameMap?[entry.key] ?? entry.key,
   837→   837→          averageScore: entry.value.avg,
   838→   838→          passRate: entry.value.passRate,
   839→   839→          studentCount: entry.value.count,
   840→   840→        ));
   841→   841→      }
   842→   842→    }
   843→   843→
   844→   844→    // Build historical trend data from the entity.
   845→   845→    final trendData = <TrendDataPoint>[];
   846→   846→    for (final point in schoolPerformance.trendData) {
   847→   847→      if (point is Map<String, dynamic>) {
   848→   848→        trendData.add(TrendDataPoint(
   849→   849→          period: point['period'] as String? ?? '',
   850→   850→          score: (point['score'] as num?)?.toDouble() ?? 0,
   851→   851→        ));
   852→   852→      }
   853→   853→    }
   854→   854→
   855→   855→    // Build grade distribution.
   856→   856→    final gradeDist = <String, int>{};
   857→   857→    for (final entry in schoolPerformance.gradeDistribution.entries) {
   858→   858→      gradeDist[entry.key] = (entry.value is num
   859→   859→          ? (entry.value as num).toInt()
   860→   860→          : int.tryParse(entry.value.toString()) ?? 0);
   861→   861→    }
   862→   862→
   863→   863→    final content = SchoolReportContent(
   864→   864→      schoolName: schoolName,
   865→   865→      sessionName: sessionName,
   866→   866→      totalStudents: schoolPerformance.totalStudents,
   867→   867→      totalClasses: schoolPerformance.totalClasses,
   868→   868→      averageScore: schoolPerformance.averageScore,
   869→   869→      passRate: schoolPerformance.passRate,
   870→   870→      distinctionRate: schoolPerformance.distinctionRate,
   871→   871→      gradeDistribution: gradeDist,
   872→   872→      classRankings: classRankings,
   873→   873→      subjectRankings: subjectRankings,
   874→   874→      trendData: trendData,
   875→   875→    );
   876→   876→
   877→   877→    return ReportData(
   878→   878→      type: ReportType.school,
   879→   879→      title: 'School Report — $schoolName',
   880→   880→      content: content.toJson(),
   881→   881→      metadata: _buildMetadata(),
   882→   882→    );
   883→   883→  }
   884→   884→
   885→   885→  // ── Subject Report ─────────────────────────────────────────────────
   886→   886→
   887→   887→  /// Creates a subject-level report with statistics, score distribution,
   888→   888→  /// topic mastery, and student performance table.
   889→   889→  ///
   890→   890→  /// Uses [ClassPerformanceEntity] scoped to a subject for statistics,
   891→   891→  /// and [studentSubjectResults] for the per-student breakdown.
   892→   892→  ReportData generateSubjectReport({
   893→   893→    required ClassPerformanceEntity subjectPerformance,
   894→   894→    required List<StudentSubjectResultEntity> studentSubjectResults,
   895→   895→    required String subjectName,
   896→   896→    required String className,
   897→   897→    required String sessionName,
   898→   898→    List<TopicMasteryEntity>? topicMasteryData,
   899→   899→    Map<String, String>? studentNameMap,
   900→   900→    Map<String, String>? topicNameMap,
   901→   901→    GradeScaleEntity? gradeScale,
   902→   902→  }) {
   903→   903→    // Build score distribution buckets.
   904→   904→    final buckets = _computeScoreBuckets(studentSubjectResults);
   905→   905→
   906→   906→    // Build topic mastery overview.
   907→   907→    final topicEntries = <TopicMasteryEntry>[];
   908→   908→    if (topicMasteryData != null) {
   909→   909→      // Group by topic and compute averages.
   910→   910→      final byTopic = <String, List<TopicMasteryEntity>>{};

   912→   912→        byTopic.putIfAbsent(tm.topicId, () => []).add(tm);

   915→   915→        final items = entry.value;
   916→   916→        final avgAccuracy = items.isEmpty
   917→   917→            ? 0.0

   920→   920→        topicEntries.add(TopicMasteryEntry(
   921→   921→          topicName: topicNameMap?[entry.key] ?? entry.key,
   922→   922→          masteryLevel: _dominantMasteryLevel(items),

   924→   924→          studentsAttempted: items
   925→   925→              .where((t) => t.questionsAttempted > 0)
   926→   926→              .length,
   927→   927→        ));
   928→   928→      }
   929→   929→    }
   930→   930→
   931→   931→    // Build student performance table sorted by percentage descending.
   932→   932→    final sorted = List<StudentSubjectResultEntity>.from(studentSubjectResults)
   933→   933→      ..sort((a, b) => b.percentage.compareTo(a.percentage));
   934→   934→
   935→   935→    final studentRows = sorted.asMap().entries.map((entry) {
   936→   936→      final sr = entry.value;
   937→   937→      final grade = sr.grade ??
   938→   938→          gradeScale?.applyToPercentage(sr.percentage)?.grade ??
   939→   939→          _defaultGrade(sr.percentage);
   940→   940→      return SubjectStudentRow(
   941→   941→        position: entry.key + 1,
   942→   942→        studentName: studentNameMap?[sr.studentId] ?? sr.studentId,
   943→   943→        score: sr.totalMarksObtained,
   944→   944→        percentage: sr.percentage,
   945→   945→        grade: grade,
   946→   946→        isPassed: sr.isPassed,
   947→   947→      );
   948→   948→    }).toList();
   949→   949→
   950→   950→    final content = SubjectReportContent(
   951→   951→      subjectName: subjectName,
   952→   952→      className: className,
   953→   953→      sessionName: sessionName,
   954→   954→      totalStudents: subjectPerformance.totalStudents,
   955→   955→      averageScore: subjectPerformance.averageScore,
   956→   956→      highestScore: subjectPerformance.highestScore,
   957→   957→      lowestScore: subjectPerformance.lowestScore,
   958→   958→      passRate: subjectPerformance.passRate,
   959→   959→      scoreDistribution: buckets,
   960→   960→      topicMastery: topicEntries,
   961→   961→      studentPerformance: studentRows,
   962→   962→    );
   963→   963→
   964→   964→    return ReportData(
   965→   965→      type: ReportType.subject,
   966→   966→      title: 'Subject Report — $subjectName',
   967→   967→      content: content.toJson(),
   968→   968→      metadata: _buildMetadata(),
   969→   969→    );
   970→   970→  }
   971→   971→
   972→   972→  // ── Exam Summary ───────────────────────────────────────────────────
   973→   973→
   974→   974→  /// Creates an exam summary report with overview, statistics, question
   975→   975→  /// analysis, and top/bottom performers.
   976→   976→  ///
   977→   977→  /// Uses [ExamEntity] for exam metadata, [ExamStatistics] for computed
   978→   978→  /// statistics, and [ExamResultEntity] list for student-level data.
   979→   979→  ReportData generateExamSummary({
   980→   980→    required ExamEntity exam,
   981→   981→    required ExamStatistics statistics,
   982→   982→    required List<ExamResultEntity> results,
   983→   983→    Map<String, String>? studentNameMap,
   984→   984→    GradeScaleEntity? gradeScale,
   985→   985→  }) {
   986→   986→    // Build question analysis from statistics.
   987→   987→    final questionAnalysis = <QuestionAnalysisEntry>[];
   988→   988→    final sortedQuestions = statistics.questionsByCorrectRate.entries.toList()
   989→   989→      ..sort((a, b) => a.key.compareTo(b.key));
   990→   990→    for (var i = 0; i < sortedQuestions.length; i++) {
   991→   991→      final entry = sortedQuestions[i];
   992→   992→      questionAnalysis.add(QuestionAnalysisEntry(
   993→   993→        questionNumber: i + 1,
   994→   994→        questionId: entry.key,
   995→   995→        correctRate: entry.value,
   996→   996→        averageTimeSeconds: 0, // Derived from attempt data if available
   997→   997→        skippedCount: 0, // Derived from attempt data if available
   998→   998→      ));
   999→   999→    }
  1000→  1000→
  1001→  1001→    // Build grade distribution from results.
  1002→  1002→    final gradeDist = <String, int>{};
  1003→  1003→    for (final result in results) {
  1004→  1004→      final grade = result.grade ??
  1005→  1005→          gradeScale?.applyToPercentage(result.scorePercentage)?.grade ??
  1006→  1006→          _defaultGrade(result.scorePercentage);
  1007→  1007→      gradeDist[grade] = (gradeDist[grade] ?? 0) + 1;
  1008→  1008→    }
  1009→  1009→
  1010→  1010→    // Sort results for top/bottom performers.
  1011→  1011→    final sorted = List<ExamResultEntity>.from(results)
