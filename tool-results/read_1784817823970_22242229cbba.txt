   283→  // ── Exception mapping helper ───────────────────────────────────────────
   284→  Never _mapPostgrestException(sb.PostgrestException e) {
   285→    AppLogger.error('Postgrest error: ${e.message}', error: e);
   286→    switch (e.code) {
   287→      case 'PGRST116':
   288→        throw NotFoundException(message: e.message);
   289→      case '23505':
   290→        throw const ServerException(
   291→          message: 'A record with this data already exists.',
   292→          statusCode: 409,
   293→        );
   294→      case '23503':
   295→        throw const ServerException(
   296→          message: 'Referenced record not found.',
   297→          statusCode: 404,
   298→        );
   299→      case '42501':
   300→        throw const ForbiddenException(
   301→          message: 'You do not have permission for this action.',
   302→        );
   303→      default:
   304→        throw ServerException(
   305→          message: e.message,
   306→          statusCode: int.tryParse(e.code ?? '') ?? 500,
   307→        );
   308→    }
   309→  }

   315→  @override
   316→  Future<List<ConversationModel>> getConversations({
   317→    int page = 1,
   318→    int perPage = 20,
   319→  }) async {
   320→    try {
   321→      AppLogger.info('Fetching conversations (page: $page, perPage: $perPage)');
   322→      final response = await _supabase.rpc(
   323→        'get_user_conversations',
   324→        params: {
   325→          'p_user_id': _supabase.auth.currentUser?.id,
   326→          'p_page': page,
   327→          'p_per_page': perPage,
   328→        },
   329→      );
   330→      final list = response as List<dynamic>;
   331→      AppLogger.info('Fetched ${list.length} conversations');
   332→      return list
   333→          .map((e) => ConversationModel.fromJson(e as Map<String, dynamic>))
   334→          .toList();
   335→    } on sb.PostgrestException catch (e) {
   336→      _mapPostgrestException(e);
   337→    } catch (e) {
   338→      AppLogger.error('Failed to fetch conversations', error: e);
   339→      throw ServerException(message: e.toString(), statusCode: 500);
   340→    }
   341→  }
   342→
   343→  @override
   344→  Future<ConversationModel> getConversation(String conversationId) async {
   345→    try {
   346→      AppLogger.info('Fetching conversation: $conversationId');
   347→      final response = await _supabase
   348→          .from(_conversationsTable)
   349→          .select('*, participants:$_participantsTable(*)')
   350→          .eq('id', conversationId)
   351→          .single();
   352→      AppLogger.info('Conversation fetched successfully');
   353→      return ConversationModel.fromJson(response);
   354→    } on sb.PostgrestException catch (e) {
   355→      if (e.code == 'PGRST116') {
   356→        throw NotFoundException(
   357→          message: 'Conversation not found: $conversationId',
   358→        );
   359→      }
   360→      _mapPostgrestException(e);
   361→    } catch (e) {
   362→      AppLogger.error('Failed to fetch conversation: $conversationId', error: e);
   363→      throw ServerException(message: e.toString(), statusCode: 500);
   364→    }
   365→  }
