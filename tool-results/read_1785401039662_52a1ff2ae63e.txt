   882→-- Get platform-wide dashboard metrics (cached in dashboard_snapshots)
   883→CREATE OR REPLACE FUNCTION get_platform_dashboard_metrics()
   884→RETURNS JSONB AS $$
   885→DECLARE
   886→  result JSONB;
   887→BEGIN
   888→  SELECT jsonb_build_object(
   889→    'total_schools', (SELECT COUNT(*) FROM schools WHERE is_active = TRUE),
   890→    'total_teachers', (SELECT COUNT(*) FROM user_profiles WHERE role = 'teacher' AND is_active = TRUE),
   891→    'total_students', (SELECT COUNT(*) FROM user_profiles WHERE role = 'student' AND is_active = TRUE),
   892→    'total_parents', (SELECT COUNT(*) FROM user_profiles WHERE role = 'parent' AND is_active = TRUE),
   893→    'active_exams', (SELECT COUNT(*) FROM exams WHERE status IN ('active', 'in_progress')),
   894→    'daily_active_users', (SELECT COUNT(DISTINCT user_id) FROM login_monitoring WHERE is_success = TRUE AND created_at >= now() - INTERVAL '24 hours'),
   895→    'monthly_active_users', (SELECT COUNT(DISTINCT user_id) FROM login_monitoring WHERE is_success = TRUE AND created_at >= now() - INTERVAL '30 days'),
   896→    'ai_requests_today', (SELECT COUNT(*) FROM ai_request_logs WHERE created_at >= CURRENT_DATE),
   897→    'revenue_today', (SELECT COALESCE(SUM(amount), 0) FROM transactions WHERE status = 'completed' AND created_at >= CURRENT_DATE),
   898→    'monthly_revenue', (SELECT COALESCE(SUM(amount), 0) FROM transactions WHERE status = 'completed' AND created_at >= DATE_TRUNC('month', CURRENT_DATE)),
   899→    'annual_revenue', (SELECT COALESCE(SUM(amount), 0) FROM transactions WHERE status = 'completed' AND created_at >= DATE_TRUNC('year', CURRENT_DATE)),
   900→    'active_subscriptions', (SELECT COUNT(*) FROM subscriptions WHERE status = 'active'),
   901→    'trial_accounts', (SELECT COUNT(*) FROM subscriptions WHERE status = 'trialing'),
   902→    'computed_at', now()
   903→  ) INTO result;
   904→
   905→  -- Cache the result
   906→  INSERT INTO dashboard_snapshots (snapshot_type, data, expires_at)
   907→  VALUES ('platform_metrics', result, now() + INTERVAL '5 minutes')
   908→  ON CONFLICT (snapshot_type) DO UPDATE SET data = result, computed_at = now(), expires_at = now() + INTERVAL '5 minutes';
   909→
   910→  RETURN result;
   911→END;
   912→$$ LANGUAGE plpgsql SECURITY DEFINER;

   914→-- Get school growth metrics
   915→CREATE OR REPLACE FUNCTION get_school_growth_metrics(
   916→  period_start DATE DEFAULT CURRENT_DATE - INTERVAL '12 months',
   917→  period_end DATE DEFAULT CURRENT_DATE
   918→)
   919→RETURNS TABLE(month DATE, new_schools BIGINT, total_schools BIGINT, churned_schools BIGINT) AS $$
   920→BEGIN
   921→  RETURN QUERY
   922→  WITH months AS (
   923→    SELECT generate_series(period_start, period_end, INTERVAL '1 month')::DATE AS month
   924→  ),
   925→  new_counts AS (
   926→    SELECT DATE_TRUNC('month', created_at)::DATE AS month, COUNT(*)::BIGINT AS new_schools
   927→    FROM schools WHERE created_at >= period_start AND created_at <= period_end
   928→    GROUP BY DATE_TRUNC('month', created_at)
   929→  ),
   930→  churn_counts AS (
   931→    SELECT DATE_TRUNC('month', updated_at)::DATE AS month, COUNT(*)::BIGINT AS churned_schools
   932→    FROM schools WHERE is_active = FALSE AND updated_at >= period_start AND updated_at <= period_end
   933→    GROUP BY DATE_TRUNC('month', updated_at)
   934→  )
   935→  SELECT m.month,
   936→    COALESCE(nc.new_schools, 0),
   937→    (SELECT COUNT(*) FROM schools WHERE created_at <= m.month + INTERVAL '1 month' AND is_active = TRUE)::BIGINT,
   938→    COALESCE(cc.churned_schools, 0)
   939→  FROM months m
   940→  LEFT JOIN new_counts nc ON nc.month = m.month
   941→  LEFT JOIN churn_counts cc ON cc.month = m.month
   942→  ORDER BY m.month;
   943→END;
   944→$$ LANGUAGE plpgsql SECURITY DEFINER;

   948→  period_start DATE DEFAULT CURRENT_DATE - INTERVAL '30 days',
   949→  period_end DATE DEFAULT CURRENT_DATE
   950→)
   951→RETURNS TABLE(date DATE, total_requests BIGINT, successful_requests BIGINT, failed_requests BIGINT,
   952→  total_tokens BIGINT, total_cost DECIMAL, avg_latency_ms DECIMAL) AS $$

   956→    COUNT(*)::BIGINT,

   962→  FROM ai_request_logs

   964→  GROUP BY created_at::DATE
   965→  ORDER BY date;
   966→END;

   970→CREATE OR REPLACE FUNCTION detect_suspicious_logins(
   971→  lookback_hours INTEGER DEFAULT 24,
   972→  failure_threshold INTEGER DEFAULT 5
   973→)
   974→RETURNS TABLE(user_id UUID, email VARCHAR, failure_count BIGINT, distinct_ips BIGINT, last_attempt TIMESTAMPTZ) AS $$
   975→BEGIN
   976→  RETURN QUERY
   977→  SELECT lm.user_id, lm.email,
   978→    COUNT(*)::BIGINT AS failure_count,
   979→    COUNT(DISTINCT lm.ip_address)::BIGINT AS distinct_ips,
   980→    MAX(lm.created_at) AS last_attempt
   981→  FROM login_monitoring lm
   982→  WHERE lm.is_success = FALSE
   983→    AND lm.created_at >= now() - (lookback_hours || ' hours')::INTERVAL
   984→  GROUP BY lm.user_id, lm.email
   985→  HAVING COUNT(*) >= failure_threshold
   986→  ORDER BY failure_count DESC;
   987→END;
   988→$$ LANGUAGE plpgsql SECURITY DEFINER;
