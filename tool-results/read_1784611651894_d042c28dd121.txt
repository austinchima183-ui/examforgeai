   248→  @override
   249→  Future<List<SubscriptionPlanModel>> getSubscriptionPlans({
   250→    String? billingModel,
   251→    bool activeOnly = true,
   252→    int limit = PaginatedQueryMixin.dropdownPageSize,
   253→  }) async {
   254→    try {
   255→      var query = _supabase.from(_plansTable).select();
   256→
   257→      if (billingModel != null) {
   258→        query = query.eq('billing_model', billingModel);
   259→      }
   260→      if (activeOnly) {
   261→        query = query.eq('is_active', true);
   262→      }
   263→
   264→      // PERF: Added limit to prevent unbounded query on subscription_plans
   265→      final response = query.order('sort_order', ascending: true).limit(limit);
   266→
   267→      final list = await response;
   268→      AppLogger.info('Fetched ${list.length} subscription plans');
   269→      return list
   270→          .map<SubscriptionPlanModel>(
   271→            (row) => SubscriptionPlanModel.fromJson(row),
   272→          )
   273→          .toList();
   274→    } on sb.PostgrestException catch (e) {
   275→      throw _mapPostgrestException(e);
   276→    } catch (e) {
   277→      AppLogger.error('Unexpected getSubscriptionPlans error', error: e);
   278→      throw const ServerException(
   279→        message: 'Failed to fetch subscription plans.',
   280→        statusCode: 500,
   281→      );
   282→    }
   283→  }
   284→
   285→  @override
   286→  Future<SubscriptionPlanModel> getSubscriptionPlan(String planId) async {
   287→    try {
   288→      final response = await _supabase
   289→          .from(_plansTable)
   290→          .select()
   291→          .eq('id', planId)
   292→          .limit(1);
   293→
   294→      if (response.isEmpty) {
   295→        throw const NotFoundException('Subscription plan not found.');
   296→      }
   297→
   298→      return SubscriptionPlanModel.fromJson(response.first);
   299→    } on sb.PostgrestException catch (e) {
   300→      throw _mapPostgrestException(e);
   301→    } on NotFoundException {
   302→      rethrow;
   303→    } catch (e) {
   304→      AppLogger.error('Unexpected getSubscriptionPlan error', error: e);
   305→      throw const ServerException(
   306→        message: 'Failed to fetch subscription plan.',
   307→        statusCode: 500,
   308→      );
   309→    }
   310→  }

   313→  Future<SubscriptionPlanModel> upsertSubscriptionPlan(

   318→
   319→      final response = await _supabase
   320→          .from(_plansTable)

   322→          .select();

   325→        throw const ServerException(
   326→          message: 'Subscription plan upsert returned no data.',
   327→          statusCode: 500,
   328→        );
   329→      }
   330→
   331→      AppLogger.info('Subscription plan upserted: ${response.first['id']}');
   332→      return SubscriptionPlanModel.fromJson(response.first);
   333→    } on sb.PostgrestException catch (e) {
   334→      throw _mapPostgrestException(e);
   335→    } on ServerException {
   336→      rethrow;
   337→    } catch (e) {

   383→  @override
   384→  Future<List<SubscriptionModel>> getSubscriptions({
   385→    String? schoolId,
   386→    String? subscriberType,
   387→    String? status,
   388→    int page = 1,
   389→    int perPage = 20,
   390→  }) async {
   391→    try {
   392→      var query = _supabase.from(_subscriptionsTable).select();
   393→
   394→      if (schoolId != null) {
   395→        query = query.eq('school_id', schoolId);
   396→      }
   397→      if (subscriberType != null) {
   398→        query = query.eq('subscriber_type', subscriberType);
   399→      }
   400→      if (status != null) {
   401→        query = query.eq('status', status);
   402→      }
   403→
   404→      final filtered = _applyFilters(query, {
   405→        if (schoolId != null) 'school_id': schoolId,
   406→        if (subscriberType != null) 'subscriber_type': subscriberType,
   407→        if (status != null) 'status': status,
   408→      });
   409→
   410→      var transformed =
   411→          filtered.order('created_at', ascending: false) as sb.PostgrestTransformBuilder<dynamic>;
   412→      transformed = _applyPagination(transformed, page: page, perPage: perPage);
   413→
   414→      final list = await transformed;
   415→      AppLogger.info('Fetched ${list.length} subscriptions');
   416→      return list
   417→          .map<SubscriptionModel>(
   418→            (row) => SubscriptionModel.fromJson(row),
   419→          )

  1070→  @override
  1071→  Future<CouponModel> validateCoupon({
  1072→    required String code,
  1073→    required String billingModel,
  1074→    String? planId,
  1075→  }) async {
  1076→    try {
  1077→      var query = _supabase
  1078→          .from(_couponsTable)
  1079→          .select()
  1080→          .eq('code', code)
  1081→          .eq('billing_model', billingModel)
  1082→          .eq('is_active', true) as sb.PostgrestFilterBuilder<dynamic>;
  1083→
  1084→      if (planId != null) {
  1085→        query = query.eq('plan_id', planId);
  1086→      }
  1087→
  1088→      final response = await query.limit(1);
  1089→
  1090→      if (response.isEmpty) {
  1091→        throw const NotFoundException('Invalid or expired coupon code.');
  1092→      }
  1093→
  1094→      final couponData = response.first;
  1095→
  1096→      // Validate expiry
  1097→      final validUntil = couponData['valid_until'] != null
  1098→          ? DateTime.tryParse(couponData['valid_until'] as String)
  1099→          : null;
  1100→      if (validUntil != null && validUntil.isBefore(DateTime.now())) {
  1101→        throw const ValidationException(
  1102→          message: 'This coupon has expired.',
  1103→          fieldErrors: {'code': 'Coupon has expired.'},
  1104→        );
  1105→      }
  1106→
  1107→      final validFrom = couponData['valid_from'] != null
  1108→          ? DateTime.tryParse(couponData['valid_from'] as String)
  1109→          : null;
  1110→      if (validFrom != null && validFrom.isAfter(DateTime.now())) {
  1111→        throw const ValidationException(
  1112→          message: 'This coupon is not yet valid.',
  1113→          fieldErrors: {'code': 'Coupon is not yet valid.'},
  1114→        );
  1115→      }
  1116→
  1117→      // Validate redemption count
  1118→      final maxRedemptions = couponData['max_redemptions'] as int?;
  1119→      final currentRedemptions =
  1120→          couponData['current_redemptions'] as int? ?? 0;
  1121→      if (maxRedemptions != null &&
  1122→          currentRedemptions >= maxRedemptions) {
  1123→        throw const ValidationException(
  1124→          message: 'This coupon has reached its maximum redemptions.',
  1125→          fieldErrors: {'code': 'Coupon redemption limit reached.'},
  1126→        );
  1127→      }
  1128→
  1129→      AppLogger.info('Coupon validated: $code');
  1130→      return CouponModel.fromJson(couponData);
  1131→    } on sb.PostgrestException catch (e) {
  1132→      throw _mapPostgrestException(e);
  1133→    } on NotFoundException {
  1134→      rethrow;
  1135→    } on ValidationException {
  1136→      rethrow;
  1137→    } catch (e) {
  1138→      AppLogger.error('Unexpected validateCoupon error', error: e);
  1139→      throw const ServerException(
  1140→        message: 'Failed to validate coupon.',
  1141→        statusCode: 500,
  1142→      );
  1143→    }
  1144→  }

  1181→            'updated_at': DateTime.now().toIso8601String(),

  1887→  /// Applies pagination using Supabase range.
  1888→  sb.PostgrestTransformBuilder<dynamic> _applyPagination(
  1889→    sb.PostgrestTransformBuilder<dynamic> query, {
  1890→    int? page,
  1891→    int? perPage,
  1892→  }) {
  1893→    final p = page ?? 1;
  1894→    final pp = perPage ?? 20;
  1895→    final from = (p - 1) * pp;
  1896→    final to = from + pp - 1;
  1897→    return query.range(from, to);
  1898→  }

  1900→  /// Maps a Supabase [sb.PostgrestException] to a domain exception.
  1901→  Exception _mapPostgrestException(sb.PostgrestException e) {
  1902→    final statusCode = e.code != null ? int.tryParse(e.code!) ?? 0 : 0;
  1903→    final message = e.message ?? 'An unexpected database error occurred.';
  1904→
  1905→    AppLogger.warning(
  1906→      'Supabase PostgrestException — code: ${e.code}, message: $message',
  1907→    );
  1908→
  1909→    switch (statusCode) {
  1910→      case 401:
  1911→        return UnauthorizedException(message);
  1912→      case 403:
  1913→        return ForbiddenException(message);
  1914→      case 404:
  1915→        return NotFoundException(message);
  1916→      case 422:
  1917→        return ValidationException(
  1918→          message: message,
  1919→          fieldErrors: e.details is Map<String, dynamic>
  1920→              ? e.details as Map<String, String>
  1921→              : {},
  1922→        );
  1923→      default:
  1924→        return ServerException(
  1925→          message: message,
  1926→          statusCode: statusCode,
  1927→          data: e.details,
  1928→        );
  1929→    }
  1930→  }

  1932→  /// Generates a unique referral code from the referrer ID and type.
  1933→  String _generateReferralCode(String referrerId, String referrerType) {
  1934→    final prefix = referrerType == 'teacher_saas'
  1935→        ? 'TCH'
  1936→        : referrerType == 'school_saas'
  1937→            ? 'SCH'
  1938→            : 'ENT';
  1939→    final suffix = referrerId.substring(0, referrerId.length > 8 ? 8 : referrerId.length).toUpperCase();
  1940→    final timestamp = DateTime.now().millisecondsSinceEpoch.toRadixString(36).toUpperCase();
  1941→    return '$prefix-$suffix-$timestamp';
  1942→  }
