    12→/// AI Tutor chat interface page.
    13→///
    14→/// Features:
    15→/// - Conversation list sidebar (desktop) / drawer (mobile)
    16→/// - Chat area with message bubbles
    17→/// - Input field with send button at bottom
    18→/// - New conversation FAB
    19→/// - Subject/topic selection when creating new conversation
    20→/// - Auto-scroll to bottom on new messages
    21→/// - Typing indicator for AI responses
    22→/// - Markdown rendering for AI responses
    23→/// - Empty state for no conversations
    24→class AiTutorPage extends ConsumerStatefulWidget {
    25→  const AiTutorPage({super.key});
    26→
    27→  @override
    28→  ConsumerState<AiTutorPage> createState() => _AiTutorPageState();
    29→}
    30→
    31→class _AiTutorPageState extends ConsumerState<AiTutorPage> {
    32→  final TextEditingController _messageController = TextEditingController();
    33→  final ScrollController _scrollController = ScrollController();
    34→  final GlobalKey<AnimatedListState> _listKey = GlobalKey();
    35→
    36→  @override
    37→  void initState() {
    38→    super.initState();
    39→    WidgetsBinding.instance.addPostFrameCallback((_) {
    40→      ref.read(aiTutorProvider.notifier).loadConversations();
    41→    });
    42→  }
    43→
    44→  @override
    45→  void dispose() {
    46→    _messageController.dispose();
    47→    _scrollController.dispose();
    48→    super.dispose();
    49→  }
    50→
    51→  void _scrollToBottom() {
    52→    WidgetsBinding.instance.addPostFrameCallback((_) {
    53→      if (_scrollController.hasClients) {
    54→        _scrollController.animateTo(
    55→          _scrollController.position.maxScrollExtent,
    56→          duration: const Duration(milliseconds: 300),
    57→          curve: Curves.easeOut,
    58→        );
    59→      }
    60→    });
    61→  }
    62→
    63→  @override
    64→  Widget build(BuildContext context) {
    65→    final tutorState = ref.watch(aiTutorProvider);
    66→
    67→    if (tutorState.isLoading && tutorState.conversations.isEmpty) {
    68→      return const Center(child: AppLoadingSpinner());
    69→    }
    70→
    71→    if (tutorState.error != null && tutorState.conversations.isEmpty) {
    72→      return AppErrorState(
    73→        icon: Icons.error_outline_rounded,
    74→        title: 'Failed to Load Conversations',
    75→        message: tutorState.error,
    76→        onRetry: () =>
    77→            ref.read(aiTutorProvider.notifier).loadConversations(),
    78→      );
    79→    }
    80→
    81→    if (context.isDesktop) {
    82→      return _buildDesktopLayout(context, tutorState);
    83→    }
    84→    return _buildMobileLayout(context, tutorState);
    85→  }

   505→  // ─── Actions ────────────────────────────────────────────────────────
   506→
   507→  void _sendCurrentMessage() {
   508→    final text = _messageController.text.trim();
   509→    if (text.isEmpty) return;
   510→    _messageController.clear();
   511→    ref.read(aiTutorProvider.notifier).sendMessage(text);
   512→    _scrollToBottom();
   513→  }
   514→
   515→  void _sendMessage(String text) {
   516→    _messageController.text = text;
   517→    _sendCurrentMessage();
   518→  }

   633→class _ConversationTile extends StatelessWidget {

   646→  @override
