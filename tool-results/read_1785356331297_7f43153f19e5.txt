   167→/// Data-layer representation of a configurable grading scale.
   168→class GradeScaleModel {
   169→  const GradeScaleModel({
   170→    required this.id,
   171→    required this.schoolId,
   172→    required this.name,
   173→    required this.gradeType,
   174→    this.isDefault = false,
   175→    this.isActive = true,
   176→    this.scaleEntries = const [],
   177→    this.createdBy,
   178→    this.settings = const {},
   179→    required this.createdAt,
   180→    required this.updatedAt,
   181→  });
   182→
   183→  final String id;
   184→  final String schoolId;
   185→  final String name;
   186→  final String gradeType;
   187→  final bool isDefault;
   188→  final bool isActive;
   189→  final List<GradeScaleEntryModel> scaleEntries;
   190→  final String? createdBy;
   191→  final Map<String, dynamic> settings;
   192→  final DateTime createdAt;
   193→  final DateTime updatedAt;
   194→
   195→  // ─── JSON Serialization ────────────────────────────────────────────
   196→
   197→  factory GradeScaleModel.fromJson(Map<String, dynamic> json) {
   198→    // Parse nested scale entries
   199→    final rawEntries = json['scale_entries'] ?? json['scaleEntries'];
   200→    final List<GradeScaleEntryModel> entries = rawEntries is List
   201→        ? rawEntries
   202→            .map(
   203→                (e) => GradeScaleEntryModel.fromJson(e as Map<String, dynamic>),)
   204→            .toList()
   205→        : <GradeScaleEntryModel>[];
   206→
   207→    return GradeScaleModel(
   208→      id: json['id'] as String? ?? '',
   209→      schoolId:
   210→          json['school_id'] as String? ?? json['schoolId'] as String? ?? '',
   211→      name: json['name'] as String? ?? '',
   212→      gradeType: json['grade_type'] as String? ??
   213→          json['gradeType'] as String? ??
   214→          'percentage',
   215→      isDefault:
   216→          json['is_default'] as bool? ?? json['isDefault'] as bool? ?? false,
   217→      isActive:
   218→          json['is_active'] as bool? ?? json['isActive'] as bool? ?? true,
   219→      scaleEntries: entries,
   220→      createdBy:
   221→          json['created_by'] as String? ?? json['createdBy'] as String?,
   222→      settings: json['settings'] as Map<String, dynamic>? ?? {},
   223→      createdAt: json['created_at'] != null
   224→          ? DateTime.parse(json['created_at'] as String)
   225→          : json['createdAt'] != null
   226→              ? DateTime.parse(json['createdAt'] as String)
   227→              : DateTime.now(),
   228→      updatedAt: json['updated_at'] != null
   229→          ? DateTime.parse(json['updated_at'] as String)
   230→          : json['updatedAt'] != null
   231→              ? DateTime.parse(json['updatedAt'] as String)
   232→              : DateTime.now(),
   233→    );
   234→  }
   235→
   236→  Map<String, dynamic> toJson() {
   237→    return {
   238→      'id': id,
   239→      'school_id': schoolId,
   240→      'name': name,
   241→      'grade_type': gradeType,
   242→      'is_default': isDefault,
   243→      'is_active': isActive,
   244→      'scale_entries': scaleEntries.map((e) => e.toJson()).toList(),
   245→      'created_by': createdBy,
   246→      'settings': settings,
   247→      'created_at': createdAt.toIso8601String(),
   248→      'updated_at': updatedAt.toIso8601String(),
   249→    };
   250→  }
   251→
   252→  // ─── Entity Conversion ─────────────────────────────────────────────
   253→
