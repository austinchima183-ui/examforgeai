    11→/// all operations to [SchoolManagementRemoteDataSource].
    12→///
    13→/// Responsibilities:
    14→/// - Converts domain entities to/from data models
    15→/// - Catches data-layer exceptions and converts them to domain [Failure] types
    16→/// - Returns [Result] to force callers to handle both success and failure

    18→  SchoolManagementRepositoryImpl({

    22→  final SchoolManagementRemoteDataSource _remoteDataSource;

    24→  // ═══════════════════════════════════════════════════════════════════════
    25→  // Helper: Convert exceptions to Failures
    26→  // ═══════════════════════════════════════════════════════════════════════
    27→
    28→  Failure _mapExceptionToFailure(Object e) {
    29→    if (e is AuthException) {
    30→      return Failure.auth(message: e.message, code: e.code);
    31→    } else if (e is ServerException) {
    32→      return Failure.server(
    33→        message: e.message,
    34→        statusCode: e.statusCode,
    35→        data: e.data,
    36→      );
    37→    } else if (e is CacheException) {
    38→      return Failure.cache(message: e.message);
    39→    } else if (e is NetworkException) {
    40→      return Failure.network(message: e.message);
    41→    } else if (e is ValidationException) {
    42→      return Failure.validation(
    43→        message: e.message,
    44→        fieldErrors: e.fieldErrors,
    45→      );
    46→    } else if (e is NotFoundException) {
    47→      return Failure.notFound(message: e.message);
    48→    } else if (e is UnauthorizedException) {
    49→      return Failure.unauthorized(message: e.message);
    50→    } else if (e is ForbiddenException) {
    51→      return Failure.forbidden(message: e.message);
    52→    } else {
    53→      AppLogger.error(
    54→        'Unexpected exception in SchoolManagementRepositoryImpl',
    55→        error: e,
    56→      );
    57→      return Failure.server(
    58→        message: 'An unexpected error occurred: $e',
    59→        statusCode: 500,
    60→      );
    61→    }
    62→  }
