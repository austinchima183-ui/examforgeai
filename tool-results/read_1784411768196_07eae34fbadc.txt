   157→// APP ROUTER PROVIDER
   158→// ═══════════════════════════════════════════════════════════════════════

   160→/// Riverpod provider for the application's [GoRouter].
   161→///
   162→/// Watches [authStateProvider] so that the router automatically
   163→/// re-evaluates its [redirect] callback whenever the authentication
   164→/// state changes (sign-in, sign-out, token refresh).
   165→///
   166→/// Usage in `main.dart`:
   167→/// ```dart
   168→/// final router = ref.read(appRouterProvider);
   169→/// MaterialApp.router(routerConfig: router);
   170→/// ```
   171→final appRouterProvider = Provider<GoRouter>((ref) {
   172→  // Watch auth state so the router re-evaluates redirects on change.
   173→  final authState = ref.watch(authStateProvider);
   174→
   175→  // Also watch onboarding state and role so those redirects stay live.
   176→  ref.watch(onboardingCompleteProvider);
   177→  ref.watch(userRoleProvider);
   178→
   179→  return GoRouter(
   180→    navigatorKey: _rootNavigatorKey,
   181→    initialLocation: RouteNames.splash,
   182→    debugLogDiagnostics: true,
   183→
   184→    // ─── Global Redirect ──────────────────────────────────────────
   185→    redirect: (context, state) {
   186→      final currentPath = state.matchedLocation;
   187→      final isAuthenticated = authState.when(
   188→        data: (authData) => authData.session != null,
   189→        loading: () {
   190→          // While loading, only allow splash. Everything else redirects
   191→          // to splash so we don't flash the login page briefly.
   192→          if (currentPath != RouteNames.splash) {
   193→            return RouteNames.splash;
   194→          }
   195→          return null;
   196→        },
   197→        error: (_, __) => false,
   198→      );
   199→
   200→      // ── Auth Guard ────────────────────────────────────────────
   201→      final authRedirect = AuthGuard.evaluate(
   202→        isAuthenticated: isAuthenticated,
   203→        currentPath: currentPath,
   204→      );
   205→      if (authRedirect != null) return authRedirect;
   206→
   207→      // ── Onboarding Guard ─────────────────────────────────────
   208→      final onboardingAsync = ref.read(onboardingCompleteProvider);
   209→      final isOnboardingComplete = onboardingAsync.when(
   210→        data: (complete) => complete,
   211→        loading: () => true, // Assume complete while loading to avoid flicker.
   212→        error: (_, __) => true,
   213→      );
   214→
   215→      final onboardingRedirect = OnboardingGuard.evaluate(
   216→        isAuthenticated: isAuthenticated,
   217→        isOnboardingComplete: isOnboardingComplete,
   218→        currentPath: currentPath,
   219→      );
   220→      if (onboardingRedirect != null) return onboardingRedirect;
   221→
   222→      // ── Role-Based Guard ─────────────────────────────────────
   223→      final roleAsync = ref.read(userRoleProvider);
   224→      final userRole = roleAsync.when(
   225→        data: UserRole.fromString,
   226→        loading: () => null,
   227→        error: (_, __) => null,
   228→      );
   229→
   230→      final roleRedirect = RoleBasedGuard.evaluate(
   231→        userRole: userRole,
   232→        currentPath: currentPath,
   233→      );
   234→      if (roleRedirect != null) return roleRedirect;
   235→
   236→      // No redirect needed — proceed as requested.
   237→      return null;
   238→    },
   239→

   241→    routes: [
   242→      // ── Splash ────────────────────────────────────────────────

   247→      ),
