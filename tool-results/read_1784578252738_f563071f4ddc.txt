    11→// ───────────────────────────────────────────────────────────────────────
    12→// Helper: parse DateTime accepting both snake_case and camelCase keys
    13→// ───────────────────────────────────────────────────────────────────────
    14→
    15→DateTime _parseDateTime(dynamic value) {
    16→  if (value is String) return DateTime.parse(value);
    17→  if (value is DateTime) return value;
    18→  return DateTime.now();
    19→}
    20→
    21→DateTime? _parseDateTimeNullable(dynamic value) {
    22→  if (value == null) return null;
    23→  if (value is String) return DateTime.parse(value);
    24→  if (value is DateTime) return value;
    25→  return null;
    26→}
