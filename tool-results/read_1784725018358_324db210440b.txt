    71→class BulkUpdateSettingsUseCase {
    72→  BulkUpdateSettingsUseCase(this._repository);
    73→  final SuperAdminRepository _repository;
    74→
    75→  Future<Result<List<PlatformSetting>>> call(BulkUpdateSettingsParams params) {
    76→    if (params.settings.isEmpty) {
    77→      return const FailureResult(Failure.validation(message: 'No settings provided'));
    78→    }
    79→    final readonlySettings = params.settings.where((s) => s.isReadonly).toList();
    80→    if (readonlySettings.isNotEmpty) {
    81→      return FailureResult(Failure.validation(
    82→        message: 'Cannot modify readonly settings: ${readonlySettings.map((s) => s.key).join(', ')}',
    83→      ));
    84→    }
    85→    return _repository.bulkUpdateSettings(params.settings);
    86→  }
