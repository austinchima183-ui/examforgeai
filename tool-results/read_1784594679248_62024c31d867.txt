    22→/// - Attendance summary card with rate and day breakdown

    60→  /// Loads attendance data for the currently selected month.
    61→  void _loadAttendanceForMonth() {
    62→    final start = DateTime(_selectedMonth.year, _selectedMonth.month, 1);
    63→    final end = DateTime(_selectedMonth.year, _selectedMonth.month + 1, 0);
    64→
    65→    ref.read(childAttendanceProvider.notifier).loadAttendance(
    66→          widget.studentId,
    67→          startDate: start,
    68→          endDate: end,
    69→        );
    70→  }

    86→  Widget _buildBody(BuildContext context, ChildAttendanceState state) {
    87→    // Loading state
    88→    if (state.isLoading && state.attendance == null) {
    89→      return _buildShimmerLoading(context);
    90→    }
    91→
    92→    // Error state
    93→    if (state.error != null && state.attendance == null) {
    94→      return AppErrorState.genericError(
    95→        message: state.error,
    96→        onRetry: () => _loadAttendanceForMonth(),
    97→      );
    98→    }
    99→
   100→    // Empty state
   101→    final attendance = state.attendance;
   102→    if (attendance == null) {
   103→      return AppEmptyState.noData(
   104→        title: 'No Attendance Data',
   105→        subtitle:
   106→            'Attendance records will appear here once available.',
   107→        actionLabel: 'Retry',
   108→        onAction: () => _loadAttendanceForMonth(),
   109→      );
   110→    }
   111→
   112→    // Success — render attendance
   113→    return RefreshIndicator(
   114→      onRefresh: () async {
   115→        ref
   116→            .read(childAttendanceProvider.notifier)
   117→            .refreshAttendance(
   118→              widget.studentId,
   119→              startDate: DateTime(
   120→                  _selectedMonth.year, _selectedMonth.month, 1),
   121→              endDate: DateTime(
   122→                  _selectedMonth.year, _selectedMonth.month + 1, 0),
   123→            );
   124→      },
   125→      child: SingleChildScrollView(
   126→        physics: const AlwaysScrollableScrollPhysics(),
   127→        padding: const EdgeInsets.only(bottom: Spacings.xxl),
   128→        child: Column(
   129→          crossAxisAlignment: CrossAxisAlignment.start,
   130→          children: [
   131→            // ─── Attendance Summary Card ────────────────────────────
   132→            _buildAttendanceSummaryCard(context, attendance),
   133→
   134→            const SizedBox(height: Spacings.xl),
   135→
   136→            // ─── Month Selector ─────────────────────────────────────
   137→            _buildMonthSelector(context),
   138→
   139→            const SizedBox(height: Spacings.md),
   140→
   141→            // ─── Monthly Attendance Calendar ────────────────────────
   142→            _buildMonthlyCalendar(context, attendance),

   416→  Widget _buildMonthlyCalendar(
   417→    BuildContext context,
   418→    ChildAttendanceEntity attendance,
   419→  ) {
   420→    final cs = context.colorScheme;
   421→    final tt = context.textTheme;
   422→
   423→    // Build a map of date -> status from records
   424→    final attendanceMap = <int, String>{};
   425→    for (final record in attendance.records) {
   426→      final dateStr = record['date'] as String?;
   427→      final status = record['status'] as String?;
   428→      if (dateStr != null && status != null) {
   429→        final date = DateTime.tryParse(dateStr);
   430→        if (date != null &&
   431→            date.year == _selectedMonth.year &&
   432→            date.month == _selectedMonth.month) {
   433→          attendanceMap[date.day] = status;
   434→        }
   435→      }
   436→    }
   437→
   438→    final daysInMonth =
   439→        DateTime(_selectedMonth.year, _selectedMonth.month + 1, 0).day;
   440→    final firstWeekday =
   441→        DateTime(_selectedMonth.year, _selectedMonth.month, 1).weekday;
   442→    // Convert Monday=1 to Monday=0 for our grid
   443→    final startOffset = firstWeekday - 1;
   444→
   445→    return Padding(
   446→      padding: const EdgeInsets.symmetric(horizontal: Spacings.lg),
   447→      child: Card(
   448→        elevation: Spacings.elevationSm,
   449→        shape: RoundedRectangleBorder(
   450→          borderRadius: Spacings.borderRadiusLg,
   451→        ),
   452→        child: Padding(
   453→          padding: Spacings.paddingCard,
   454→          child: Column(
   455→            children: [
   456→              // Weekday headers
   457→              Row(
   458→                children: ['Mon', 'Tue', 'Wed', 'Thu', 'Fri', 'Sat', 'Sun']
   459→                    .map((day) => Expanded(
   460→                          child: Center(
   461→                            child: Text(
   462→                              day,
   463→                              style: tt.labelSmall?.copyWith(
   464→                                color: cs.onSurfaceVariant,
   465→                                fontWeight: AppTypography.wSemiBold,
   466→                              ),
   467→                            ),
   468→                          ),
   469→                        ))
   470→                    .toList(),
   471→              ),
   472→              const SizedBox(height: Spacings.sm),
   473→              // Day grid
   474→              Wrap(
   475→                spacing: 0,
   476→                runSpacing: 0,
   477→                children: [
   478→                  // Empty cells for offset
   479→                  for (int i = 0; i < startOffset; i++)
   480→                    SizedBox(
   481→                      width: (MediaQuery.of(context).size.width -
   482→                              Spacings.lg * 2 -
   483→                              Spacings.lg * 2) /
   484→                          7,
   485→                      height: 36,
   486→                    ),
   487→                  // Day cells
   488→                  for (int day = 1; day <= daysInMonth; day++)
   489→                    _buildCalendarDay(
   490→                      context,
   491→                      day: day,
   492→                      status: attendanceMap[day],
   493→                      isWeekend: (startOffset + day - 1) % 7 >= 5,
   494→                    ),
   495→                ],
   496→              ),
   497→              const SizedBox(height: Spacings.md),
   498→              // Legend
   499→              _buildCalendarLegend(context),
   500→            ],
   501→          ),
   502→        ),
   503→      ),
   504→    );
   505→  }

   507→  Widget _buildCalendarDay(
   508→    BuildContext context, {
   509→    required int day,
   510→    String? status,
   511→    required bool isWeekend,
   512→  }) {
   513→    final cs = context.colorScheme;
   514→    final tt = context.textTheme;
   515→    final brightness = cs.brightness;
   516→
   517→    Color bgColor;
   518→    Color textColor;
   519→    switch (status?.toLowerCase()) {
   520→      case 'present':
   521→        bgColor = AppColors.successLight;
   522→        textColor = AppColors.successOf(brightness);
   523→        break;
   524→      case 'absent':
   525→        bgColor = AppColors.errorLight;
   526→        textColor = AppColors.errorOf(brightness);
   527→        break;
   528→      case 'late':
   529→        bgColor = AppColors.warningLight;
   530→        textColor = AppColors.warningOf(brightness);
   531→        break;
   532→      case 'excused':
   533→        bgColor = AppColors.infoLight;
   534→        textColor = AppColors.infoOf(brightness);
   535→        break;
   536→      default:
   537→        bgColor = isWeekend
   538→            ? cs.surfaceContainerHighest
   539→            : Colors.transparent;
   540→        textColor = isWeekend
   541→            ? cs.onSurfaceVariant
   542→            : cs.onSurface;
   543→    }
   544→
   545→    final width = (MediaQuery.of(context).size.width -
   546→            Spacings.lg * 2 -
   547→            Spacings.lg * 2) /
   548→        7;
   549→
   550→    return SizedBox(
   551→      width: width,
   552→      height: 36,
   553→      child: Container(
   554→        margin: const EdgeInsets.all(2),
   555→        decoration: BoxDecoration(
   556→          color: bgColor,
   557→          borderRadius: Spacings.borderRadiusSm,
   558→        ),
   559→        child: Center(
   560→          child: Text(
   561→            '$day',
   562→            style: tt.labelMedium?.copyWith(
   563→              color: textColor,
   564→              fontWeight: status != null
   565→                  ? AppTypography.wSemiBold
   566→                  : AppTypography.wRegular,
   567→            ),
   568→          ),
   569→        ),
   570→      ),
   571→    );
   572→  }

   622→  Widget _buildTrendsCard(
   623→    BuildContext context,
   624→    ChildAttendanceEntity attendance,
   625→  ) {
   626→    final cs = context.colorScheme;
   627→    final tt = context.textTheme;
   628→    final summary = _parseAttendanceSummary(attendance.records);
   629→    final rate = summary.rate;
   630→
   631→    String trendText;
   632→    IconData trendIcon;
   633→    Color trendColor;
   634→    if (rate >= 0.9) {
   635→      trendText = 'Attendance is excellent this month. Keep it up!';
   636→      trendIcon = Icons.trending_up;
   637→      trendColor = AppColors.successOf(cs.brightness);
   638→    } else if (rate >= 0.75) {
   639→      trendText =
   640→          'Attendance is acceptable but could be improved.';
   641→      trendIcon = Icons.trending_flat;
   642→      trendColor = AppColors.warningOf(cs.brightness);
   643→    } else {
   644→      trendText =
   645→          'Attendance has declined this month. Please ensure regular attendance.';
   646→      trendIcon = Icons.trending_down;
   647→      trendColor = AppColors.errorOf(cs.brightness);
   648→    }
   649→
   650→    return Padding(
   651→      padding: const EdgeInsets.symmetric(horizontal: Spacings.lg),
   652→      child: Card(
   653→        elevation: Spacings.elevationNone,
   654→        color: cs.surfaceContainerLow,
   655→        shape: RoundedRectangleBorder(
   656→          borderRadius: Spacings.borderRadiusMd,
   657→        ),
   658→        child: Padding(

   808→  /// Parses attendance records into a summary with counts and rate.
   809→  _AttendanceSummary _parseAttendanceSummary(
   810→    List<Map<String, dynamic>> records,
   811→  ) {
   812→    int present = 0;
   813→    int absent = 0;
   814→    int late = 0;
   815→    int excused = 0;
   816→
   817→    for (final record in records) {
   818→      final status = (record['status'] as String?)?.toLowerCase();
   819→      switch (status) {
   820→        case 'present':
   821→          present++;
   822→          break;
   823→        case 'absent':
   824→          absent++;
   825→          break;
   826→        case 'late':
   827→          late++;
   828→          break;
   829→        case 'excused':
   830→          excused++;
   831→          break;
   832→      }
   833→    }
   834→
   835→    final total = present + absent + late + excused;
   836→    final rate = total > 0 ? (present + late) / total : 0.0;
   837→
   838→    return _AttendanceSummary(
   839→      present: present,
   840→      absent: absent,
   841→      late: late,
   842→      excused: excused,
   843→      total: total,
   844→      rate: rate,
   845→    );
   846→  }

   848→  /// Returns the colour for an attendance rate.
   849→  ///
   850→  /// Green if >90%, amber if 75–90%, red if <75%.
   851→  Color _attendanceRateColor(double rate, Brightness brightness) {
   852→    if (rate > 0.9) return AppColors.successOf(brightness);
   853→    if (rate >= 0.75) return AppColors.warningOf(brightness);
   854→    return AppColors.errorOf(brightness);
   855→  }
