   248→   248→  @override
   249→   249→  Future<List<SubscriptionPlanModel>> getSubscriptionPlans({
   250→   250→    String? billingModel,
   251→   251→    bool activeOnly = true,
   252→   252→    int limit = PaginatedQueryMixin.dropdownPageSize,
   253→   253→  }) async {
   254→   254→    try {
   255→   255→      var query = _supabase.from(_plansTable).select();
   256→   256→
   257→   257→      if (billingModel != null) {
   258→   258→        query = query.eq('billing_model', billingModel);
   259→   259→      }
   260→   260→      if (activeOnly) {
   261→   261→        query = query.eq('is_active', true);
   262→   262→      }
   263→   263→
   264→   264→      // PERF: Added limit to prevent unbounded query on subscription_plans
   265→   265→      final response = query.order('sort_order', ascending: true).limit(limit);
   266→   266→
   267→   267→      final list = await response;
   268→   268→      AppLogger.info('Fetched ${list.length} subscription plans');
   269→   269→      return list
   270→   270→          .map<SubscriptionPlanModel>(
   271→   271→            (row) => SubscriptionPlanModel.fromJson(row),
   272→   272→          )
   273→   273→          .toList();
   274→   274→    } on sb.PostgrestException catch (e) {
   275→   275→      throw _mapPostgrestException(e);
   276→   276→    } catch (e) {
   277→   277→      AppLogger.error('Unexpected getSubscriptionPlans error', error: e);
   278→   278→      throw const ServerException(
   279→   279→        message: 'Failed to fetch subscription plans.',
   280→   280→        statusCode: 500,
   281→   281→      );
   282→   282→    }
   283→   283→  }

   288→   288→      final response = await _supabase
   289→   289→          .from(_plansTable)
   290→   290→          .select()
   291→   291→          .eq('id', planId)
   292→   292→          .limit(1);
   293→   293→
   294→   294→      if (response.isEmpty) {

   296→   296→      }
   297→   297→
   298→   298→      return SubscriptionPlanModel.fromJson(response.first);
   299→   299→    } on sb.PostgrestException catch (e) {
   300→   300→      throw _mapPostgrestException(e);
   301→   301→    } on NotFoundException {
   302→   302→      rethrow;
   303→   303→    } catch (e) {
   304→   304→      AppLogger.error('Unexpected getSubscriptionPlan error', error: e);
   305→   305→      throw const ServerException(
   306→   306→        message: 'Failed to fetch subscription plan.',

   383→   383→  @override
   384→   384→  Future<List<SubscriptionModel>> getSubscriptions({
   385→   385→    String? schoolId,
   386→   386→    String? subscriberType,
   387→   387→    String? status,
   388→   388→    int page = 1,
   389→   389→    int perPage = 20,
   390→   390→  }) async {
   391→   391→    try {
   392→   392→      var query = _supabase.from(_subscriptionsTable).select();
   393→   393→
   394→   394→      if (schoolId != null) {
   395→   395→        query = query.eq('school_id', schoolId);
   396→   396→      }
   397→   397→      if (subscriberType != null) {
   398→   398→        query = query.eq('subscriber_type', subscriberType);
   399→   399→      }
   400→   400→      if (status != null) {
   401→   401→        query = query.eq('status', status);
   402→   402→      }
   403→   403→
   404→   404→      final filtered = _applyFilters(query, {
   405→   405→        if (schoolId != null) 'school_id': schoolId,
   406→   406→        if (subscriberType != null) 'subscriber_type': subscriberType,
   407→   407→        if (status != null) 'status': status,
   408→   408→      });
   409→   409→
   410→   410→      var transformed =
   411→   411→          filtered.order('created_at', ascending: false);
   412→   412→      transformed = _applyPagination(transformed, page: page, perPage: perPage);
   413→   413→
   414→   414→      final list = await transformed;
   415→   415→      AppLogger.info('Fetched ${list.length} subscriptions');
   416→   416→      return list
   417→   417→          .map<SubscriptionModel>(
   418→   418→            (row) => SubscriptionModel.fromJson(row),
   419→   419→          )
   420→   420→          .toList();
   421→   421→    } on sb.PostgrestException catch (e) {
   422→   422→      throw _mapPostgrestException(e);
   423→   423→    } catch (e) {
   424→   424→      AppLogger.error('Unexpected getSubscriptions error', error: e);
   425→   425→      throw const ServerException(
   426→   426→        message: 'Failed to fetch subscriptions.',
   427→   427→        statusCode: 500,
   428→   428→      );
   429→   429→    }
