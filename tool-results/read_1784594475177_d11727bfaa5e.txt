    14→/// Comprehensive student dashboard page for the Student Portal.
    15→///
    16→/// Displays:
    17→/// - Welcome section with student name and learning streak
    18→/// - Stat cards row: Upcoming Exams, Pending Assignments, Average Score, Practice Sessions
    19→/// - Quick Actions grid: AI Tutor, Practice Mode, Assignments, Flashcards, Study Planner, Resources
    20→/// - Today's Schedule section
    21→/// - Recent Activity list
    22→/// - Notifications summary
    23→/// - AI Study Suggestions section
    24→/// - Pull-to-refresh with [RefreshIndicator]
    25→/// - Loading skeleton state
    26→class StudentPortalDashboardPage extends ConsumerStatefulWidget {
    27→  const StudentPortalDashboardPage({super.key});
    28→
    29→  @override
    30→  ConsumerState<StudentPortalDashboardPage> createState() =>
    31→      _StudentPortalDashboardPageState();
    32→}
    33→
    34→class _StudentPortalDashboardPageState
    35→    extends ConsumerState<StudentPortalDashboardPage> {
    36→  @override
    37→  void initState() {
    38→    super.initState();
    39→    WidgetsBinding.instance.addPostFrameCallback((_) {
    40→      ref.read(studentDashboardProvider.notifier).loadDashboard();
    41→    });
    42→  }
    43→
    44→  @override
    45→  Widget build(BuildContext context) {
    46→    final dashboardState = ref.watch(studentDashboardProvider);
    47→
    48→    if (dashboardState.isLoading && dashboardState.stats.upcomingExams == 0) {
    49→      return const _DashboardLoadingSkeleton();
    50→    }
    51→
    52→    if (dashboardState.error != null &&
    53→        dashboardState.stats.upcomingExams == 0) {
    54→      return AppErrorState(
    55→        icon: Icons.error_outline_rounded,
    56→        title: 'Failed to Load Dashboard',
    57→        message: dashboardState.error,
    58→        onRetry: () =>
    59→            ref.read(studentDashboardProvider.notifier).refresh(),
    60→      );
    61→    }
    62→
    63→    final stats = dashboardState.stats;
    64→
    65→    return RefreshIndicator(
    66→      onRefresh: () =>
    67→          ref.read(studentDashboardProvider.notifier).refresh(),
    68→      child: SingleChildScrollView(
    69→        physics: const AlwaysScrollableScrollPhysics(),
    70→        padding: Spacings.paddingScreen,
    71→        child: context.isDesktop
    72→            ? _buildDesktopLayout(context, ref, dashboardState, stats)
    73→            : _buildMobileLayout(context, ref, dashboardState, stats),
    74→      ),
    75→    );
    76→  }
    77→

   638→  Widget _buildStatusBadge(BuildContext context, SubmissionStatus status) {
   639→    final cs = context.colorScheme;
   640→    final tt = context.textTheme;
   641→    final (color, label) = switch (status) {
   642→      SubmissionStatus.draft => (AppColors.warning, 'Draft'),
   643→      SubmissionStatus.submitted => (AppColors.info, 'Submitted'),
   644→      SubmissionStatus.graded => (AppColors.success, 'Graded'),
   645→      SubmissionStatus.returned => (AppColors.error, 'Returned'),
   646→      SubmissionStatus.lateSubmitted => (AppColors.warning, 'Late'),
   647→      SubmissionStatus.resubmitted => (AppColors.info, 'Resubmitted'),
   648→    };
   649→

   696→  String _formatDate(DateTime? date) {
   697→    if (date == null) return 'N/A';
   698→    final now = DateTime.now();
   699→    final diff = date.difference(now).inDays;
   700→    if (diff == 0) return 'Today';
   701→    if (diff == 1) return 'Tomorrow';
   702→    if (diff > 0 && diff <= 7) return 'In $diff days';
   703→    return '${date.day}/${date.month}/${date.year}';
   704→  }
   705→}
