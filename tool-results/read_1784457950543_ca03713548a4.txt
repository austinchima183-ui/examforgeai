   396→   396→  @override
   397→   397→  Future<Result<TransactionEntity>> verifyPayment(String txRef) async {
   398→   398→    try {
   399→   399→      final flutterwaveData = await _flutterwaveDataSource.verifyTransaction(txRef);
   400→   400→
   401→   401→      // Find and update our local transaction record
   402→   402→      final localTx = await _remoteDataSource.getTransaction(
   403→   403→        flutterwaveData['tx_ref'] as String? ?? txRef,
   404→   404→      );
   405→   405→
   406→   406→      final status = _mapFlutterwaveStatus(
   407→   407→        flutterwaveData['status'] as String? ?? 'pending',
   408→   408→      );
   409→   409→      final updatedTx = await _remoteDataSource.updateTransaction(
   410→   410→        localTx.id,
   411→   411→        {
   412→   412→          'status': status.value,
   413→   413→          'flutterwave_transaction_id': flutterwaveData['id']?.toString(),
   414→   414→          'flutterwave_flw_ref': flutterwaveData['flw_ref'] as String?,
   415→   415→          'flutterwave_fee': flutterwaveData['app_fee'] as double? ?? 0,
   416→   416→          'net_amount': flutterwaveData['amount_settled'] as double? ?? 0,
   417→   417→          'payment_method_summary': flutterwaveData['payment_type'] as String?,
   418→   418→          'processor_response': flutterwaveData,
   419→   419→          'verified_at': DateTime.now().toIso8601String(),
   420→   420→          if (status == TransactionStatus.successful)
   421→   421→            'completed_at': DateTime.now().toIso8601String(),
   422→   422→        },
   423→   423→      );
   424→   424→      return Success(updatedTx.toEntity());
   425→   425→    } on AuthException catch (e) {
   426→   426→      return FailureResult(Failure.auth(message: e.message, code: e.code));
   427→   427→    } on ServerException catch (e) {
   428→   428→      return FailureResult(Failure.server(
   429→   429→        message: e.message, statusCode: e.statusCode, data: e.data,
   430→   430→      ));
   431→   431→    } on NotFoundException catch (e) {
   432→   432→      return FailureResult(Failure.notFound(message: e.message));
   433→   433→    } catch (e) {
   434→   434→      AppLogger.error('Unexpected verifyPayment error', error: e);
   435→   435→      return const FailureResult(Failure.server(
   436→   436→        message: 'Failed to verify payment.', statusCode: 500,
   437→   437→      ));
   438→   438→    }

   441→   441→  @override
   442→   442→  Future<Result<bool>> processWebhookEvent(Map<String, dynamic> payload) async {
   443→   443→    try {
   444→   444→      final event = payload['event'] as String? ?? '';
   445→   445→      final data = payload['data'] as Map<String, dynamic>? ?? {};
   446→   446→
   447→   447→      AppLogger.info('Processing Flutterwave webhook event: $event');
   448→   448→
   449→   449→      switch (event) {
   450→   450→        case 'charge.completed':
   451→   451→          final txRef = data['tx_ref'] as String?;
   452→   452→          if (txRef != null) {
   453→   453→            // Verify and update the transaction
   454→   454→            await verifyPayment(txRef);
   455→   455→          }
   456→   456→          break;
   457→   457→        case 'transfer.completed':
   458→   458→        case 'transfer.failed':
   459→   459→          AppLogger.info('Transfer event received: $event');
   460→   460→          break;
   461→   461→        case 'subscription.cancelled':
   462→   462→          final subId = data['subscription_id'] as String?;
   463→   463→          if (subId != null) {
   464→   464→            await _remoteDataSource.updateSubscription(subId, {
   465→   465→              'status': 'cancelled',
   466→   466→              'cancelled_at': DateTime.now().toIso8601String(),
   467→   467→            });
   468→   468→          }
   469→   469→          break;
   470→   470→        default:
   471→   471→          AppLogger.info('Unhandled webhook event: $event');
   472→   472→      }
   473→   473→      return const Success(true);
   474→   474→    } on AuthException catch (e) {
   475→   475→      return FailureResult(Failure.auth(message: e.message, code: e.code));
   476→   476→    } on ServerException catch (e) {
   477→   477→      return FailureResult(Failure.server(
   478→   478→        message: e.message, statusCode: e.statusCode, data: e.data,
   479→   479→      ));
