   302→/// Every method maps Supabase-specific responses and errors into the
   303→/// domain-agnostic types defined in the data layer. Supabase
   304→/// [sb.PostgrestException] instances are converted to our custom
   305→/// exceptions with user-friendly messages.
   306→class MarketplaceRemoteDataSourceImpl implements MarketplaceRemoteDataSource {
   307→  MarketplaceRemoteDataSourceImpl({
   308→    required sb.SupabaseClient supabaseClient,

   343→  @override
   344→  Future<List<MarketplaceCategoryModel>> getCategories({
   345→    bool activeOnly = true,
   346→    int limit = PaginatedQueryMixin.dropdownPageSize,
   347→  }) async {
   348→    try {
   349→      var query = _supabase.from(_categoriesTable).select();
   350→
   351→      if (activeOnly) {
   352→        query = query.eq('is_active', true);
   353→      }
   354→
   355→      // PERF: Added limit to prevent unbounded query on marketplace_categories
   356→      final response = await query.order('sort_order', ascending: true).limit(limit);
   357→
   358→      AppLogger.info('Fetched ${response.length} marketplace categories');
   359→      return response
   360→          .map<MarketplaceCategoryModel>(
   361→            (row) => MarketplaceCategoryModel.fromJson(row),
   362→          )
   363→          .toList();
   364→    } on sb.PostgrestException catch (e) {
   365→      throw _mapPostgrestException(e);
   366→    } catch (e) {
   367→      AppLogger.error('Unexpected getCategories error', error: e);
   368→      throw const ServerException(
   369→        message: 'Failed to fetch marketplace categories.',
   370→        statusCode: 500,
   371→      );
   372→    }
   373→  }
   374→
   375→  @override
   376→  Future<MarketplaceCategoryModel> getCategory(String categoryId) async {
   377→    try {
   378→      final response = await _supabase
   379→          .from(_categoriesTable)
   380→          .select()
   381→          .eq('id', categoryId)
   382→          .limit(1);
   383→
   384→      if (response.isEmpty) {
   385→        throw const NotFoundException(message: 'Category not found.');
   386→      }
   387→
   388→      return MarketplaceCategoryModel.fromJson(response.first);
   389→    } on sb.PostgrestException catch (e) {
   390→      throw _mapPostgrestException(e);
   391→    } on NotFoundException {
   392→      rethrow;
   393→    } catch (e) {
   394→      AppLogger.error('Unexpected getCategory error', error: e);
   395→      throw const ServerException(
   396→        message: 'Failed to fetch category.',
   397→        statusCode: 500,
   398→      );
   399→    }
   400→  }
   401→
   402→  @override
   403→  Future<MarketplaceCategoryModel> upsertCategory(
   404→    MarketplaceCategoryModel category,
   405→  ) async {
   406→    try {
   407→      final data = category.toJson();
   408→      data['updated_at'] = DateTime.now().toIso8601String();
   409→
   410→      final response = await _supabase
   411→          .from(_categoriesTable)
   412→          .upsert(data)
   413→          .select();
   414→
   415→      if (response.isEmpty) {
   416→        throw const ServerException(
   417→          message: 'Category upsert returned no data.',
   418→          statusCode: 500,
   419→        );
   420→      }
   421→
   422→      AppLogger.info('Category upserted: ${response.first['id']}');
   423→      return MarketplaceCategoryModel.fromJson(response.first);
   424→    } on sb.PostgrestException catch (e) {
   425→      throw _mapPostgrestException(e);
   426→    } on ServerException {
   427→      rethrow;
   428→    } catch (e) {
   429→      AppLogger.error('Unexpected upsertCategory error', error: e);
   430→      throw const ServerException(
   431→        message: 'Failed to upsert category.',
   432→        statusCode: 500,
   433→      );
   434→    }
   435→  }

   448→      AppLogger.info('Category soft-deleted: $categoryId');

   629→  @override
   630→  Future<List<MarketplaceProductModel>> getProducts({
   631→    String? status,
   632→    String? productType,
   633→    String? categoryId,
   634→    String? sellerId,
   635→    String? subject,
   636→    String? classLevel,
   637→    String? curriculum,
   638→    String? search,
   639→    String? sortBy,
   640→    int limit = 20,
   641→    int offset = 0,
   642→  }) async {
   643→    try {
   644→      // Use RPC for full-text search when a search query is provided
   645→      if (search != null && search.isNotEmpty) {
   646→        final response = await _supabase.rpc(
   647→          'marketplace_search',
   648→          params: {
   649→            'search_query': search,
   650→            'p_status': status,
   651→            'p_product_type': productType,
   652→            'p_category_id': categoryId,
   653→            'p_seller_id': sellerId,
   654→            'p_subject': subject,
   655→            'p_class_level': classLevel,
   656→            'p_curriculum': curriculum,
   657→            'p_sort_by': sortBy ?? 'newest',
   658→            'p_limit': limit,
   659→            'p_offset': offset,
   660→          },
   661→        );
   662→
   663→        final list = response as List<dynamic>;
   664→        AppLogger.info('Search returned ${list.length} products');
   665→        return list
   666→            .map<MarketplaceProductModel>(
   667→              (row) => MarketplaceProductModel.fromJson(row as Map<String, dynamic>),
   668→            )
   669→            .toList();
   670→      }
   671→
   672→      // Standard query builder when no search
   673→      var filterQuery = _supabase.from(_productsTable).select();
   674→
   675→      filterQuery = filterQuery.filter('deleted_at', 'is', null); // exclude soft-deleted
   676→
   677→      if (status != null) {
   678→        filterQuery = filterQuery.eq('status', status);
   679→      }
   680→      if (productType != null) {
   681→        filterQuery = filterQuery.eq('product_type', productType);
   682→      }
   683→      if (categoryId != null) {
   684→        filterQuery = filterQuery.eq('category_id', categoryId);
   685→      }
   686→      if (sellerId != null) {
   687→        filterQuery = filterQuery.eq('seller_id', sellerId);
   688→      }
   689→      if (subject != null) {
   690→        filterQuery = filterQuery.eq('subject', subject);
   691→      }
   692→      if (classLevel != null) {
   693→        filterQuery = filterQuery.eq('class_level', classLevel);
   694→      }
   695→      if (curriculum != null) {
   696→        filterQuery = filterQuery.eq('curriculum', curriculum);
   697→      }
   698→
   699→      // Apply sort
   700→      final transformQuery = _applySortBy(filterQuery, sortBy);
   701→
   702→      final response = await transformQuery.range(offset, offset + limit - 1);
   703→
   704→      AppLogger.info('Fetched ${response.length} products');
   705→      return response
   706→          .map<MarketplaceProductModel>(
   707→            (row) => MarketplaceProductModel.fromJson(row),
   708→          )
   709→          .toList();
   710→    } on sb.PostgrestException catch (e) {
   711→      throw _mapPostgrestException(e);
   712→    } catch (e) {
   713→      AppLogger.error('Unexpected getProducts error', error: e);
   714→      throw const ServerException(
   715→        message: 'Failed to fetch products.',
   716→        statusCode: 500,
   717→      );
   718→    }
   719→  }

  1329→        await _supabase.from(_orderItemsTable).insert(orderItems);

  1332→      AppLogger.info('Order created: $orderId');
  1333→
  1334→      // Fetch the complete order with items
