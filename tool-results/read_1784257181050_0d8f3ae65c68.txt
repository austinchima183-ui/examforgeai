   242→/// Riverpod [StateNotifier] that manages the question editor form state.

   244→/// Handles loading an existing question for editing, managing all form
   245→/// fields and type-specific answer data, and creating/updating questions
   246→/// through the use cases.
   247→class QuestionEditorNotifier extends StateNotifier<QuestionEditorState> {
   248→  QuestionEditorNotifier({
   249→    required CreateQuestionUseCase createQuestionUseCase,
   250→    required UpdateQuestionUseCase updateQuestionUseCase,
   251→    required ManageQuestionStatusUseCase manageQuestionStatusUseCase,
   252→    required QuestionBankRepository repository,
   253→  })  : _createQuestionUseCase = createQuestionUseCase,
   254→        _updateQuestionUseCase = updateQuestionUseCase,
   255→        _manageQuestionStatusUseCase = manageQuestionStatusUseCase,
   256→        _repository = repository,
   257→        super(const QuestionEditorState());
   258→
   259→  final CreateQuestionUseCase _createQuestionUseCase;
   260→  final UpdateQuestionUseCase _updateQuestionUseCase;
   261→  final ManageQuestionStatusUseCase _manageQuestionStatusUseCase;
   262→  final QuestionBankRepository _repository;
   263→
   264→  // ─── Load Question for Edit ─────────────────────────────────────────
   265→
   266→  /// Loads a question by [questionId] and populates all form fields.
   267→  ///
   268→  /// Used when editing an existing question. The form fields are
   269→  /// populated from the loaded entity.
   270→  Future<void> loadQuestionForEdit(String questionId) async {
   271→    state = state.copyWith(isLoading: true, error: null);
   272→
   273→    final result = await _repository.getQuestionWithDetails(questionId);
   274→
   275→    result.fold(
   276→      onSuccess: (question) {
   277→        state = state.copyWith(
   278→          isLoading: false,
   279→          question: question,
   280→          content: question.content,
   281→          explanation: question.explanation ?? '',
   282→          teacherNotes: question.teacherNotes ?? '',
   283→          referenceMaterials: question.referenceMaterials ?? '',
   284→          selectedQuestionType: question.questionType,
   285→          selectedDifficulty: question.difficulty,
   286→          selectedExamType: question.examType ?? ExamType.schoolExam,
   287→          selectedSubjectId: question.subjectId,
   288→          selectedTopicId: question.topicId,
   289→          selectedSubtopicId: question.subtopicId,
   290→          selectedClassId: question.classId,
   291→          selectedCategoryId: question.categoryId,
   292→          marks: question.marks,
   293→          negativeMarks: question.negativeMarks,
   294→          timeAllowedSeconds: question.timeAllowedSeconds,
   295→          answerOptions: question.answerOptions,
   296→          matchingPairs: question.matchingPairs,
   297→          orderingItems: question.orderingItems,
   298→          fillInBlankAnswers: question.fillInBlankAnswers,
   299→          attachments: question.attachments,
   300→          selectedTags: question.tags,
   301→          error: null,
   302→        );
   303→
   304→        // Load topics for the selected subject.
   305→        if (question.subjectId.isNotEmpty) {
   306→          _loadTopicsForSubject(question.subjectId);
   307→        }
   308→        // Load subtopics for the selected topic.
   309→        if (question.topicId != null && question.topicId!.isNotEmpty) {
   310→          _loadSubtopicsForTopic(question.topicId!);
   311→        }
   312→
   313→        AppLogger.info('Loaded question for edit: ${question.id}');
   314→      },
   315→      onFailure: (failure) {
   316→        state = state.copyWith(
   317→          isLoading: false,
   318→          error: _mapFailureToMessage(failure),
   319→        );
   320→        AppLogger.warning('Failed to load question for edit: $failure');
   321→      },
   322→    );
   323→  }

   431→  // ─── Answer Options Management ──────────────────────────────────────
   432→
   433→  /// Adds a new blank answer option.
   434→  void addAnswerOption() {
   435→    final now = DateTime.now();
   436→    final index = state.answerOptions.length;
   437→    final option = AnswerOptionEntity(
   438→      id: 'new-${index + 1}',
   439→      questionId: state.question?.id ?? '',
   440→      content: '',
   441→      isCorrect: false,
   442→      marks: 0.0,
   443→      sortOrder: index,
   444→      createdAt: now,
   445→      updatedAt: now,
   446→    );
   447→    state = state.copyWith(
   448→      answerOptions: [...state.answerOptions, option],
   449→    );
   450→  }
   451→
   452→  /// Removes the answer option at [index].
   453→  void removeAnswerOption(int index) {
   454→    if (index < 0 || index >= state.answerOptions.length) return;
   455→    final updated = [...state.answerOptions];
   456→    updated.removeAt(index);
   457→    // Re-sort remaining options.
   458→    final reindexed = updated
   459→        .asMap()
   460→        .entries
   461→        .map((e) => e.value.copyWith(sortOrder: e.key))
   462→        .toList();
   463→    state = state.copyWith(answerOptions: reindexed);
   464→  }
   465→
   466→  /// Updates the answer option at [index] with the provided [option].
   467→  void updateAnswerOption(int index, AnswerOptionEntity option) {
   468→    if (index < 0 || index >= state.answerOptions.length) return;
   469→    final updated = [...state.answerOptions];
   470→    updated[index] = option;
   471→    state = state.copyWith(answerOptions: updated);
   472→  }

   480→    }).toList();

   573→  /// Reorders ordering items by moving an item from [oldIndex] to
   574→  /// [newIndex].
   575→  void reorderOrderingItems(int oldIndex, int newIndex) {
   576→    final items = [...state.orderingItems];
   577→    if (oldIndex < 0 ||
   578→        oldIndex >= items.length ||
   579→        newIndex < 0 ||
   580→        newIndex >= items.length) {
   581→      return;
   582→    }
   583→    final item = items.removeAt(oldIndex);
   584→    items.insert(newIndex, item);
   585→    // Re-index correct positions.
   586→    final reindexed = items.asMap().entries.map((e) {
   587→      return e.value.copyWith(correctPosition: e.key);
   588→    }).toList();
   589→    state = state.copyWith(orderingItems: reindexed);
   590→  }

   601→      blankIndex: index,
   602→      acceptableAnswers: const [''],
   603→      isCaseSensitive: false,
   604→      marks: 0.0,
   605→      createdAt: now,
   606→    );
   607→    state = state.copyWith(
   608→      fillInBlankAnswers: [...state.fillInBlankAnswers, answer],
   609→    );
   610→  }
   611→
   612→  /// Removes the fill-in-the-blank answer at [index].
   613→  void removeFillInBlank(int index) {
   614→    if (index < 0 || index >= state.fillInBlankAnswers.length) return;
   615→    final updated = [...state.fillInBlankAnswers];
   616→    updated.removeAt(index);
   617→    // Re-index blanks.
   618→    final reindexed = updated.asMap().entries.map((e) {
   619→      return e.value.copyWith(blankIndex: e.key);
   620→    }).toList();
   621→    state = state.copyWith(fillInBlankAnswers: reindexed);
   622→  }
   623→
   624→  /// Updates the fill-in-the-blank answer at [index] with the provided
   625→  /// [answer].
   626→  void updateFillInBlank(int index, FillInBlankAnswerEntity answer) {
   627→    if (index < 0 || index >= state.fillInBlankAnswers.length) return;
   628→    final updated = [...state.fillInBlankAnswers];
   629→    updated[index] = answer;
   630→    state = state.copyWith(fillInBlankAnswers: updated);
   631→  }

   642→  /// Removes the attachment at [index].

   646→    updated.removeAt(index);

   666→  // ─── Save Question ──────────────────────────────────────────────────
   667→
   668→  /// Saves the question by creating a new one or updating an existing one,
   669→  /// depending on whether a question ID exists.
   670→  ///
   671→  /// Validates all required fields before saving.
   672→  Future<void> saveQuestion() async {
   673→    if (!validateForm()) return;
   674→
   675→    state = state.copyWith(isSaving: true, error: null);
   676→
   677→    final now = DateTime.now();
   678→    final questionEntity = QuestionEntity(
   679→      id: state.question?.id ?? '',
   680→      schoolId: state.question?.schoolId,
   681→      subjectId: state.selectedSubjectId ?? '',
   682→      topicId: state.selectedTopicId,
   683→      subtopicId: state.selectedSubtopicId,
   684→      classId: state.selectedClassId,
   685→      categoryId: state.selectedCategoryId,
   686→      questionType: state.selectedQuestionType,
   687→      difficulty: state.selectedDifficulty,
   688→      examType: state.selectedExamType,
   689→      content: state.content,
   690→      explanation: state.explanation.isNotEmpty ? state.explanation : null,
   691→      teacherNotes:
   692→          state.teacherNotes.isNotEmpty ? state.teacherNotes : null,
   693→      referenceMaterials: state.referenceMaterials.isNotEmpty
   694→          ? state.referenceMaterials
   695→          : null,
   696→      marks: state.marks,
   697→      negativeMarks: state.negativeMarks,
   698→      timeAllowedSeconds: state.timeAllowedSeconds,
   699→      isPublished: state.question?.isPublished ?? false,
   700→      isArchived: state.question?.isArchived ?? false,
   701→      isFeatured: state.question?.isFeatured ?? false,
   702→      version: state.question?.version ?? 1,
   703→      createdBy: state.question?.createdBy,
   704→      updatedBy: state.question?.updatedBy,
   705→      createdAt: state.question?.createdAt ?? now,
   706→      updatedAt: now,
   707→      answerOptions: state.answerOptions,
   708→      matchingPairs: state.matchingPairs,
   709→      orderingItems: state.orderingItems,
   710→      fillInBlankAnswers: state.fillInBlankAnswers,
   711→      attachments: state.attachments,
   712→      tags: state.selectedTags,
   713→    );
   714→
   715→    if (state.isEditMode) {
   716→      final result = await _updateQuestionUseCase(
   717→        UpdateQuestionParams(
   718→          question: questionEntity,
   719→          answerOptions: state.answerOptions,
   720→          matchingPairs: state.matchingPairs,
   721→          orderingItems: state.orderingItems,
   722→          fillInBlankAnswers: state.fillInBlankAnswers,
   723→          attachments: state.attachments,
   724→          tagIds: state.selectedTags.map((t) => t.id).toList(),
   725→        ),
   726→      );
   727→
   728→      result.fold(
   729→        onSuccess: (updatedQuestion) {
   730→          state = state.copyWith(
   731→            isSaving: false,
   732→            question: updatedQuestion,
   733→            error: null,
   734→          );
   735→          AppLogger.info('Question updated: ${updatedQuestion.id}');
   736→        },
   737→        onFailure: (failure) {
   738→          state = state.copyWith(
   739→            isSaving: false,
   740→            error: _mapFailureToMessage(failure),
   741→          );
   742→          AppLogger.warning('Failed to update question: $failure');
   743→        },
   744→      );
   745→    } else {
   746→      final result = await _createQuestionUseCase(
   747→        CreateQuestionParams(
   748→          question: questionEntity,
   749→          answerOptions: state.answerOptions,
   750→          matchingPairs: state.matchingPairs,
   751→          orderingItems: state.orderingItems,
   752→          fillInBlankAnswers: state.fillInBlankAnswers,
   753→          attachments: state.attachments,
   754→          tagIds: state.selectedTags.map((t) => t.id).toList(),
   755→        ),
   756→      );
   757→
   758→      result.fold(
   759→        onSuccess: (createdQuestion) {
   760→          state = state.copyWith(
   761→            isSaving: false,
   762→            question: createdQuestion,
   763→            error: null,
   764→          );
   765→          AppLogger.info('Question created: ${createdQuestion.id}');
   766→        },
   767→        onFailure: (failure) {
   768→          state = state.copyWith(
   769→            isSaving: false,
   770→            error: _mapFailureToMessage(failure),
   771→          );
   772→          AppLogger.warning('Failed to create question: $failure');
   773→        },
   774→      );
   775→    }
   776→  }

   778→  // ─── Save and Publish ───────────────────────────────────────────────
   779→
   780→  /// Saves the question and then publishes it.
   781→  Future<void> saveAndPublish() async {
   782→    await saveQuestion();
   783→
   784→    // Only publish if the save succeeded (no error and question has an ID).
   785→    if (state.error == null && state.question != null) {
   786→      final questionId = state.question!.id;
   787→      if (questionId.isNotEmpty) {
   788→        final result = await _manageQuestionStatusUseCase(
   789→          ManageQuestionStatusParams(
   790→            questionId: questionId,
   791→            action: QuestionStatusAction.publish,
   792→          ),
   793→        );
   794→
   795→        result.fold(
   796→          onSuccess: (_) {
   797→            state = state.copyWith(
   798→              question: state.question?.copyWith(isPublished: true),
   799→            );
   800→            AppLogger.info('Question published: $questionId');
   801→          },
   802→          onFailure: (failure) {
   803→            state = state.copyWith(
   804→              error: _mapFailureToMessage(failure),
   805→            );
   806→            AppLogger.warning('Failed to publish question: $failure');
   807→          },
   808→        );
   809→      }
   810→    }
   811→  }

   827→  // ─── Validate Form ──────────────────────────────────────────────────
   828→
   829→  /// Validates all required form fields and type-specific data.
   830→  ///
   831→  /// Returns `true` if the form is valid, `false` otherwise.
   832→  /// Sets the [error] field with a validation message when invalid.
   833→  bool validateForm() {
   834→    // ── Content ────────────────────────────────────────────────────────
   835→    if (state.content.trim().isEmpty) {
   836→      state = state.copyWith(error: 'Question content is required');
   837→      return false;
   838→    }
   839→
   840→    // ── Subject ────────────────────────────────────────────────────────
   841→    if (state.selectedSubjectId == null ||
   842→        state.selectedSubjectId!.isEmpty) {
   843→      state = state.copyWith(error: 'Please select a subject');
   844→      return false;
   845→    }
   846→
   847→    // ── Marks ──────────────────────────────────────────────────────────
   848→    if (state.marks <= 0) {
   849→      state = state.copyWith(error: 'Marks must be greater than zero');
   850→      return false;
   851→    }
   852→
   853→    // ── Type-specific validation ───────────────────────────────────────
   854→    final type = state.selectedQuestionType;
   855→
   856→    // Answer options for choice-based types.
   857→    if (type.hasOptions && state.answerOptions.isEmpty) {
   858→      state = state.copyWith(
   859→        error: '${type.label} questions require at least one answer option',
   860→      );
   861→      return false;
   862→    }
   863→
   864→    if (type.hasOptions && type.hasCorrectAnswer) {
   865→      final hasCorrect = state.answerOptions.any((o) => o.isCorrect);
   866→      if (!hasCorrect) {
   867→        state = state.copyWith(
   868→          error: 'At least one answer option must be marked as correct',
   869→        );
   870→        return false;
   871→      }
   872→    }
   873→
   874→    // Non-empty content for answer options.
   875→    if (type.hasOptions) {
   876→      final emptyOption = state.answerOptions.any(
   877→        (o) => o.content.trim().isEmpty,
   878→      );
   879→      if (emptyOption) {
   880→        state = state.copyWith(
   881→          error: 'All answer options must have content',
   882→        );
   883→        return false;
   884→      }
   885→    }
   886→
   887→    // Matching pairs.
   888→    if (type == QuestionType.matching && state.matchingPairs.isEmpty) {
   889→      state = state.copyWith(
   890→        error: 'Matching questions require at least one matching pair',
   891→      );
   892→      return false;
   893→    }
   894→
   895→    if (type == QuestionType.matching) {
   896→      final incompletePair = state.matchingPairs.any(
   897→        (p) => p.leftContent.trim().isEmpty || p.rightContent.trim().isEmpty,
   898→      );
   899→      if (incompletePair) {
   900→        state = state.copyWith(
   901→          error: 'All matching pairs must have both left and right content',
   902→        );
   903→        return false;
   904→      }
   905→    }
   906→
   907→    // Ordering items.
   908→    if (type == QuestionType.ordering && state.orderingItems.isEmpty) {
   909→      state = state.copyWith(
   910→        error: 'Ordering questions require at least one ordering item',
   911→      );
   912→      return false;
   913→    }
   914→
   915→    if (type == QuestionType.ordering) {
   916→      final emptyItem = state.orderingItems.any(
   917→        (i) => i.content.trim().isEmpty,
   918→      );
   919→      if (emptyItem) {
   920→        state = state.copyWith(
   921→          error: 'All ordering items must have content',
   922→        );
   923→        return false;
   924→      }
   925→    }
   926→
   927→    // Fill-in-blank answers.
   928→    if (type == QuestionType.fillInBlank &&
   929→        state.fillInBlankAnswers.isEmpty) {
   930→      state = state.copyWith(
   931→        error: 'Fill-in-the-blank questions require at least one blank',
   932→      );
   933→      return false;
   934→    }
   935→
   936→    if (type == QuestionType.fillInBlank) {
   937→      final emptyBlank = state.fillInBlankAnswers.any(
   938→        (a) => a.acceptableAnswers.isEmpty ||
   939→            a.acceptableAnswers.every((s) => s.trim().isEmpty),
   940→      );
   941→      if (emptyBlank) {
   942→        state = state.copyWith(
   943→          error: 'Each blank must have at least one acceptable answer',
   944→        );
   945→        return false;
   946→      }
   947→    }
   948→
   949→    // Clear any previous error since form is valid.
   950→    state = state.copyWith(error: null);
   951→    return true;
   952→  }
