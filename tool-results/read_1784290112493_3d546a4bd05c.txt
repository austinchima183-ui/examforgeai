   122→   122→  final dio = Dio(
   123→   123→    BaseOptions(
   124→   124→      baseUrl: AppConfig.baseUrl,
   125→   125→      connectTimeout: AppConfig.connectTimeoutDuration,
   126→   126→      receiveTimeout: AppConfig.receiveTimeoutDuration,
   127→   127→      sendTimeout: AppConfig.sendTimeoutDuration,
   128→   128→      headers: ApiConstants.baseHeaders(),
   129→   129→      responseType: ResponseType.json,
   130→   130→      validateStatus: (status) => status != null && status < 500,
   131→   131→    ),
   132→   132→  );
   133→   133→  return dio;
   134→   134→});
