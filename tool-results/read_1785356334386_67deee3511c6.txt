     7→/// Helper to read a field from JSON that may be in snake_case (Supabase)
     8→/// or camelCase (internal).
     9→T? _readField<T>(Map<String, dynamic> json, String snakeKey, String camelKey) {
    10→  return json[snakeKey] as T? ?? json[camelKey] as T?;
    11→}
    12→
    13→/// Helper to read a DateTime field from JSON (snake_case or camelCase).
    14→DateTime _readDateTime(Map<String, dynamic> json, String snakeKey, String camelKey) {
    15→  final raw = json[snakeKey] as String? ?? json[camelKey] as String;
    16→  return DateTime.parse(raw);
    17→}
    18→
    19→/// Helper to read a nullable DateTime field from JSON.
    20→DateTime? _readNullableDateTime(Map<String, dynamic> json, String snakeKey, String camelKey) {
    21→  final raw = json[snakeKey] as String? ?? json[camelKey] as String?;
    22→  if (raw == null) return null;
    23→  return DateTime.parse(raw);
    24→}
    25→
    26→/// Helper to read a List<String> from JSON.
    27→List<String> _readListOfStrings(dynamic value) {
    28→  if (value == null) return [];
    29→  return (value as List).cast<String>();
    30→}
    31→
    32→/// Helper to read a Map<String, dynamic> from JSON.
    33→Map<String, dynamic> _readMap(dynamic value) {
    34→  if (value == null) return {};
    35→  return Map<String, dynamic>.from(value as Map);
    36→}
    37→
    38→/// Helper to read a List<Map<String, dynamic>> from JSON.
    39→List<Map<String, dynamic>> _readListOfMaps(dynamic value) {
    40→  if (value == null) return [];
    41→  return (value as List).map((e) => Map<String, dynamic>.from(e as Map)).toList();
    42→}
