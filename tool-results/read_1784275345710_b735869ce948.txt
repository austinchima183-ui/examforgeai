   492→   492→/// Represents a variable placeholder within a prompt template.
   493→   493→class PromptVariable extends Equatable {
   494→   494→  const PromptVariable({
   495→   495→    required this.name,
   496→   496→    this.description,
   497→   497→    this.isRequired = true,
   498→   498→    this.defaultValue,
   499→   499→  });
   500→   500→
   501→   501→  /// The variable name used in the template (e.g., `{{subject}}`).
   502→   502→  final String name;
   503→   503→
   504→   504→  /// A description of what this variable represents.
   505→   505→  final String? description;
   506→   506→
   507→   507→  /// Whether this variable must be provided when using the template.
   508→   508→  final bool isRequired;
   509→   509→
   510→   510→  /// Default value used when the variable is not provided.
   511→   511→  final String? defaultValue;
   512→   512→
   513→   513→  PromptVariable copyWith({
   514→   514→    String? name,
   515→   515→    String? description,
   516→   516→    bool? isRequired,
   517→   517→    String? defaultValue,
   518→   518→  }) {
   519→   519→    return PromptVariable(
   520→   520→      name: name ?? this.name,
   521→   521→      description: description ?? this.description,
   522→   522→      isRequired: isRequired ?? this.isRequired,
   523→   523→      defaultValue: defaultValue ?? this.defaultValue,
   524→   524→    );
   525→   525→  }
   526→   526→
   527→   527→  @override
   528→   528→  List<Object?> get props => [name, description, isRequired, defaultValue];
   529→   529→}
