   325→   325→  @override
   326→   326→  Future<List<PlatformSettingModel>> getPlatformSettings({
   327→   327→    String? scope,
   328→   328→  }) async {
   329→   329→    try {
   330→   330→      var query = _supabase.from(_platformSettingsTable).select();
   331→   331→
   332→   332→      if (scope != null) {
   333→   333→        query = query.eq('scope', scope);
   334→   334→      }
   335→   335→
   336→   336→      final list = await query.order('key', ascending: true);
   337→   337→      AppLogger.info('Fetched ${list.length} platform settings');
   338→   338→      return list
   339→   339→          .map<PlatformSettingModel>(
   340→   340→            (row) => PlatformSettingModel.fromJson(row),
   341→   341→          )
   342→   342→          .toList();
   343→   343→    } on sb.PostgrestException catch (e) {
   344→   344→      throw _mapPostgrestException(e);
   345→   345→    } catch (e) {
   346→   346→      AppLogger.error('Unexpected getPlatformSettings error', error: e);
   347→   347→      throw const ServerException(
   348→   348→        message: 'Failed to fetch platform settings.',
   349→   349→        statusCode: 500,
   350→   350→      );
   351→   351→    }
   352→   352→  }

   354→   354→  @override
   355→   355→  Future<PlatformSettingModel> getPlatformSetting(
   356→   356→    String key, {
   357→   357→    String? scope,
   358→   358→  }) async {
   359→   359→    try {
   360→   360→      var query = _supabase
   361→   361→          .from(_platformSettingsTable)
   362→   362→          .select()
   363→   363→          .eq('key', key);
   364→   364→
   365→   365→      if (scope != null) {
   366→   366→        query = query.eq('scope', scope);
   367→   367→      }
   368→   368→
   369→   369→      final response = await query.limit(1);
   370→   370→
   371→   371→      if (response.isEmpty) {
   372→   372→        throw const NotFoundException('Platform setting not found.');
   373→   373→      }
   374→   374→
   375→   375→      return PlatformSettingModel.fromJson(response.first);
   376→   376→    } on sb.PostgrestException catch (e) {
   377→   377→      throw _mapPostgrestException(e);
   378→   378→    } on NotFoundException {
   379→   379→      rethrow;
   380→   380→    } catch (e) {
   381→   381→      AppLogger.error('Unexpected getPlatformSetting error', error: e);
   382→   382→      throw const ServerException(
   383→   383→        message: 'Failed to fetch platform setting.',
   384→   384→        statusCode: 500,
   385→   385→      );
   386→   386→    }
   387→   387→  }

   389→   389→  @override
   390→   390→  Future<PlatformSettingModel> updatePlatformSetting(
   391→   391→    Map<String, dynamic> settingData,
   392→   392→  ) async {
   393→   393→    try {
   394→   394→      settingData['updated_at'] = DateTime.now().toIso8601String();
   395→   395→
   396→   396→      final response = await _supabase
   397→   397→          .from(_platformSettingsTable)
   398→   398→          .update(settingData)
   399→   399→          .eq('id', settingData['id'] as String)
   400→   400→          .select();
   401→   401→
   402→   402→      if (response.isEmpty) {
   403→   403→        throw const NotFoundException('Platform setting not found for update.');
   404→   404→      }
   405→   405→
   406→   406→      AppLogger.info('Platform setting updated: ${response.first['key']}');
   407→   407→      return PlatformSettingModel.fromJson(response.first);
   408→   408→    } on sb.PostgrestException catch (e) {
   409→   409→      throw _mapPostgrestException(e);
   410→   410→    } on NotFoundException {
   411→   411→      rethrow;
   412→   412→    } catch (e) {
   413→   413→      AppLogger.error('Unexpected updatePlatformSetting error', error: e);
   414→   414→      throw const ServerException(
   415→   415→        message: 'Failed to update platform setting.',
   416→   416→        statusCode: 500,
   417→   417→      );
   418→   418→    }
   419→   419→  }

   421→   421→  @override
   422→   422→  Future<List<PlatformSettingModel>> bulkUpdateSettings(
   423→   423→    List<Map<String, dynamic>> settingsData,
   424→   424→  ) async {
   425→   425→    try {
   426→   426→      final results = <PlatformSettingModel>[];
   427→   427→
   428→   428→      for (final data in settingsData) {
   429→   429→        data['updated_at'] = DateTime.now().toIso8601String();
   430→   430→
   431→   431→        final response = await _supabase
   432→   432→            .from(_platformSettingsTable)
   433→   433→            .update(data)
   434→   434→            .eq('id', data['id'] as String)
   435→   435→            .select();
   436→   436→
   437→   437→        if (response.isNotEmpty) {
   438→   438→          results.add(PlatformSettingModel.fromJson(response.first));
   439→   439→        }
   440→   440→      }
   441→   441→
   442→   442→      AppLogger.info('Bulk updated ${results.length} platform settings');
   443→   443→      return results;
   444→   444→    } on sb.PostgrestException catch (e) {
   445→   445→      throw _mapPostgrestException(e);
   446→   446→    } catch (e) {
   447→   447→      AppLogger.error('Unexpected bulkUpdateSettings error', error: e);
   448→   448→      throw const ServerException(
   449→   449→        message: 'Failed to bulk update platform settings.',
   450→   450→        statusCode: 500,
   451→   451→      );
   452→   452→    }
   453→   453→  }
