    13→/// Abstract interface for all remote CBT engine data operations.
    14→///
    15→/// Implementations handle all network communication with the Supabase
    16→/// backend and return plain model instances. Exceptions are allowed to
    17→/// propagate so the repository layer can catch and convert them to
    18→/// domain [Failure] types.
    19→abstract class CbtRemoteDataSource {
    20→  // ─── Exam CRUD ──────────────────────────────────────────────────────
    21→
    22→  Future<ExamModel> createExam(Map<String, dynamic> examData);

    27→  Future<List<ExamModel>> getExams(Map<String, dynamic> filters);
    28→
    29→  // ─── Exam Lifecycle ─────────────────────────────────────────────────

   171→      );

   305→  @override
   306→  Future<List<ExamModel>> getExams(Map<String, dynamic> filters) async {
   307→    try {
   308→      var query = _supabaseClient.from(_examsTable).select();
   309→
   310→      if (filters['school_id'] != null) {
   311→        query = query.eq('school_id', filters['school_id'] as String);
   312→      }
   313→      if (filters['subject_id'] != null) {
   314→        query = query.eq('subject_id', filters['subject_id'] as String);
   315→      }
   316→      if (filters['status'] != null) {
   317→        query = query.eq('status', filters['status'] as String);
   318→      }
   319→      if (filters['class_id'] != null) {
   320→        query = query.eq('class_id', filters['class_id'] as String);
   321→      }
   322→
   323→      final page = filters['page'] as int? ?? 1;
   324→      final perPage = filters['per_page'] as int? ?? filters['perPage'] as int? ?? 20;
   325→      final offset = (page - 1) * perPage;
   326→
   327→      query = query.range(offset, offset + perPage - 1);
   328→      query = query.order('created_at', ascending: false);
   329→
   330→      final response = await query;
   331→
   332→      return response
   333→          .map<ExamModel>((json) => ExamModel.fromJson(json))
   334→          .toList();
   335→    } on sb.PostgrestException catch (e) {
   336→      AppLogger.error('Get exams failed', error: e);
   337→      throw ServerException(
   338→        message: e.message,
   339→        statusCode: int.tryParse(e.code ?? '') ?? 500,
   340→        data: e.details,
   341→      );
   342→    } catch (e) {
   343→      AppLogger.error('Get exams unexpected error', error: e);
   344→      throw ServerException(
   345→        message: 'Failed to get exams: $e',
   346→        statusCode: 500,
   347→      );
   348→    }
   349→  }

   412→  @override
   413→  Future<ExamModel> cloneExam(String examId) async {
   414→    try {
   415→      // Fetch the original exam with details
   416→      final original = await getExamWithDetails(examId);
   417→
   418→      final now = DateTime.now();
   419→      final cloneData = {
   420→        'school_id': original.schoolId,
   421→        'created_by': original.createdBy,
   422→        'title': '${original.title} (Copy)',
   423→        'description': original.description,
   424→        'subject_id': original.subjectId,
   425→        'class_id': original.classId,
   426→        'academic_session_id': original.academicSessionId,
   427→        'exam_type': original.examType,
   428→        'status': 'draft',
   429→        'start_time': now.add(const Duration(days: 7)).toIso8601String(),
   430→        'end_time': now.add(const Duration(days: 7, hours: 2)).toIso8601String(),
   431→        'time_limit_minutes': original.timeLimitMinutes,
   432→        'total_marks': original.totalMarks,
   433→        'pass_mark': original.passMark,
   434→        'pass_mark_type': original.passMarkType,
   435→        'instructions': original.instructions,
   436→        'allowed_attempts': original.allowedAttempts,
   437→        'negative_marking_enabled': original.negativeMarkingEnabled,
   438→        'negative_mark_value': original.negativeMarkValue,
   439→        'grace_period_minutes': original.gracePeriodMinutes,
   440→        'auto_submit': original.autoSubmit,
   441→        'randomize_questions': original.randomizeQuestions,
   442→        'randomize_options': original.randomizeOptions,
   443→        'show_results': original.showResults,
   444→        'show_correct_answers': original.showCorrectAnswers,
   445→        'show_explanations': original.showExplanations,
   446→        'is_template': original.isTemplate,
   447→        'template_id': original.id,
   448→        'max_students': original.maxStudents,
   449→        'ip_restriction': original.ipRestriction,
   450→        'require_full_screen': original.requireFullScreen,
   451→        'allow_resume': original.allowResume,
   452→        'browser_lockdown': original.browserLockdown,
   453→        'metadata': original.metadata,
   454→      };
   455→
   456→      final clonedExam = await createExam(cloneData);
   457→
   458→      // Clone sections
   459→      for (final section in original.sections) {
   460→        await _supabaseClient.from(_examSectionsTable).insert({
   461→          'exam_id': clonedExam.id,
   462→          'title': section.title,
   463→          'description': section.description,
   464→          'instructions': section.instructions,
   465→          'sort_order': section.sortOrder,
   466→          'time_limit_minutes': section.timeLimitMinutes,
   467→          'randomize_questions': section.randomizeQuestions,
   468→        });
   469→      }
   470→
   471→      // Clone questions
   472→      for (final question in original.questions) {
   473→        await _supabaseClient.from(_examQuestionsTable).insert({
   474→          'exam_id': clonedExam.id,
   475→          'section_id': question.sectionId,
   476→          'question_id': question.questionId,
   477→          'sort_order': question.sortOrder,
   478→          'marks': question.marks,
   479→          'negative_marks': question.negativeMarks,
   480→          'is_compulsory': question.isCompulsory,
   481→        });
   482→      }
   483→
   484→      return clonedExam;
   485→    } on ServerException {
   486→      rethrow;
   487→    } on sb.PostgrestException catch (e) {
   488→      AppLogger.error('Clone exam failed', error: e);
   489→      throw ServerException(
   490→        message: e.message,
   491→        statusCode: int.tryParse(e.code ?? '') ?? 500,
   492→        data: e.details,
   493→      );
   494→    } catch (e) {
   495→      AppLogger.error('Clone exam unexpected error', error: e);
   496→      throw ServerException(
   497→        message: 'Failed to clone exam: $e',
   498→        statusCode: 500,
   499→      );
   500→    }
   501→  }

   562→  @override

   565→    List<String> questionIds,
   566→  ) async {
   567→    try {
   568→      // Update sort_order for each question in the new order
   569→      for (var i = 0; i < questionIds.length; i++) {
   570→        await _supabaseClient
   571→            .from(_examQuestionsTable)
   572→            .update({'sort_order': i})
   573→            .eq('exam_id', examId)
   574→            .eq('question_id', questionIds[i]);
   575→      }
   576→    } on sb.PostgrestException catch (e) {
   577→      AppLogger.error('Reorder questions failed', error: e);
   578→      throw ServerException(
   579→        message: e.message,
   580→        statusCode: int.tryParse(e.code ?? '') ?? 500,

   724→  @override
   725→  Future<ExamResultModel> submitAttempt(
   726→    String attemptId, {
   727→    String submissionType = 'manual',
   728→  }) async {
   729→    try {
   730→      final now = DateTime.now();
   731→
   732→      // Fetch the attempt
   733→      final attemptResponse = await _supabaseClient
   734→          .from(_examAttemptsTable)
   735→          .select()
   736→          .eq('id', attemptId)
   737→          .single();
   738→
   739→      final attempt = ExamAttemptModel.fromJson(attemptResponse);
   740→
   741→      // Fetch all answers for this attempt
   742→      final answersResponse = await _supabaseClient
   743→          .from(_studentAnswersTable)
   744→          .select()
   745→          .eq('attempt_id', attemptId);
   746→
   747→      final answers = answersResponse
   748→          .map<StudentAnswerModel>((a) => StudentAnswerModel.fromJson(a))
   749→          .toList();
   750→
   751→      // Calculate total marks
   752→      double totalMarks = 0;
   753→      for (final answer in answers) {
   754→        totalMarks += answer.marksAwarded - answer.marksDeducted;
   755→      }
   756→
   757→      // Calculate time spent
   758→      final startedAt = attempt.startedAt;
   759→      final timeSpent = now.difference(startedAt).inSeconds;
   760→
   761→      // Determine if all objective questions are graded
   762→      final hasUngradedSubjective = answers.any(
   763→        (a) => a.isCorrect == null && a.marksAwarded == 0,
   764→      );
   765→      final gradingStatus =
   766→          hasUngradedSubjective ? 'partially_graded' : 'auto_graded';
   767→
   768→      // Update the attempt
   769→      await _supabaseClient
   770→          .from(_examAttemptsTable)
   771→          .update({
   772→            'status': submissionType == 'auto_submit' ? 'auto_submitted' : 'submitted',
   773→            'submitted_at': now.toIso8601String(),
   774→            'submission_type': submissionType,
   775→            'time_spent_seconds': timeSpent,
   776→            'total_marks': totalMarks,
   777→            'grading_status': gradingStatus,
   778→            'last_activity_at': now.toIso8601String(),
   779→          })
   780→          .eq('id', attemptId);
   781→
   782→      // Deactivate the session
   783→      await _supabaseClient
   784→          .from(_examSessionsTable)
   785→          .update({
   786→            'is_active': false,
   787→            'connection_status': 'disconnected',
   788→          })
   789→          .eq('attempt_id', attemptId);
   790→
   791→      // Calculate score percentage
   792→      // Fetch exam total marks
   793→      final examResponse = await _supabaseClient
   794→          .from(_examsTable)
   795→          .select('total_marks, pass_mark, pass_mark_type')
   796→          .eq('id', attempt.examId)
   797→          .single();
   798→
   799→      final examTotalMarks =
   800→          (examResponse['total_marks'] as num?)?.toDouble() ?? 0.0;
   801→      final passMark =
   802→          (examResponse['pass_mark'] as num?)?.toDouble() ?? 0.0;
   803→      final passMarkType =
   804→          examResponse['pass_mark_type'] as String? ?? 'percentage';
   805→
   806→      final scorePercentage = examTotalMarks > 0
   807→          ? (totalMarks / examTotalMarks) * 100
   808→          : 0.0;
   809→
   810→      final effectivePassMark = passMarkType == 'percentage'
   811→          ? passMark
   812→          : (examTotalMarks > 0 ? (passMark / examTotalMarks) * 100 : 0.0);
   813→
   814→      final isPassed = scorePercentage >= effectivePassMark;
   815→
   816→      // Create or update result
   817→      final existingResult = await _supabaseClient
   818→          .from(_examResultsTable)
   819→          .select('id')
   820→          .eq('attempt_id', attemptId)
   821→          .maybeSingle();
   822→
   823→      final resultData = {
   824→        'exam_id': attempt.examId,
   825→        'student_id': attempt.studentId,
   826→        'attempt_id': attemptId,
   827→        'total_marks': totalMarks,
   828→        'total_possible': examTotalMarks,
   829→        'score_percentage': scorePercentage,
   830→        'is_passed': isPassed,
   831→        'time_spent_seconds': timeSpent,
   832→        'grading_status': gradingStatus,
   833→        'is_released': false,
   834→      };
   835→
   836→      if (existingResult != null) {
   837→        await _supabaseClient
   838→            .from(_examResultsTable)
   839→            .update(resultData)
   840→            .eq('id', existingResult['id'] as String);
   841→
   842→        final updatedResult = await _supabaseClient
   843→            .from(_examResultsTable)
   844→            .select()
   845→            .eq('id', existingResult['id'] as String)
   846→            .single();
   847→
   848→        return ExamResultModel.fromJson(updatedResult);
   849→      } else {
   850→        final insertResponse = await _supabaseClient
   851→            .from(_examResultsTable)
   852→            .insert(resultData)
   853→            .select()
   854→            .single();
   855→
   856→        return ExamResultModel.fromJson(insertResponse);
   857→      }
   858→    } on sb.PostgrestException catch (e) {
   859→      AppLogger.error('Submit attempt failed', error: e);
   860→      throw ServerException(
   861→        message: e.message,
   862→        statusCode: int.tryParse(e.code ?? '') ?? 500,
   863→        data: e.details,
   864→      );
   865→    } catch (e) {
   866→      if (e is ServerException) rethrow;
   867→      AppLogger.error('Submit attempt unexpected error', error: e);
   868→      throw ServerException(
   869→        message: 'Failed to submit attempt: $e',
   870→        statusCode: 500,
   871→      );
   872→    }
   873→  }

  1353→  @override
  1354→  Future<ExamStatisticsModel> getExamStatistics(String examId) async {
  1355→    try {
  1356→      // Fetch all results for this exam
  1357→      final results = await _supabaseClient
  1358→          .from(_examResultsTable)
  1359→          .select()
  1360→          .eq('exam_id', examId);
  1361→
  1362→      // Fetch exam students count
  1363→      final students = await _supabaseClient
  1364→          .from(_examStudentsTable)
  1365→          .select('id')
  1366→          .eq('exam_id', examId);
  1367→
  1368→      // Fetch attempts
  1369→      final attempts = await _supabaseClient
  1370→          .from(_examAttemptsTable)
  1371→          .select()
  1372→          .eq('exam_id', examId);
  1373→
  1374→      final totalStudents = students.length;
  1375→      final startedStudents = attempts
  1376→          .where((a) => a['status'] != 'not_started')
  1377→          .length;
  1378→      final completedStudents = attempts
  1379→          .where((a) =>
  1380→              a['status'] == 'submitted' ||
  1381→              a['status'] == 'auto_submitted' ||
  1382→              a['status'] == 'timed_out')
  1383→          .length;
  1384→      final submittedStudents = completedStudents;
  1385→
  1386→      // Calculate score statistics
  1387→      final scores = results
  1388→          .map<double>((r) => (r['score_percentage'] as num?)?.toDouble() ?? 0.0)
  1389→          .toList();
  1390→
  1391→      final averageScore = scores.isNotEmpty
  1392→          ? scores.reduce((a, b) => a + b) / scores.length
  1393→          : 0.0;
  1394→      final highestScore = scores.isNotEmpty
  1395→          ? scores.reduce((a, b) => a > b ? a : b)
  1396→          : 0.0;
  1397→      final lowestScore = scores.isNotEmpty
  1398→          ? scores.reduce((a, b) => a < b ? a : b)
  1399→          : 0.0;
  1400→
  1401→      // Calculate median
  1402→      double medianScore = 0.0;
  1403→      if (scores.isNotEmpty) {
  1404→        final sorted = List<double>.from(scores)..sort();
  1405→        final mid = sorted.length ~/ 2;
  1406→        medianScore = sorted.length.isEven
  1407→            ? (sorted[mid - 1] + sorted[mid]) / 2
  1408→            : sorted[mid];
  1409→      }
  1410→
  1411→      // Calculate pass rate
  1412→      final passedCount = results
  1413→          .where((r) => r['is_passed'] == true)
  1414→          .length;
  1415→      final passRate =
  1416→          results.isNotEmpty ? (passedCount / results.length) * 100 : 0.0;
  1417→
  1418→      // Average time spent
  1419→      final timeSpentList = attempts
  1420→          .where((a) => a['time_spent_seconds'] != null)
  1421→          .map<int>((a) => a['time_spent_seconds'] as int)
  1422→          .toList();
  1423→      final averageTimeSpent = timeSpentList.isNotEmpty
  1424→          ? timeSpentList.reduce((a, b) => a + b) ~/ timeSpentList.length
  1425→          : 0;
  1426→
  1427→      // Grading completion
  1428→      final gradedCount = results
  1429→          .where((r) =>
  1430→              r['grading_status'] == 'auto_graded' ||
  1431→              r['grading_status'] == 'fully_graded')
  1432→          .length;
  1433→      final gradingCompletion = results.isNotEmpty
  1434→          ? (gradedCount / results.length) * 100
  1435→          : 0.0;
  1436→
  1437→      return ExamStatisticsModel(
  1438→        examId: examId,
  1439→        totalStudents: totalStudents,
  1440→        startedStudents: startedStudents,
  1441→        completedStudents: completedStudents,
  1442→        submittedStudents: submittedStudents,
  1443→        averageScore: averageScore,
  1444→        highestScore: highestScore,
  1445→        lowestScore: lowestScore,
  1446→        medianScore: medianScore,
  1447→        passRate: passRate,
  1448→        questionsByCorrectRate: {}, // Requires per-question analysis
  1449→        averageTimeSpentSeconds: averageTimeSpent,
  1450→        gradingCompletionPercentage: gradingCompletion,
  1451→      );
  1452→    } on sb.PostgrestException catch (e) {
  1453→      AppLogger.error('Get exam statistics failed', error: e);
  1454→      throw ServerException(
  1455→        message: e.message,
  1456→        statusCode: int.tryParse(e.code ?? '') ?? 500,
  1457→        data: e.details,
  1458→      );
  1459→    } catch (e) {
  1460→      AppLogger.error('Get exam statistics unexpected error', error: e);
  1461→      throw ServerException(
  1462→        message: 'Failed to get exam statistics: $e',
  1463→        statusCode: 500,
  1464→      );
  1465→    }
  1466→  }

  1468→  @override
  1469→  Future<LiveExamStatsModel> getLiveExamStats(String examId) async {
  1470→    try {
  1471→      // Get active sessions
  1472→      final activeSessions = await _supabaseClient
  1473→          .from(_examSessionsTable)
  1474→          .select()
  1475→          .eq('exam_id', examId)
  1476→          .eq('is_active', true);
  1477→
  1478→      // Get all assigned students
  1479→      final students = await _supabaseClient
  1480→          .from(_examStudentsTable)
  1481→          .select('id')
  1482→          .eq('exam_id', examId);
  1483→
  1484→      // Get submitted attempts
  1485→      final submittedAttempts = await _supabaseClient
  1486→          .from(_examAttemptsTable)
  1487→          .select()
  1488→          .eq('exam_id', examId)
  1489→          .inFilter('status', ['submitted', 'auto_submitted', 'timed_out']);
  1490→
  1491→      // Recent submissions (last 10)
  1492→      final recentSubmissions = await _supabaseClient
  1493→          .from(_examAttemptsTable)
  1494→          .select()
  1495→          .eq('exam_id', examId)
  1496→          .inFilter('status', ['submitted', 'auto_submitted', 'timed_out'])
  1497→          .order('submitted_at', ascending: false)
  1498→          .limit(10);
  1499→
  1500→      // Recent monitoring events (last 20)
  1501→      final recentEvents = await _supabaseClient
  1502→          .from(_monitoringLogsTable)
  1503→          .select()
  1504→          .eq('exam_id', examId)
  1505→          .order('created_at', ascending: false)
  1506→          .limit(20);
  1507→
  1508→      final totalEligible = students.length;
  1509→      final activeNow = activeSessions.length;
  1510→      final completed = submittedAttempts.length;
  1511→      final notStarted = totalEligible - activeNow - completed;
  1512→
  1513→      // Calculate average progress
  1514→      double averageProgress = 0;
  1515→      if (activeSessions.isNotEmpty) {
  1516→        final totalQuestionsAnswered = activeSessions.fold<int>(
  1517→          0,
  1518→          (sum, s) =>
  1519→              sum + ((s['questions_answered'] as int?) ?? 0),
  1520→        );
  1521→        // Estimate total questions per exam
  1522→        averageProgress = totalEligible > 0
  1523→            ? totalQuestionsAnswered / (activeNow * 10) * 100
  1524→            : 0;
  1525→        averageProgress = averageProgress.clamp(0.0, 100.0);
  1526→      }
  1527→
  1528→      return LiveExamStatsModel(
  1529→        examId: examId,
  1530→        totalEligible: totalEligible,
  1531→        activeNow: activeNow,
  1532→        completed: completed,
  1533→        notStarted: notStarted < 0 ? 0 : notStarted,
  1534→        averageProgress: averageProgress,
  1535→        recentSubmissions: recentSubmissions
  1536→            .map<ExamAttemptModel>((a) => ExamAttemptModel.fromJson(a))
  1537→            .toList(),
  1538→        activeSessions: activeSessions
  1539→            .map<ExamSessionModel>((s) => ExamSessionModel.fromJson(s))
  1540→            .toList(),
  1541→        recentMonitoringEvents: recentEvents
  1542→            .map<MonitoringLogModel>((e) => MonitoringLogModel.fromJson(e))
  1543→            .toList(),
  1544→      );
  1545→    } on sb.PostgrestException catch (e) {
  1546→      AppLogger.error('Get live exam stats failed', error: e);
  1547→      throw ServerException(
  1548→        message: e.message,
  1549→        statusCode: int.tryParse(e.code ?? '') ?? 500,
  1550→        data: e.details,
  1551→      );
  1552→    } catch (e) {
  1553→      AppLogger.error('Get live exam stats unexpected error', error: e);
  1554→      throw ServerException(
  1555→        message: 'Failed to get live exam stats: $e',
  1556→        statusCode: 500,
  1557→      );
