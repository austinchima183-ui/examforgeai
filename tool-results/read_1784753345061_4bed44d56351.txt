   384→  /// Reorders questions according to the given list of question IDs.
   385→  void reorderQuestions(List<String> questionIds) {
   386→    final reordered = <ExamQuestionEntity>[];
   387→    for (var i = 0; i < questionIds.length; i++) {
   388→      final question = state.questions
   389→          .where((q) => q.questionId == questionIds[i])
   390→          .firstOrNull;
   391→      if (question != null) {
   392→        reordered.add(question.copyWith(sortOrder: i));
   393→      }
   394→    }
   395→    state = state.copyWith(questions: reordered);
   396→    AppLogger.info('Questions reordered: ${questionIds.length} items');
   397→  }

   463→  /// Saves the exam — creates if new, updates if editing.
   464→  Future<void> saveExam() async {
   465→    if (state.isSaving) return;
   466→
   467→    state = state.copyWith(isSaving: true, error: null);
   468→
   469→    if (state.isEditMode) {

   476→  Future<void> _createExam() async {
   477→    if (state.startTime == null || state.endTime == null) {
   478→      state = state.copyWith(
   479→        isSaving: false,
   480→        error: 'Start time and end time are required',
   481→      );
   482→      return;
   483→    }
   484→
   485→    final input = ExamCreateInput(
   486→      title: state.title,
   487→      description:
   488→          state.description.trim().isEmpty ? null : state.description,
   489→      subjectId: state.subjectId ?? '',
   490→      classId: state.classId ?? '',
   491→      academicSessionId: state.academicSessionId ?? '',
   492→      examType: state.examType,
   493→      startTime: state.startTime!,
   494→      endTime: state.endTime!,
   495→      timeLimitMinutes: state.timeLimitMinutes,
   496→      passMark: state.passMark,
   497→      passMarkType: state.passMarkType,
   498→      instructions:
   499→          state.instructions.trim().isEmpty ? null : state.instructions,
   500→      allowedAttempts: state.allowedAttempts,
   501→      negativeMarkingEnabled: state.negativeMarkingEnabled,
   502→      negativeMarkValue: state.negativeMarkValue,
   503→      gracePeriodMinutes: state.gracePeriodMinutes,
   504→      autoSubmit: state.autoSubmit,
   505→      randomizeQuestions: state.randomizeQuestions,
   506→      randomizeOptions: state.randomizeOptions,
   507→      showResults: state.showResults,
   508→      showCorrectAnswers: state.showCorrectAnswers,
   509→      showExplanations: state.showExplanations,
   510→      maxStudents: state.maxStudents,
   511→      requireFullScreen: state.requireFullScreen,
   512→      allowResume: state.allowResume,
   513→      browserLockdown: state.browserLockdown,
   514→    );
   515→
   516→    final result = await _createExamUseCase(
   517→      CreateExamParams(
   518→        exam: input,
   519→        sections: state.sections,
   520→        questions: state.questions,
   521→        studentIds: state.assignedStudents,
   522→      ),
   523→    );
   524→
   525→    result.fold(
   526→      onSuccess: (exam) {
   527→        state = state.copyWith(
   528→          isSaving: false,
   529→          exam: exam,
   530→          status: exam.status,
   531→          successMessage: 'Exam created successfully',
   532→          error: null,
   533→        );
   534→        AppLogger.info('Exam created: ${exam.id}');
   535→      },
   536→      onFailure: (failure) {
   537→        state = state.copyWith(
   538→          isSaving: false,
   539→          error: _mapFailureToMessage(failure),
   540→        );
   541→        AppLogger.warning('Failed to create exam: $failure');
   542→      },
   543→    );
   544→  }

   555→      academicSessionId:
   556→          state.academicSessionId ?? state.exam!.academicSessionId,
   557→      examType: state.examType,
   558→      startTime: state.startTime ?? state.exam!.startTime,
   559→      endTime: state.endTime ?? state.exam!.endTime,
   560→      timeLimitMinutes: state.timeLimitMinutes,
   561→      totalMarks: state.totalMarks,
   562→      passMark: state.passMark,
   563→      passMarkType: state.passMarkType,
   564→      instructions:
   565→          state.instructions.trim().isEmpty ? null : state.instructions,
   566→      allowedAttempts: state.allowedAttempts,
   567→      negativeMarkingEnabled: state.negativeMarkingEnabled,
   568→      negativeMarkValue: state.negativeMarkValue,
   569→      gracePeriodMinutes: state.gracePeriodMinutes,
   570→      autoSubmit: state.autoSubmit,
   571→      randomizeQuestions: state.randomizeQuestions,
   572→      randomizeOptions: state.randomizeOptions,
   573→      showResults: state.showResults,
   574→      showCorrectAnswers: state.showCorrectAnswers,
   575→      showExplanations: state.showExplanations,
   576→      maxStudents: state.maxStudents,
   577→      requireFullScreen: state.requireFullScreen,
   578→      allowResume: state.allowResume,
   579→      browserLockdown: state.browserLockdown,
   580→      sections: state.sections,
   581→      questions: state.questions,
   582→      updatedAt: DateTime.now(),
   583→    );
   584→
   585→    final result = await _updateExamUseCase(
   586→      UpdateExamParams(
   587→        exam: updatedExam,
   588→        sections: state.sections,
   589→        questions: state.questions,
   590→        studentIds: state.assignedStudents,
   591→      ),
   592→    );
   593→
   594→    result.fold(
   595→      onSuccess: (exam) {
   596→        state = state.copyWith(
   597→          isSaving: false,
   598→          exam: exam,
   599→          status: exam.status,
   600→          successMessage: 'Exam updated successfully',
   601→          error: null,
   602→        );
   603→        AppLogger.info('Exam updated: ${exam.id}');
   604→      },
   605→      onFailure: (failure) {
   606→        state = state.copyWith(
   607→          isSaving: false,
   608→          error: _mapFailureToMessage(failure),
   609→        );
   610→        AppLogger.warning('Failed to update exam: $failure');
   611→      },
   612→    );
   613→  }
   614→
   615→  // ═══════════════════════════════════════════════════════════════════════
   616→  // PUBLISH EXAM
   617→  // ═══════════════════════════════════════════════════════════════════════
   618→
   619→  /// Publishes the exam, making it visible to assigned students.
   620→  ///
   621→  /// Must be called after saving the exam.
   622→  Future<void> publishExam() async {
   623→    if (state.isPublishing) return;
   624→    if (state.exam == null) {
   625→      state = state.copyWith(error: 'Please save the exam first');
   626→      return;
   627→    }
   628→
   629→    state = state.copyWith(isPublishing: true, error: null);
   630→
   631→    final result = await _manageExamStatusUseCase(
   632→      ManageStatusParams(
   633→        examId: state.exam!.id,
   634→        action: ManageStatusAction.publish,
   635→      ),
   636→    );
   637→
   638→    result.fold(
   639→      onSuccess: (exam) {
   640→        state = state.copyWith(
   641→          isPublishing: false,
   642→          exam: exam,
   643→          status: exam.status,
   644→          successMessage: 'Exam published successfully',
   645→          error: null,
   646→        );
   647→        AppLogger.info('Exam published: ${exam.id}');
   648→      },
   649→      onFailure: (failure) {
   650→        state = state.copyWith(
   651→          isPublishing: false,
   652→          error: _mapFailureToMessage(failure),
   653→        );
   654→        AppLogger.warning('Failed to publish exam: $failure');
   655→      },
   656→    );
   657→  }
   658→
   659→  // ═══════════════════════════════════════════════════════════════════════
   660→  // CLONE EXAM
   661→  // ═══════════════════════════════════════════════════════════════════════
   662→
   663→  /// Creates a deep copy of the current exam with a new ID and draft status.
   664→  Future<void> cloneExam() async {
   665→    if (state.exam == null) return;
   666→
   667→    state = state.copyWith(isSaving: true, error: null);
   668→

   676→    result.fold(

   679→          isSaving: false,
   680→          exam: exam,
   681→          status: exam.status,
   682→          successMessage: 'Exam cloned successfully',
   683→          error: null,
   684→        );
   685→        AppLogger.info('Exam cloned: ${exam.id}');
   686→      },
   687→      onFailure: (failure) {
   688→        state = state.copyWith(
   689→          isSaving: false,
   690→          error: _mapFailureToMessage(failure),

   692→        AppLogger.warning('Failed to clone exam: $failure');
   693→      },

   695→  }
   696→
   697→  // ═══════════════════════════════════════════════════════════════════════
   698→  // LOAD EXAM FOR EDIT

   783→  /// Maps a [Failure] to a user-friendly error message.
   784→  String _mapFailureToMessage(Failure failure) {
   785→    return failure.when(
   786→      server: (message, _, __) => message,
   787→      cache: (message) => message,
   788→      auth: (message, _) => message,
   789→      network: (message) => message,
   790→      validation: (message, _) => message,
   791→      notFound: (message) => message,
   792→      unauthorized: (message) => message,
   793→      forbidden: (message) => message,
   794→    );
   795→  }
