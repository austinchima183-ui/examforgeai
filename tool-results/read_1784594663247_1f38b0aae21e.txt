    26→    26→/// AI-powered Worksheet Generator page.

    67→    67→  // ─── Generate Worksheet ───────────────────────────────────────────
    68→    68→
    69→    69→  void _generateWorksheet() {
    70→    70→    if (!_formKey.currentState!.validate()) return;
    71→    71→
    72→    72→    final questionCount = int.tryParse(_questionCountController.text.trim()) ?? 10;
    73→    73→
    74→    74→    ref.read(worksheetProvider.notifier).generateWorksheet(
    75→    75→          GenerateWorksheetParams(
    76→    76→            subject: _subjectController.text.trim(),
    77→    77→            className: _classNameController.text.trim().isNotEmpty
    78→    78→                ? _classNameController.text.trim()
    79→    79→                : null,
    80→    80→            topic: _topicController.text.trim().isNotEmpty
    81→    81→                ? _topicController.text.trim()
    82→    82→                : null,
    83→    83→            worksheetType: _worksheetType,
    84→    84→            difficulty: _difficulty,
    85→    85→            questionCount: questionCount,
    86→    86→          ),
    87→    87→        );
    88→    88→    setState(() => _showResult = true);
    89→    89→  }
    90→    90→
    91→    91→  // ─── Save Worksheet ───────────────────────────────────────────────
    92→    92→
    93→    93→  void _saveWorksheet() {
    94→    94→    final worksheet = ref.read(worksheetProvider).currentWorksheet;
    95→    95→    if (worksheet == null) return;
    96→    96→
    97→    97→    ref.read(worksheetProvider.notifier).createWorksheet(
    98→    98→          CreateWorksheetParams(worksheet: worksheet),
    99→    99→        );
   100→   100→  }

   112→   112→          ),

   114→   114→  }
