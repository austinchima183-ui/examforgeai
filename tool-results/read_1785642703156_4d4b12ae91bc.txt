    20→    20→## 1. All Environment Variables and Their Purpose
    21→    21→
    22→    22→### 1.1 Tier 1: Critical Secrets
    23→    23→
    24→    24→These variables contain sensitive values that must never be committed to source code or logged.
    25→    25→
    26→    26→| Variable | Purpose | Example Value | Used By |
    27→    27→|----------|---------|---------------|---------|
    28→    28→| `SUPABASE_URL` | Supabase project URL (environment-specific, not a secret per se) | `https://abc123.supabase.co` | Flutter app, Edge Functions, deploy scripts |
    29→    29→| `SUPABASE_ANON_KEY` | Public anonymous key (safe in client code; rotates yearly) | `eyJhbGciOiJIUzI1NiIs...` | Flutter app (`SupabaseConfig.initialize()`) |
    30→    30→| `SUPABASE_SERVICE_KEY` | Service role key — **CRITICAL: server-side only, never in client** | `eyJhbGciOiJIUzI1NiIs...` | Edge Functions, deploy scripts, monitoring |
    31→    31→| `SUPABASE_SERVICE_ROLE_KEY` | Alias for `SUPABASE_SERVICE_KEY` in Edge Functions | Same as above | Edge Functions (`Deno.env.get('SUPABASE_SERVICE_ROLE_KEY')`) |
    32→    32→| `FLUTTERWAVE_PUBLIC_KEY` | Client-side Flutterwave key (safe in Flutter app) | `FLWPUBK_TEST-...` | Flutter app (payment widget) |
    33→    33→| `FLUTTERWAVE_SECRET_KEY` | Server-side Flutterwave key — **CRITICAL: Edge Functions only** | `FLWSECK_TEST-...` | Edge Functions (`flutterwave-webhook`, `process-refund`, `health-check`) |
    34→    34→| `FLUTTERWAVE_WEBHOOK_SECRET_HASH` | Webhook verification hash — **CRITICAL: Edge Functions only** | `a1b2c3d4e5f6...` | `flutterwave-webhook` Edge Function |
    35→    35→| `FCM_SERVER_KEY` | Firebase Cloud Messaging server key (server-side only) | `AAAAxxxx...` | Push notification service |
    36→    36→| `PRODUCTION_DATABASE_URL` | PostgreSQL connection string for production | `postgresql://examforge_app:pass@prod-db.examforge.ai:5432/examforge_production` | Deploy scripts, backup scripts |
    37→    37→| `STAGING_DATABASE_URL` | PostgreSQL connection string for staging | `postgresql://examforge_app:pass@staging-db.examforge.ai:5432/examforge_staging` | Deploy scripts, backup scripts |
    38→    38→| `DATABASE_URL` | Active environment DB URL (set by deploy script at runtime) | Same as production or staging URL | Runtime — set by `deploy.sh` based on target |
    39→    39→| `PRODUCTION_HOST` | Production server hostname | `prod-db.examforge.ai` | Deploy scripts (SSH target) |
    40→    40→| `PRODUCTION_USER` | Production SSH deploy user | `deploy` | Deploy scripts (SSH user) |
    41→    41→| `STAGING_HOST` | Staging server hostname | `staging-db.examforge.ai` | Deploy scripts (SSH target) |
    42→    42→| `STAGING_USER` | Staging SSH deploy user | `deploy` | Deploy scripts (SSH user) |
    43→    43→| `ENCRYPTION_MASTER_KEY` | Master key for key derivation (future use) | 256-bit hex string | Local encryption service |
    44→    44→| `GPG_RECIPIENT` | GPG key ID for backup encryption | `admin@examforge.ai` | Backup scripts |
    45→    45→| `AWS_ACCESS_KEY_ID` | S3 access key for backups | `AKIAIOSFODNN7EXAMPLE` | Backup scripts (S3 upload) |
    46→    46→| `AWS_SECRET_ACCESS_KEY` | S3 secret key for backups | `wJalrXUtnFEMI/K7MDENG/...` | Backup scripts (S3 upload) |
    47→    47→| `SENTRY_DSN` | Sentry error tracking DSN | `https://key@sentry.io/project` | Application error reporting |
    48→    48→| `GRAFANA_API_KEY` | Grafana API key (if self-hosted) | `glsa_xxxxxxxx...` | Monitoring dashboards |

    50→    50→### 1.2 Tier 2: Configuration (Environment-Specific, Not Secrets)
    51→    51→
    52→    52→| Variable | Purpose | Default | Used By |
    53→    53→|----------|---------|---------|---------|
    54→    54→| `ENVIRONMENT` | Current environment identifier | `development` | All services — controls feature flags, timeouts, CORS origins |
    55→    55→| `APP_URL` | Public-facing application URL | `http://localhost:3000` | CORS, health checks, deep links |
    56→    56→| `API_URL` | API base URL | `http://localhost:3001` | Flutter app API calls |
    57→    57→| `DB_HOST` | Database host | `localhost` | Backup scripts, direct DB connections |
    58→    58→| `DB_PORT` | Database port | `5432` | Backup scripts, direct DB connections |
    59→    59→| `DB_NAME` | Database name | `examforge_dev` | Backup scripts |
    60→    60→| `DB_USER` | Database user (least-privilege application user) | `postgres` | Backup scripts |
    61→    61→| `DB_PASSWORD` | Database password (Tier 1 when production) | — | Backup scripts |
    62→    62→| `DB_POOL_MIN` | Connection pool minimum | `2` | Database pool manager |
    63→    63→| `DB_POOL_MAX` | Connection pool maximum | `20` (prod: `50`) | Database pool manager |
    64→    64→| `DEPLOY_PATH` | Remote deployment directory | `/var/www/examforge-dev` | Deploy scripts |
    65→    65→| `DEPLOY_STRATEGY` | Deployment strategy | `standard` | Deploy scripts (`standard` or `blue-green`) |
    66→    66→| `BACKUP_ENABLED` | Whether to run pre-deploy backups | `false` (dev) / `true` (staging/prod) | Deploy scripts |
    67→    67→| `ROLLBACK_ENABLED` | Whether to enable automatic rollback | `true` | Deploy scripts |
    68→    68→| `HEALTH_CHECK_URL` | Health check endpoint URL | `${APP_URL}/health` | Deploy scripts |
    69→    69→| `SLOW_QUERY_THRESHOLD_MS` | Slow query threshold in milliseconds | `500` (prod: `300`) | Monitoring |
    70→    70→| `HEALTH_CHECK_INTERVAL_SEC` | Health check interval in seconds | `300` | Monitoring |
    71→    71→| `LOG_LEVEL` | Logging verbosity | `info` | Application logger |
    72→    72→| `AWS_REGION` | AWS region for S3 backups | `af-south-1` | Backup scripts |
    73→    73→| `AWS_ENDPOINT` | Custom S3 endpoint (for S3-compatible storage) | — (use default AWS) | Backup scripts |
    74→    74→| `S3_BUCKET` | S3 bucket name for backups | `examforge-backups-prod` | Backup scripts |
    75→    75→| `DEPLOY_HOST` | Resolved deployment host (set by deploy script) | Same as `PRODUCTION_HOST` or `STAGING_HOST` | Deploy scripts |
    76→    76→| `DEPLOY_USER` | Resolved deployment user (set by deploy script) | Same as `PRODUCTION_USER` or `STAGING_USER` | Deploy scripts |

    78→    78→### 1.3 Compile-Time Variables (Flutter --dart-define)
    79→    79→
    80→    80→In release builds, environment variables are passed via `--dart-define` flags since the `.env` file is not available:
    81→    81→
    82→    82→```bash
    83→    83→flutter build web --release \
    84→    84→  --dart-define=SUPABASE_URL=https://abc123.supabase.co \
    85→    85→  --dart-define=SUPABASE_ANON_KEY=eyJhbGciOiJIUzI1NiIs... \
    86→    86→  --dart-define=SUPABASE_SERVICE_KEY=eyJhbGciOiJIUzI1NiIs... \
    87→    87→  --dart-define=FCM_SERVER_KEY=AAAAxxxx... \
    88→    88→  --dart-define=FLUTTERWAVE_PUBLIC_KEY=FLWPUBK_TEST-... \
    89→    89→  --dart-define=FLUTTERWAVE_SECRET_KEY=FLWSECK_TEST-... \
    90→    90→  --dart-define=ENVIRONMENT=production
    91→    91→```
    92→    92→
    93→    93→These are read in `EnvConfig.initialize()` via `String.fromEnvironment()` when the `.env` file is unavailable (release builds).
    94→    94→

   257→   257→## 3. Secret Management
   258→   258→
   259→   259→### 3.1 Secret Storage Architecture
   260→   260→
   261→   261→```
   262→   262→┌─────────────────────────────────────────────────────────────┐
   263→   263→│                     SECRET SOURCES                          │
   264→   264→│                                                             │
   265→   265→│  ┌──────────────┐  ┌──────────────┐  ┌──────────────┐     │
   266→   266→│  │   GitHub     │  │   Supabase   │  │    AWS       │     │
   267→   267→│  │   Secrets    │  │   Vault      │  │    IAM       │     │
   268→   268→│  │              │  │              │  │              │     │
   269→   269→│  │ • CI/CD vars │  │ • Edge Fn    │  │ • S3 access  │     │
   270→   270→│  │ • Deploy     │  │   secrets    │  │ • Backup     │     │
   271→   271→│  │   scripts    │  │ • App        │  │   uploads    │     │
   272→   272→│  │ • Build      │  │   secrets    │  │              │     │
   273→   273→│  └──────┬───────┘  └──────┬───────┘  └──────┬───────┘     │
   274→   274→│         │                  │                  │             │
   275→   275→│         └──────────────────┼──────────────────┘             │
   276→   276→│                            │                                │
   277→   277→│                            ▼                                │
   278→   278→│                    ┌──────────────┐                         │
   279→   279→│                    │  Application │                         │
   280→   280→│                    │  Runtime     │                         │
   281→   281→│                    └──────────────┘                         │
   282→   282→└─────────────────────────────────────────────────────────────┘
   283→   283→```

   318→   318→### 3.3 Supabase Vault
   319→   319→
   320→   320→Supabase Vault stores secrets needed by Edge Functions at runtime.
   321→   321→
   322→   322→**Setting a secret in Supabase Vault:**
   323→   323→```bash
   324→   324→# Via Supabase CLI
   325→   325→supabase secrets set SUPABASE_SERVICE_ROLE_KEY=eyJhbGciOiJIUzI1NiIs... --project-id examforge-production
   326→   326→supabase secrets set FLUTTERWAVE_SECRET_KEY=FLWSECK_TEST-... --project-id examforge-production
   327→   327→supabase secrets set FLUTTERWAVE_WEBHOOK_SECRET_HASH=a1b2c3d4e5f6... --project-id examforge-production
   328→   328→supabase secrets set ENVIRONMENT=production --project-id examforge-production
   329→   329→
   330→   330→# Listing secrets
   331→   331→supabase secrets list --project-id examforge-production
   332→   332→```
   333→   333→
   334→   334→**Secrets stored in Supabase Vault (per environment):**
   335→   335→
   336→   336→| Secret | Production | Staging | Development |
   337→   337→|--------|-----------|---------|-------------|
   338→   338→| `SUPABASE_URL` | Yes | Yes | No (local) |
   339→   339→| `SUPABASE_SERVICE_ROLE_KEY` | Yes | Yes | No (local) |
   340→   340→| `FLUTTERWAVE_SECRET_KEY` | Yes | Yes (test key) | No |
   341→   341→| `FLUTTERWAVE_WEBHOOK_SECRET_HASH` | Yes | Yes (test hash) | No |
   342→   342→| `ENVIRONMENT` | `production` | `staging` | — |
   343→   343→| `DEPLOY_VERSION` | Updated per deploy | Updated per deploy | — |
   344→   344→
   345→   345→### 3.4 Local Secret Management (Development)
   346→   346→
   347→   347→For local development, secrets are stored in the `.env` file:
   348→   348→
   349→   349→```bash
   350→   350→# Copy the template
   351→   351→cp .env.example .env
   352→   352→
   353→   353→# Fill in development values
   354→   354→# .env is listed in .gitignore and will NEVER be committed
   355→   355→```
   356→   356→
   357→   357→**Fallback behavior (from `EnvConfig`):** In release builds without a `.env` file, the app falls back to `--dart-define` compile-time constants. In debug builds, safe placeholder values are used so the app can still launch.
   358→   358→
   359→   359→### 3.5 Secret Access Audit Logging
   360→   360→
   361→   361→All secret access is logged with:
   362→   362→
   363→   363→| Field | Description |
   364→   364→|-------|-------------|
   365→   365→| Timestamp (UTC) | When the secret was accessed |
   366→   366→| User/service account | Who accessed the secret |
   367→   367→| Secret name | The name of the secret (NOT the value) |
   368→   368→| Action | `read`, `rotate`, or `delete` |
   369→   369→| Source IP | Where the access originated |
   370→   370→| Correlation ID | For tracing across services |
   371→   371→
   372→   372→**Audit log locations:**
   373→   373→
   374→   374→| System | Audit Location |
   375→   375→|--------|---------------|
   376→   376→| GitHub Actions | GitHub audit log (Settings → Audit log) |
   377→   377→| Supabase | `admin_access_log` table + Supabase Dashboard audit |

   588→   588→
   589→   589→### 5.1 Pre-Deployment Configuration Checklist
   590→   590→
   591→   591→Run through this checklist before every production deployment:
   592→   592→
   593→   593→- [ ] **All required environment variables are set** — Verify in GitHub Secrets and Supabase Vault
   594→   594→- [ ] **No secrets in source code** — Run `gitleaks detect --source .`
   595→   595→- [ ] **`.env` file is in `.gitignore`** — Verify `.env` is listed
   596→   596→- [ ] **Database URL is correct** — Verify connectivity: `psql "${PRODUCTION_DATABASE_URL}" -c "SELECT 1;"`
   597→   597→- [ ] **Supabase URL and keys match the production project** — Verify in Supabase Dashboard
   598→   598→- [ ] **Flutterwave keys are for production (not test)** — Verify in Flutterwave Dashboard
   599→   599→- [ ] **CORS origins match the environment** — Check Edge Function CORS configuration
   600→   600→- [ ] **Feature flags are set correctly for production** — Review `feature_flags` table
   601→   601→- [ ] **Connection pool size is appropriate** — `DB_POOL_MAX=50` for production
   602→   602→- [ ] **Timeouts are set for production** — 15s for all timeouts
   603→   603→- [ ] **Slow query threshold is 300ms** — Verify `SLOW_QUERY_THRESHOLD_MS=300`
   604→   604→- [ ] **Backup is enabled** — Verify `BACKUP_ENABLED=true`
   605→   605→- [ ] **Logging level is appropriate** — `LOG_LEVEL=warning` in production
   606→   606→- [ ] **Health check URL is correct** — Verify `curl -s https://examforge.ai/health | jq .`

   608→   608→### 5.2 Post-Configuration Verification Script
   609→   609→
   610→   610→```bash
   611→   611→#!/bin/bash
   612→   612→# Verify production configuration
   613→   613→set -euo pipefail
   614→   614→
   615→   615→echo "═══ ExamForge AI — Configuration Verification ═══"
   616→   616→echo ""
   617→   617→
   618→   618→# Check environment variables are set
   619→   619→for var in SUPABASE_URL SUPABASE_ANON_KEY SUPABASE_SERVICE_KEY \
   620→   620→           FLUTTERWAVE_PUBLIC_KEY FLUTTERWAVE_SECRET_KEY \
   621→   621→           ENVIRONMENT; do
   622→   622→  if [ -z "${!var:-}" ]; then
   623→   623→    echo "❌ MISSING: $var"
   624→   624→  else
   625→   625→    echo "✅ SET: $var (length: ${#!var})"
   626→   626→  fi
   627→   627→done
   628→   628→
   629→   629→echo ""
   630→   630→
   631→   631→# Check database connectivity
   632→   632→if psql "${DATABASE_URL:-}" -c "SELECT 1;" &>/dev/null; then
   633→   633→  echo "✅ Database: reachable"
   634→   634→else
   635→   635→  echo "❌ Database: unreachable"
   636→   636→fi
   637→   637→
   638→   638→# Check health endpoint
   639→   639→HTTP_CODE=$(curl -s -o /dev/null -w "%{http_code}" "${HEALTH_CHECK_URL:-http://localhost:3000/health}" 2>/dev/null || echo "000")
   640→   640→if [ "${HTTP_CODE}" = "200" ]; then
   641→   641→  echo "✅ Health check: 200 OK"
   642→   642→elif [ "${HTTP_CODE}" = "503" ]; then
   643→   643→  echo "⚠️  Health check: 503 Service Unavailable"
   644→   644→else
   645→   645→  echo "❌ Health check: HTTP ${HTTP_CODE}"
   646→   646→fi
   647→   647→
   648→   648→# Check for secrets in source code
   649→   649→if command -v gitleaks &>/dev/null; then
   650→   650→  if gitleaks detect --source . --no-banner 2>/dev/null; then
   651→   651→    echo "✅ No secrets in source code"
   652→   652→  else
   653→   653→    echo "❌ Secrets detected in source code!"
   654→   654→  fi
   655→   655→else
   656→   656→  echo "⚠️  gitleaks not installed — skipping secret scan"
   657→   657→fi
   658→   658→
   659→   659→echo ""
   660→   660→echo "═══ Verification Complete ═══"
   661→   661→```

   666→   666→
   667→   667→```sql
   668→   668→-- Check if RLS is disabled on any table (should be 0 rows)
   669→   669→SELECT tablename FROM pg_tables
   670→   670→WHERE schemaname = 'public' AND rowsecurity = false;
   671→   671→
   672→   672→-- Check if any feature flags have been changed from expected values
   673→   673→SELECT name, is_enabled, rollout_percentage
   674→   674→FROM feature_flags
   675→   675→WHERE (name = 'ai_exam_hints' AND is_enabled = true)
   676→   676→   OR (name = 'offline_mode' AND is_enabled = true)
   677→   677→   OR (name = 'biometric_login' AND is_enabled = true)
   678→   678→   OR (name = 'multi_school' AND is_enabled = true)
   679→   679→   OR (name = 'advanced_analytics' AND is_enabled = true);
   680→   680→-- All should return 0 rows in production
   681→   681→
   682→   682→-- Check for unauthorized database roles
   683→   683→SELECT rolname FROM pg_roles
   684→   684→WHERE rolname NOT IN (
   685→   685→  'postgres', 'authenticated', 'anon', 'service_role',
   686→   686→  'webhook_processor', 'refund_processor', 'monitoring_agent',

   690→   690→  -- Add other expected roles

   692→   692→)
