   256→  Widget _buildRevenueByBillingModel(RevenueAnalytics analytics, ColorScheme cs) {
   257→    final modelData = analytics.revenueByBillingModel;
   258→    if (modelData.isEmpty) {
   259→      return const AdminEmptyState(
   260→        message: 'No billing model data available.',
   261→        icon: Icons.pie_chart_outline,
   262→      );
   263→    }
   264→
   265→    // Extract billing model entries
   266→    final entries = <_BillingModelEntry>[];
   267→    double maxValue = 0;
   268→
   269→    modelData.forEach((key, value) {
   270→      final amount = (value is num) ? value.toDouble() : 0.0;
   271→      if (amount > maxValue) maxValue = amount;
   272→      entries.add(_BillingModelEntry(
   273→        name: _formatModelName(key),
   274→        revenue: amount,
   275→        color: _modelColor(entries.length),
   276→      ));
   277→    });
   278→
   279→    if (maxValue == 0) maxValue = 1; // Prevent division by zero
   280→
   281→    return Card(
   282→      elevation: Spacings.elevationSm,
   283→      shape: RoundedRectangleBorder(borderRadius: Spacings.borderRadiusMd),
   284→      child: Padding(
   285→        padding: Spacings.paddingAll,
   286→        child: Column(
   287→          children: entries.map((entry) {
   288→            final percentage = (entry.revenue / maxValue * 100).clamp(0, 100);
   289→            return Padding(
   290→              padding: const EdgeInsets.only(bottom: Spacings.md),
   291→              child: Column(
   292→                crossAxisAlignment: CrossAxisAlignment.start,
   293→                children: [
   294→                  Row(
   295→                    children: [
   296→                      Container(
   297→                        width: 12,
   298→                        height: 12,
   299→                        decoration: BoxDecoration(
   300→                          color: entry.color,
   301→                          borderRadius: Spacings.borderRadiusSm,
   302→                        ),
   303→                      ),
   304→                      const SizedBox(width: Spacings.sm),
   305→                      Expanded(
   306→                        child: Text(
   307→                          entry.name,
   308→                          style: AppTypography.wSemiBold.copyWith(fontSize: 13),
   309→                        ),
   310→                      ),
   311→                      Text(
   312→                        _formatCurrency(entry.revenue),
   313→                        style: AppTypography.wSemiBold.copyWith(

   339→  Color _modelColor(int index) {
   340→    const colors = [
   341→      AppColors.success,
   342→      AppColors.info,
   343→      Colors.deepPurple,
   344→      Colors.orange,
   345→      Colors.teal,
   346→      Colors.pink,
   347→    ];
   348→    return colors[index % colors.length];
   349→  }
   350→
   351→  String _formatModelName(String key) {
   352→    // Convert snake_case or camelCase to Title Case
   353→    return key
   354→        .replaceAllMapped(
   355→          RegExp(r'[_\s]|(?<=[a-z])(?=[A-Z])'),
   356→          (match) => ' ',
   357→        )
   358→        .split(' ')
   359→        .where((s) => s.isNotEmpty)
   360→        .map((s) => s[0].toUpperCase() + s.substring(1).toLowerCase())
   361→        .join(' ');
   362→  }

   489→    for (final entry in monthlyData) {
   490→      final revenue = (entry['revenue'] is num)
   491→          ? (entry['revenue'] as num).toDouble()
   492→          : 0.0;
   493→      if (revenue > maxRevenue) maxRevenue = revenue;
   494→    }
   495→    if (maxRevenue == 0) maxRevenue = 1;
   496→
   497→    return Card(
   498→      elevation: Spacings.elevationSm,
   499→      shape: RoundedRectangleBorder(borderRadius: Spacings.borderRadiusMd),

   510→                  final barCount = monthlyData.length;
   511→                  final barWidth =
   512→                      (constraints.maxWidth - (barCount - 1) * Spacings.sm) /

   582→  String _shortMonth(String month) {
   583→    // Expecting format like "2024-01" or "Jan 2024"
   584→    if (month.length >= 7 && month.contains('-')) {
   585→      const months = [
   586→        '', 'Jan', 'Feb', 'Mar', 'Apr', 'May', 'Jun',
   587→        'Jul', 'Aug', 'Sep', 'Oct', 'Nov', 'Dec'
   588→      ];
   589→      final parts = month.split('-');
   590→      final m = int.tryParse(parts[1]) ?? 0;
   591→      return m > 0 && m < 13 ? '${months[m]} ${parts[0].substring(2)}' : month;
   592→    }
   593→    return month.length > 6 ? month.substring(0, 6) : month;
   594→  }

   647→            final growth = (entry['growth'] is num)

   709→  // ─── Formatting Helpers ─────────────────────────────────────────────────

   711→  /// Formats a double as Naira currency with comma separators.
   712→  static String _formatCurrency(double n) {
   713→    final formatted = n.toStringAsFixed(0).replaceAllMapped(
   714→          RegExp(r'(\d{1,3})(?=(\d{3})+(?!\d))'),
   715→          (Match m) => '${m[1]},',
   716→        );
   717→    return '₦$formatted';
   718→  }
   719→
   720→  /// Compact currency format for bar chart labels (e.g. "₦1.2M").
   721→  static String _formatCompactCurrency(double n) {
   722→    if (n >= 1000000) {
   723→      return '₦${(n / 1000000).toStringAsFixed(1)}M';
   724→    } else if (n >= 1000) {
   725→      return '₦${(n / 1000).toStringAsFixed(0)}K';
   726→    }
   727→    return '₦${n.toStringAsFixed(0)}';
   728→  }
