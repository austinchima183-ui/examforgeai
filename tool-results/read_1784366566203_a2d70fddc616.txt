    97→/// Riverpod provider for the application's [GoRouter].
    98→///
    99→/// Watches [authStateProvider] so that the router automatically
   100→/// re-evaluates its [redirect] callback whenever the authentication
   101→/// state changes (sign-in, sign-out, token refresh).
   102→///
   103→/// Usage in `main.dart`:
   104→/// ```dart
   105→/// final router = ref.read(appRouterProvider);
   106→/// MaterialApp.router(routerConfig: router);
   107→/// ```
   108→final appRouterProvider = Provider<GoRouter>((ref) {
   109→  // Watch auth state so the router re-evaluates redirects on change.
   110→  final authState = ref.watch(authStateProvider);
   111→
   112→  // Also watch onboarding state and role so those redirects stay live.
   113→  ref.watch(onboardingCompleteProvider);
   114→  ref.watch(userRoleProvider);
   115→
   116→  return GoRouter(
   117→    navigatorKey: _rootNavigatorKey,
   118→    initialLocation: RouteNames.splash,
   119→    debugLogDiagnostics: true,
   120→
   121→    // ─── Global Redirect ──────────────────────────────────────────
   122→    redirect: (context, state) {
   123→      final currentPath = state.matchedLocation;
   124→      final isAuthenticated = authState.when(
   125→        data: (authData) => authData.session != null,
   126→        loading: () {
   127→          // While loading, only allow splash. Everything else redirects
   128→          // to splash so we don't flash the login page briefly.
   129→          if (currentPath != RouteNames.splash) {
   130→            return RouteNames.splash;
   131→          }
   132→          return null;
   133→        },
   134→        error: (_, __) => false,
   135→      );
   136→
   137→      // ── Auth Guard ────────────────────────────────────────────
   138→      final authRedirect = AuthGuard.evaluate(
   139→        isAuthenticated: isAuthenticated,
   140→        currentPath: currentPath,
   141→      );
   142→      if (authRedirect != null) return authRedirect;
   143→
   144→      // ── Onboarding Guard ─────────────────────────────────────
   145→      final onboardingAsync = ref.read(onboardingCompleteProvider);
   146→      final isOnboardingComplete = onboardingAsync.when(
   147→        data: (complete) => complete,
   148→        loading: () => true, // Assume complete while loading to avoid flicker.
   149→        error: (_, __) => true,
   150→      );
   151→
   152→      final onboardingRedirect = OnboardingGuard.evaluate(
   153→        isAuthenticated: isAuthenticated,
   154→        isOnboardingComplete: isOnboardingComplete,
   155→        currentPath: currentPath,
   156→      );
   157→      if (onboardingRedirect != null) return onboardingRedirect;
   158→
   159→      // ── Role-Based Guard ─────────────────────────────────────
   160→      final roleAsync = ref.read(userRoleProvider);
   161→      final userRole = roleAsync.when(
   162→        data: UserRole.fromString,
   163→        loading: () => null,
   164→        error: (_, __) => null,
   165→      );
   166→
   167→      final roleRedirect = RoleBasedGuard.evaluate(
   168→        userRole: userRole,
   169→        currentPath: currentPath,
   170→      );
   171→      if (roleRedirect != null) return roleRedirect;
   172→
   173→      // No redirect needed — proceed as requested.
   174→      return null;
   175→    },
   176→

   679→// NOT FOUND PAGE (404)
   680→// ═══════════════════════════════════════════════════════════════════════
   681→
   682→/// Shown when the user navigates to a route that does not exist in
   683→/// the route tree.
   684→///
   685→/// Provides a friendly message and a button to navigate back to the
   686→/// dashboard (or login if unauthenticated).
   687→class NotFoundPage extends ConsumerWidget {
   688→  const NotFoundPage({
   689→    required this.error,
   690→    this.path,
   691→    super.key,
   692→  });
   693→
   694→  /// The [GoException] from GoRouter describing the error.
   695→  final GoException? error;
   696→
   697→  /// The unmatched path the user attempted to visit.
   698→  final String? path;
   699→
   700→  @override
   701→  Widget build(BuildContext context, WidgetRef ref) {
   702→    final theme = Theme.of(context);
   703→    final isAuthenticated = ref.watch(isAuthenticatedProvider);
   704→
   705→    return Scaffold(
   706→      body: Center(
   707→        child: Padding(
   708→          padding: const EdgeInsets.all(32),
   709→          child: Column(
   710→            mainAxisSize: MainAxisSize.min,
   711→            children: [
   712→              Icon(
   713→                Icons.search_off_rounded,
   714→                size: 80,
   715→                color: theme.colorScheme.error,
   716→              ),
   717→              const SizedBox(height: 24),
   718→              Text(
   719→                'Page Not Found',
   720→                style: theme.textTheme.headlineMedium?.copyWith(
   721→                  fontWeight: FontWeight.bold,
   722→                ),
   723→                textAlign: TextAlign.center,
   724→              ),
   725→              const SizedBox(height: 12),
   726→              if (path != null)
   727→                Text(
   728→                  'The page "$path" does not exist.',
   729→                  style: theme.textTheme.bodyLarge?.copyWith(
   730→                    color: theme.colorScheme.onSurfaceVariant,
   731→                  ),
   732→                  textAlign: TextAlign.center,
   733→                ),
   734→              if (error != null) ...[
   735→                const SizedBox(height: 8),
   736→                Text(
   737→                  error!.message,
   738→                  style: theme.textTheme.bodySmall?.copyWith(
   739→                    color: theme.colorScheme.outline,
   740→                  ),
   741→                  textAlign: TextAlign.center,
   742→                ),
   743→              ],
   744→              const SizedBox(height: 32),
   745→              FilledButton.icon(
   746→                onPressed: () => context.go(
   747→                  isAuthenticated
   748→                      ? RouteNames.dashboard
   749→                      : RouteNames.login,
   750→                ),
   751→                icon: const Icon(Icons.home),
   752→                label: Text(
   753→                  isAuthenticated ? 'Go to Dashboard' : 'Go to Login',
   754→                ),
   755→              ),
   756→            ],
   757→          ),
   758→        ),
   759→      ),
   760→    );
   761→  }
   762→}
