    12→// ───────────────────────────────────────────────────────────────────────
    13→// Helper: parse DateTime accepting both snake_case and camelCase keys
    14→// ───────────────────────────────────────────────────────────────────────
    15→
    16→DateTime _parseDateTime(dynamic value) {
    17→  if (value is String) return DateTime.parse(value);
    18→  if (value is DateTime) return value;
    19→  return DateTime.now();
    20→}
    21→
    22→DateTime? _parseDateTimeNullable(dynamic value) {
    23→  if (value == null) return null;
    24→  if (value is String) return DateTime.parse(value);
    25→  if (value is DateTime) return value;
    26→  return null;
    27→}
