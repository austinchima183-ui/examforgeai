    21→/// Academic performance page for a specific child in the Parent Portal.
    22→///
    23→/// Displays comprehensive academic information including:
    24→/// - Overall performance card with average score and attendance rate
    25→/// - Subject performance list with colour-coded scores
    26→/// - Performance chart placeholder
    27→/// - Teacher remarks section
    28→/// - Download report card button with format options
    29→///
    30→/// Receives [studentId] as a route parameter and loads performance
    31→/// data using [childPerformanceProvider].
    32→class ChildPerformancePage extends ConsumerStatefulWidget {
    33→  const ChildPerformancePage({
    34→    super.key,
    35→    required this.studentId,
    36→  });
    37→
    38→  /// Unique identifier of the student whose performance is displayed.
    39→  final String studentId;
    40→
    41→  @override
    42→  ConsumerState<ChildPerformancePage> createState() => _State();
    43→}
    44→
    45→class _State extends ConsumerState<ChildPerformancePage> {
    46→  // ─── Lifecycle ──────────────────────────────────────────────────────
    47→
    48→  @override
    49→  void initState() {
    50→    super.initState();
    51→    WidgetsBinding.instance.addPostFrameCallback((_) {
    52→      ref
    53→          .read(childPerformanceProvider.notifier)
    54→          .loadPerformance(widget.studentId);
    55→    });
    56→  }
    57→
    58→  // ─── Build ──────────────────────────────────────────────────────────
    59→
    60→  @override
    61→  Widget build(BuildContext context) {
    62→    final performanceState = ref.watch(childPerformanceProvider);
    63→
    64→    return Scaffold(
    65→      appBar: const AppAppBar(title: 'Academic Performance'),
    66→      body: _buildBody(context, performanceState),
    67→    );
    68→  }
    69→
    70→  // ─── Body Router ────────────────────────────────────────────────────
    71→
    72→  Widget _buildBody(BuildContext context, ChildPerformanceState state) {
    73→    // Loading state
    74→    if (state.isLoading && state.performance == null) {
    75→      return _buildShimmerLoading(context);
    76→    }
    77→
    78→    // Error state
    79→    if (state.error != null && state.performance == null) {
    80→      return AppErrorState.genericError(
    81→        message: state.error,
    82→        onRetry: () => ref
    83→            .read(childPerformanceProvider.notifier)
    84→            .loadPerformance(widget.studentId),
    85→      );
    86→    }
    87→
    88→    // Empty state
    89→    final performance = state.performance;
    90→    if (performance == null) {
    91→      return AppEmptyState.noData(
    92→        title: 'No Performance Data',
    93→        subtitle:
    94→            'Academic performance data will appear here once available.',
    95→        actionLabel: 'Retry',
    96→        onAction: () => ref
    97→            .read(childPerformanceProvider.notifier)
    98→            .loadPerformance(widget.studentId),
    99→      );
   100→    }
   101→
   102→    // Success — render performance
   103→    return RefreshIndicator(
   104→      onRefresh: () => ref
   105→          .read(childPerformanceProvider.notifier)
   106→          .refreshPerformance(widget.studentId),
   107→      child: SingleChildScrollView(
   108→        physics: const AlwaysScrollableScrollPhysics(),
   109→        padding: const EdgeInsets.only(bottom: Spacings.xxl),
   110→        child: Column(
   111→          crossAxisAlignment: CrossAxisAlignment.start,
   112→          children: [
   113→            // ─── Overall Performance Card ───────────────────────────
   114→            _buildOverallPerformanceCard(context, performance),
   115→
   116→            const SizedBox(height: Spacings.xl),
   117→
   118→            // ─── Subject Performance List ───────────────────────────
   119→            _buildSubjectPerformanceList(context, performance),
   120→
   121→            const SizedBox(height: Spacings.xl),
   122→
   123→            // ─── Performance Chart Placeholder ──────────────────────
   124→            _buildChartPlaceholder(context),
   125→
   126→            const SizedBox(height: Spacings.xl),
   127→
   128→            // ─── Teacher Remarks Section ────────────────────────────
   129→            _buildTeacherRemarksSection(context, performance),
   130→
   131→            const SizedBox(height: Spacings.xl),
   132→
   133→            // ─── Download Report Card Button ────────────────────────
   134→            _buildDownloadReportButton(context),
   135→
   136→            const SizedBox(height: Spacings.lg),
   137→          ],
   138→        ),
   139→      ),
   140→    );
   141→  }

   814→  /// Returns a colour based on the score percentage.
   815→  ///
   816→  /// Green for >70%, amber for 50–70%, red for <50%.
   817→  Color _scoreColor(double score, Brightness brightness) {
   818→    if (score > 70) return AppColors.successOf(brightness);
   819→    if (score >= 50) return AppColors.warningOf(brightness);
   820→    return AppColors.errorOf(brightness);
   821→  }
   822→
   823→  /// Returns a background colour based on the score percentage.
   824→  Color _scoreBackgroundColor(double score, Brightness brightness) {
   825→    if (score > 70) return AppColors.successLight;
   826→    if (score >= 50) return AppColors.warningLight;
   827→    return AppColors.errorLight;
   828→  }

   830→  /// Formats a [DateTime] as a short date string.
   831→  String _formatDate(DateTime dt) {
   832→    const months = [
   833→      '',
   834→      'Jan',
   835→      'Feb',
   836→      'Mar',
   837→      'Apr',
   838→      'May',
   839→      'Jun',
   840→      'Jul',
   841→      'Aug',
   842→      'Sep',
   843→      'Oct',
   844→      'Nov',
   845→      'Dec',
   846→    ];
   847→    return '${months[dt.month]} ${dt.day}, ${dt.year}';
   848→  }
