   300→   300→/// Supabase-backed implementation of [MarketplaceRemoteDataSource].
   301→   301→///
   302→   302→/// Every method maps Supabase-specific responses and errors into the

   305→   305→/// exceptions with user-friendly messages.

   307→   307→  MarketplaceRemoteDataSourceImpl({
   308→   308→    required sb.SupabaseClient supabaseClient,
   309→   309→  }) : _supabase = supabaseClient;

   343→   343→  @override
   344→   344→  Future<List<MarketplaceCategoryModel>> getCategories({
   345→   345→    bool activeOnly = true,
   346→   346→    int limit = PaginatedQueryMixin.dropdownPageSize,
   347→   347→  }) async {
   348→   348→    try {
   349→   349→      var query = _supabase.from(_categoriesTable).select();
   350→   350→
   351→   351→      if (activeOnly) {
   352→   352→        query = query.eq('is_active', true);
   353→   353→      }
   354→   354→
   355→   355→      // PERF: Added limit to prevent unbounded query on marketplace_categories
   356→   356→      final response = await query.order('sort_order', ascending: true).limit(limit);
   357→   357→
   358→   358→      AppLogger.info('Fetched ${response.length} marketplace categories');
   359→   359→      return response
   360→   360→          .map<MarketplaceCategoryModel>(
   361→   361→            (row) => MarketplaceCategoryModel.fromJson(row),
   362→   362→          )
   363→   363→          .toList();
   364→   364→    } on sb.PostgrestException catch (e) {
   365→   365→      throw _mapPostgrestException(e);
   366→   366→    } catch (e) {
   367→   367→      AppLogger.error('Unexpected getCategories error', error: e);
   368→   368→      throw const ServerException(
   369→   369→        message: 'Failed to fetch marketplace categories.',
   370→   370→        statusCode: 500,
   371→   371→      );
   372→   372→    }
   373→   373→  }
   374→   374→
   375→   375→  @override
   376→   376→  Future<MarketplaceCategoryModel> getCategory(String categoryId) async {
   377→   377→    try {
   378→   378→      final response = await _supabase
   379→   379→          .from(_categoriesTable)
   380→   380→          .select()
   381→   381→          .eq('id', categoryId)
   382→   382→          .limit(1);
   383→   383→
   384→   384→      if (response.isEmpty) {
   385→   385→        throw const NotFoundException(message: 'Category not found.');
   386→   386→      }
   387→   387→
   388→   388→      return MarketplaceCategoryModel.fromJson(response.first);
   389→   389→    } on sb.PostgrestException catch (e) {
   390→   390→      throw _mapPostgrestException(e);
   391→   391→    } on NotFoundException {
   392→   392→      rethrow;
   393→   393→    } catch (e) {
   394→   394→      AppLogger.error('Unexpected getCategory error', error: e);
   395→   395→      throw const ServerException(
   396→   396→        message: 'Failed to fetch category.',
   397→   397→        statusCode: 500,
   398→   398→      );
   399→   399→    }
   400→   400→  }
   401→   401→
   402→   402→  @override
   403→   403→  Future<MarketplaceCategoryModel> upsertCategory(
   404→   404→    MarketplaceCategoryModel category,
   405→   405→  ) async {
   406→   406→    try {
   407→   407→      final data = category.toJson();
   408→   408→      data['updated_at'] = DateTime.now().toIso8601String();
   409→   409→
   410→   410→      final response = await _supabase
   411→   411→          .from(_categoriesTable)
   412→   412→          .upsert(data)
   413→   413→          .select();
   414→   414→
   415→   415→      if (response.isEmpty) {
   416→   416→        throw const ServerException(
   417→   417→          message: 'Category upsert returned no data.',
   418→   418→          statusCode: 500,
   419→   419→        );
   420→   420→      }
   421→   421→
   422→   422→      AppLogger.info('Category upserted: ${response.first['id']}');
   423→   423→      return MarketplaceCategoryModel.fromJson(response.first);
   424→   424→    } on sb.PostgrestException catch (e) {
   425→   425→      throw _mapPostgrestException(e);
   426→   426→    } on ServerException {
   427→   427→      rethrow;
   428→   428→    } catch (e) {
   429→   429→      AppLogger.error('Unexpected upsertCategory error', error: e);
   430→   430→      throw const ServerException(
   431→   431→        message: 'Failed to upsert category.',
   432→   432→        statusCode: 500,
   433→   433→      );
   434→   434→    }
   435→   435→  }
   436→   436→
   437→   437→  @override
   438→   438→  Future<bool> deleteCategory(String categoryId) async {
   439→   439→    try {
   440→   440→      await _supabase
   441→   441→          .from(_categoriesTable)
   442→   442→          .update({
   443→   443→            'is_active': false,
   444→   444→            'updated_at': DateTime.now().toIso8601String(),
   445→   445→          })
   446→   446→          .eq('id', categoryId);
   447→   447→
   448→   448→      AppLogger.info('Category soft-deleted: $categoryId');
   449→   449→      return true;
   450→   450→    } on sb.PostgrestException catch (e) {
   451→   451→      throw _mapPostgrestException(e);
   452→   452→    } catch (e) {
   453→   453→      AppLogger.error('Unexpected deleteCategory error', error: e);
   454→   454→      throw const ServerException(
   455→   455→        message: 'Failed to delete category.',
   456→   456→        statusCode: 500,
   457→   457→      );
   458→   458→    }
