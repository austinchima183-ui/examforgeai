   258→// NAVIGATOR KEYS
   259→// ═══════════════════════════════════════════════════════════════════════
   260→
   261→/// Root navigator key — owns the top-level navigation stack.
   262→final _rootNavigatorKey = GlobalKey<NavigatorState>();
   263→
   264→/// Shell navigator key — owns the nested navigation stack inside the
   265→/// [DashboardShell]. This allows the shell (sidebar / bottom nav) to
   266→/// remain mounted while inner pages transition.
   267→final _shellNavigatorKey = GlobalKey<NavigatorState>();
   268→
   269→// ═══════════════════════════════════════════════════════════════════════
   270→// APP ROUTER PROVIDER
   271→// ═══════════════════════════════════════════════════════════════════════
   272→
   273→/// Riverpod provider for the application's [GoRouter].
   274→///
   275→/// Watches [authStateProvider] so that the router automatically
   276→/// re-evaluates its [redirect] callback whenever the authentication
   277→/// state changes (sign-in, sign-out, token refresh).
   278→///
   279→/// Usage in `main.dart`:
   280→/// ```dart
   281→/// final router = ref.read(appRouterProvider);
   282→/// MaterialApp.router(routerConfig: router);
   283→/// ```
   284→final appRouterProvider = Provider<GoRouter>((ref) {
   285→  // Watch auth state so the router re-evaluates redirects on change.
   286→  final authState = ref.watch(authStateProvider);
   287→
   288→  // Also watch onboarding state and role so those redirects stay live.
   289→  ref.watch(onboardingCompleteProvider);
   290→  ref.watch(userRoleProvider);
   291→
   292→  return GoRouter(
   293→    navigatorKey: _rootNavigatorKey,
   294→    initialLocation: RouteNames.splash,
   295→    debugLogDiagnostics: true,
   296→
   297→    // ─── Global Redirect ──────────────────────────────────────────
   298→    redirect: (context, state) {
   299→      final currentPath = state.matchedLocation;
   300→      final isAuthenticated = authState.when(
   301→        data: (authData) => authData.session != null,
   302→        loading: () {
   303→          // While loading, only allow splash. Everything else redirects
   304→          // to splash so we don't flash the login page briefly.
   305→          if (currentPath != RouteNames.splash) {
   306→            return RouteNames.splash;
   307→          }
   308→          return null;
   309→        },
   310→        error: (_, __) => false,
   311→      );
   312→
   313→      // ── Auth Guard ────────────────────────────────────────────
   314→      final authRedirect = AuthGuard.evaluate(
   315→        isAuthenticated: isAuthenticated,
   316→        currentPath: currentPath,
   317→      );
   318→      if (authRedirect != null) return authRedirect;
   319→
   320→      // ── Onboarding Guard ─────────────────────────────────────
   321→      final onboardingAsync = ref.read(onboardingCompleteProvider);
   322→      final isOnboardingComplete = onboardingAsync.when(
   323→        data: (complete) => complete,
   324→        loading: () => true, // Assume complete while loading to avoid flicker.
   325→        error: (_, __) => true,
   326→      );
   327→
   328→      final onboardingRedirect = OnboardingGuard.evaluate(
   329→        isAuthenticated: isAuthenticated,
   330→        isOnboardingComplete: isOnboardingComplete,
   331→        currentPath: currentPath,
   332→      );
   333→      if (onboardingRedirect != null) return onboardingRedirect;
   334→
   335→      // ── Role-Based Guard ─────────────────────────────────────
   336→      final roleAsync = ref.read(userRoleProvider);
   337→      final userRole = roleAsync.when(
   338→        data: UserRole.fromString,
   339→        loading: () => null,
   340→        error: (_, __) => null,
   341→      );
   342→
   343→      final roleRedirect = RoleBasedGuard.evaluate(
   344→        userRole: userRole,
   345→        currentPath: currentPath,
   346→      );
   347→      if (roleRedirect != null) return roleRedirect;
   348→
   349→      // No redirect needed — proceed as requested.
   350→      return null;
   351→    },
   352→
   353→    // ─── Route Tree ───────────────────────────────────────────────
