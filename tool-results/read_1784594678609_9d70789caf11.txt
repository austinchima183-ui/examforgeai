    19→
    20→/// Secure messaging page for parent-teacher communication.
    21→///
    22→/// Presents two views:
    23→/// - **Threads list** (default): Shows all conversation threads with
    24→///   avatar, name, role badge, last message preview, unread count,
    25→///   timestamp, and optional student name tag.
    26→/// - **Conversation view** (shown when a thread is selected): Shows
    27→///   message bubbles with date separators, read receipts, and an
    28→///   input bar for composing new messages.
    29→///
    30→/// Uses [ConsumerStatefulWidget] with a private [_State] class pattern,
    31→/// loads data in [initState] via [WidgetsBinding.instance.addPostFrameCallback],
    32→/// and renders a responsive layout using [Scaffold] with [AppAppBar].
    33→class ParentMessagingPage extends ConsumerStatefulWidget {
    34→  const ParentMessagingPage({super.key});
    35→
    36→  @override
    37→  ConsumerState<ParentMessagingPage> createState() => _State();
    38→}
    39→
    40→class _State extends ConsumerState<ParentMessagingPage> {
    41→  // ─── State ──────────────────────────────────────────────────────────
    42→
    43→  /// The currently selected thread, or `null` to show the threads list.
    44→  ParentMessageThreadEntity? _selectedThread;
    45→
    46→  /// Search query for filtering threads.
    47→  String _searchQuery = '';
    48→
    49→  /// Controller for the message input field.
    50→  final TextEditingController _messageController = TextEditingController();
    51→
    52→  /// Scroll controller for the conversation messages list.

    56→

    59→    super.initState();

    65→  @override

    67→    _messageController.dispose();
    68→    _scrollController.dispose();
    69→    super.dispose();
    70→  }

   165→
   166→  Widget _buildThreadsView(
   167→    BuildContext context,
   168→    ParentMessagingState state,
   169→  ) {
   170→    // Loading state
   171→    if (state.isLoading && state.threads.isEmpty) {
   172→      return _buildShimmerLoading(context);
   173→    }
   174→
   175→    // Error state
   176→    if (state.error != null && state.threads.isEmpty) {
   177→      return AppErrorState.genericError(
   178→        message: state.error,
   179→        onRetry: () => ref.read(parentMessagingProvider.notifier).loadThreads(),
   180→      );
   181→    }
   182→
   183→    // Filter threads by search query
   184→    final threads = _filterThreads(state.threads);
   185→
   186→    return Column(
   187→      children: [
   188→        // ─── Search Bar ───────────────────────────────────────────
   189→        _buildSearchBar(context),
   190→
   191→        // ─── Threads List ─────────────────────────────────────────
   192→        Expanded(
   193→          child: threads.isEmpty
   194→              ? AppEmptyState.noMessages(
   195→                  title: 'No Messages Yet',
   196→                  subtitle:
   197→                      'Start a conversation with a teacher.',
   198→                  actionLabel: 'Compose',
   199→                  onAction: () {
   200→                    // TODO: Navigate to compose
   201→                  },
   202→                )
   203→              : RefreshIndicator(
   204→                  onRefresh: () =>
   205→                      ref.read(parentMessagingProvider.notifier).loadThreads(),
   206→                  child: ListView.separated(
   207→                    padding: const EdgeInsets.only(bottom: Spacings.xxl),
   208→                    itemCount: threads.length,
   209→                    separatorBuilder: (_, __) =>
   210→                        const SizedBox(height: Spacings.xs),
   211→                    itemBuilder: (_, index) =>
   212→                        _buildThreadCard(context, threads[index]),
   213→                  ),
   214→                ),
   215→        ),
   216→      ],
   217→    );
   218→  }

   430→  Widget _buildConversationView(
   431→    BuildContext context,
   432→    ParentMessagingState state,
   433→  ) {
   434→    final messages = state.messages;
   435→
   436→    return Column(
   437→      children: [
   438→        // ─── Messages List ────────────────────────────────────────
   439→        Expanded(
   440→          child: state.isLoading && messages.isEmpty
   441→              ? _buildShimmerLoading(context)
   442→              : messages.isEmpty
   443→                  ? AppEmptyState.noMessages(
   444→                      title: 'No Messages',
   445→                      subtitle: 'Start the conversation below.',
   446→                    )
   447→                  : RefreshIndicator(
   448→                      onRefresh: () => ref
   449→                          .read(parentMessagingProvider.notifier)
   450→                          .loadMessages(_selectedThread!.threadId),
   451→                      child: ListView.builder(
   452→                        controller: _scrollController,
   453→                        padding: const EdgeInsets.symmetric(
   454→                          horizontal: Spacings.lg,
   455→                          vertical: Spacings.md,
   456→                        ),
   457→                        itemCount: messages.length,
   458→                        itemBuilder: (_, index) {
   459→                          final message = messages[index];
   460→                          final showDateSeparator = _shouldShowDateSeparator(
   461→                            messages,
   462→                            index,
   463→                          );
   464→                          return Column(
   465→                            crossAxisAlignment: CrossAxisAlignment.stretch,
   466→                            children: [
   467→                              if (showDateSeparator)
   468→                                _buildDateSeparator(
   469→                                  context,
   470→                                  message.createdAt,
   471→                                ),
   472→                              _buildMessageBubble(context, message),
   473→                            ],
   474→                          );
   475→                        },
   476→                      ),
   477→                    ),
   478→        ),
   479→
   480→        // ─── Input Bar ────────────────────────────────────────────
   481→        _buildInputBar(context, state),
   482→      ],
   483→    );
   484→  }

   486→  // ─── Message Bubble ─────────────────────────────────────────────────
   487→
   488→  Widget _buildMessageBubble(
   489→    BuildContext context,
   490→    ParentMessageEntity message,
   491→  ) {
   492→    final cs = context.colorScheme;
   493→    final tt = context.textTheme;
   494→    final isSent = message.direction == MessageDirection.outgoing;
   495→    final isRead = message.readAt != null;
   496→
   497→    return Padding(
   498→      padding: const EdgeInsets.only(bottom: Spacings.md),
   499→      child: Align(
   500→        alignment: isSent ? Alignment.centerRight : Alignment.centerLeft,
   501→        child: ConstrainedBox(
   502→          constraints: BoxConstraints(
   503→            maxWidth: context.width * 0.75,
   504→          ),
   505→          child: Container(
   506→            padding: const EdgeInsets.symmetric(
   507→              horizontal: Spacings.md,
   508→              vertical: Spacings.sm,
   509→            ),
   510→            decoration: BoxDecoration(
   511→              color: isSent ? cs.primary : cs.surfaceContainerHighest,
   512→              borderRadius: BorderRadius.only(
   513→                topLeft: const Radius.circular(Spacings.mdRadius),
   514→                topRight: const Radius.circular(Spacings.mdRadius),
   515→                bottomLeft: Radius.circular(
   516→                  isSent ? Spacings.mdRadius : Spacings.xs,
   517→                ),
   518→                bottomRight: Radius.circular(
   519→                  isSent ? Spacings.xs : Spacings.mdRadius,
   520→                ),
   521→              ),
   522→            ),
   523→            child: Column(
   524→              crossAxisAlignment: isSent
   525→                  ? CrossAxisAlignment.end
   526→                  : CrossAxisAlignment.start,
   527→              children: [
   528→                // Message body
   529→                Text(
   530→                  message.body,
   531→                  style: tt.bodyMedium?.copyWith(
   532→                    color: isSent ? cs.onPrimary : cs.onSurface,
   533→                  ),
   534→                ),
   535→                const SizedBox(height: Spacings.xs),
   536→                // Timestamp + read receipt
   537→                Row(
   538→                  mainAxisSize: MainAxisSize.min,
   539→                  children: [
   540→                    Text(
   541→                      _formatMessageTime(message.createdAt),
   542→                      style: tt.labelSmall?.copyWith(
   543→                        color: isSent
   544→                            ? cs.onPrimary.withValues(alpha: 0.7)
   545→                            : cs.onSurfaceVariant,
   546→                      ),
   547→                    ),
   548→                    if (isSent) ...[
   549→                      const SizedBox(width: Spacings.xs),
   550→                      Icon(
   551→                        isRead
   552→                            ? Icons.done_all
   553→                            : Icons.done,
   554→                        size: Spacings.smIcon,
   555→                        color: isRead
   556→                            ? cs.onPrimary.withValues(alpha: 0.7)
   557→                            : cs.onPrimary.withValues(alpha: 0.5),
   558→                      ),
   559→                    ],
   560→                  ],
   561→                ),
   562→              ],
   563→            ),
   564→          ),
   565→        ),
   566→      ),
   567→    );
   568→  }

   731→  /// Opens a conversation thread and loads its messages.
   732→  void _openThread(ParentMessageThreadEntity thread) {
   733→    setState(() => _selectedThread = thread);
   734→    ref.read(parentMessagingProvider.notifier).loadMessages(thread.threadId);

   737→  /// Sends a message in the current conversation.
   738→  void _sendMessage(ParentMessagingState state) {
   739→    final text = _messageController.text.trim();
   740→    if (text.isEmpty || _selectedThread == null) return;
   741→
   742→    _messageController.clear();
   743→    ref.read(parentMessagingProvider.notifier).sendMessage(
   744→      SendMessageParams(
   745→        recipientId: _selectedThread!.otherUserId,
   746→        subject: _selectedThread!.lastMessage?.subject ?? 'Message',
   747→        body: text,
   748→      ),
   749→    );
   750→
   751→    // Auto-scroll to bottom after sending
   752→    WidgetsBinding.instance.addPostFrameCallback((_) {
   753→      if (_scrollController.hasClients) {
   754→        _scrollController.animateTo(
   755→          _scrollController.position.maxScrollExtent,
   756→          duration: const Duration(milliseconds: 300),
   757→          curve: Curves.easeOut,
   758→        );
   759→      }
   760→    });
   761→  }

   767→  /// Filters threads by the current search query.
   768→  List<ParentMessageThreadEntity> _filterThreads(
   769→    List<ParentMessageThreadEntity> threads,
   770→  ) {
   771→    if (_searchQuery.isEmpty) return threads;
   772→    final query = _searchQuery.toLowerCase();
   773→    return threads.where((thread) {
   774→      return thread.otherUserName.toLowerCase().contains(query) ||
   775→          (thread.studentName?.toLowerCase().contains(query) ?? false) ||
   776→          (thread.lastMessage?.body.toLowerCase().contains(query) ?? false);
   777→    }).toList();
   778→  }

   784→  /// Returns a human-readable role label.
   785→  String _roleLabel(String role) {
   786→    switch (role.toLowerCase()) {
   787→      case 'teacher':
   788→        return 'Teacher';
   789→      case 'admin':
   790→        return 'Admin';
   791→      case 'principal':
   792→        return 'Principal';
   793→      default:
   794→        return role[0].toUpperCase() + role.substring(1);
   795→    }
   796→  }
   797→
   798→  /// Determines whether a date separator should be shown before [index].
   799→  bool _shouldShowDateSeparator(
   800→    List<ParentMessageEntity> messages,
   801→    int index,
   802→  ) {
   803→    if (index == 0) return true;
   804→    final current = messages[index].createdAt;
   805→    final previous = messages[index - 1].createdAt;
   806→    return current.year != previous.year ||
   807→        current.month != previous.month ||
   808→        current.day != previous.day;
   809→  }
   810→
   811→  /// Formats a [DateTime] as a short date string for separators.
   812→  String _formatDate(DateTime dt) {
   813→    final now = DateTime.now();
   814→    final today = DateTime(now.year, now.month, now.day);
   815→    final date = DateTime(dt.year, dt.month, dt.day);
   816→
   817→    if (date == today) return 'Today';
   818→    if (date == today.subtract(const Duration(days: 1))) return 'Yesterday';
   819→
   820→    const months = [
   821→      '',
   822→      'Jan',
   823→      'Feb',
   824→      'Mar',
   825→      'Apr',
   826→      'May',
   827→      'Jun',
   828→      'Jul',
   829→      'Aug',
   830→      'Sep',
   831→      'Oct',
   832→      'Nov',
   833→      'Dec',
   834→    ];
   835→    return '${months[dt.month]} ${dt.day}, ${dt.year}';
   836→  }
   837→
   838→  /// Formats a [DateTime] as a short time string for message bubbles.
   839→  String _formatMessageTime(DateTime dt) {
   840→    final hour = dt.hour.toString().padLeft(2, '0');
   841→    final minute = dt.minute.toString().padLeft(2, '0');
   842→    return '$hour:$minute';
   843→  }
   844→
   845→  /// Formats a [DateTime] as a relative time-ago string.
   846→  String _formatTimeAgo(DateTime dt) {
   847→    final now = DateTime.now();
   848→    final diff = now.difference(dt);
   849→
   850→    if (diff.inMinutes < 1) return 'Just now';
   851→    if (diff.inMinutes < 60) return '${diff.inMinutes}m ago';
   852→    if (diff.inHours < 24) return '${diff.inHours}h ago';
   853→    if (diff.inDays < 7) return '${diff.inDays}d ago';
   854→    return _formatDate(dt);
   855→  }
