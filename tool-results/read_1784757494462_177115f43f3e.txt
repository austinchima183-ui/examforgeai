    31→
    32→String _roleLabel(String role) {
    33→  for (final entry in _kRoleOptions) {
    34→    if (entry.key == role) return entry.value;
    35→  }
    36→  return role;
    37→}
    38→
    39→Color _roleColor(String role) {
    40→  switch (role) {
    41→    case 'super_admin':
    42→      return AppColors.error;
    43→    case 'school_admin':
    44→      return AppColors.info;
    45→    case 'teacher':
    46→      return AppColors.success;
    47→    case 'student':
    48→      return AppColors.seed;
    49→    case 'parent':
    50→      return AppColors.warning;
    51→    default:
    52→      return Colors.grey;
    53→  }
    54→}

    56→IconData _roleIcon(String role) {
    57→  switch (role) {
    58→    case 'super_admin':
    59→      return Icons.admin_panel_settings_outlined;
    60→    case 'school_admin':
    61→      return Icons.domain_outlined;
    62→    case 'teacher':
    63→      return Icons.school_outlined;
    64→    case 'student':
    65→      return Icons.person_outline_rounded;
    66→    case 'parent':

    73→String _formatLastLogin(DateTime? lastLoginAt) {
    74→  if (lastLoginAt == null) return 'Never';
    75→  final now = DateTime.now();
    76→  final diff = now.difference(lastLoginAt);
    77→  if (diff.inMinutes < 1) return 'Just now';
    78→  if (diff.inHours < 1) return '${diff.inMinutes}m ago';
    79→  if (diff.inDays < 1) return '${diff.inHours}h ago';
    80→  if (diff.inDays < 30) return '${diff.inDays}d ago';
    81→  return '${lastLoginAt.day}/${lastLoginAt.month}/${lastLoginAt.year}';
    82→}

   643→    return Card(

   645→      shape: RoundedRectangleBorder(borderRadius: Spacings.borderRadiusMd),
   646→      child: Padding(
