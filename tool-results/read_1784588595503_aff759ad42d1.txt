   183→  @override
   184→  Future<QuestionModel> createQuestion(
   185→    Map<String, dynamic> questionData,
   186→  ) async {
   187→    try {
   188→      final response =
   189→          await _supabase.from(_questionsTable).insert(questionData).select();
   190→
   191→      if (response.isEmpty) {
   192→        throw const ServerException(
   193→          message: 'Question creation returned no data.',
   194→          statusCode: 500,
   195→        );
   196→      }
   197→
   198→      AppLogger.info('Question created: ${response.first['id']}');
   199→      return QuestionModel.fromJson(response.first);
   200→    } on sb.PostgrestException catch (e) {
   201→      throw _mapPostgrestException(e);
   202→    } on ServerException {
   203→      rethrow;
   204→    } catch (e) {
   205→      AppLogger.error('Unexpected createQuestion error', error: e);
   206→      throw const ServerException(
   207→        message: 'Failed to create question.',
   208→        statusCode: 500,
   209→      );
   210→    }
   211→  }

   293→  @override
   294→  Future<Map<String, dynamic>> getQuestionWithDetails(
   295→    String questionId,
   296→  ) async {
   297→    try {
   298→      // Fetch the base question
   299→      final questionResponse = await _supabase
   300→          .from(_questionsTable)
   301→          .select()
   302→          .eq('id', questionId)
   303→          .limit(1);
   304→
   305→      if (questionResponse.isEmpty) {
   306→        throw const NotFoundException('Question not found.');
   307→      }
   308→
   309→      final questionJson = Map<String, dynamic>.from(questionResponse.first);
   310→
   311→      // Fetch related data in parallel
   312→      final results = await Future.wait([
   313→        _supabase
   314→            .from(_answerOptionsTable)
   315→            .select()
   316→            .eq('question_id', questionId)
   317→            .order('sort_order'),
   318→        _supabase
   319→            .from(_matchingPairsTable)
   320→            .select()
   321→            .eq('question_id', questionId)
   322→            .order('sort_order'),
   323→        _supabase
   324→            .from(_orderingItemsTable)
   325→            .select()
   326→            .eq('question_id', questionId)
   327→            .order('correct_position'),
   328→        _supabase
   329→            .from(_fillInBlankTable)
   330→            .select()
   331→            .eq('question_id', questionId)
   332→            .order('blank_index'),
   333→        _supabase
   334→            .from(_attachmentsTable)
   335→            .select()
   336→            .eq('question_id', questionId)
   337→            .order('sort_order'),
   338→        // Tags need a join through the relation table
   339→        _supabase
   340→            .from(_tagRelationsTable)
   341→            .select('question_tags(*)')
   342→            .eq('question_id', questionId),
   343→      ]);
   344→
   345→      questionJson['answer_options'] = results[0] as List;
   346→      questionJson['matching_pairs'] = results[1] as List;
   347→      questionJson['ordering_items'] = results[2] as List;
   348→      questionJson['fill_in_blank_answers'] = results[3] as List;
   349→      questionJson['question_attachments'] = results[4] as List;
   350→
   351→      // Flatten tag join results
   352→      final tagRows = results[5] as List;
   353→      final tagList = tagRows
   354→          .map((row) => row['question_tags'] as Map<String, dynamic>?)
   355→          .whereType<Map<String, dynamic>>()
   356→          .toList();
   357→      questionJson['question_tags'] = tagList;
   358→
   359→      return questionJson;
   360→    } on sb.PostgrestException catch (e) {
   361→      throw _mapPostgrestException(e);
   362→    } on NotFoundException {
   363→      rethrow;
   364→    } catch (e) {
   365→      AppLogger.error('Unexpected getQuestionWithDetails error', error: e);
   366→      throw const ServerException(
   367→        message: 'Failed to retrieve question details.',
   368→        statusCode: 500,
   369→      );
   370→    }
   371→  }

   374→  Future<List<QuestionModel>> getQuestions(
   375→    Map<String, dynamic> filters,
   376→  ) async {
   377→    try {
   378→      var query = _supabase.from(_questionsTable).select();
   379→
   380→      query = _applyFilters(query, filters);
   381→      query = _applySorting(query, filters['sort_by'] as String?);
   382→      query = _applyPagination(
   383→        query,
   384→        page: filters['page'] as int?,
   385→        perPage: filters['per_page'] as int?,
   386→      );
   387→
   388→      final response = await query;
   389→      return response
   390→          .map((json) => QuestionModel.fromJson(json as Map<String, dynamic>))
   391→          .toList();
   392→    } on sb.PostgrestException catch (e) {
   393→      throw _mapPostgrestException(e);
   394→    } catch (e) {
   395→      AppLogger.error('Unexpected getQuestions error', error: e);
   396→      throw const ServerException(

   500→  @override
   501→  Future<QuestionModel> duplicateQuestion(String questionId) async {
   502→    try {
   503→      // Fetch the original question
   504→      final original = await getQuestion(questionId);
   505→
   506→      // Prepare the duplicate data (strip ID, reset version)
   507→      final duplicateData = original.toJson()
   508→        ..remove('id')
   509→        ..remove('created_at')
   510→        ..remove('updated_at')
   511→        ..['version'] = 1
   512→        ..['parent_id'] = original.id
   513→        ..['is_published'] = false
   514→        ..['is_archived'] = false
   515→        ..['usage_count'] = 0
   516→        ..['avg_score'] = 0.0;
   517→
   518→      return createQuestion(duplicateData);
   519→    } on sb.PostgrestException catch (e) {
   520→      throw _mapPostgrestException(e);
   521→    } on ServerException {
   522→      rethrow;
   523→    } on NotFoundException {
   524→      rethrow;
   525→    } catch (e) {
   526→      AppLogger.error('Unexpected duplicateQuestion error', error: e);
   527→      throw const ServerException(
   528→        message: 'Failed to duplicate question.',
   529→        statusCode: 500,
   530→      );
   531→    }

   551→      for (final id in questionIds) {
   552→        await _supabase
   553→            .from(_questionsTable)

   556→      }
   557→
   558→      AppLogger.info('Moved ${questionIds.length} questions');

   600→  @override
   601→  Future<List<AnswerOptionModel>> updateAnswerOptions(
   602→    String questionId,
   603→    List<Map<String, dynamic>> options,
   604→  ) async {
   605→    try {
   606→      // Delete existing options
   607→      await _supabase
   608→          .from(_answerOptionsTable)
   609→          .delete()
   610→          .eq('question_id', questionId);
   611→
   612→      // Insert new options
   613→      if (options.isNotEmpty) {
   614→        final optionsData = options
   615→            .map((o) => {
   616→                  ...o,
   617→                  'question_id': questionId,
   618→                })
   619→            .toList();
   620→
   621→        await _supabase
   622→            .from(_answerOptionsTable)
   623→            .insert(optionsData);
   624→      }
   625→
   626→      // Fetch the newly inserted options
   627→      return getAnswerOptions(questionId);
   628→    } on sb.PostgrestException catch (e) {
   629→      throw _mapPostgrestException(e);
   630→    } catch (e) {
   631→      AppLogger.error('Unexpected updateAnswerOptions error', error: e);
   632→      throw const ServerException(
   633→        message: 'Failed to update answer options.',
   634→        statusCode: 500,
   635→      );
   636→    }
   637→  }

   770→      // Check if already favorited

   773→          .select('id')

   775→          .eq('question_id', questionId);
   776→

   779→        await _supabase
   780→            .from(_favoritesTable)

   785→      } else {

   788→          'user_id': userId,
   789→          'question_id': questionId,
   790→        });
   791→        AppLogger.info('Added favorite: $questionId');
   792→      }
   793→    } on sb.PostgrestException catch (e) {
   794→      throw _mapPostgrestException(e);

   797→    } catch (e) {

   799→      throw const ServerException(
   800→        message: 'Failed to toggle favorite.',
   801→        statusCode: 500,

   803→    }

  1331→  @override
  1332→  Future<QuestionBankStatsModel> getStats({String? schoolId}) async {
  1333→    try {
  1334→      // Base query builder
  1335→      var baseQuery = _supabase.from(_questionsTable).select();
  1336→
  1337→      if (schoolId != null) {
  1338→        baseQuery = baseQuery.eq('school_id', schoolId);
  1339→      }
  1340→
  1341→      // Fetch all questions for stats computation
  1342→      final allQuestions = await baseQuery;
  1343→
  1344→      final totalQuestions = allQuestions.length;
  1345→      final publishedQuestions = allQuestions
  1346→          .where((q) => q['is_published'] == true)
  1347→          .length;
  1348→      final draftQuestions = allQuestions
  1349→          .where((q) => q['is_published'] == false && q['is_archived'] == false)
  1350→          .length;
  1351→      final archivedQuestions = allQuestions
  1352→          .where((q) => q['is_archived'] == true)
  1353→          .length;
  1354→
  1355→      // Compute breakdowns
  1356→      final Map<String, int> bySubject = {};
  1357→      final Map<String, int> byDifficulty = {};
  1358→      final Map<String, int> byType = {};
  1359→      final Map<String, int> byExamType = {};
  1360→
  1361→      for (final q in allQuestions) {
  1362→        final subjectId = q['subject_id'] as String?;
  1363→        if (subjectId != null) {
  1364→          bySubject[subjectId] = (bySubject[subjectId] ?? 0) + 1;
  1365→        }
  1366→
  1367→        final diff = q['difficulty'] as String?;
  1368→        if (diff != null) {
  1369→          byDifficulty[diff] = (byDifficulty[diff] ?? 0) + 1;
  1370→        }
  1371→
  1372→        final type = q['question_type'] as String?;
  1373→        if (type != null) {
  1374→          byType[type] = (byType[type] ?? 0) + 1;
  1375→        }
  1376→
  1377→        final exam = q['exam_type'] as String?;
  1378→        if (exam != null) {
  1379→          byExamType[exam] = (byExamType[exam] ?? 0) + 1;
  1380→        }
  1381→      }
  1382→
  1383→      // Recent questions (last 7 days)
  1384→      final weekAgo = DateTime.now().subtract(const Duration(days: 7));
  1385→      final recentQuestions = allQuestions
  1386→          .where((q) {
  1387→            final createdAt = q['created_at'] as String?;
  1388→            if (createdAt == null) return false;
  1389→            return DateTime.parse(createdAt).isAfter(weekAgo);
  1390→          })
  1391→          .length;
  1392→
  1393→      // Collection count
  1394→      var collectionQuery =
  1395→          _supabase.from(_collectionsTable).select('id');
  1396→      if (schoolId != null) {
  1397→        collectionQuery = collectionQuery.eq('school_id', schoolId);
  1398→      }
  1399→      final collections = await collectionQuery;
  1400→
  1401→      // Favorites count for current user
  1402→      final userId = _currentUserId;
  1403→      int totalFavorites = 0;
  1404→      if (userId != null) {
  1405→        final favorites = await _supabase
  1406→            .from(_favoritesTable)
  1407→            .select('id')
  1408→            .eq('user_id', userId);
  1409→        totalFavorites = favorites.length;
  1410→      }
  1411→
  1412→      // Most used questions (top 5)
  1413→      final sortedByUsage = List<Map<String, dynamic>>.from(allQuestions)
  1414→        ..sort((a, b) => (b['usage_count'] as int? ?? 0)
  1415→            .compareTo(a['usage_count'] as int? ?? 0));
  1416→      final mostUsed = sortedByUsage
  1417→          .take(5)
  1418→          .map((q) => QuestionModel.fromJson(q))
  1419→          .toList();
  1420→
  1421→      return QuestionBankStatsModel(
  1422→        totalQuestions: totalQuestions,
  1423→        publishedQuestions: publishedQuestions,
  1424→        draftQuestions: draftQuestions,
  1425→        archivedQuestions: archivedQuestions,
  1426→        questionsBySubject: bySubject,
  1427→        questionsByDifficulty: byDifficulty,
  1428→        questionsByType: byType,
  1429→        questionsByExamType: byExamType,
  1430→        recentQuestions: recentQuestions,
  1431→        totalCollections: collections.length,
  1432→        totalFavorites: totalFavorites,
  1433→        mostUsedQuestions: mostUsed,
  1434→      );
  1435→    } on sb.PostgrestException catch (e) {
  1436→      throw _mapPostgrestException(e);
  1437→    } catch (e) {
  1438→      AppLogger.error('Unexpected getStats error', error: e);
  1439→      throw const ServerException(
  1440→        message: 'Failed to retrieve question bank stats.',
  1441→        statusCode: 500,
  1442→      );
  1443→    }
  1444→  }

  1593→    String questionId,
  1594→    int version,
  1595→  ) async {

  1597→      // Fetch the version snapshot
  1598→      final versionResponse = await _supabase

  1600→          .select()
  1601→          .eq('question_id', questionId)
  1602→          .eq('version', version)
  1603→          .limit(1);
  1604→
  1605→      if (versionResponse.isEmpty) {
  1606→        throw NotFoundException(
  1607→          'Version $version not found for question $questionId.',
  1608→        );
  1609→      }
  1610→
  1611→      final versionData = versionResponse.first;
  1612→      final snapshot = versionData['snapshot'] as Map<String, dynamic>?;
  1613→
  1614→      if (snapshot == null) {
  1615→        throw const ServerException(
  1616→          message: 'Version snapshot is empty.',
  1617→          statusCode: 500,
  1618→        );
  1619→      }
  1620→
  1621→      // Update the question with the snapshot data
  1622→      final restoredData = Map<String, dynamic>.from(snapshot)
  1623→        ..remove('id')
  1624→        ..remove('created_at')
  1625→        ..['updated_at'] = DateTime.now().toIso8601String()
  1626→        ..['version'] = (await getQuestion(questionId)).version + 1;
  1627→
  1628→      return updateQuestion(questionId, restoredData);
  1629→    } on sb.PostgrestException catch (e) {
  1630→      throw _mapPostgrestException(e);
  1631→    } on ServerException {
  1632→      rethrow;
  1633→    } on NotFoundException {
  1634→      rethrow;
  1635→    } catch (e) {
  1636→      AppLogger.error('Unexpected restoreVersion error', error: e);
  1637→      throw const ServerException(
  1638→        message: 'Failed to restore question version.',
  1639→        statusCode: 500,
  1640→      );
  1641→    }
  1642→  }

  1707→  /// Applies sorting based on the [sortBy] value.
  1708→  sb.PostgrestTransformBuilder<dynamic> _applySorting(
  1709→    sb.PostgrestTransformBuilder<dynamic> query,
  1710→    String? sortBy,
  1711→  ) {
  1712→    switch (sortBy) {
  1713→      case 'oldest':
  1714→        return query.order('created_at', ascending: true);
  1715→      case 'most_used':
  1716→        return query.order('usage_count', ascending: false);
  1717→      case 'least_used':
  1718→        return query.order('usage_count', ascending: true);
  1719→      case 'highest_rated':
  1720→        return query.order('avg_score', ascending: false);
  1721→      case 'a_z':
  1722→        return query.order('content', ascending: true);
  1723→      case 'z_a':
  1724→        return query.order('content', ascending: false);
  1725→      case 'newest':
  1726→      default:
  1727→        return query.order('created_at', ascending: false);
  1728→    }
  1729→  }

  1731→  /// Applies pagination using Supabase range.
  1732→  sb.PostgrestTransformBuilder<dynamic> _applyPagination(
  1733→    sb.PostgrestTransformBuilder<dynamic> query, {
  1734→    int? page,
  1735→    int? perPage,
  1736→  }) {
  1737→    final p = page ?? 1;
  1738→    final pp = perPage ?? 20;
  1739→    final from = (p - 1) * pp;
  1740→    final to = from + pp - 1;
  1741→    return query.range(from, to);
  1742→  }

  1744→  /// Maps a Supabase [sb.PostgrestException] to a domain exception.
  1745→  Exception _mapPostgrestException(sb.PostgrestException e) {
  1746→    final statusCode = e.code != null ? int.tryParse(e.code!) ?? 0 : 0;
  1747→    final message = e.message ?? 'An unexpected database error occurred.';
  1748→
  1749→    AppLogger.warning(
  1750→      'Supabase PostgrestException — code: ${e.code}, message: $message',
  1751→    );
  1752→
  1753→    switch (statusCode) {
  1754→      case 401:
  1755→        return UnauthorizedException(message);
  1756→      case 403:
  1757→        return ForbiddenException(message);
  1758→      case 404:
  1759→        return NotFoundException(message);
  1760→      case 422:
  1761→        return ValidationException(
  1762→          message: message,
  1763→          fieldErrors: e.details is Map<String, dynamic>
  1764→              ? e.details as Map<String, String>
  1765→              : {},
  1766→        );
  1767→      default:
  1768→        return ServerException(
  1769→          message: message,
  1770→          statusCode: statusCode,
  1771→          data: e.details,
  1772→        );
  1773→    }
  1774→  }
