   172→-- 2. HELPER FUNCTION: set_updated_at
   173→-- ============================================================================
   174→-- Reusable trigger function that sets updated_at = now() on any row update.
   175→-- Applied to every table with an updated_at column.
   176→-- ============================================================================
   177→
   178→CREATE OR REPLACE FUNCTION set_updated_at()
   179→RETURNS TRIGGER AS $$
   180→BEGIN
   181→  NEW.updated_at = now();
   182→  RETURN NEW;
   183→END;

   190→-- 3. HELPER FUNCTION: is_super_admin
   191→-- ============================================================================
   192→-- Checks whether the currently authenticated user has the super_admin role.
   193→-- Used throughout RLS policies to grant full access.
   194→-- ============================================================================
   195→
   196→CREATE OR REPLACE FUNCTION is_super_admin()
   197→RETURNS BOOLEAN AS $$
   198→  SELECT EXISTS (
   199→    SELECT 1 FROM users
   200→    WHERE id = auth.uid()
   201→      AND role = 'super_admin'
   202→      AND is_active = true
   203→  );
   204→$$ LANGUAGE sql STABLE SECURITY DEFINER;

   371→-- ============================================================================
   372→-- 9. TABLE: sync_queue
   373→-- ============================================================================
   374→-- Queue of pending mutations created while offline.
   375→-- The client enqueues operations when offline; the sync engine processes
   376→-- them when connectivity is restored, respecting priority ordering.
   377→-- ============================================================================
   378→
   379→CREATE TABLE IF NOT EXISTS sync_queue (
   380→  id                    UUID PRIMARY KEY DEFAULT gen_random_uuid(),
   381→  user_id               UUID NOT NULL REFERENCES users(id) ON DELETE CASCADE,
   382→  table_name            TEXT NOT NULL,                        -- Target table for the mutation
   383→  record_id             TEXT,                                 -- Primary key of the affected record
   384→  operation             sync_operation_type NOT NULL,         -- insert / update / delete
   385→  payload               JSONB NOT NULL,                       -- The mutation data (full row or diff)
   386→  priority              INTEGER DEFAULT 5,                    -- 1 = critical, 5 = normal, 10 = low
   387→  attempts              INTEGER DEFAULT 0,                    -- Number of processing attempts
   388→  max_attempts          INTEGER DEFAULT 5,                    -- Max retries before marking dead
   389→  last_attempt_at       TIMESTAMPTZ,                          -- Timestamp of most recent attempt
   390→  next_retry_at         TIMESTAMPTZ DEFAULT now(),            -- When to retry next (exponential backoff)
   391→  status                sync_queue_status DEFAULT 'pending',
   392→  error_message         TEXT,                                 -- Last error encountered during processing
   393→  created_at            TIMESTAMPTZ DEFAULT now(),
   394→  updated_at            TIMESTAMPTZ DEFAULT now(),
   395→
   396→  -- Validate priority range
   397→  CONSTRAINT chk_sync_queue_priority CHECK (priority >= 1 AND priority <= 10),
   398→  -- Validate attempts are non-negative
   399→  CONSTRAINT chk_sync_queue_attempts CHECK (attempts >= 0),
   400→  -- Prevent excessive retry configurations
   401→  CONSTRAINT chk_sync_queue_max_attempts CHECK (max_attempts >= 1 AND max_attempts <= 20)
   402→);
   403→
   404→COMMENT ON TABLE sync_queue IS
   405→  'Pending mutation queue for offline-first sync; processed in priority order when online';
   406→COMMENT ON COLUMN sync_queue.priority IS 'Processing priority: 1 = critical (exam submissions), 5 = normal, 10 = low (analytics)';
   407→COMMENT ON COLUMN sync_queue.payload IS 'Full row data for inserts, diff for updates, or empty for deletes';
   408→COMMENT ON COLUMN sync_queue.next_retry_at IS 'Earliest time to retry; supports exponential backoff strategies';
   409→
   410→-- Indexes for sync_queue (optimized for the processing query pattern)
   411→CREATE INDEX IF NOT EXISTS idx_sync_queue_user_id
   412→  ON sync_queue (user_id);
   413→CREATE INDEX IF NOT EXISTS idx_sync_queue_status
   414→  ON sync_queue (status);
   415→CREATE INDEX IF NOT EXISTS idx_sync_queue_priority
   416→  ON sync_queue (priority ASC);
   417→CREATE INDEX IF NOT EXISTS idx_sync_queue_next_retry
   418→  ON sync_queue (next_retry_at);
   419→-- Composite index for the primary processing query: find pending items for a user
   420→CREATE INDEX IF NOT EXISTS idx_sync_queue_process
   421→  ON sync_queue (user_id, status, priority ASC, next_retry_at)
   422→  WHERE status IN ('pending', 'failed');
   423→-- GIN index for JSONB payload queries
   424→CREATE INDEX IF NOT EXISTS idx_sync_queue_payload
   425→  ON sync_queue USING GIN (payload jsonb_path_ops);
   426→
   427→-- Trigger: auto-update updated_at
   428→DROP TRIGGER IF EXISTS trg_sync_queue_updated_at ON sync_queue;
   429→CREATE TRIGGER trg_sync_queue_updated_at
   430→  BEFORE UPDATE ON sync_queue
   431→  FOR EACH ROW EXECUTE FUNCTION set_updated_at();

   434→-- 10. TABLE: sync_conflicts
   435→-- ============================================================================
   436→-- Detected sync conflicts requiring resolution.
   437→-- When both client and server modify the same record, a conflict is logged
   438→-- here for manual or automated resolution.
   439→-- ============================================================================
   440→
   441→CREATE TABLE IF NOT EXISTS sync_conflicts (
   442→  id                    UUID PRIMARY KEY DEFAULT gen_random_uuid(),
   443→  user_id               UUID NOT NULL REFERENCES users(id) ON DELETE CASCADE,
   444→  table_name            TEXT NOT NULL,
   445→  record_id             TEXT NOT NULL,
   446→  local_data            JSONB NOT NULL,                       -- Client-side version of the record
   447→  server_data           JSONB NOT NULL,                       -- Server-side version of the record
   448→  resolution            conflict_resolution DEFAULT 'pending',
   449→  resolved_data         JSONB,                                -- Merged/selected data after resolution
   450→  resolved_by           UUID REFERENCES users(id) ON DELETE SET NULL,
   451→  resolved_at           TIMESTAMPTZ,
   452→  created_at            TIMESTAMPTZ DEFAULT now(),
   453→
   454→  -- Ensure resolution is set when resolved_data is provided
   455→  CONSTRAINT chk_sync_conflicts_resolution CHECK (
   456→    resolution = 'pending' OR resolved_at IS NOT NULL
   457→  )
   458→);
   459→
   460→COMMENT ON TABLE sync_conflicts IS
   461→  'Detected sync conflicts between local and server data, requiring resolution';
   462→COMMENT ON COLUMN sync_conflicts.local_data IS 'The client-side version of the conflicting record';
   463→COMMENT ON COLUMN sync_conflicts.server_data IS 'The server-side version of the conflicting record';
   464→COMMENT ON COLUMN sync_conflicts.resolution IS 'How the conflict was resolved: pending, local_wins, server_wins, merge, or manual';
   465→COMMENT ON COLUMN sync_conflicts.resolved_data IS 'The final merged/selected data after resolution';
   466→
   467→-- Indexes for sync_conflicts
   468→CREATE INDEX IF NOT EXISTS idx_sync_conflicts_user_id
   469→  ON sync_conflicts (user_id);
   470→CREATE INDEX IF NOT EXISTS idx_sync_conflicts_resolution
   471→  ON sync_conflicts (resolution);
   472→CREATE INDEX IF NOT EXISTS idx_sync_conflicts_pending
   473→  ON sync_conflicts (user_id, created_at DESC)
   474→  WHERE resolution = 'pending';
   475→-- GIN indexes for JSONB conflict data
   476→CREATE INDEX IF NOT EXISTS idx_sync_conflicts_local_data
   477→  ON sync_conflicts USING GIN (local_data jsonb_path_ops);
   478→CREATE INDEX IF NOT EXISTS idx_sync_conflicts_server_data
   479→  ON sync_conflicts USING GIN (server_data jsonb_path_ops);

   536→-- ============================================================================
   537→-- Track locally cached resources and their validity.
   538→-- Enables the client to determine which cached resources are stale and
   539→-- need refreshing, and supports cache eviction based on expiry and size.
   540→-- ============================================================================
   541→
   542→CREATE TABLE IF NOT EXISTS offline_cache_metadata (
   543→  id                    UUID PRIMARY KEY DEFAULT gen_random_uuid(),
   544→  user_id               UUID NOT NULL REFERENCES users(id) ON DELETE CASCADE,
   545→  resource_type         TEXT NOT NULL,                        -- e.g. "exam", "question_bank", "media"
   546→  resource_id           TEXT NOT NULL,                        -- PK of the cached resource
   547→  cache_key             TEXT NOT NULL,                        -- Unique cache identifier (for lookup)
   548→  version               INTEGER DEFAULT 1,                   -- Cache version for invalidation
   549→  expires_at            TIMESTAMPTZ,                          -- When this cache entry becomes stale
   550→  file_size_bytes       INTEGER,                              -- Size of the cached resource
   551→  checksum              TEXT,                                 -- Integrity hash of the cached content
   552→  is_encrypted          BOOLEAN DEFAULT false,                -- Whether the cached data is encrypted
   553→  access_count          INTEGER DEFAULT 0,                    -- Number of times accessed (for LRU eviction)
   554→  last_accessed_at      TIMESTAMPTZ DEFAULT now(),
   555→  created_at            TIMESTAMPTZ DEFAULT now(),
   556→  updated_at            TIMESTAMPTZ DEFAULT now(),
   557→
   558→  -- Each user can only have one cache entry per key
   559→  CONSTRAINT uq_offline_cache_user_key UNIQUE (user_id, cache_key),
   560→  -- Validate non-negative values
   561→  CONSTRAINT chk_offline_cache_version CHECK (version >= 1),
   562→  CONSTRAINT chk_offline_cache_access_count CHECK (access_count >= 0),
   563→  CONSTRAINT chk_offline_cache_file_size CHECK (file_size_bytes IS NULL OR file_size_bytes >= 0)
   564→);
   565→
   566→COMMENT ON TABLE offline_cache_metadata IS
   567→  'Locally cached resource metadata for offline access, with expiry and integrity tracking';
   568→COMMENT ON COLUMN offline_cache_metadata.cache_key IS 'Unique cache key combining resource type and ID for client-side lookup';
   569→COMMENT ON COLUMN offline_cache_metadata.expires_at IS 'When this cache entry becomes stale and should be refreshed';
   570→COMMENT ON COLUMN offline_cache_metadata.is_encrypted IS 'Whether the cached content is encrypted at rest on the device';
   571→
   572→-- Indexes for offline_cache_metadata
   573→CREATE INDEX IF NOT EXISTS idx_offline_cache_user_id
   574→  ON offline_cache_metadata (user_id);
   575→CREATE INDEX IF NOT EXISTS idx_offline_cache_resource_type
   576→  ON offline_cache_metadata (resource_type);
   577→CREATE INDEX IF NOT EXISTS idx_offline_cache_expires_at
   578→  ON offline_cache_metadata (expires_at) WHERE expires_at IS NOT NULL;
   579→CREATE INDEX IF NOT EXISTS idx_offline_cache_last_accessed
   580→  ON offline_cache_metadata (user_id, last_accessed_at DESC);

  1299→-- ---------------------------------------------------------------------------
  1300→-- register_device(): Upsert a device registration for push notifications
  1301→-- ---------------------------------------------------------------------------
  1302→-- Called when a user opens the app on a device. If the device_token already
  1303→-- exists for this user, it updates the existing record; otherwise it inserts.
  1304→-- Also deactivates stale devices (not seen in 90 days).
  1305→-- ---------------------------------------------------------------------------
  1306→
  1307→CREATE OR REPLACE FUNCTION register_device(
  1308→  p_user_id       UUID,
  1309→  p_device_token  TEXT,
  1310→  p_platform      mobile_platform,
  1311→  p_device_name   TEXT DEFAULT NULL,
  1312→  p_device_model  TEXT DEFAULT NULL,
  1313→  p_os_version    TEXT DEFAULT NULL,
  1314→  p_app_version   TEXT DEFAULT NULL,
  1315→  p_push_enabled  BOOLEAN DEFAULT true,
  1316→  p_topics        TEXT[] DEFAULT '{}'
  1317→)
  1318→RETURNS UUID AS $$
  1319→DECLARE
  1320→  v_device_id UUID;
  1321→BEGIN
  1322→  -- Upsert the device registration
  1323→  INSERT INTO device_registrations (
  1324→    user_id, device_token, platform, device_name, device_model,
  1325→    os_version, app_version, push_enabled, push_notification_topics,
  1326→    last_active_at
  1327→  ) VALUES (
  1328→    p_user_id, p_device_token, p_platform, p_device_name, p_device_model,
  1329→    p_os_version, p_app_version, p_push_enabled, p_topics, now()
  1330→  )
  1331→  ON CONFLICT (user_id, device_token) DO UPDATE SET
  1332→    platform              = EXCLUDED.platform,
  1333→    device_name           = COALESCE(EXCLUDED.device_name, device_registrations.device_name),
  1334→    device_model          = COALESCE(EXCLUDED.device_model, device_registrations.device_model),
  1335→    os_version            = COALESCE(EXCLUDED.os_version, device_registrations.os_version),
  1336→    app_version           = COALESCE(EXCLUDED.app_version, device_registrations.app_version),
  1337→    push_enabled          = EXCLUDED.push_enabled,
  1338→    push_notification_topics = EXCLUDED.push_notification_topics,
  1339→    is_active             = true,
  1340→    last_active_at        = now()
  1341→  RETURNING id INTO v_device_id;
  1342→
  1343→  -- Also upsert the push notification token
  1344→  INSERT INTO push_notification_tokens (user_id, token, platform, is_active)
  1345→  VALUES (p_user_id, p_device_token, p_platform, true)
  1346→  ON CONFLICT (token) DO UPDATE SET
  1347→    user_id   = EXCLUDED.user_id,
  1348→    platform  = EXCLUDED.platform,
  1349→    is_active = true;
  1350→
  1351→  -- Deactivate devices that haven't been seen in 90 days (for this user)
  1352→  UPDATE device_registrations
  1353→  SET is_active = false
  1354→  WHERE user_id = p_user_id
  1355→    AND is_active = true
  1356→    AND last_active_at < now() - INTERVAL '90 days'
  1357→    AND id != v_device_id;
  1358→
  1359→  RETURN v_device_id;
  1360→END;
  1361→$$ LANGUAGE plpgsql SECURITY DEFINER;
  1362→
  1363→COMMENT ON FUNCTION register_device(UUID, TEXT, mobile_platform, TEXT, TEXT, TEXT, TEXT, BOOLEAN, TEXT[]) IS
  1364→  'Upsert a device registration for a user, sync the push token, and deactivate stale devices (90+ days)';
  1365→
  1366→-- ---------------------------------------------------------------------------
  1367→-- record_sync_event(): Log a sync operation for monitoring
  1368→-- ---------------------------------------------------------------------------
  1369→-- Creates a sync_logs entry and updates the corresponding sync_metadata
  1370→-- for each table that was synced.
  1371→-- ---------------------------------------------------------------------------
  1372→
  1373→CREATE OR REPLACE FUNCTION record_sync_event(
  1374→  p_user_id             UUID,
  1375→  p_device_id           TEXT DEFAULT NULL,
  1376→  p_sync_type           sync_log_type,
  1377→  p_tables_synced       TEXT[] DEFAULT '{}',
  1378→  p_records_pushed      INTEGER DEFAULT 0,
  1379→  p_records_pulled      INTEGER DEFAULT 0,
  1380→  p_conflicts_count     INTEGER DEFAULT 0,
  1381→  p_errors_count        INTEGER DEFAULT 0,
  1382→  p_duration_ms         INTEGER DEFAULT NULL,
  1383→  p_connection_quality  connection_quality DEFAULT NULL,
  1384→  p_status              sync_log_status,
  1385→  p_error_details       JSONB DEFAULT NULL,
  1386→  p_started_at          TIMESTAMPTZ DEFAULT now(),
  1387→  p_completed_at        TIMESTAMPTZ DEFAULT NULL
  1388→)
  1389→RETURNS UUID AS $$
  1390→DECLARE
  1391→  v_log_id UUID;
  1392→  v_table_name TEXT;
  1393→BEGIN
  1394→  -- Insert the sync log entry
  1395→  INSERT INTO sync_logs (
  1396→    user_id, device_id, sync_type, tables_synced, records_pushed,
  1397→    records_pulled, conflicts_count, errors_count, duration_ms,
  1398→    connection_quality, status, error_details, started_at, completed_at
  1399→  ) VALUES (
  1400→    p_user_id, p_device_id, p_sync_type, p_tables_synced, p_records_pushed,

  1402→    p_connection_quality, p_status, p_error_details, p_started_at,
  1403→    COALESCE(p_completed_at, now())
  1404→  ) RETURNING id INTO v_log_id;
  1405→
  1406→  -- Update sync_metadata for each synced table
  1407→  FOREACH v_table_name IN ARRAY p_tables_synced LOOP
  1408→    INSERT INTO sync_metadata (user_id, table_name, last_synced_at, is_full_sync)
  1409→    VALUES (p_user_id, v_table_name, now(), p_sync_type = 'full')
  1410→    ON CONFLICT (user_id, table_name) DO UPDATE SET
  1411→      last_synced_at = now(),
  1412→      is_full_sync   = CASE WHEN p_sync_type = 'full' THEN true ELSE sync_metadata.is_full_sync END;
  1413→  END LOOP;
  1414→
  1415→  RETURN v_log_id;
  1416→END;
  1417→$$ LANGUAGE plpgsql SECURITY DEFINER;

  1460→-- ---------------------------------------------------------------------------
  1461→-- process_sync_queue(): Process pending sync items for a user
  1462→-- ---------------------------------------------------------------------------
  1463→-- Marks the next batch of pending items as in_progress for processing.
  1464→-- Items are selected by priority (ASC) then next_retry_at (ASC).
  1465→-- Returns the selected item IDs for the caller to process.
  1466→-- ---------------------------------------------------------------------------
  1467→
  1468→CREATE OR REPLACE FUNCTION process_sync_queue(
  1469→  p_user_id  UUID,
  1470→  p_batch_size INTEGER DEFAULT 50
  1471→)
  1472→RETURNS TABLE (
  1473→  id          UUID,
  1474→  table_name  TEXT,
  1475→  record_id   TEXT,
  1476→  operation   sync_operation_type,
  1477→  payload     JSONB,
  1478→  priority    INTEGER,
  1479→  attempts    INTEGER
  1480→) AS $$
  1481→BEGIN
  1482→  -- Atomically select and lock the next batch of pending items
  1483→  RETURN QUERY
  1484→  UPDATE sync_queue
  1485→  SET status         = 'in_progress',
  1486→      last_attempt_at = now(),
  1487→      attempts        = attempts + 1,
  1488→      updated_at      = now()
  1489→  WHERE id IN (
  1490→    SELECT sq.id
  1491→    FROM sync_queue sq
  1492→    WHERE sq.user_id = p_user_id
  1493→      AND sq.status IN ('pending', 'failed')
  1494→      AND sq.next_retry_at <= now()
  1495→      AND sq.attempts < sq.max_attempts
  1496→    ORDER BY sq.priority ASC, sq.next_retry_at ASC
  1497→    LIMIT p_batch_size
  1498→    FOR UPDATE SKIP LOCKED
  1499→  )
  1500→  RETURNING sync_queue.id, sync_queue.table_name, sync_queue.record_id,
  1501→            sync_queue.operation, sync_queue.payload, sync_queue.priority,
  1502→            sync_queue.attempts;
  1503→
  1504→  -- Mark items that exceeded max_attempts as dead
  1505→  UPDATE sync_queue
  1506→  SET status = 'dead',
  1507→      updated_at = now()
  1508→  WHERE user_id = p_user_id
  1509→    AND status IN ('pending', 'failed')
  1510→    AND attempts >= max_attempts
  1511→    AND next_retry_at <= now();
  1512→END;
  1513→$$ LANGUAGE plpgsql SECURITY DEFINER;
  1514→
  1515→COMMENT ON FUNCTION process_sync_queue(UUID, INTEGER) IS
  1516→  'Claim the next batch of pending sync items for processing, ordered by priority and retry time';
  1517→
  1518→-- ---------------------------------------------------------------------------
  1519→-- complete_sync_queue_item(): Mark a sync queue item as completed or failed
  1520→-- ---------------------------------------------------------------------------
  1521→-- Helper to finalize processing of an item claimed by process_sync_queue.
  1522→-- ---------------------------------------------------------------------------
  1523→
  1524→CREATE OR REPLACE FUNCTION complete_sync_queue_item(
  1525→  p_id            UUID,
  1526→  p_success       BOOLEAN DEFAULT true,
  1527→  p_error_message TEXT DEFAULT NULL,
  1528→  p_retry_delay   INTERVAL DEFAULT INTERVAL '30 seconds'
  1529→)
  1530→RETURNS VOID AS $$
  1531→BEGIN
  1532→  IF p_success THEN
  1533→    UPDATE sync_queue

  1565→CREATE OR REPLACE FUNCTION record_connectivity_event(
  1566→  p_user_id                   UUID,
  1567→  p_device_id                 TEXT DEFAULT NULL,

  1744→-- ---------------------------------------------------------------------------
  1745→-- validate_offline_attempt(): Validate an offline exam attempt
  1746→-- ---------------------------------------------------------------------------
  1747→-- Validates an offline exam attempt before accepting it into the main
  1748→-- system. Checks:
  1749→--   1. The exam allows offline mode
  1750→--   2. The student hasn't exceeded max offline attempts
  1751→--   3. The integrity hash is valid
  1752→--   4. The time taken is reasonable (not negative, not exceeding exam duration)
  1753→--   5. The attempt data and answers are present and well-formed
  1754→-- Returns a JSONB with validation results.
  1755→-- ---------------------------------------------------------------------------
  1756→
  1757→CREATE OR REPLACE FUNCTION validate_offline_attempt(
  1758→  p_attempt_id UUID
  1759→)
  1760→RETURNS JSONB AS $$
  1761→DECLARE
  1762→  v_attempt         offline_exam_attempts%ROWTYPE;
  1763→  v_config          offline_exam_config%ROWTYPE;
  1764→  v_existing_count  INTEGER;
  1765→  v_errors          JSONB := '[]';
  1766→  v_is_valid        BOOLEAN := true;
  1767→  v_result          JSONB;
  1768→BEGIN
  1769→  -- Fetch the attempt
  1770→  SELECT * INTO v_attempt FROM offline_exam_attempts WHERE id = p_attempt_id;
  1771→  IF NOT FOUND THEN
  1772→    RETURN jsonb_build_object(
  1773→      'valid', false,
  1774→      'errors', jsonb_build_array('Attempt not found')
  1775→    );
  1776→  END IF;
  1777→
  1778→  -- Fetch the offline config for this exam
  1779→  SELECT * INTO v_config
  1780→  FROM offline_exam_config
  1781→  WHERE exam_id = v_attempt.exam_id
  1782→    AND school_id = v_attempt.school_id;
  1783→
  1784→  -- Validation 1: Exam allows offline mode
  1785→  IF NOT FOUND OR NOT v_config.allows_offline THEN
  1786→    v_errors := v_errors || jsonb_build_array(jsonb_build_object(
  1787→      'field', 'allows_offline',
  1788→      'message', 'This exam does not allow offline attempts'
  1789→    ));
  1790→    v_is_valid := false;
  1791→  END IF;
  1792→
  1793→  -- Validation 2: Max offline attempts not exceeded
  1794→  IF FOUND THEN
  1795→    SELECT COUNT(*) INTO v_existing_count
  1796→    FROM offline_exam_attempts
  1797→    WHERE exam_id = v_attempt.exam_id
  1798→      AND student_id = v_attempt.student_id
  1799→      AND sync_status IN ('synced', 'validated', 'pending')
  1800→      AND id != p_attempt_id;
  1801→
  1802→    IF v_existing_count >= v_config.max_offline_attempts THEN
  1803→      v_errors := v_errors || jsonb_build_array(jsonb_build_object(
  1804→        'field', 'max_offline_attempts',
  1805→        'message', format('Maximum offline attempts (%s) exceeded', v_config.max_offline_attempts)
  1806→      ));
  1807→      v_is_valid := false;
  1808→    END IF;
  1809→  END IF;
  1810→
  1811→  -- Validation 3: Answers are present
  1812→  IF v_attempt.answers IS NULL OR jsonb_typeof(v_attempt.answers) = 'null' THEN
  1813→    v_errors := v_errors || jsonb_build_array(jsonb_build_object(
  1814→      'field', 'answers',
  1815→      'message', 'Answers data is missing'
  1816→    ));
  1817→    v_is_valid := false;
  1818→  END IF;
  1819→
  1820→  -- Validation 4: Attempt data is present
  1821→  IF v_attempt.attempt_data IS NULL OR jsonb_typeof(v_attempt.attempt_data) = 'null' THEN
  1822→    v_errors := v_errors || jsonb_build_array(jsonb_build_object(
  1823→      'field', 'attempt_data',
  1824→      'message', 'Attempt metadata is missing'
  1825→    ));
  1826→    v_is_valid := false;
  1827→  END IF;
  1828→
  1829→  -- Validation 5: Time taken is reasonable
  1830→  IF v_attempt.time_taken_seconds IS NOT NULL AND v_attempt.time_taken_seconds < 0 THEN
  1831→    v_errors := v_errors || jsonb_build_array(jsonb_build_object(
  1832→      'field', 'time_taken_seconds',
  1833→      'message', 'Time taken cannot be negative'
  1834→    ));
  1835→    v_is_valid := false;
  1836→  END IF;
  1837→
  1838→  -- Validation 6: Completed at is after started at
  1839→  IF v_attempt.completed_at IS NOT NULL AND v_attempt.completed_at < v_attempt.started_at THEN
  1840→    v_errors := v_errors || jsonb_build_array(jsonb_build_object(
  1841→      'field', 'completed_at',
  1842→      'message', 'Completion time cannot be before start time'
  1843→    ));
  1844→    v_is_valid := false;
  1845→  END IF;
  1846→
  1847→  -- Update the attempt status
  1848→  IF v_is_valid THEN
  1849→    UPDATE offline_exam_attempts
  1850→    SET sync_status = 'validated',
  1851→        validation_errors = NULL
  1852→    WHERE id = p_attempt_id;
  1853→  ELSE
  1854→    UPDATE offline_exam_attempts
  1855→    SET sync_status = 'rejected',
  1856→        validation_errors = v_errors
  1857→    WHERE id = p_attempt_id;
  1858→  END IF;
  1859→
  1860→  -- Build result

  1862→    'valid', v_is_valid,
  1863→    'attempt_id', p_attempt_id,
  1864→    'exam_id', v_attempt.exam_id,
  1865→    'student_id', v_attempt.student_id,
  1866→    'errors', v_errors,
  1867→    'validated_at', now()
  1868→  );
  1869→
  1870→  RETURN v_result;
  1871→END;
