    85→  Never _mapPostgrestException(sb.PostgrestException e) {
    86→    AppLogger.error('Postgrest error: ${e.message}', error: e);
    87→    switch (e.code) {
    88→      case 'PGRST116':
    89→        throw NotFoundException(message: e.message);
    90→      case '23505':
    91→        throw ServerException(
    92→          message: 'A record with this data already exists.',
    93→          statusCode: 409,
    94→        );
    95→      case '23503':
    96→        throw ServerException(
    97→          message: 'Referenced record not found.',
    98→          statusCode: 404,
    99→        );
   100→      case '42501':
   101→        throw ForbiddenException(
   102→          message: 'You do not have permission for this action.',
   103→        );
   104→      default:
   105→        throw ServerException(
   106→          message: e.message,
   107→          statusCode: e.statusCode ?? 500,
   108→        );
   109→    }
   110→  }

   236→            check_out_time,

   239→          .eq('student_id', studentId)

   247→      }

   250→      AppLogger.info('Child attendance fetched successfully');
   251→      return {
   252→        'student_id': studentId,
   253→        'records': response,
   254→        'start_date': startDate?.toIso8601String(),
   255→        'end_date': endDate?.toIso8601String(),
   256→      };
   257→    } on sb.PostgrestException catch (e) {
   258→      _mapPostgrestException(e);

   343→  @override
   344→  Future<List<ParentMessageModel>> getMessages(
   345→    Map<String, dynamic> params,
   346→  ) async {
   347→    try {
   348→      AppLogger.info('Fetching parent messages with params: $params');
   349→      var query = _supabase
   350→          .from(_messagesTable)
   351→          .select()
   352→          .order('created_at', ascending: false);
   353→
   354→      // Apply filters from params
   355→      if (params.containsKey('thread_id')) {
   356→        query = query.eq('thread_id', params['thread_id'] as String);
   357→      }
   358→      if (params.containsKey('student_id')) {
   359→        query = query.eq('student_id', params['student_id'] as String);
   360→      }
   361→      if (params.containsKey('direction')) {
   362→        query = query.eq('direction', params['direction'] as String);
   363→      }
   364→      if (params.containsKey('is_archived')) {
   365→        query = query.eq('is_archived', params['is_archived'] as bool);
   366→      }
   367→      if (params.containsKey('page') && params.containsKey('per_page')) {
   368→        final page = params['page'] as int;
   369→        final perPage = params['per_page'] as int;
   370→        query = query.range((page - 1) * perPage, page * perPage - 1);
   371→      }
   372→
   373→      final response = await query;
   374→      AppLogger.info('Parent messages fetched: ${response.length} records');
   375→      return response
   376→          .map((e) => ParentMessageModel.fromJson(e))
   377→          .toList();
   378→    } on sb.PostgrestException catch (e) {
   379→      _mapPostgrestException(e);
   380→    } catch (e) {
   381→      AppLogger.error('Failed to fetch parent messages', error: e);
   382→      throw ServerException(message: e.toString(), statusCode: 500);
   383→    }
   384→  }

   607→  @override
   608→  Future<List<ParentAiInsightModel>> getInsights(
   609→    Map<String, dynamic> params,
   610→  ) async {
   611→    try {
   612→      AppLogger.info('Fetching parent AI insights with params: $params');
   613→      final parentId = _supabase.auth.currentUser?.id ?? '';
   614→      var query = _supabase
   615→          .from(_insightsTable)
   616→          .select()
   617→          .eq('parent_id', parentId)
   618→          .order('created_at', ascending: false);
   619→
   620→      // Apply filters from params
   621→      if (params.containsKey('student_id')) {
   622→        query = query.eq('student_id', params['student_id'] as String);
   623→      }
   624→      if (params.containsKey('is_read')) {
   625→        query = query.eq('is_read', params['is_read'] as bool);
   626→      }
   627→      if (params.containsKey('is_dismissed')) {
   628→        query = query.eq('is_dismissed', params['is_dismissed'] as bool);
   629→      } else {
   630→        // Default: exclude dismissed insights
   631→        query = query.eq('is_dismissed', false);
   632→      }
   633→      if (params.containsKey('insight_type')) {
   634→        query = query.eq('insight_type', params['insight_type'] as String);
   635→      }
   636→      if (params.containsKey('severity')) {
   637→        query = query.eq('severity', params['severity'] as String);
   638→      }
   639→
   640→      final response = await query;
   641→      AppLogger.info(
   642→        'Parent AI insights fetched: ${response.length} records',
   643→      );
   644→      return response
   645→          .map((e) => ParentAiInsightModel.fromJson(e))
   646→          .toList();
   647→    } on sb.PostgrestException catch (e) {
   648→      _mapPostgrestException(e);
   649→    } catch (e) {
   650→      AppLogger.error('Failed to fetch parent AI insights', error: e);
   651→      throw ServerException(message: e.toString(), statusCode: 500);
