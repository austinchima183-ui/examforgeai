   251→/// Provides the Supabase client singleton.
   252→///
   253→/// Depends on [SupabaseConfig.initialize] having been called in `main()`.
   254→final supabaseClientProvider = Provider<sb.SupabaseClient>((ref) {
   255→  try {
   256→    return SupabaseConfig.client;
   257→  } catch (e) {
   258→    AppLogger.error('Failed to resolve Supabase client', error: e);
   259→    rethrow;
   260→  }
   261→});
   262→
   263→/// Provides a configured [Dio] instance with base options pulled from
   264→/// [AppConfig] and [ApiConstants].
   265→final dioProvider = Provider<Dio>((ref) {
   266→  final dio = Dio(
   267→    BaseOptions(
   268→      baseUrl: AppConfig.baseUrl,
   269→      connectTimeout: AppConfig.connectTimeoutDuration,
   270→      receiveTimeout: AppConfig.receiveTimeoutDuration,
   271→      sendTimeout: AppConfig.sendTimeoutDuration,
   272→      headers: ApiConstants.baseHeaders(),
   273→      responseType: ResponseType.json,
   274→      validateStatus: (status) => status != null && status < 500,
   275→    ),
   276→  );
   277→  return dio;
   278→});
   279→
   280→/// Provides the [ApiClient] wrapper around [Dio].
   281→///
   282→/// FIX: Now injects [StorageService] so that auth tokens are actually
   283→/// read from secure storage instead of returning null (the root cause
   284→/// of the "ApiClient auth always returning null" bug).
   285→final apiClientProvider = Provider<ApiClient>((ref) {
   286→  final dio = ref.watch(dioProvider);
   287→  final storageService = ref.watch(storageServiceProvider);
   288→  return ApiClient(dio, storageService: storageService);
   289→});

   294→  // Delegate to the core-layer provider.
   295→  return ref.watch(coreNetworkInfoProvider);
   296→});

   339→/// Stream provider that emits the current Supabase [sb.AuthState].

   341→/// Automatically reconnects on transient failures and emits the latest
   342→/// state to any listener.
   343→final authStateProvider = StreamProvider<sb.AuthState>((ref) {

   345→  return supabaseClient.auth.onAuthStateChange;

   347→
   348→/// Convenience provider that yields the currently authenticated [sb.User],
   349→/// or `null` if not signed in.
   350→final currentUserProvider = Provider<sb.User?>((ref) {
   351→  final authState = ref.watch(authStateProvider);
   352→  return authState.when(
   353→    data: (state) => state.session?.user,
