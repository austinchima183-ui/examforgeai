    25→/// Detail page for a single lesson plan with all sections, edit mode toggle,
    26→/// and the "Generate Questions" button prominently displayed.
    27→class LessonPlanDetailPage extends ConsumerStatefulWidget {
    28→  const LessonPlanDetailPage({
    29→    super.key,
    30→    required this.planId,
    31→  });
    32→
    33→  /// The ID of the lesson plan to display.
    34→  final String planId;
    35→
    36→  @override
    37→  ConsumerState<LessonPlanDetailPage> createState() =>
    38→      _LessonPlanDetailPageState();
    39→}
    40→
    41→class _LessonPlanDetailPageState extends ConsumerState<LessonPlanDetailPage> {
    42→  bool _isEditing = false;
    43→
    44→  // ─── Edit Controllers ────────────────────────────────────────────────
    45→
    46→  final _titleCtrl = TextEditingController();
    47→  final _notesCtrl = TextEditingController();
    48→  final List<TextEditingController> _objectiveControllers = [];
    49→  final List<TextEditingController> _outcomeControllers = [];
    50→  final List<TextEditingController> _materialControllers = [];
    51→  final List<TextEditingController> _homeworkControllers = [];
    52→  final List<TextEditingController> _referenceControllers = [];
    53→  final List<TextEditingController> _extensionControllers = [];
    54→
    55→  bool _controllersInitialized = false;
    56→
    57→  @override
    58→  void initState() {
    59→    super.initState();
    60→    WidgetsBinding.instance.addPostFrameCallback((_) {
    61→      _loadPlan();
    62→    });
    63→  }
    64→
    65→  @override
    66→  void dispose() {
    67→    _titleCtrl.dispose();
    68→    _notesCtrl.dispose();
    69→    _disposeControllers(_objectiveControllers);
    70→    _disposeControllers(_outcomeControllers);
    71→    _disposeControllers(_materialControllers);
    72→    _disposeControllers(_homeworkControllers);
    73→    _disposeControllers(_referenceControllers);
    74→    _disposeControllers(_extensionControllers);
    75→    super.dispose();
    76→  }
    77→
    78→  void _disposeControllers(List<TextEditingController> controllers) {
    79→    for (final c in controllers) {
    80→      c.dispose();
    81→    }
    82→  }
    83→
    84→  // ─── Data Loading ────────────────────────────────────────────────────
    85→
    86→  void _loadPlan() {
    87→    final state = ref.read(lessonPlanProvider);
    88→    // First check if the plan is already in the list
    89→    final plan = state.lessonPlans.where((p) => p.id == widget.planId).firstOrNull;
    90→    if (plan != null) {
    91→      ref.read(lessonPlanProvider.notifier).setCurrentPlan(plan);
    92→    }
    93→    // If no plan is loaded, try loading from the API
    94→    if (ref.read(lessonPlanProvider).currentPlan == null) {
    95→      ref.read(lessonPlanProvider.notifier).loadLessonPlans();
    96→    }
    97→  }
    98→
    99→  void _initEditControllers(LessonPlanEntity plan) {
   100→    if (_controllersInitialized) return;
   101→    _controllersInitialized = true;
   102→
   103→    _titleCtrl.text = plan.title;
   104→    _notesCtrl.text = plan.notes ?? '';
   105→    _initListControllers(plan.learningObjectives, _objectiveControllers);
   106→    _initListControllers(plan.learningOutcomes, _outcomeControllers);
   107→    _initListControllers(plan.teachingMaterials, _materialControllers);
   108→    _initListControllers(plan.homework, _homeworkControllers);
   109→    _initListControllers(plan.referencesList, _referenceControllers);
   110→    _initListControllers(plan.extensionActivities, _extensionControllers);
   111→  }
   112→
   113→  void _initListControllers(
   114→    List<String> items,
   115→    List<TextEditingController> controllers,
   116→  ) {
   117→    for (final c in controllers) {
   118→      c.dispose();
   119→    }
   120→    controllers.clear();
   121→    for (final item in items) {
   122→      controllers.add(TextEditingController(text: item));
   123→    }
   124→  }
   125→
   126→  // ─── Helpers ─────────────────────────────────────────────────────────
   127→
   128→  void _listenForMessages() {
   129→    final state = ref.read(lessonPlanProvider);
   130→    if (state.successMessage != null) {
   131→      _showSnackBar(state.successMessage!, isError: false);
   132→      ref.read(lessonPlanProvider.notifier).clearSuccessMessage();
   133→    }
   134→    if (state.error != null) {
   135→      _showSnackBar(state.error!, isError: true);
   136→      ref.read(lessonPlanProvider.notifier).clearError();
   137→    }
   138→  }
   139→
   140→  void _showSnackBar(String message, {required bool isError}) {
   141→    if (!mounted) return;
   142→    final cs = context.colorScheme;
   143→    context.scaffoldMessenger.showSnackBar(
   144→      SnackBar(
   145→        content: Text(message),
   146→        backgroundColor: isError ? cs.error : cs.primary,
   147→        behavior: SnackBarBehavior.floating,
   148→        shape: RoundedRectangleBorder(
   149→          borderRadius: BorderRadius.circular(Spacings.smRadius),
   150→        ),
   151→        duration: const Duration(seconds: 3),
   152→      ),
   153→    );
   154→  }
   155→
   156→  Future<void> _handleSave() async {
   157→    final plan = ref.read(lessonPlanProvider).currentPlan;
   158→    if (plan == null) return;
   159→
   160→    final updatedPlan = plan.copyWith(
   161→      title: _titleCtrl.text.trim().isNotEmpty
   162→          ? _titleCtrl.text.trim()
   163→          : plan.title,
   164→      notes: _notesCtrl.text.trim().isNotEmpty
   165→          ? _notesCtrl.text.trim()
   166→          : plan.notes,
   167→      learningObjectives: _controllersToList(_objectiveControllers),
   168→      learningOutcomes: _controllersToList(_outcomeControllers),
   169→      teachingMaterials: _controllersToList(_materialControllers),
   170→      homework: _controllersToList(_homeworkControllers),
   171→      referencesList: _controllersToList(_referenceControllers),
   172→      extensionActivities: _controllersToList(_extensionControllers),
   173→    );
   174→
   175→    await ref.read(lessonPlanProvider.notifier).updateLessonPlan(
   176→      UpdateLessonPlanParams(plan: updatedPlan),
   177→    );
   178→
   179→    setState(() => _isEditing = false);
   180→    _controllersInitialized = false;
   181→    _listenForMessages();
   182→  }
   183→
   184→  List<String> _controllersToList(List<TextEditingController> controllers) {
   185→    return controllers
   186→        .map((c) => c.text.trim())
   187→        .where((s) => s.isNotEmpty)
   188→        .toList();
   189→  }
   190→
   191→  void _handleDelete() {
   192→    final plan = ref.read(lessonPlanProvider).currentPlan;
   193→    if (plan == null) return;
   194→
   195→    final cs = context.colorScheme;
   196→    showDialog(
   197→      context: context,
   198→      builder: (ctx) => AlertDialog(
   199→        title: const Text('Delete Lesson Plan'),
   200→        content: Text(
   201→          'Are you sure you want to delete "${plan.title}"? This action cannot be undone.',
   202→        ),
   203→        shape: RoundedRectangleBorder(
   204→          borderRadius: BorderRadius.circular(Spacings.lgRadius),
   205→        ),
   206→        actions: [
   207→          TextButton(
   208→            onPressed: () => Navigator.pop(ctx),
   209→            child: const Text('Cancel'),
   210→          ),
   211→          FilledButton(
   212→            onPressed: () async {
   213→              Navigator.pop(ctx);
   214→              await ref.read(lessonPlanProvider.notifier).deleteLessonPlan(plan.id);
   215→              _listenForMessages();
   216→              if (mounted) context.pop();
   217→            },
   218→            style: FilledButton.styleFrom(
   219→              backgroundColor: cs.error,
   220→              foregroundColor: cs.onError,
   221→            ),
   222→            child: const Text('Delete'),
   223→          ),
   224→        ],
   225→      ),
   226→    );
   227→  }
   228→
   229→  void _handlePublish() {
   230→    ref.read(lessonPlanProvider.notifier).publishLessonPlan(widget.planId);
   231→    _listenForMessages();
   232→  }
   233→
   234→  // ─── Build ───────────────────────────────────────────────────────────
   235→
   236→  @override
   237→  Widget build(BuildContext context) {
   238→    final state = ref.watch(lessonPlanProvider);
   239→    final plan = state.currentPlan;
   240→
   241→    if (state.isLoading && plan == null) {
   242→      return Scaffold(
   243→        appBar: AppAppBar(title: 'Lesson Plan'),
   244→        body: const Center(
   245→          child: AppLoadingSpinner(size: AppLoadingSpinnerSize.large),
   246→        ),
   247→      );
   248→    }
   249→
   250→    if (plan == null && state.error != null) {
   251→      return Scaffold(
   252→        appBar: AppAppBar(title: 'Lesson Plan'),
   253→        body: AppErrorState.genericError(
   254→          message: state.error,
   255→          onRetry: () => ref.read(lessonPlanProvider.notifier).loadLessonPlans(),
   256→        ),
   257→      );
   258→    }
   259→
   260→    if (plan == null) {
   261→      return Scaffold(
   262→        appBar: AppAppBar(title: 'Lesson Plan'),
   263→        body: AppEmptyState(
   264→          icon: Icons.description_outlined,
   265→          title: 'Lesson Plan Not Found',
   266→          subtitle: 'The lesson plan you are looking for does not exist or has been removed.',
   267→          actionLabel: 'Go Back',
   268→          onAction: () => context.pop(),
   269→        ),
   270→      );
   271→    }
   272→
   273→    // Initialize edit controllers when plan is available
   274→    _initEditControllers(plan);
   275→
   276→    return Scaffold(
   277→      appBar: AppAppBar(
   278→        title: _isEditing ? 'Edit Lesson Plan' : 'Lesson Plan Details',
   279→        leading: IconButton(
   280→          icon: const Icon(Icons.arrow_back_rounded),
   281→          onPressed: () => context.pop(),
   282→        ),
   283→        actions: [
   284→          if (!_isEditing) ...[
   285→            AppIconButton(
   286→              icon: Icons.edit_rounded,
   287→              onPressed: () => setState(() => _isEditing = true),
   288→              tooltip: 'Edit',
   289→              variant: AppIconButtonVariant.tonal,
   290→            ),
   291→            AppIconButton(
   292→              icon: Icons.delete_outline_rounded,
   293→              onPressed: _handleDelete,
   294→              tooltip: 'Delete',
   295→              variant: AppIconButtonVariant.standard,
   296→            ),
   297→          ] else ...[
   298→            AppIconButton(
   299→              icon: Icons.save_rounded,
   300→              onPressed: _handleSave,
   301→              tooltip: 'Save',
   302→              variant: AppIconButtonVariant.filled,
   303→              isLoading: state.isUpdating,
   304→            ),
   305→            AppIconButton(
   306→              icon: Icons.close_rounded,
   307→              onPressed: () {
   308→                setState(() {
   309→                  _isEditing = false;
   310→                  _controllersInitialized = false;
   311→                });
   312→              },
   313→              tooltip: 'Cancel',
   314→              variant: AppIconButtonVariant.standard,
   315→            ),
   316→          ],
   317→        ],
   318→      ),
   319→      body: RefreshIndicator(
   320→        onRefresh: () async {
   321→          await ref.read(lessonPlanProvider.notifier).loadLessonPlans();
   322→          _listenForMessages();
   323→        },
   324→        child: _buildDetailContent(plan),
   325→      ),
   326→    );
   327→  }
