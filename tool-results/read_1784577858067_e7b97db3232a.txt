   865→-- Get teacher dashboard summary
   866→CREATE OR REPLACE FUNCTION get_teacher_workspace_summary(
   867→  p_teacher_id UUID,
   868→  p_school_id UUID
   869→)
   870→RETURNS JSONB AS $$
   871→DECLARE
   872→  v_result JSONB;
   873→BEGIN
   874→  SELECT jsonb_build_object(
   875→    'total_lesson_plans', (SELECT COUNT(*) FROM lesson_plans WHERE teacher_id = p_teacher_id),
   876→    'total_schemes', (SELECT COUNT(*) FROM schemes_of_work WHERE teacher_id = p_teacher_id),
   877→    'total_worksheets', (SELECT COUNT(*) FROM worksheets WHERE teacher_id = p_teacher_id),
   878→    'total_assignments', (SELECT COUNT(*) FROM workspace_assignments WHERE teacher_id = p_teacher_id),
   879→    'pending_assignments', (SELECT COUNT(*) FROM workspace_assignments WHERE teacher_id = p_teacher_id AND assignment_status = 'draft'),
   880→    'total_resources', (SELECT COUNT(*) FROM teaching_resources WHERE teacher_id = p_teacher_id),
   881→    'today_events', (SELECT COALESCE(jsonb_agg(jsonb_build_object(
   882→      'id', id, 'title', title, 'event_type', event_type,
   883→      'start_time', start_time, 'end_time', end_time, 'color', color
   884→    )), '[]'::jsonb) FROM calendar_events
   885→    WHERE teacher_id = p_teacher_id
   886→      AND start_time::date = CURRENT_DATE
   887→    ORDER BY start_time),
   888→    'upcoming_events', (SELECT COALESCE(jsonb_agg(jsonb_build_object(
   889→      'id', id, 'title', title, 'event_type', event_type,
   890→      'start_time', start_time, 'end_time', end_time, 'color', color
   891→    )), '[]'::jsonb) FROM calendar_events
   892→    WHERE teacher_id = p_teacher_id
   893→      AND start_time > now()
   894→      AND start_time < now() + INTERVAL '7 days'
   895→    ORDER BY start_time LIMIT 10),
   896→    'recent_ai_content', (SELECT COALESCE(jsonb_agg(jsonb_build_object(
   897→      'id', id, 'action_type', action_type, 'generated_content', LEFT(generated_content, 100),
   898→      'created_at', created_at
   899→    )), '[]'::jsonb) FROM ai_content_history
   900→    WHERE teacher_id = p_teacher_id
   901→    ORDER BY created_at DESC LIMIT 5),
   902→    'draft_lesson_plans', (SELECT COALESCE(jsonb_agg(jsonb_build_object(
   903→      'id', id, 'title', title, 'subject', subject, 'updated_at', updated_at
   904→    )), '[]'::jsonb) FROM lesson_plans
   905→    WHERE teacher_id = p_teacher_id AND is_published = false AND is_archived = false
   906→    ORDER BY updated_at DESC LIMIT 5)
   907→  ) INTO v_result;
   908→
   909→  RETURN v_result;
   910→END;
   911→$$ LANGUAGE plpgsql SECURITY DEFINER;

   913→-- Save version snapshot before update
   914→CREATE OR REPLACE FUNCTION save_workspace_version()
   915→RETURNS TRIGGER AS $$
   916→BEGIN
   917→  IF OLD IS DISTINCT FROM NEW THEN
   918→    INSERT INTO workspace_version_history (teacher_id, resource_type, resource_id, version_number, snapshot, change_summary)
   919→    VALUES (
   920→      COALESCE(NEW.teacher_id, OLD.teacher_id),
   921→      TG_TABLE_NAME,
   922→      COALESCE(NEW.id, OLD.id),
   923→      COALESCE(OLD.version, 1),
   924→      to_jsonb(OLD),
   925→      'Auto-saved version ' || COALESCE(OLD.version, 1)
   926→    );
   927→  END IF;
   928→  RETURN NEW;
   929→END;
   930→$$ LANGUAGE plpgsql;
