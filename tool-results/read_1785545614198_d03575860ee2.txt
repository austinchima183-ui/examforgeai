   251→// ═══════════════════════════════════════════════════════════════════════
   252→
   253→/// Root navigator key — owns the top-level navigation stack.
   254→final _rootNavigatorKey = GlobalKey<NavigatorState>();
   255→
   256→/// Shell navigator key — owns the nested navigation stack inside the
   257→/// [DashboardShell]. This allows the shell (sidebar / bottom nav) to
   258→/// remain mounted while inner pages transition.
   259→final _shellNavigatorKey = GlobalKey<NavigatorState>();
   260→
   261→// ═══════════════════════════════════════════════════════════════════════
   262→// APP ROUTER PROVIDER
   263→// ═══════════════════════════════════════════════════════════════════════
   264→
   265→/// Riverpod provider for the application's [GoRouter].
   266→///
   267→/// Watches [authStateProvider] so that the router automatically
   268→/// re-evaluates its [redirect] callback whenever the authentication
   269→/// state changes (sign-in, sign-out, token refresh).
   270→///
   271→/// Usage in `main.dart`:
   272→/// ```dart
   273→/// final router = ref.read(appRouterProvider);
   274→/// MaterialApp.router(routerConfig: router);
   275→/// ```
   276→final appRouterProvider = Provider<GoRouter>((ref) {
   277→  // Watch auth state so the router re-evaluates redirects on change.
   278→  final authState = ref.watch(authStateProvider);
   279→
   280→  // Also watch onboarding state and role so those redirects stay live.
   281→  ref.watch(onboardingCompleteProvider);
   282→  ref.watch(userRoleProvider);
   283→
   284→  return GoRouter(
   285→    navigatorKey: _rootNavigatorKey,
   286→    initialLocation: RouteNames.splash,
   287→    debugLogDiagnostics: true,
   288→
   289→    // ─── Global Redirect ──────────────────────────────────────────
   290→    redirect: (context, state) {
   291→      final currentPath = state.matchedLocation;
   292→      // ── Loading Guard ───────────────────────────────────────
   293→      // While loading, only allow splash to avoid flicker.
   294→      if (authState.isLoading && currentPath != RouteNames.splash) {
   295→        return RouteNames.splash;
   296→      }
   297→
   298→      final isAuthenticated = authState.when(
   299→        data: (authData) => authData.session != null,
   300→        loading: () => false, // Handled by loading guard above.
   301→        error: (_, __) => false,
   302→      );
   303→
   304→      // ── Auth Guard ────────────────────────────────────────────
   305→      final authRedirect = AuthGuard.evaluate(
   306→        isAuthenticated: isAuthenticated,
   307→        currentPath: currentPath,
   308→      );
   309→      if (authRedirect != null) return authRedirect;
   310→
   311→      // ── Onboarding Guard ─────────────────────────────────────
   312→      final onboardingAsync = ref.read(onboardingCompleteProvider);
   313→      final isOnboardingComplete = onboardingAsync.when(
   314→        data: (complete) => complete,
   315→        loading: () => true, // Assume complete while loading to avoid flicker.
   316→        error: (_, __) => true,
   317→      );
   318→
   319→      final onboardingRedirect = OnboardingGuard.evaluate(
   320→        isAuthenticated: isAuthenticated,
   321→        isOnboardingComplete: isOnboardingComplete,
   322→        currentPath: currentPath,
   323→      );
   324→      if (onboardingRedirect != null) return onboardingRedirect;
   325→
   326→      // ── Role-Based Guard ─────────────────────────────────────
   327→      final roleAsync = ref.read(userRoleProvider);
   328→      final userRole = roleAsync.when(
   329→        data: UserRole.fromString,
   330→        loading: () => null,
   331→        error: (_, __) => null,
   332→      );
   333→
   334→      final roleRedirect = RoleBasedGuard.evaluate(
   335→        userRole: userRole,
   336→        currentPath: currentPath,
   337→      );
   338→      if (roleRedirect != null) return roleRedirect;
   339→
   340→      // No redirect needed — proceed as requested.
   341→      return null;
   342→    },
   343→
   344→    // ─── Route Tree ───────────────────────────────────────────────
   345→    routes: [
   346→      // ── Splash ────────────────────────────────────────────────
   347→      GoRoute(
   348→        path: RouteNames.splash,
   349→        name: 'splash',
   350→        builder: (context, state) => const SplashPage(),
   351→      ),
   352→
   353→      // ── Onboarding ───────────────────────────────────────────
   354→      GoRoute(
   355→        path: RouteNames.onboarding,
   356→        name: 'onboarding',
   357→        builder: (context, state) => const OnboardingPage(),
   358→      ),
   359→
   360→      // ── Auth Routes ──────────────────────────────────────────
   361→      GoRoute(
   362→        path: RouteNames.login,
   363→        name: 'login',
   364→        builder: (context, state) => const LoginPage(),
   365→      ),
   366→      GoRoute(
   367→        path: RouteNames.register,
   368→        name: 'register',
   369→        builder: (context, state) => const RegisterPage(),
   370→      ),
   371→      GoRoute(
   372→        path: RouteNames.forgotPassword,
   373→        name: 'forgotPassword',
   374→        builder: (context, state) => const ForgotPasswordPage(),
   375→      ),
   376→      GoRoute(
   377→        path: RouteNames.verifyEmail,
   378→        name: 'verifyEmail',
   379→        builder: (context, state) => const VerifyEmailPage(),
   380→      ),
   381→      GoRoute(
   382→        path: RouteNames.resetPassword,
   383→        name: 'resetPassword',
   384→        builder: (context, state) => const ResetPasswordPage(),
   385→      ),
   386→
   387→      // ── Protected Shell Route ─────────────────────────────────
   388→      // All routes inside this shell share the [DashboardShell]
   389→      // wrapper which provides the sidebar / bottom nav and app bar.
   390→      ShellRoute(
   391→        navigatorKey: _shellNavigatorKey,
   392→        builder: (context, state, child) => DashboardShell(child: child),
   393→        routes: [
   394→          // Dashboard with role-based sub-routes.
   395→          GoRoute(
   396→            path: RouteNames.dashboard,
   397→            name: 'dashboard',
   398→            builder: (context, state) => const DashboardRedirector(),
   399→            routes: [
   400→              GoRoute(
   401→                path: 'teacher',
