   321→  @override
   322→  Future<List<MarketplaceCategoryModel>> getCategories({
   323→    bool activeOnly = true,
   324→  }) async {
   325→    try {
   326→      var query = _supabase.from(_categoriesTable).select();
   327→
   328→      if (activeOnly) {
   329→        query = query.eq('is_active', true);
   330→      }
   331→
   332→      final response = await query.order('sort_order', ascending: true);
   333→
   334→      AppLogger.info('Fetched ${response.length} marketplace categories');
   335→      return response
   336→          .map<MarketplaceCategoryModel>(
   337→            (row) => MarketplaceCategoryModel.fromJson(row),
   338→          )
   339→          .toList();
   340→    } on sb.PostgrestException catch (e) {
   341→      throw _mapPostgrestException(e);
   342→    } catch (e) {
   343→      AppLogger.error('Unexpected getCategories error', error: e);
   344→      throw const ServerException(
   345→        message: 'Failed to fetch marketplace categories.',
   346→        statusCode: 500,
   347→      );
   348→    }
   349→  }
   350→
   351→  @override
   352→  Future<MarketplaceCategoryModel> getCategory(String categoryId) async {
   353→    try {
   354→      final response = await _supabase
   355→          .from(_categoriesTable)
   356→          .select()
   357→          .eq('id', categoryId)
   358→          .limit(1);
   359→
   360→      if (response.isEmpty) {
   361→        throw const NotFoundException('Category not found.');
   362→      }
   363→
   364→      return MarketplaceCategoryModel.fromJson(response.first);
   365→    } on sb.PostgrestException catch (e) {
   366→      throw _mapPostgrestException(e);
   367→    } on NotFoundException {
   368→      rethrow;
   369→    } catch (e) {
   370→      AppLogger.error('Unexpected getCategory error', error: e);
   371→      throw const ServerException(
   372→        message: 'Failed to fetch category.',
   373→        statusCode: 500,
   374→      );
   375→    }
   376→  }
   377→
   378→  @override
   379→  Future<MarketplaceCategoryModel> upsertCategory(
   380→    MarketplaceCategoryModel category,
   381→  ) async {
   382→    try {
   383→      final data = category.toJson();
   384→      data['updated_at'] = DateTime.now().toIso8601String();
   385→
   386→      final response = await _supabase
   387→          .from(_categoriesTable)
   388→          .upsert(data)
   389→          .select();
   390→
   391→      if (response.isEmpty) {
   392→        throw const ServerException(
   393→          message: 'Category upsert returned no data.',
   394→          statusCode: 500,
   395→        );
   396→      }
   397→
   398→      AppLogger.info('Category upserted: ${response.first['id']}');
   399→      return MarketplaceCategoryModel.fromJson(response.first);
   400→    } on sb.PostgrestException catch (e) {
   401→      throw _mapPostgrestException(e);
   402→    } on ServerException {
   403→      rethrow;
   404→    } catch (e) {
   405→      AppLogger.error('Unexpected upsertCategory error', error: e);
   406→      throw const ServerException(
   407→        message: 'Failed to upsert category.',
   408→        statusCode: 500,
   409→      );
   410→    }
   411→  }

   424→      AppLogger.info('Category soft-deleted: $categoryId');

   607→    String? status,
   608→    String? productType,

   611→    String? subject,

   613→    String? curriculum,
   614→    String? search,
   615→    String? sortBy,
   616→    int limit = 20,
   617→    int offset = 0,
   618→  }) async {
   619→    try {
   620→      // Use RPC for full-text search when a search query is provided
   621→      if (search != null && search.isNotEmpty) {
   622→        final response = await _supabase.rpc(
   623→          'marketplace_search',
   624→          params: {
   625→            'search_query': search,
   626→            'p_status': status,

  1272→  @override
  1273→  Future<MarketplaceOrderModel> createOrder(MarketplaceOrderModel order) async {
  1274→    try {
  1275→      final orderData = order.toJson();
  1276→      orderData['created_at'] = DateTime.now().toIso8601String();
  1277→      orderData['updated_at'] = DateTime.now().toIso8601String();
  1278→
  1279→      // Extract order items for nested insert
  1280→      final itemsJson = orderData.remove('items') as List<dynamic>? ?? [];
  1281→
  1282→      // Insert the order
  1283→      final orderResponse = await _supabase
  1284→          .from(_ordersTable)
  1285→          .insert(orderData)
  1286→          .select()
  1287→          .single();
  1288→
  1289→      final orderId = orderResponse['id'] as String;
  1290→
  1291→      // Insert order items with the order reference
  1292→      if (itemsJson.isNotEmpty) {
  1293→        final orderItems = itemsJson.map((item) {
  1294→          final itemMap = Map<String, dynamic>.from(item as Map<String, dynamic>);
  1295→          itemMap['order_id'] = orderId;
  1296→          itemMap['created_at'] = DateTime.now().toIso8601String();
  1297→          return itemMap;
  1298→        }).toList();
  1299→
  1300→        await _supabase.from(_orderItemsTable).insert(orderItems);
  1301→      }
  1302→
  1303→      AppLogger.info('Order created: $orderId');
  1304→
  1305→      // Fetch the complete order with items
  1306→      return getOrder(orderId);
  1307→    } on sb.PostgrestException catch (e) {
  1308→      throw _mapPostgrestException(e);
  1309→    } on ServerException {
  1310→      rethrow;
  1311→    } catch (e) {
  1312→      AppLogger.error('Unexpected createOrder error', error: e);
  1313→      throw const ServerException(
  1314→        message: 'Failed to create order.',
  1315→        statusCode: 500,
  1316→      );
  1317→    }
  1318→  }
