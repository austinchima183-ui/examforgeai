   218→/// Mastery level for topics.
   219→enum MasteryLevel {
   220→  notStarted(value: 'not_started', label: 'Not Started', percentage: 0, color: '#9CA3AF'),
   221→  beginner(value: 'beginner', label: 'Beginner', percentage: 20, color: '#EF4444'),
   222→  developing(value: 'developing', label: 'Developing', percentage: 40, color: '#F97316'),
   223→  proficient(value: 'proficient', label: 'Proficient', percentage: 60, color: '#FACC15'),
   224→  advanced(value: 'advanced', label: 'Advanced', percentage: 80, color: '#84CC16'),
   225→  expert(value: 'expert', label: 'Expert', percentage: 100, color: '#22C55E');
   226→
   227→  const MasteryLevel({
   228→    required this.value,
   229→    required this.label,
   230→    required this.percentage,
   231→    required this.color,
   232→  });
   233→
   234→  final String value;
   235→  final String label;
   236→  final int percentage;
   237→  final String color;
   238→
   239→  static MasteryLevel? fromString(String? value) {
   240→    if (value == null) return null;
   241→    return MasteryLevel.values.cast<MasteryLevel?>().firstWhere(
   242→          (e) => e?.value == value,
   243→          orElse: () => null,
   244→        );
   245→  }
   246→
   247→  /// Derive mastery level from an accuracy percentage.
   248→  static MasteryLevel fromAccuracy(double accuracy) {
   249→    if (accuracy < 5) return notStarted;
   250→    if (accuracy < 30) return beginner;
   251→    if (accuracy < 50) return developing;
   252→    if (accuracy < 70) return proficient;
   253→    if (accuracy < 90) return advanced;
   254→    return expert;
   255→  }
   256→}

   345→  /// Apply this grade scale to a percentage and return the matching entry.
   346→  GradeScaleEntryEntity? applyToPercentage(double percentage) {
   347→    for (final entry in scaleEntries) {
   348→      if (percentage >= entry.minPercentage && percentage <= entry.maxPercentage) {
   349→        return entry;
   350→      }
   351→    }
   352→    return null;
   353→  }
