   201→// APP ROUTER PROVIDER
   202→// ═══════════════════════════════════════════════════════════════════════
   203→
   204→/// Riverpod provider for the application's [GoRouter].
   205→///
   206→/// Watches [authStateProvider] so that the router automatically
   207→/// re-evaluates its [redirect] callback whenever the authentication
   208→/// state changes (sign-in, sign-out, token refresh).
   209→///
   210→/// Usage in `main.dart`:
   211→/// ```dart
   212→/// final router = ref.read(appRouterProvider);
   213→/// MaterialApp.router(routerConfig: router);
   214→/// ```
   215→final appRouterProvider = Provider<GoRouter>((ref) {
   216→  // Watch auth state so the router re-evaluates redirects on change.
   217→  final authState = ref.watch(authStateProvider);
   218→
   219→  // Also watch onboarding state and role so those redirects stay live.
   220→  ref.watch(onboardingCompleteProvider);
   221→  ref.watch(userRoleProvider);
   222→
   223→  return GoRouter(
   224→    navigatorKey: _rootNavigatorKey,
   225→    initialLocation: RouteNames.splash,
   226→    debugLogDiagnostics: true,
   227→
   228→    // ─── Global Redirect ──────────────────────────────────────────
   229→    redirect: (context, state) {
   230→      final currentPath = state.matchedLocation;
   231→      final isAuthenticated = authState.when(
   232→        data: (authData) => authData.session != null,
   233→        loading: () {
   234→          // While loading, only allow splash. Everything else redirects
   235→          // to splash so we don't flash the login page briefly.
   236→          if (currentPath != RouteNames.splash) {
   237→            return RouteNames.splash;
   238→          }
   239→          return null;
   240→        },
   241→        error: (_, __) => false,
   242→      );
   243→
   244→      // ── Auth Guard ────────────────────────────────────────────
   245→      final authRedirect = AuthGuard.evaluate(
   246→        isAuthenticated: isAuthenticated,
   247→        currentPath: currentPath,
   248→      );
   249→      if (authRedirect != null) return authRedirect;
   250→
   251→      // ── Onboarding Guard ─────────────────────────────────────
   252→      final onboardingAsync = ref.read(onboardingCompleteProvider);
   253→      final isOnboardingComplete = onboardingAsync.when(
   254→        data: (complete) => complete,
   255→        loading: () => true, // Assume complete while loading to avoid flicker.
   256→        error: (_, __) => true,
   257→      );
   258→
   259→      final onboardingRedirect = OnboardingGuard.evaluate(
   260→        isAuthenticated: isAuthenticated,
   261→        isOnboardingComplete: isOnboardingComplete,
   262→        currentPath: currentPath,
   263→      );
   264→      if (onboardingRedirect != null) return onboardingRedirect;
   265→
   266→      // ── Role-Based Guard ─────────────────────────────────────
   267→      final roleAsync = ref.read(userRoleProvider);
   268→      final userRole = roleAsync.when(
   269→        data: UserRole.fromString,
   270→        loading: () => null,
   271→        error: (_, __) => null,
   272→      );
   273→
   274→      final roleRedirect = RoleBasedGuard.evaluate(
   275→        userRole: userRole,
   276→        currentPath: currentPath,
   277→      );
   278→      if (roleRedirect != null) return roleRedirect;
   279→
   280→      // No redirect needed — proceed as requested.
   281→      return null;
   282→    },
   283→

   285→    routes: [
   286→      // ── Splash ────────────────────────────────────────────────
   287→      GoRoute(
   288→        path: RouteNames.splash,
   289→        name: 'splash',

   291→      ),
   292→

  1472→// NOT FOUND PAGE (404)
  1473→// ═══════════════════════════════════════════════════════════════════════
  1474→
  1475→/// Shown when the user navigates to a route that does not exist in
  1476→/// the route tree.
  1477→///
  1478→/// Provides a friendly message and a button to navigate back to the
  1479→/// dashboard (or login if unauthenticated).
  1480→class NotFoundPage extends ConsumerWidget {
  1481→  const NotFoundPage({
  1482→    required this.error,
  1483→    this.path,
  1484→    super.key,
  1485→  });
  1486→
  1487→  /// The [GoException] from GoRouter describing the error.
  1488→  final GoException? error;
  1489→
  1490→  /// The unmatched path the user attempted to visit.
  1491→  final String? path;
  1492→
  1493→  @override
  1494→  Widget build(BuildContext context, WidgetRef ref) {
  1495→    final theme = Theme.of(context);
  1496→    final isAuthenticated = ref.watch(isAuthenticatedProvider);
  1497→
  1498→    return Scaffold(
  1499→      body: Center(
  1500→        child: Padding(
  1501→          padding: const EdgeInsets.all(32),
  1502→          child: Column(
  1503→            mainAxisSize: MainAxisSize.min,
  1504→            children: [
  1505→              Icon(
  1506→                Icons.search_off_rounded,
  1507→                size: 80,
  1508→                color: theme.colorScheme.error,
  1509→              ),
  1510→              const SizedBox(height: 24),
  1511→              Text(
  1512→                'Page Not Found',
  1513→                style: theme.textTheme.headlineMedium?.copyWith(
  1514→                  fontWeight: FontWeight.bold,
  1515→                ),
  1516→                textAlign: TextAlign.center,
  1517→              ),
  1518→              const SizedBox(height: 12),
  1519→              if (path != null)
  1520→                Text(
  1521→                  'The page "$path" does not exist.',
  1522→                  style: theme.textTheme.bodyLarge?.copyWith(
  1523→                    color: theme.colorScheme.onSurfaceVariant,
  1524→                  ),
  1525→                  textAlign: TextAlign.center,
  1526→                ),
  1527→              if (error != null) ...[
  1528→                const SizedBox(height: 8),
  1529→                Text(
  1530→                  error!.message,
  1531→                  style: theme.textTheme.bodySmall?.copyWith(
  1532→                    color: theme.colorScheme.outline,
  1533→                  ),
  1534→                  textAlign: TextAlign.center,
  1535→                ),
  1536→              ],
  1537→              const SizedBox(height: 32),
  1538→              FilledButton.icon(
  1539→                onPressed: () => context.go(
  1540→                  isAuthenticated
  1541→                      ? RouteNames.dashboard
  1542→                      : RouteNames.login,
  1543→                ),
  1544→                icon: const Icon(Icons.home),
  1545→                label: Text(
  1546→                  isAuthenticated ? 'Go to Dashboard' : 'Go to Login',
  1547→                ),
  1548→              ),
  1549→            ],
  1550→          ),
  1551→        ),
  1552→      ),
  1553→    );
  1554→  }
