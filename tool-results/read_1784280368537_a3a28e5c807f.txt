   157→/// Stream provider that emits the current Supabase [sb.AuthState].

   166→/// Convenience provider that yields the currently authenticated [sb.User],
   167→/// or `null` if not signed in.
   168→final currentUserProvider = Provider<sb.User?>((ref) {
   169→  final authState = ref.watch(authStateProvider);
   170→  return authState.when(
   171→    data: (state) => state.session?.user,
   172→    loading: () => SupabaseConfig.currentUser,
   173→    error: (_, __) => null,
   174→  );
   175→});
