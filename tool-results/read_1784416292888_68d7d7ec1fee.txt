   172→// ═══════════════════════════════════════════════════════════════════════
   173→// APP ROUTER PROVIDER
   174→// ═══════════════════════════════════════════════════════════════════════
   175→
   176→/// Riverpod provider for the application's [GoRouter].
   177→///
   178→/// Watches [authStateProvider] so that the router automatically
   179→/// re-evaluates its [redirect] callback whenever the authentication
   180→/// state changes (sign-in, sign-out, token refresh).
   181→///
   182→/// Usage in `main.dart`:
   183→/// ```dart
   184→/// final router = ref.read(appRouterProvider);
   185→/// MaterialApp.router(routerConfig: router);
   186→/// ```
   187→final appRouterProvider = Provider<GoRouter>((ref) {
   188→  // Watch auth state so the router re-evaluates redirects on change.
   189→  final authState = ref.watch(authStateProvider);
   190→
   191→  // Also watch onboarding state and role so those redirects stay live.
   192→  ref.watch(onboardingCompleteProvider);
   193→  ref.watch(userRoleProvider);
   194→
   195→  return GoRouter(
   196→    navigatorKey: _rootNavigatorKey,
   197→    initialLocation: RouteNames.splash,
   198→    debugLogDiagnostics: true,
   199→
   200→    // ─── Global Redirect ──────────────────────────────────────────
   201→    redirect: (context, state) {
   202→      final currentPath = state.matchedLocation;
   203→      final isAuthenticated = authState.when(
   204→        data: (authData) => authData.session != null,
   205→        loading: () {
   206→          // While loading, only allow splash. Everything else redirects
   207→          // to splash so we don't flash the login page briefly.
   208→          if (currentPath != RouteNames.splash) {
   209→            return RouteNames.splash;
   210→          }
   211→          return null;
   212→        },
   213→        error: (_, __) => false,
   214→      );
   215→
   216→      // ── Auth Guard ────────────────────────────────────────────
   217→      final authRedirect = AuthGuard.evaluate(
   218→        isAuthenticated: isAuthenticated,
   219→        currentPath: currentPath,
   220→      );
   221→      if (authRedirect != null) return authRedirect;
   222→
   223→      // ── Onboarding Guard ─────────────────────────────────────
   224→      final onboardingAsync = ref.read(onboardingCompleteProvider);
   225→      final isOnboardingComplete = onboardingAsync.when(
   226→        data: (complete) => complete,
   227→        loading: () => true, // Assume complete while loading to avoid flicker.
   228→        error: (_, __) => true,
   229→      );
   230→
   231→      final onboardingRedirect = OnboardingGuard.evaluate(
   232→        isAuthenticated: isAuthenticated,
   233→        isOnboardingComplete: isOnboardingComplete,
   234→        currentPath: currentPath,
   235→      );
   236→      if (onboardingRedirect != null) return onboardingRedirect;
   237→
   238→      // ── Role-Based Guard ─────────────────────────────────────
   239→      final roleAsync = ref.read(userRoleProvider);
   240→      final userRole = roleAsync.when(
   241→        data: UserRole.fromString,
   242→        loading: () => null,
   243→        error: (_, __) => null,
   244→      );
   245→
   246→      final roleRedirect = RoleBasedGuard.evaluate(
   247→        userRole: userRole,
   248→        currentPath: currentPath,
   249→      );
   250→      if (roleRedirect != null) return roleRedirect;
   251→
   252→      // No redirect needed — proceed as requested.
   253→      return null;
   254→    },
   255→

   257→    routes: [
   258→      // ── Splash ────────────────────────────────────────────────
   259→      GoRoute(
   260→        path: RouteNames.splash,
   261→        name: 'splash',
   262→        builder: (context, state) => const SplashPage(),
   263→      ),
   264→
   265→      // ── Onboarding ───────────────────────────────────────────
   266→      GoRoute(

  1307→
  1308→/// Shown when the user navigates to a route that does not exist in
  1309→/// the route tree.
  1310→///
  1311→/// Provides a friendly message and a button to navigate back to the
  1312→/// dashboard (or login if unauthenticated).
  1313→class NotFoundPage extends ConsumerWidget {
  1314→  const NotFoundPage({
  1315→    required this.error,
  1316→    this.path,
  1317→    super.key,
  1318→  });
  1319→
  1320→  /// The [GoException] from GoRouter describing the error.
  1321→  final GoException? error;
  1322→
  1323→  /// The unmatched path the user attempted to visit.
  1324→  final String? path;
  1325→
  1326→  @override
  1327→  Widget build(BuildContext context, WidgetRef ref) {
  1328→    final theme = Theme.of(context);
  1329→    final isAuthenticated = ref.watch(isAuthenticatedProvider);
  1330→
  1331→    return Scaffold(
  1332→      body: Center(
  1333→        child: Padding(
  1334→          padding: const EdgeInsets.all(32),
  1335→          child: Column(
  1336→            mainAxisSize: MainAxisSize.min,
  1337→            children: [
  1338→              Icon(
  1339→                Icons.search_off_rounded,
  1340→                size: 80,
  1341→                color: theme.colorScheme.error,
  1342→              ),
  1343→              const SizedBox(height: 24),
  1344→              Text(
  1345→                'Page Not Found',
  1346→                style: theme.textTheme.headlineMedium?.copyWith(
  1347→                  fontWeight: FontWeight.bold,
  1348→                ),
  1349→                textAlign: TextAlign.center,
  1350→              ),
  1351→              const SizedBox(height: 12),
  1352→              if (path != null)
  1353→                Text(
  1354→                  'The page "$path" does not exist.',
  1355→                  style: theme.textTheme.bodyLarge?.copyWith(
  1356→                    color: theme.colorScheme.onSurfaceVariant,
  1357→                  ),
  1358→                  textAlign: TextAlign.center,
  1359→                ),
  1360→              if (error != null) ...[
  1361→                const SizedBox(height: 8),
  1362→                Text(
  1363→                  error!.message,
  1364→                  style: theme.textTheme.bodySmall?.copyWith(
  1365→                    color: theme.colorScheme.outline,
  1366→                  ),
  1367→                  textAlign: TextAlign.center,
  1368→                ),
  1369→              ],
  1370→              const SizedBox(height: 32),

  1372→                onPressed: () => context.go(
  1373→                  isAuthenticated

  1375→                      : RouteNames.login,
  1376→                ),
  1377→                icon: const Icon(Icons.home),
