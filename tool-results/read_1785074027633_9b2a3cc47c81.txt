   201→  @override
   202→  Future<GenerationRequestModel> createGenerationRequest(

   204→  ) async {
   205→    try {
   206→      final response = await _supabase
   207→          .from(_generationRequestsTable)
   208→          .insert(requestData)
   209→          .select();
   210→
   211→      if (response.isEmpty) {
   212→        throw const ServerException(
   213→          message: 'Generation request creation returned no data.',
   214→          statusCode: 500,
   215→        );
   216→      }
   217→
   218→      AppLogger.info('Generation request created: ${response.first['id']}');
   219→      return GenerationRequestModel.fromJson(response.first);
   220→    } on sb.PostgrestException catch (e) {
   221→      throw _mapPostgrestException(e);
   222→    } on ServerException {
   223→      rethrow;
   224→    } catch (e) {
   225→      AppLogger.error('Unexpected createGenerationRequest error', error: e);
   226→      throw const ServerException(
   227→        message: 'Failed to create generation request.',
   228→        statusCode: 500,

   292→  @override
   293→  Future<List<GenerationRequestModel>> getGenerationRequests(
   294→    Map<String, dynamic> filters,
   295→  ) async {
   296→    try {
   297→      // Build filter chain first (on PostgrestFilterBuilder),
   298→      // then apply transforms (order, range) which return PostgrestTransformBuilder.
   299→      var filterQuery = _supabase
   300→          .from(_generationRequestsTable)
   301→          .select();
   302→
   303→      // Apply filters on PostgrestFilterBuilder
   304→      if (filters.containsKey('school_id')) {
   305→        filterQuery = filterQuery.eq('school_id', filters['school_id'] as String);
   306→      }
   307→      if (filters.containsKey('status')) {
   308→        filterQuery = filterQuery.eq('status', filters['status'] as String);
   309→      }
   310→      if (filters.containsKey('requested_by')) {
   311→        filterQuery = filterQuery.eq('requested_by', filters['requested_by'] as String);
   312→      }
   313→
   314→      // Now apply transforms on PostgrestTransformBuilder
   315→      var transformQuery = filterQuery.order('created_at', ascending: false);
   316→
   317→      // Pagination
   318→      final page = filters['page'] as int? ?? 1;
   319→      final perPage = filters['per_page'] as int? ?? filters['perPage'] as int? ?? 20;
   320→      final from = (page - 1) * perPage;
   321→      transformQuery = transformQuery.range(from, from + perPage - 1);
   322→
   323→      final response = await transformQuery;
   324→      return response
   325→          .map<GenerationRequestModel>(
   326→              (json) => GenerationRequestModel.fromJson(json),)
   327→          .toList();
   328→    } on sb.PostgrestException catch (e) {
   329→      throw _mapPostgrestException(e);
   330→    } catch (e) {
   331→      AppLogger.error('Unexpected getGenerationRequests error', error: e);
   332→      throw const ServerException(
   333→        message: 'Failed to get generation requests.',
   334→        statusCode: 500,
   335→      );
   336→    }
   337→  }

  1618→  @override
  1619→  Future<Map<String, dynamic>> getDashboardStats(String? schoolId) async {
  1620→    try {
  1621→      // Use an RPC call if available, otherwise aggregate manually
  1622→      try {
  1623→        final response = await _supabase.rpc(
  1624→          'get_ai_dashboard_stats',
  1625→          params: schoolId != null ? {'p_school_id': schoolId} : {},
  1626→        );
  1627→        return response as Map<String, dynamic>;
  1628→      } on sb.PostgrestException catch (e) {
  1629→        if (e.code != '42883') rethrow; // not undefined function
  1630→
  1631→        // Fallback: manually aggregate stats
  1632→        final questionsQuery = _supabase
  1633→            .from(_generatedQuestionsTable)
  1634→            .select('review_status, question_type, difficulty, bloom_level, '
  1635→                'confidence_score, school_id');
  1636→
  1637→        var filteredQuery = questionsQuery;
  1638→        if (schoolId != null) {
  1639→          filteredQuery = filteredQuery.eq('school_id', schoolId);
  1640→        }
  1641→
  1642→        final questions = await filteredQuery;
  1643→
  1644→        final int totalGenerated = questions.length;
  1645→        int totalApproved = 0;
  1646→        int totalRejected = 0;
  1647→        int pendingReview = 0;
  1648→        final questionsByType = <String, int>{};
  1649→        final questionsByDifficulty = <String, int>{};
  1650→        final questionsByBloomLevel = <String, int>{};
  1651→        double totalConfidence = 0;
  1652→        int confidenceCount = 0;
  1653→
  1654→        for (final q in questions) {
  1655→          final status = q['review_status'] as String? ?? 'pending';
  1656→          switch (status) {
  1657→            case 'approved':
  1658→              totalApproved++;
  1659→              break;
  1660→            case 'rejected':
  1661→              totalRejected++;
  1662→              break;
  1663→            default:
  1664→              pendingReview++;
  1665→          }
  1666→
  1667→          final type = q['question_type'] as String? ?? 'unknown';
  1668→          questionsByType[type] = (questionsByType[type] ?? 0) + 1;
  1669→
  1670→          final diff = q['difficulty'] as String? ?? 'unknown';
  1671→          questionsByDifficulty[diff] =
  1672→              (questionsByDifficulty[diff] ?? 0) + 1;
  1673→
  1674→          final bloom = q['bloom_level'] as String?;
  1675→          if (bloom != null) {
  1676→            questionsByBloomLevel[bloom] =
  1677→                (questionsByBloomLevel[bloom] ?? 0) + 1;
  1678→          }
  1679→
  1680→          final conf = q['confidence_score'] as num?;
  1681→          if (conf != null) {
  1682→            totalConfidence += conf.toDouble();
  1683→            confidenceCount++;
  1684→          }
  1685→        }
  1686→
  1687→        // Get cost info from usage stats
  1688→        var usageQuery = _supabase
  1689→            .from(_usageStatsTable)
  1690→            .select('total_cost, total_input_tokens, total_output_tokens, provider');
  1691→        if (schoolId != null) {
  1692→          usageQuery = usageQuery.eq('school_id', schoolId);
  1693→        }
  1694→
  1695→        final usage = await usageQuery;
  1696→        double totalCost = 0;
  1697→        int totalTokensUsed = 0;
  1698→        final costByProvider = <String, double>{};
  1699→
  1700→        for (final u in usage) {
  1701→          final cost = (u['total_cost'] as num?)?.toDouble() ?? 0;
  1702→          totalCost += cost;
  1703→          totalTokensUsed +=
  1704→              (u['total_input_tokens'] as int? ?? 0) +
  1705→                  (u['total_output_tokens'] as int? ?? 0);
  1706→          final prov = u['provider'] as String? ?? 'unknown';
  1707→          costByProvider[prov] = (costByProvider[prov] ?? 0) + cost;
  1708→        }
  1709→
  1710→        return {
  1711→          'total_generated': totalGenerated,
  1712→          'total_approved': totalApproved,
  1713→          'total_rejected': totalRejected,
  1714→          'pending_review': pendingReview,
  1715→          'total_cost': totalCost,
  1716→          'total_tokens_used': totalTokensUsed,
  1717→          'questions_by_type': questionsByType,
  1718→          'questions_by_difficulty': questionsByDifficulty,
  1719→          'questions_by_bloom_level': questionsByBloomLevel,
  1720→          'avg_confidence_score':
  1721→              confidenceCount > 0 ? totalConfidence / confidenceCount : null,
  1722→          'avg_generation_time_ms': null,
  1723→          'recent_generations': [],
  1724→          'cost_by_provider': costByProvider,
  1725→          'daily_usage': [],
  1726→        };
  1727→      }
  1728→    } on sb.PostgrestException catch (e) {
  1729→      throw _mapPostgrestException(e);
  1730→    } catch (e) {

  1783→  /// Maps a Supabase PostgrestException to a domain-appropriate exception.
  1784→  ServerException _mapPostgrestException(sb.PostgrestException e) {
  1785→    AppLogger.error('Supabase error: ${e.message}', error: e);
  1786→
  1787→    switch (e.code) {
  1788→      case '23505':
  1789→        return ServerException(
  1790→          message: 'A record with this data already exists.',
  1791→          statusCode: 409,
  1792→          data: e.details,
  1793→        );
  1794→      case '23503':
  1795→        return ServerException(
  1796→          message: 'Referenced record not found.',
  1797→          statusCode: 400,
  1798→          data: e.details,
  1799→        );
  1800→      case '42501':
  1801→      case 'PGRST301':
  1802→        return ServerException(
  1803→          message: 'Insufficient permissions.',
  1804→          statusCode: 403,
  1805→          data: e.details,
  1806→        );
  1807→      case 'PGRST116':
  1808→        return ServerException(
  1809→          message: 'Resource not found.',
  1810→          statusCode: 404,
  1811→          data: e.details,
  1812→        );
  1813→      default:
  1814→        return ServerException(
  1815→          message: e.message ?? 'An unexpected database error occurred.',
  1816→          statusCode: int.tryParse(e.code ?? '') ?? 500,
  1817→          data: e.details,
  1818→        );
  1819→    }
