     1→     1→/// Analytics Engine — pure computation service for results analytics.
     2→     2→///
     3→     3→/// This service provides powerful insights for teachers, schools, and students
     4→     4→/// by operating on raw result data and producing computed analytics. It is a
     5→     5→/// computation-only module with zero database access and no external
     6→     6→/// dependencies beyond the entity types defined in the results and CBT domains.
     7→     7→///
     8→     8→/// ## Design Principles
     9→     9→///
    10→    10→/// - **Pure functions**: Every public method is deterministic — same inputs
    11→    11→///   always produce the same outputs. No hidden state, no I/O.
    12→    12→/// - **Static helpers**: Core statistical primitives are exposed as `static`
    13→    13→///   methods for maximum testability and reusability.
    14→    14→/// - **Null-safe**: All methods handle empty input lists gracefully, returning
    15→    15→///   sensible defaults rather than throwing.
    16→    16→/// - **Self-contained DTOs**: Output types are defined in this file to avoid
    17→    17→///   coupling to persistence-layer entities.
    18→    18→///
    19→    19→/// ## Usage
    20→    20→///
    21→    21→/// ```dart
    22→    22→/// final engine = AnalyticsEngine();
    23→    23→/// final summary = engine.computeClassPerformance(overallResults,
    24→    24→///     subjectResults: subjectResults);
    25→    25→/// print('Class average: ${summary.averageScore}');
    26→    26→/// print('Pass rate: ${summary.passRate}%');
    27→    27→/// ```

   385→   385→// ANALYTICS ENGINE
   386→   386→// ═══════════════════════════════════════════════════════════════════════
   387→   387→
   388→   388→/// Pure-computation analytics engine for exam result data.
   389→   389→///
   390→   390→/// All methods are side-effect-free and operate solely on the data passed
   391→   391→/// to them. No database access, no network calls, no mutable state.
   392→   392→///
   393→   393→/// The engine is designed to be instantiated once and reused across the
   394→   394→/// application lifecycle, though it carries no state of its own.
   395→   395→///
   396→   396→/// ### Threading
   397→   397→///
   398→   398→/// Because all operations are pure and synchronous, this class is safe to
   399→   399→/// use from any isolate. For large datasets, consider running computations
   400→   400→/// in a background isolate via `Isolate.run`.
   401→   401→///
   402→   402→/// ### Distinction Threshold
   403→   403→///
   404→   404→/// The default distinction threshold is **75%**. This can be overridden via
   405→   405→/// the [distinctionThreshold] parameter on relevant methods.
   406→   406→class AnalyticsEngine {
   407→   407→  // ── Constants ──────────────────────────────────────────────────────
   408→   408→
   409→   409→  /// Default distinction threshold (percentage).
   410→   410→  static const double defaultDistinctionThreshold = 75.0;
   411→   411→
   412→   412→  /// Default pass mark (percentage).
   413→   413→  static const double defaultPassMark = 50.0;
   414→   414→
   415→   415→  /// Number of standard histogram buckets (0–10, 10–20, …, 90–100).
   416→   416→  static const int defaultBucketCount = 10;
   417→   417→
   418→   418→  /// Minimum data points required for a reliable trend calculation.
   419→   419→  static const int minTrendDataPoints = 3;
   420→   420→
   421→   421→  /// Attendance threshold for the "high attendance" group.
   422→   422→  static const double highAttendanceThreshold = 80.0;
   423→   423→
   424→   424→  // ── 1. Class Performance ──────────────────────────────────────────
   425→   425→
   426→   426→  /// Computes a comprehensive performance summary for a class.
   427→   427→  ///
   428→   428→  /// Given a list of [results] (overall results for each student in the
   429→   429→  /// class), this method calculates central tendency, pass/distinction
   430→   430→  /// rates, grade and score distributions, and topic-level aggregations.
   431→   431→  ///
   432→   432→  /// When [subjectResults] are provided, a per-subject breakdown is
   433→   433→  /// included in [ClassPerformanceSummary.subjectBreakdown] and topic
   434→   434→  /// performance is aggregated from those results' metadata.
   435→   435→  ///
   436→   436→  /// Set [previousResults] to compute the [ClassPerformanceSummary.improvementRate]
   437→   437→  /// by comparing the current period's average to the previous period's.
   438→   438→  ///
   439→   439→  /// [distinctionThreshold] defaults to [defaultDistinctionThreshold] (75%).
   440→   440→  /// [passMark] defaults to [defaultPassMark] (50%).
   441→   441→  ClassPerformanceSummary computeClassPerformance(
   442→   442→    List<StudentOverallResultEntity> results, {
   443→   443→    List<StudentSubjectResultEntity>? subjectResults,
   444→   444→    List<StudentOverallResultEntity>? previousResults,
   445→   445→    double distinctionThreshold = defaultDistinctionThreshold,
   446→   446→    double passMark = defaultPassMark,
   447→   447→  }) {
   448→   448→    if (results.isEmpty) {
   449→   449→      return ClassPerformanceSummary(
   450→   450→        classId: '',
   451→   451→        totalStudents: 0,
   452→   452→        averageScore: 0,
   453→   453→        highestScore: 0,
   454→   454→        lowestScore: 0,
   455→   455→        medianScore: 0,
   456→   456→        passRate: 0,
   457→   457→        distinctionRate: 0,
   458→   458→        gradeDistribution: {},
   459→   459→        scoreDistribution: _buildEmptyBuckets(),
   460→   460→        topicPerformance: {},
   461→   461→        improvementRate: 0,
   462→   462→      );
   463→   463→    }
   464→   464→
   465→   465→    final scores = results.map((r) => r.overallPercentage).toList();
   466→   466→    final classId = results.first.classId;
   467→   467→
   468→   468→    // Central tendency
   469→   469→    final average = _mean(scores);
   470→   470→    final highest = _max(scores);
   471→   471→    final lowest = _min(scores);
   472→   472→    final median = _median(scores);
   473→   473→
   474→   474→    // Pass & distinction rates
   475→   475→    final passCount = scores.where((s) => s >= passMark).length;
   476→   476→    final distinctionCount = scores.where((s) => s >= distinctionThreshold).length;
   477→   477→    final passRate = (passCount / scores.length) * 100;
   478→   478→    final distinctionRate = (distinctionCount / scores.length) * 100;
   479→   479→
   480→   480→    // Grade distribution
   481→   481→    final gradeDistribution = <String, int>{};
   482→   482→    for (final r in results) {
   483→   483→      final grade = r.overallGrade ?? _gradeFromPercentage(r.overallPercentage);
   484→   484→      gradeDistribution[grade] = (gradeDistribution[grade] ?? 0) + 1;
   485→   485→    }
   486→   486→
   487→   487→    // Score distribution
   488→   488→    final scoreDistribution = computeScoreDistribution(scores);
   489→   489→
   490→   490→    // Topic performance from subject results metadata
   491→   491→    final topicPerformance = _computeTopicPerformanceFromResults(
   492→   492→      subjectResults ?? [],
   493→   493→    );
   494→   494→
   495→   495→    // Subject breakdown
   496→   496→    final subjectBreakdown = <String, ClassPerformanceSummary>{};
   497→   497→    if (subjectResults != null && subjectResults.isNotEmpty) {
   498→   498→      final bySubject = _groupBy<String, StudentSubjectResultEntity>(
   499→   499→        subjectResults,
   500→   500→        (r) => r.subjectId,
   501→   501→      );
   502→   502→      for (final entry in bySubject.entries) {
   503→   503→        subjectBreakdown[entry.key] = _computeSubjectSummary(
   504→   504→          entry.value,
   505→   505→          distinctionThreshold: distinctionThreshold,
   506→   506→          passMark: passMark,
   507→   507→        );
   508→   508→      }
   509→   509→    }
   510→   510→
   511→   511→    // Improvement rate
   512→   512→    final improvementRate = _computeImprovementRate(results, previousResults);
   513→   513→
   514→   514→    return ClassPerformanceSummary(
   515→   515→      classId: classId,
   516→   516→      totalStudents: results.length,
   517→   517→      averageScore: _round2(average),
   518→   518→      highestScore: _round2(highest),
   519→   519→      lowestScore: _round2(lowest),
   520→   520→      medianScore: _round2(median),
   521→   521→      passRate: _round2(passRate),
   522→   522→      distinctionRate: _round2(distinctionRate),
   523→   523→      gradeDistribution: gradeDistribution,
   524→   524→      scoreDistribution: scoreDistribution,
   525→   525→      topicPerformance: topicPerformance,
   526→   526→      improvementRate: _round2(improvementRate),
   527→   527→      subjectBreakdown: subjectBreakdown,
   528→   528→    );
   529→   529→  }

   532→   532→
   533→   533→  /// Aggregates multiple [ClassPerformanceSummary] instances into a
   534→   534→  /// school-level summary with cross-class and cross-subject rankings.
   535→   535→  ///
   536→   536→  /// [topicMasteryData] is used to identify the most difficult topic
   537→   537→  /// across the entire school.
   538→   538→  ///
   539→   539→  /// [trendData] provides historical performance data for trend
   540→   540→  /// visualization on dashboards.
   541→   541→  SchoolPerformanceSummary computeSchoolPerformance(
   542→   542→    List<ClassPerformanceSummary> classPerformances, {
   543→   543→    required String schoolId,
   544→   544→    required String academicSessionId,
   545→   545→    List<TopicMasteryEntity>? topicMasteryData,
   546→   546→    List<TrendData>? trendData,
   547→   547→    int totalExams = 0,
   548→   548→  }) {
   549→   549→    if (classPerformances.isEmpty) {
   550→   550→      return SchoolPerformanceSummary(
   551→   551→        schoolId: schoolId,
   552→   552→        academicSessionId: academicSessionId,
   553→   553→        totalStudents: 0,
   554→   554→        totalClasses: 0,
   555→   555→        averageScore: 0,
   556→   556→        passRate: 0,
   557→   557→        distinctionRate: 0,
   558→   558→        bestClassId: null,
   559→   559→        bestSubjectId: null,
   560→   560→        mostDifficultTopicId: null,
   561→   561→        classRankings: [],
   562→   562→        subjectRankings: [],
   563→   563→        gradeDistribution: {},
   564→   564→        trendData: trendData ?? [],
   565→   565→        totalExams: totalExams,
   566→   566→      );
   567→   567→    }
   568→   568→
   569→   569→    final totalStudents = classPerformances.fold<int>(
   570→   570→      0,
   571→   571→      (sum, c) => sum + c.totalStudents,
   572→   572→    );
   573→   573→
   574→   574→    // Weighted average: weight each class by its student count.
   575→   575→    final weightedSum = classPerformances.fold<double>(
   576→   576→      0,
   577→   577→      (sum, c) => sum + (c.averageScore * c.totalStudents),
   578→   578→    );
   579→   579→    final averageScore = totalStudents > 0 ? weightedSum / totalStudents : 0.0;
   580→   580→
   581→   581→    // Weighted pass rate
   582→   582→    final passWeightedSum = classPerformances.fold<double>(
   583→   583→      0,
   584→   584→      (sum, c) => sum + (c.passRate * c.totalStudents),
   585→   585→    );
   586→   586→    final passRate = totalStudents > 0 ? passWeightedSum / totalStudents : 0.0;
   587→   587→
   588→   588→    // Weighted distinction rate
   589→   589→    final distinctionWeightedSum = classPerformances.fold<double>(
   590→   590→      0,
   591→   591→      (sum, c) => sum + (c.distinctionRate * c.totalStudents),
   592→   592→    );
   593→   593→    final distinctionRate =
   594→   594→        totalStudents > 0 ? distinctionWeightedSum / totalStudents : 0.0;
   595→   595→
   596→   596→    // Class rankings
   597→   597→    final classRankings = _rankEntities(
   598→   598→      classPerformances.map((c) => MapEntry(c.classId, c.averageScore)).toList(),
   599→   599→    );
   600→   600→
   601→   601→    // Subject rankings from breakdowns
   602→   602→    final subjectScores = <String, List<double>>{};
   603→   603→    for (final cp in classPerformances) {
   604→   604→      for (final sb in cp.subjectBreakdown.entries) {
   605→   605→        subjectScores.putIfAbsent(sb.key, () => []).add(sb.value.averageScore);
   606→   606→      }
   607→   607→    }
   608→   608→    final subjectAverages = subjectScores.map(
   609→   609→      (k, v) => MapEntry(k, _mean(v)),
   610→   610→    );
   611→   611→    final subjectRankings = _rankEntities(
   612→   612→      subjectAverages.entries.map((e) => MapEntry(e.key, e.value)).toList(),
   613→   613→    );
   614→   614→
   615→   615→    // Best class & subject
   616→   616→    final bestClassId = classRankings.isNotEmpty ? classRankings.first.entityId : null;
   617→   617→    final bestSubjectId = subjectRankings.isNotEmpty ? subjectRankings.first.entityId : null;
   618→   618→
   619→   619→    // Most difficult topic
   620→   620→    String? mostDifficultTopicId;
   621→   621→    if (topicMasteryData != null && topicMasteryData.isNotEmpty) {
   622→   622→      final difficultTopics = identifyDifficultTopics(topicMasteryData);
   623→   623→      if (difficultTopics.isNotEmpty) {
   624→   624→        mostDifficultTopicId = difficultTopics.first.topicId;
   625→   625→      }
   626→   626→    } else {
   627→   627→      // Fallback: look at topic performance from class summaries
   628→   628→      final allTopics = <String, List<TopicPerformanceData>>{};
   629→   629→      for (final cp in classPerformances) {
   630→   630→        for (final tp in cp.topicPerformance.entries) {
   631→   631→          allTopics.putIfAbsent(tp.key, () => []).add(tp.value);
   632→   632→        }
   633→   633→      }
   634→   634→      if (allTopics.isNotEmpty) {
   635→   635→        String? worstTopic;
   636→   636→        double worstAccuracy = 101;
   637→   637→        for (final entry in allTopics.entries) {
   638→   638→          final avg = _mean(entry.value.map((t) => t.averageAccuracy).toList());
   639→   639→          if (avg < worstAccuracy) {
   640→   640→            worstAccuracy = avg;
   641→   641→            worstTopic = entry.key;
   642→   642→          }
   643→   643→        }
   644→   644→        mostDifficultTopicId = worstTopic;
   645→   645→      }
   646→   646→    }
   647→   647→
   648→   648→    // Grade distribution: merge all class distributions
   649→   649→    final gradeDistribution = <String, int>{};
   650→   650→    for (final cp in classPerformances) {
   651→   651→      for (final entry in cp.gradeDistribution.entries) {
   652→   652→        gradeDistribution[entry.key] =
   653→   653→            (gradeDistribution[entry.key] ?? 0) + entry.value;
   654→   654→      }
   655→   655→    }
   656→   656→
   657→   657→    return SchoolPerformanceSummary(
   658→   658→      schoolId: schoolId,
   659→   659→      academicSessionId: academicSessionId,
   660→   660→      totalStudents: totalStudents,
   661→   661→      totalClasses: classPerformances.length,
   662→   662→      averageScore: _round2(averageScore),
   663→   663→      passRate: _round2(passRate),
   664→   664→      distinctionRate: _round2(distinctionRate),
   665→   665→      bestClassId: bestClassId,
   666→   666→      bestSubjectId: bestSubjectId,
   667→   667→      mostDifficultTopicId: mostDifficultTopicId,
   668→   668→      classRankings: classRankings,
   669→   669→      subjectRankings: subjectRankings,
   670→   670→      gradeDistribution: gradeDistribution,
   671→   671→      trendData: trendData ?? [],
   672→   672→      totalExams: totalExams,
   673→   673→    );
   674→   674→  }

   676→   676→  // ── 3. Student Trends ─────────────────────────────────────────────
   677→   677→
   678→   678→  /// Computes a student's performance trend from historical scores using
   679→   679→  /// simple linear regression.
   680→   680→  ///
   681→   681→  /// [historicalScores] should be ordered chronologically (oldest first).
   682→   682→  /// At least [minTrendDataPoints] data points are needed for a reliable
   683→   683→  /// result; fewer will still produce a trend but with low confidence.
   684→   684→  ///
   685→   685→  /// [strengths] and [weaknesses] are subject/topic IDs where the student
   686→   686→  /// consistently scores above or below their own average.
   687→   687→  StudentTrendAnalysis computeStudentTrends({
   688→   688→    required String studentId,
   689→   689→    required List<double> historicalScores,
   690→   690→    List<String> strengths = const [],
   691→   691→    List<String> weaknesses = const [],
   692→   692→  }) {
   693→   693→    if (historicalScores.length < 2) {
   694→   694→      return StudentTrendAnalysis(
   695→   695→        studentId: studentId,
   696→   696→        trend: PerformanceTrend.stable,
   697→   697→        slope: 0,
   698→   698→        confidence: 0,
   699→   699→        strengths: strengths,
   700→   700→        weaknesses: weaknesses,
   701→   701→        recommendations: _generateRecommendations(
   702→   702→          PerformanceTrend.stable,
   703→   703→          0,
   704→   704→          strengths,
   705→   705→          weaknesses,
   706→   706→        ),
   707→   707→      );
   708→   708→    }
   709→   709→
   710→   710→    final regression = _linearRegression(historicalScores);
   711→   711→    final slope = regression.slope;
   712→   712→    final confidence = regression.rSquared;
   713→   713→
   714→   714→    // Determine trend from slope
   715→   715→    PerformanceTrend trend;
   716→   716→    if (slope > 1.5) {
   717→   717→      trend = PerformanceTrend.improving;
   718→   718→    } else if (slope < -1.5) {
   719→   719→      trend = PerformanceTrend.declining;
   720→   720→    } else {
   721→   721→      trend = PerformanceTrend.stable;
   722→   722→    }
   723→   723→
   724→   724→    final recommendations = _generateRecommendations(
   725→   725→      trend,
   726→   726→      slope,
   727→   727→      strengths,
   728→   728→      weaknesses,
   729→   729→    );
   730→   730→
   731→   731→    return StudentTrendAnalysis(
   732→   732→      studentId: studentId,
   733→   733→      trend: trend,
   734→   734→      slope: _round2(slope),
   735→   735→      confidence: _round4(confidence),
   736→   736→      strengths: strengths,
   737→   737→      weaknesses: weaknesses,
   738→   738→      recommendations: recommendations,
   739→   739→    );
   740→   740→  }

   766→   766→  // ── 5. Rankings ───────────────────────────────────────────────────
   767→   767→
   768→   768→  /// Generates a ranked list of students from their overall results.
   769→   769→  ///
   770→   770→  /// Uses standard competition ranking (also known as "1224" ranking):
   771→   771→  /// students with identical scores share the same rank, and the next
   772→   772→  /// rank is skipped. For example: 1st, 2nd, 2nd, 4th.
   773→   773→  ///
   774→   774→  /// Results are sorted by [StudentOverallResultEntity.overallPercentage]
   775→   775→  /// in descending order.
   776→   776→  List<RankedStudent> generateRankings(
   777→   777→    List<StudentOverallResultEntity> results,
   778→   778→  ) {
   779→   779→    if (results.isEmpty) return [];
   780→   780→
   781→   781→    // Sort descending by percentage
   782→   782→    final sorted = List<StudentOverallResultEntity>.from(results)
   783→   783→      ..sort((a, b) => b.overallPercentage.compareTo(a.overallPercentage));
   784→   784→
   785→   785→    final rankings = <RankedStudent>[];
   786→   786→    int? lastRank;
   787→   787→    double? lastScore;
   788→   788→
   789→   789→    for (int i = 0; i < sorted.length; i++) {
   790→   790→      final result = sorted[i];
   791→   791→      final position = i + 1;
   792→   792→
   793→   793→      // Same score → same rank; otherwise rank = position
   794→   794→      final rank = (lastScore != null &&
   795→   795→              _round2(result.overallPercentage) == _round2(lastScore))
   796→   796→          ? lastRank!
   797→   797→          : position;
   798→   798→
   799→   799→      rankings.add(RankedStudent(
   800→   800→        studentId: result.studentId,
   801→   801→        rank: rank,
   802→   802→        score: _round2(result.overallPercentage),
   803→   803→        classId: result.classId,
   804→   804→      ));
   805→   805→
   806→   806→      lastRank = rank;
   807→   807→      lastScore = _round2(result.overallPercentage);
   808→   808→    }
   809→   809→
   810→   810→    return rankings;
   811→   811→  }

   813→   813→  // ── 6. Score Distribution ─────────────────────────────────────────
   814→   814→
   815→   815→  /// Creates histogram buckets for score visualization.
   816→   816→  ///
   817→   817→  /// By default, creates [defaultBucketCount] buckets of equal width
   818→   818→  /// (0–10, 10–20, …, 90–100). Each bucket is inclusive of its lower
   819→   819→  /// bound and inclusive of its upper bound, except the last bucket
   820→   820→  /// which includes 100.
   821→   821→  ///
   822→   822→  /// Returns a map of range label → midpoint for quick chart integration.
   823→   823→  Map<String, double> computeScoreDistributionMap(List<double> scores) {
   824→   824→    final buckets = computeScoreDistribution(scores);
   825→   825→    return {for (final b in buckets) b.rangeLabel: (b.min + b.max) / 2};
   826→   826→  }
   827→   827→
   828→   828→  /// Creates histogram buckets for score visualization.
   829→   829→  ///
   830→   830→  /// Returns a list of [ScoreBucket] instances with count and percentage
   831→   831→  /// for each range.
   832→   832→  List<ScoreBucket> computeScoreDistribution(
   833→   833→    List<double> scores, {
   834→   834→    int bucketCount = defaultBucketCount,
   835→   835→  }) {
   836→   836→    if (scores.isEmpty) return _buildEmptyBuckets(bucketCount: bucketCount);
   837→   837→
   838→   838→    final bucketWidth = 100.0 / bucketCount;
   839→   839→    final buckets = <ScoreBucket>[];
   840→   840→
   841→   841→    for (int i = 0; i < bucketCount; i++) {
   842→   842→      final min = i * bucketWidth;
   843→   843→      final max = (i + 1) * bucketWidth;
   844→   844→
   845→   845→      // For the last bucket, include 100. For others, use strict upper bound.
   846→   846→      final count = scores.where((s) {
   847→   847→        if (i == bucketCount - 1) {
   848→   848→          return s >= min && s <= max;
   849→   849→        }
   850→   850→        return s >= min && s < max;
   851→   851→      }).length;
   852→   852→
   853→   853→      final percentage = (count / scores.length) * 100;
   854→   854→
   855→   855→      buckets.add(ScoreBucket(
   856→   856→        rangeLabel: '${min.round()}–${max.round()}',
   857→   857→        min: min,
   858→   858→        max: max,
   859→   859→        count: count,
   860→   860→        percentage: _round2(percentage),
   861→   861→      ));
   862→   862→    }
   863→   863→
   864→   864→    return buckets;
   865→   865→  }

   867→   867→  // ── 7. Identify Difficult Topics ──────────────────────────────────
   868→   868→
   869→   869→  /// Identifies topics with the lowest accuracy across a class or school.

   871→   871→  /// Topics are sorted by ascending accuracy (worst first). Use
   872→   872→  /// [maxResults] to limit the number of topics returned.
   873→   873→  List<IdentifiedTopic> identifyDifficultTopics(
   874→   874→    List<TopicMasteryEntity> masteryData, {
   875→   875→    int? maxResults,
   876→   876→  }) {
   877→   877→    if (masteryData.isEmpty) return [];
   878→   878→
   879→   879→    // Aggregate by topic ID
   880→   880→    final topicAggregation = <String, _TopicAggregation>{};
   881→   881→    for (final m in masteryData) {
   882→   882→      final agg = topicAggregation.putIfAbsent(
   883→   883→        m.topicId,
   884→   884→        () => _TopicAggregation(topicId: m.topicId, subjectId: m.subjectId),
   885→   885→      );
   886→   886→      agg.addAccuracy(m.accuracyPercentage);
   887→   887→    }
   888→   888→
   889→   889→    // Convert and sort by accuracy ascending
   890→   890→    var topics = topicAggregation.values.map((agg) {
   891→   891→      final avgAccuracy = agg.averageAccuracy;
   892→   892→      return IdentifiedTopic(
   893→   893→        topicId: agg.topicId,
   894→   894→        averageAccuracy: _round2(avgAccuracy),
   895→   895→        studentCount: agg.studentCount,
   896→   896→        masteryLevel: MasteryLevel.fromAccuracy(avgAccuracy),
   897→   897→        subjectId: agg.subjectId,
   898→   898→      );
   899→   899→    }).toList()
   900→   900→      ..sort((a, b) => a.averageAccuracy.compareTo(b.averageAccuracy));
   901→   901→
   902→   902→    if (maxResults != null && maxResults > 0) {
   903→   903→      topics = topics.take(maxResults).toList();
   904→   904→    }
   905→   905→
   906→   906→    return topics;

   909→   909→  // ── 8. Attendance Correlation ─────────────────────────────────────
   910→   910→
   911→   911→  /// Computes the Pearson correlation coefficient between attendance
   912→   912→  /// percentage and performance score.
   913→   913→  ///
   914→   914→  /// [attendancePerformancePairs] is a list of (attendance%, score%)
   915→   915→  /// pairs for each student.
   916→   916→  ///
   917→   917→  /// Returns an [AttendanceCorrelation] with the coefficient, sample
   918→   918→  /// size, and a human-readable interpretation.
   919→   919→  AttendanceCorrelation computeAttendanceCorrelation(
   920→   920→    List<({double attendance, double performance})> attendancePerformancePairs,
   921→   921→  ) {
   922→   922→    if (attendancePerformancePairs.length < 3) {
   923→   923→      return AttendanceCorrelation(
   924→   924→        correlationCoefficient: 0,
   925→   925→        sampleSize: attendancePerformancePairs.length,
   926→   926→        interpretation: 'Insufficient data for correlation analysis '
   927→   927→            '(need at least 3 data points)',
   928→   928→      );
   929→   929→    }
   930→   930→
   931→   931→    final n = attendancePerformancePairs.length;
   932→   932→    final attendanceValues =
   933→   933→        attendancePerformancePairs.map((p) => p.attendance).toList();
   934→   934→    final performanceValues =
   935→   935→        attendancePerformancePairs.map((p) => p.performance).toList();
   936→   936→
   937→   937→    final coefficient =
   938→   938→        _pearsonCorrelation(attendanceValues, performanceValues);
   939→   939→
   940→   940→    // Compute average scores for high/low attendance groups
   941→   941→    final highAttendance = attendancePerformancePairs
   942→   942→        .where((p) => p.attendance >= highAttendanceThreshold)
   943→   943→        .map((p) => p.performance)
   944→   944→        .toList();
   945→   945→    final lowAttendance = attendancePerformancePairs
   946→   946→        .where((p) => p.attendance < highAttendanceThreshold)
   947→   947→        .map((p) => p.performance)
   948→   948→        .toList();
   949→   949→
   950→   950→    final highAvg = highAttendance.isNotEmpty ? _mean(highAttendance) : null;
   951→   951→    final lowAvg = lowAttendance.isNotEmpty ? _mean(lowAttendance) : null;
   952→   952→
   953→   953→    final interpretation = _interpretCorrelation(coefficient);
   954→   954→
   955→   955→    return AttendanceCorrelation(
   956→   956→      correlationCoefficient: _round4(coefficient),
   957→   957→      sampleSize: n,
   958→   958→      interpretation: interpretation,
   959→   959→      highAttendanceAvgScore: highAvg != null ? _round2(highAvg) : null,
   960→   960→      lowAttendanceAvgScore: lowAvg != null ? _round2(lowAvg) : null,
   961→   961→    );
   962→   962→  }

   965→   965→
   966→   966→  /// Computes the performance trend from a list of historical scores

   968→   968→  ///
   969→   969→  /// Returns the slope of the best-fit line. A positive slope indicates
   970→   970→  /// improvement, negative indicates decline, and near-zero indicates

   973→   973→  /// Requires at least 2 data points; returns 0 otherwise.

   983→   983→  // ── Statistics ─────────────────────────────────────────────────────
   984→   984→
   985→   985→  /// Arithmetic mean of a list of numbers.
   986→   986→  static double _mean(List<double> values) {
   987→   987→    if (values.isEmpty) return 0;
   988→   988→    return values.reduce((a, b) => a + b) / values.length;
   989→   989→  }
   990→   990→
   991→   991→  /// Maximum value in a list, or 0 if empty.
   992→   992→  static double _max(List<double> values) {
   993→   993→    if (values.isEmpty) return 0;
   994→   994→    return values.reduce((a, b) => a > b ? a : b);
   995→   995→  }
   996→   996→
   997→   997→  /// Minimum value in a list, or 0 if empty.
   998→   998→  static double _min(List<double> values) {
   999→   999→    if (values.isEmpty) return 0;
  1000→  1000→    return values.reduce((a, b) => a < b ? a : b);
  1001→  1001→  }
  1002→  1002→
  1003→  1003→  /// Median (50th percentile) of a sorted copy of [values].
  1004→  1004→  static double _median(List<double> values) {
  1005→  1005→    if (values.isEmpty) return 0;
  1006→  1006→    final sorted = List<double>.from(values)..sort();
  1007→  1007→    final mid = sorted.length ~/ 2;
  1008→  1008→    if (sorted.length.isEven) {
  1009→  1009→      return (sorted[mid - 1] + sorted[mid]) / 2;
  1010→  1010→    }
  1011→  1011→    return sorted[mid];
  1012→  1012→  }

  1014→  1014→  /// Standard deviation (population) of a list.
  1015→  1015→  static double _standardDeviation(List<double> values) {
  1016→  1016→    if (values.length < 2) return 0;
  1017→  1017→    final mean = _mean(values);
  1018→  1018→    final sumSquaredDiff = values.fold<double>(
  1019→  1019→      0,
  1020→  1020→      (sum, v) => sum + (v - mean) * (v - mean),
  1021→  1021→    );
  1022→  1022→    return _sqrt(sumSquaredDiff / values.length);
  1023→  1023→  }

  1025→  1025→  /// Integer square root approximation for standard deviation.
  1026→  1026→  /// Uses Newton's method for reasonable precision.
  1027→  1027→  static double _sqrt(double value) {
  1028→  1028→    if (value < 0) return 0;
  1029→  1029→    if (value == 0) return 0;
  1030→  1030→    double x = value;
  1031→  1031→    double prev = 0;
  1032→  1032→    for (int i = 0; i < 20; i++) {
  1033→  1033→      prev = x;
  1034→  1034→      x = 0.5 * (x + value / x);
  1035→  1035→      if ((x - prev).abs() < 0.000001) break;
  1036→  1036→    }
  1037→  1037→    return x;
  1038→  1038→  }

  1040→  1040→  // ── Linear Regression ──────────────────────────────────────────────
  1041→  1041→
  1042→  1042→  /// Simple linear regression (OLS) on [y] values with implicit
  1043→  1043→  /// x = 0, 1, 2, …, n−1.
  1044→  1044→  ///
  1045→  1045→  /// Returns the slope, intercept, and R² value.
  1046→  1046→  static _LinearRegressionResult _linearRegression(List<double> y) {
  1047→  1047→    final n = y.length;
  1048→  1048→    if (n < 2) {
  1049→  1049→      return _LinearRegressionResult(slope: 0, intercept: _mean(y), rSquared: 0);
  1050→  1050→    }
  1051→  1051→
  1052→  1052→    // x values: 0, 1, 2, ..., n-1
  1053→  1053→    final xMean = (n - 1) / 2.0;
  1054→  1054→    final yMean = _mean(y);
  1055→  1055→
  1056→  1056→    double sumXY = 0;
  1057→  1057→    double sumX2 = 0;
  1058→  1058→    for (int i = 0; i < n; i++) {
  1059→  1059→      final xDiff = i - xMean;
  1060→  1060→      final yDiff = y[i] - yMean;
  1061→  1061→      sumXY += xDiff * yDiff;
  1062→  1062→      sumX2 += xDiff * xDiff;
  1063→  1063→    }
  1064→  1064→
  1065→  1065→    final slope = sumX2 != 0 ? sumXY / sumX2 : 0.0;
  1066→  1066→    final intercept = yMean - slope * xMean;
  1067→  1067→
  1068→  1068→    // R² calculation
  1069→  1069→    final ssTotal = y.fold<double>(0, (sum, v) => sum + (v - yMean) * (v - yMean));
  1070→  1070→    double ssResidual = 0;
  1071→  1071→    for (int i = 0; i < n; i++) {
  1072→  1072→      final predicted = intercept + slope * i;
  1073→  1073→      ssResidual += (y[i] - predicted) * (y[i] - predicted);
  1074→  1074→    }
  1075→  1075→    final rSquared = ssTotal != 0 ? 1 - (ssResidual / ssTotal) : 0.0;
  1076→  1076→
  1077→  1077→    return _LinearRegressionResult(
  1078→  1078→      slope: slope,
  1079→  1079→      intercept: intercept,
  1080→  1080→      rSquared: rSquared.clamp(0, 1),
  1081→  1081→    );
  1082→  1082→  }

  1084→  1084→  // ── Pearson Correlation ────────────────────────────────────────────
  1085→  1085→
  1086→  1086→  /// Computes the Pearson correlation coefficient between [x] and [y].
  1087→  1087→  static double _pearsonCorrelation(List<double> x, List<double> y) {
  1088→  1088→    assert(x.length == y.length);
  1089→  1089→    final n = x.length;
  1090→  1090→    if (n < 2) return 0;
  1091→  1091→
  1092→  1092→    final xMean = _mean(x);
  1093→  1093→    final yMean = _mean(y);
  1094→  1094→
  1095→  1095→    double sumXY = 0;
  1096→  1096→    double sumX2 = 0;
  1097→  1097→    double sumY2 = 0;
  1098→  1098→
  1099→  1099→    for (int i = 0; i < n; i++) {
  1100→  1100→      final dx = x[i] - xMean;
  1101→  1101→      final dy = y[i] - yMean;
  1102→  1102→      sumXY += dx * dy;
  1103→  1103→      sumX2 += dx * dx;
  1104→  1104→      sumY2 += dy * dy;
  1105→  1105→    }
  1106→  1106→
  1107→  1107→    final denominator = _sqrt(sumX2) * _sqrt(sumY2);
  1108→  1108→    if (denominator == 0) return 0;
  1109→  1109→    return (sumXY / denominator).clamp(-1.0, 1.0);
  1110→  1110→  }

  1127→  1127→  /// Ranks entities by score (descending) using competition ranking.
