   254→  // ─── Helper: Handle Supabase errors ────────────────────────────────
   255→
   256→  Never _handlePostgrestError(sb.PostgrestException e) {
   257→    AppLogger.error('Supabase error: ${e.message}', error: e);
   258→    if (e.code == '401') {
   259→      throw UnauthorizedException(message: e.message);
   260→    } else if (e.code == '403') {
   261→      throw ForbiddenException(message: e.message);
   262→    } else if (e.code == '404' || e.code == 'PGRST116') {
   263→      throw NotFoundException(message: e.message);
   264→    }
   265→    throw ServerException(
   266→      message: e.message,
   267→      statusCode: int.tryParse(e.code ?? '500') ?? 500,
   268→      data: e.details,
   269→    );
   270→  }

   274→  @override
   275→  Future<List<Map<String, dynamic>>> getEducationalLevels({
   276→    int limit = PaginatedQueryMixin.dropdownPageSize,
   277→  }) async {
   278→    try {
   279→      // PERF: Added limit to prevent unbounded query on educational_levels
   280→      final response = await _supabase
   281→          .from('educational_levels')
   282→          .select()
   283→          .eq('is_active', true)
   284→          .order('level_order')
   285→          .limit(limit);
   286→      return List<Map<String, dynamic>>.from(response);
   287→    } on sb.PostgrestException catch (e) {
   288→      _handlePostgrestError(e);
   289→    }
   290→  }

   341→  @override
   342→  Future<List<Map<String, dynamic>>> getCurricula(
   343→    Map<String, dynamic> filters, {
   344→    int limit = PaginatedQueryMixin.defaultPageSize,
   345→    int offset = 0,
   346→  }) async {
   347→    try {
   348→      var query = _supabase.from('curricula').select();
   349→
   350→      if (filters.containsKey('country_code')) {
   351→        query = query.eq('country_code', filters['country_code'] as String);
   352→      }
   353→      if (filters.containsKey('curriculum_type')) {
   354→        query =
   355→            query.eq('curriculum_type', filters['curriculum_type'] as String);
   356→      }
   357→      if (filters.containsKey('is_active')) {
   358→        query = query.eq('is_active', filters['is_active'] as bool);
   359→      }
   360→
   361→      // PERF: Added range-based pagination to prevent unbounded query
   362→      final response = await query.order('name').range(offset, offset + limit - 1);
   363→      return List<Map<String, dynamic>>.from(response);
   364→    } on sb.PostgrestException catch (e) {
   365→      _handlePostgrestError(e);
   366→    }
   367→  }
