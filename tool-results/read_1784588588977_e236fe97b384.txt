   299→  @override
   300→  Future<SchoolModel> createSchool(Map<String, dynamic> schoolData) async {
   301→    try {
   302→      final response = await _supabaseClient
   303→          .from(_schoolsTable)
   304→          .insert(schoolData)
   305→          .select()
   306→          .single();
   307→
   308→      return SchoolModel.fromJson(response);
   309→    } on sb.PostgrestException catch (e) {
   310→      AppLogger.error('Create school failed', error: e);
   311→      throw ServerException(
   312→        message: e.message,
   313→        statusCode: int.tryParse(e.code ?? '') ?? 500,
   314→        data: e.details,
   315→      );
   316→    } catch (e) {
   317→      AppLogger.error('Create school unexpected error', error: e);
   318→      throw ServerException(
   319→        message: 'Failed to create school: $e',
   320→        statusCode: 500,
   321→      );
   322→    }
   323→  }

   410→  @override
   411→  Future<List<SchoolModel>> getSchools(Map<String, dynamic> filters) async {
   412→    try {
   413→      var query = _supabaseClient.from(_schoolsTable).select();
   414→
   415→      if (filters['is_active'] != null) {
   416→        query = query.eq('is_active', filters['is_active'] as bool);
   417→      }
   418→      if (filters['search'] != null) {
   419→        final search = filters['search'] as String;
   420→        query = query.or('name.ilike.%$search%,code.ilike.%$search%');
   421→      }
   422→      if (filters['school_level'] != null) {
   423→        query = query.eq('school_level', filters['school_level'] as String);
   424→      }
   425→      if (filters['subscription_status'] != null) {
   426→        query = query.eq(
   427→          'subscription_status',
   428→          filters['subscription_status'] as String,
   429→        );
   430→      }
   431→
   432→      final page = filters['page'] as int? ?? 1;
   433→      final perPage = filters['per_page'] as int? ?? filters['perPage'] as int? ?? 20;
   434→      final offset = (page - 1) * perPage;
   435→
   436→      query = query.range(offset, offset + perPage - 1);
   437→      query = query.order('created_at', ascending: false);
   438→
   439→      final response = await query;
   440→
   441→      return response
   442→          .map<SchoolModel>((json) => SchoolModel.fromJson(json))
   443→          .toList();
   444→    } on sb.PostgrestException catch (e) {
   445→      AppLogger.error('Get schools failed', error: e);
   446→      throw ServerException(
   447→        message: e.message,
   448→        statusCode: int.tryParse(e.code ?? '') ?? 500,
   449→        data: e.details,
   450→      );
   451→    } catch (e) {
   452→      AppLogger.error('Get schools unexpected error', error: e);
   453→      throw ServerException(
   454→        message: 'Failed to get schools: $e',
   455→        statusCode: 500,
   456→      );
   457→    }
   458→  }

   791→  }) async {

   793→      var query = _supabaseClient
   794→          .from(_studentProfilesTable)
