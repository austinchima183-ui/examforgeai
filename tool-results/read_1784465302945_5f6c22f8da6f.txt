    16→
    17→/// Super Admin moderation page with tab-based navigation for managing
    18→/// pending products, reported reviews, sellers, and disputes.
    19→///
    20→/// ```dart
    21→/// Navigator.push(
    22→///   context,
    23→///   MaterialPageRoute(builder: (_) => MarketplaceModerationPage()),
    24→/// );
    25→/// ```
    26→class MarketplaceModerationPage extends ConsumerStatefulWidget {
    27→  const MarketplaceModerationPage({super.key});
    28→
    29→  @override
    30→  ConsumerState<MarketplaceModerationPage> createState() =>
    31→      _MarketplaceModerationPageState();
    32→}
    33→
    34→class _MarketplaceModerationPageState
    35→    extends ConsumerState<MarketplaceModerationPage>
    36→    with SingleTickerProviderStateMixin {
    37→  late final TabController _tabController;
    38→
    39→  @override
    40→  void initState() {
    41→    super.initState();
    42→    _tabController = TabController(
    43→      length: _ModerationTab.values.length,
    44→      vsync: this,
    45→    );
    46→    WidgetsBinding.instance.addPostFrameCallback((_) => _loadAllData());
    47→  }
    48→
    49→  @override
    50→  void dispose() {
    51→    _tabController.dispose();
    52→    super.dispose();
    53→  }
    54→
    55→  Future<void> _loadAllData() async {
    56→    final notifier = ref.read(moderationProvider.notifier);
    57→    await Future.wait([
    58→      notifier.loadPendingProducts(),
    59→      notifier.loadSellers(),
    60→      notifier.loadDisputes(),
    61→    ]);
    62→  }
    63→
    64→  void _showSnackBar(String message, {bool isError = false}) {
    65→    context.scaffoldMessenger.showSnackBar(
    66→      SnackBar(
    67→        content: Text(message),
    68→        backgroundColor: isError ? AppColors.error : null,
    69→        behavior: SnackBarBehavior.floating,
    70→      ),
    71→    );
    72→  }
    73→
    74→  // ─── Stats Bar ────────────────────────────────────────────────────────
    75→
    76→  Widget _buildStatsBar(ModerationState state) {
    77→    final cs = context.colorScheme;
    78→    final tt = context.textTheme;
    79→
    80→    return Container(
    81→      padding: const EdgeInsets.symmetric(
    82→        horizontal: Spacings.lg,
    83→        vertical: Spacings.md,
    84→      ),
    85→      color: cs.surfaceContainerLow,
    86→      child: Row(
    87→        children: [
    88→          _StatChip(
    89→            label: 'Pending Products',
    90→            count: state.pendingProducts.length,
    91→            color: AppColors.warning,
    92→            icon: Icons.inventory_2_outlined,
    93→          ),
    94→          const SizedBox(width: Spacings.lg),
    95→          _StatChip(
    96→            label: 'Open Disputes',
    97→            count: state.openDisputeCount,
    98→            color: AppColors.error,
    99→            icon: Icons.gavel_outlined,
   100→          ),
   101→          const SizedBox(width: Spacings.lg),
   102→          _StatChip(
   103→            label: 'Reported Reviews',
   104→            count: state.reportedReviews.length,
   105→            color: AppColors.info,
   106→            icon: Icons.flag_outlined,
   107→          ),
   108→        ],
   109→      ),
   110→    );
   111→  }
   112→
   113→  @override
   114→  Widget build(BuildContext context) {
   115→    final state = ref.watch(moderationProvider);
   116→
   117→    // Listen for error/success messages
   118→    ref.listen<ModerationState>(moderationProvider, (prev, next) {
   119→      if (next.error != null && prev?.error != next.error) {
   120→        _showSnackBar(next.error!, isError: true);
   121→        ref.read(moderationProvider.notifier).clearError();
   122→      }
   123→      if (next.successMessage != null &&
   124→          prev?.successMessage != next.successMessage) {
   125→        _showSnackBar(next.successMessage!);
   126→        ref.read(moderationProvider.notifier).clearSuccess();
   127→      }
   128→    });
   129→
   130→    return Scaffold(
   131→      appBar: AppAppBar(
   132→        title: 'Moderation',
   133→        bottom: TabBar(
   134→          controller: _tabController,
   135→          tabs: _ModerationTab.values
   136→              .map((tab) => Tab(
   137→                    text: tab.label,
   138→                    icon: Icon(tab.icon),
   139→                  ))
   140→              .toList(),
   141→        ),
   142→      ),
   143→      body: Column(
   144→        children: [
   145→          // ── Stats bar ───────────────────────────────────────────────
   146→          _buildStatsBar(state),
   147→
   148→          // ── Tab content ─────────────────────────────────────────────
   149→          Expanded(
   150→            child: state.isLoading && state.pendingProducts.isEmpty
   151→                ? const Center(child: AppLoadingSpinner())
   152→                : TabBarView(
   153→                    controller: _tabController,
   154→                    children: [
   155→                      // ── Pending Products ────────────────────────────
   156→                      _PendingProductsTab(
   157→                        products: state.pendingProducts,
   158→                        isLoading: state.isLoading,
   159→                      ),
   160→
   161→                      // ── Reported Reviews ────────────────────────────
   162→                      _ReportedReviewsTab(
   163→                        reviews: state.reportedReviews,
   164→                        isLoading: state.isLoading,
   165→                      ),
   166→
   167→                      // ── Sellers ─────────────────────────────────────
   168→                      _SellersTab(
   169→                        sellers: state.sellers,
   170→                        isLoading: state.isLoading,
   171→                      ),
   172→
   173→                      // ── Disputes ────────────────────────────────────
   174→                      _DisputesTab(
   175→                        disputes: state.disputes,
   176→                        isLoading: state.isLoading,
   177→                      ),
   178→                    ],
   179→                  ),
   180→          ),
   181→        ],
   182→      ),
   183→    );
   184→  }
   185→}

   261→class _PendingProductsTab extends ConsumerWidget {
   262→  const _PendingProductsTab({
   263→    required this.products,
   264→    required this.isLoading,
   265→  });
   266→
   267→  final List<MarketplaceProductEntity> products;
   268→  final bool isLoading;
   269→
   270→  @override
   271→  Widget build(BuildContext context, WidgetRef ref) {
   272→    if (products.isEmpty && !isLoading) {
   273→      return AppEmptyState(
   274→        icon: Icons.check_circle_outline_rounded,
   275→        title: 'No Pending Products',
   276→        subtitle: 'All products have been reviewed.',
   277→      );
   278→    }
   279→
   280→    return Column(
   281→      children: [
   282→        // Bulk actions
   283→        if (products.isNotEmpty)
   284→          Padding(
   285→            padding: const EdgeInsets.all(Spacings.lg),
   286→            child: Row(
   287→              children: [
   288→                Expanded(
   289→                  child: AppButton(
   290→                    label: 'Approve All',
   291→                    onPressed: () async {
   292→                      final confirmed = await AppDialog.showConfirm(
   293→                        context: context,
   294→                        title: 'Approve All Products?',
   295→                        message:
   296→                            'This will approve ${products.length} pending products.',
   297→                        confirmText: 'Approve All',
   298→                      );
   299→                      if (confirmed == true) {
   300→                        for (final p in products) {
   301→                          await ref
   302→                              .read(moderationProvider.notifier)
   303→                              .approveProduct(
   304→                                productId: p.id,
   305→                                moderatorId: 'admin',
   306→                              );
   307→                        }
   308→                      }
   309→                    },
   310→                    variant: AppButtonVariant.tonal,
   311→                    icon: Icons.check_circle_outline,
   312→                    size: AppButtonSize.small,
   313→                  ),
   314→                ),
   315→                const SizedBox(width: Spacings.md),
   316→                Expanded(
   317→                  child: AppButton(
   318→                    label: 'Reject All',
   319→                    onPressed: () async {
   320→                      final confirmed = await AppDialog.showConfirm(
   321→                        context: context,
   322→                        title: 'Reject All Products?',
   323→                        message:
   324→                            'This will reject ${products.length} pending products.',
   325→                        confirmText: 'Reject All',
   326→                        isDestructive: true,
   327→                      );
   328→                      if (confirmed == true) {
   329→                        for (final p in products) {
   330→                          await ref
   331→                              .read(moderationProvider.notifier)
   332→                              .rejectProduct(
   333→                                productId: p.id,
   334→                                reason: 'Bulk rejection',
   335→                                moderatorId: 'admin',
   336→                              );
   337→                        }
   338→                      }
   339→                    },
   340→                    variant: AppButtonVariant.outlined,
   341→                    icon: Icons.cancel_outlined,
   342→                    size: AppButtonSize.small,
   343→                  ),
   344→                ),
   345→              ],
   346→            ),
   347→          ),
   348→
   349→        // Product list
   350→        Expanded(
   351→          child: RefreshIndicator(
   352→            onRefresh: () =>
   353→                ref.read(moderationProvider.notifier).loadPendingProducts(),
   354→            child: ListView.separated(
   355→              padding: const EdgeInsets.symmetric(horizontal: Spacings.lg),
   356→              itemCount: products.length,
   357→              separatorBuilder: (_, __) =>
   358→                  const SizedBox(height: Spacings.md),
   359→              itemBuilder: (context, index) {
   360→                final product = products[index];
   361→                return _PendingProductCard(product: product);
   362→              },
   363→            ),
   364→          ),
   365→        ),
   366→      ],
   367→    );
   368→  }
   369→}

   532→  void _showRejectDialog(BuildContext context, WidgetRef ref) {
   533→    final reasonController = TextEditingController();
   534→
   535→    AppDialog.showCustom(
   536→      context: context,
   537→      builder: (ctx) {
   538→        return Column(
   539→          mainAxisSize: MainAxisSize.min,
   540→          crossAxisAlignment: CrossAxisAlignment.start,
   541→          children: [
   542→            Text(
   543→              'Reject Product',
   544→              style: ctx.textTheme.titleLarge?.copyWith(
   545→                fontWeight: AppTypography.wSemiBold,
   546→                color: ctx.colorScheme.onSurface,
   547→              ),
   548→            ),
   549→            const SizedBox(height: Spacings.md),
   550→            Text(
   551→              'Provide a reason for rejecting "${product.title}":',
   552→              style: ctx.textTheme.bodyMedium?.copyWith(
   553→                color: ctx.colorScheme.onSurfaceVariant,
   554→              ),
   555→            ),
   556→            const SizedBox(height: Spacings.md),
   557→            AppTextField(
   558→              controller: reasonController,
   559→              label: 'Reason',
   560→              hint: 'Enter rejection reason',
   561→              maxLines: 3,
   562→              minLines: 2,
   563→              isRequired: true,

   565→            const SizedBox(height: Spacings.xl),
   566→            Row(
   567→              mainAxisAlignment: MainAxisAlignment.end,

   575→                AppButton(
   576→                  label: 'Reject',

   578→                    final reason = reasonController.text.trim();
   579→                    if (reason.isEmpty) return;

   581→                    ref.read(moderationProvider.notifier).rejectProduct(
   582→                          productId: product.id,

   586→                  },

   590→            ),
