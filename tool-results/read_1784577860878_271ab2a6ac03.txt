    34→    34→// PROMPT ENGINE
    35→    35→// ═══════════════════════════════════════════════════════════════════════
    36→    36→
    37→    37→/// The Prompt Engine is the brain of the AI question generation system.
    38→    38→///
    39→    39→/// It is responsible for:
    40→    40→/// - Resolving prompt templates with input variables
    41→    41→/// - Selecting the best template for a given generation request
    42→    42→/// - Building generation, improvement, validation, and extraction prompts
    43→    43→/// - Parsing AI responses into structured domain entities
    44→    44→class PromptEngine {
    45→    45→  // ─── Template Resolution ──────────────────────────────────────────
    46→    46→
    47→    47→  /// Resolve a prompt template by replacing `{{variable_name}}` placeholders
    48→    48→  /// with actual values from [variables].
    49→    49→  ///
    50→    50→  /// Required variables that are missing will produce a warning log but
    51→    51→  /// will be replaced with an empty string so generation can continue.
    52→    52→  String resolvePrompt(
    53→    53→    PromptTemplateEntity template,
    54→    54→    Map<String, dynamic> variables,
    55→    55→  ) {
    56→    56→    String resolved = template.userPromptTemplate;
    57→    57→
    58→    58→    // Replace all {{variable_name}} occurrences
    59→    59→    final variablePattern = RegExp(r'\{\{(\w+)\}\}');
    60→    60→    resolved = resolved.replaceAllMapped(variablePattern, (match) {
    61→    61→      final varName = match.group(1)!;
    62→    62→      if (variables.containsKey(varName)) {
    63→    63→        return variables[varName].toString();
    64→    64→      }
    65→    65→      // Check if template declares this variable with a default
    66→    66→      final templateVar = template.variables
    67→    67→          .where((v) => v.name == varName)
    68→    68→          .firstOrNull;
    69→    69→      if (templateVar?.defaultValue != null) {
    70→    70→        return templateVar!.defaultValue!;
    71→    71→      }
    72→    72→      if (templateVar?.isRequired ?? false) {
    73→    73→        AppLogger.warning('Missing required variable: $varName in template '
    74→    74→            '"${template.name}"');
    75→    75→      }
    76→    76→      return '';
    77→    77→    });
    78→    78→
    79→    79→    return resolved;
    80→    80→  }
    81→    81→
    82→    82→  /// Build a generation prompt from teacher inputs.
    83→    83→  ///
    84→    84→  /// Selects the best matching template from [templates] based on
    85→    85→  /// subject, question type, difficulty, curriculum, and language.
    86→    86→  /// If no suitable template is found, falls back to the built-in
    87→    87→  /// default generation prompt.
    88→    88→  PromptResolution resolveGenerationPrompt(
    89→    89→    GenerationInputEntity input,
    90→    90→    List<PromptTemplateEntity> templates,
    91→    91→  ) {
    92→    92→    final activeTemplates = templates
    93→    93→        .where((t) =>
    94→    94→            t.isActive &&
    95→    95→            t.promptType == PromptType.questionGeneration)
    96→    96→        .toList();
    97→    97→
    98→    98→    // Score each template based on how well it matches the input
    99→    99→    PromptTemplateEntity? bestMatch;
   100→   100→    int bestScore = -1;
   101→   101→
   102→   102→    for (final template in activeTemplates) {
   103→   103→      int score = 0;
   104→   104→
   105→   105→      // Match on question type (high weight)
   106→   106→      if (template.questionType == input.questionType) {
   107→   107→        score += 50;
   108→   108→      }
   109→   109→
   110→   110→      // Match on difficulty (medium weight)
   111→   111→      if (template.difficulty == input.difficulty) {
   112→   112→        score += 20;
   113→   113→      }
   114→   114→
   115→   115→      // Match on subject (high weight)
   116→   116→      if (template.subjectId == input.subjectId) {
   117→   117→        score += 30;
   118→   118→      }
   119→   119→
   120→   120→      // Match on curriculum (medium weight)
   121→   121→      if (template.curriculum == input.curriculum) {
   122→   122→        score += 15;
   123→   123→      }
   124→   124→
   125→   125→      // Match on Bloom's level (medium weight)
   126→   126→      if (template.bloomLevel == input.bloomLevel) {
   127→   127→        score += 15;
   128→   128→      }
   129→   129→
   130→   130→      // Match on language (low weight)
   131→   131→      if (template.language == input.language) {
   132→   132→        score += 10;
   133→   133→      }
   134→   134→
   135→   135→      // Match on provider preference (low weight)
   136→   136→      if (template.provider == input.provider) {
   137→   137→        score += 5;
   138→   138→      }
   139→   139→
   140→   140→      // Default templates get a small bonus
   141→   141→      if (template.isDefault) {
   142→   142→        score += 3;
   143→   143→      }
   144→   144→
   145→   145→      // Higher quality score gets a bonus
   146→   146→      if (template.qualityScore != null) {
   147→   147→        score += (template.qualityScore! * 10).round();
   148→   148→      }
   149→   149→
   150→   150→      // Higher success rate gets a bonus
   151→   151→      if (template.successRate != null) {
   152→   152→        score += (template.successRate! * 10).round();
   153→   153→      }
   154→   154→
   155→   155→      if (score > bestScore) {
   156→   156→        bestScore = score;
   157→   157→        bestMatch = template;
   158→   158→      }
   159→   159→    }
   160→   160→
   161→   161→    // Build resolved variables from input
   162→   162→    final resolvedVariables = _buildGenerationVariables(input);
   163→   163→
   164→   164→    if (bestMatch != null) {
   165→   165→      final systemPrompt = bestMatch.systemPrompt;
   166→   166→      final userPrompt = resolvePrompt(bestMatch, resolvedVariables);
   167→   167→
   168→   168→      // Append few-shot examples if available
   169→   169→      final fullUserPrompt = _appendFewShotExamples(userPrompt, bestMatch);
   170→   170→
   171→   171→      // Append chain-of-thought instruction if enabled
   172→   172→      final finalUserPrompt = bestMatch.chainOfThought
   173→   173→          ? '$fullUserPrompt\n\nThink through this step by step before generating the questions.'
   174→   174→          : fullUserPrompt;
   175→   175→
   176→   176→      return PromptResolution(
   177→   177→        systemPrompt: systemPrompt,
   178→   178→        userPrompt: finalUserPrompt,
   179→   179→        templateUsed: bestMatch,
   180→   180→        resolvedVariables: resolvedVariables,
   181→   181→      );
   182→   182→    }
   183→   183→
   184→   184→    // Fallback to built-in default prompt
   185→   185→    return PromptResolution(
   186→   186→      systemPrompt: _defaultSystemPrompt,
   187→   187→      userPrompt: _buildDefaultUserPrompt(input),
   188→   188→      templateUsed: null,
   189→   189→      resolvedVariables: resolvedVariables,
   190→   190→    );
   191→   191→  }

   295→   295→    buffer.writeln('Return a JSON object with the following structure:');

   440→   440→  // ─── Response Parsing ─────────────────────────────────────────────
   441→   441→
   442→   442→  /// Parse AI response into structured generated questions.
   443→   443→  ///
   444→   444→  /// The response should be a JSON object or a list of question objects.
   445→   445→  /// This method handles multiple response formats:
   446→   446→  /// 1. `{ "questions": [...] }` — standard format
   447→   447→  /// 2. `[...]` — direct list of questions
   448→   448→  /// 3. `{ "question": {...} }` — single question wrapper
   449→   449→  List<GeneratedQuestionEntity> parseGeneratedQuestions(
   450→   450→    Map<String, dynamic> response,
   451→   451→    GenerationRequestEntity request,
   452→   452→  ) {
   453→   453→    try {
   454→   454→      List<dynamic> questionsList;
   455→   455→
   456→   456→      if (response.containsKey('questions')) {
   457→   457→        questionsList = response['questions'] as List<dynamic>;
   458→   458→      } else if (response.containsKey('question')) {
   459→   459→        questionsList = [response['question']];
   460→   460→      } else if (response.values.any((v) => v is List)) {
   461→   461→        // Try to find the list of questions
   462→   462→        final listEntry = response.entries
   463→   463→            .firstWhere((e) => e.value is List && (e.value as List).isNotEmpty);
   464→   464→        questionsList = listEntry.value as List<dynamic>;
   465→   465→      } else {
   466→   466→        // Assume the entire response is a single question
   467→   467→        questionsList = [response];
   468→   468→      }
   469→   469→
   470→   470→      return questionsList
   471→   471→          .whereType<Map<String, dynamic>>()
   472→   472→          .map((q) => _mapToGeneratedQuestion(q, request))

   511→   511→  /// Parse validation results from AI response.
   512→   512→  List<ValidationResultEntity> parseValidationResults(
   513→   513→    Map<String, dynamic> response,
   514→   514→    String questionId,
   515→   515→  ) {
   516→   516→    try {
   517→   517→      final results = <ValidationResultEntity>[];
   518→   518→
   519→   519→      // If response has validation_results array
   520→   520→      if (response.containsKey('validation_results')) {
   521→   521→        final validationResults =
   522→   522→            response['validation_results'] as List<dynamic>;
   523→   523→        for (int i = 0; i < validationResults.length; i++) {
   524→   524→          final result = validationResults[i] as Map<String, dynamic>;
   525→   525→          results.add(ValidationResultEntity(
   526→   526→            id: '',
   527→   527→            generatedQuestionId: questionId,
   528→   528→            validationType: result['type'] as String? ?? 'general',
   529→   529→            severity: ValidationSeverity.fromString(
   530→   530→                    result['severity'] as String?) ??
   531→   531→                ValidationSeverity.info,
   532→   532→            message: result['message'] as String? ?? 'No message provided',
   533→   533→            suggestion: result['suggestion'] as String?,
   534→   534→            isResolved: false,
   535→   535→            resolvedBy: null,
   536→   536→            resolvedAt: null,
   537→   537→            createdAt: DateTime.now(),
   538→   538→          ));
   539→   539→        }
   540→   540→      }
   541→   541→
   542→   542→      // If no structured results, create one from the overall validity
   543→   543→      if (results.isEmpty) {
   544→   544→        final isValid = response['is_valid'] as bool? ?? true;
   545→   545→        results.add(ValidationResultEntity(
   546→   546→          id: '',
   547→   547→          generatedQuestionId: questionId,
   548→   548→          validationType: 'overall',
   549→   549→          severity: isValid ? ValidationSeverity.info : ValidationSeverity.error,
   550→   550→          message: isValid
   551→   551→              ? 'Question passed validation'
   552→   552→              : 'Question failed validation',
   553→   553→          suggestion: response['suggestion'] as String?,
   554→   554→          isResolved: false,
   555→   555→          resolvedBy: null,
   556→   556→          resolvedAt: null,
   557→   557→          createdAt: DateTime.now(),
   558→   558→        ));
   559→   559→      }
   560→   560→
   561→   561→      return results;
   562→   562→    } catch (e) {
   563→   563→      AppLogger.error('Failed to parse validation results', error: e);
   564→   564→      return [
   565→   565→        ValidationResultEntity(
   566→   566→          id: '',
   567→   567→          generatedQuestionId: questionId,
   568→   568→          validationType: 'parse_error',
   569→   569→          severity: ValidationSeverity.warning,
   570→   570→          message: 'Could not parse validation results from AI response',
   571→   571→          isResolved: false,
   572→   572→          resolvedBy: null,
   573→   573→          resolvedAt: null,
   574→   574→          createdAt: DateTime.now(),
   575→   575→        ),
   576→   576→      ];
   577→   577→    }
   578→   578→  }

   622→   622→  /// Maps a raw question JSON object to a [GeneratedQuestionEntity].
   623→   623→  GeneratedQuestionEntity _mapToGeneratedQuestion(
   624→   624→    Map<String, dynamic> q,
   625→   625→    GenerationRequestEntity request,
   626→   626→  ) {
   627→   627→    // Determine question type
   628→   628→    final questionTypeStr = q['question_type'] as String? ??
   629→   629→        q['type'] as String? ??
   630→   630→        request.inputParams['question_type'] as String? ??
   631→   631→        'multiple_choice';
   632→   632→    final questionType = QuestionType.values.cast<QuestionType?>().firstWhere(
   633→   633→          (t) => t?.value == questionTypeStr,
   634→   634→          orElse: () => QuestionType.multipleChoice,
   635→   635→        ) ?? QuestionType.multipleChoice;
   636→   636→
   637→   637→    // Determine difficulty
   638→   638→    final difficultyStr = q['difficulty'] as String? ??
   639→   639→        request.inputParams['difficulty'] as String? ??
   640→   640→        'medium';
   641→   641→    final difficulty = DifficultyLevel.values.cast<DifficultyLevel?>().firstWhere(
   642→   642→          (d) => d?.value == difficultyStr,
   643→   643→          orElse: () => DifficultyLevel.medium,
   644→   644→        ) ?? DifficultyLevel.medium;
   645→   645→
   646→   646→    // Determine Bloom's level
   647→   647→    final bloomStr = q['bloom_level'] as String? ??
   648→   648→        request.inputParams['bloom_level'] as String?;
   649→   649→    final bloomLevel = BloomTaxonomy.fromString(bloomStr);
   650→   650→
   651→   651→    // Parse answer options
   652→   652→    List<Map<String, dynamic>> answerOptions = [];
   653→   653→    if (q['answer_options'] != null) {
   654→   654→      answerOptions = (q['answer_options'] as List<dynamic>)
   655→   655→          .whereType<Map<String, dynamic>>()
   656→   656→          .toList();
   657→   657→    } else if (q['options'] != null) {
   658→   658→      answerOptions = (q['options'] as List<dynamic>)
   659→   659→          .whereType<Map<String, dynamic>>()
   660→   660→          .toList();
   661→   661→    }
   662→   662→
   663→   663→    // Parse matching pairs
   664→   664→    List<Map<String, dynamic>> matchingPairs = [];
   665→   665→    if (q['matching_pairs'] != null) {
   666→   666→      matchingPairs = (q['matching_pairs'] as List<dynamic>)
   667→   667→          .whereType<Map<String, dynamic>>()
   668→   668→          .toList();
   669→   669→    }
   670→   670→
   671→   671→    // Parse ordering items
   672→   672→    List<Map<String, dynamic>> orderingItems = [];
   673→   673→    if (q['ordering_items'] != null) {
   674→   674→      orderingItems = (q['ordering_items'] as List<dynamic>)
   675→   675→          .whereType<Map<String, dynamic>>()
   676→   676→          .toList();
   677→   677→    }
   678→   678→
   679→   679→    // Parse fill-in-blank answers
   680→   680→    List<Map<String, dynamic>> fillInBlankAnswers = [];
   681→   681→    if (q['fill_in_blank_answers'] != null) {
   682→   682→      fillInBlankAnswers = (q['fill_in_blank_answers'] as List<dynamic>)
   683→   683→          .whereType<Map<String, dynamic>>()
   684→   684→          .toList();
   685→   685→    }
   686→   686→
   687→   687→    // Parse suggested references
   688→   688→    List<String> references = [];
   689→   689→    if (q['references'] != null) {
   690→   690→      references = (q['references'] as List<dynamic>)
   691→   691→          .whereType<String>()
   692→   692→          .toList();
   693→   693→    } else if (q['suggested_references'] != null) {
   694→   694→      references = (q['suggested_references'] as List<dynamic>)
   695→   695→          .whereType<String>()
   696→   696→          .toList();
   697→   697→    }
   698→   698→
   699→   699→    return GeneratedQuestionEntity(
   700→   700→      id: '',
   701→   701→      generationRequestId: request.id,
   702→   702→      questionBankId: null,
   703→   703→      schoolId: request.schoolId,
   704→   704→      questionType: questionType,
   705→   705→      difficulty: difficulty,
   706→   706→      bloomLevel: bloomLevel,
   707→   707→      content: q['content'] as String? ?? q['question'] as String? ?? '',
   708→   708→      contentJson: q['content_json'] as Map<String, dynamic>?,
   709→   709→      answerOptions: answerOptions,
   710→   710→      matchingPairs: matchingPairs,
   711→   711→      orderingItems: orderingItems,
   712→   712→      fillInBlankAnswers: fillInBlankAnswers,
   713→   713→      explanation: q['explanation'] as String?,
   714→   714→      suggestedReferences: references,
   715→   715→      marks: q['marks'] as int? ?? 1,
   716→   716→      estimatedTimeSeconds: q['estimated_time_seconds'] as int?,
   717→   717→      confidenceScore: (q['confidence_score'] as num?)?.toDouble(),
   718→   718→      curriculumAlignment: q['curriculum_alignment'] as Map<String, dynamic>?,
   719→   719→      reviewStatus: ReviewStatus.pending,
   720→   720→      reviewedBy: null,
   721→   721→      reviewedAt: null,
   722→   722→      reviewNotes: null,
   723→   723→      teacherEdits: null,
   724→   724→      isEdited: false,
   725→   725→      isApproved: false,
   726→   726→      metadata: q['metadata'] as Map<String, dynamic>?,
   727→   727→      createdAt: DateTime.now(),
   728→   728→      updatedAt: DateTime.now(),
   729→   729→    );
   730→   730→  }
