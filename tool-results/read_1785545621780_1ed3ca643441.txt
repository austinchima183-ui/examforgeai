   201→   250→// NAVIGATOR KEYS
   202→   251→// ═══════════════════════════════════════════════════════════════════════
   203→   252→
   204→   253→/// Root navigator key — owns the top-level navigation stack.
   205→   254→final _rootNavigatorKey = GlobalKey<NavigatorState>();
   206→   255→
   207→   256→/// Shell navigator key — owns the nested navigation stack inside the
   208→   257→/// [DashboardShell]. This allows the shell (sidebar / bottom nav) to
   209→   258→/// remain mounted while inner pages transition.
   210→   259→final _shellNavigatorKey = GlobalKey<NavigatorState>();
   211→   260→
   212→   261→// ═══════════════════════════════════════════════════════════════════════
   213→   262→// APP ROUTER PROVIDER
   214→   263→// ═══════════════════════════════════════════════════════════════════════
   215→   264→
   216→   265→/// Riverpod provider for the application's [GoRouter].
   217→   266→///
   218→   267→/// Watches [authStateProvider] so that the router automatically
   219→   268→/// re-evaluates its [redirect] callback whenever the authentication
   220→   269→/// state changes (sign-in, sign-out, token refresh).
   221→   270→///
   222→   271→/// Usage in `main.dart`:
   223→   272→/// ```dart
   224→   273→/// final router = ref.read(appRouterProvider);
   225→   274→/// MaterialApp.router(routerConfig: router);
   226→   275→/// ```
   227→   276→final appRouterProvider = Provider<GoRouter>((ref) {
   228→   277→  // Watch auth state so the router re-evaluates redirects on change.
   229→   278→  final authState = ref.watch(authStateProvider);
   230→   279→
   231→   280→  // Also watch onboarding state and role so those redirects stay live.
   232→   281→  ref.watch(onboardingCompleteProvider);
   233→   282→  ref.watch(userRoleProvider);
   234→   283→
   235→   284→  return GoRouter(
   236→   285→    navigatorKey: _rootNavigatorKey,
   237→   286→    initialLocation: RouteNames.splash,
   238→   287→    debugLogDiagnostics: true,
   239→   288→
   240→   289→    // ─── Global Redirect ──────────────────────────────────────────
   241→   290→    redirect: (context, state) {
   242→   291→      final currentPath = state.matchedLocation;
   243→   292→      // ── Loading Guard ───────────────────────────────────────
   244→   293→      // While loading, only allow splash to avoid flicker.
   245→   294→      if (authState.isLoading && currentPath != RouteNames.splash) {
   246→   295→        return RouteNames.splash;
   247→   296→      }
   248→   297→
   249→   298→      final isAuthenticated = authState.when(
   250→   299→        data: (authData) => authData.session != null,
   251→   300→        loading: () => false, // Handled by loading guard above.
   252→   301→        error: (_, __) => false,
   253→   302→      );
   254→   303→
   255→   304→      // ── Auth Guard ────────────────────────────────────────────
   256→   305→      final authRedirect = AuthGuard.evaluate(
   257→   306→        isAuthenticated: isAuthenticated,
   258→   307→        currentPath: currentPath,
   259→   308→      );
   260→   309→      if (authRedirect != null) return authRedirect;
   261→   310→
   262→   311→      // ── Onboarding Guard ─────────────────────────────────────
   263→   312→      final onboardingAsync = ref.read(onboardingCompleteProvider);
   264→   313→      final isOnboardingComplete = onboardingAsync.when(
   265→   314→        data: (complete) => complete,
   266→   315→        loading: () => true, // Assume complete while loading to avoid flicker.
   267→   316→        error: (_, __) => true,
   268→   317→      );
   269→   318→
   270→   319→      final onboardingRedirect = OnboardingGuard.evaluate(
   271→   320→        isAuthenticated: isAuthenticated,
   272→   321→        isOnboardingComplete: isOnboardingComplete,
   273→   322→        currentPath: currentPath,
   274→   323→      );
   275→   324→      if (onboardingRedirect != null) return onboardingRedirect;
   276→   325→
   277→   326→      // ── Role-Based Guard ─────────────────────────────────────
   278→   327→      final roleAsync = ref.read(userRoleProvider);
   279→   328→      final userRole = roleAsync.when(
   280→   329→        data: UserRole.fromString,
   281→   330→        loading: () => null,
   282→   331→        error: (_, __) => null,
   283→   332→      );
   284→   333→
   285→   334→      final roleRedirect = RoleBasedGuard.evaluate(
   286→   335→        userRole: userRole,
   287→   336→        currentPath: currentPath,
   288→   337→      );
   289→   338→      if (roleRedirect != null) return roleRedirect;
   290→   339→
   291→   340→      // No redirect needed — proceed as requested.
   292→   341→      return null;
   293→   342→    },
   294→   343→
   295→   344→    // ─── Route Tree ───────────────────────────────────────────────
   296→   345→    routes: [
   297→   346→      // ── Splash ────────────────────────────────────────────────
   298→   347→      GoRoute(
   299→   348→        path: RouteNames.splash,
   300→   349→        name: 'splash',
   301→   350→        builder: (context, state) => const SplashPage(),
   302→   351→      ),
   303→   352→
   304→   353→      // ── Onboarding ───────────────────────────────────────────
   305→   354→      GoRoute(
   306→   355→        path: RouteNames.onboarding,
   307→   356→        name: 'onboarding',
   308→   357→        builder: (context, state) => const OnboardingPage(),
   309→   358→      ),
   310→   359→
   311→   360→      // ── Auth Routes ──────────────────────────────────────────
   312→   361→      GoRoute(
   313→   362→        path: RouteNames.login,
   314→   363→        name: 'login',
   315→   364→        builder: (context, state) => const LoginPage(),
   316→   365→      ),
   317→   366→      GoRoute(
   318→   367→        path: RouteNames.register,
   319→   368→        name: 'register',
   320→   369→        builder: (context, state) => const RegisterPage(),
   321→   370→      ),
   322→   371→      GoRoute(
   323→   372→        path: RouteNames.forgotPassword,
   324→   373→        name: 'forgotPassword',
   325→   374→        builder: (context, state) => const ForgotPasswordPage(),
   326→   375→      ),
   327→   376→      GoRoute(
   328→   377→        path: RouteNames.verifyEmail,
   329→   378→        name: 'verifyEmail',
