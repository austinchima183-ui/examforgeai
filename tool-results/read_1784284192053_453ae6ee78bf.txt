   503→   503→  /// Applies a grade scale to a percentage score and returns the

   505→   505→  Future<void> applyGradeScale(double percentage, String scaleId) async {
   506→   506→    state = state.copyWith(isLoading: true, error: null);
   507→   507→
   508→   508→    final result = await _applyGradeScaleUseCase(percentage, scaleId);
   509→   509→

   511→   511→      onSuccess: (entry) {
   512→   512→        state = state.copyWith(
   513→   513→          isLoading: false,
   514→   514→          appliedEntry: entry,
   515→   515→          successMessage: entry != null
   516→   516→              ? 'Grade: ${entry.grade} (${entry.minPercentage}%–${entry.maxPercentage}%)'

   518→   518→          error: null,
