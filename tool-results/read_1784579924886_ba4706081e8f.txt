   396→  @override
   397→  Future<Result<TransactionEntity>> verifyPayment(String txRef) async {
   398→    try {
   399→      final flutterwaveData = await _flutterwaveDataSource.verifyTransaction(txRef);
   400→
   401→      // Find and update our local transaction record
   402→      final localTx = await _remoteDataSource.getTransaction(
   403→        flutterwaveData['tx_ref'] as String? ?? txRef,
   404→      );
   405→
   406→      final status = _mapFlutterwaveStatus(
   407→        flutterwaveData['status'] as String? ?? 'pending',
   408→      );
   409→      final updatedTx = await _remoteDataSource.updateTransaction(
   410→        localTx.id,
   411→        {
   412→          'status': status.value,
   413→          'flutterwave_transaction_id': flutterwaveData['id']?.toString(),
   414→          'flutterwave_flw_ref': flutterwaveData['flw_ref'] as String?,
   415→          'flutterwave_fee': flutterwaveData['app_fee'] as double? ?? 0,
   416→          'net_amount': flutterwaveData['amount_settled'] as double? ?? 0,
   417→          'payment_method_summary': flutterwaveData['payment_type'] as String?,
   418→          'processor_response': flutterwaveData,
   419→          'verified_at': DateTime.now().toIso8601String(),
   420→          if (status == TransactionStatus.successful)
   421→            'completed_at': DateTime.now().toIso8601String(),
   422→        },
   423→      );
   424→      return Success(updatedTx.toEntity());
   425→    } on AuthException catch (e) {
   426→      return FailureResult(Failure.auth(message: e.message, code: e.code));
   427→    } on ServerException catch (e) {
   428→      return FailureResult(Failure.server(
   429→        message: e.message, statusCode: e.statusCode, data: e.data,
   430→      ));
   431→    } on NotFoundException catch (e) {
   432→      return FailureResult(Failure.notFound(message: e.message));
   433→    } catch (e) {
   434→      AppLogger.error('Unexpected verifyPayment error', error: e);
   435→      return const FailureResult(Failure.server(
   436→        message: 'Failed to verify payment.', statusCode: 500,
   437→      ));
   438→    }
   439→  }

   441→  @override
   442→  Future<Result<bool>> processWebhookEvent(Map<String, dynamic> payload) async {
   443→    try {
   444→      final event = payload['event'] as String? ?? '';
   445→      final data = payload['data'] as Map<String, dynamic>? ?? {};
   446→
   447→      AppLogger.info('Processing Flutterwave webhook event: $event');
   448→
   449→      switch (event) {
   450→        case 'charge.completed':
   451→          final txRef = data['tx_ref'] as String?;
   452→          if (txRef != null) {
   453→            // Verify and update the transaction
   454→            await verifyPayment(txRef);
   455→          }
   456→          break;
   457→        case 'transfer.completed':
   458→        case 'transfer.failed':
   459→          AppLogger.info('Transfer event received: $event');
   460→          break;
   461→        case 'subscription.cancelled':
   462→          final subId = data['subscription_id'] as String?;
   463→          if (subId != null) {
   464→            await _remoteDataSource.updateSubscription(subId, {
   465→              'status': 'cancelled',
   466→              'cancelled_at': DateTime.now().toIso8601String(),
   467→            });
   468→          }
   469→          break;
   470→        default:
   471→          AppLogger.info('Unhandled webhook event: $event');
   472→      }
   473→      return const Success(true);
   474→    } on AuthException catch (e) {
   475→      return FailureResult(Failure.auth(message: e.message, code: e.code));
   476→    } on ServerException catch (e) {
   477→      return FailureResult(Failure.server(
   478→        message: e.message, statusCode: e.statusCode, data: e.data,
   479→      ));
   480→    } catch (e) {
   481→      AppLogger.error('Unexpected processWebhookEvent error', error: e);
   482→      return const FailureResult(Failure.server(
   483→        message: 'Failed to process webhook event.', statusCode: 500,
   484→      ));
   485→    }
   486→  }

   540→  @override
   541→  Future<Result<TransactionEntity>> requestRefund({
   542→    required String transactionId,
   543→    required double amount,
   544→    String? reason,
   545→  }) async {
   546→    try {
   547→      // Get the transaction to find Flutterwave ID
   548→      final txModel = await _remoteDataSource.getTransaction(transactionId);
   549→      final flwTxId = txModel.flutterwaveTransactionId;
   550→
   551→      if (flwTxId == null || flwTxId.isEmpty) {
   552→        return const FailureResult(Failure.validation(
   553→          message: 'Cannot refund: no Flutterwave transaction ID found.',
   554→        ));
   555→      }
   556→
   557→      // Process refund via Flutterwave
   558→      await _flutterwaveDataSource.processRefund(
   559→        transactionId: flwTxId,
   560→        amount: amount,
   561→      );
   562→
   563→      // Update local transaction
   564→      final updatedTx = await _remoteDataSource.updateTransaction(
   565→        transactionId,
   566→        {
   567→          'refund_amount': amount,
   568→          if (reason != null) 'refund_reason': reason,
   569→          'refunded_at': DateTime.now().toIso8601String(),
   570→          'status': amount >= txModel.amount ? 'refunded' : 'partially_refunded',
   571→        },
   572→      );
   573→      return Success(updatedTx.toEntity());
   574→    } on AuthException catch (e) {
   575→      return FailureResult(Failure.auth(message: e.message, code: e.code));
   576→    } on ServerException catch (e) {
   577→      return FailureResult(Failure.server(
   578→        message: e.message, statusCode: e.statusCode, data: e.data,
   579→      ));
   580→    } catch (e) {
   581→      AppLogger.error('Unexpected requestRefund error', error: e);
   582→      return const FailureResult(Failure.server(
   583→        message: 'Failed to process refund.', statusCode: 500,
   584→      ));
   585→    }
   586→  }

   642→  @override
   643→  Future<Result<InvoiceEntity>> generateInvoice({
   644→    required String subscriptionId,
   645→    required List<InvoiceLineItem> lineItems,
   646→  }) async {
   647→    try {
   648→      final lineItemMaps = lineItems.map((item) => {
   649→        'description': item.description,
   650→        'quantity': item.quantity,
   651→        'unit_price': item.unitPrice,
   652→        'total': item.total,
   653→        'tax_rate': item.taxRate,
   654→        'tax_amount': item.taxAmount,
   655→      }).toList();
   656→
   657→      final subtotal = lineItems.fold<double>(0, (sum, i) => sum + i.total);
   658→      final taxAmount = lineItems.fold<double>(0, (sum, i) => sum + i.taxAmount);
   659→      final totalAmount = subtotal + taxAmount;
   660→
   661→      final data = {
   662→        'subscription_id': subscriptionId,
   663→        'line_items': lineItemMaps,
   664→        'subtotal': subtotal,
   665→        'tax_amount': taxAmount,
   666→        'total_amount': totalAmount,
   667→        'due_date': DateTime.now().add(const Duration(days: 30)).toIso8601String(),
   668→      };
   669→      final model = await _remoteDataSource.createInvoice(data);
   670→      return Success(model.toEntity());
   671→    } on AuthException catch (e) {
   672→      return FailureResult(Failure.auth(message: e.message, code: e.code));
   673→    } on ServerException catch (e) {
   674→      return FailureResult(Failure.server(
   675→        message: e.message, statusCode: e.statusCode, data: e.data,
   676→      ));
   677→    } catch (e) {
   678→      AppLogger.error('Unexpected generateInvoice error', error: e);
   679→      return const FailureResult(Failure.server(
   680→        message: 'Failed to generate invoice.', statusCode: 500,
   681→      ));
   682→    }
