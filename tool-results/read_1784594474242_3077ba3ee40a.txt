    12→/// Flashcard study interface page.
    13→///
    14→/// Features:
    15→/// - Deck list with: Title, Card count, Due count, Subject, Favorite toggle, Study button
    16→/// - Create deck dialog
    17→/// - Study mode: Card flip animation, Hint button, Rating buttons, Progress, Session summary
    18→/// - Deck detail: Card list, Add card button, Edit card, Delete card
    19→/// - Generate flashcards from content dialog
    20→class FlashcardPage extends ConsumerStatefulWidget {
    21→  const FlashcardPage({super.key});
    22→
    23→  @override
    24→  ConsumerState<FlashcardPage> createState() => _FlashcardPageState();
    25→}
    26→
    27→class _FlashcardPageState extends ConsumerState<FlashcardPage>
    28→    with SingleTickerProviderStateMixin {
    29→  late final AnimationController _flipController;
    30→
    31→  @override
    32→  void initState() {
    33→    super.initState();
    34→    _flipController = AnimationController(
    35→      vsync: this,
    36→      duration: const Duration(milliseconds: 600),
    37→    );
    38→    WidgetsBinding.instance.addPostFrameCallback((_) {
    39→      ref.read(flashcardProvider.notifier).loadDecks();
    40→    });
    41→  }
    42→
    43→  @override
    44→  void dispose() {
    45→    _flipController.dispose();
    46→    super.dispose();
    47→  }
    48→
    49→  @override
    50→  Widget build(BuildContext context) {
    51→    final flashcardState = ref.watch(flashcardProvider);
    52→
    53→    if (flashcardState.isStudying && flashcardState.currentCard != null) {
    54→      return _buildStudyMode(context, flashcardState);
    55→    }
    56→
    57→    if (flashcardState.currentDeck != null && !flashcardState.isStudying) {
    58→      return _buildDeckDetail(context, flashcardState);
    59→    }
    60→
    61→    return _buildDeckList(context, flashcardState);
    62→  }

    68→  Widget _buildDeckList(BuildContext context, FlashcardState state) {
    69→    final cs = context.colorScheme;
    70→
    71→    return Scaffold(
    72→      appBar: AppBar(title: const Text('Flashcards')),
    73→      body: state.isLoading && state.decks.isEmpty
    74→          ? const Center(child: AppLoadingSpinner())

    76→              ? AppErrorState(

    78→                  title: 'Failed to Load Decks',
    79→                  message: state.error,
    80→                  onRetry: () =>
    81→                      ref.read(flashcardProvider.notifier).loadDecks(),
    82→                )

    84→                  ? AppEmptyState(
    85→                      icon: Icons.style_outlined,
    86→                      title: 'No Flashcard Decks',
    87→                      subtitle:
    88→                          'Create a deck to start studying with spaced repetition.',

   308→                                style: tt.headlineSmall?.copyWith(
   309→                                  fontWeight: AppTypography.wSemiBold,
   310→                                  color: isFront

   312→                                      : cs.onPrimaryContainer,
   313→                                ),

   315→                              ),
   316→                              if (!isFront && card.hint != null) ...[
   317→                                const SizedBox(height: Spacings.lg),
   318→                                Text(
   319→                                  'Hint: ${card.hint}',
   320→                                  style: tt.bodyMedium?.copyWith(

   323→                                    fontStyle: FontStyle.italic,

   334→                ),
   335→              ),
   336→            ),

   339→          // Rating buttons (shown after flip)
