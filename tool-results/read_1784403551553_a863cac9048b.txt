    56→  void _listenForMessages() {
    57→    final state = ref.read(resourceLibraryProvider);
    58→    if (state.successMessage != null) {
    59→      _showSnackBar(state.successMessage!, isError: false);
    60→      ref.read(resourceLibraryProvider.notifier).clearSuccessMessage();

    68→  void _showSnackBar(String message, {required bool isError}) {
    69→    if (!mounted) return;
    70→    final cs = context.colorScheme;
    71→    context.scaffoldMessenger.showSnackBar(
    72→      SnackBar(
    73→        content: Text(message),
    74→        backgroundColor: isError ? cs.error : cs.primary,
    75→        behavior: SnackBarBehavior.floating,
    76→        shape: RoundedRectangleBorder(

    84→  List<TeachingResourceEntity> _applyLocalFilters(
    85→    List<TeachingResourceEntity> resources,
    86→  ) {
    87→    var filtered = resources;
    88→
    89→    // Content type filter
    90→    if (_filterContentType != null) {
    91→      filtered = filtered
    92→          .where((r) => r.resourceType == _filterContentType)
    93→          .toList();
    94→    }
    95→
    96→    // Subject filter
    97→    if (_filterSubject != null) {
    98→      filtered = filtered
    99→          .where((r) => r.subject == _filterSubject)
   100→          .toList();
   101→    }
   102→
   103→    // Date range filter
   104→    if (_filterDateRange != null) {
   105→      filtered = filtered
   106→          .where((r) =>
   107→              r.updatedAt.isAfter(_filterDateRange!.start) &&
   108→              r.updatedAt
   109→                  .isBefore(_filterDateRange!.end.add(const Duration(days: 1))))
   110→          .toList();
   111→    }
   112→
   113→    // Sort
   114→    switch (_sortOption) {
   115→      case _SortOption.recent:
   116→        filtered.sort((a, b) => b.updatedAt.compareTo(a.updatedAt));
   117→      case _SortOption.az:
   118→        filtered.sort((a, b) => a.title.toLowerCase().compareTo(b.title.toLowerCase()));
   119→      case _SortOption.mostUsed:
   120→        // Sort by version (proxy for usage) then by updatedAt
   121→        filtered.sort((a, b) {
   122→          final versionCompare = b.version.compareTo(a.version);
   123→          if (versionCompare != 0) return versionCompare;
   124→          return b.updatedAt.compareTo(a.updatedAt);
   125→        });
   126→    }
   127→
   128→    return filtered;
   129→  }
   130→
   131→  List<String> _getUniqueSubjects(List<TeachingResourceEntity> resources) {
   132→    final subjects = resources
   133→        .where((r) => r.subject != null)
   134→        .map((r) => r.subject!)
   135→        .toSet()
   136→        .toList()
   137→      ..sort();
   138→    return subjects;
   139→  }

   573→        items: items,
   574→        itemLabel: itemLabel,
   575→        selectedValue: value,
   576→        onSelected: onChanged,
   577→      ),
   578→      onDeleted: isSelected ? () => onChanged(null) : null,
   579→      deleteIconColor: isSelected ? cs.onPrimary : null,

   581→      backgroundColor: cs.surfaceContainerLow,
