   979→   979→CREATE OR REPLACE FUNCTION public.get_exam_readiness(p_user_id UUID, p_exam_body_id UUID DEFAULT NULL, p_subject_id UUID DEFAULT NULL)
   980→   980→RETURNS JSONB AS $$
   981→   981→DECLARE
   982→   982→  v_result JSONB;
   983→   983→BEGIN
   984→   984→  SELECT jsonb_build_object(
   985→   985→    'user_id', p_user_id,
   986→   986→    'assessments', (SELECT COALESCE(jsonb_agg(row_to_json(ra.*)), '[]'::jsonb)
   987→   987→      FROM public.readiness_assessments ra
   988→   988→      WHERE ra.user_id = p_user_id
   989→   989→        AND (p_exam_body_id IS NULL OR ra.exam_body_id = p_exam_body_id)
   990→   990→        AND (p_subject_id IS NULL OR ra.subject_id = p_subject_id)
   991→   991→      ORDER BY ra.assessed_at DESC LIMIT 10),
   992→   992→    'overall_readiness', (SELECT ROUND(AVG(readiness_score), 2)
   993→   993→      FROM public.readiness_assessments
   994→   994→      WHERE user_id = p_user_id
   995→   995→        AND (p_exam_body_id IS NULL OR exam_body_id = p_exam_body_id)),
   996→   996→    'weak_areas', (SELECT COALESCE(jsonb_agg(DISTINCT jsonb_array_elements(weak_topics)), '[]'::jsonb)
   997→   997→      FROM public.readiness_assessments
   998→   998→      WHERE user_id = p_user_id AND (p_exam_body_id IS NULL OR exam_body_id = p_exam_body_id)),
   999→   999→    'study_plans', (SELECT COUNT(*) FROM public.study_plans WHERE user_id = p_user_id AND is_active = true),
  1000→  1000→    'total_study_minutes', (SELECT COALESCE(SUM(total_study_minutes), 0) FROM public.study_plans WHERE user_id = p_user_id)
  1001→  1001→  ) INTO v_result;
  1002→  1002→  RETURN v_result;
  1003→  1003→END;
  1004→  1004→$$ LANGUAGE plpgsql STABLE SECURITY DEFINER;
