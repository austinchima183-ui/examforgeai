     1→/// Report Generator — pure computation service for structured report outputs.
     2→///
     3→/// Transforms domain entities (results, exam, and CBT data) into structured
     4→/// [ReportData] objects ready for export to PDF, Excel, or CSV by the
     5→/// datasource layer (Supabase Edge Functions or local generators).
     6→///
     7→/// ## Design Principles
     8→///
     9→/// - **Pure transformation**: Every public method accepts entities and returns
    10→///   a [ReportData] with no I/O, no database access, no side effects.
    11→/// - **Self-contained DTOs**: Output content types are defined in this file to
    12→///   avoid coupling to persistence-layer models.
    13→/// - **Null-safe**: All methods handle missing or partial data gracefully,
    14→///   providing sensible defaults rather than throwing.
    15→/// - **Format-agnostic**: This service produces structured data only; the
    16→///   actual file generation is handled downstream.
    17→///
    18→/// ## Usage
    19→///
    20→/// ```dart
    21→/// final generator = ReportGenerator();
    22→/// final report = generator.generateStudentReport(
    23→///   overallResult: studentOverall,
    24→///   subjectResults: subjectList,
    25→///   studentName: 'John Doe',
    26→///   className: 'SS2A',
    27→///   sessionName: '2025/2026',
    28→///   trendData: trendPoints,
    29→/// );
    30→/// // Pass report to datasource layer for PDF/Excel/CSV generation
    31→/// ```

   574→// REPORT GENERATOR SERVICE
   575→// ═══════════════════════════════════════════════════════════════════════
   576→
   577→/// Generates structured report data from domain entities.
   578→///
   579→/// This service is a pure computation layer — it transforms raw entity data
   580→/// into [ReportData] structures that the datasource layer can then render
   581→/// into PDF, Excel, or CSV files. No file I/O is performed here.
   582→class ReportGenerator {
   583→  // ── Student Report ─────────────────────────────────────────────────
   584→
   585→  /// Creates a comprehensive report for a single student.
   586→  ///
   587→  /// Requires the student's [overallResult] entity, their [subjectResults],
   588→  /// and contextual information like [studentName], [className], and
   589→  /// [sessionName]. Optional [trendData] provides historical performance
   590→  /// for chart rendering.
   591→  ///
   592→  /// The optional [gradeScale] is used to derive the overall grade from the
   593→  /// overall percentage if [StudentOverallResultEntity.overallGrade] is null.
   594→  ReportData generateStudentReport({
   595→    required StudentOverallResultEntity overallResult,
   596→    required List<StudentSubjectResultEntity> subjectResults,
   597→    required String studentName,
   598→    required String className,
   599→    required String sessionName,
   600→    List<TrendDataPoint>? trendData,
   601→    GradeScaleEntity? gradeScale,
   602→  }) {
   603→    // Resolve the overall grade — use the entity's grade first, then
   604→    // fall back to the grade scale if available.
   605→    final overallGrade = overallResult.overallGrade ??
   606→        gradeScale?.applyToPercentage(overallResult.overallPercentage)?.grade ??
   607→        _defaultGrade(overallResult.overallPercentage);
   608→
   609→    // Build subject-by-subject breakdown rows.
   610→    final subjectRows = subjectResults.map((sr) {
   611→      final grade = sr.grade ??
   612→          gradeScale?.applyToPercentage(sr.percentage)?.grade ??
   613→          _defaultGrade(sr.percentage);
   614→      return SubjectResultRow(
   615→        subjectName: sr.subjectId, // Caller should map ID → name upstream
   616→        score: sr.totalMarksObtained,
   617→        percentage: sr.percentage,
   618→        grade: grade,
   619→        classAverage: sr.classAverage,
   620→        position: sr.classPosition,
   621→        trend: sr.performanceTrend.label,
   622→      );
   623→    }).toList();
   624→
   625→    // Collect strengths and weaknesses from subject results.
   626→    final strengths = <String>[
   627→      ...overallResult.metadata['strengths'] as List<dynamic>? ?? [],
   628→      ...subjectResults
   629→          .where((sr) => sr.performanceTrend == PerformanceTrend.improving)
   630→          .map((sr) => 'Strong performance in ${sr.subjectId}'),
   631→    ];
   632→
   633→    final weaknesses = <String>[
   634→      ...overallResult.metadata['weaknesses'] as List<dynamic>? ?? [],
   635→      ...subjectResults
   636→          .where((sr) => sr.performanceTrend == PerformanceTrend.declining)
   637→          .map((sr) => 'Needs improvement in ${sr.subjectId}'),
   638→    ];
   639→
   640→    // AI study recommendations from the overall result entity.
   641→    final recommendations = overallResult.aiStudyRecommendations;
   642→
   643→    // Build trend data from supplied points, or default to a single point.
   644→    final trend = trendData ??
   645→        [
   646→          TrendDataPoint(
   647→            period: sessionName,
   648→            score: overallResult.overallPercentage,
   649→          ),
   650→        ];
   651→
   652→    final content = StudentReportContent(
   653→      studentName: studentName,
   654→      className: className,
   655→      sessionName: sessionName,
   656→      overallScore: overallResult.overallPercentage,
   657→      overallGrade: overallGrade,
   658→      overallGpa: overallResult.overallGpa,
   659→      classPosition: overallResult.classPosition,
   660→      classSize: overallResult.classSize,
   661→      subjectResults: subjectRows,
   662→      strengths: strengths,
   663→      weaknesses: weaknesses,
   664→      recommendations: recommendations,
   665→      teacherComment: overallResult.teacherComment,
   666→      trendData: trend,
   667→    );
   668→
   669→    return ReportData(
   670→      type: ReportType.student,
   671→      title: 'Student Report — $studentName',

   679→  /// Creates a class-wide performance report.
   680→  ///
   681→  /// Accepts a [ClassPerformanceEntity] for the class statistics,
   682→  /// [studentResults] for building the rankings table, and
   683→  /// [subjectResults] for subject performance comparisons.

   699→

   701→      final sr = entry.value;

   703→          gradeScale?.applyToPercentage(sr.overallPercentage)?.grade ??

   715→    // Build subject performance comparison entries.
   716→    final subjectComparisons = <SubjectComparisonEntry>[];

   718→      // Group by subject and compute aggregated stats.
   719→      final bySubject = <String, List<StudentSubjectResultEntity>>{};
   720→      for (final sr in subjectResults) {
   721→        bySubject.putIfAbsent(sr.subjectId, () => []).add(sr);

   724→        final scores = entry.value.map((e) => e.percentage).toList();

   730→          subjectName: subjectNameMap?[entry.key] ?? entry.key,
   731→          classAverage: avg,

   735→              ? (passed / scores.length) * 100

  1063→  /// Formats a score to 1 decimal place (e.g., `85.3`).

  1068→  /// Formats a percentage with 1 decimal place and % sign (e.g., `85.3%`).

  1073→  /// Formats a position as an ordinal string (e.g., `1st`, `2nd`, `3rd`,
  1074→  /// `4th`, `11th`, `21st`).
  1075→  ///
  1076→  /// Returns `N/A` if [position] is null.
  1077→  static String formatPosition(int? position) {
  1078→    if (position == null) return 'N/A';
  1079→    if (position < 1) return position.toString();
  1080→
  1081→    final lastTwo = position % 100;
  1082→    if (lastTwo >= 11 && lastTwo <= 13) return '${position}th';
  1083→
  1084→    switch (position % 10) {
  1085→      case 1:
  1086→        return '${position}st';
  1087→      case 2:
  1088→        return '${position}nd';
  1089→      case 3:
  1090→        return '${position}rd';
  1091→      default:
  1092→        return '${position}th';
  1093→    }
  1094→  }
  1095→
  1096→  /// Formats a duration in seconds to a human-readable string.
  1097→  ///
  1098→  /// Examples:
  1099→  /// - `3612` → `"1h 0m"`
  1100→  /// - `2712` → `"45m 12s"`
  1101→  /// - `60` → `"1m 0s"`
  1102→  /// - `30` → `"0m 30s"`
  1103→  static String formatDuration(int seconds) {
  1104→    if (seconds < 0) seconds = 0;
  1105→    final hours = seconds ~/ 3600;
  1106→    final remainingSeconds = seconds % 3600;
  1107→    final minutes = remainingSeconds ~/ 60;
  1108→    final secs = remainingSeconds % 60;
  1109→
  1110→    if (hours > 0) {
  1111→      return '${hours}h ${minutes}m';
  1112→    }
  1113→    return '${minutes}m ${secs}s';
  1114→  }
  1115→
  1116→  /// Formats a date as `"Jan 15, 2026"`.

  1118→    const months = [

  1120→      'Jul', 'Aug', 'Sep', 'Oct', 'Nov', 'Dec',
  1121→    ];
  1122→    return '${months[date.month]} ${date.day}, ${date.year}';
  1123→  }

  1129→  /// Derives a default letter grade from a percentage when no grade scale

  1148→  /// Computes score distribution buckets from a list of subject results.
  1149→  ///
  1150→  /// Standard ranges: 0–19, 20–39, 40–49, 50–59, 60–69, 70–79, 80–89,
  1151→  /// 90–100.
  1152→  static List<ScoreBucket> _computeScoreBuckets(
  1153→    List<StudentSubjectResultEntity> results,
  1154→  ) {
  1155→    const ranges = [
  1156→      (label: '0–19', min: 0.0, max: 19.99),
  1157→      (label: '20–39', min: 20.0, max: 39.99),
  1158→      (label: '40–49', min: 40.0, max: 49.99),
  1159→      (label: '50–59', min: 50.0, max: 59.99),
  1160→      (label: '60–69', min: 60.0, max: 69.99),
  1161→      (label: '70–79', min: 70.0, max: 79.99),
  1162→      (label: '80–89', min: 80.0, max: 89.99),
  1163→      (label: '90–100', min: 90.0, max: 100.0),
  1164→    ];
  1165→
  1166→    final total = results.length.toDouble();
  1167→    return ranges.map((range) {
  1168→      final count = results
  1169→          .where((r) => r.percentage >= range.min && r.percentage <= range.max)
  1170→          .length;
  1171→      return ScoreBucket(
  1172→        range: range.label,
  1173→        count: count,
  1174→        percentage: total > 0 ? (count / total) * 100 : 0,
  1175→      );
  1176→    }).toList();
  1177→  }

  1179→  /// Determines the most common mastery level among a list of topic
  1180→  /// mastery entries.
  1181→  static String _dominantMasteryLevel(List<TopicMasteryEntity> items) {
  1182→    if (items.isEmpty) return MasteryLevel.notStarted.label;
  1183→    final counts = <MasteryLevel, int>{};
  1184→    for (final item in items) {
  1185→      counts[item.masteryLevel] = (counts[item.masteryLevel] ?? 0) + 1;
  1186→    }
  1187→    final dominant = counts.entries.reduce(
  1188→      (a, b) => a.value >= b.value ? a : b,
  1189→    );
  1190→    return dominant.key.label;
  1191→  }
