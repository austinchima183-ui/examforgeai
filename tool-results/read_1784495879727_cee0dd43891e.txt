   215→/// Root navigator key — owns the top-level navigation stack.
   216→final _rootNavigatorKey = GlobalKey<NavigatorState>();
   217→
   218→/// Shell navigator key — owns the nested navigation stack inside the
   219→/// [DashboardShell]. This allows the shell (sidebar / bottom nav) to
   220→/// remain mounted while inner pages transition.
   221→final _shellNavigatorKey = GlobalKey<NavigatorState>();
   222→
   223→// ═══════════════════════════════════════════════════════════════════════
   224→// APP ROUTER PROVIDER
   225→// ═══════════════════════════════════════════════════════════════════════
   226→
   227→/// Riverpod provider for the application's [GoRouter].
   228→///
   229→/// Watches [authStateProvider] so that the router automatically
   230→/// re-evaluates its [redirect] callback whenever the authentication
   231→/// state changes (sign-in, sign-out, token refresh).
   232→///
   233→/// Usage in `main.dart`:
   234→/// ```dart
   235→/// final router = ref.read(appRouterProvider);
   236→/// MaterialApp.router(routerConfig: router);
   237→/// ```
   238→final appRouterProvider = Provider<GoRouter>((ref) {
   239→  // Watch auth state so the router re-evaluates redirects on change.
   240→  final authState = ref.watch(authStateProvider);
   241→
   242→  // Also watch onboarding state and role so those redirects stay live.
   243→  ref.watch(onboardingCompleteProvider);
   244→  ref.watch(userRoleProvider);
   245→
   246→  return GoRouter(
   247→    navigatorKey: _rootNavigatorKey,
   248→    initialLocation: RouteNames.splash,
   249→    debugLogDiagnostics: true,
   250→
   251→    // ─── Global Redirect ──────────────────────────────────────────
   252→    redirect: (context, state) {
   253→      final currentPath = state.matchedLocation;
   254→      final isAuthenticated = authState.when(
   255→        data: (authData) => authData.session != null,
   256→        loading: () {
   257→          // While loading, only allow splash. Everything else redirects
   258→          // to splash so we don't flash the login page briefly.
   259→          if (currentPath != RouteNames.splash) {
   260→            return RouteNames.splash;
   261→          }
   262→          return null;
   263→        },
   264→        error: (_, __) => false,
   265→      );
   266→
   267→      // ── Auth Guard ────────────────────────────────────────────
   268→      final authRedirect = AuthGuard.evaluate(
   269→        isAuthenticated: isAuthenticated,
   270→        currentPath: currentPath,
   271→      );
   272→      if (authRedirect != null) return authRedirect;
   273→
   274→      // ── Onboarding Guard ─────────────────────────────────────
   275→      final onboardingAsync = ref.read(onboardingCompleteProvider);
   276→      final isOnboardingComplete = onboardingAsync.when(
   277→        data: (complete) => complete,
   278→        loading: () => true, // Assume complete while loading to avoid flicker.
   279→        error: (_, __) => true,
   280→      );
   281→
   282→      final onboardingRedirect = OnboardingGuard.evaluate(
   283→        isAuthenticated: isAuthenticated,
   284→        isOnboardingComplete: isOnboardingComplete,
   285→        currentPath: currentPath,
   286→      );
   287→      if (onboardingRedirect != null) return onboardingRedirect;
   288→
   289→      // ── Role-Based Guard ─────────────────────────────────────
   290→      final roleAsync = ref.read(userRoleProvider);
   291→      final userRole = roleAsync.when(
   292→        data: UserRole.fromString,
   293→        loading: () => null,
   294→        error: (_, __) => null,
   295→      );
   296→
   297→      final roleRedirect = RoleBasedGuard.evaluate(
   298→        userRole: userRole,
   299→        currentPath: currentPath,
   300→      );
   301→      if (roleRedirect != null) return roleRedirect;
   302→
   303→      // No redirect needed — proceed as requested.
   304→      return null;
   305→    },
   306→
   307→    // ─── Route Tree ───────────────────────────────────────────────
   308→    routes: [
   309→      // ── Splash ────────────────────────────────────────────────
   310→      GoRoute(
   311→        path: RouteNames.splash,
   312→        name: 'splash',
   313→        builder: (context, state) => const SplashPage(),
   314→      ),
   315→
   316→      // ── Onboarding ───────────────────────────────────────────
   317→      GoRoute(
   318→        path: RouteNames.onboarding,
   319→        name: 'onboarding',
   320→        builder: (context, state) => const OnboardingPage(),
   321→      ),
   322→
   323→      // ── Auth Routes ──────────────────────────────────────────
   324→      GoRoute(
   325→        path: RouteNames.login,
   326→        name: 'login',
   327→        builder: (context, state) => const LoginPage(),
   328→      ),
   329→      GoRoute(
   330→        path: RouteNames.register,
   331→        name: 'register',
   332→        builder: (context, state) => const RegisterPage(),
   333→      ),
   334→      GoRoute(
   335→        path: RouteNames.forgotPassword,
   336→        name: 'forgotPassword',
   337→        builder: (context, state) => const ForgotPasswordPage(),
   338→      ),
   339→      GoRoute(
   340→        path: RouteNames.verifyEmail,
   341→        name: 'verifyEmail',
   342→        builder: (context, state) => const VerifyEmailPage(),
   343→      ),
   344→      GoRoute(
   345→        path: RouteNames.resetPassword,
   346→        name: 'resetPassword',
   347→        builder: (context, state) => const ResetPasswordPage(),
   348→      ),
   349→
   350→      // ── Protected Shell Route ─────────────────────────────────
   351→      // All routes inside this shell share the [DashboardShell]
   352→      // wrapper which provides the sidebar / bottom nav and app bar.
   353→      ShellRoute(
   354→        navigatorKey: _shellNavigatorKey,
   355→        builder: (context, state, child) => DashboardShell(child: child),
   356→        routes: [
   357→          // Dashboard with role-based sub-routes.
   358→          GoRoute(
   359→            path: RouteNames.dashboard,
   360→            name: 'dashboard',
   361→            builder: (context, state) => const DashboardRedirector(),
   362→            routes: [
   363→              GoRoute(
   364→                path: 'teacher',
   365→                name: 'teacherDashboard',
   366→                builder: (context, state) => const TeacherDashboardPage(),
   367→              ),
   368→              GoRoute(
   369→                path: 'student',
   370→                name: 'studentDashboard',
   371→                builder: (context, state) => const StudentDashboardPage(),
   372→              ),
   373→              GoRoute(
   374→                path: 'school-admin',
   375→                name: 'schoolAdminDashboard',
   376→                builder: (context, state) => const SchoolAdminDashboardPage(),
   377→              ),
   378→              GoRoute(
   379→                path: 'super-admin',
   380→                name: 'superAdminDashboard',
   381→                redirect: (context, state) => RouteNames.superAdminDashboard,
   382→              ),
   383→            ],
   384→          ),
   385→
   386→          // Profile.
   387→          GoRoute(
   388→            path: RouteNames.profile,
   389→            name: 'profile',
   390→            builder: (context, state) => const ProfilePage(),
   391→          ),
   392→
   393→          // Settings.
   394→          GoRoute(
   395→            path: RouteNames.settings,
   396→            name: 'settings',
   397→            builder: (context, state) => const SettingsPage(),
   398→          ),
   399→
   400→          // Notifications.
   401→          GoRoute(
   402→            path: RouteNames.notifications,
   403→            name: 'notifications',
   404→            builder: (context, state) => const NotificationsPage(),
   405→          ),
   406→
   407→          // ── Question Bank Routes ──────────────────────────────────
   408→          GoRoute(
   409→            path: RouteNames.questionBank,
   410→            name: 'questionBank',
   411→            builder: (context, state) => const QuestionBankDashboardPage(),
   412→          ),
