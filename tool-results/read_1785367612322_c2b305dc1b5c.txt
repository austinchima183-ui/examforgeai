   248→  @override
   249→  Future<List<SubscriptionPlanModel>> getSubscriptionPlans({
   250→    String? billingModel,
   251→    bool activeOnly = true,
   252→    int limit = PaginatedQueryMixin.dropdownPageSize,
   253→  }) async {
   254→    try {
   255→      var query = _supabase.from(_plansTable).select();
   256→
   257→      if (billingModel != null) {
   258→        query = query.eq('billing_model', billingModel);
   259→      }
   260→      if (activeOnly) {
   261→        query = query.eq('is_active', true);
   262→      }
   263→
   264→      // PERF: Added limit to prevent unbounded query on subscription_plans
   265→      final response = query.order('sort_order', ascending: true).limit(limit);
   266→
   267→      final list = await response;
   268→      AppLogger.info('Fetched ${list.length} subscription plans');
   269→      return list
   270→          .map<SubscriptionPlanModel>(
   271→            (row) => SubscriptionPlanModel.fromJson(row),
   272→          )
   273→          .toList();
   274→    } on sb.PostgrestException catch (e) {
   275→      throw _mapPostgrestException(e);
   276→    } catch (e) {
   277→      AppLogger.error('Unexpected getSubscriptionPlans error', error: e);
   278→      throw const ServerException(
   279→        message: 'Failed to fetch subscription plans.',
   280→        statusCode: 500,
   281→      );
   282→    }
   283→  }
   284→
   285→  @override
   286→  Future<SubscriptionPlanModel> getSubscriptionPlan(String planId) async {
   287→    try {
   288→      final response = await _supabase
   289→          .from(_plansTable)
   290→          .select()
   291→          .eq('id', planId)
   292→          .limit(1);
   293→
   294→      if (response.isEmpty) {
   295→        throw const NotFoundException(message: 'Subscription plan not found.');
   296→      }
   297→
   298→      return SubscriptionPlanModel.fromJson(response.first);
   299→    } on sb.PostgrestException catch (e) {
   300→      throw _mapPostgrestException(e);
   301→    } on NotFoundException {
   302→      rethrow;
   303→    } catch (e) {
   304→      AppLogger.error('Unexpected getSubscriptionPlan error', error: e);
   305→      throw const ServerException(
   306→        message: 'Failed to fetch subscription plan.',
   307→        statusCode: 500,
   308→      );
   309→    }
   310→  }

   331→      AppLogger.info('Subscription plan upserted: ${response.first['id']}');
   332→      return SubscriptionPlanModel.fromJson(response.first);
   333→    } on sb.PostgrestException catch (e) {
   334→      throw _mapPostgrestException(e);
   335→    } on ServerException {
   336→      rethrow;

   383→  @override

   385→    String? schoolId,
   386→    String? subscriberType,
   387→    String? status,
   388→    int page = 1,
   389→    int perPage = 20,
   390→  }) async {
   391→    try {
   392→      // PERF: Fix duplicate filter application bug
   393→      // Previous code applied filters TWICE: once inline (lines 394-402)
   394→      // then again via _applyFilters (lines 404-408). Now uses only _applyFilters.
   395→      final query = _supabase.from(_subscriptionsTable).select(
   396→        'id, subscriber_id, subscriber_type, plan_id, '
   397→        'status, current_period_start, current_period_end, created_at',

   399→
   400→      final filters = <String, String>{};
   401→      if (schoolId != null) filters['school_id'] = schoolId;

   405→      final filtered = _applyFilters(query, filters);

   407→      var transformed =
   408→          filtered.order('created_at', ascending: false);
   409→      transformed = _applyPagination(transformed, page: page, perPage: perPage);
   410→
   411→      final list = await transformed;
   412→      AppLogger.info('Fetched ${list.length} subscriptions');
   413→      return list
   414→          .map<SubscriptionModel>(
   415→            (row) => SubscriptionModel.fromJson(row),
   416→          )
   417→          .toList();
   418→    } on sb.PostgrestException catch (e) {

  1057→  @override
  1058→  Future<CouponModel> validateCoupon({
  1059→    required String code,
  1060→    required String billingModel,
  1061→    String? planId,
  1062→  }) async {
  1063→    try {
  1064→      var query = _supabase
  1065→          .from(_couponsTable)
  1066→          .select()
  1067→          .eq('code', code)
  1068→          .eq('billing_model', billingModel)
  1069→          .eq('is_active', true) as sb.PostgrestFilterBuilder<dynamic>;
  1070→
  1071→      if (planId != null) {
  1072→        query = query.eq('plan_id', planId);
  1073→      }
  1074→
  1075→      final response = await query.limit(1);
  1076→
  1077→      if (response.isEmpty) {
  1078→        throw const NotFoundException(message: 'Invalid or expired coupon code.');
  1079→      }
  1080→
  1081→      final couponData = response.first;
  1082→
  1083→      // Validate expiry
  1084→      final validUntil = couponData['valid_until'] != null
  1085→          ? DateTime.tryParse(couponData['valid_until'] as String)
  1086→          : null;
  1087→      if (validUntil != null && validUntil.isBefore(DateTime.now())) {
  1088→        throw const ValidationException(
  1089→          message: 'This coupon has expired.',
  1090→          fieldErrors: {'code': 'Coupon has expired.'},
  1091→        );
  1092→      }
  1093→
  1094→      final validFrom = couponData['valid_from'] != null
  1095→          ? DateTime.tryParse(couponData['valid_from'] as String)
  1096→          : null;
  1097→      if (validFrom != null && validFrom.isAfter(DateTime.now())) {
  1098→        throw const ValidationException(
  1099→          message: 'This coupon is not yet valid.',
  1100→          fieldErrors: {'code': 'Coupon is not yet valid.'},
  1101→        );
  1102→      }
  1103→
  1104→      // Validate redemption count
  1105→      final maxRedemptions = couponData['max_redemptions'] as int?;
  1106→      final currentRedemptions =
  1107→          couponData['current_redemptions'] as int? ?? 0;
  1108→      if (maxRedemptions != null &&
  1109→          currentRedemptions >= maxRedemptions) {
  1110→        throw const ValidationException(
  1111→          message: 'This coupon has reached its maximum redemptions.',
  1112→          fieldErrors: {'code': 'Coupon redemption limit reached.'},
  1113→        );
  1114→      }
  1115→
  1116→      AppLogger.info('Coupon validated: $code');
  1117→      return CouponModel.fromJson(couponData);
  1118→    } on sb.PostgrestException catch (e) {
  1119→      throw _mapPostgrestException(e);
  1120→    } on NotFoundException {
  1121→      rethrow;
  1122→    } on ValidationException {
  1123→      rethrow;
  1124→    } catch (e) {
  1125→      AppLogger.error('Unexpected validateCoupon error', error: e);
  1126→      throw const ServerException(
  1127→        message: 'Failed to validate coupon.',
  1128→        statusCode: 500,
  1129→      );
  1130→    }
  1131→  }

  1874→  /// Applies pagination using Supabase range.
  1875→  sb.PostgrestTransformBuilder<dynamic> _applyPagination(
  1876→    sb.PostgrestTransformBuilder<dynamic> query, {
  1877→    int? page,
  1878→    int? perPage,
  1879→  }) {
  1880→    final p = page ?? 1;
  1881→    final pp = perPage ?? 20;
  1882→    final from = (p - 1) * pp;
  1883→    final to = from + pp - 1;

  1887→  /// Maps a Supabase [sb.PostgrestException] to a domain exception.
  1888→  Exception _mapPostgrestException(sb.PostgrestException e) {
  1889→    final statusCode = e.code != null ? int.tryParse(e.code!) ?? 0 : 0;
  1890→    final message = e.message ?? 'An unexpected database error occurred.';
  1891→
  1892→    AppLogger.warning(
  1893→      'Supabase PostgrestException — code: ${e.code}, message: $message',
  1894→    );
  1895→
  1896→    switch (statusCode) {
  1897→      case 401:
  1898→        return UnauthorizedException(message: message);
  1899→      case 403:
  1900→        return ForbiddenException(message: message);
  1901→      case 404:
  1902→        return NotFoundException(message: message);
  1903→      case 422:
  1904→        return ValidationException(
  1905→          message: message,
  1906→          fieldErrors: e.details is Map<String, dynamic>
  1907→              ? e.details as Map<String, String>
  1908→              : {},
  1909→        );
  1910→      default:
  1911→        return ServerException(
  1912→          message: message,
  1913→          statusCode: statusCode,
  1914→          data: e.details,
  1915→        );
  1916→    }
  1917→  }
