   460→-- TEACHING STATISTICS MATERIALIZED VIEW
   461→-- ═══════════════════════════════════════════════════════════════════════════════
   462→
   463→CREATE MATERIALIZED VIEW IF NOT EXISTS teacher_statistics AS
   464→SELECT
   465→  t.teacher_id,
   466→  t.school_id,
   467→  COUNT(DISTINCT lp.id) AS total_lesson_plans,
   468→  COUNT(DISTINCT lp.id) FILTER (WHERE lp.is_published) AS published_lesson_plans,
   469→  COUNT(DISTINCT sw.id) AS total_schemes,
   470→  COUNT(DISTINCT ws.id) AS total_worksheets,
   471→  COUNT(DISTINCT wa.id) AS total_assignments,
   472→  COUNT(DISTINCT wa.id) FILTER (WHERE wa.status = 'published') AS published_assignments,
   473→  COUNT(DISTINCT p.id) AS total_presentations,
   474→  COUNT(DISTINCT r.id) AS total_rubrics,
   475→  COUNT(DISTINCT tr.id) FILTER (WHERE tr.is_favorite) AS favorite_resources,
   476→  COUNT(DISTINCT ach.id) AS total_ai_generations,
   477→  SUM(COALESCE(ach.tokens_used, 0)) AS total_tokens_used,
   478→  COUNT(DISTINCT tsk.id) FILTER (WHERE tsk.status = 'completed') AS completed_tasks,
   479→  COUNT(DISTINCT tsk.id) FILTER (WHERE tsk.status = 'pending') AS pending_tasks,
   480→  COUNT(DISTINCT tsk.id) FILTER (WHERE tsk.due_date < now() AND tsk.status = 'pending') AS overdue_tasks
   481→FROM (
   482→  SELECT DISTINCT teacher_id, school_id FROM lesson_plans
   483→  UNION DISTINCT SELECT teacher_id, school_id FROM schemes_of_work
   484→  UNION DISTINCT SELECT teacher_id, school_id FROM worksheets
   485→) t
   486→LEFT JOIN lesson_plans lp ON lp.teacher_id = t.teacher_id
   487→LEFT JOIN schemes_of_work sw ON sw.teacher_id = t.teacher_id
   488→LEFT JOIN worksheets ws ON ws.teacher_id = t.teacher_id
   489→LEFT JOIN workspace_assignments wa ON wa.teacher_id = t.teacher_id
   490→LEFT JOIN presentations p ON p.teacher_id = t.teacher_id
   491→LEFT JOIN rubrics r ON r.teacher_id = t.teacher_id
   492→LEFT JOIN teaching_resources tr ON tr.teacher_id = t.teacher_id
   493→LEFT JOIN ai_content_history ach ON ach.teacher_id = t.teacher_id
   494→LEFT JOIN tasks tsk ON tsk.teacher_id = t.teacher_id
   495→GROUP BY t.teacher_id, t.school_id;
   496→
   497→CREATE UNIQUE INDEX idx_teacher_statistics_teacher ON teacher_statistics(teacher_id);

   760→-- Auto-create version snapshot on presentation update
   761→CREATE OR REPLACE FUNCTION create_presentation_version()
   762→RETURNS TRIGGER AS $$
   763→BEGIN
   764→  IF OLD.slides IS DISTINCT FROM NEW.slides OR OLD.title IS DISTINCT FROM NEW.title THEN
   765→    INSERT INTO presentation_versions (presentation_id, version_number, snapshot, change_summary, created_by)
   766→    VALUES (
   767→      NEW.id,
   768→      COALESCE((SELECT MAX(version_number) FROM presentation_versions WHERE presentation_id = NEW.id), 0) + 1,
   769→      jsonb_build_object(
   770→        'title', NEW.title,
   771→        'description', NEW.description,
   772→        'slides', NEW.slides,
   773→        'speaker_notes', NEW.speaker_notes
   774→      ),
   775→      'Auto-saved version',
   776→      NEW.teacher_id
   777→    );
   778→    NEW.version := NEW.version + 1;
   779→  END IF;
   780→  RETURN NEW;
   781→END;
   782→$$ LANGUAGE plpgsql SECURITY DEFINER;

   785→  BEFORE UPDATE ON presentations

   792→-- Get enhanced workspace dashboard data
   793→CREATE OR REPLACE FUNCTION get_enhanced_workspace_dashboard(p_teacher_id UUID)
   794→RETURNS JSONB AS $$
   795→DECLARE
   796→  v_result JSONB;
   797→  v_school_id UUID;
   798→BEGIN
   799→  SELECT school_id INTO v_school_id FROM users WHERE id = p_teacher_id;
   800→
   801→  SELECT jsonb_build_object(
   802→    'stats', jsonb_build_object(
   803→      'lesson_plans', (SELECT COUNT(*) FROM lesson_plans WHERE teacher_id = p_teacher_id AND is_archived = FALSE),
   804→      'worksheets', (SELECT COUNT(*) FROM worksheets WHERE teacher_id = p_teacher_id),
   805→      'assignments', (SELECT COUNT(*) FROM workspace_assignments WHERE teacher_id = p_teacher_id),
   806→      'presentations', (SELECT COUNT(*) FROM presentations WHERE teacher_id = p_teacher_id AND is_archived = FALSE),
   807→      'rubrics', (SELECT COUNT(*) FROM rubrics WHERE teacher_id = p_teacher_id AND is_archived = FALSE),
   808→      'resources', (SELECT COUNT(*) FROM teaching_resources WHERE teacher_id = p_teacher_id),
   809→      'ai_generations', (SELECT COUNT(*) FROM ai_content_history WHERE teacher_id = p_teacher_id),
   810→      'pending_tasks', (SELECT COUNT(*) FROM tasks WHERE teacher_id = p_teacher_id AND status = 'pending'),
   811→      'overdue_tasks', (SELECT COUNT(*) FROM tasks WHERE teacher_id = p_teacher_id AND status = 'pending' AND due_date < now()),
   812→      'shared_with_me', (SELECT COUNT(*) FROM shared_resources WHERE shared_with = p_teacher_id AND is_accepted = TRUE)
   813→    ),
   814→    'today_classes', (
   815→      SELECT COALESCE(jsonb_agg(
   816→        jsonb_build_object(
   817→          'id', ce.id,
   818→          'title', ce.title,
   819→          'start_time', ce.start_time,
   820→          'end_time', ce.end_time,
   821→          'event_type', ce.event_type
   822→        )
   823→      ), '[]'::jsonb)
   824→      FROM calendar_events ce
   825→      WHERE ce.teacher_id = p_teacher_id
   826→      AND ce.start_time::date = CURRENT_DATE
   827→      ORDER BY ce.start_time
   828→    ),
   829→    'pending_assignments', (
   830→      SELECT COALESCE(jsonb_agg(
   831→        jsonb_build_object(
   832→          'id', wa.id,
   833→          'title', wa.title,
   834→          'deadline', wa.deadline,
   835→          'status', wa.status
   836→        )
   837→      ), '[]'::jsonb)
   838→      FROM workspace_assignments wa
   839→      WHERE wa.teacher_id = p_teacher_id
   840→      AND wa.status IN ('draft', 'published')
   841→      ORDER BY wa.deadline NULLS LAST
   842→      LIMIT 5
   843→    ),
   844→    'recent_documents', (
   845→      SELECT COALESCE(jsonb_agg(subq), '[]'::jsonb) FROM (
   846→        SELECT jsonb_build_object('id', id, 'title', title, 'type', 'lesson_plan', 'updated_at', updated_at) FROM lesson_plans WHERE teacher_id = p_teacher_id ORDER BY updated_at DESC LIMIT 2
   847→        UNION ALL
   848→        SELECT jsonb_build_object('id', id, 'title', title, 'type', 'worksheet', 'updated_at', updated_at) FROM worksheets WHERE teacher_id = p_teacher_id ORDER BY updated_at DESC LIMIT 2
   849→        UNION ALL
   850→        SELECT jsonb_build_object('id', id, 'title', title, 'type', 'presentation', 'updated_at', updated_at) FROM presentations WHERE teacher_id = p_teacher_id AND is_archived = FALSE ORDER BY updated_at DESC LIMIT 2
   851→        ORDER BY updated_at DESC LIMIT 6
   852→      ) subq
   853→    ),
   854→    'saved_templates', (
   855→      SELECT COALESCE(jsonb_agg(
   856→        jsonb_build_object('id', id, 'name', name, 'template_type', template_type, 'usage_count', usage_count)
   857→      ), '[]'::jsonb)
   858→      FROM workspace_templates
   859→      WHERE teacher_id = p_teacher_id OR is_public = TRUE
   860→      ORDER BY usage_count DESC LIMIT 6
   861→    ),
   862→    'upcoming_events', (
   863→      SELECT COALESCE(jsonb_agg(
   864→        jsonb_build_object('id', id, 'title', title, 'start_time', start_time, 'end_time', end_time, 'event_type', event_type)
   865→      ), '[]'::jsonb)
   866→      FROM calendar_events
   867→      WHERE teacher_id = p_teacher_id
   868→      AND start_time > now()
   869→      ORDER BY start_time LIMIT 5
   870→    ),
   871→    'teaching_statistics', (
   872→      SELECT jsonb_build_object(
   873→        'total_students', COALESCE(SUM(jsonb_array_length(COALESCE(students, '[]'::jsonb))), 0),
   874→        'classes_taught', (SELECT COUNT(DISTINCT class_id) FROM class_subjects WHERE teacher_id = p_teacher_id),
   875→        'questions_generated', (SELECT COUNT(*) FROM ai_content_history WHERE teacher_id = p_teacher_id AND action = 'generate_questions'),
   876→        'resources_shared', (SELECT COUNT(*) FROM shared_resources WHERE shared_by = p_teacher_id)
   877→      )
   878→      FROM classes c WHERE c.id IN (SELECT class_id FROM class_subjects WHERE teacher_id = p_teacher_id)
   879→    ),
   880→    'notifications_count', (
   881→      SELECT COUNT(*) FROM shared_resources WHERE shared_with = p_teacher_id AND is_accepted IS NULL
   882→    )
   883→  ) INTO v_result;
   884→
   885→  RETURN v_result;
   886→END;
   887→$$ LANGUAGE plpgsql SECURITY DEFINER;
