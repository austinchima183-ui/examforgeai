   189→            ),

   584→  Widget _buildImportResult(BuildContext context, ImportExportState state) {
   585→    final cs = context.colorScheme;
   586→    final tt = context.textTheme;
   587→    final isDark = context.isDarkMode;
   588→    final importStatus = state.importStatus;
   589→
   590→    final isSuccess = state.isImportComplete;
   591→    final importedCount = importStatus?.importedCount ?? 0;
   592→    final failedCount = importStatus?.failedCount ?? 0;
   593→    final total = importStatus?.totalQuestions ?? 0;
   594→
   595→    return AppCard(
   596→      padding: const EdgeInsets.all(Spacings.lg),
   597→      child: Column(
   598→        crossAxisAlignment: CrossAxisAlignment.start,
   599→        children: [
   600→          // Header
   601→          Row(
   602→            children: [
   603→              Icon(
   604→                isSuccess
   605→                    ? Icons.check_circle_rounded
   606→                    : Icons.error_outline_rounded,
   607→                size: Spacings.lgIcon,
   608→                color: isSuccess
   609→                    ? AppColors.successOf(cs.brightness)
   610→                    : AppColors.errorOf(cs.brightness),
   611→              ),
   612→              const SizedBox(width: Spacings.md),
   613→              Text(
   614→                isSuccess ? 'Import Complete' : 'Import Failed',
   615→                style: tt.titleMedium?.copyWith(
   616→                  fontWeight: AppTypography.wSemiBold,
   617→                  color: cs.onSurface,
   618→                ),
   619→              ),
   620→            ],
   621→          ),
   622→          const SizedBox(height: Spacings.lg),
   623→
   624→          // Summary
   625→          Row(
   626→            children: [
   627→              Expanded(
   628→                child: _resultStat(
   629→                  context,
   630→                  label: 'Total',
   631→                  value: '$total',
   632→                  color: cs.onSurface,
   633→                ),
   634→              ),
   635→              Expanded(
   636→                child: _resultStat(
   637→                  context,
   638→                  label: 'Imported',
   639→                  value: '$importedCount',
   640→                  color: AppColors.success,
   641→                ),
   642→              ),
   643→              Expanded(
   644→                child: _resultStat(
   645→                  context,
   646→                  label: 'Failed',
   647→                  value: '$failedCount',
   648→                  color: failedCount > 0 ? AppColors.error : cs.onSurfaceVariant,
   649→                ),
   650→              ),
   651→            ],
   652→          ),
   653→
   654→          // Error details
   655→          if (failedCount > 0 && importStatus?.errors != null) ...[

   741→  Widget _resultStat(
   742→    BuildContext context, {
   743→    required String label,
   744→    required String value,
   745→    required Color color,

   749→    return Container(

   753→        borderRadius: BorderRadius.circular(Spacings.smRadius),
   754→      ),
   755→      child: Column(
   756→        children: [
   757→          Text(
   758→            value,
   759→            style: tt.headlineSmall?.copyWith(
   760→              fontWeight: AppTypography.wBold,
   761→              color: color,
   762→            ),
   763→          ),
   764→          Text(
   765→            label,
   766→            style: tt.bodySmall?.copyWith(color: color),
   767→          ),
   768→        ],
   769→      ),
   770→    );
   771→  }

   773→  Widget _tableCell(BuildContext context, String text,
   774→      {bool isHeader = false}) {
   775→    final tt = context.textTheme;
   776→    return Padding(
   777→      padding: const EdgeInsets.all(Spacings.sm),
   778→      child: Text(
   779→        text,
   780→        style: (isHeader ? tt.labelSmall : tt.bodySmall)?.copyWith(
   781→          fontWeight: isHeader ? AppTypography.wSemiBold : AppTypography.wRegular,
   782→          color: isHeader
   783→              ? context.colorScheme.onSurface
   784→              : context.colorScheme.onSurfaceVariant,
   785→        ),
   786→        maxLines: 2,
   787→        overflow: TextOverflow.ellipsis,
   788→      ),
   789→    );
   790→  }

   794→  void _pickFile() {
   795→    // In production, use file_picker package
   796→    setState(() {
   797→      _hasFile = true;
   798→      _selectedFileName =
   799→          'questions_import.${_selectedFormat == 'excel' ? 'xlsx' : _selectedFormat}';
   800→    });
   801→  }
   802→
   803→  void _startImport() {
   804→    ref.read(_importExportProvider.notifier).startImport(
   805→          ImportQuestionsParams(
   806→            source: _selectedFormat,
   807→            fileName: _selectedFileName,
   808→            defaultSubjectId: _subjectController.text.isNotEmpty
   809→                ? _subjectController.text
   810→                : null,
   811→            defaultClassId: _classController.text.isNotEmpty
   812→                ? _classController.text
   813→                : null,
   814→            defaultTopicId: _topicController.text.isNotEmpty
   815→                ? _topicController.text
   816→                : null,
   817→          ),
   818→        );
   819→  }
   820→
   821→  void _resetImport() {
   822→    ref.read(_importExportProvider.notifier).resetState();
   823→    setState(() {
   824→      _hasFile = false;
   825→      _selectedFileName = null;
   826→    });
   827→  }
