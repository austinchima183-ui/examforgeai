    85→    85→
    86→    86→/// Riverpod provider for the application's [GoRouter].
    87→    87→///
    88→    88→/// Watches [authStateProvider] so that the router automatically
    89→    89→/// re-evaluates its [redirect] callback whenever the authentication
    90→    90→/// state changes (sign-in, sign-out, token refresh).
    91→    91→///
    92→    92→/// Usage in `main.dart`:
    93→    93→/// ```dart
    94→    94→/// final router = ref.read(appRouterProvider);
    95→    95→/// MaterialApp.router(routerConfig: router);
    96→    96→/// ```
    97→    97→final appRouterProvider = Provider<GoRouter>((ref) {
    98→    98→  // Watch auth state so the router re-evaluates redirects on change.
    99→    99→  final authState = ref.watch(authStateProvider);
   100→   100→
   101→   101→  // Also watch onboarding state and role so those redirects stay live.
   102→   102→  ref.watch(onboardingCompleteProvider);
   103→   103→  ref.watch(userRoleProvider);
   104→   104→
   105→   105→  return GoRouter(
   106→   106→    navigatorKey: _rootNavigatorKey,
   107→   107→    initialLocation: RouteNames.splash,
   108→   108→    debugLogDiagnostics: true,
   109→   109→
   110→   110→    // ─── Global Redirect ──────────────────────────────────────────
   111→   111→    redirect: (context, state) {
   112→   112→      final currentPath = state.matchedLocation;
   113→   113→      final isAuthenticated = authState.when(
   114→   114→        data: (authData) => authData.session != null,
   115→   115→        loading: () {
   116→   116→          // While loading, only allow splash. Everything else redirects
   117→   117→          // to splash so we don't flash the login page briefly.
   118→   118→          if (currentPath != RouteNames.splash) {
   119→   119→            return RouteNames.splash;
   120→   120→          }
   121→   121→          return null;
   122→   122→        },
   123→   123→        error: (_, __) => false,
   124→   124→      );
   125→   125→
   126→   126→      // ── Auth Guard ────────────────────────────────────────────
   127→   127→      final authRedirect = AuthGuard.evaluate(
   128→   128→        isAuthenticated: isAuthenticated,
   129→   129→        currentPath: currentPath,
   130→   130→      );
   131→   131→      if (authRedirect != null) return authRedirect;
   132→   132→
   133→   133→      // ── Onboarding Guard ─────────────────────────────────────
   134→   134→      final onboardingAsync = ref.read(onboardingCompleteProvider);
   135→   135→      final isOnboardingComplete = onboardingAsync.when(
   136→   136→        data: (complete) => complete,
   137→   137→        loading: () => true, // Assume complete while loading to avoid flicker.
   138→   138→        error: (_, __) => true,
   139→   139→      );
   140→   140→
   141→   141→      final onboardingRedirect = OnboardingGuard.evaluate(
   142→   142→        isAuthenticated: isAuthenticated,
   143→   143→        isOnboardingComplete: isOnboardingComplete,
   144→   144→        currentPath: currentPath,
   145→   145→      );
   146→   146→      if (onboardingRedirect != null) return onboardingRedirect;
   147→   147→
   148→   148→      // ── Role-Based Guard ─────────────────────────────────────
   149→   149→      final roleAsync = ref.read(userRoleProvider);
   150→   150→      final userRole = roleAsync.when(
   151→   151→        data: UserRole.fromString,
   152→   152→        loading: () => null,
   153→   153→        error: (_, __) => null,
   154→   154→      );
   155→   155→
   156→   156→      final roleRedirect = RoleBasedGuard.evaluate(
   157→   157→        userRole: userRole,
   158→   158→        currentPath: currentPath,
   159→   159→      );
   160→   160→      if (roleRedirect != null) return roleRedirect;
   161→   161→
   162→   162→      // No redirect needed — proceed as requested.
   163→   163→      return null;
   164→   164→    },
