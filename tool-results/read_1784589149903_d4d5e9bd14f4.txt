   262→   262→
   263→   263→/// Provides a configured [Dio] instance with base options pulled from
   264→   264→/// [AppConfig] and [ApiConstants].
   265→   265→final dioProvider = Provider<Dio>((ref) {
   266→   266→  final dio = Dio(
   267→   267→    BaseOptions(
   268→   268→      baseUrl: AppConfig.baseUrl,
   269→   269→      connectTimeout: AppConfig.connectTimeoutDuration,
   270→   270→      receiveTimeout: AppConfig.receiveTimeoutDuration,
   271→   271→      sendTimeout: AppConfig.sendTimeoutDuration,
   272→   272→      headers: ApiConstants.baseHeaders(),
   273→   273→      responseType: ResponseType.json,
   274→   274→      validateStatus: (status) => status != null && status < 500,
   275→   275→    ),
   276→   276→  );
   277→   277→  return dio;
   278→   278→});
   279→   279→
   280→   280→/// Provides the [ApiClient] wrapper around [Dio].
   281→   281→///
   282→   282→/// FIX: Now injects [StorageService] so that auth tokens are actually
   283→   283→/// read from secure storage instead of returning null (the root cause
   284→   284→/// of the "ApiClient auth always returning null" bug).
   285→   285→final apiClientProvider = Provider<ApiClient>((ref) {
   286→   286→  final dio = ref.watch(dioProvider);
   287→   287→  final storageService = ref.watch(storageServiceProvider);
   288→   288→  return ApiClient(dio, storageService: storageService);
   289→   289→});
