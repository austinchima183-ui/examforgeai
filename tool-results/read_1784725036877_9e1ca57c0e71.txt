   111→  /// Loads all dashboard data for the given school and session.
   112→  Future<void> loadDashboard({
   113→    required String schoolId,
   114→    required String classId,
   115→    required String academicSessionId,
   116→  }) async {
   117→    state = state.copyWith(isLoading: true, error: null);
   118→
   119→    // Load grade scales
   120→    final scalesResult = await _getGradeScalesUseCase(schoolId, isActive: true);
   121→
   122→    scalesResult.fold(
   123→      onSuccess: (scales) {
   124→        state = state.copyWith(gradeScales: scales);
   125→      },
   126→      onFailure: (failure) {
   127→        AppLogger.warning('Failed to load grade scales: $failure');
   128→      },
   129→    );
   130→
   131→    // Load class performance
   132→    final classResult = await _getClassPerformanceUseCase(
   133→      classId: classId,
   134→      academicSessionId: academicSessionId,
   135→    );
   136→
   137→    classResult.fold(
   138→      onSuccess: (performance) {
   139→        state = state.copyWith(classPerformance: performance);
   140→      },
   141→      onFailure: (failure) {
   142→        AppLogger.warning('Failed to load class performance: $failure');
   143→      },
   144→    );
   145→
   146→    // Load school performance
   147→    final schoolResult = await _getSchoolPerformanceUseCase(
   148→      schoolId: schoolId,
   149→      academicSessionId: academicSessionId,
   150→    );
   151→
   152→    schoolResult.fold(
   153→      onSuccess: (performance) {
   154→        state = state.copyWith(
   155→          schoolPerformance: performance,
   156→          isLoading: false,
   157→          error: null,
   158→        );
   159→        AppLogger.info('Dashboard data loaded successfully');
   160→      },
   161→      onFailure: (failure) {
   162→        state = state.copyWith(
   163→          isLoading: false,
   164→          error: _mapFailureToMessage(failure),
   165→        );
   166→        AppLogger.warning('Failed to load school performance: $failure');
   167→      },
   168→    );
   169→  }
