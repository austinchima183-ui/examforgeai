    39→  RealtimeChannel(
    40→    this.topic,
    41→    this.socket, {
    42→    RealtimeChannelConfig params = const RealtimeChannelConfig(),
    43→  })  : _timeout = socket.timeout,
    44→        params = params.toMap(),
    45→        subTopic = topic.replaceFirst(
    46→            RegExp(r"^realtime:", caseSensitive: false), "") {
    47→    broadcastEndpointURL = '${httpEndpointURL(socket.endPoint)}/api/broadcast';
    48→    _private = params.private;
    49→
    50→    joinPush = Push(
    51→      this,
    52→      ChannelEvents.join,
    53→      this.params,
    54→      _timeout,
    55→    );
    56→    _rejoinTimer =
    57→        RetryTimer(() => rejoinUntilConnected(), socket.reconnectAfterMs);
    58→    joinPush.receive('ok', (_) {
    59→      _state = ChannelStates.joined;
    60→      _rejoinTimer.reset();
    61→      for (final pushEvent in _pushBuffer) {
    62→        pushEvent.send();
    63→      }
    64→      _pushBuffer = [];
    65→    });
    66→
    67→    _onClose(() {
    68→      _rejoinTimer.reset();
    69→      socket.log('channel', 'close $topic $joinRef');
    70→      _state = ChannelStates.closed;
    71→      socket.remove(this);
    72→    });
    73→
    74→    _onError((reason) {
    75→      if (isLeaving || isClosed) {
    76→        return;
    77→      }
    78→      socket.log('channel', 'error $topic', reason);
    79→      _state = ChannelStates.errored;
    80→      _rejoinTimer.scheduleTimeout();
    81→    });
    82→
    83→    joinPush.receive('timeout', (_) {
    84→      if (!isJoining) {
    85→        return;
    86→      }
    87→      socket.log('channel', 'timeout $topic', joinPush.timeout);
    88→      _state = ChannelStates.errored;
    89→      _rejoinTimer.scheduleTimeout();
    90→    });
    91→
    92→    onEvents(ChannelEvents.reply.eventName(), ChannelFilter(), (payload,
    93→        [ref]) {
    94→      trigger(replyEventName(ref), payload);
    95→    });
    96→
    97→    presence = RealtimePresence(this);
    98→  }

   104→      rejoin();

   108→  bool _shouldEnablePresence() {

   110→        (params['config']['presence']['enabled'] == true);

   113→  void _handlePresenceUpdate() {
   114→    if (joinedOnce && isJoined) {
   115→      final currentPresenceEnabled = params['config']['presence']['enabled'];
   116→      final shouldEnablePresence = _shouldEnablePresence();
   117→
   118→      if (!currentPresenceEnabled && shouldEnablePresence) {
   119→        final config = Map<String, dynamic>.from(params['config']);
   120→        config['presence'] = Map<String, dynamic>.from(config['presence']);
   121→        config['presence']['enabled'] = true;
   122→        params['config'] = config;
   123→        updateJoinPayload({'config': config});
   124→        rejoin();
   125→      }
   126→    }
   127→  }

   129→  /// Subscribes to receive real-time changes
   130→  ///
   131→  /// Pass a [callback] to react to different status changes.
   132→  ///
   133→  /// [timeout] parameter can be used to override the default timeout set on [RealtimeClient].
   134→  RealtimeChannel subscribe([
   135→    void Function(RealtimeSubscribeStatus status, Object? error)? callback,
   136→    Duration? timeout,
   137→  ]) {
   138→    if (!socket.isConnected) {
   139→      unawaited(socket.connect());
   140→    }
   141→    if (joinedOnce == true) {
   142→      throw "tried to subscribe multiple times. 'subscribe' can only be called a single time per channel instance";
   143→    }
   144→    final broadcast = params['config']['broadcast'];
   145→    final presenceConfig = params['config']['presence'];
   146→    final isPrivate = params['config']['private'];
   147→
   148→    _onError((e) {
   149→      if (callback != null) callback(RealtimeSubscribeStatus.channelError, e);
   150→    });
   151→    _onClose(() {
   152→      if (callback != null) callback(RealtimeSubscribeStatus.closed, null);
   153→    });
   154→
   155→    // A `postgres_changes` subscription's replication setup happens
   156→    // asynchronously on the server, AFTER the phoenix join succeeds. The join
   157→    // reply optimistically echoes the requested config (with server-assigned
   158→    // binding ids), so the join is reported as `subscribed` before the
   159→    // replication is actually live. If that setup then fails (e.g. RLS denies
   160→    // under a stale/expired token, or the server declines), the verdict
   161→    // arrives on a later `system` event with status `error` -- which is
   162→    // otherwise only exposed via `onSystemEvents` (for debugging) and never
   163→    // reaches this status callback. The result is a channel that reports
   164→    // `subscribed` yet delivers nothing. Forward it as `channelError` so
   165→    // consumers can detect and recover from the failed subscription.
   166→    onSystemEvents((payload) {
   167→      if (payload is Map && payload['status'] == 'error') {
   168→        if (callback != null) {
   169→          callback(
   170→            RealtimeSubscribeStatus.channelError,
   171→            Exception(payload['message']?.toString() ??
   172→                'postgres_changes subscription failed'),
   173→          );
   174→        }
   175→      }
   176→    });
   177→
   178→    final presenceEnabled = _shouldEnablePresence();
   179→
   180→    final accessTokenPayload = <String, String>{};
   181→    final config = {
   182→      'broadcast': broadcast,
   183→      'presence': {...presenceConfig, 'enabled': presenceEnabled},
   184→      'postgres_changes':
   185→          _bindings['postgres_changes']?.map((r) => r.filter).toList() ?? [],
   186→      'private': isPrivate == true,
   187→    };
   188→
   189→    if (socket.accessToken != null) {
   190→      accessTokenPayload['access_token'] = socket.accessToken!;
   191→    }
   192→
   193→    updateJoinPayload({'config': config, ...accessTokenPayload});
   194→
   195→    joinedOnce = true;
   196→    rejoin(timeout ?? _timeout);
   197→
   198→    joinPush
   199→        .receive(
   200→      'ok',
   201→      (response) =>
   202→          unawaited(_handleJoinOk(response as Map<String, dynamic>, callback)),
   203→    )
   204→        .receive('error', (error) {
   205→      if (callback != null) {
   206→        callback(
   207→          RealtimeSubscribeStatus.channelError,
   208→          Exception(
   209→            jsonEncode((error as Map<String, dynamic>).isNotEmpty
   210→                ? (error).values.join(', ')
   211→                : 'error'),
   212→          ),
   213→        );
   214→      }
   215→    }).receive('timeout', (_) {
   216→      if (callback != null) callback(RealtimeSubscribeStatus.timedOut, null);
   217→    });
   218→    return this;
   219→  }
   220→
   221→  Future<void> _handleJoinOk(
   222→    Map<String, dynamic> response,
   223→    void Function(RealtimeSubscribeStatus status, Object? error)? callback,
   224→  ) async {
   225→    final serverPostgresFilters = response['postgres_changes'];
   226→    if (socket.accessToken != null) {
   227→      try {
   228→        // ignore: avoid-passing-self-as-argument
   229→        await socket.setAuth(socket.accessToken);
   230→      } on FormatException catch (e) {
   231→        // The cached access token may have expired by the time the
   232→        // channel rejoins (e.g. after the device wakes from a long
   233→        // sleep). Auth state listeners will re-call setAuth with a
   234→        // fresh token shortly after, so swallow this specific error
   235→        // to avoid surfacing it as an uncaught exception. The same
   236→        // filter is applied in `SupabaseClient._handleTokenChanged`.
   237→        if (!e.message.contains('InvalidJWTToken')) {
   238→          rethrow;
   239→        }
   240→      }
   241→    }
   242→
   243→    if (serverPostgresFilters == null) {
   244→      if (callback != null) {
   245→        callback(RealtimeSubscribeStatus.subscribed, null);
   246→      }
   247→      return;
   248→    }
   249→    final clientPostgresBindings = _bindings['postgres_changes'];
   250→    final bindingsLen = clientPostgresBindings?.length ?? 0;
   251→    final newPostgresBindings = <Binding>[];
   252→
   253→    for (var i = 0; i < bindingsLen; i++) {
   254→      final clientPostgresBinding = clientPostgresBindings![i];
   255→
   256→      final event = clientPostgresBinding.filter['event'];
   257→      final schema = clientPostgresBinding.filter['schema'];
   258→      final table = clientPostgresBinding.filter['table'];
   259→      final filter = clientPostgresBinding.filter['filter'];
   260→      final serverPostgresFilter = serverPostgresFilters[i];
   261→
   262→      if (serverPostgresFilter != null &&
   263→          serverPostgresFilter['event'] == event &&
   264→          serverPostgresFilter['schema'] == schema &&
   265→          serverPostgresFilter['table'] == table &&
   266→          serverPostgresFilter['filter'] == filter) {
   267→        newPostgresBindings.add(clientPostgresBinding.copyWith(
   268→          id: serverPostgresFilter['id']?.toString(),
   269→        ));
   270→      } else {
   271→        unawaited(unsubscribe());
   272→        if (callback != null) {
   273→          callback(
   274→            RealtimeSubscribeStatus.channelError,
   275→            Exception(
   276→                'mismatch between server and client bindings for postgres changes'),
   277→          );
   278→        }
   279→        return;
   280→      }
   281→    }
   282→
   283→    _bindings['postgres_changes'] = newPostgresBindings;
   284→
   285→    if (callback != null) {
   286→      callback(RealtimeSubscribeStatus.subscribed, null);
   287→    }
   288→  }

   333→  /// Sets up a listener on your Supabase database.
   334→  ///
   335→  /// [event] determines whether you listen to `insert`, `update`, `delete`, or all of the events.
   336→  ///
   337→  /// [schema] is the schema of the database on which to set up the listener.
   338→  /// The listener will return all changes from every listenable schema if omitted.
   339→  ///
   340→  /// [table] is the table of the database on which to setup the listener.
   341→  /// The listener will return all changes from every listenable table if omitted.
   342→  ///
   343→  /// [filter] can be used to further control which rows to listen to within the given [schema] and [table].
   344→  ///
   345→  /// ```dart
   346→  /// supabase.channel('my_channel').onPostgresChanges(
   347→  ///     event: PostgresChangeEvent.all,
   348→  ///     schema: 'public',
   349→  ///     table: 'messages',
   350→  ///     filter: PostgresChangeFilter(
   351→  ///       type: PostgresChangeFilterType.eq,
   352→  ///       column: 'room_id',
   353→  ///       value: 200,
   354→  ///     ),
   355→  ///     callback: (payload) {
   356→  ///       print(payload);
   357→  ///     }).subscribe();
   358→  /// ```

   360→    required PostgresChangeEvent event,
   361→    String? schema,

   363→    PostgresChangeFilter? filter,

   365→  }) {
   366→    return onEvents(
   367→      'postgres_changes',
   368→      ChannelFilter(
   369→        event: event.toRealtimeEvent(),
   370→        schema: schema,
   371→        table: table,
   372→        filter: filter?.toString(),
   373→      ),
   374→      (payload, [ref]) => callback(PostgresChangePayload.fromPayload(payload)),
   375→    );
   376→  }

   495→  @internal
   496→  RealtimeChannel onEvents(
   497→      String type, ChannelFilter filter, BindingCallback callback) {
   498→    final typeLower = type.toLowerCase();
   499→
   500→    final binding = Binding(typeLower, filter.toMap(), callback);
   501→
   502→    if (_bindings[typeLower] != null) {
   503→      _bindings[typeLower]!.add(binding);
   504→    } else {
   505→      _bindings[typeLower] = [binding];
   506→    }
   507→
   508→    return this;
   509→  }
   510→
   511→  @internal
   512→  RealtimeChannel off(String type, Map<String, String> filter) {
   513→    final typeLower = type.toLowerCase();
   514→
   515→    _bindings[typeLower] = _bindings[typeLower]!.where((bind) {
   516→      return !(bind.type.toLowerCase() == typeLower &&
   517→          RealtimeChannel._isEqual(bind.filter, filter));
   518→    }).toList();
   519→    return this;
   520→  }

   522→  /// Returns `true` if the socket is connected and the channel has been joined.
   523→  bool get canPush {
   524→    return socket.isConnected && isJoined;

   527→  @internal
   528→  Push push(
   529→    ChannelEvents event,
   530→    Map<String, dynamic> payload, [
   531→    Duration? timeout,
   532→  ]) {
   533→    if (!joinedOnce) {
   534→      throw "tried to push '${event.eventName()}' to '$topic' before joining. Use channel.subscribe() before pushing events";
   535→    }
   536→    final pushEvent = Push(this, event, payload, timeout ?? _timeout);
   537→    if (canPush) {
   538→      pushEvent.send();
   539→    } else {
   540→      pushEvent.startTimeout();
   541→      _pushBuffer.add(pushEvent);
   542→    }
   543→
   544→    return pushEvent;
   545→  }

   547→  /// Sends a broadcast message explicitly via REST API.
   548→  ///
   549→  /// This method always uses the REST API endpoint regardless of WebSocket connection state.
   550→  /// Useful when you want to guarantee REST delivery or when gradually migrating from implicit REST fallback.
   551→  ///
   552→  /// [payload] must be either a `Map<String, dynamic>`, which is JSON-encoded,
   553→  /// or binary data ([TypedData] such as [Uint8List], or [ByteBuffer]), which
   554→  /// is sent as `application/octet-stream`. Map payloads keep `httpSend`
   555→  /// consistent with [onBroadcast], which delivers received payloads as
   556→  /// `Map<String, dynamic>`.
   557→  ///
   558→  /// [event] is the name of the broadcast event.
   559→  /// [timeout] is an optional timeout duration.
   560→  ///
   561→  /// Requires a Realtime server running v2.97.0 or newer.
   562→  ///
   563→  /// Returns a [Future] that resolves when the message is sent successfully,
   564→  /// or throws an error if the message fails to send.
   565→  ///
   566→  /// ```dart
   567→  /// try {
   568→  ///   await channel.httpSend(
   569→  ///     event: 'cursor-pos',
   570→  ///     payload: {'x': 123, 'y': 456},
   571→  ///   );
   572→  /// } catch (e) {
   573→  ///   print('Failed to send message: $e');
   574→  /// }
   575→  /// ```
   576→  Future<void> httpSend({
   577→    required String event,
   578→    required Object payload,
   579→    Duration? timeout,
   580→  }) async {
   581→    final isBinary = payload is TypedData || payload is ByteBuffer;
   582→
   583→    assert(
   584→      isBinary || payload is Map<String, dynamic>,
   585→      'httpSend payload must be a Map<String, dynamic> or binary '
   586→      '(TypedData/ByteBuffer)',
   587→    );
   588→
   589→    final headers = {
   590→      ..._broadcastHeaders,
   591→      'Content-Type':
   592→          isBinary ? 'application/octet-stream' : 'application/json',
   593→    };
   594→
   595→    final url = Uri.parse(
   596→      '$broadcastEndpointURL'
   597→      '/${Uri.encodeComponent(subTopic)}'
   598→      '/events/${Uri.encodeComponent(event)}'
   599→      '${_private ? '?private=true' : ''}',
   600→    );
   601→
   602→    final body = isBinary ? _asBytes(payload) : json.encode(payload);
   603→
   604→    final response = await (socket.httpClient?.post ?? post)(url,
   605→            headers: headers, body: body)
   606→        .timeout(
   607→      timeout ?? _timeout,
   608→      onTimeout: () => throw TimeoutException('Request timeout'),
   609→    );
   610→
   611→    if (response.statusCode == 202) {
   612→      return;
   613→    }
   614→
   615→    if (response.statusCode == 404) {
   616→      throw Exception(
   617→        'httpSend() requires Realtime server v2.97.0 or newer; the endpoint '
   618→        'returned 404. Update your Supabase CLI to a recent version, or upgrade '
   619→        'the Realtime server in your self-hosted setup.',
   620→      );
   621→    }
   622→
   623→    String errorMessage = response.reasonPhrase ?? 'Unknown error';
   624→    try {
   625→      final errorBody = json.decode(response.body) as Map<String, dynamic>;
   626→      errorMessage = (errorBody['error'] ??
   627→          errorBody['message'] ??
   628→          errorMessage) as String;
   629→    } catch (_) {
   630→      // If JSON parsing fails, use the default error message
   631→    }
   632→
   633→    throw Exception(errorMessage);
   634→  }
   635→
   636→  Uint8List _asBytes(Object payload) {
   637→    if (payload is ByteBuffer) {
   638→      return payload.asUint8List();
   639→    }
   640→    final typed = payload as TypedData;
   641→    return typed.buffer.asUint8List(typed.offsetInBytes, typed.lengthInBytes);
   642→  }

   644→  /// Sends a realtime broadcast message.
   645→  ///
   646→  /// With protocol `2.0.0` the message is sent as a positional JSON text frame.
   647→  ///
   648→  /// To send a raw binary payload over a WebSocket binary frame (avoiding JSON
   649→  /// encoding on the server), set a `Uint8List` (or any `TypedData`) under the
   650→  /// `payload` key:
   651→  ///
   652→  /// ```dart
   653→  /// channel.sendBroadcastMessage(
   654→  ///   event: 'file',
   655→  ///   payload: {'payload': myUint8List},
   656→  /// );
   657→  /// ```
   658→  Future<ChannelResponse> sendBroadcastMessage({
   659→    required String event,
   660→    required Map<String, dynamic> payload,
   661→  }) {
   662→    return send(
   663→      type: RealtimeListenTypes.broadcast,
   664→      event: event,
   665→      payload: payload,
   666→    );
   667→  }

   669→  @internal
   670→  Future<ChannelResponse> send({
   671→    required RealtimeListenTypes type,
   672→    String? event,
   673→    required Map<String, dynamic> payload,
   674→    Map<String, dynamic> opts = const {},
   675→  }) async {
   676→    final completer = Completer<ChannelResponse>();
   677→
   678→    payload['type'] = type.toType();
   679→    if (event != null) {
   680→      payload['event'] = event;
   681→    }
   682→
   683→    if (!canPush && type == RealtimeListenTypes.broadcast) {
   684→      socket.log(
   685→        'channel',
   686→        'send() is automatically falling back to REST API. '
   687→            'This behavior will be deprecated in the future. '
   688→            'Please use httpSend() explicitly for REST delivery.',
   689→      );
   690→
   691→      try {
   692→        final response = await (socket.httpClient?.post ?? post)(
   693→          Uri.parse(broadcastEndpointURL),
   694→          headers: _broadcastHeaders,
   695→          body: json.encode(_broadcastBody(event, payload)),
   696→        );
   697→        if (200 <= response.statusCode && response.statusCode < 300) {
   698→          completer.complete(ChannelResponse.ok);
   699→        } else {
   700→          completer.complete(ChannelResponse.error);
   701→        }
   702→      } catch (e) {
   703→        completer.complete(ChannelResponse.error);
   704→      }
   705→    } else {
   706→      final push = this.push(
   707→        ChannelEventsExtended.fromType(payload['type']),
   708→        payload,
   709→        opts['timeout'] ?? _timeout,
   710→      );
   711→
   712→      if (payload['type'] == 'broadcast' &&
   713→          (params['config']?['broadcast']?['ack'] == null ||
   714→              params['config']?['broadcast']?['ack'] == false)) {
   715→        if (!completer.isCompleted) {
   716→          completer.complete(ChannelResponse.ok);
   717→        }
   718→      }
   719→
   720→      push.receive('ok', (_) {
   721→        if (!completer.isCompleted) {
   722→          completer.complete(ChannelResponse.ok);
   723→        }
   724→      });
   725→      push.receive('error', (_) {
   726→        if (!completer.isCompleted) {
   727→          completer.complete(ChannelResponse.error);
   728→        }
   729→      });
   730→      push.receive('timeout', (_) {
   731→        if (!completer.isCompleted) {
   732→          completer.complete(ChannelResponse.timedOut);
   733→        }
   734→      });
   735→    }
   736→    return completer.future;
   737→  }

   768→  /// Leaves the channel
   769→  ///
   770→  /// Unsubscribes from server events, and instructs channel to terminate on server.
   771→  /// Triggers onClose() hooks.
   772→  ///
   773→  /// To receive leave acknowledgements, use the a `receive` hook to bind to the server ack,
   774→  /// ```dart
   775→  /// channel.unsubscribe().receive("ok", (_){print("left!");} );
   776→  /// ```
   777→  Future<String> unsubscribe([Duration? timeout]) {
   778→    _state = ChannelStates.leaving;
   779→    void onClose() {
   780→      socket.log('channel', 'leave $topic');
   781→      trigger(ChannelEvents.close.eventName(), 'leave', joinRef);
   782→    }
   783→
   784→    // Destroy joinPush to avoid connection timeouts during unscription phase
   785→    joinPush.destroy();
   786→
   787→    final completer = Completer<String>();
   788→
   789→    final leavePush = Push(this, ChannelEvents.leave, {}, timeout ?? _timeout);
   790→
   791→    leavePush.receive('ok', (_) {
   792→      onClose();
   793→      if (!completer.isCompleted) {
   794→        completer.complete('ok');
   795→      }
   796→    }).receive('timeout', (_) {
   797→      if (!completer.isCompleted) {
   798→        onClose();
   799→      }
   800→      completer.complete('timed out');
   801→    }).receive('error', (_) {
   802→      onClose();
   803→      if (!completer.isCompleted) {
   804→        completer.complete('error');
   805→      }
   806→    });
   807→
   808→    leavePush.send();
   809→
   810→    if (!canPush) {
   811→      leavePush.trigger('ok', {});
   812→    }
   813→
   814→    return completer.future;
   815→  }

   844→  /// Resends [joinPush] to tell the server we join this channel again and marks
   845→  /// the channel as [ChannelStates.joining].
   846→  ///
   847→  /// Usually [rejoin] only happens when the channel timeouts or errors out.
   848→  /// When manually disconnecting, the channel is still marked as
   849→  /// [ChannelStates.joined]. Calling [RealtimeClient.leaveOpenTopic] will
   850→  /// unsubscribe itself, which causes issues when trying to rejoin. This method
   851→  /// therefore doesn't call [RealtimeClient.leaveOpenTopic].
   852→  @internal
   853→  void forceRejoin([Duration? timeout]) {
   854→    if (isLeaving) {
   855→      return;
   856→    }
   857→    _state = ChannelStates.joining;
   858→    joinPush.resend(timeout ?? _timeout);
   859→  }

   861→  void trigger(String type, [dynamic payload, String? ref]) {
   862→    final typeLower = type.toLowerCase();
   863→
   864→    final events = [
   865→      ChannelEvents.close,
   866→      ChannelEvents.error,
   867→      ChannelEvents.leave,
   868→      ChannelEvents.join,
   869→    ].map((e) => e.eventName()).toSet();
   870→
   871→    if (ref != null && events.contains(typeLower) && ref != joinRef) {
   872→      return;
   873→    }
   874→
   875→    var handledPayload = onMessage(typeLower, payload, ref);
   876→    if (payload != null && handledPayload == null) {
   877→      throw 'channel onMessage callbacks must return the payload, modified or unmodified';
   878→    }
   879→
   880→    if (['insert', 'update', 'delete'].contains(typeLower)) {
   881→      final bindings = _bindings['postgres_changes']?.where((bind) {
   882→        return (bind.filter['event'] == '*' ||
   883→            bind.filter['event']?.toLowerCase() == typeLower);
   884→      });
   885→
   886→      for (final bind in (bindings ?? <Binding>[])) {
   887→        handledPayload = getEnrichedPayload(handledPayload);
   888→        bind.callback(handledPayload, ref);
   889→      }
   890→    } else {
   891→      final bindings = (_bindings[typeLower] ?? []).where((bind) {
   892→        if (['broadcast', 'presence', 'postgres_changes'].contains(typeLower)) {
   893→          final bindId = bind.id;
   894→          if (bindId != null) {
   895→            final bindEvent = bind.filter['event'];
   896→
   897→            return ((payload['ids'] as List?)?.contains(int.parse(bindId)) ==
   898→                    true &&
   899→                (bindEvent == '*' ||
   900→                    bindEvent?.toLowerCase() ==
   901→                        (payload['data']?['type'] as String?)?.toLowerCase()));
   902→          }
   903→          final bindEvent = bind.filter['event']?.toLowerCase();
   904→          return (bindEvent == '*' ||
   905→              bindEvent == (payload?['event'] as String?)?.toLowerCase());
   906→        }
   907→        return bind.type.toLowerCase() == typeLower;
   908→      });
   909→      for (final bind in bindings) {
   910→        if (handledPayload is Map<String, dynamic> &&
   911→            handledPayload.containsKey('ids')) {
   912→          handledPayload = getEnrichedPayload(handledPayload);
   913→        }
   914→
   915→        bind.callback(handledPayload, ref);
   916→      }
   917→    }
   918→  }

   940→  static bool _isEqual(Map<String, String> obj1, Map<String, String> obj2) {
   941→    if (obj1.keys.length != obj2.keys.length) {
   942→      return false;
   943→    }
   944→
   945→    for (final k in obj1.keys) {
   946→      if (obj1[k] != obj2[k]) {
   947→        return false;
   948→      }
   949→    }
   950→
   951→    return true;
   952→  }
