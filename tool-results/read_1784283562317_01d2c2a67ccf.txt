   164→/// Data-layer representation of a configurable grading scale.
   165→class GradeScaleModel {
   166→  const GradeScaleModel({
   167→    required this.id,
   168→    required this.schoolId,
   169→    required this.name,
   170→    required this.gradeType,
   171→    this.isDefault = false,
   172→    this.isActive = true,
   173→    this.scaleEntries = const [],
   174→    this.createdBy,
   175→    this.settings = const {},
   176→    required this.createdAt,
   177→    required this.updatedAt,
   178→  });
   179→
   180→  final String id;
   181→  final String schoolId;
   182→  final String name;
   183→  final String gradeType;
   184→  final bool isDefault;
   185→  final bool isActive;
   186→  final List<GradeScaleEntryModel> scaleEntries;
   187→  final String? createdBy;
   188→  final Map<String, dynamic> settings;
   189→  final DateTime createdAt;
   190→  final DateTime updatedAt;
   191→
   192→  // ─── JSON Serialization ────────────────────────────────────────────
   193→
   194→  factory GradeScaleModel.fromJson(Map<String, dynamic> json) {
   195→    // Parse nested scale entries
   196→    final rawEntries = json['scale_entries'] ?? json['scaleEntries'];
   197→    final List<GradeScaleEntryModel> entries = rawEntries is List
   198→        ? rawEntries
   199→            .map(
   200→                (e) => GradeScaleEntryModel.fromJson(e as Map<String, dynamic>))
   201→            .toList()
   202→        : <GradeScaleEntryModel>[];
   203→
   204→    return GradeScaleModel(
   205→      id: json['id'] as String? ?? '',
   206→      schoolId:
   207→          json['school_id'] as String? ?? json['schoolId'] as String? ?? '',
   208→      name: json['name'] as String? ?? '',
   209→      gradeType: json['grade_type'] as String? ??
   210→          json['gradeType'] as String? ??
   211→          'percentage',
   212→      isDefault:
   213→          json['is_default'] as bool? ?? json['isDefault'] as bool? ?? false,
   214→      isActive:
   215→          json['is_active'] as bool? ?? json['isActive'] as bool? ?? true,
   216→      scaleEntries: entries,
   217→      createdBy:
   218→          json['created_by'] as String? ?? json['createdBy'] as String?,
   219→      settings: json['settings'] as Map<String, dynamic>? ?? {},
   220→      createdAt: json['created_at'] != null
   221→          ? DateTime.parse(json['created_at'] as String)
   222→          : json['createdAt'] != null
   223→              ? DateTime.parse(json['createdAt'] as String)
   224→              : DateTime.now(),
   225→      updatedAt: json['updated_at'] != null
   226→          ? DateTime.parse(json['updated_at'] as String)
   227→          : json['updatedAt'] != null
   228→              ? DateTime.parse(json['updatedAt'] as String)
   229→              : DateTime.now(),
   230→    );
   231→  }
   232→
   233→  Map<String, dynamic> toJson() {
   234→    return {
   235→      'id': id,
   236→      'school_id': schoolId,
   237→      'name': name,
   238→      'grade_type': gradeType,
   239→      'is_default': isDefault,
   240→      'is_active': isActive,
   241→      'scale_entries': scaleEntries.map((e) => e.toJson()).toList(),
   242→      'created_by': createdBy,
   243→      'settings': settings,
   244→      'created_at': createdAt.toIso8601String(),
   245→      'updated_at': updatedAt.toIso8601String(),
   246→    };
   247→  }
