    66→    66→  // ─── Body Router ────────────────────────────────────────────────────
    67→    67→
    68→    68→  Widget _buildBody(BuildContext context, WorkspaceDashboardState state) {
    69→    69→    // Loading state
    70→    70→    if (state.isLoading && state.dashboardSummary == null) {
    71→    71→      return const Center(child: AppLoadingSpinner());
    72→    72→    }
    73→    73→
    74→    74→    // Error state
    75→    75→    if (state.error != null && state.dashboardSummary == null) {
    76→    76→      return AppErrorState.genericError(
    77→    77→        message: state.error,
    78→    78→        onRetry: () =>
    79→    79→            ref.read(workspaceDashboardProvider.notifier).loadDashboard(),
    80→    80→      );
    81→    81→    }
    82→    82→
    83→    83→    // Empty state
    84→    84→    final summary = state.dashboardSummary;
    85→    85→    if (summary == null) {
    86→    86→      return AppEmptyState.noData(
    87→    87→        title: 'No Workspace Data',
    88→    88→        subtitle: 'Your workspace dashboard will appear here once loaded.',
    89→    89→        actionLabel: 'Refresh',
    90→    90→        onAction: () =>
    91→    91→            ref.read(workspaceDashboardProvider.notifier).loadDashboard(),
    92→    92→      );
    93→    93→    }
    94→    94→
    95→    95→    // Success — render dashboard
    96→    96→    return RefreshIndicator(
    97→    97→      onRefresh: () =>
    98→    98→          ref.read(workspaceDashboardProvider.notifier).refresh(),
    99→    99→      child: CustomScrollView(
   100→   100→        physics: const AlwaysScrollableScrollPhysics(),
   101→   101→        slivers: [
   102→   102→          // ─── Welcome Section ──────────────────────────────────────

   104→   104→
   105→   105→          // ─── Stats Cards ──────────────────────────────────────────
   106→   106→          SliverToBoxAdapter(child: _buildStatsCards(context, summary)),
   107→   107→
   108→   108→          // ─── Quick AI Actions ─────────────────────────────────────
   109→   109→          SliverToBoxAdapter(child: _buildQuickActionsSection(context)),
   110→   110→
   111→   111→          // ─── Schedule & Events (2-column on desktop) ──────────────
   112→   112→          SliverToBoxAdapter(
   113→   113→            child: _buildScheduleAndEvents(context, summary),
   114→   114→          ),
   115→   115→
   116→   116→          // ─── Recently Generated AI Content ────────────────────────

   118→   118→            child: _buildRecentAiContent(context, summary),
