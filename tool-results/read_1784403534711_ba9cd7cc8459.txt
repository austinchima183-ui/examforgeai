    24→
    25→/// AI Lesson Plan Generator page where teachers fill in parameters and AI
    26→/// generates a complete lesson plan.
    27→///
    28→/// Has two states: input form (no plan generated yet) and generated result
    29→/// (editable lesson plan with all sections).
    30→class LessonPlanGeneratorPage extends ConsumerStatefulWidget {
    31→  const LessonPlanGeneratorPage({super.key});
    32→
    33→  @override
    34→  ConsumerState<LessonPlanGeneratorPage> createState() =>
    35→      _LessonPlanGeneratorPageState();
    36→}
    37→
    38→class _LessonPlanGeneratorPageState
    39→    extends ConsumerState<LessonPlanGeneratorPage> {
    40→  // ─── Form Controllers ────────────────────────────────────────────────
    41→
    42→  final _formKey = GlobalKey<FormState>();
    43→  final _subjectCtrl = TextEditingController();
    44→  final _classNameCtrl = TextEditingController();
    45→  final _topicCtrl = TextEditingController();
    46→  final _subtopicCtrl = TextEditingController();
    47→  final _objectivesCtrl = TextEditingController();
    48→  final _durationCtrl = TextEditingController(text: '40');
    49→
    50→  CurriculumType _curriculum = CurriculumType.nigerian;
    51→  TeachingStyle _teachingStyle = TeachingStyle.interactive;
    52→  StudentLevel _studentLevel = StudentLevel.intermediate;
    53→

    55→

    58→  final List<TextEditingController> _outcomeControllers = [];
    59→  bool _isEditing = false;
    60→
    61→  @override
    62→  void dispose() {
    63→    _subjectCtrl.dispose();
    64→    _classNameCtrl.dispose();
    65→    _topicCtrl.dispose();
    66→    _subtopicCtrl.dispose();
    67→    _objectivesCtrl.dispose();
    68→    _durationCtrl.dispose();
    69→    _titleEditCtrl.dispose();
    70→    for (final c in _objectiveControllers) {
    71→      c.dispose();
    72→    }
    73→    for (final c in _outcomeControllers) {
    74→      c.dispose();
    75→    }
    76→    super.dispose();
    77→  }

    79→  // ─── Helpers ─────────────────────────────────────────────────────────
    80→
    81→  void _listenForMessages() {
    82→    final state = ref.read(lessonPlanProvider);
    83→    if (state.successMessage != null) {
    84→      _showSnackBar(state.successMessage!, isError: false);
    85→      ref.read(lessonPlanProvider.notifier).clearSuccessMessage();
    86→    }
    87→    if (state.error != null) {
    88→      _showSnackBar(state.error!, isError: true);
    89→      ref.read(lessonPlanProvider.notifier).clearError();
    90→    }
    91→  }
    92→
    93→  void _showSnackBar(String message, {required bool isError}) {
    94→    if (!mounted) return;
    95→    final cs = context.colorScheme;
    96→    context.scaffoldMessenger.showSnackBar(
    97→      SnackBar(
    98→        content: Text(message),
    99→        backgroundColor: isError ? cs.error : cs.primary,
   100→        behavior: SnackBarBehavior.floating,
   101→        shape: RoundedRectangleBorder(
   102→          borderRadius: BorderRadius.circular(Spacings.smRadius),
   103→        ),
   104→        duration: const Duration(seconds: 3),
   105→      ),
   106→    );
   107→  }
   108→
   109→  Future<void> _handleGenerate() async {
   110→    if (!_formKey.currentState!.validate()) return;
   111→
   112→    final objectives = _objectivesCtrl.text
   113→        .split('\n')
   114→        .map((s) => s.trim())
   115→        .where((s) => s.isNotEmpty)
   116→        .toList();
   117→
   118→    final duration = int.tryParse(_durationCtrl.text) ?? 40;
   119→
   120→    await ref.read(lessonPlanProvider.notifier).generateLessonPlan(
   121→      GenerateLessonPlanParams(
   122→        subject: _subjectCtrl.text.trim(),
   123→        className: _classNameCtrl.text.trim(),
   124→        topic: _topicCtrl.text.trim(),
   125→        subtopic: _subtopicCtrl.text.trim().isNotEmpty
   126→            ? _subtopicCtrl.text.trim()
   127→            : null,
   128→        curriculum: _curriculum,
   129→        learningObjectives: objectives.isNotEmpty ? objectives : null,
   130→        durationMinutes: duration,
   131→        teachingStyle: _teachingStyle,
   132→        studentLevel: _studentLevel,
   133→      ),
   134→    );
   135→
   136→    _listenForMessages();
   137→
   138→    // Initialize edit controllers from the generated plan
   139→    final plan = ref.read(lessonPlanProvider).currentPlan;
   140→    if (plan != null) {
   141→      _titleEditCtrl.text = plan.title;
   142→      _initListControllers(
   143→        plan.learningObjectives,
   144→        _objectiveControllers,
   145→      );
   146→      _initListControllers(
   147→        plan.learningOutcomes,
   148→        _outcomeControllers,
   149→      );
   150→    }
   151→  }
   152→
   153→  void _initListControllers(
   154→    List<String> items,
   155→    List<TextEditingController> controllers,
   156→  ) {
   157→    for (final c in controllers) {
   158→      c.dispose();
   159→    }
   160→    controllers.clear();
   161→    for (final item in items) {
   162→      controllers.add(TextEditingController(text: item));
   163→    }
   164→  }
   165→
   166→  void _handleSave() {
   167→    final plan = ref.read(lessonPlanProvider).currentPlan;
   168→    if (plan == null) return;
   169→
   170→    final updatedObjectives = _objectiveControllers
   171→        .map((c) => c.text.trim())
   172→        .where((s) => s.isNotEmpty)
   173→        .toList();
   174→    final updatedOutcomes = _outcomeControllers
   175→        .map((c) => c.text.trim())
   176→        .where((s) => s.isNotEmpty)
   177→        .toList();
   178→
   179→    ref.read(lessonPlanProvider.notifier).updateLessonPlan(
   180→      UpdateLessonPlanParams(
   181→        plan: plan.copyWith(
   182→          title: _titleEditCtrl.text.trim().isNotEmpty
   183→              ? _titleEditCtrl.text.trim()
   184→              : plan.title,
   185→          learningObjectives: updatedObjectives,
   186→          learningOutcomes: updatedOutcomes,
   187→        ),
   188→      ),
   189→    );
   190→
   191→    setState(() => _isEditing = false);
   192→    _listenForMessages();
   193→  }
   194→
   195→  void _handleReset() {
   196→    ref.read(lessonPlanProvider.notifier).setCurrentPlan(null);
   197→    _titleEditCtrl.clear();
   198→    for (final c in _objectiveControllers) {
   199→      c.dispose();
   200→    }
   201→    _objectiveControllers.clear();
   202→    for (final c in _outcomeControllers) {
   203→      c.dispose();
   204→    }
   205→    _outcomeControllers.clear();
   206→    setState(() => _isEditing = false);
   207→  }
