     9→--   - Timetable system with conflict detection
    10→--   - Attendance tracking (student & teacher)
    11→--   - Homework & assignments
    12→--   - Announcements & document center
    13→--   - Promotion history & graduation tracking

    55→    );

    57→

   116→      'monday',

   170→-- Add new columns to the existing schools table
   171→DO $$
   172→BEGIN
   173→  -- Motto

   180→

   187→  END IF;
   188→

   195→  END IF;
   196→
   197→  -- Branding secondary color

   199→    SELECT 1 FROM information_schema.columns

   928→-- Ensures only one session per school is marked as current.
   929→-- ---------------------------------------------------------------------------
   930→CREATE OR REPLACE FUNCTION ensure_single_current_session()
   931→RETURNS TRIGGER AS $$
   932→BEGIN
   933→  IF NEW.is_current = true THEN
   934→    UPDATE academic_sessions
   935→    SET is_current = false
   936→    WHERE school_id = NEW.school_id
   937→      AND id != NEW.id
   938→      AND is_current = true;
   939→  END IF;
   940→  RETURN NEW;
   941→END;
   942→$$ LANGUAGE plpgsql SECURITY DEFINER;

   976→-- ---------------------------------------------------------------------------
   977→-- detect_timetable_conflicts()
   978→-- Raises an exception if a teacher or class is double-booked.
   979→-- ---------------------------------------------------------------------------
   980→CREATE OR REPLACE FUNCTION detect_timetable_conflicts()
   981→RETURNS TRIGGER AS $$
   982→BEGIN
   983→  -- Check teacher conflict (same teacher, same day, same period, different slot)
   984→  IF NEW.teacher_id IS NOT NULL THEN
   985→    IF EXISTS (
   986→      SELECT 1 FROM timetable_slots ts
   987→      JOIN timetables t ON t.id = ts.timetable_id
   988→      WHERE ts.teacher_id = NEW.teacher_id
   989→        AND ts.day_of_week = NEW.day_of_week
   990→        AND ts.period_number = NEW.period_number
   991→        AND ts.id != NEW.id
   992→        AND t.is_active = true
   993→    ) THEN
   994→      RAISE EXCEPTION 'Teacher is already assigned to another class at % period % on %',
   995→        NEW.teacher_id, NEW.period_number, NEW.day_of_week;
   996→    END IF;
   997→  END IF;
   998→
   999→  -- Check class conflict (same class, same day, same period, different slot)
  1000→  IF NEW.class_id IS NOT NULL THEN
  1001→    IF EXISTS (
  1002→      SELECT 1 FROM timetable_slots ts
  1003→      JOIN timetables t ON t.id = ts.timetable_id
  1004→      WHERE ts.class_id = NEW.class_id
  1005→        AND ts.day_of_week = NEW.day_of_week
  1006→        AND ts.period_number = NEW.period_number
  1007→        AND ts.id != NEW.id
  1008→        AND t.is_active = true
  1009→    ) THEN
  1010→      RAISE EXCEPTION 'Class is already occupied at period % on %',
  1011→        NEW.period_number, NEW.day_of_week;
  1012→    END IF;
  1013→  END IF;
  1014→
  1015→  RETURN NEW;
  1016→END;
  1017→$$ LANGUAGE plpgsql SECURITY DEFINER;
  1018→
  1019→DROP TRIGGER IF EXISTS trg_detect_timetable_conflicts ON timetable_slots;
  1020→CREATE TRIGGER trg_detect_timetable_conflicts
  1021→  BEFORE INSERT OR UPDATE ON timetable_slots
  1022→  FOR EACH ROW
  1023→  EXECUTE FUNCTION detect_timetable_conflicts();

  1035→  IF NEW.is_published = true AND (OLD.is_published = false OR OLD.is_published IS NULL) THEN
  1036→    notification_message := CASE NEW.priority
  1037→      WHEN 'urgent' THEN 'URGENT: ' || NEW.title
  1038→      WHEN 'high' THEN 'IMPORTANT: ' || NEW.title
  1039→      ELSE NEW.title
  1040→    END;
  1041→
  1042→    -- Notify all users in the school based on target audience
  1043→    FOR target_user IN
  1044→      SELECT id FROM users
  1045→      WHERE school_id = NEW.school_id
  1046→        AND is_active = true
  1047→        AND CASE NEW.target_audience
  1048→          WHEN 'students' THEN role = 'student'
  1049→          WHEN 'teachers' THEN role = 'teacher'
  1050→          WHEN 'parents' THEN role = 'parent'
  1051→          ELSE true
  1052→        END
  1053→    LOOP
  1054→      INSERT INTO notifications (user_id, type, title, message, data)
  1055→      VALUES (
  1056→        target_user.id,
  1057→        'system',
  1058→        'New Announcement',
  1059→        notification_message,
  1060→        jsonb_build_object('announcement_id', NEW.id, 'type', 'announcement')
  1061→      );
  1062→    END LOOP;
  1063→  END IF;
  1064→
  1065→  RETURN NEW;
  1066→END;
