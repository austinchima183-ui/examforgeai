   200→  @override
   201→  Future<GenerationRequestModel> createGenerationRequest(
   202→    Map<String, dynamic> requestData,
   203→  ) async {
   204→    try {
   205→      final response = await _supabase
   206→          .from(_generationRequestsTable)
   207→          .insert(requestData)
   208→          .select();
   209→
   210→      if (response.isEmpty) {
   211→        throw const ServerException(
   212→          message: 'Generation request creation returned no data.',
   213→          statusCode: 500,
   214→        );
   215→      }
   216→
   217→      AppLogger.info('Generation request created: ${response.first['id']}');
   218→      return GenerationRequestModel.fromJson(response.first);
   219→    } on sb.PostgrestException catch (e) {
   220→      throw _mapPostgrestException(e);
   221→    } on ServerException {
   222→      rethrow;
   223→    } catch (e) {
   224→      AppLogger.error('Unexpected createGenerationRequest error', error: e);
   225→      throw const ServerException(
   226→        message: 'Failed to create generation request.',
   227→        statusCode: 500,
   228→      );
   229→    }
   230→  }

   291→  @override
   292→  Future<List<GenerationRequestModel>> getGenerationRequests(
   293→    Map<String, dynamic> filters,
   294→  ) async {
   295→    try {
   296→      var query = _supabase
   297→          .from(_generationRequestsTable)
   298→          .select()
   299→          .order('created_at', ascending: false);
   300→
   301→      // Apply filters
   302→      if (filters.containsKey('school_id')) {
   303→        query = query.eq('school_id', filters['school_id'] as String);
   304→      }
   305→      if (filters.containsKey('status')) {
   306→        query = query.eq('status', filters['status'] as String);
   307→      }
   308→      if (filters.containsKey('requested_by')) {
   309→        query = query.eq('requested_by', filters['requested_by'] as String);
   310→      }
   311→
   312→      // Pagination
   313→      final page = filters['page'] as int? ?? 1;
   314→      final perPage = filters['per_page'] as int? ?? filters['perPage'] as int? ?? 20;
   315→      final from = (page - 1) * perPage;
   316→      query = query.range(from, from + perPage - 1);
   317→
   318→      final response = await query;
   319→      return response
   320→          .map<GenerationRequestModel>(
   321→              (json) => GenerationRequestModel.fromJson(json))
   322→          .toList();
   323→    } on sb.PostgrestException catch (e) {
   324→      throw _mapPostgrestException(e);
   325→    } catch (e) {
   326→      AppLogger.error('Unexpected getGenerationRequests error', error: e);
   327→      throw const ServerException(
   328→        message: 'Failed to get generation requests.',
   329→        statusCode: 500,
   330→      );
   331→    }
   332→  }

  1479→          .eq('id', id)
  1480→          .select();

  1482→      return GenerationQueueModel.fromJson(
  1483→          updated.isNotEmpty ? updated.first : entry);

  1596→  @override
  1597→  Future<Map<String, dynamic>> getDashboardStats(String? schoolId) async {
  1598→    try {
  1599→      // Use an RPC call if available, otherwise aggregate manually
  1600→      try {
  1601→        final response = await _supabase.rpc(
  1602→          'get_ai_dashboard_stats',
  1603→          params: schoolId != null ? {'p_school_id': schoolId} : {},
  1604→        );
  1605→        return response as Map<String, dynamic>;
  1606→      } on sb.PostgrestException catch (e) {
  1607→        if (e.code != '42883') rethrow; // not undefined function
  1608→
  1609→        // Fallback: manually aggregate stats
  1610→        final questionsQuery = _supabase
  1611→            .from(_generatedQuestionsTable)
  1612→            .select('review_status, question_type, difficulty, bloom_level, '
  1613→                'confidence_score, school_id');
  1614→
  1615→        var filteredQuery = questionsQuery;
  1616→        if (schoolId != null) {
  1617→          filteredQuery = filteredQuery.eq('school_id', schoolId);
  1618→        }
  1619→
  1620→        final questions = await filteredQuery;
  1621→
  1622→        int totalGenerated = questions.length;
  1623→        int totalApproved = 0;
  1624→        int totalRejected = 0;
  1625→        int pendingReview = 0;
  1626→        final questionsByType = <String, int>{};
  1627→        final questionsByDifficulty = <String, int>{};
  1628→        final questionsByBloomLevel = <String, int>{};
  1629→        double totalConfidence = 0;
  1630→        int confidenceCount = 0;
  1631→
  1632→        for (final q in questions) {
  1633→          final status = q['review_status'] as String? ?? 'pending';
  1634→          switch (status) {
  1635→            case 'approved':
  1636→              totalApproved++;
  1637→              break;
  1638→            case 'rejected':
  1639→              totalRejected++;
  1640→              break;
  1641→            default:
  1642→              pendingReview++;
  1643→          }
  1644→
  1645→          final type = q['question_type'] as String? ?? 'unknown';
  1646→          questionsByType[type] = (questionsByType[type] ?? 0) + 1;
  1647→
  1648→          final diff = q['difficulty'] as String? ?? 'unknown';
  1649→          questionsByDifficulty[diff] =
  1650→              (questionsByDifficulty[diff] ?? 0) + 1;
  1651→
  1652→          final bloom = q['bloom_level'] as String?;
  1653→          if (bloom != null) {
  1654→            questionsByBloomLevel[bloom] =
  1655→                (questionsByBloomLevel[bloom] ?? 0) + 1;
  1656→          }
  1657→
  1658→          final conf = q['confidence_score'] as num?;
  1659→          if (conf != null) {
  1660→            totalConfidence += conf.toDouble();
  1661→            confidenceCount++;
  1662→          }
  1663→        }
  1664→
  1665→        // Get cost info from usage stats
  1666→        var usageQuery = _supabase
  1667→            .from(_usageStatsTable)
  1668→            .select('total_cost, total_input_tokens, total_output_tokens, provider');
  1669→        if (schoolId != null) {
  1670→          usageQuery = usageQuery.eq('school_id', schoolId);
  1671→        }
  1672→
  1673→        final usage = await usageQuery;
  1674→        double totalCost = 0;
  1675→        int totalTokensUsed = 0;
  1676→        final costByProvider = <String, double>{};
  1677→
  1678→        for (final u in usage) {
  1679→          final cost = (u['total_cost'] as num?)?.toDouble() ?? 0;
  1680→          totalCost += cost;
  1681→          totalTokensUsed +=
  1682→              (u['total_input_tokens'] as int? ?? 0) +
  1683→                  (u['total_output_tokens'] as int? ?? 0);
  1684→          final prov = u['provider'] as String? ?? 'unknown';
  1685→          costByProvider[prov] = (costByProvider[prov] ?? 0) + cost;
  1686→        }
  1687→
  1688→        return {
  1689→          'total_generated': totalGenerated,
  1690→          'total_approved': totalApproved,
  1691→          'total_rejected': totalRejected,
  1692→          'pending_review': pendingReview,
  1693→          'total_cost': totalCost,
  1694→          'total_tokens_used': totalTokensUsed,
  1695→          'questions_by_type': questionsByType,
  1696→          'questions_by_difficulty': questionsByDifficulty,
  1697→          'questions_by_bloom_level': questionsByBloomLevel,
  1698→          'avg_confidence_score':
  1699→              confidenceCount > 0 ? totalConfidence / confidenceCount : null,
  1700→          'avg_generation_time_ms': null,
  1701→          'recent_generations': [],
  1702→          'cost_by_provider': costByProvider,
  1703→          'daily_usage': [],
  1704→        };
  1705→      }
  1706→    } on sb.PostgrestException catch (e) {
  1707→      throw _mapPostgrestException(e);
  1708→    } catch (e) {
  1709→      AppLogger.error('Unexpected getDashboardStats error', error: e);
  1710→      throw const ServerException(
  1711→        message: 'Failed to get dashboard stats.',
  1712→        statusCode: 500,
  1713→      );
  1714→    }
  1715→  }

  1759→  /// Maps a Supabase PostgrestException to a domain-appropriate exception.
  1760→  ServerException _mapPostgrestException(sb.PostgrestException e) {
  1761→    AppLogger.error('Supabase error: ${e.message}', error: e);
  1762→
  1763→    switch (e.code) {
  1764→      case '23505':
  1765→        return ServerException(
  1766→          message: 'A record with this data already exists.',
  1767→          statusCode: 409,
  1768→          data: e.details,
  1769→        );
  1770→      case '23503':
  1771→        return ServerException(
  1772→          message: 'Referenced record not found.',
  1773→          statusCode: 400,
  1774→          data: e.details,
  1775→        );
  1776→      case '42501':
  1777→      case 'PGRST301':
  1778→        return ServerException(
  1779→          message: 'Insufficient permissions.',
  1780→          statusCode: 403,
  1781→          data: e.details,
  1782→        );
  1783→      case 'PGRST116':
  1784→        return ServerException(
  1785→          message: 'Resource not found.',
  1786→          statusCode: 404,
  1787→          data: e.details,
  1788→        );
  1789→      default:
  1790→        return ServerException(
  1791→          message: e.message ?? 'An unexpected database error occurred.',
  1792→          statusCode: int.tryParse(e.code ?? '') ?? 500,
  1793→          data: e.details,
  1794→        );
  1795→    }
  1796→  }
