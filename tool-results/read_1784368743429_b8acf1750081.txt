   144→   144→/// Provides a configured [Dio] instance with base options pulled from
   145→   145→/// [AppConfig] and [ApiConstants].
   146→   146→final dioProvider = Provider<Dio>((ref) {
   147→   147→  final dio = Dio(
   148→   148→    BaseOptions(
   149→   149→      baseUrl: AppConfig.baseUrl,
   150→   150→      connectTimeout: AppConfig.connectTimeoutDuration,
   151→   151→      receiveTimeout: AppConfig.receiveTimeoutDuration,
   152→   152→      sendTimeout: AppConfig.sendTimeoutDuration,
   153→   153→      headers: ApiConstants.baseHeaders(),
   154→   154→      responseType: ResponseType.json,
   155→   155→      validateStatus: (status) => status != null && status < 500,
   156→   156→    ),
   157→   157→  );
   158→   158→  return dio;
   159→   159→});
