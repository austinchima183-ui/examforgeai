     1→/// Analytics Engine — pure computation service for results analytics.
     2→///
     3→/// This service provides powerful insights for teachers, schools, and students
     4→/// by operating on raw result data and producing computed analytics. It is a
     5→/// computation-only module with zero database access and no external
     6→/// dependencies beyond the entity types defined in the results and CBT domains.
     7→///
     8→/// ## Design Principles
     9→///
    10→/// - **Pure functions**: Every public method is deterministic — same inputs
    11→///   always produce the same outputs. No hidden state, no I/O.
    12→/// - **Static helpers**: Core statistical primitives are exposed as `static`
    13→///   methods for maximum testability and reusability.
    14→/// - **Null-safe**: All methods handle empty input lists gracefully, returning
    15→///   sensible defaults rather than throwing.
    16→/// - **Self-contained DTOs**: Output types are defined in this file to avoid
    17→///   coupling to persistence-layer entities.
    18→///
    19→/// ## Usage
    20→///
    21→/// ```dart
    22→/// final engine = AnalyticsEngine();
    23→/// final summary = engine.computeClassPerformance(overallResults,
    24→///     subjectResults: subjectResults);
    25→/// print('Class average: ${summary.averageScore}');
    26→/// print('Pass rate: ${summary.passRate}%');
    27→/// ```

   385→// ANALYTICS ENGINE
   386→// ═══════════════════════════════════════════════════════════════════════

   388→/// Pure-computation analytics engine for exam result data.
   389→///
   390→/// All methods are side-effect-free and operate solely on the data passed
   391→/// to them. No database access, no network calls, no mutable state.
   392→///
   393→/// The engine is designed to be instantiated once and reused across the
   394→/// application lifecycle, though it carries no state of its own.
   395→///
   396→/// ### Threading
   397→///
   398→/// Because all operations are pure and synchronous, this class is safe to
   399→/// use from any isolate. For large datasets, consider running computations
   400→/// in a background isolate via `Isolate.run`.
   401→///
   402→/// ### Distinction Threshold
   403→///
   404→/// The default distinction threshold is **75%**. This can be overridden via
   405→/// the [distinctionThreshold] parameter on relevant methods.
   406→class AnalyticsEngine {
   407→  // ── Constants ──────────────────────────────────────────────────────
   408→
   409→  /// Default distinction threshold (percentage).
   410→  static const double defaultDistinctionThreshold = 75.0;
   411→
   412→  /// Default pass mark (percentage).
   413→  static const double defaultPassMark = 50.0;
   414→
   415→  /// Number of standard histogram buckets (0–10, 10–20, …, 90–100).
   416→  static const int defaultBucketCount = 10;
   417→
   418→  /// Minimum data points required for a reliable trend calculation.
   419→  static const int minTrendDataPoints = 3;
   420→
   421→  /// Attendance threshold for the "high attendance" group.
   422→  static const double highAttendanceThreshold = 80.0;
   423→
   424→  // ── 1. Class Performance ──────────────────────────────────────────
   425→
   426→  /// Computes a comprehensive performance summary for a class.
   427→  ///
   428→  /// Given a list of [results] (overall results for each student in the
   429→  /// class), this method calculates central tendency, pass/distinction
   430→  /// rates, grade and score distributions, and topic-level aggregations.
   431→  ///
   432→  /// When [subjectResults] are provided, a per-subject breakdown is
   433→  /// included in [ClassPerformanceSummary.subjectBreakdown] and topic
   434→  /// performance is aggregated from those results' metadata.
   435→  ///
   436→  /// Set [previousResults] to compute the [ClassPerformanceSummary.improvementRate]
   437→  /// by comparing the current period's average to the previous period's.
   438→  ///
   439→  /// [distinctionThreshold] defaults to [defaultDistinctionThreshold] (75%).
   440→  /// [passMark] defaults to [defaultPassMark] (50%).
   441→  ClassPerformanceSummary computeClassPerformance(
   442→    List<StudentOverallResultEntity> results, {
   443→    List<StudentSubjectResultEntity>? subjectResults,
   444→    List<StudentOverallResultEntity>? previousResults,
   445→    double distinctionThreshold = defaultDistinctionThreshold,
   446→    double passMark = defaultPassMark,
   447→  }) {
   448→    if (results.isEmpty) {
   449→      return ClassPerformanceSummary(
   450→        classId: '',
   451→        totalStudents: 0,
   452→        averageScore: 0,
   453→        highestScore: 0,
   454→        lowestScore: 0,
   455→        medianScore: 0,
   456→        passRate: 0,
   457→        distinctionRate: 0,
   458→        gradeDistribution: {},
   459→        scoreDistribution: _buildEmptyBuckets(),
   460→        topicPerformance: {},
   461→        improvementRate: 0,
   462→      );
   463→    }
   464→
   465→    final scores = results.map((r) => r.overallPercentage).toList();
   466→    final classId = results.first.classId;
   467→
   468→    // Central tendency
   469→    final average = _mean(scores);
   470→    final highest = _max(scores);
   471→    final lowest = _min(scores);
   472→    final median = _median(scores);
   473→
   474→    // Pass & distinction rates
   475→    final passCount = scores.where((s) => s >= passMark).length;
   476→    final distinctionCount = scores.where((s) => s >= distinctionThreshold).length;
   477→    final passRate = (passCount / scores.length) * 100;
   478→    final distinctionRate = (distinctionCount / scores.length) * 100;
   479→
   480→    // Grade distribution
   481→    final gradeDistribution = <String, int>{};
   482→    for (final r in results) {
   483→      final grade = r.overallGrade ?? _gradeFromPercentage(r.overallPercentage);
   484→      gradeDistribution[grade] = (gradeDistribution[grade] ?? 0) + 1;
   485→    }
   486→
   487→    // Score distribution
   488→    final scoreDistribution = computeScoreDistribution(scores);
   489→
   490→    // Topic performance from subject results metadata
   491→    final topicPerformance = _computeTopicPerformanceFromResults(
   492→      subjectResults ?? [],
   493→    );
   494→
   495→    // Subject breakdown
   496→    final subjectBreakdown = <String, ClassPerformanceSummary>{};
   497→    if (subjectResults != null && subjectResults.isNotEmpty) {
   498→      final bySubject = _groupBy<String, StudentSubjectResultEntity>(
   499→        subjectResults,
   500→        (r) => r.subjectId,
   501→      );
   502→      for (final entry in bySubject.entries) {
   503→        subjectBreakdown[entry.key] = _computeSubjectSummary(
   504→          entry.value,
   505→          distinctionThreshold: distinctionThreshold,
   506→          passMark: passMark,
   507→        );
   508→      }
   509→    }
   510→
   511→    // Improvement rate
   512→    final improvementRate = _computeImprovementRate(results, previousResults);
   513→
   514→    return ClassPerformanceSummary(
   515→      classId: classId,
   516→      totalStudents: results.length,
   517→      averageScore: _round2(average),
   518→      highestScore: _round2(highest),
   519→      lowestScore: _round2(lowest),
   520→      medianScore: _round2(median),
   521→      passRate: _round2(passRate),
   522→      distinctionRate: _round2(distinctionRate),
   523→      gradeDistribution: gradeDistribution,
   524→      scoreDistribution: scoreDistribution,
   525→      topicPerformance: topicPerformance,
   526→      improvementRate: _round2(improvementRate),
   527→      subjectBreakdown: subjectBreakdown,
   528→    );
   529→  }
   530→
   531→  // ── 2. School Performance ──────────────────────────────────────────
   532→
   533→  /// Aggregates multiple [ClassPerformanceSummary] instances into a
   534→  /// school-level summary with cross-class and cross-subject rankings.
   535→  ///
   536→  /// [topicMasteryData] is used to identify the most difficult topic
   537→  /// across the entire school.
   538→  ///
   539→  /// [trendData] provides historical performance data for trend
   540→  /// visualization on dashboards.
   541→  SchoolPerformanceSummary computeSchoolPerformance(
   542→    List<ClassPerformanceSummary> classPerformances, {
   543→    required String schoolId,
   544→    required String academicSessionId,
   545→    List<TopicMasteryEntity>? topicMasteryData,
   546→    List<TrendData>? trendData,
   547→    int totalExams = 0,
   548→  }) {
   549→    if (classPerformances.isEmpty) {
   550→      return SchoolPerformanceSummary(
   551→        schoolId: schoolId,
   552→        academicSessionId: academicSessionId,
   553→        totalStudents: 0,
   554→        totalClasses: 0,
   555→        averageScore: 0,
   556→        passRate: 0,
   557→        distinctionRate: 0,
   558→        bestClassId: null,
   559→        bestSubjectId: null,
   560→        mostDifficultTopicId: null,
   561→        classRankings: [],
   562→        subjectRankings: [],
   563→        gradeDistribution: {},
   564→        trendData: trendData ?? [],
   565→        totalExams: totalExams,
   566→      );
   567→    }
   568→
   569→    final totalStudents = classPerformances.fold<int>(
   570→      0,
   571→      (sum, c) => sum + c.totalStudents,
   572→    );
   573→
   574→    // Weighted average: weight each class by its student count.
   575→    final weightedSum = classPerformances.fold<double>(
   576→      0,
   577→      (sum, c) => sum + (c.averageScore * c.totalStudents),
   578→    );
   579→    final averageScore = totalStudents > 0 ? weightedSum / totalStudents : 0.0;
   580→
   581→    // Weighted pass rate
   582→    final passWeightedSum = classPerformances.fold<double>(
   583→      0,
   584→      (sum, c) => sum + (c.passRate * c.totalStudents),
   585→    );
   586→    final passRate = totalStudents > 0 ? passWeightedSum / totalStudents : 0.0;
   587→
   588→    // Weighted distinction rate
   589→    final distinctionWeightedSum = classPerformances.fold<double>(
   590→      0,
   591→      (sum, c) => sum + (c.distinctionRate * c.totalStudents),
   592→    );
   593→    final distinctionRate =
   594→        totalStudents > 0 ? distinctionWeightedSum / totalStudents : 0.0;
   595→
   596→    // Class rankings
   597→    final classRankings = _rankEntities(
   598→      classPerformances.map((c) => MapEntry(c.classId, c.averageScore)).toList(),
   599→    );
   600→
   601→    // Subject rankings from breakdowns
   602→    final subjectScores = <String, List<double>>{};
   603→    for (final cp in classPerformances) {
   604→      for (final sb in cp.subjectBreakdown.entries) {
   605→        subjectScores.putIfAbsent(sb.key, () => []).add(sb.value.averageScore);
   606→      }
   607→    }
   608→    final subjectAverages = subjectScores.map(
   609→      (k, v) => MapEntry(k, _mean(v)),
   610→    );
   611→    final subjectRankings = _rankEntities(
   612→      subjectAverages.entries.map((e) => MapEntry(e.key, e.value)).toList(),
   613→    );
   614→
   615→    // Best class & subject
   616→    final bestClassId = classRankings.isNotEmpty ? classRankings.first.entityId : null;
   617→    final bestSubjectId = subjectRankings.isNotEmpty ? subjectRankings.first.entityId : null;
   618→
   619→    // Most difficult topic
   620→    String? mostDifficultTopicId;
   621→    if (topicMasteryData != null && topicMasteryData.isNotEmpty) {
   622→      final difficultTopics = identifyDifficultTopics(topicMasteryData);
   623→      if (difficultTopics.isNotEmpty) {
   624→        mostDifficultTopicId = difficultTopics.first.topicId;
   625→      }
   626→    } else {
   627→      // Fallback: look at topic performance from class summaries
   628→      final allTopics = <String, List<TopicPerformanceData>>{};
   629→      for (final cp in classPerformances) {
   630→        for (final tp in cp.topicPerformance.entries) {
   631→          allTopics.putIfAbsent(tp.key, () => []).add(tp.value);
   632→        }
   633→      }
   634→      if (allTopics.isNotEmpty) {
   635→        String? worstTopic;
   636→        double worstAccuracy = 101;
   637→        for (final entry in allTopics.entries) {
   638→          final avg = _mean(entry.value.map((t) => t.averageAccuracy).toList());
   639→          if (avg < worstAccuracy) {
   640→            worstAccuracy = avg;
   641→            worstTopic = entry.key;
   642→          }
   643→        }
   644→        mostDifficultTopicId = worstTopic;
   645→      }
   646→    }
   647→
   648→    // Grade distribution: merge all class distributions
   649→    final gradeDistribution = <String, int>{};
   650→    for (final cp in classPerformances) {
   651→      for (final entry in cp.gradeDistribution.entries) {
   652→        gradeDistribution[entry.key] =
   653→            (gradeDistribution[entry.key] ?? 0) + entry.value;
   654→      }
   655→    }
   656→
   657→    return SchoolPerformanceSummary(

   678→  /// Computes a student's performance trend from historical scores using
   679→  /// simple linear regression.
   680→  ///
   681→  /// [historicalScores] should be ordered chronologically (oldest first).
   682→  /// At least [minTrendDataPoints] data points are needed for a reliable
   683→  /// result; fewer will still produce a trend but with low confidence.
   684→  ///
   685→  /// [strengths] and [weaknesses] are subject/topic IDs where the student
   686→  /// consistently scores above or below their own average.
   687→  StudentTrendAnalysis computeStudentTrends({
   688→    required String studentId,
   689→    required List<double> historicalScores,
   690→    List<String> strengths = const [],
   691→    List<String> weaknesses = const [],
   692→  }) {
   693→    if (historicalScores.length < 2) {
   694→      return StudentTrendAnalysis(
   695→        studentId: studentId,
   696→        trend: PerformanceTrend.stable,
   697→        slope: 0,
   698→        confidence: 0,
   699→        strengths: strengths,
   700→        weaknesses: weaknesses,
   701→        recommendations: _generateRecommendations(
   702→          PerformanceTrend.stable,
   703→          0,
   704→          strengths,
   705→          weaknesses,
   706→        ),
   707→      );
   708→    }
   709→
   710→    final regression = _linearRegression(historicalScores);
   711→    final slope = regression.slope;
   712→    final confidence = regression.rSquared;
   713→
   714→    // Determine trend from slope
   715→    PerformanceTrend trend;
   716→    if (slope > 1.5) {
   717→      trend = PerformanceTrend.improving;
   718→    } else if (slope < -1.5) {
   719→      trend = PerformanceTrend.declining;
   720→    } else {
   721→      trend = PerformanceTrend.stable;
   722→    }
   723→
   724→    final recommendations = _generateRecommendations(
   725→      trend,
   726→      slope,
   727→      strengths,
   728→      weaknesses,
   729→    );
   730→
   731→    return StudentTrendAnalysis(
   732→      studentId: studentId,
   733→      trend: trend,
   734→      slope: _round2(slope),

   737→      weaknesses: weaknesses,

   766→  // ── 5. Rankings ───────────────────────────────────────────────────
   767→
   768→  /// Generates a ranked list of students from their overall results.
   769→  ///
   770→  /// Uses standard competition ranking (also known as "1224" ranking):
   771→  /// students with identical scores share the same rank, and the next
   772→  /// rank is skipped. For example: 1st, 2nd, 2nd, 4th.
   773→  ///
   774→  /// Results are sorted by [StudentOverallResultEntity.overallPercentage]
   775→  /// in descending order.
   776→  List<RankedStudent> generateRankings(
   777→    List<StudentOverallResultEntity> results,
   778→  ) {
   779→    if (results.isEmpty) return [];
   780→
   781→    // Sort descending by percentage
   782→    final sorted = List<StudentOverallResultEntity>.from(results)
   783→      ..sort((a, b) => b.overallPercentage.compareTo(a.overallPercentage));
   784→
   785→    final rankings = <RankedStudent>[];
   786→    int? lastRank;
   787→    double? lastScore;
   788→
   789→    for (int i = 0; i < sorted.length; i++) {
   790→      final result = sorted[i];
   791→      final position = i + 1;
   792→
   793→      // Same score → same rank; otherwise rank = position
   794→      final rank = (lastScore != null &&
   795→              _round2(result.overallPercentage) == _round2(lastScore))
   796→          ? lastRank!
   797→          : position;
   798→
   799→      rankings.add(RankedStudent(
   800→        studentId: result.studentId,
   801→        rank: rank,
   802→        score: _round2(result.overallPercentage),
   803→        classId: result.classId,
   804→      ));
   805→
   806→      lastRank = rank;
   807→      lastScore = _round2(result.overallPercentage);
   808→    }
   809→
   810→    return rankings;
   811→  }

   815→  /// Creates histogram buckets for score visualization.
   816→  ///
   817→  /// By default, creates [defaultBucketCount] buckets of equal width
   818→  /// (0–10, 10–20, …, 90–100). Each bucket is inclusive of its lower
   819→  /// bound and inclusive of its upper bound, except the last bucket
   820→  /// which includes 100.
   821→  ///
   822→  /// Returns a map of range label → midpoint for quick chart integration.
   823→  Map<String, double> computeScoreDistributionMap(List<double> scores) {
   824→    final buckets = computeScoreDistribution(scores);
   825→    return {for (final b in buckets) b.rangeLabel: (b.min + b.max) / 2};
   826→  }
   827→
   828→  /// Creates histogram buckets for score visualization.
   829→  ///
   830→  /// Returns a list of [ScoreBucket] instances with count and percentage
   831→  /// for each range.
   832→  List<ScoreBucket> computeScoreDistribution(
   833→    List<double> scores, {
   834→    int bucketCount = defaultBucketCount,
   835→  }) {
   836→    if (scores.isEmpty) return _buildEmptyBuckets(bucketCount: bucketCount);
   837→
   838→    final bucketWidth = 100.0 / bucketCount;
   839→    final buckets = <ScoreBucket>[];
   840→
   841→    for (int i = 0; i < bucketCount; i++) {
   842→      final min = i * bucketWidth;
   843→      final max = (i + 1) * bucketWidth;
   844→
   845→      // For the last bucket, include 100. For others, use strict upper bound.
   846→      final count = scores.where((s) {
   847→        if (i == bucketCount - 1) {
   848→          return s >= min && s <= max;
   849→        }
   850→        return s >= min && s < max;
   851→      }).length;
   852→
   853→      final percentage = (count / scores.length) * 100;
   854→
   855→      buckets.add(ScoreBucket(
   856→        rangeLabel: '${min.round()}–${max.round()}',
   857→        min: min,
   858→        max: max,
   859→        count: count,
   860→        percentage: _round2(percentage),
   861→      ));
   862→    }
   863→
   864→    return buckets;
   865→  }

   911→  /// Computes the Pearson correlation coefficient between attendance
   912→  /// percentage and performance score.
   913→  ///
   914→  /// [attendancePerformancePairs] is a list of (attendance%, score%)
   915→  /// pairs for each student.
   916→  ///
   917→  /// Returns an [AttendanceCorrelation] with the coefficient, sample
   918→  /// size, and a human-readable interpretation.
   919→  AttendanceCorrelation computeAttendanceCorrelation(
   920→    List<({double attendance, double performance})> attendancePerformancePairs,
   921→  ) {
   922→    if (attendancePerformancePairs.length < 3) {
   923→      return AttendanceCorrelation(
   924→        correlationCoefficient: 0,
   925→        sampleSize: attendancePerformancePairs.length,
   926→        interpretation: 'Insufficient data for correlation analysis '
   927→            '(need at least 3 data points)',
   928→      );
   929→    }
   930→
   931→    final n = attendancePerformancePairs.length;
   932→    final attendanceValues =
   933→        attendancePerformancePairs.map((p) => p.attendance).toList();
   934→    final performanceValues =
   935→        attendancePerformancePairs.map((p) => p.performance).toList();
   936→
   937→    final coefficient =
   938→        _pearsonCorrelation(attendanceValues, performanceValues);
   939→
   940→    // Compute average scores for high/low attendance groups
   941→    final highAttendance = attendancePerformancePairs
   942→        .where((p) => p.attendance >= highAttendanceThreshold)
   943→        .map((p) => p.performance)
   944→        .toList();
   945→    final lowAttendance = attendancePerformancePairs
   946→        .where((p) => p.attendance < highAttendanceThreshold)
   947→        .map((p) => p.performance)
   948→        .toList();
   949→
   950→    final highAvg = highAttendance.isNotEmpty ? _mean(highAttendance) : null;
   951→    final lowAvg = lowAttendance.isNotEmpty ? _mean(lowAttendance) : null;
   952→
   953→    final interpretation = _interpretCorrelation(coefficient);
   954→
   955→    return AttendanceCorrelation(
   956→      correlationCoefficient: _round4(coefficient),
   957→      sampleSize: n,
   958→      interpretation: interpretation,
   959→      highAttendanceAvgScore: highAvg != null ? _round2(highAvg) : null,
   960→      lowAttendanceAvgScore: lowAvg != null ? _round2(lowAvg) : null,
   961→    );
   962→  }

   985→  /// Arithmetic mean of a list of numbers.
   986→  static double _mean(List<double> values) {
   987→    if (values.isEmpty) return 0;
   988→    return values.reduce((a, b) => a + b) / values.length;
   989→  }
   990→
   991→  /// Maximum value in a list, or 0 if empty.
   992→  static double _max(List<double> values) {
   993→    if (values.isEmpty) return 0;
   994→    return values.reduce((a, b) => a > b ? a : b);
   995→  }
   996→
   997→  /// Minimum value in a list, or 0 if empty.
   998→  static double _min(List<double> values) {
   999→    if (values.isEmpty) return 0;
  1000→    return values.reduce((a, b) => a < b ? a : b);
  1001→  }
  1002→
  1003→  /// Median (50th percentile) of a sorted copy of [values].
  1004→  static double _median(List<double> values) {
  1005→    if (values.isEmpty) return 0;
  1006→    final sorted = List<double>.from(values)..sort();
  1007→    final mid = sorted.length ~/ 2;
  1008→    if (sorted.length.isEven) {
  1009→      return (sorted[mid - 1] + sorted[mid]) / 2;
  1010→    }
  1011→    return sorted[mid];
  1012→  }
  1013→
  1014→  /// Standard deviation (population) of a list.
  1015→  static double _standardDeviation(List<double> values) {
  1016→    if (values.length < 2) return 0;
  1017→    final mean = _mean(values);
  1018→    final sumSquaredDiff = values.fold<double>(
  1019→      0,
  1020→      (sum, v) => sum + (v - mean) * (v - mean),
  1021→    );
  1022→    return _sqrt(sumSquaredDiff / values.length);
  1023→  }
  1024→
  1025→  /// Integer square root approximation for standard deviation.
  1026→  /// Uses Newton's method for reasonable precision.
  1027→  static double _sqrt(double value) {
  1028→    if (value < 0) return 0;
  1029→    if (value == 0) return 0;
  1030→    double x = value;
  1031→    double prev = 0;
  1032→    for (int i = 0; i < 20; i++) {
  1033→      prev = x;
  1034→      x = 0.5 * (x + value / x);
  1035→      if ((x - prev).abs() < 0.000001) break;
  1036→    }
  1037→    return x;
  1038→  }
  1039→
  1040→  // ── Linear Regression ──────────────────────────────────────────────
  1041→
  1042→  /// Simple linear regression (OLS) on [y] values with implicit
  1043→  /// x = 0, 1, 2, …, n−1.
  1044→  ///
  1045→  /// Returns the slope, intercept, and R² value.
  1046→  static _LinearRegressionResult _linearRegression(List<double> y) {
  1047→    final n = y.length;
  1048→    if (n < 2) {
  1049→      return _LinearRegressionResult(slope: 0, intercept: _mean(y), rSquared: 0);
  1050→    }
  1051→
  1052→    // x values: 0, 1, 2, ..., n-1
  1053→    final xMean = (n - 1) / 2.0;
  1054→    final yMean = _mean(y);
  1055→
  1056→    double sumXY = 0;
  1057→    double sumX2 = 0;
  1058→    for (int i = 0; i < n; i++) {
  1059→      final xDiff = i - xMean;
  1060→      final yDiff = y[i] - yMean;
  1061→      sumXY += xDiff * yDiff;
  1062→      sumX2 += xDiff * xDiff;
  1063→    }
  1064→
  1065→    final slope = sumX2 != 0 ? sumXY / sumX2 : 0.0;
  1066→    final intercept = yMean - slope * xMean;
  1067→
  1068→    // R² calculation
  1069→    final ssTotal = y.fold<double>(0, (sum, v) => sum + (v - yMean) * (v - yMean));
  1070→    double ssResidual = 0;
  1071→    for (int i = 0; i < n; i++) {
  1072→      final predicted = intercept + slope * i;
  1073→      ssResidual += (y[i] - predicted) * (y[i] - predicted);
  1074→    }
  1075→    final rSquared = ssTotal != 0 ? 1 - (ssResidual / ssTotal) : 0.0;
  1076→
  1077→    return _LinearRegressionResult(
  1078→      slope: slope,
  1079→      intercept: intercept,
  1080→      rSquared: rSquared.clamp(0, 1),
  1081→    );
  1082→  }
  1083→
  1084→  // ── Pearson Correlation ────────────────────────────────────────────
  1085→
  1086→  /// Computes the Pearson correlation coefficient between [x] and [y].
  1087→  static double _pearsonCorrelation(List<double> x, List<double> y) {
  1088→    assert(x.length == y.length);
  1089→    final n = x.length;
  1090→    if (n < 2) return 0;
  1091→
  1092→    final xMean = _mean(x);
  1093→    final yMean = _mean(y);
  1094→
  1095→    double sumXY = 0;
  1096→    double sumX2 = 0;
  1097→    double sumY2 = 0;
  1098→
  1099→    for (int i = 0; i < n; i++) {
  1100→      final dx = x[i] - xMean;
  1101→      final dy = y[i] - yMean;
  1102→      sumXY += dx * dy;
  1103→      sumX2 += dx * dx;
  1104→      sumY2 += dy * dy;
  1105→    }
  1106→
  1107→    final denominator = _sqrt(sumX2) * _sqrt(sumY2);
  1108→    if (denominator == 0) return 0;
  1109→    return (sumXY / denominator).clamp(-1.0, 1.0);
  1110→  }

  1114→  /// Groups [items] by a key extracted via [keyExtractor].

  1116→    List<T> items,
  1117→    K Function(T) keyExtractor,
  1118→  ) {
  1119→    final map = <K, List<T>>{};
  1120→    for (final item in items) {
  1121→      final key = keyExtractor(item);
  1122→      map.putIfAbsent(key, () => []).add(item);
  1123→    }
  1124→    return map;
  1125→  }

  1234→  ClassPerformanceSummary _computeSubjectSummary(
  1235→    List<StudentSubjectResultEntity> results, {
  1236→    double distinctionThreshold = defaultDistinctionThreshold,
  1237→    double passMark = defaultPassMark,
  1238→  }) {
  1239→    if (results.isEmpty) {
  1240→      return ClassPerformanceSummary(
  1241→        classId: '',
  1242→        totalStudents: 0,
  1243→        averageScore: 0,
  1244→        highestScore: 0,
  1245→        lowestScore: 0,
  1246→        medianScore: 0,
  1247→        passRate: 0,
  1248→        distinctionRate: 0,
  1249→        gradeDistribution: {},
  1250→        scoreDistribution: _buildEmptyBuckets(),
  1251→        topicPerformance: {},
  1252→        improvementRate: 0,
  1253→      );
  1254→    }
  1255→
  1256→    final scores = results.map((r) => r.percentage).toList();
  1257→    final subjectId = results.first.subjectId;
  1258→    final classId = results.first.classId;
  1259→
  1260→    final gradeDistribution = <String, int>{};
  1261→    for (final r in results) {
  1262→      final grade = r.grade ?? _gradeFromPercentage(r.percentage);
  1263→      gradeDistribution[grade] = (gradeDistribution[grade] ?? 0) + 1;
  1264→    }
  1265→
  1266→    return ClassPerformanceSummary(
  1267→      classId: classId,
  1268→      subjectId: subjectId,
  1269→      totalStudents: results.length,
  1270→      averageScore: _round2(_mean(scores)),
  1271→      highestScore: _round2(_max(scores)),
  1272→      lowestScore: _round2(_min(scores)),
  1273→      medianScore: _round2(_median(scores)),
  1274→      passRate: _round2(
  1275→        (scores.where((s) => s >= passMark).length / scores.length) * 100,
  1276→      ),
  1277→      distinctionRate: _round2(
  1278→        (scores.where((s) => s >= distinctionThreshold).length / scores.length) *
  1279→            100,
  1280→      ),
  1281→      gradeDistribution: gradeDistribution,
  1282→      scoreDistribution: computeScoreDistribution(scores),
  1283→      topicPerformance: _computeTopicPerformanceFromResults(results),
  1284→      improvementRate: 0,
  1285→    );
  1286→  }

  1290→  /// Computes the improvement rate between current and previous results.
  1291→  ///
  1292→  /// Returns the average delta in percentage points (positive = improvement).
  1293→  /// Matches students by ID. Students not in both lists are excluded.
  1294→  static double _computeImprovementRate(
  1295→    List<StudentOverallResultEntity> current,
  1296→    List<StudentOverallResultEntity>? previous,
  1297→  ) {
  1298→    if (previous == null || previous.isEmpty) return 0;
  1299→
  1300→    final previousMap = <String, double>{};
  1301→    for (final r in previous) {
  1302→      previousMap[r.studentId] = r.overallPercentage;
  1303→    }
  1304→
  1305→    double totalDelta = 0;
  1306→    int matchedCount = 0;
  1307→    for (final r in current) {
  1308→      final prevScore = previousMap[r.studentId];
  1309→      if (prevScore != null) {
  1310→        totalDelta += r.overallPercentage - prevScore;
  1311→        matchedCount++;
  1312→      }
  1313→    }
  1314→
  1315→    return matchedCount > 0 ? totalDelta / matchedCount : 0;
  1316→  }

  1383→  /// Returns a human-readable interpretation of a correlation coefficient.
  1384→  static String _interpretCorrelation(double r) {
  1385→    final absR = r.abs();
  1386→    final direction = r >= 0 ? 'positive' : 'negative';
  1387→
  1388→    String strength;
  1389→    if (absR >= 0.8) {
  1390→      strength = 'very strong';
  1391→    } else if (absR >= 0.6) {
  1392→      strength = 'strong';
  1393→    } else if (absR >= 0.4) {
  1394→      strength = 'moderate';
  1395→    } else if (absR >= 0.2) {
  1396→      strength = 'weak';
  1397→    } else {
  1398→      strength = 'very weak';
  1399→    }
  1400→
  1401→    return 'There is a $strength $direction correlation between attendance '
  1402→        'and performance (r = ${r.toStringAsFixed(3)}).';
  1403→  }
