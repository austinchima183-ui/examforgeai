   927→   927→-- ensure_single_current_session()
   928→   928→-- Ensures only one session per school is marked as current.
   929→   929→-- ---------------------------------------------------------------------------
   930→   930→CREATE OR REPLACE FUNCTION ensure_single_current_session()
   931→   931→RETURNS TRIGGER AS $$
   932→   932→BEGIN
   933→   933→  IF NEW.is_current = true THEN
   934→   934→    UPDATE academic_sessions
   935→   935→    SET is_current = false
   936→   936→    WHERE school_id = NEW.school_id
   937→   937→      AND id != NEW.id
   938→   938→      AND is_current = true;
   939→   939→  END IF;
   940→   940→  RETURN NEW;
   941→   941→END;
   942→   942→$$ LANGUAGE plpgsql SECURITY DEFINER;
   943→   943→
   944→   944→DROP TRIGGER IF EXISTS trg_ensure_single_current_session ON academic_sessions;
   945→   945→CREATE TRIGGER trg_ensure_single_current_session
   946→   946→  BEFORE INSERT OR UPDATE OF is_current ON academic_sessions
   947→   947→  FOR EACH ROW
   948→   948→  WHEN (NEW.is_current = true)
   949→   949→  EXECUTE FUNCTION ensure_single_current_session();

   977→   977→-- detect_timetable_conflicts()
   978→   978→-- Raises an exception if a teacher or class is double-booked.
   979→   979→-- ---------------------------------------------------------------------------
   980→   980→CREATE OR REPLACE FUNCTION detect_timetable_conflicts()
   981→   981→RETURNS TRIGGER AS $$
   982→   982→BEGIN
