    33→  String? _filterProviderId;
    34→  DateTimeRange? _filterDateRange;
    35→
    36→  @override
    37→  void initState() {
    38→    super.initState();
    39→    WidgetsBinding.instance.addPostFrameCallback((_) {
    40→      _loadData();

    44→  void _loadData() {
    45→    ref.read(aiManagementProvider.notifier).loadProviders();
    46→    ref.read(aiManagementProvider.notifier).loadRequestLogs(
    47→      providerId: _filterProviderId,
    48→      startDate: _filterDateRange?.start,
    49→      endDate: _filterDateRange?.end,

    52→
    53→  // ─── Provider Status Color ──────────────────────────────────────────────
    54→
    55→  Color _providerStatusColor(AIProviderStatus status) {
    56→    switch (status) {
    57→      case AIProviderStatus.active:
    58→        return AppColors.success;
    59→      case AIProviderStatus.inactive:
    60→        return AppColors.error;
    61→      case AIProviderStatus.degraded:
    62→        return AppColors.warning;
    63→      case AIProviderStatus.maintenance:
    64→        return AppColors.info;
    65→      case AIProviderStatus.suspended:
    66→        return AppColors.error;
    67→    }
    68→  }
    69→
    70→  IconData _providerStatusIcon(AIProviderStatus status) {
    71→    switch (status) {
    72→      case AIProviderStatus.active:
    73→        return Icons.check_circle;
    74→      case AIProviderStatus.inactive:
    75→        return Icons.cancel;
    76→      case AIProviderStatus.degraded:
    77→        return Icons.warning_amber;
    78→      case AIProviderStatus.maintenance:
    79→        return Icons.build;
    80→      case AIProviderStatus.suspended:
    81→        return Icons.block;
    82→    }
    83→  }
    84→
    85→  Color _budgetBarColor(double utilization) {
    86→    if (utilization >= 90) return AppColors.error;
    87→    if (utilization >= 70) return AppColors.warning;
    88→    return AppColors.success;
    89→  }

    92→
    93→  @override

   100→      if (next.successMessage != null && next.successMessage != prev?.successMessage) {
   101→        ScaffoldMessenger.of(context).showSnackBar(
   102→          SnackBar(
   103→            content: Text(next.successMessage!),

   128→            tooltip: 'Refresh',

   257→            subtitle: 'This month',
   258→          ),
   259→        ),
   260→        const SizedBox(width: Spacings.md),
   261→        Expanded(
   262→          child: MetricCard(

   264→            value: '${avgLatency.toStringAsFixed(0)}ms',

   271→        Expanded(

   273→            title: 'Success Rate',

   275→            icon: Icons.check_circle_outline,

   341→  // ─── Request Logs DataTable ─────────────────────────────────────────────
   342→
   343→  Widget _buildRequestLogsTable(AIManagementState state, ColorScheme cs) {
   344→    final logs = state.requestLogs;
   345→
   346→    if (logs.isEmpty) {
   347→      return const AdminEmptyState(
   348→        message: 'No request logs found.',
   349→        icon: Icons.receipt_long_outlined,
   350→      );
   351→    }
   352→
   353→    // Build a name lookup map from provider id → name
   354→    final providerNameMap = <String, String>{};
   355→    for (final p in state.providers) {
   356→      providerNameMap[p.id] = p.name;
   357→    }
   358→
   359→    return Card(
   360→      elevation: Spacings.elevationSm,
   361→      shape: RoundedRectangleBorder(borderRadius: Spacings.borderRadiusMd),
   362→      child: SingleChildScrollView(
   363→        scrollDirection: Axis.horizontal,
   364→        child: DataTable(
   365→          columnSpacing: Spacings.xl,
   366→          headingRowColor: WidgetStatePropertyAll(cs.surfaceContainerHighest),
   367→          columns: [
   368→            DataColumn(label: _tableHeader('Timestamp', cs)),
   369→            DataColumn(label: _tableHeader('Provider', cs)),
   370→            DataColumn(label: _tableHeader('Model', cs)),
   371→            DataColumn(label: _tableHeader('Request Type', cs)),
   372→            DataColumn(label: _tableHeader('Tokens In', cs)),
   373→            DataColumn(label: _tableHeader('Tokens Out', cs)),
   374→            DataColumn(label: _tableHeader('Total Tokens', cs)),
   375→            DataColumn(label: _tableHeader('Cost', cs)),
   376→            DataColumn(label: _tableHeader('Latency', cs)),
   377→            DataColumn(label: _tableHeader('Status', cs)),
   378→          ],
   379→          rows: logs.map((log) {
   380→            final isSuccess = log.isSuccess;
   381→            return DataRow(
   382→              cells: [
   383→                DataCell(Text(
   384→                  _formatTimestamp(log.createdAt),
   385→                  style: AppTypography.wRegular.copyWith(fontSize: 12),
   386→                )),
   387→                DataCell(Text(
   388→                  providerNameMap[log.providerId] ?? log.providerId.substring(0, 8),
   389→                  style: AppTypography.wMedium.copyWith(fontSize: 12),
   390→                )),
   391→                DataCell(Text(
   392→                  log.model,
   393→                  style: AppTypography.wRegular.copyWith(fontSize: 12),
   394→                )),
   395→                DataCell(Text(
   396→                  log.requestType,
   397→                  style: AppTypography.wRegular.copyWith(fontSize: 12),
   398→                )),
   399→                DataCell(Text(
   400→                  _formatNumber(log.inputTokens),
   401→                  style: AppTypography.wRegular.copyWith(fontSize: 12),
   402→                )),
   403→                DataCell(Text(
   404→                  _formatNumber(log.outputTokens),
   405→                  style: AppTypography.wRegular.copyWith(fontSize: 12),
   406→                )),
   407→                DataCell(Text(
   408→                  _formatNumber(log.totalTokens),
   409→                  style: AppTypography.wSemiBold.copyWith(fontSize: 12),
   410→                )),
   411→                DataCell(Text(
   412→                  '\$${log.cost.toStringAsFixed(4)}',
   413→                  style: AppTypography.wSemiBold.copyWith(fontSize: 12),
   414→                )),
   415→                DataCell(Text(
   416→                  '${log.latencyMs}ms',
   417→                  style: AppTypography.wRegular.copyWith(
   418→                    fontSize: 12,
   419→                    color: log.latencyMs > 3000
   420→                        ? AppColors.error
   421→                        : log.latencyMs > 1500
   422→                            ? AppColors.warning
   423→                            : null,
   424→                  ),
   425→                )),
   426→                DataCell(
   427→                  StatusBadge(
   428→                    label: isSuccess ? 'Success' : 'Failed',
   429→                    color: isSuccess ? AppColors.success : AppColors.error,
   430→                    icon: isSuccess ? Icons.check_circle : Icons.error,
   431→                  ),
   432→                ),
   433→              ],
   434→            );
   435→          }).toList(),
   436→        ),
   437→      ),
   438→    );
   439→  }

   451→  // ─── Date Range Picker ──────────────────────────────────────────────────
   452→
   453→  Future<void> _pickDateRange() async {
   454→    final now = DateTime.now();
   455→    final picked = await showDateRangePicker(
   456→      context: context,
   457→      firstDate: now.subtract(const Duration(days: 90)),
   458→      lastDate: now,
   459→      initialDateRange: _filterDateRange,
   460→    );
   461→    if (picked != null) {
   462→      setState(() => _filterDateRange = picked);
   463→      ref.read(aiManagementProvider.notifier).loadRequestLogs(
   464→            providerId: _filterProviderId,
   465→            startDate: picked.start,
   466→            endDate: picked.end,
   467→          );
   468→    }
   469→  }

   733→  String _formatTimestamp(DateTime dt) {
   734→    return '${dt.year}-${_pad(dt.month)}-${_pad(dt.day)} '
   735→        '${_pad(dt.hour)}:${_pad(dt.minute)}';
   736→  }
   737→
   738→  String _pad(int n) => n.toString().padLeft(2, '0');
   739→
   740→  String _formatNumber(int n) {
   741→    if (n >= 1000000) return '${(n / 1000000).toStringAsFixed(1)}M';
   742→    if (n >= 1000) return '${(n / 1000).toStringAsFixed(1)}K';
   743→    return n.toString();
   744→  }
