   148→/// Provides the Supabase client singleton.

   152→  try {

   155→    AppLogger.error('Failed to resolve Supabase client', error: e);
   156→    rethrow;
   157→  }
   158→});
   159→
   160→/// Provides a configured [Dio] instance with base options pulled from
   161→/// [AppConfig] and [ApiConstants].
   162→final dioProvider = Provider<Dio>((ref) {
   163→  final dio = Dio(
   164→    BaseOptions(
   165→      baseUrl: AppConfig.baseUrl,
   166→      connectTimeout: AppConfig.connectTimeoutDuration,
   167→      receiveTimeout: AppConfig.receiveTimeoutDuration,
   168→      sendTimeout: AppConfig.sendTimeoutDuration,
   169→      headers: ApiConstants.baseHeaders(),
   170→      responseType: ResponseType.json,
   171→      validateStatus: (status) => status != null && status < 500,
   172→    ),
   173→  );
   174→  return dio;
   175→});
   176→
   177→/// Provides the [ApiClient] wrapper around [Dio].
   178→final apiClientProvider = Provider<ApiClient>((ref) {
   179→  final dio = ref.watch(dioProvider);

   231→/// Stream provider that emits the current Supabase [sb.AuthState].

   233→/// Automatically reconnects on transient failures and emits the latest
   234→/// state to any listener.
   235→final authStateProvider = StreamProvider<sb.AuthState>((ref) {
   236→  final supabaseClient = ref.watch(supabaseClientProvider);
   237→  return supabaseClient.auth.onAuthStateChange;
   238→});
   239→
   240→/// Convenience provider that yields the currently authenticated [sb.User],
   241→/// or `null` if not signed in.
   242→final currentUserProvider = Provider<sb.User?>((ref) {
   243→  final authState = ref.watch(authStateProvider);
   244→  return authState.when(
   245→    data: (state) => state.session?.user,
   246→    loading: () => SupabaseConfig.currentUser,
   247→    error: (_, __) => null,
   248→  );
   249→});

   251→/// Convenience provider that yields `true` when the user is authenticated.
   252→final isAuthenticatedProvider = Provider<bool>((ref) {
   253→  final user = ref.watch(currentUserProvider);
   254→  return user != null;
   255→});
