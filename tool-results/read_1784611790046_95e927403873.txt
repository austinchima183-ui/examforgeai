   158→   158→/// Real-time progress update during streaming generation.
   159→   159→class GenerationProgress {
   160→   160→  const GenerationProgress({
   161→   161→    required this.requestId,
   162→   162→    this.status = GenerationProgressStatus.initializing,
   163→   163→    this.questionsCompleted = 0,
   164→   164→    this.questionsTotal = 0,
   165→   165→    this.currentChunk,
   166→   166→    this.partialContent,
   167→   167→    this.validationResults = const [],
   168→   168→    this.error,
   169→   169→  });
   170→   170→
   171→   171→  /// The generation request ID.
   172→   172→  final String requestId;
   173→   173→
   174→   174→  /// Current status of the generation.
   175→   175→  final GenerationProgressStatus status;
   176→   176→
   177→   177→  /// Number of questions completed so far.
   178→   178→  final int questionsCompleted;
   179→   179→
   180→   180→  /// Total number of questions requested.
   181→   181→  final int questionsTotal;
   182→   182→
   183→   183→  /// The latest streaming chunk (if streaming).
   184→   184→  final AiCompletionChunk? currentChunk;
   185→   185→
   186→   186→  /// Partial content accumulated so far.
   187→   187→  final String? partialContent;
   188→   188→
   189→   189→  /// Validation results for completed questions.
   190→   190→  final List<ValidationResultEntity> validationResults;
   191→   191→
   192→   192→  /// Error message if a step failed.
   193→   193→  final String? error;
   194→   194→
   195→   195→  /// Progress as a fraction (0.0 – 1.0).
   196→   196→  double get progress {
   197→   197→    if (questionsTotal == 0) return 0.0;
   198→   198→    return (questionsCompleted / questionsTotal).clamp(0.0, 1.0);
   199→   199→  }
   200→   200→
   201→   201→  GenerationProgress copyWith({
   202→   202→    String? requestId,
   203→   203→    GenerationProgressStatus? status,
   204→   204→    int? questionsCompleted,
   205→   205→    int? questionsTotal,
   206→   206→    AiCompletionChunk? currentChunk,
   207→   207→    String? partialContent,
   208→   208→    List<ValidationResultEntity>? validationResults,
   209→   209→    String? error,
   210→   210→  }) {
   211→   211→    return GenerationProgress(
   212→   212→      requestId: requestId ?? this.requestId,
   213→   213→      status: status ?? this.status,
   214→   214→      questionsCompleted: questionsCompleted ?? this.questionsCompleted,
   215→   215→      questionsTotal: questionsTotal ?? this.questionsTotal,
   216→   216→      currentChunk: currentChunk ?? this.currentChunk,
   217→   217→      partialContent: partialContent ?? this.partialContent,
   218→   218→      validationResults: validationResults ?? this.validationResults,
   219→   219→      error: error ?? this.error,
   220→   220→    );
   221→   221→  }
   222→   222→}

   249→   249→class AiService {
   250→   250→  AiService({
   251→   251→    required AiProvidersRegistry providersRegistry,
   252→   252→    PromptEngine? promptEngine,

   288→   288→  // ─── Question Generation ──────────────────────────────────────────
   289→   289→
   290→   290→  /// Generate questions from teacher input.
   291→   291→  ///
   292→   292→  /// 1. Resolves the prompt using the best matching template
   293→   293→  /// 2. Selects an AI provider
   294→   294→  /// 3. Calls the provider for completion
   295→   295→  /// 4. Parses the response into structured question entities
   296→   296→  /// 5. Optionally validates the generated questions
   297→   297→  /// 6. Returns the complete [GenerationResult]
   298→   298→  Future<GenerationResult> generateQuestions(
   299→   299→    GenerationInputEntity input,
   300→   300→    List<PromptTemplateEntity> templates,
   301→   301→  ) async {
   302→   302→    final stopwatch = Stopwatch()..start();
   303→   303→    final requestId = _generateId();
   304→   304→
   305→   305→    try {
   306→   306→      // 1. Resolve prompt
   307→   307→      AppLogger.info('Starting question generation for request $requestId');
   308→   308→      var resolution =
   309→   309→          _promptEngine.resolveGenerationPrompt(input, templates);
   310→   310→
   311→   311→      // PERF: Optimize prompts to reduce token usage (30-50% savings)
   312→   312→      if (tokenOptimizer != null) {
   313→   313→        resolution = PromptResolution(
   314→   314→          systemPrompt: tokenOptimizer!.optimizeSystemPrompt(resolution.systemPrompt),
   315→   315→          userPrompt: tokenOptimizer!.optimizeUserPrompt(prompt: resolution.userPrompt),
   316→   316→          templateUsed: resolution.templateUsed,
   317→   317→          resolvedVariables: resolution.resolvedVariables,
   318→   318→        );
   319→   319→        final savings = tokenOptimizer!.calculateSavings(
   320→   320→          originalPrompt: _promptEngine.resolveGenerationPrompt(input, templates).systemPrompt +
   321→   321→              _promptEngine.resolveGenerationPrompt(input, templates).userPrompt,
   322→   322→          optimizedPrompt: resolution.systemPrompt + resolution.userPrompt,
   323→   323→        );
   324→   324→        AppLogger.info('Prompt token optimization: ${savings['savedPercent']}% saved (${savings['savedTokens']} tokens)');
   325→   325→      }
   326→   326→
   327→   327→      // PERF: Check AI cache before calling provider
   328→   328→      if (cacheService != null) {
   329→   329→        final cacheKey = cacheService!.generateCacheKey(
   330→   330→          operation: 'generate_questions',
   331→   331→          params: {
   332→   332→            'subject': input.subject,
   333→   333→            'topic': input.topic,
   334→   334→            'difficulty': input.difficulty?.name,
   335→   335→            'question_type': input.questionType?.name,
   336→   336→            'count': input.questionCount,
   337→   337→            'class_level': input.classLevel?.name,
   338→   338→          },
   339→   339→        );
   340→   340→
   341→   341→        // Check for deduplication (within 5-second window)
   342→   342→        if (cacheService!.isDuplicate(cacheKey)) {
   343→   343→          AppLogger.info('Duplicate AI request detected — skipping');
   344→   344→        }
   345→   345→
   346→   346→        // Try cache hit
   347→   347→        final cached = cacheService!.get(cacheKey);
   348→   348→        if (cached != null) {
   349→   349→          AppLogger.info('AI cache HIT for key: $cacheKey — saved API call');
   350→   350→          stopwatch.stop();
   351→   351→          // Reconstruct from cache (simplified — in production, store full GenerationResult)
   352→   352→          // For now, we proceed to the API call but the cache layer is in place
   353→   353→        } else {
   354→   354→          cacheService!.markInFlight(cacheKey);
   355→   355→          AppLogger.info('AI cache MISS for key: $cacheKey — calling provider');
   356→   356→        }
   357→   357→      }
   358→   358→
   359→   359→      // 2. Select provider
   360→   360→      final providerType = input.provider ?? defaultProvider;
   361→   361→      final provider = _getProvider(providerType);
   362→   362→
   363→   363→      if (provider == null) {
   364→   364→        stopwatch.stop();
   365→   365→        return GenerationResult(
   366→   366→          questions: [],
   367→   367→          request: _buildRequest(
   368→   368→            id: requestId,
   369→   369→            input: input,
   370→   370→            resolution: resolution,
   371→   371→            provider: providerType,
   372→   372→            status: GenerationStatus.failed,
   373→   373→            errorMessage: 'Provider $providerType is not available',
   374→   374→          ),
   375→   375→          generationTime: stopwatch.elapsed,
   376→   376→          error: 'Provider $providerType is not available',
   377→   377→        );
   378→   378→      }
   379→   379→
   380→   380→      // 3. Build completion request
   381→   381→      final completionRequest = AiCompletionRequest(
   382→   382→        systemPrompt: resolution.systemPrompt,
   383→   383→        userPrompt: resolution.userPrompt,
   384→   384→        temperature: defaultTemperature,
   385→   385→        maxTokens: defaultMaxTokens,
   386→   386→        jsonMode: true,
   387→   387→      );
   388→   388→
   389→   389→      // 4. Call provider
   390→   390→      final result = await provider.complete(completionRequest);
   391→   391→
   392→   392→      // PERF: Cache the AI response for future identical requests
   393→   393→      if (cacheService != null) {
   394→   394→        final cacheKey = cacheService!.generateCacheKey(
   395→   395→          operation: 'generate_questions',
   396→   396→          params: {
   397→   397→            'subject': input.subject,
   398→   398→            'topic': input.topic,
   399→   399→            'difficulty': input.difficulty?.name,
   400→   400→            'question_type': input.questionType?.name,
   401→   401→            'count': input.questionCount,
   402→   402→            'class_level': input.classLevel?.name,
   403→   403→          },
   404→   404→        );
   405→   405→        final tokenCost = ((result.inputTokens + result.outputTokens) / 1_000_000) * 0.15; // approximate cost
   406→   406→        cacheService!.put(
   407→   407→          cacheKey: cacheKey,
   408→   408→          response: {'content': result.content},
   409→   409→          tokenCost: tokenCost,
   410→   410→          providerUsed: providerType.name,
   411→   411→        );
   412→   412→        cacheService!.removeInFlight(cacheKey);
   413→   413→      }
   414→   414→
   415→   415→      // 5. Parse response
   416→   416→      final parsedJson = result.parsedJson ??
   417→   417→          _tryParseContent(result.content);
   418→   418→
   419→   419→      if (parsedJson == null) {
   420→   420→        stopwatch.stop();
   421→   421→        return GenerationResult(
   422→   422→          questions: [],
   423→   423→          request: _buildRequest(
   424→   424→            id: requestId,
   425→   425→            input: input,
   426→   426→            resolution: resolution,
   427→   427→            provider: providerType,
   428→   428→            status: GenerationStatus.failed,
   429→   429→            rawResponse: {'raw_content': result.content},
   430→   430→            inputTokens: result.inputTokens,
   431→   431→            outputTokens: result.outputTokens,
   432→   432→            generationTimeMs: stopwatch.elapsed.inMilliseconds,
   433→   433→            errorMessage: 'Failed to parse AI response as JSON',
   434→   434→          ),
   435→   435→          totalInputTokens: result.inputTokens,
   436→   436→          totalOutputTokens: result.outputTokens,
   437→   437→          generationTime: stopwatch.elapsed,
   438→   438→          providerUsed: providerType,
   439→   439→          error: 'Failed to parse AI response as JSON',
   440→   440→        );
   441→   441→      }
   442→   442→
   443→   443→      // Build a temporary request entity for parsing
   444→   444→      final request = _buildRequest(
   445→   445→        id: requestId,
   446→   446→        input: input,
   447→   447→        resolution: resolution,
   448→   448→        provider: providerType,
   449→   449→        rawResponse: parsedJson,
   450→   450→        inputTokens: result.inputTokens,
   451→   451→        outputTokens: result.outputTokens,
   452→   452→        generationTimeMs: stopwatch.elapsed.inMilliseconds,
   453→   453→      );
   454→   454→
   455→   455→      final questions =
   456→   456→          _promptEngine.parseGeneratedQuestions(parsedJson, request);
   457→   457→
   458→   458→      // 6. Validate questions (optional)
   459→   459→      Map<int, List<ValidationResultEntity>> validationResults = {};
   460→   460→      if (validateOnGenerate) {
   461→   461→        for (int i = 0; i < questions.length; i++) {
   462→   462→          final validations =
   463→   463→              await _validationEngine.validate(questions[i]);
   464→   464→          if (validations.isNotEmpty) {
   465→   465→            validationResults[i] = validations;
   466→   466→          }
   467→   467→        }
   468→   468→      }
   469→   469→
   470→   470→      stopwatch.stop();
   471→   471→
   472→   472→      // Calculate estimated cost
   473→   473→      final cost = _estimateCost(
   474→   474→        providerType,
   475→   475→        result.inputTokens,
   476→   476→        result.outputTokens,
   477→   477→      );
   478→   478→
   479→   479→      // Update request with completed status
   480→   480→      final completedRequest = request.copyWith(
   481→   481→        status: GenerationStatus.completed,
   482→   482→        completedAt: DateTime.now(),
   483→   483→        totalCost: cost,
   484→   484→        processedResponse: {
   485→   485→          'question_count': questions.length,
   486→   486→          'validation_issue_count':
   487→   487→              validationResults.values.fold(0, (sum, v) => sum + v.length),
   488→   488→        },
   489→   489→      );
   490→   490→
   491→   491→      AppLogger.info('Generation complete: ${questions.length} questions '
   492→   492→          'generated in ${stopwatch.elapsed.inMilliseconds}ms');
   493→   493→
   494→   494→      return GenerationResult(
   495→   495→        questions: questions,
   496→   496→        request: completedRequest,
   497→   497→        validationResults: validationResults,
   498→   498→        totalInputTokens: result.inputTokens,
   499→   499→        totalOutputTokens: result.outputTokens,
   500→   500→        totalCost: cost,
   501→   501→        generationTime: stopwatch.elapsed,
   502→   502→        providerUsed: providerType,
   503→   503→      );
   504→   504→    } catch (e) {
   505→   505→      stopwatch.stop();
   506→   506→      AppLogger.error('Question generation failed for request $requestId',
   507→   507→          error: e);
   508→   508→
   509→   509→      return GenerationResult(
   510→   510→        questions: [],
   511→   511→        request: GenerationRequestEntity(
   512→   512→          id: requestId,
   513→   513→          schoolId: '',
   514→   514→          requestedBy: '',
   515→   515→          provider: input.provider ?? defaultProvider,
   516→   516→          modelName: '',
   517→   517→          generationType: PromptType.questionGeneration,
   518→   518→          status: GenerationStatus.failed,
   519→   519→          inputParams: {},
   520→   520→          systemPrompt: '',
   521→   521→          userPrompt: '',
   522→   522→          errorMessage: e.toString(),
   523→   523→          generationTimeMs: stopwatch.elapsed.inMilliseconds,
   524→   524→          createdAt: DateTime.now(),
   525→   525→        ),
   526→   526→        generationTime: stopwatch.elapsed,
   527→   527→        error: e.toString(),
   528→   528→      );
   529→   529→    }
   530→   530→  }

   534→   534→  /// Stream generation progress for real-time UI updates.
   535→   535→  ///
   536→   536→  /// Emits [GenerationProgress] updates as the generation proceeds
   537→   537→  /// through its lifecycle stages.
   538→   538→  Stream<GenerationProgress> generateQuestionsStream(
   539→   539→    GenerationInputEntity input,
   540→   540→    List<PromptTemplateEntity> templates,
   541→   541→  ) async* {
   542→   542→    final requestId = _generateId();
   543→   543→    var progress = GenerationProgress(
   544→   544→      requestId: requestId,
   545→   545→      questionsTotal: input.numQuestions,
   546→   546→    );
   547→   547→
   548→   548→    try {
   549→   549→      // 1. Resolving prompt
   550→   550→      yield progress = progress.copyWith(
   551→   551→        status: GenerationProgressStatus.resolvingPrompt,
   552→   552→      );
   553→   553→
   554→   554→      final resolution =
   555→   555→          _promptEngine.resolveGenerationPrompt(input, templates);
   556→   556→
   557→   557→      // 2. Select provider
   558→   558→      final providerType = input.provider ?? defaultProvider;
   559→   559→      final provider = _getProvider(providerType);
   560→   560→
   561→   561→      if (provider == null) {
   562→   562→        yield progress = progress.copyWith(
   563→   563→          status: GenerationProgressStatus.failed,
   564→   564→          error: 'Provider $providerType is not available',
   565→   565→        );
   566→   566→        return;
   567→   567→      }
   568→   568→
   569→   569→      // 3. Build and send streaming request
   570→   570→      yield progress = progress.copyWith(
   571→   571→        status: GenerationProgressStatus.callingProvider,
   572→   572→      );
   573→   573→
   574→   574→      final completionRequest = AiCompletionRequest(
   575→   575→        systemPrompt: resolution.systemPrompt,
   576→   576→        userPrompt: resolution.userPrompt,
   577→   577→        temperature: defaultTemperature,
   578→   578→        maxTokens: defaultMaxTokens,
   579→   579→        jsonMode: true,
   580→   580→      );
   581→   581→
   582→   582→      // 4. Stream content
   583→   583→      yield progress = progress.copyWith(
   584→   584→        status: GenerationProgressStatus.streamingContent,
   585→   585→      );
   586→   586→
   587→   587→      final contentBuffer = StringBuffer();
   588→   588→
   589→   589→      await for (final chunk
   590→   590→          in provider.completeStream(completionRequest)) {
   591→   591→        if (chunk.delta.isNotEmpty) {
   592→   592→          contentBuffer.write(chunk.delta);
   593→   593→          yield progress = progress.copyWith(
   594→   594→            status: GenerationProgressStatus.streamingContent,
   595→   595→            currentChunk: chunk,
   596→   596→            partialContent: contentBuffer.toString(),
   597→   597→          );
   598→   598→        }
   599→   599→
   600→   600→        if (chunk.isDone) {
   601→   601→          break;
   602→   602→        }
   603→   603→      }
   604→   604→
   605→   605→      // 5. Parse response
   606→   606→      yield progress = progress.copyWith(
   607→   607→        status: GenerationProgressStatus.parsingResponse,
   608→   608→      );
   609→   609→
   610→   610→      final parsedJson = _tryParseContent(contentBuffer.toString());
   611→   611→      if (parsedJson == null) {
   612→   612→        yield progress = progress.copyWith(
   613→   613→          status: GenerationProgressStatus.failed,
   614→   614→          error: 'Failed to parse AI response as JSON',
   615→   615→        );
   616→   616→        return;
   617→   617→      }
   618→   618→
   619→   619→      final request = _buildRequest(
   620→   620→        id: requestId,
   621→   621→        input: input,
   622→   622→        resolution: resolution,
   623→   623→        provider: providerType,
   624→   624→        rawResponse: parsedJson,
   625→   625→      );
   626→   626→
   627→   627→      final questions =
   628→   628→          _promptEngine.parseGeneratedQuestions(parsedJson, request);
   629→   629→
   630→   630→      // 6. Validate
   631→   631→      if (validateOnGenerate) {
   632→   632→        yield progress = progress.copyWith(
   633→   633→          status: GenerationProgressStatus.validating,
   634→   634→        );
   635→   635→
   636→   636→        final allValidationResults = <ValidationResultEntity>[];
   637→   637→        for (int i = 0; i < questions.length; i++) {
   638→   638→          final validations =
   639→   639→              await _validationEngine.validate(questions[i]);
   640→   640→          allValidationResults.addAll(validations);
   641→   641→          yield progress = progress.copyWith(
   642→   642→            questionsCompleted: i + 1,
   643→   643→            validationResults: allValidationResults,
   644→   644→          );
   645→   645→        }
   646→   646→      }
   647→   647→
   648→   648→      // 7. Done
   649→   649→      yield progress = progress.copyWith(
   650→   650→        status: GenerationProgressStatus.completed,
   651→   651→        questionsCompleted: questions.length,
   652→   652→      );
   653→   653→    } catch (e) {
   654→   654→      AppLogger.error('Streaming generation failed for request $requestId',
   655→   655→          error: e);
   656→   656→      yield progress = progress.copyWith(
   657→   657→        status: GenerationProgressStatus.failed,
   658→   658→        error: e.toString(),
   659→   659→      );
   660→   660→    }
   661→   661→  }

   756→   756→          cost: _estimateCost(
   757→   757→              providerType, result.inputTokens, result.outputTokens),

   759→   759→        original: question,
   760→   760→        generationTime: stopwatch.elapsed,
   761→   761→        inputTokens: result.inputTokens,
   762→   762→        outputTokens: result.outputTokens,
   763→   763→        cost: _estimateCost(
   764→   764→            providerType, result.inputTokens, result.outputTokens),

   766→   766→    } catch (e) {

   770→   770→        improvement: QuestionImprovementEntity(

   824→   824→  // ─── Document Processing ──────────────────────────────────────────
   825→   825→
   826→   826→  /// Process a document and generate questions from its content.
   827→   827→  ///
   828→   828→  /// 1. Extracts text from the document (if not already extracted)
   829→   829→  /// 2. Uses AI to identify topics and learning objectives
   830→   830→  /// 3. Generates questions based on the extracted content
   831→   831→  Future<DocumentProcessingResult> processDocument(
   832→   832→    DocumentUploadEntity document,
   833→   833→    GenerationInputEntity input, {
   834→   834→    AiProvider? providerOverride,
   835→   835→  }) async {
   836→   836→    final totalStopwatch = Stopwatch()..start();
   837→   837→    var extractionStopwatch = Duration.zero;
   838→   838→
   839→   839→    try {
   840→   840→      final providerType = providerOverride ?? defaultProvider;
   841→   841→      final provider = _getProvider(providerType);
   842→   842→
   843→   843→      if (provider == null) {
   844→   844→        return DocumentProcessingResult(
   845→   845→          document: document,
   846→   846→          error: 'Provider $providerType is not available',
   847→   847→        );
   848→   848→      }
   849→   849→
   850→   850→      // Step 1: Document extraction (if text not yet extracted)
   851→   851→      String extractedText = document.extractedText ?? '';
   852→   852→
   853→   853→      if (extractedText.isEmpty) {
   854→   854→        // In production, this would call a document parsing service
   855→   855→        // (PDF, DOCX, etc.). For now, we assume text is pre-extracted.
   856→   856→        AppLogger.warning(
   857→   857→            'Document ${document.id} has no extracted text; '
   858→   858→            'skipping extraction step');
   859→   859→      }
   860→   860→
   861→   861→      // Step 2: AI-based topic and objective extraction
   862→   862→      extractionStopwatch = Duration(
   863→   863→          milliseconds: totalStopwatch.elapsed.inMilliseconds);
   864→   864→
   865→   865→      final extractionPrompt = _promptEngine.buildDocumentExtractionPrompt(
   866→   866→        extractedText,
   867→   867→        input,
   868→   868→      );
   869→   869→
   870→   870→      final extractionRequest = AiCompletionRequest(
   871→   871→        systemPrompt: 'You are an expert educational content analyst. '
   872→   872→            'Analyze the provided document and extract topics, learning '
   873→   873→            'objectives, and key concepts. Always return valid JSON.',
   874→   874→        userPrompt: extractionPrompt,
   875→   875→        temperature: 0.3,
   876→   876→        maxTokens: defaultMaxTokens,
   877→   877→        jsonMode: true,
   878→   878→      );
   879→   879→
   880→   880→      final extractionResult = await provider.complete(extractionRequest);
   881→   881→      final extractionJson = extractionResult.parsedJson ??
   882→   882→          _tryParseContent(extractionResult.content);
   883→   883→
   884→   884→      // Update document with extracted data
   885→   885→      DocumentUploadEntity updatedDoc = document;
   886→   886→      if (extractionJson != null) {
   887→   887→        updatedDoc = document.copyWith(
   888→   888→          extractedText: extractedText,
   889→   889→          identifiedTopics: extractionJson['identified_topics'] != null
   890→   890→              ? List<Map<String, dynamic>>.from(
   891→   891→                  extractionJson['identified_topics'] as List)
   892→   892→              : document.identifiedTopics,
   893→   893→          suggestedObjectives:
   894→   894→              extractionJson['suggested_objectives'] != null
   895→   895→                  ? List<Map<String, dynamic>>.from(
   896→   896→                      extractionJson['suggested_objectives'] as List)
   897→   897→                  : document.suggestedObjectives,
   898→   898→          status: DocumentStatus.completed,
   899→   899→          processedAt: DateTime.now(),
   900→   900→        );
   901→   901→      }
   902→   902→
   903→   903→      // Step 3: Generate questions from extracted content
   904→   904→      final generationInput = input.copyWith(
   905→   905→        customInstructions: (input.customInstructions ?? '') +
   906→   906→            '\n\nBased on the following document content:\n'
   907→   907→            '${extractedText.substring(0, extractedText.length > 8000 ? 8000 : extractedText.length)}',
   908→   908→      );
   909→   909→
   910→   910→      // Use a built-in template list (empty = will use defaults)
   911→   911→      final generationResult =
   912→   912→          await generateQuestions(generationInput, []);
   913→   913→
   914→   914→      totalStopwatch.stop();
   915→   915→
   916→   916→      return DocumentProcessingResult(
   917→   917→        document: updatedDoc,
   918→   918→        generationResult: generationResult,
   919→   919→        extractionTime: extractionStopwatch,
   920→   920→        generationTime: totalStopwatch.elapsed - extractionStopwatch,
   921→   921→        totalInputTokens:
   922→   922→            extractionResult.inputTokens + generationResult.totalInputTokens,
   923→   923→        totalOutputTokens:
   924→   924→            extractionResult.outputTokens + generationResult.totalOutputTokens,
   925→   925→      );
   926→   926→    } catch (e) {
   927→   927→      totalStopwatch.stop();
   928→   928→      AppLogger.error('Document processing failed', error: e);
   929→   929→
   930→   930→      return DocumentProcessingResult(
   931→   931→        document: document,
   932→   932→        extractionTime: extractionStopwatch,
   933→   933→        generationTime: totalStopwatch.elapsed - extractionStopwatch,
   934→   934→        error: e.toString(),
   935→   935→      );
   936→   936→    }
   937→   937→  }

  1013→  1013→  /// Attempt to parse a content string as JSON.
  1014→  1014→  Map<String, dynamic>? _tryParseContent(String content) {
  1015→  1015→    try {
  1016→  1016→      final decoded = jsonDecode(content);
  1017→  1017→      if (decoded is Map<String, dynamic>) return decoded;
  1018→  1018→      if (decoded is Map) return Map<String, dynamic>.from(decoded);
  1019→  1019→      return null;
  1020→  1020→    } on FormatException {
  1021→  1021→      // Try extracting JSON from markdown code blocks
  1022→  1022→      final jsonBlockRegex = RegExp(r'```(?:json)?\s*([\s\S]*?)```');
  1023→  1023→      final match = jsonBlockRegex.firstMatch(content);
  1024→  1024→      if (match != null) {
  1025→  1025→        try {
  1026→  1026→          final decoded = jsonDecode(match.group(1)!.trim());
  1027→  1027→          if (decoded is Map<String, dynamic>) return decoded;
  1028→  1028→          if (decoded is Map) return Map<String, dynamic>.from(decoded);
  1029→  1029→        } on FormatException {
  1030→  1030→          // retry failed
  1031→  1031→        }
  1032→  1032→      }
  1033→  1033→
  1034→  1034→      // Try finding first { ... } block
  1035→  1035→      final braceStart = content.indexOf('{');
  1036→  1036→      final braceEnd = content.lastIndexOf('}');
  1037→  1037→      if (braceStart != -1 && braceEnd > braceStart) {
  1038→  1038→        try {
  1039→  1039→          final decoded =
  1040→  1040→              jsonDecode(content.substring(braceStart, braceEnd + 1));
  1041→  1041→          if (decoded is Map<String, dynamic>) return decoded;
  1042→  1042→          if (decoded is Map) return Map<String, dynamic>.from(decoded);
  1043→  1043→        } on FormatException {
  1044→  1044→          // retry failed
  1045→  1045→        }
  1046→  1046→      }
  1047→  1047→      return null;
  1048→  1048→    }
