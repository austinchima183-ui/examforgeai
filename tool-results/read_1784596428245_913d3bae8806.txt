    25→    25→
    26→    26→/// The main Billing Dashboard page.
    27→    27→///
    28→    28→/// This is the entry point for the billing feature, presenting a modern
    29→    29→/// SaaS-style billing dashboard with:
    30→    30→/// - Current subscription overview (plan name, status, period end)
    31→    31→/// - AI credit balance card
    32→    32→/// - Quick actions (Upgrade, Manage, View Invoices, Buy Credits)
    33→    33→/// - Recent transactions list
    34→    34→/// - Billing alerts/notifications
    35→    35→///
    36→    36→/// Uses [ConsumerStatefulWidget] with a private [_State] class pattern,
    37→    37→/// loads data in [initState] via [WidgetsBinding.instance.addPostFrameCallback],
    38→    38→/// and renders a responsive layout using [SingleChildScrollView] with
    39→    39→/// vertically arranged Card sections.
    40→    40→class BillingDashboardPage extends ConsumerStatefulWidget {
    41→    41→  const BillingDashboardPage({super.key});
    42→    42→
    43→    43→  @override
    44→    44→  ConsumerState<BillingDashboardPage> createState() => _State();
    45→    45→}
    46→    46→
    47→    47→class _State extends ConsumerState<BillingDashboardPage> {
    48→    48→  // ─── Lifecycle ──────────────────────────────────────────────────────
    49→    49→
    50→    50→  @override
    51→    51→  void initState() {
    52→    52→    super.initState();
    53→    53→    WidgetsBinding.instance.addPostFrameCallback((_) {
    54→    54→      _loadData();
    55→    55→    });
    56→    56→  }
    57→    57→
    58→    58→  /// Loads all billing data in parallel.
    59→    59→  void _loadData() {
    60→    60→    final userId = ref.read(userIdProvider);
    61→    61→    if (userId == null) return;
    62→    62→
    63→    63→    final subscriberType = BillingModel.teacherSaas;
    64→    64→
    65→    65→    ref.read(subscriptionProvider.notifier).loadCurrentSubscription(
    66→    66→          subscriberId: userId,
    67→    67→          subscriberType: subscriberType,
    68→    68→        );
    69→    69→
    70→    70→    ref.read(aiCreditsProvider.notifier).loadCreditBalance(
    71→    71→          ownerId: userId,
    72→    72→          ownerType: subscriberType,
    73→    73→        );
    74→    74→
    75→    75→    ref.read(paymentProvider.notifier).loadTransactions(
    76→    76→          userId: userId,
    77→    77→          page: 1,
    78→    78→          perPage: 5,
    79→    79→        );
    80→    80→
    81→    81→    ref.read(billingNotificationProvider.notifier).loadNotifications(
    82→    82→          userId: userId,
    83→    83→          unreadOnly: false,
    84→    84→          page: 1,
    85→    85→          perPage: 10,
    86→    86→        );
    87→    87→  }
    88→    88→
    89→    89→  // ─── Build ──────────────────────────────────────────────────────────
    90→    90→
    91→    91→  @override
    92→    92→  Widget build(BuildContext context) {
    93→    93→    final subscriptionState = ref.watch(subscriptionProvider);
    94→    94→    final creditsState = ref.watch(aiCreditsProvider);
    95→    95→    final paymentState = ref.watch(paymentProvider);
    96→    96→    final notificationState = ref.watch(billingNotificationProvider);
    97→    97→
    98→    98→    final isLoading = subscriptionState.isLoading ||
    99→    99→        creditsState.isLoading ||
   100→   100→        (paymentState.isLoading && paymentState.transactions.isEmpty) ||
   101→   101→        (notificationState.isLoading && notificationState.notifications.isEmpty);
   102→   102→
   103→   103→    final hasError = subscriptionState.error != null &&
   104→   104→        subscriptionState.subscription == null;
   105→   105→
   106→   106→    return Scaffold(
   107→   107→      appBar: AppAppBar(
   108→   108→        title: 'Billing',
   109→   109→        actions: [
   110→   110→          if (notificationState.hasUnread)
   111→   111→            IconButton(
   112→   112→              icon: Badge(
   113→   113→                label: Text('${notificationState.unreadCount}'),
   114→   114→                child: const Icon(Icons.notifications_outlined),
   115→   115→              ),
   116→   116→              onPressed: () => context.go(RouteNames.billingNotifications),
   117→   117→              tooltip: 'Notifications',
   118→   118→            )
   119→   119→          else
   120→   120→            IconButton(
   121→   121→              icon: const Icon(Icons.notifications_outlined),
   122→   122→              onPressed: () => context.go(RouteNames.billingNotifications),
   123→   123→              tooltip: 'Notifications',
   124→   124→            ),
   125→   125→        ],
   126→   126→      ),
   127→   127→      body: _buildBody(
   128→   128→        context,
   129→   129→        isLoading: isLoading,
   130→   130→        hasError: hasError,
   131→   131→        subscriptionState: subscriptionState,
   132→   132→        creditsState: creditsState,
   133→   133→        paymentState: paymentState,
   134→   134→        notificationState: notificationState,
   135→   135→      ),
   136→   136→    );
   137→   137→  }
   138→   138→
   139→   139→  // ─── Body Router ────────────────────────────────────────────────────
   140→   140→
   141→   141→  Widget _buildBody(
   142→   142→    BuildContext context, {
   143→   143→    required bool isLoading,
   144→   144→    required bool hasError,
   145→   145→    required SubscriptionState subscriptionState,
   146→   146→    required AiCreditsState creditsState,
   147→   147→    required PaymentState paymentState,
   148→   148→    required BillingNotificationState notificationState,
   149→   149→  }) {
   150→   150→    // Loading state — initial load with no data
   151→   151→    if (isLoading &&
   152→   152→        subscriptionState.subscription == null &&
   153→   153→        creditsState.creditBalance == null &&
   154→   154→        paymentState.transactions.isEmpty) {
   155→   155→      return const Center(child: AppLoadingSpinner());
   156→   156→    }
   157→   157→
   158→   158→    // Error state — no subscription data available
   159→   159→    if (hasError) {
   160→   160→      return AppErrorState.genericError(
   161→   161→        message: subscriptionState.error,
   162→   162→        onRetry: _loadData,
   163→   163→      );
   164→   164→    }
   165→   165→
   166→   166→    // Success — render dashboard
   167→   167→    return RefreshIndicator(
   168→   168→      onRefresh: () async => _loadData(),
   169→   169→      child: SingleChildScrollView(
   170→   170→        physics: const AlwaysScrollableScrollPhysics(),
   171→   171→        padding: const EdgeInsets.symmetric(
   172→   172→          horizontal: Spacings.lg,
   173→   173→          vertical: Spacings.md,
   174→   174→        ),
   175→   175→        child: Column(
   176→   176→          crossAxisAlignment: CrossAxisAlignment.start,
   177→   177→          children: [
   178→   178→            // ─── Subscription Overview ────────────────────────────
   179→   179→            _buildSubscriptionOverview(context, subscriptionState),
   180→   180→
   181→   181→            const SizedBox(height: Spacings.lg),
   182→   182→
   183→   183→            // ─── AI Credit Balance Card ───────────────────────────
   184→   184→            _buildCreditBalanceCard(context, creditsState),
   185→   185→
   186→   186→            const SizedBox(height: Spacings.lg),
   187→   187→
   188→   188→            // ─── Quick Actions ────────────────────────────────────
   189→   189→            _buildQuickActions(context),
   190→   190→
   191→   191→            const SizedBox(height: Spacings.lg),
   192→   192→
   193→   193→            // ─── Billing Alerts / Notifications ───────────────────
   194→   194→            _buildBillingAlerts(context, notificationState),
   195→   195→
   196→   196→            const SizedBox(height: Spacings.lg),
   197→   197→
   198→   198→            // ─── Recent Transactions ──────────────────────────────
   199→   199→            _buildRecentTransactions(context, paymentState),
   200→   200→
   201→   201→            // Bottom padding
   202→   202→            const SizedBox(height: Spacings.xxl),
   203→   203→          ],
   204→   204→        ),
   205→   205→      ),
   206→   206→    );
   207→   207→  }

  1203→  1203→  /// Formats a [DateTime] for display in the transaction list.
  1204→  1204→  String _formatTransactionDate(DateTime date) {
  1205→  1205→    final now = DateTime.now();
  1206→  1206→    final diff = now.difference(date);
  1207→  1207→
  1208→  1208→    if (diff.inDays == 0) {
  1209→  1209→      if (diff.inHours == 0) {
  1210→  1210→        if (diff.inMinutes == 0) return 'Just now';
  1211→  1211→        return '${diff.inMinutes}m ago';
  1212→  1212→      }
  1213→  1213→      return '${diff.inHours}h ago';
  1214→  1214→    } else if (diff.inDays == 1) {
  1215→  1215→      return 'Yesterday';

  1218→  1218→    } else {
  1219→  1219→      const months = [
  1220→  1220→        '',

  1231→  1231→        'Nov',
  1232→  1232→        'Dec',
  1233→  1233→      ];
  1234→  1234→      return '${months[date.month]} ${date.day}, ${date.year}';
  1235→  1235→    }
  1236→  1236→  }
