    41→  Widget build(BuildContext context) {
    42→    final practiceState = ref.watch(practiceProvider);
    43→
    44→    // Determine which screen to show
    45→    if (practiceState.currentSession != null &&
    46→        practiceState.currentSession!.status ==
    47→            PracticeSessionStatus.completed) {
    48→      return _buildResultsScreen(context, practiceState);
    49→    }
    50→
    51→    if (practiceState.currentSession != null &&
    52→        practiceState.currentSession!.status ==
    53→            PracticeSessionStatus.inProgress) {
    54→      return _buildSessionScreen(context, practiceState);
    55→    }
    56→
    57→    return _buildSetupScreen(context, practiceState);
    58→  }

   658→  void _startSession() {
   659→    ref.read(practiceProvider.notifier).createSession(
   660→      subjectId: _selectedSubject,
   661→      topicId: _selectedTopic,
   662→      difficulty: _selectedDifficulty,
   663→      mode: _selectedMode,
   664→      timeLimitSec: _selectedMode == PracticeMode.timed
   665→          ? _timeLimitMin * 60
   666→          : null,
   667→      questionCount: _questionCount.toInt(),
   668→    );
   669→  }

   700→  String _formatDuration(Duration duration) {
   701→    final hours = duration.inHours;
   702→    final minutes = duration.inMinutes.remainder(60);
   703→    final seconds = duration.inSeconds.remainder(60);
   704→    if (hours > 0) {
   705→      return '${hours}h ${minutes}m';
   706→    }
   707→    return '${minutes.toString().padLeft(2, '0')}:${seconds.toString().padLeft(2, '0')}';
   708→  }
   709→}
