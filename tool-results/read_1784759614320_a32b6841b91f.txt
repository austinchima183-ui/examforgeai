     1→// =============================================================================
     2→// ExamForge AI — Offline Exam Mode Page
     3→// =============================================================================
     4→//
     5→// Full exam-taking interface that works entirely offline. When the user
     6→// submits, the attempt is saved locally and queued for sync. If the device
     7→// is online, the attempt is synced immediately; otherwise it is persisted
     8→// to local storage and will be synced when connectivity is restored.
     9→//
    10→// Features:
    11→//   - Question display with question number, text, options / images
    12→//   - Answer selection (multiple choice, true/false, short answer)
    13→//   - Navigation: Previous / Next / Jump-to-question sidebar
    14→//   - Local countdown timer from exam duration
    15→//   - Auto-save every 30 seconds to local storage
    16→//   - Submit button (queues for sync if offline)
    17→//   - Progress indicator (answered / total)
    18→//   - Offline warning banner
    19→//   - Integrity tracking: hash of answers + timestamps
    20→//   - Error handling: network failure, session timeout, data corruption
    21→// =============================================================================

    46→/// A full-screen exam-taking page that works offline.
    47→///
    48→/// Requires [examId] and [examTitle]. The exam data (questions, options,
    49→/// duration) must have been previously downloaded for offline access and
    50→/// stored in local storage via [CacheManager].
    51→class OfflineExamPage extends ConsumerStatefulWidget {
    52→  const OfflineExamPage({
    53→    super.key,
    54→    required this.examId,
    55→    required this.examTitle,
    56→  });
    57→
    58→  final String examId;
    59→  final String examTitle;
    60→
    61→  @override
    62→  ConsumerState<OfflineExamPage> createState() => _OfflineExamPageState();
    63→}

    65→// ═══════════════════════════════════════════════════════════════════════════════
    66→// STATE
    67→// ═══════════════════════════════════════════════════════════════════════════════
    68→
    69→class _OfflineExamPageState extends ConsumerState<OfflineExamPage>
    70→    with WidgetsBindingObserver {
    71→  // ─── Controllers / State ──────────────────────────────────────────────
    72→
    73→  int _currentQuestionIndex = 0;
    74→  Map<int, Map<String, dynamic>> _answers = {};
    75→  Map<int, bool> _flaggedQuestions = {};
    76→  Duration _timeRemaining = Duration.zero;
    77→  Timer? _timer;
    78→  Timer? _autoSaveTimer;
    79→  bool _isSubmitting = false;
    80→  bool _isSubmitted = false;
    81→  bool _isOnline = true;
    82→  String? _errorMessage;
    83→  String? _successMessage;
    84→  DateTime? _examStartedAt;
    85→
    86→  // ─── Offline Exam Data (loaded from local storage) ────────────────────
    87→
    88→  List<Map<String, dynamic>> _questions = [];
    89→  int _examDurationMinutes = 0;
    90→  String? _studentId;
    91→  String? _schoolId;
    92→  bool _isLoading = true;
    93→
    94→  // ─── Integrity Tracking ──────────────────────────────────────────────
    95→
    96→  final List<Map<String, dynamic>> _integrityLog = [];
    97→
    98→  @override
    99→  void initState() {
   100→    super.initState();
   101→    WidgetsBinding.instance.addObserver(this);
   102→    _examStartedAt = DateTime.now();
   103→    _loadOfflineExamData();
   104→  }
   105→
   106→  @override
   107→  void dispose() {
   108→    WidgetsBinding.instance.removeObserver(this);
   109→    _timer?.cancel();
   110→    _autoSaveTimer?.cancel();
   111→    super.dispose();
   112→  }
   113→
   114→  @override
   115→  void didChangeAppLifecycleState(AppLifecycleState state) {
   116→    // Log integrity event when app goes to background during exam.
   117→    if (state == AppLifecycleState.paused && !_isSubmitted) {
   118→      _logIntegrityEvent('app_backgrounded');
   119→    } else if (state == AppLifecycleState.resumed && !_isSubmitted) {
   120→      _logIntegrityEvent('app_foregrounded');
   121→    }
   122→  }

   124→  // ═══════════════════════════════════════════════════════════════════════
   125→  // DATA LOADING
   126→  // ═══════════════════════════════════════════════════════════════════════
   127→
   128→  /// Loads the exam data from local cache that was previously downloaded
   129→  /// for offline access.
   130→  Future<void> _loadOfflineExamData() async {
   131→    try {
   132→      final cacheManager = ref.read(cacheManagerProvider);
   133→      final userId = cacheManager.storageService.getUserId();
   134→
   135→      // Try to load the cached exam data.
   136→      final cachedData = await cacheManager.getCachedData(
   137→        key: 'offline_exam_${widget.examId}',
   138→        userId: userId ?? '',
   139→      );
   140→
   141→      if (cachedData == null) {
   142→        if (mounted) {
   143→          setState(() {
   144→            _isLoading = false;
   145→            _errorMessage = 'Exam data not found. Please download this exam '
   146→                'for offline use while connected.';
   147→          });
   148→        }
   149→        return;
   150→      }
   151→
   152→      final examData = cachedData;
   153→
   154→      setState(() {
   155→        _questions = List<Map<String, dynamic>>.from(
   156→          (examData['questions'] as List?)?.cast<Map<String, dynamic>>() ?? [],
   157→        );
   158→        _examDurationMinutes = examData['duration_minutes'] as int? ?? 60;
   159→        _studentId = examData['student_id'] as String? ?? userId ?? '';
   160→        _schoolId = examData['school_id'] as String? ?? '';
   161→        _timeRemaining = Duration(minutes: _examDurationMinutes);
   162→        _isLoading = false;
   163→      });
   164→
   165→      // Check connectivity.
   166→      _checkConnectivity();
   167→
   168→      // Start the countdown timer.
   169→      _startTimer();
   170→
   171→      // Start the auto-save timer (every 30 seconds).
   172→      _startAutoSaveTimer();
   173→
   174→      _logIntegrityEvent('exam_started');
   175→    } catch (e) {
   176→      AppLogger.error('[OfflineExam] Failed to load exam data', error: e);
   177→      if (mounted) {
   178→        setState(() {
   179→          _isLoading = false;
   180→          _errorMessage = 'Failed to load exam data. The cached data may '
   181→              'be corrupted. Please reconnect and try again.';
   182→        });
   183→      }
   184→    }
   185→  }
   186→
   187→  /// Checks current connectivity status.
   188→  Future<void> _checkConnectivity() async {
   189→    try {
   190→      final networkInfo = ref.read(coreNetworkInfoProvider);
   191→      _isOnline = await networkInfo.isConnected;
   192→    } catch (_) {
   193→      _isOnline = false;
   194→    }
   195→    if (mounted) setState(() {});
   196→  }
   197→
   198→  // ═══════════════════════════════════════════════════════════════════════
   199→  // TIMER
   200→  // ═══════════════════════════════════════════════════════════════════════
   201→
   202→  void _startTimer() {
   203→    _timer = Timer.periodic(const Duration(seconds: 1), (timer) {
   204→      if (_timeRemaining.inSeconds <= 0) {
   205→        timer.cancel();
   206→        _handleTimeUp();
   207→        return;
   208→      }
   209→      if (mounted) {
   210→        setState(() {
   211→          _timeRemaining = _timeRemaining - const Duration(seconds: 1);
   212→        });
   213→      }
   214→    });
   215→  }
   216→
   217→  void _handleTimeUp() {
   218→    _logIntegrityEvent('time_up_auto_submit');
   219→    _submitExam();
   220→  }
   221→
   222→  // ═══════════════════════════════════════════════════════════════════════
   223→  // AUTO-SAVE
   224→  // ═══════════════════════════════════════════════════════════════════════
   225→
   226→  void _startAutoSaveTimer() {
   227→    _autoSaveTimer = Timer.periodic(const Duration(seconds: 30), (_) {
   228→      _autoSave();
   229→    });
   230→  }
   231→
   232→  /// Saves the current exam state to local storage.
   233→  Future<void> _autoSave() async {
   234→    if (_isSubmitted) return;
   235→
   236→    try {
   237→      final cacheManager = ref.read(cacheManagerProvider);
   238→      final userId = _studentId ?? '';
   239→
   240→      final saveData = <String, dynamic>{
   241→        'exam_id': widget.examId,
   242→        'current_question_index': _currentQuestionIndex,
   243→        'answers': _answers,
   244→        'flagged_questions': _flaggedQuestions,
   245→        'time_remaining_seconds': _timeRemaining.inSeconds,
   246→        'last_saved_at': DateTime.now().toIso8601String(),
   247→      };
   248→
   249→      await cacheManager.cacheData(
   250→        key: 'offline_exam_progress_${widget.examId}',
   251→        userId: userId,
   252→        resourceType: 'exam_progress',
   253→        resourceId: widget.examId,
   254→        data: saveData,
   255→      );
   256→
   257→      AppLogger.debug('[OfflineExam] Auto-saved progress');
   258→    } catch (e) {
   259→      AppLogger.warning('[OfflineExam] Auto-save failed: $e');
   260→    }
   261→  }
   262→
   263→  // ═══════════════════════════════════════════════════════════════════════
   264→  // INTEGRITY TRACKING
   265→  // ═══════════════════════════════════════════════════════════════════════
   266→
   267→  void _logIntegrityEvent(String eventType) {
   268→    _integrityLog.add({
   269→      'event': eventType,
   270→      'timestamp': DateTime.now().toIso8601String(),
   271→      'current_question': _currentQuestionIndex,
   272→      'answers_count': _answers.length,
   273→    });
   274→  }
   275→
   276→  /// Computes an integrity hash of the answers and timestamps.
   277→  String _computeIntegrityHash() {
   278→    final payload = jsonEncode({
   279→      'exam_id': widget.examId,
   280→      'student_id': _studentId,
   281→      'answers': _answers,
   282→      'integrity_log': _integrityLog,
   283→      'started_at': _examStartedAt?.toIso8601String(),
   284→      'submitted_at': DateTime.now().toIso8601String(),
   285→    });
   286→    return sha256.convert(utf8.encode(payload)).toString();
   287→  }
   288→
   289→  // ═══════════════════════════════════════════════════════════════════════
   290→  // ANSWER HANDLING
   291→  // ═══════════════════════════════════════════════════════════════════════
   292→
   293→  void _selectAnswer(int questionIndex, Map<String, dynamic> answer) {
   294→    if (_isSubmitted) return;
   295→    setState(() {
   296→      _answers[questionIndex] = {
   297→        ...answer,
   298→        'answered_at': DateTime.now().toIso8601String(),
   299→      };
   300→    });
   301→  }
   302→
   303→  void _toggleFlag(int questionIndex) {
   304→    if (_isSubmitted) return;
   305→    setState(() {
   306→      _flaggedQuestions[questionIndex] =
   307→          !(_flaggedQuestions[questionIndex] ?? false);
   308→    });
   309→  }
   310→
   311→  // ═══════════════════════════════════════════════════════════════════════
   312→  // NAVIGATION
   313→  // ═══════════════════════════════════════════════════════════════════════
   314→
   315→  void _goToQuestion(int index) {
   316→    if (index < 0 || index >= _questions.length) return;
   317→    setState(() {
   318→      _currentQuestionIndex = index;
   319→    });
   320→  }
   321→
   322→  void _goNext() => _goToQuestion(_currentQuestionIndex + 1);
   323→  void _goPrevious() => _goToQuestion(_currentQuestionIndex - 1);
   324→
   325→  // ═══════════════════════════════════════════════════════════════════════
   326→  // SUBMIT EXAM
   327→  // ═══════════════════════════════════════════════════════════════════════
   328→
   329→  Future<void> _submitExam() async {
   330→    if (_isSubmitting || _isSubmitted) return;
   331→
   332→    // Confirm submission (unless auto-submitted due to time-up).
   333→    final shouldSubmit = await _showSubmitConfirmation();
   334→    if (!shouldSubmit) return;
   335→
   336→    setState(() {
   337→      _isSubmitting = true;
   338→      _errorMessage = null;
   339→    });
   340→
   341→    _logIntegrityEvent('exam_submitted');
   342→    _timer?.cancel();
   343→    _autoSaveTimer?.cancel();
   344→
   345→    final integrityHash = _computeIntegrityHash();
   346→
   347→    final attemptData = <String, dynamic>{
   348→      'exam_id': widget.examId,
   349→      'student_id': _studentId,
   350→      'school_id': _schoolId,
   351→      'started_at': _examStartedAt?.toIso8601String(),
   352→      'completed_at': DateTime.now().toIso8601String(),
   353→      'time_taken_seconds': _examDurationMinutes * 60 - _timeRemaining.inSeconds,
   354→      'integrity_hash': integrityHash,
   355→      'integrity_log': _integrityLog,
   356→    };
   357→
   358→    final answersData = <String, dynamic>{};
   359→    _answers.forEach((index, answer) {
   360→      final question = index < _questions.length ? _questions[index] : null;
   361→      final questionId = question?['id'] as String? ?? 'q_$index';
   362→      answersData[questionId] = answer;
   363→    });
   364→
   365→    final attempt = OfflineExamAttempt(
   366→      id: '', // Will be generated by CacheManager
   367→      examId: widget.examId,
   368→      studentId: _studentId ?? '',
   369→      schoolId: _schoolId ?? '',
   370→      attemptData: attemptData,
   371→      answers: answersData,
   372→      startedAt: _examStartedAt ?? DateTime.now(),
   373→      completedAt: DateTime.now(),
   374→      timeTakenSeconds:
   375→          _examDurationMinutes * 60 - _timeRemaining.inSeconds,
   376→      integrityHash: integrityHash,
   377→      syncStatus: AttemptSyncStatus.pending,
   378→      syncAttempts: 0,
   379→      createdAt: DateTime.now(),
   380→    );
   381→
   382→    try {
   383→      // Save locally first (always).
   384→      final cacheManager = ref.read(cacheManagerProvider);
   385→      await cacheManager.saveOfflineExamAttempt(
   386→        examId: attempt.examId,
   387→        studentId: attempt.studentId,
   388→        schoolId: {'id': attempt.schoolId},
   389→        attemptData: attempt.attemptData,
   390→        answers: attempt.answers,
   391→        startedAt: attempt.startedAt,
   392→        completedAt: attempt.completedAt,
   393→        timeTakenSeconds: attempt.timeTakenSeconds ?? 0,
   394→        integrityHash: attempt.integrityHash,
   395→      );
   396→
   397→      // Try to sync if online.
   398→      await _checkConnectivity();
   399→
   400→      if (_isOnline) {
   401→        // Attempt immediate sync.
   402→        try {
   403→          final syncResult = await ref
   404→              .read(offlineProvider.notifier)
   405→              .loadSyncStatus(_studentId ?? '');
   406→
   407→          // If sync engine is available, trigger it.
   408→          // Otherwise, the attempt stays in pending queue and will
   409→          // be synced when the SyncEngine processes it.
   410→          setState(() {
   411→            _isSubmitting = false;
   412→            _isSubmitted = true;
   413→            _successMessage = 'Exam submitted successfully! Your answers '
   414→                'have been synced to the server.';
   415→          });
   416→
   417→          AppLogger.info('[OfflineExam] Exam submitted and synced');
   418→        } catch (e) {
   419→          // Network failure during submit — save locally, queue for sync.
   420→          setState(() {
   421→            _isSubmitting = false;
   422→            _isSubmitted = true;
   423→            _successMessage = 'Exam saved locally. Your answers will be '
   424→                'synced when you reconnect.';
   425→          });
   426→
   427→          AppLogger.warning('[OfflineExam] Sync failed, saved locally: $e');
   428→        }
   429→      } else {
   430→        // Offline — mark as pending sync.
   431→        setState(() {
   432→          _isSubmitting = false;
   433→          _isSubmitted = true;
   434→          _successMessage = 'You are offline. Your answers have been saved '
   435→              'locally and will be synced when you reconnect.';
   436→        });
   437→
   438→        AppLogger.info('[OfflineExam] Exam saved locally (offline)');
   439→      }
   440→    } catch (e) {
   441→      AppLogger.error('[OfflineExam] Failed to save exam attempt', error: e);
   442→
   443→      setState(() {
   444→        _isSubmitting = false;
   445→        _errorMessage = 'Failed to save your exam. Please try again. '
   446→            'If the problem persists, your progress has been auto-saved '
   447→            'and can be recovered.';
   448→      });
   449→    }
   450→  }

   452→  /// Shows a confirmation dialog before submitting.
   453→  Future<bool> _showSubmitConfirmation() async {
   454→    final unanswered = _questions.length - _answers.length;
   455→    final flagged = _flaggedQuestions.values.where((f) => f).length;
   456→
   457→    final result = await showDialog<bool>(
   458→      context: context,
   459→      builder: (ctx) => AppDialog(
   460→        title: 'Submit Exam?',
   461→        content: Column(
   462→          mainAxisSize: MainAxisSize.min,
   463→          crossAxisAlignment: CrossAxisAlignment.start,
   464→          children: [
   465→            Text(
   466→              'You are about to submit "${widget.examTitle}".',
   467→              style: ctx.textTheme.bodyMedium,
   468→            ),
   469→            Spacings.itemGap,
   470→            Text(
   471→              'Answered: ${_answers.length} / ${_questions.length}',
   472→              style: ctx.textTheme.bodyMedium,
   473→            ),
   474→            if (unanswered > 0) ...[
   475→              Spacings.smallGap,
   476→              Text(
   477→                '⚠ $unanswered question${unanswered > 1 ? 's' : ''} unanswered',
   478→                style: ctx.textTheme.bodyMedium?.copyWith(
   479→                  color: AppColors.warning,
   480→                ),
   481→              ),
   482→            ],
   483→            if (flagged > 0) ...[
   484→              Spacings.smallGap,
   485→              Text(
   486→              '🚩 $flagged question${flagged > 1 ? 's' : ''} flagged for review',
   487→                style: ctx.textTheme.bodyMedium?.copyWith(
   488→                  color: AppColors.info,
   489→                ),
   490→              ),
   491→            ],
   492→            Spacings.itemGap,
   493→            Text(
   494→              'This action cannot be undone.',
   495→              style: ctx.textTheme.bodySmall?.copyWith(
   496→                color: ctx.colorScheme.error,
   497→              ),
   498→            ),
   499→          ],
   500→        ),
   501→        actions: [
   502→          AppButton(
   503→            label: 'Cancel',
   504→            variant: AppButtonVariant.outlined,
   505→            onPressed: () => Navigator.of(ctx).pop(false),
   506→          ),
   507→          AppButton(
   508→            label: 'Submit',
   509→            variant: AppButtonVariant.elevated,
   510→            onPressed: () => Navigator.of(ctx).pop(true),
   511→          ),
   512→        ],
   513→      ),
   514→    );
   515→
   516→    return result ?? false;
   517→  }

   523→  @override
   524→  Widget build(BuildContext context) {
   525→    final cs = context.colorScheme;
   526→    final tt = context.textTheme;
   527→
   528→    // ─── Loading State ────────────────────────────────────────────────

   530→      return Scaffold(
   531→        backgroundColor: cs.surface,
   532→        appBar: AppBar(title: Text(widget.examTitle)),

   537→    // ─── Error State (no exam data) ──────────────────────────────────

   540→        backgroundColor: cs.surface,

   546→              mainAxisSize: MainAxisSize.min,
   547→              children: [
   548→                Icon(Icons.cloud_off, size: 64, color: cs.error),
   549→                Spacings.sectionGap,
   550→                Text(
   551→                  _errorMessage ?? 'Exam data not available offline.',
   552→                  style: tt.titleMedium?.copyWith(color: cs.error),
   553→                  textAlign: TextAlign.center,
   554→                ),
   555→                Spacings.itemGap,
   556→                AppButton(
   557→                  label: 'Go Back',
   558→                  variant: AppButtonVariant.outlined,

   578→    final isFlagged = _flaggedQuestions[_currentQuestionIndex] ?? false;
   579→
   580→    return PopScope(

   582→      onPopInvokedWithResult: (didPop, _) {

   586→      },
   587→      child: Scaffold(
   588→        backgroundColor: cs.surface,
   589→        appBar: _buildAppBar(context, cs, tt),
   590→        body: Column(
   591→          children: [
   592→            // ─── Offline Warning Banner ─────────────────────────────
   593→            if (!_isOnline) _buildOfflineBanner(context, cs, tt),
   594→
   595→            // ─── Progress Bar ───────────────────────────────────────
   596→            _buildProgressBar(context, cs, progress, answeredCount, totalCount),
   597→
   598→            // ─── Question Content ───────────────────────────────────
   599→            Expanded(
   600→              child: SingleChildScrollView(
   601→                padding: Spacings.paddingScreen,
   602→                child: Column(
   603→                  crossAxisAlignment: CrossAxisAlignment.start,
   604→                  children: [

   839→    TextTheme tt,

   841→  ) {
