    21→/// The main Teacher Workspace Dashboard page.
    22→///
    23→/// This is the entry point for the workspace module, presenting a modern
    24→/// SaaS-style dashboard with:
    25→/// - Welcome section with teacher name and current date
    26→/// - Stats cards row (Lesson Plans, Worksheets, Assignments, Resources)
    27→/// - Quick AI Actions grid (8 action buttons)
    28→/// - Today's Schedule timeline
    29→/// - Upcoming Events
    30→/// - Recently Generated AI Content
    31→/// - Draft Lesson Plans with "Generate Questions" buttons
    32→///
    33→/// Uses [ConsumerStatefulWidget] with a private [_State] class pattern,
    34→/// loads data in [initState] via [WidgetsBinding.instance.addPostFrameCallback],
    35→/// and renders a responsive layout using [CustomScrollView] with slivers.
    36→class TeacherWorkspaceDashboardPage extends ConsumerStatefulWidget {
    37→  const TeacherWorkspaceDashboardPage({super.key});
    38→
    39→  @override
    40→  ConsumerState<TeacherWorkspaceDashboardPage> createState() => _State();
    41→}
    42→
    43→class _State extends ConsumerState<TeacherWorkspaceDashboardPage> {
    44→  // ─── Lifecycle ──────────────────────────────────────────────────────
    45→
    46→  @override
    47→  void initState() {
    48→    super.initState();
    49→    WidgetsBinding.instance.addPostFrameCallback((_) {
    50→      ref.read(workspaceDashboardProvider.notifier).loadDashboard();
    51→    });
    52→  }
    53→
    54→  // ─── Build ──────────────────────────────────────────────────────────
    55→
    56→  @override
    57→  Widget build(BuildContext context) {
    58→    final dashboardState = ref.watch(workspaceDashboardProvider);
    59→
    60→    return Scaffold(
    61→      appBar: const AppAppBar(title: 'Workspace'),
    62→      body: _buildBody(context, dashboardState),
    63→    );
    64→  }
    65→
    66→  // ─── Body Router ────────────────────────────────────────────────────
    67→
    68→  Widget _buildBody(BuildContext context, WorkspaceDashboardState state) {
    69→    // Loading state
    70→    if (state.isLoading && state.dashboardSummary == null) {
    71→      return const Center(child: AppLoadingSpinner());
    72→    }
    73→
    74→    // Error state
    75→    if (state.error != null && state.dashboardSummary == null) {
    76→      return AppErrorState.genericError(
    77→        message: state.error,
    78→        onRetry: () =>
    79→            ref.read(workspaceDashboardProvider.notifier).loadDashboard(),
    80→      );
    81→    }
    82→
    83→    // Empty state
    84→    final summary = state.dashboardSummary;
    85→    if (summary == null) {
    86→      return AppEmptyState.noData(
    87→        title: 'No Workspace Data',
    88→        subtitle: 'Your workspace dashboard will appear here once loaded.',
    89→        actionLabel: 'Refresh',
    90→        onAction: () =>
    91→            ref.read(workspaceDashboardProvider.notifier).loadDashboard(),
    92→      );
    93→    }
    94→
    95→    // Success — render dashboard
    96→    return RefreshIndicator(
    97→      onRefresh: () =>
    98→          ref.read(workspaceDashboardProvider.notifier).refresh(),
    99→      child: CustomScrollView(
   100→        physics: const AlwaysScrollableScrollPhysics(),
   101→        slivers: [
   102→          // ─── Welcome Section ──────────────────────────────────────
   103→          SliverToBoxAdapter(child: _buildWelcomeSection(context)),
   104→
   105→          // ─── Stats Cards ──────────────────────────────────────────
   106→          SliverToBoxAdapter(child: _buildStatsCards(context, summary)),
   107→
   108→          // ─── Quick AI Actions ─────────────────────────────────────
   109→          SliverToBoxAdapter(child: _buildQuickActionsSection(context)),
   110→
   111→          // ─── Schedule & Events (2-column on desktop) ──────────────
   112→          SliverToBoxAdapter(
   113→            child: _buildScheduleAndEvents(context, summary),
   114→          ),
   115→
   116→          // ─── Recently Generated AI Content ────────────────────────
   117→          SliverToBoxAdapter(
   118→            child: _buildRecentAiContent(context, summary),
   119→          ),
   120→
   121→          // ─── Draft Lesson Plans ───────────────────────────────────
   122→          SliverToBoxAdapter(
   123→            child: _buildDraftLessonPlans(context, summary),
   124→          ),
   125→
   126→          // Bottom padding
   127→          const SliverPadding(padding: EdgeInsets.only(bottom: Spacings.xxl)),
   128→        ],
   129→      ),
   130→    );
   131→  }

   568→      elevation: Spacings.elevationNone,
   569→      shape: RoundedRectangleBorder(

   615→              resourceType: type,
   616→              resourceId: resourceId,

   619→          ],
   620→        ),
   621→      ),

   625→  IconData _typeIcon(String type) {
   626→    switch (type) {
   627→      case 'lesson_plan':
   628→        return Icons.auto_stories_rounded;
   629→      case 'worksheet':
   630→        return Icons.assignment_rounded;
   631→      case 'scheme_of_work':
   632→        return Icons.calendar_month_rounded;
   633→      case 'assignment':
   634→        return Icons.task_rounded;
   635→      default:
   636→        return Icons.smart_toy_rounded;
   637→    }
   638→  }
   639→
   640→  String _formatRelativeTime(String isoDate) {
   641→    final dt = DateTime.tryParse(isoDate);
   642→    if (dt == null) return '';
   643→    final diff = DateTime.now().difference(dt);
   644→    if (diff.inMinutes < 1) return 'Just now';
   645→    if (diff.inMinutes < 60) return '${diff.inMinutes}m ago';
   646→    if (diff.inHours < 24) return '${diff.inHours}h ago';
   647→    return '${diff.inDays}d ago';
   648→  }

   678→          color: cs.outlineVariant.withValues(alpha: 0.5),
   679→        ),
   680→      ),
   681→      child: Padding(
   682→        padding: const EdgeInsets.all(Spacings.lg),
   683→        child: Column(
   684→          crossAxisAlignment: CrossAxisAlignment.start,
   685→          children: [
   686→            Row(
   687→              children: [
   688→                Container(

   714→                        Text(
   715→                          [subject, topic].whereType<String>().join(' · '),
   716→                          style: tt.bodySmall?.copyWith(
   717→                            color: cs.onSurfaceVariant,
   718→                          ),
   719→                          maxLines: 1,
   720→                          overflow: TextOverflow.ellipsis,
   721→                        ),
   722→                    ],
   723→                  ),
   724→                ),
   725→                // Draft badge
   726→                Container(
   727→                  padding: const EdgeInsets.symmetric(
   728→                    horizontal: Spacings.sm,
   729→                    vertical: Spacings.xs,

   749→                style: tt.bodySmall?.copyWith(

   773→  String _formatRelativeTime(String isoDate) {
   774→    final dt = DateTime.tryParse(isoDate);
   775→    if (dt == null) return '';
   776→    final diff = DateTime.now().difference(dt);
   777→    if (diff.inMinutes < 1) return 'just now';
   778→    if (diff.inMinutes < 60) return '${diff.inMinutes}m ago';
   779→    if (diff.inHours < 24) return '${diff.inHours}h ago';
   780→    return '${diff.inDays}d ago';
   781→  }
