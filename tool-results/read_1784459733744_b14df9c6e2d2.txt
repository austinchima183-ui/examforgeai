    16→Color _severityColor(AuditSeverity severity) {
    17→  switch (severity) {
    18→    case AuditSeverity.info:
    19→      return AppColors.info;
    20→    case AuditSeverity.warning:
    21→      return AppColors.warning;
    22→    case AuditSeverity.error:
    23→      return AppColors.error;
    24→    case AuditSeverity.critical:
    25→      return AppColors.errorDark;
    26→  }
    27→}
    28→
    29→String _formatTimestamp(DateTime dt) {
    30→  final now = DateTime.now();
    31→  final diff = now.difference(dt);
    32→  if (diff.inMinutes < 1) return 'Just now';
    33→  if (diff.inHours < 1) return '${diff.inMinutes}m ago';
    34→  if (diff.inDays < 1) return '${diff.inHours}h ago';
    35→  if (diff.inDays < 7) return '${diff.inDays}d ago';
    36→  return '${dt.day.toString().padLeft(2, '0')}/${dt.month.toString().padLeft(2, '0')}/${dt.year} ${dt.hour.toString().padLeft(2, '0')}:${dt.minute.toString().padLeft(2, '0')}';
    37→}
    38→
    39→String _formatDuration(int? ms) {
    40→  if (ms == null) return '—';
    41→  if (ms < 1000) return '${ms}ms';
    42→  if (ms < 60000) return '${(ms / 1000).toStringAsFixed(1)}s';
    43→  return '${(ms / 60000).toStringAsFixed(1)}m';
    44→}

   105→    if (!_tabController.indexIsChanging) {

   110→  void _loadData() {

   114→  void _loadTabData(int index) {
   115→    final notifier = ref.read(securityCenterProvider.notifier);
   116→    switch (index) {
   117→      case 0:
   118→        notifier.loadAuditLogs(category: _filterCategory);
   119→        break;
   120→      case 1:
   121→        notifier.loadLoginMonitoring(failedOnly: _failedOnly);
   122→        break;
   123→      case 2:
   124→        // Active sessions are loaded as part of login monitoring
   125→        notifier.loadLoginMonitoring();
   126→        break;
   127→      case 3:
   128→        notifier.detectSuspicious();
   129→        break;
   130→    }
   131→  }
