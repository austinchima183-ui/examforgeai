   207→   207→  @override
   208→   208→  Future<GradeScaleModel> createGradeScale(GradeScaleModel scale) async {
   209→   209→    try {
   210→   210→      // 1. Insert the grade_scale row (without nested entries)
   211→   211→      final scaleJson = scale.toJson()
   212→   212→        ..remove('scale_entries');
   213→   213→
   214→   214→      final response = await _supabaseClient
   215→   215→          .from(_gradeScalesTable)
   216→   216→          .insert(scaleJson)
   217→   217→          .select()
   218→   218→          .single();
   219→   219→
   220→   220→      final createdScale = GradeScaleModel.fromJson(response);
   221→   221→
   222→   222→      // 2. Insert entries
   223→   223→      if (scale.scaleEntries.isNotEmpty) {
   224→   224→        final entriesRows = scale.scaleEntries.map((e) {
   225→   225→          final json = e.toJson();
   226→   226→          json['grade_scale_id'] = createdScale.id;
   227→   227→          return json;
   228→   228→        }).toList();
   229→   229→
   230→   230→        await _supabaseClient
   231→   231→            .from(_gradeScaleEntriesTable)
   232→   232→            .insert(entriesRows);
   233→   233→      }
   234→   234→
   235→   235→      // 3. Return scale with entries populated
   236→   236→      return createdScale.copyWith(
   237→   237→        scaleEntries: scale.scaleEntries,
   238→   238→      );
   239→   239→    } on sb.PostgrestException catch (e) {
   240→   240→      AppLogger.error('Create grade scale failed', error: e);
   241→   241→      throw ServerException(
   242→   242→        message: e.message,
   243→   243→        statusCode: int.tryParse(e.code ?? '') ?? 500,
   244→   244→        data: e.details,
   245→   245→      );
   246→   246→    } catch (e) {
   247→   247→      AppLogger.error('Create grade scale unexpected error', error: e);
   248→   248→      throw ServerException(
   249→   249→        message: 'Failed to create grade scale: $e',
   250→   250→        statusCode: 500,
   251→   251→      );
   252→   252→    }
   253→   253→  }

   255→   255→  @override
   256→   256→  Future<GradeScaleModel> updateGradeScale(GradeScaleModel scale) async {
   257→   257→    try {
   258→   258→      // 1. Update the grade_scale row
   259→   259→      final scaleJson = scale.toJson()
   260→   260→        ..remove('scale_entries');
   261→   261→
   262→   262→      final response = await _supabaseClient
   263→   263→          .from(_gradeScalesTable)
   264→   264→          .update(scaleJson)
   265→   265→          .eq('id', scale.id)
   266→   266→          .select()
   267→   267→          .single();
   268→   268→
   269→   269→      final updatedScale = GradeScaleModel.fromJson(response);
   270→   270→
   271→   271→      // 2. Delete existing entries and re-insert
   272→   272→      await _supabaseClient
   273→   273→          .from(_gradeScaleEntriesTable)
   274→   274→          .delete()
   275→   275→          .eq('grade_scale_id', scale.id);
   276→   276→
   277→   277→      if (scale.scaleEntries.isNotEmpty) {
   278→   278→        final entriesRows = scale.scaleEntries.map((e) {
   279→   279→          final json = e.toJson();
   280→   280→          json['grade_scale_id'] = scale.id;
   281→   281→          return json;
   282→   282→        }).toList();
   283→   283→
   284→   284→        await _supabaseClient
   285→   285→            .from(_gradeScaleEntriesTable)
   286→   286→            .insert(entriesRows);
   287→   287→      }
   288→   288→
   289→   289→      // 3. Return scale with entries populated
   290→   290→      return updatedScale.copyWith(
   291→   291→        scaleEntries: scale.scaleEntries,
   292→   292→      );
   293→   293→    } on sb.PostgrestException catch (e) {
   294→   294→      AppLogger.error('Update grade scale failed', error: e);
   295→   295→      if (e.code == 'PGRST116') {
   296→   296→        throw NotFoundException('Grade scale not found: ${scale.id}');
   297→   297→      }
   298→   298→      throw ServerException(
   299→   299→        message: e.message,
   300→   300→        statusCode: int.tryParse(e.code ?? '') ?? 500,
   301→   301→        data: e.details,
   302→   302→      );
   303→   303→    } catch (e) {
   304→   304→      AppLogger.error('Update grade scale unexpected error', error: e);
   305→   305→      throw ServerException(
   306→   306→        message: 'Failed to update grade scale: $e',
   307→   307→        statusCode: 500,
   308→   308→      );
   309→   309→    }
   310→   310→  }

   367→   367→  @override
   368→   368→  Future<List<GradeScaleModel>> getGradeScales(
   369→   369→    String schoolId, {
   370→   370→    bool? isActive,
   371→   371→    String? gradeType,
   372→   372→  }) async {
   373→   373→    try {
   374→   374→      var query = _supabaseClient
   375→   375→          .from(_gradeScalesTable)
   376→   376→          .select('*, grade_scale_entries(*)');
   377→   377→
   378→   378→      query = query.eq('school_id', schoolId);
   379→   379→
   380→   380→      if (isActive != null) {
   381→   381→        query = query.eq('is_active', isActive);
   382→   382→      }
   383→   383→      if (gradeType != null) {
   384→   384→        query = query.eq('grade_type', gradeType);
   385→   385→      }
   386→   386→
   387→   387→      query = query.order('created_at', ascending: false);
   388→   388→
   389→   389→      final response = await query;
   390→   390→
   391→   391→      return response.map<GradeScaleModel>((json) {
   392→   392→        final mappedJson = _mapGradeScaleEntries(json);
   393→   393→        return GradeScaleModel.fromJson(mappedJson);
   394→   394→      }).toList();
   395→   395→    } on sb.PostgrestException catch (e) {
   396→   396→      AppLogger.error('Get grade scales failed', error: e);
   397→   397→      throw ServerException(
   398→   398→        message: e.message,
   399→   399→        statusCode: int.tryParse(e.code ?? '') ?? 500,
   400→   400→        data: e.details,
   401→   401→      );
   402→   402→    } catch (e) {
   403→   403→      AppLogger.error('Get grade scales unexpected error', error: e);
   404→   404→      throw ServerException(
   405→   405→        message: 'Failed to get grade scales: $e',
   406→   406→        statusCode: 500,
   407→   407→      );
   408→   408→    }
   409→   409→  }

   414→   414→    final entries = json['grade_scale_entries'];
   415→   415→    if (entries is List) {
   416→   416→      return {
   417→   417→        ...json,
   418→   418→        'scale_entries': entries,
   419→   419→      };
   420→   420→    }
   421→   421→    return json;
   422→   422→  }
