   106→   106→  void _listenForMessages() {
   107→   107→    final state = ref.read(rubricProvider);
   108→   108→    if (state.successMessage != null) {
   109→   109→      _showSnackBar(state.successMessage!, isError: false);
   110→   110→      ref.read(rubricProvider.notifier).clearSuccessMessage();
   111→   111→    }
   112→   112→    if (state.error != null) {
   113→   113→      _showSnackBar(state.error!, isError: true);
   114→   114→      ref.read(rubricProvider.notifier).clearError();
   115→   115→    }

   118→   118→  void _showSnackBar(String message, {required bool isError}) {
   119→   119→    if (!mounted) return;
   120→   120→    final cs = context.colorScheme;
   121→   121→    context.scaffoldMessenger.showSnackBar(
   122→   122→      SnackBar(

   125→   125→        behavior: SnackBarBehavior.floating,
   126→   126→        shape: RoundedRectangleBorder(
   127→   127→          borderRadius: BorderRadius.circular(Spacings.smRadius),
   128→   128→        ),
   129→   129→        duration: const Duration(seconds: 3),
   130→   130→      ),
   131→   131→    );
   132→   132→  }

   134→   134→  void _initEditControllers(RubricEntity rubric) {
   135→   135→    _rubricTitleCtrl.text = rubric.title;
   136→   136→
   137→   137→    // Clear old controllers
   138→   138→    for (final c in _criterionNameControllers) {
   139→   139→      c.dispose();
   140→   140→    }
   141→   141→    _criterionNameControllers.clear();
   142→   142→    for (final c in _criterionWeightControllers) {
   143→   143→      c.dispose();
   144→   144→    }
   145→   145→    _criterionWeightControllers.clear();
   146→   146→    for (final row in _cellControllers.values) {
   147→   147→      for (final ctrl in row.values) {
   148→   148→        ctrl.dispose();
   149→   149→      }
   150→   150→    }
   151→   151→    _cellControllers.clear();
   152→   152→
   153→   153→    // Create new controllers for each criterion
   154→   154→    for (int i = 0; i < rubric.criteria.length; i++) {
   155→   155→      final criterion = rubric.criteria[i];
   156→   156→      _criterionNameControllers
   157→   157→          .add(TextEditingController(text: criterion.criterion));
   158→   158→      _criterionWeightControllers
   159→   159→          .add(TextEditingController(text: criterion.weight.toString()));
   160→   160→
   161→   161→      final levelControllers = <RubricCriterionLevel, TextEditingController>{};
   162→   162→      for (final level in criterion.levels) {
   163→   163→        levelControllers[level.level] =
   164→   164→            TextEditingController(text: level.description);
   165→   165→      }
   166→   166→      // Ensure all levels have controllers even if missing
   167→   167→      for (final levelValue in RubricCriterionLevel.values) {
   168→   168→        levelControllers.putIfAbsent(
   169→   169→          levelValue,
   170→   170→          () => TextEditingController(text: ''),
   171→   171→        );
   172→   172→      }
   173→   173→      _cellControllers[criterion.criterion] = levelControllers;
   174→   174→    }
   175→   175→  }

   177→   177→  Future<void> _handleGenerate() async {
   178→   178→    if (!_formKey.currentState!.validate()) return;
   179→   179→
   180→   180→    await ref.read(rubricProvider.notifier).generateRubric(
   181→   181→      GenerateRubricParams(
   182→   182→        subjectId: _selectedSubject,
   183→   183→        topic: _topicCtrl.text.trim(),
   184→   184→        criteriaCount: _criteriaCount.round(),
   185→   185→        totalPoints: double.tryParse(_totalPointsCtrl.text) ?? 100,
   186→   186→      ),
   187→   187→    );
   188→   188→
   189→   189→    _listenForMessages();
   190→   190→
   191→   191→    final rubric = ref.read(rubricProvider).currentRubric;
   192→   192→    if (rubric != null) {
   193→   193→      _initEditControllers(rubric);
   194→   194→    }
   195→   195→  }

   217→   217→  void _handleAddCriterion() {
   218→   218→    final rubric = ref.read(rubricProvider).currentRubric;
   219→   219→    if (rubric == null) return;
   220→   220→
   221→   221→    final newCriterion = RubricCriterionEntity(
   222→   222→      criterion: 'New Criterion',
   223→   223→      weight: 1.0,
   224→   224→      levels: RubricCriterionLevel.values
   225→   225→          .map((level) => RubricLevelEntity(
   226→   226→                level: level,
   227→   227→                description: '',
   228→   228→                score: 0,
   229→   229→              ))
   230→   230→          .toList(),
   231→   231→    );
   232→   232→
   233→   233→    final updatedRubric = rubric.copyWith(
   234→   234→      criteria: [...rubric.criteria, newCriterion],
   235→   235→    );
   236→   236→    ref.read(rubricProvider.notifier).setCurrentRubric(updatedRubric);
   237→   237→    _initEditControllers(updatedRubric);
   238→   238→    setState(() {});
   239→   239→  }
   240→   240→
   241→   241→  void _handleRemoveCriterion(int index) {
   242→   242→    final rubric = ref.read(rubricProvider).currentRubric;
   243→   243→    if (rubric == null || rubric.criteria.length <= 1) return;
   244→   244→
   245→   245→    final updatedCriteria = List<RubricCriterionEntity>.from(rubric.criteria)
   246→   246→      ..removeAt(index);
   247→   247→    final updatedRubric = rubric.copyWith(criteria: updatedCriteria);
   248→   248→    ref.read(rubricProvider.notifier).setCurrentRubric(updatedRubric);
   249→   249→    _initEditControllers(updatedRubric);
   250→   250→    setState(() {});
   251→   251→  }
   252→   252→
   253→   253→  void _handleReset() {
   254→   254→    ref.read(rubricProvider.notifier).setCurrentRubric(null);
   255→   255→    _rubricTitleCtrl.clear();
   256→   256→    for (final c in _criterionNameControllers) {
   257→   257→      c.dispose();
   258→   258→    }
   259→   259→    _criterionNameControllers.clear();
   260→   260→    for (final c in _criterionWeightControllers) {
   261→   261→      c.dispose();
   262→   262→    }
   263→   263→    _criterionWeightControllers.clear();
   264→   264→    for (final row in _cellControllers.values) {
   265→   265→      for (final ctrl in row.values) {
   266→   266→        ctrl.dispose();
   267→   267→      }
   268→   268→    }
   269→   269→    _cellControllers.clear();
   270→   270→    setState(() => _isEditing = false);
   271→   271→  }
