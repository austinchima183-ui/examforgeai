    15→/// Abstract interface for all remote CBT engine data operations.
    16→///

    18→/// backend and return plain model instances. Exceptions are allowed to
    19→/// propagate so the repository layer can catch and convert them to
    20→/// domain [Failure] types.

   307→  @override
   308→  Future<List<ExamModel>> getExams(Map<String, dynamic> filters) async {
   309→    try {
   310→      var filterQuery = _supabaseClient.from(_examsTable).select();
   311→
   312→      if (filters['school_id'] != null) {
   313→        filterQuery = filterQuery.eq('school_id', filters['school_id'] as String);
   314→      }
   315→      if (filters['subject_id'] != null) {
   316→        filterQuery = filterQuery.eq('subject_id', filters['subject_id'] as String);
   317→      }
   318→      if (filters['status'] != null) {
   319→        filterQuery = filterQuery.eq('status', filters['status'] as String);
   320→      }
   321→      if (filters['class_id'] != null) {
   322→        filterQuery = filterQuery.eq('class_id', filters['class_id'] as String);
   323→      }
   324→
   325→      final page = filters['page'] as int? ?? 1;
   326→      final perPage = filters['per_page'] as int? ?? filters['perPage'] as int? ?? 20;
   327→      final offset = (page - 1) * perPage;
   328→
   329→      final transformQuery = filterQuery.order('created_at', ascending: false).range(offset, offset + perPage - 1);
   330→
   331→      final response = await transformQuery;
   332→
   333→      return response
   334→          .map<ExamModel>((json) => ExamModel.fromJson(json))
   335→          .toList();
   336→    } on sb.PostgrestException catch (e) {
   337→      AppLogger.error('Get exams failed', error: e);
   338→      throw ServerException(
   339→        message: e.message,
   340→        statusCode: int.tryParse(e.code ?? '') ?? 500,
   341→        data: e.details,
   342→      );
   343→    } catch (e) {
   344→      AppLogger.error('Get exams unexpected error', error: e);
   345→      throw ServerException(
   346→        message: 'Failed to get exams: $e',
   347→        statusCode: 500,
   348→      );
   349→    }
   350→  }

   413→  @override
   414→  Future<ExamModel> cloneExam(String examId) async {
   415→    try {
   416→      // Fetch the original exam with details
   417→      final original = await getExamWithDetails(examId);
   418→
   419→      final now = DateTime.now();
   420→      final cloneData = {
   421→        'school_id': original.schoolId,
   422→        'created_by': original.createdBy,
   423→        'title': '${original.title} (Copy)',
   424→        'description': original.description,
   425→        'subject_id': original.subjectId,
   426→        'class_id': original.classId,
   427→        'academic_session_id': original.academicSessionId,
   428→        'exam_type': original.examType,
   429→        'status': 'draft',
   430→        'start_time': now.add(const Duration(days: 7)).toIso8601String(),
   431→        'end_time': now.add(const Duration(days: 7, hours: 2)).toIso8601String(),
   432→        'time_limit_minutes': original.timeLimitMinutes,
   433→        'total_marks': original.totalMarks,
   434→        'pass_mark': original.passMark,
   435→        'pass_mark_type': original.passMarkType,
   436→        'instructions': original.instructions,
   437→        'allowed_attempts': original.allowedAttempts,
   438→        'negative_marking_enabled': original.negativeMarkingEnabled,
   439→        'negative_mark_value': original.negativeMarkValue,
   440→        'grace_period_minutes': original.gracePeriodMinutes,
   441→        'auto_submit': original.autoSubmit,
   442→        'randomize_questions': original.randomizeQuestions,
   443→        'randomize_options': original.randomizeOptions,
   444→        'show_results': original.showResults,
   445→        'show_correct_answers': original.showCorrectAnswers,
   446→        'show_explanations': original.showExplanations,
   447→        'is_template': original.isTemplate,
   448→        'template_id': original.id,
   449→        'max_students': original.maxStudents,
   450→        'ip_restriction': original.ipRestriction,
   451→        'require_full_screen': original.requireFullScreen,
   452→        'allow_resume': original.allowResume,
   453→        'browser_lockdown': original.browserLockdown,
   454→        'metadata': original.metadata,
   455→      };
   456→
   457→      final clonedExam = await createExam(cloneData);
   458→
   459→      // Clone sections
   460→      for (final section in original.sections) {
   461→        await _supabaseClient.from(_examSectionsTable).insert({
   462→          'exam_id': clonedExam.id,
   463→          'title': section.title,
   464→          'description': section.description,
   465→          'instructions': section.instructions,
   466→          'sort_order': section.sortOrder,
   467→          'time_limit_minutes': section.timeLimitMinutes,
   468→          'randomize_questions': section.randomizeQuestions,
   469→        });
   470→      }
   471→
   472→      // Clone questions
   473→      for (final question in original.questions) {
   474→        await _supabaseClient.from(_examQuestionsTable).insert({
   475→          'exam_id': clonedExam.id,
   476→          'section_id': question.sectionId,
   477→          'question_id': question.questionId,
   478→          'sort_order': question.sortOrder,
   479→          'marks': question.marks,
   480→          'negative_marks': question.negativeMarks,
   481→          'is_compulsory': question.isCompulsory,
   482→        });
   483→      }
   484→
   485→      return clonedExam;
   486→    } on ServerException {
   487→      rethrow;
   488→    } on sb.PostgrestException catch (e) {
   489→      AppLogger.error('Clone exam failed', error: e);
   490→      throw ServerException(
   491→        message: e.message,
   492→        statusCode: int.tryParse(e.code ?? '') ?? 500,
   493→        data: e.details,
   494→      );
   495→    } catch (e) {
   496→      AppLogger.error('Clone exam unexpected error', error: e);
   497→      throw ServerException(
   498→        message: 'Failed to clone exam: $e',
   499→        statusCode: 500,
   500→      );
   501→    }
   502→  }

   580→        message: e.message,

   655→  @override
   656→  Future<ExamAttemptModel> startAttempt(String examId) async {
   657→    try {
   658→      final userId = _supabaseClient.auth.currentUser?.id;
   659→      if (userId == null) {
   660→        throw const AuthException(
   661→          message: 'Not authenticated',
   662→          code: 'NOT_AUTHENTICATED',
   663→        );
   664→      }
   665→
   666→      // ── Call server-side SQL function which validates eligibility,
   667→      //    creates the attempt AND session atomically ───────────────
   668→      final response = await _supabaseClient.rpc(
   669→        'start_exam_attempt',
   670→        params: {'p_exam_id': examId},
   671→      );
   672→
   673→      // Server returns the created attempt row as JSON
   674→      if (response == null) {
   675→        throw const ServerException(
   676→          message: 'Server returned no data for start_exam_attempt',
   677→          statusCode: 500,
   678→        );
   679→      }
   680→
   681→      final attemptMap = response as Map<String, dynamic>;
   682→
   683→      // Validate the server confirmed the attempt was started
   684→      final attemptStatus = attemptMap['status'] as String?;
   685→      if (attemptStatus != 'in_progress') {
   686→        AppLogger.warning(
   687→          'Server returned unexpected attempt status: $attemptStatus',
   688→        );
   689→        throw ServerException(
   690→          message: 'Server did not start attempt. Status: $attemptStatus',
   691→          statusCode: 409,
   692→          data: attemptMap,
   693→        );
   694→      }
   695→
   696→      return ExamAttemptModel.fromJson(attemptMap);
   697→    } on AuthException {
   698→      rethrow;
   699→    } on sb.PostgrestException catch (e) {
   700→      AppLogger.error('Start attempt RPC failed', error: e);
   701→      // Map known Postgrest error codes to domain-meaningful exceptions
   702→      final msg = e.message;
   703→      if (msg.contains('not assigned') || msg.contains('not eligible')) {
   704→        throw ForbiddenException(message: msg);
   705→      }
   706→      if (msg.contains('max attempts') || msg.contains('already in progress')) {
   707→        throw ValidationException(message: msg);
   708→      }
   709→      throw ServerException(
   710→        message: e.message,
   711→        statusCode: int.tryParse(e.code ?? '') ?? 500,
   712→        data: e.details,
   713→      );
   714→    } on ForbiddenException {
   715→      rethrow;
   716→    } on ValidationException {
   717→      rethrow;
   718→    } catch (e) {
   719→      if (e is ServerException || e is AuthException) rethrow;
   720→      AppLogger.error('Start attempt unexpected error', error: e);
   721→      throw ServerException(
   722→        message: 'Failed to start attempt: $e',
   723→        statusCode: 500,
   724→      );
   725→    }
   726→  }

   898→  @override
   899→  Future<StudentAnswerModel> saveAnswer(
   900→    String attemptId,
   901→    String questionId,
   902→    Map<String, dynamic> answerData,
   903→  ) async {
   904→    try {
   905→      // ── Validate exam timing server-side before accepting the answer ──
   906→      final timingResponse = await _supabaseClient.rpc(
   907→        'validate_exam_timing',
   908→        params: {'p_attempt_id': attemptId},
   909→      );
   910→
   911→      final timingResult = timingResponse as Map<String, dynamic>?;
   912→      final isWithinTime = timingResult?['is_within_time'] as bool? ?? false;
   913→
   914→      if (!isWithinTime) {
   915→        throw const ValidationException(
   916→          message: 'Cannot save answer: exam time has exceeded',
   917→          fieldErrors: {
   918→            'time_exceeded': 'The exam time limit has been exceeded. '
   919→                'No further answers can be saved.',
   920→          },
   921→        );
   922→      }
   923→
   924→      // Use server time for the answer timestamp to avoid client clock skew
   925→      final serverTime = timingResult?['server_time'] as String?;
   926→      final now = serverTime != null
   927→          ? DateTime.parse(serverTime)
   928→          : DateTime.now();
   929→
   930→      // Upsert: update existing answer or insert new one
   931→      final existing = await _supabaseClient
   932→          .from(_studentAnswersTable)
   933→          .select('id')
   934→          .eq('attempt_id', attemptId)
   935→          .eq('question_id', questionId)
   936→          .maybeSingle();
   937→
   938→      if (existing != null) {
   939→        final response = await _supabaseClient
   940→            .from(_studentAnswersTable)
   941→            .update({
   942→              'answer_data': answerData,
   943→              'answered_at': now.toIso8601String(),
   944→              'updated_at': now.toIso8601String(),
   945→            })
   946→            .eq('id', existing['id'] as String)
   947→            .select()
   948→            .single();
   949→
   950→        return StudentAnswerModel.fromJson(response);
   951→      } else {
   952→        final response = await _supabaseClient
   953→            .from(_studentAnswersTable)
   954→            .insert({
   955→              'attempt_id': attemptId,
   956→              'question_id': questionId,
   957→              'answer_data': answerData,
   958→              'answered_at': now.toIso8601String(),
   959→            })
   960→            .select()
   961→            .single();
   962→
   963→        return StudentAnswerModel.fromJson(response);
   964→      }
   965→    } on sb.PostgrestException catch (e) {
   966→      AppLogger.error('Save answer failed', error: e);
   967→      throw ServerException(
   968→        message: e.message,
   969→        statusCode: int.tryParse(e.code ?? '') ?? 500,
   970→        data: e.details,
   971→      );
   972→    } on ValidationException {
   973→      rethrow;
   974→    } catch (e) {
   975→      if (e is ServerException || e is ValidationException) rethrow;
   976→      AppLogger.error('Save answer unexpected error', error: e);
   977→      throw ServerException(
   978→        message: 'Failed to save answer: $e',
   979→        statusCode: 500,
   980→      );
   981→    }
   982→  }

  1313→  @override
  1314→  Future<ExamStatisticsModel> getExamStatistics(String examId) async {
  1315→    try {
  1316→      // PERF: Use DatabasePoolManager.executeMonitored for performance tracking
  1317→      // on the most expensive CBT queries
  1318→      final results = await DatabasePoolManager.executeMonitored(
  1319→        queryType: 'select',
  1320→        table: _examResultsTable,
  1321→        operation: 'get_exam_statistics',
  1322→        query: () => _supabaseClient
  1323→            .from(_examResultsTable)
  1324→            .select('score_percentage, is_passed, grading_status')
  1325→            .eq('exam_id', examId),
  1326→      );
  1327→
  1328→      // PERF: Only count IDs, don't fetch full rows
  1329→      final students = await _supabaseClient
  1330→          .from(_examStudentsTable)
  1331→          .select('id')
  1332→          .eq('exam_id', examId)
  1333→          .limit(PaginatedQueryMixin.maxPageSize); // Safety cap
  1334→
  1335→      // PERF: Select only columns needed for statistics calculation
  1336→      final attempts = await _supabaseClient
  1337→          .from(_examAttemptsTable)
  1338→          .select('status, time_spent_seconds')
  1339→          .eq('exam_id', examId)
  1340→          .limit(PaginatedQueryMixin.maxPageSize); // Safety cap
  1341→
  1342→      final totalStudents = students.length;
  1343→      final startedStudents = attempts
  1344→          .where((a) => a['status'] != 'not_started')
  1345→          .length;
  1346→      final completedStudents = attempts
  1347→          .where((a) =>
  1348→              a['status'] == 'submitted' ||
  1349→              a['status'] == 'auto_submitted' ||
  1350→              a['status'] == 'timed_out',)
  1351→          .length;
  1352→      final submittedStudents = completedStudents;
  1353→
  1354→      // Calculate score statistics
  1355→      final scores = results
  1356→          .map<double>((r) => (r['score_percentage'] as num?)?.toDouble() ?? 0.0)
  1357→          .toList();
  1358→
  1359→      final averageScore = scores.isNotEmpty
  1360→          ? scores.reduce((a, b) => a + b) / scores.length
  1361→          : 0.0;
  1362→      final highestScore = scores.isNotEmpty
  1363→          ? scores.reduce((a, b) => a > b ? a : b)
  1364→          : 0.0;
  1365→      final lowestScore = scores.isNotEmpty
  1366→          ? scores.reduce((a, b) => a < b ? a : b)
  1367→          : 0.0;
  1368→
  1369→      // Calculate median
  1370→      double medianScore = 0.0;
  1371→      if (scores.isNotEmpty) {
  1372→        final sorted = List<double>.from(scores)..sort();
  1373→        final mid = sorted.length ~/ 2;
  1374→        medianScore = sorted.length.isEven
  1375→            ? (sorted[mid - 1] + sorted[mid]) / 2
  1376→            : sorted[mid];
  1377→      }
  1378→
  1379→      // Calculate pass rate
  1380→      final passedCount = results
  1381→          .where((r) => r['is_passed'] == true)
  1382→          .length;
  1383→      final passRate =
  1384→          results.isNotEmpty ? (passedCount / results.length) * 100 : 0.0;
  1385→
  1386→      // Average time spent
  1387→      final timeSpentList = attempts
  1388→          .where((a) => a['time_spent_seconds'] != null)
  1389→          .map<int>((a) => a['time_spent_seconds'] as int)
  1390→          .toList();
  1391→      final averageTimeSpent = timeSpentList.isNotEmpty
  1392→          ? timeSpentList.reduce((a, b) => a + b) ~/ timeSpentList.length
  1393→          : 0;
  1394→
  1395→      // Grading completion
  1396→      final gradedCount = results
  1397→          .where((r) =>
  1398→              r['grading_status'] == 'auto_graded' ||
  1399→              r['grading_status'] == 'fully_graded',)
  1400→          .length;
  1401→      final gradingCompletion = results.isNotEmpty
  1402→          ? (gradedCount / results.length) * 100
  1403→          : 0.0;
  1404→
  1405→      return ExamStatisticsModel(
  1406→        examId: examId,
  1407→        totalStudents: totalStudents,
  1408→        startedStudents: startedStudents,
  1409→        completedStudents: completedStudents,
  1410→        submittedStudents: submittedStudents,
  1411→        averageScore: averageScore,
  1412→        highestScore: highestScore,
  1413→        lowestScore: lowestScore,
  1414→        medianScore: medianScore,
  1415→        passRate: passRate,
  1416→        questionsByCorrectRate: {}, // Requires per-question analysis
  1417→        averageTimeSpentSeconds: averageTimeSpent,
  1418→        gradingCompletionPercentage: gradingCompletion,
  1419→      );
  1420→    } on sb.PostgrestException catch (e) {
  1421→      AppLogger.error('Get exam statistics failed', error: e);
  1422→      throw ServerException(
  1423→        message: e.message,
  1424→        statusCode: int.tryParse(e.code ?? '') ?? 500,
  1425→        data: e.details,
  1426→      );
  1427→    } catch (e) {
  1428→      AppLogger.error('Get exam statistics unexpected error', error: e);
  1429→      throw ServerException(
  1430→        message: 'Failed to get exam statistics: $e',
  1431→        statusCode: 500,
  1432→      );
  1433→    }
  1434→  }

  1436→  @override
  1437→  Future<LiveExamStatsModel> getLiveExamStats(String examId) async {
  1438→    try {
  1439→      // PERF: Use DatabasePoolManager monitoring for the live stats query
  1440→      // which is polled every 5-10 seconds during active exams
  1441→      final activeSessions = await DatabasePoolManager.executeMonitored(
  1442→        queryType: 'select',
  1443→        table: _examSessionsTable,
  1444→        operation: 'get_live_stats_sessions',
  1445→        query: () => _supabaseClient
  1446→            .from(_examSessionsTable)
  1447→            .select('id, questions_answered, questions_flagged')
  1448→            .eq('exam_id', examId)
  1449→            .eq('is_active', true)
  1450→            .limit(PaginatedQueryMixin.maxPageSize),
  1451→      );
  1452→
  1453→      // Count students (bounded by exam size)
  1454→      final students = await _supabaseClient
  1455→          .from(_examStudentsTable)
  1456→          .select('id')
  1457→          .eq('exam_id', examId)
  1458→          .limit(PaginatedQueryMixin.maxPageSize);
  1459→
  1460→      // PERF: Only select columns needed for counting
  1461→      final submittedAttempts = await _supabaseClient
  1462→          .from(_examAttemptsTable)
  1463→          .select('id, status')
  1464→          .eq('exam_id', examId)
  1465→          .inFilter('status', ['submitted', 'auto_submitted', 'timed_out']);
  1466→
  1467→      // Recent submissions (last 10)
  1468→      final recentSubmissions = await _supabaseClient
  1469→          .from(_examAttemptsTable)
  1470→          .select()
  1471→          .eq('exam_id', examId)
  1472→          .inFilter('status', ['submitted', 'auto_submitted', 'timed_out'])
  1473→          .order('submitted_at', ascending: false)
  1474→          .limit(10);
  1475→
  1476→      // Recent monitoring events (last 20)
  1477→      final recentEvents = await _supabaseClient
  1478→          .from(_monitoringLogsTable)
  1479→          .select()
  1480→          .eq('exam_id', examId)
  1481→          .order('created_at', ascending: false)
  1482→          .limit(20);
  1483→
  1484→      final totalEligible = students.length;
  1485→      final activeNow = activeSessions.length;
  1486→      final completed = submittedAttempts.length;
  1487→      final notStarted = totalEligible - activeNow - completed;
  1488→
  1489→      // Calculate average progress
  1490→      double averageProgress = 0;
  1491→      if (activeSessions.isNotEmpty) {
  1492→        final totalQuestionsAnswered = activeSessions.fold<int>(
  1493→          0,
  1494→          (sum, s) =>
  1495→              sum + ((s['questions_answered'] as int?) ?? 0),
  1496→        );
  1497→        // Estimate total questions per exam
  1498→        averageProgress = totalEligible > 0
  1499→            ? totalQuestionsAnswered / (activeNow * 10) * 100
  1500→            : 0;
  1501→        averageProgress = averageProgress.clamp(0.0, 100.0);
  1502→      }
  1503→
  1504→      return LiveExamStatsModel(
  1505→        examId: examId,
  1506→        totalEligible: totalEligible,
  1507→        activeNow: activeNow,
  1508→        completed: completed,
  1509→        notStarted: notStarted < 0 ? 0 : notStarted,
  1510→        averageProgress: averageProgress,
  1511→        recentSubmissions: recentSubmissions
  1512→            .map<ExamAttemptModel>((a) => ExamAttemptModel.fromJson(a))
  1513→            .toList(),
  1514→        activeSessions: activeSessions
  1515→            .map<ExamSessionModel>((s) => ExamSessionModel.fromJson(s))
  1516→            .toList(),
  1517→        recentMonitoringEvents: recentEvents
  1518→            .map<MonitoringLogModel>((e) => MonitoringLogModel.fromJson(e))
  1519→            .toList(),
  1520→      );
  1521→    } on sb.PostgrestException catch (e) {
  1522→      AppLogger.error('Get live exam stats failed', error: e);
  1523→      throw ServerException(
  1524→        message: e.message,
  1525→        statusCode: int.tryParse(e.code ?? '') ?? 500,
  1526→        data: e.details,
  1527→      );
  1528→    } catch (e) {
  1529→      AppLogger.error('Get live exam stats unexpected error', error: e);
  1530→      throw ServerException(
  1531→        message: 'Failed to get live exam stats: $e',
  1532→        statusCode: 500,
  1533→      );
  1534→    }
  1535→  }
