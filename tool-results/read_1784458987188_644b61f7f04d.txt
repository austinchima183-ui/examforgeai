    90→    // Listen for success/error messages

   107→            content: Text(next.error!),
   108→            backgroundColor: AppColors.error,

   128→      body: _buildBody(context, state, cs),

   219→  Widget _buildFilterChips(ColorScheme cs) {
   220→    return Wrap(
   221→      spacing: Spacings.sm,
   222→      runSpacing: Spacings.sm,
   223→      children: [
   224→        // Role filters

   226→          (role) => FilterChip(
   227→            label: Text(role.value),
   228→            selected: _roleFilter == role.key,
   229→            onSelected: (selected) {

   231→              _loadData();
   232→            },
   233→          ),
   234→        ),
   235→        const SizedBox(width: Spacings.xs),
   236→        // Active / Inactive
   237→        FilterChip(
   238→          label: const Text('Active'),

   241→            setState(() => _activeFilter = selected ? true : null);

   247→          selected: _activeFilter == false,
   248→          onSelected: (selected) {
   249→            setState(() => _activeFilter = selected ? false : null);

   251→          },

   253→      ],

   557→  String _roleLabel(String role) {

   568→      case 'school_admin':

   598→  String _formatLastLogin(DateTime? lastLoginAt) {
   599→    if (lastLoginAt == null) return 'Never';
   600→    final now = DateTime.now();
   601→    final diff = now.difference(lastLoginAt);
   602→    if (diff.inMinutes < 1) return 'Just now';
   603→    if (diff.inHours < 1) return '${diff.inMinutes}m ago';
   604→    if (diff.inDays < 1) return '${diff.inHours}h ago';
   605→    if (diff.inDays < 30) return '${diff.inDays}d ago';
   606→    return '${lastLoginAt.day}/${lastLoginAt.month}/${lastLoginAt.year}';
   607→  }
   608→}
