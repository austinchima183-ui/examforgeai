   858→   858→-- ---- Auto-update updated_at on all tables with that column ----
   859→   859→DO $$
   860→   860→DECLARE
   861→   861→  t TEXT;
   862→   862→BEGIN
   863→   863→  FOR t IN
   864→   864→    SELECT table_name FROM information_schema.columns
   865→   865→    WHERE table_schema = 'public'
   866→   866→      AND column_name = 'updated_at'
   867→   867→      AND table_name IN (
   868→   868→        'marketplace_categories',
   869→   869→        'seller_profiles',
   870→   870→        'marketplace_products',
   871→   871→        'marketplace_carts',
   872→   872→        'marketplace_orders',
   873→   873→        'marketplace_reviews',
   874→   874→        'marketplace_promo_codes',
   875→   875→        'marketplace_commission_rates',
   876→   876→        'marketplace_disputes'
   877→   877→      )
   878→   878→    GROUP BY table_name
   879→   879→  LOOP
   880→   880→    EXECUTE format(
   881→   881→      'CREATE OR REPLACE TRIGGER trg_%s_updated_at
   882→   882→         BEFORE UPDATE ON %I
   883→   883→         FOR EACH ROW
   884→   884→         EXECUTE FUNCTION update_updated_at_column()',
   885→   885→      t, t
   886→   886→    );
   887→   887→  END LOOP;
   888→   888→END;
   889→   889→$$;

   891→   891→-- ---- Auto-update product average_rating and total_reviews ----
   892→   892→CREATE OR REPLACE FUNCTION update_product_rating()
   893→   893→RETURNS TRIGGER AS $$
   894→   894→BEGIN
   895→   895→  IF TG_OP = 'INSERT' THEN
   896→   896→    UPDATE marketplace_products
   897→   897→    SET
   898→   898→      average_rating = (
   899→   899→        SELECT COALESCE(AVG(rating), 0.00)
   900→   900→        FROM marketplace_reviews
   901→   901→        WHERE product_id = NEW.product_id AND status = 'published'
   902→   902→      ),
   903→   903→      total_reviews = (
   904→   904→        SELECT COUNT(*)
   905→   905→        FROM marketplace_reviews
   906→   906→        WHERE product_id = NEW.product_id AND status = 'published'
   907→   907→      )
   908→   908→    WHERE id = NEW.product_id;
   909→   909→  ELSIF TG_OP = 'UPDATE' THEN
   910→   910→    UPDATE marketplace_products
   911→   911→    SET
   912→   912→      average_rating = (
   913→   913→        SELECT COALESCE(AVG(rating), 0.00)
   914→   914→        FROM marketplace_reviews
   915→   915→        WHERE product_id = NEW.product_id AND status = 'published'
   916→   916→      ),
   917→   917→      total_reviews = (
   918→   918→        SELECT COUNT(*)
   919→   919→        FROM marketplace_reviews
   920→   920→        WHERE product_id = NEW.product_id AND status = 'published'
   921→   921→      )
   922→   922→    WHERE id = NEW.product_id;
   923→   923→  ELSIF TG_OP = 'DELETE' THEN
   924→   924→    UPDATE marketplace_products
   925→   925→    SET
   926→   926→      average_rating = (
   927→   927→        SELECT COALESCE(AVG(rating), 0.00)
   928→   928→        FROM marketplace_reviews
   929→   929→        WHERE product_id = OLD.product_id AND status = 'published'

   931→   931→      total_reviews = (
   932→   932→        SELECT COUNT(*)
   933→   933→        FROM marketplace_reviews
   934→   934→        WHERE product_id = OLD.product_id AND status = 'published'
   935→   935→      )
   936→   936→    WHERE id = OLD.product_id;
   937→   937→  END IF;
   938→   938→  RETURN COALESCE(NEW, OLD);
   939→   939→END;
