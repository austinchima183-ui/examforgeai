   312→   312→          .select()

   315→   315→      return SchoolModel.fromJson(response);
   316→   316→    } on sb.PostgrestException catch (e) {

   417→   417→  @override
   418→   418→  Future<List<SchoolModel>> getSchools(Map<String, dynamic> filters) async {
   419→   419→    try {
   420→   420→      // PERF: Explicit columns avoid fetching all school fields for list view
   421→   421→      var query = _supabaseClient.from(_schoolsTable).select('id, name, address, is_active, created_at');
   422→   422→
   423→   423→      if (filters['is_active'] != null) {
   424→   424→        query = query.eq('is_active', filters['is_active'] as bool);
   425→   425→      }
   426→   426→      if (filters['search'] != null) {
   427→   427→        final search = filters['search'] as String;
   428→   428→        query = query.or('name.ilike.%$search%,code.ilike.%$search%');
   429→   429→      }
   430→   430→      if (filters['school_level'] != null) {
   431→   431→        query = query.eq('school_level', filters['school_level'] as String);
   432→   432→      }
   433→   433→      if (filters['subscription_status'] != null) {
   434→   434→        query = query.eq(
   435→   435→          'subscription_status',
   436→   436→          filters['subscription_status'] as String,
   437→   437→        );
   438→   438→      }
   439→   439→
   440→   440→      final page = filters['page'] as int? ?? 1;
   441→   441→      final perPage = filters['per_page'] as int? ?? filters['perPage'] as int? ?? 20;
   442→   442→      final offset = (page - 1) * perPage;
   443→   443→
   444→   444→      final response = await query.range(offset, offset + perPage - 1).order('created_at', ascending: false);
   445→   445→
   446→   446→      return response
   447→   447→          .map<SchoolModel>((json) => SchoolModel.fromJson(json))
   448→   448→          .toList();
   449→   449→    } on sb.PostgrestException catch (e) {
   450→   450→      AppLogger.error('Get schools failed', error: e);
   451→   451→      throw ServerException(
   452→   452→        message: e.message,
   453→   453→        statusCode: int.tryParse(e.code ?? '') ?? 500,
   454→   454→        data: e.details,
