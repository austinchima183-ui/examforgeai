   253→/// Provides the Supabase client singleton.
   254→///
   255→/// Depends on [SupabaseConfig.initialize] having been called in `main()`.
   256→final supabaseClientProvider = Provider<sb.SupabaseClient>((ref) {
   257→  try {
   258→    return SupabaseConfig.client;
   259→  } catch (e) {
   260→    AppLogger.error('Failed to resolve Supabase client', error: e);
   261→    rethrow;
   262→  }
   263→});
   264→
   265→/// Provides a configured [Dio] instance with base options pulled from
   266→/// [AppConfig] and [ApiConstants].
   267→final dioProvider = Provider<Dio>((ref) {
   268→  final dio = Dio(
   269→    BaseOptions(
   270→      baseUrl: AppConfig.baseUrl,
   271→      connectTimeout: AppConfig.connectTimeoutDuration,
   272→      receiveTimeout: AppConfig.receiveTimeoutDuration,
   273→      sendTimeout: AppConfig.sendTimeoutDuration,
   274→      headers: ApiConstants.baseHeaders(),
   275→      responseType: ResponseType.json,
   276→      validateStatus: (status) => status != null && status < 500,
   277→    ),
   278→  );
   279→  return dio;
   280→});
   281→
   282→/// Provides the [ApiClient] wrapper around [Dio].
   283→///
   284→/// FIX: Now injects [StorageService] so that auth tokens are actually
   285→/// read from secure storage instead of returning null (the root cause
   286→/// of the "ApiClient auth always returning null" bug).
   287→final apiClientProvider = Provider<ApiClient>((ref) {
   288→  final dio = ref.watch(dioProvider);
   289→  final storageService = ref.watch(storageServiceProvider);
   290→  return ApiClient(dio, storageService: storageService);
   291→});

   297→  return ref.watch(coreNetworkInfoProvider);

   341→/// Stream provider that emits the current Supabase [sb.AuthState].

   343→/// Automatically reconnects on transient failures and emits the latest
   344→/// state to any listener.

   349→
   350→/// Convenience provider that yields the currently authenticated [sb.User],
   351→/// or `null` if not signed in.
   352→final currentUserProvider = Provider<sb.User?>((ref) {
