   156→/// Provides a configured [Dio] instance with base options pulled from
   157→/// [AppConfig] and [ApiConstants].
   158→final dioProvider = Provider<Dio>((ref) {
   159→  final dio = Dio(
   160→    BaseOptions(
   161→      baseUrl: AppConfig.baseUrl,
   162→      connectTimeout: AppConfig.connectTimeoutDuration,
   163→      receiveTimeout: AppConfig.receiveTimeoutDuration,
   164→      sendTimeout: AppConfig.sendTimeoutDuration,
   165→      headers: ApiConstants.baseHeaders(),
   166→      responseType: ResponseType.json,
   167→      validateStatus: (status) => status != null && status < 500,
   168→    ),
   169→  );
   170→  return dio;
   171→});

   227→/// Stream provider that emits the current Supabase [sb.AuthState].

   236→/// Convenience provider that yields the currently authenticated [sb.User],
   237→/// or `null` if not signed in.
   238→final currentUserProvider = Provider<sb.User?>((ref) {
   239→  final authState = ref.watch(authStateProvider);
   240→  return authState.when(
   241→    data: (state) => state.session?.user,
   242→    loading: () => SupabaseConfig.currentUser,
   243→    error: (_, __) => null,
   244→  );
   245→});

   247→/// Convenience provider that yields `true` when the user is authenticated.
