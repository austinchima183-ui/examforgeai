    32→  @override
    33→  Future<Result<List<GeneratedQuestionEntity>>> generateQuestions(
    34→    GenerationInputEntity input,
    35→  ) async {
    36→    try {
    37→      // Create the generation request
    38→      final inputModel = GenerationInputModel.fromEntity(input);
    39→      final requestData = {
    40→        'school_id': '', // Will be set by the caller
    41→        'requested_by': '', // Will be set by the caller
    42→        'provider': input.provider?.value ?? 'openai',
    43→        'model_name': input.provider?.defaultModel ?? 'gpt-4o',
    44→        'prompt_template_id': input.promptTemplateId,
    45→        'generation_type': 'question_generation',
    46→        'status': 'pending',
    47→        'input_params': inputModel.toJson(),
    48→        'system_prompt': '',
    49→        'user_prompt': '',
    50→      };
    51→
    52→      final requestModel =
    53→          await _remoteDataSource.createGenerationRequest(requestData);
    54→
    55→      // Update with processing status
    56→      await _remoteDataSource.updateGenerationRequest(
    57→        requestModel.id,
    58→        {'status': 'processing', 'started_at': DateTime.now().toIso8601String()},
    59→      );
    60→
    61→      // Fetch prompt templates for generation
    62→      final templatesData =
    63→          await _remoteDataSource.getPromptTemplates({
    64→        'prompt_type': 'question_generation',
    65→        if (input.subjectId.isNotEmpty) 'subject_id': input.subjectId,
    66→      });
    67→
    68→      // Note: The actual AI generation happens in the AiService layer.
    69→      // This repository method creates the request and returns the
    70→      // generated questions after the AI service processes them.
    71→      // For now, we return the generated questions associated with
    72→      // the request after marking it as completed.
    73→
    74→      final questions = await _remoteDataSource.getGeneratedQuestions({
    75→        'generation_request_id': requestModel.id,
    76→      });
    77→
    78→      // Update request as completed
    79→      await _remoteDataSource.updateGenerationRequest(
    80→        requestModel.id,
    81→        {
    82→          'status': 'completed',
    83→          'completed_at': DateTime.now().toIso8601String(),
    84→        },
    85→      );
    86→
    87→      return Success(questions.map((m) => m.toEntity()).toList());
    88→    } on AuthException catch (e) {
    89→      return FailureResult(Failure.auth(message: e.message, code: e.code));
    90→    } on ServerException catch (e) {
    91→      return FailureResult(Failure.server(
    92→        message: e.message,
    93→        statusCode: e.statusCode,
    94→        data: e.data,
    95→      ));
    96→    } on NetworkException catch (e) {
    97→      return FailureResult(Failure.network(message: e.message));
    98→    } on NotFoundException catch (e) {
    99→      return FailureResult(Failure.notFound(message: e.message));
   100→    } catch (e) {
   101→      AppLogger.error('Unexpected generateQuestions error', error: e);
   102→      return FailureResult(Failure.server(
   103→        message: 'An unexpected error occurred during generation.',
   104→        statusCode: 500,
   105→      ));
   106→    }
   107→  }

   309→  @override
   310→  Future<Result<GeneratedQuestionEntity>> acceptImprovement(
   311→    String improvementId,
   312→  ) async {
   313→    try {
   314→      // Get the improvement
   315→      final improvementModel =
   316→          await _remoteDataSource.getImprovement(improvementId);
   317→      final improvement = improvementModel.toEntity();
   318→
   319→      // Mark improvement as accepted
   320→      await _remoteDataSource.updateImprovement(
   321→        improvementId,
   322→        {'is_accepted': true},
   323→      );
   324→
   325→      // Update the generated question with improved content
   326→      final updateData = <String, dynamic>{
   327→        'content': improvement.improvedContent,
   328→        'is_edited': true,
   329→        'updated_at': DateTime.now().toIso8601String(),
   330→      };
   331→
   332→      if (improvement.improvedAnswerOptions != null) {
   333→        updateData['answer_options'] = improvement.improvedAnswerOptions;
   334→      }
   335→
   336→      final updatedQuestion = await _remoteDataSource.updateGeneratedQuestion(
   337→        improvement.generatedQuestionId,
   338→        updateData,
   339→      );
   340→
   341→      return Success(updatedQuestion.toEntity());
   342→    } on AuthException catch (e) {
   343→      return FailureResult(Failure.auth(message: e.message, code: e.code));
   344→    } on ServerException catch (e) {
   345→      return FailureResult(Failure.server(
   346→        message: e.message,
   347→        statusCode: e.statusCode,
   348→        data: e.data,
   349→      ));
   350→    } on NotFoundException catch (e) {
   351→      return FailureResult(Failure.notFound(message: e.message));
   352→    } catch (e) {
   353→      AppLogger.error('Unexpected acceptImprovement error', error: e);
   354→      return FailureResult(Failure.server(
   355→        message: 'Failed to accept improvement.',
   356→        statusCode: 500,
   357→      ));
   358→    }
   359→  }
