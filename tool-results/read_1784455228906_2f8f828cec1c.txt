   187→// APP ROUTER PROVIDER
   188→// ═══════════════════════════════════════════════════════════════════════

   190→/// Riverpod provider for the application's [GoRouter].
   191→///
   192→/// Watches [authStateProvider] so that the router automatically
   193→/// re-evaluates its [redirect] callback whenever the authentication
   194→/// state changes (sign-in, sign-out, token refresh).
   195→///
   196→/// Usage in `main.dart`:
   197→/// ```dart
   198→/// final router = ref.read(appRouterProvider);
   199→/// MaterialApp.router(routerConfig: router);
   200→/// ```
   201→final appRouterProvider = Provider<GoRouter>((ref) {
   202→  // Watch auth state so the router re-evaluates redirects on change.
   203→  final authState = ref.watch(authStateProvider);
   204→
   205→  // Also watch onboarding state and role so those redirects stay live.
   206→  ref.watch(onboardingCompleteProvider);
   207→  ref.watch(userRoleProvider);
   208→
   209→  return GoRouter(
   210→    navigatorKey: _rootNavigatorKey,
   211→    initialLocation: RouteNames.splash,
   212→    debugLogDiagnostics: true,
   213→
   214→    // ─── Global Redirect ──────────────────────────────────────────
   215→    redirect: (context, state) {
   216→      final currentPath = state.matchedLocation;
   217→      final isAuthenticated = authState.when(
   218→        data: (authData) => authData.session != null,
   219→        loading: () {
   220→          // While loading, only allow splash. Everything else redirects
   221→          // to splash so we don't flash the login page briefly.
   222→          if (currentPath != RouteNames.splash) {
   223→            return RouteNames.splash;
   224→          }
   225→          return null;
   226→        },
   227→        error: (_, __) => false,
   228→      );
   229→
   230→      // ── Auth Guard ────────────────────────────────────────────
   231→      final authRedirect = AuthGuard.evaluate(
   232→        isAuthenticated: isAuthenticated,
   233→        currentPath: currentPath,
   234→      );
   235→      if (authRedirect != null) return authRedirect;
   236→
   237→      // ── Onboarding Guard ─────────────────────────────────────
   238→      final onboardingAsync = ref.read(onboardingCompleteProvider);
   239→      final isOnboardingComplete = onboardingAsync.when(
   240→        data: (complete) => complete,
   241→        loading: () => true, // Assume complete while loading to avoid flicker.
   242→        error: (_, __) => true,
   243→      );
   244→
   245→      final onboardingRedirect = OnboardingGuard.evaluate(
   246→        isAuthenticated: isAuthenticated,
   247→        isOnboardingComplete: isOnboardingComplete,
   248→        currentPath: currentPath,
   249→      );
   250→      if (onboardingRedirect != null) return onboardingRedirect;
   251→
   252→      // ── Role-Based Guard ─────────────────────────────────────
   253→      final roleAsync = ref.read(userRoleProvider);
   254→      final userRole = roleAsync.when(
   255→        data: UserRole.fromString,
   256→        loading: () => null,
   257→        error: (_, __) => null,
   258→      );
   259→
   260→      final roleRedirect = RoleBasedGuard.evaluate(
   261→        userRole: userRole,
   262→        currentPath: currentPath,
   263→      );
   264→      if (roleRedirect != null) return roleRedirect;
   265→
   266→      // No redirect needed — proceed as requested.
   267→      return null;
   268→    },
   269→

   271→    routes: [
   272→      // ── Splash ────────────────────────────────────────────────
   273→      GoRoute(
   274→        path: RouteNames.splash,
   275→        name: 'splash',
   276→        builder: (context, state) => const SplashPage(),
   277→      ),
   278→
   279→      // ── Onboarding ───────────────────────────────────────────
   280→      GoRoute(

  1398→
  1399→/// Shown when the user navigates to a route that does not exist in
  1400→/// the route tree.
  1401→///
  1402→/// Provides a friendly message and a button to navigate back to the
  1403→/// dashboard (or login if unauthenticated).
  1404→class NotFoundPage extends ConsumerWidget {
  1405→  const NotFoundPage({
  1406→    required this.error,
  1407→    this.path,
  1408→    super.key,
  1409→  });
  1410→
  1411→  /// The [GoException] from GoRouter describing the error.
  1412→  final GoException? error;
  1413→
  1414→  /// The unmatched path the user attempted to visit.
  1415→  final String? path;
  1416→
  1417→  @override
  1418→  Widget build(BuildContext context, WidgetRef ref) {
  1419→    final theme = Theme.of(context);
  1420→    final isAuthenticated = ref.watch(isAuthenticatedProvider);
  1421→
  1422→    return Scaffold(
  1423→      body: Center(
  1424→        child: Padding(
  1425→          padding: const EdgeInsets.all(32),
  1426→          child: Column(
  1427→            mainAxisSize: MainAxisSize.min,
  1428→            children: [
  1429→              Icon(
  1430→                Icons.search_off_rounded,
  1431→                size: 80,
  1432→                color: theme.colorScheme.error,
  1433→              ),
  1434→              const SizedBox(height: 24),
  1435→              Text(
  1436→                'Page Not Found',
  1437→                style: theme.textTheme.headlineMedium?.copyWith(
  1438→                  fontWeight: FontWeight.bold,
  1439→                ),
  1440→                textAlign: TextAlign.center,
  1441→              ),
  1442→              const SizedBox(height: 12),
  1443→              if (path != null)
  1444→                Text(
  1445→                  'The page "$path" does not exist.',
  1446→                  style: theme.textTheme.bodyLarge?.copyWith(
  1447→                    color: theme.colorScheme.onSurfaceVariant,
  1448→                  ),
  1449→                  textAlign: TextAlign.center,
  1450→                ),
  1451→              if (error != null) ...[
  1452→                const SizedBox(height: 8),
  1453→                Text(
  1454→                  error!.message,
  1455→                  style: theme.textTheme.bodySmall?.copyWith(
  1456→                    color: theme.colorScheme.outline,
  1457→                  ),
  1458→                  textAlign: TextAlign.center,
  1459→                ),
  1460→              ],
  1461→              const SizedBox(height: 32),
  1462→              FilledButton.icon(
  1463→                onPressed: () => context.go(
  1464→                  isAuthenticated
  1465→                      ? RouteNames.dashboard
  1466→                      : RouteNames.login,
  1467→                ),
  1468→                icon: const Icon(Icons.home),
  1469→                label: Text(
  1470→                  isAuthenticated ? 'Go to Dashboard' : 'Go to Login',
  1471→                ),
  1472→              ),
  1473→            ],
  1474→          ),
  1475→        ),
  1476→      ),
  1477→    );
  1478→  }
