   220→   220→  @override
   221→   221→  Future<GradeScaleModel> createGradeScale(GradeScaleModel scale) async {
   222→   222→    try {
   223→   223→      // 1. Insert the grade_scale row (without nested entries)
   224→   224→      final scaleJson = scale.toJson()
   225→   225→        ..remove('scale_entries');
   226→   226→
   227→   227→      final response = await _supabaseClient
   228→   228→          .from(_gradeScalesTable)
   229→   229→          .insert(scaleJson)
   230→   230→          .select()
   231→   231→          .single();
   232→   232→
   233→   233→      final createdScale = GradeScaleModel.fromJson(response);
   234→   234→
   235→   235→      // 2. Insert entries
   236→   236→      if (scale.scaleEntries.isNotEmpty) {
   237→   237→        final entriesRows = scale.scaleEntries.map((e) {
   238→   238→          final json = e.toJson();
   239→   239→          json['grade_scale_id'] = createdScale.id;
   240→   240→          return json;
   241→   241→        }).toList();
   242→   242→
   243→   243→        await _supabaseClient
   244→   244→            .from(_gradeScaleEntriesTable)
   245→   245→            .insert(entriesRows);
   246→   246→      }
   247→   247→
   248→   248→      // 3. Return scale with entries populated
   249→   249→      return createdScale.copyWith(
   250→   250→        scaleEntries: scale.scaleEntries,
   251→   251→      );
   252→   252→    } on sb.PostgrestException catch (e) {
   253→   253→      AppLogger.error('Create grade scale failed', error: e);
   254→   254→      throw ServerException(
   255→   255→        message: e.message,
   256→   256→        statusCode: int.tryParse(e.code ?? '') ?? 500,
   257→   257→        data: e.details,
   258→   258→      );
   259→   259→    } catch (e) {
   260→   260→      AppLogger.error('Create grade scale unexpected error', error: e);
   261→   261→      throw ServerException(
   262→   262→        message: 'Failed to create grade scale: $e',
   263→   263→        statusCode: 500,
   264→   264→      );
   265→   265→    }
   266→   266→  }

   268→   268→  @override
   269→   269→  Future<GradeScaleModel> updateGradeScale(GradeScaleModel scale) async {
   270→   270→    try {
   271→   271→      // 1. Update the grade_scale row
   272→   272→      final scaleJson = scale.toJson()
   273→   273→        ..remove('scale_entries');
   274→   274→
   275→   275→      final response = await _supabaseClient
   276→   276→          .from(_gradeScalesTable)
   277→   277→          .update(scaleJson)
   278→   278→          .eq('id', scale.id)
   279→   279→          .select()
   280→   280→          .single();
   281→   281→
   282→   282→      final updatedScale = GradeScaleModel.fromJson(response);
   283→   283→
   284→   284→      // 2. Delete existing entries and re-insert
   285→   285→      await _supabaseClient
   286→   286→          .from(_gradeScaleEntriesTable)
   287→   287→          .delete()
   288→   288→          .eq('grade_scale_id', scale.id);
   289→   289→
   290→   290→      if (scale.scaleEntries.isNotEmpty) {
   291→   291→        final entriesRows = scale.scaleEntries.map((e) {
   292→   292→          final json = e.toJson();
   293→   293→          json['grade_scale_id'] = scale.id;
   294→   294→          return json;
   295→   295→        }).toList();
   296→   296→
   297→   297→        await _supabaseClient
   298→   298→            .from(_gradeScaleEntriesTable)
   299→   299→            .insert(entriesRows);
   300→   300→      }
   301→   301→
   302→   302→      // 3. Return scale with entries populated
   303→   303→      return updatedScale.copyWith(
   304→   304→        scaleEntries: scale.scaleEntries,
   305→   305→      );
   306→   306→    } on sb.PostgrestException catch (e) {
   307→   307→      AppLogger.error('Update grade scale failed', error: e);
   308→   308→      if (e.code == 'PGRST116') {
   309→   309→        throw NotFoundException(message: 'Grade scale not found: ${scale.id}');
   310→   310→      }
   311→   311→      throw ServerException(
   312→   312→        message: e.message,
   313→   313→        statusCode: int.tryParse(e.code ?? '') ?? 500,
   314→   314→        data: e.details,
   315→   315→      );
   316→   316→    } catch (e) {
   317→   317→      AppLogger.error('Update grade scale unexpected error', error: e);
   318→   318→      throw ServerException(
   319→   319→        message: 'Failed to update grade scale: $e',
   320→   320→        statusCode: 500,
   321→   321→      );
   322→   322→    }
   323→   323→  }

   380→   380→  @override
   381→   381→  Future<List<GradeScaleModel>> getGradeScales(
   382→   382→    String schoolId, {
   383→   383→    bool? isActive,
   384→   384→    String? gradeType,
   385→   385→    int limit = PaginatedQueryMixin.defaultPageSize,
   386→   386→    int offset = 0,
   387→   387→  }) async {
   388→   388→    try {
   389→   389→      // PERF: Added pagination params + limit to grade scales list query
   390→   390→      var filterQuery = _supabaseClient
   391→   391→          .from(_gradeScalesTable)
   392→   392→          .select('*, grade_scale_entries(*)');
   393→   393→
   394→   394→      filterQuery = filterQuery.eq('school_id', schoolId);
   395→   395→
   396→   396→      if (isActive != null) {
   397→   397→        filterQuery = filterQuery.eq('is_active', isActive);
   398→   398→      }
   399→   399→      if (gradeType != null) {
   400→   400→        filterQuery = filterQuery.eq('grade_type', gradeType);
   401→   401→      }
   402→   402→
   403→   403→      final transformQuery = filterQuery.order('created_at', ascending: false).range(offset, offset + limit - 1);
   404→   404→
   405→   405→      final response = await transformQuery;
   406→   406→
   407→   407→      return response.map<GradeScaleModel>((json) {
   408→   408→        final mappedJson = _mapGradeScaleEntries(json);
   409→   409→        return GradeScaleModel.fromJson(mappedJson);
   410→   410→      }).toList();
   411→   411→    } on sb.PostgrestException catch (e) {
   412→   412→      AppLogger.error('Get grade scales failed', error: e);
   413→   413→      throw ServerException(
   414→   414→        message: e.message,
   415→   415→        statusCode: int.tryParse(e.code ?? '') ?? 500,
   416→   416→        data: e.details,
   417→   417→      );
   418→   418→    } catch (e) {
   419→   419→      AppLogger.error('Get grade scales unexpected error', error: e);
   420→   420→      throw ServerException(
   421→   421→        message: 'Failed to get grade scales: $e',
   422→   422→        statusCode: 500,
   423→   423→      );
   424→   424→    }
   425→   425→  }

   427→   427→  /// Maps the nested `grade_scale_entries` join result to the model's
   428→   428→  /// expected `scale_entries` key.
   429→   429→  Map<String, dynamic> _mapGradeScaleEntries(Map<String, dynamic> json) {
   430→   430→    final entries = json['grade_scale_entries'];
   431→   431→    if (entries is List) {
   432→   432→      return {
   433→   433→        ...json,
   434→   434→        'scale_entries': entries,
   435→   435→      };
   436→   436→    }
   437→   437→    return json;
   438→   438→  }
