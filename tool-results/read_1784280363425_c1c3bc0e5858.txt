   438→   438→/// Provides the [AiProvidersRegistry] singleton.
   439→   439→///
   440→   440→/// Registers OpenAI and Gemini providers conditionally based on
   441→   441→/// whether their API keys are available in the environment config.
   442→   442→final aiProviderRegistryProvider = Provider<AiProvidersRegistry>((ref) {
   443→   443→  final openaiApiKey = EnvConfig.maybeGet('OPENAI_API_KEY') ?? '';
   444→   444→  final geminiApiKey = EnvConfig.maybeGet('GEMINI_API_KEY') ?? '';
   445→   445→
   446→   446→  if (openaiApiKey.isEmpty && geminiApiKey.isEmpty) {
   447→   447→    AppLogger.warning(
   448→   448→      'No AI API keys configured — AI features will be unavailable',
   449→   449→    );
   450→   450→    return AiProvidersRegistry.createEmpty();
   451→   451→  }
   452→   452→
   453→   453→  return AiProvidersRegistry.createDefault(
   454→   454→    openaiApiKey: openaiApiKey,
   455→   455→    geminiApiKey: geminiApiKey,
   456→   456→  );
   457→   457→});

   470→   470→/// prompt engine, and validation engine.
   471→   471→final aiServiceProvider = Provider<AiService>((ref) {
   472→   472→  final registry = ref.watch(aiProviderRegistryProvider);
   473→   473→  final promptEngine = ref.watch(promptEngineProvider);
   474→   474→  final validationEngine = ref.watch(validationEngineProvider);
   475→   475→
   476→   476→  return AiService(
   477→   477→    providersRegistry: registry,
   478→   478→    promptEngine: promptEngine,
   479→   479→    validationEngine: validationEngine,
   480→   480→  );
   481→   481→});
