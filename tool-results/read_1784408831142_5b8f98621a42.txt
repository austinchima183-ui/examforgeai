   145→// APP ROUTER PROVIDER
   146→// ═══════════════════════════════════════════════════════════════════════

   148→/// Riverpod provider for the application's [GoRouter].
   149→///
   150→/// Watches [authStateProvider] so that the router automatically
   151→/// re-evaluates its [redirect] callback whenever the authentication
   152→/// state changes (sign-in, sign-out, token refresh).
   153→///
   154→/// Usage in `main.dart`:
   155→/// ```dart
   156→/// final router = ref.read(appRouterProvider);
   157→/// MaterialApp.router(routerConfig: router);
   158→/// ```
   159→final appRouterProvider = Provider<GoRouter>((ref) {
   160→  // Watch auth state so the router re-evaluates redirects on change.
   161→  final authState = ref.watch(authStateProvider);
   162→
   163→  // Also watch onboarding state and role so those redirects stay live.
   164→  ref.watch(onboardingCompleteProvider);
   165→  ref.watch(userRoleProvider);
   166→
   167→  return GoRouter(
   168→    navigatorKey: _rootNavigatorKey,
   169→    initialLocation: RouteNames.splash,
   170→    debugLogDiagnostics: true,
   171→
   172→    // ─── Global Redirect ──────────────────────────────────────────
   173→    redirect: (context, state) {
   174→      final currentPath = state.matchedLocation;
   175→      final isAuthenticated = authState.when(
   176→        data: (authData) => authData.session != null,
   177→        loading: () {
   178→          // While loading, only allow splash. Everything else redirects
   179→          // to splash so we don't flash the login page briefly.
   180→          if (currentPath != RouteNames.splash) {
   181→            return RouteNames.splash;
   182→          }
   183→          return null;
   184→        },
   185→        error: (_, __) => false,
   186→      );
   187→
   188→      // ── Auth Guard ────────────────────────────────────────────
   189→      final authRedirect = AuthGuard.evaluate(
   190→        isAuthenticated: isAuthenticated,
   191→        currentPath: currentPath,
   192→      );
   193→      if (authRedirect != null) return authRedirect;
   194→
   195→      // ── Onboarding Guard ─────────────────────────────────────
   196→      final onboardingAsync = ref.read(onboardingCompleteProvider);
   197→      final isOnboardingComplete = onboardingAsync.when(
   198→        data: (complete) => complete,
   199→        loading: () => true, // Assume complete while loading to avoid flicker.
   200→        error: (_, __) => true,
   201→      );
   202→
   203→      final onboardingRedirect = OnboardingGuard.evaluate(
   204→        isAuthenticated: isAuthenticated,
   205→        isOnboardingComplete: isOnboardingComplete,
   206→        currentPath: currentPath,
   207→      );
   208→      if (onboardingRedirect != null) return onboardingRedirect;
   209→
   210→      // ── Role-Based Guard ─────────────────────────────────────
   211→      final roleAsync = ref.read(userRoleProvider);
   212→      final userRole = roleAsync.when(
   213→        data: UserRole.fromString,
   214→        loading: () => null,
   215→        error: (_, __) => null,
   216→      );
   217→
   218→      final roleRedirect = RoleBasedGuard.evaluate(
   219→        userRole: userRole,
   220→        currentPath: currentPath,
   221→      );
   222→      if (roleRedirect != null) return roleRedirect;
   223→
   224→      // No redirect needed — proceed as requested.
   225→      return null;
   226→    },
