   307→   307→  @override
   308→   308→  Future<List<ExamModel>> getExams(Map<String, dynamic> filters) async {
   309→   309→    try {
   310→   310→      var query = _supabaseClient.from(_examsTable).select();
   311→   311→
   312→   312→      if (filters['school_id'] != null) {
   313→   313→        query = query.eq('school_id', filters['school_id'] as String);
   314→   314→      }
   315→   315→      if (filters['subject_id'] != null) {
   316→   316→        query = query.eq('subject_id', filters['subject_id'] as String);
   317→   317→      }
   318→   318→      if (filters['status'] != null) {
   319→   319→        query = query.eq('status', filters['status'] as String);
   320→   320→      }
   321→   321→      if (filters['class_id'] != null) {
   322→   322→        query = query.eq('class_id', filters['class_id'] as String);
   323→   323→      }
   324→   324→
   325→   325→      final page = filters['page'] as int? ?? 1;
   326→   326→      final perPage = filters['per_page'] as int? ?? filters['perPage'] as int? ?? 20;
   327→   327→      final offset = (page - 1) * perPage;
   328→   328→
   329→   329→      query = query.range(offset, offset + perPage - 1);
   330→   330→      query = query.order('created_at', ascending: false);
   331→   331→
   332→   332→      final response = await query;
   333→   333→
   334→   334→      return response
   335→   335→          .map<ExamModel>((json) => ExamModel.fromJson(json))
   336→   336→          .toList();
   337→   337→    } on sb.PostgrestException catch (e) {
   338→   338→      AppLogger.error('Get exams failed', error: e);
   339→   339→      throw ServerException(
   340→   340→        message: e.message,
   341→   341→        statusCode: int.tryParse(e.code ?? '') ?? 500,
   342→   342→        data: e.details,
   343→   343→      );
   344→   344→    } catch (e) {
   345→   345→      AppLogger.error('Get exams unexpected error', error: e);
   346→   346→      throw ServerException(
   347→   347→        message: 'Failed to get exams: $e',
   348→   348→        statusCode: 500,
   349→   349→      );
   350→   350→    }
   351→   351→  }

   414→   414→  @override
   415→   415→  Future<ExamModel> cloneExam(String examId) async {
   416→   416→    try {
   417→   417→      // Fetch the original exam with details
   418→   418→      final original = await getExamWithDetails(examId);
   419→   419→
   420→   420→      final now = DateTime.now();
   421→   421→      final cloneData = {
   422→   422→        'school_id': original.schoolId,
   423→   423→        'created_by': original.createdBy,
   424→   424→        'title': '${original.title} (Copy)',
   425→   425→        'description': original.description,
   426→   426→        'subject_id': original.subjectId,
   427→   427→        'class_id': original.classId,
   428→   428→        'academic_session_id': original.academicSessionId,
   429→   429→        'exam_type': original.examType,
   430→   430→        'status': 'draft',
   431→   431→        'start_time': now.add(const Duration(days: 7)).toIso8601String(),
   432→   432→        'end_time': now.add(const Duration(days: 7, hours: 2)).toIso8601String(),
   433→   433→        'time_limit_minutes': original.timeLimitMinutes,
   434→   434→        'total_marks': original.totalMarks,
   435→   435→        'pass_mark': original.passMark,
   436→   436→        'pass_mark_type': original.passMarkType,
   437→   437→        'instructions': original.instructions,
   438→   438→        'allowed_attempts': original.allowedAttempts,
   439→   439→        'negative_marking_enabled': original.negativeMarkingEnabled,
   440→   440→        'negative_mark_value': original.negativeMarkValue,
   441→   441→        'grace_period_minutes': original.gracePeriodMinutes,
   442→   442→        'auto_submit': original.autoSubmit,
   443→   443→        'randomize_questions': original.randomizeQuestions,
   444→   444→        'randomize_options': original.randomizeOptions,
   445→   445→        'show_results': original.showResults,
   446→   446→        'show_correct_answers': original.showCorrectAnswers,
   447→   447→        'show_explanations': original.showExplanations,
   448→   448→        'is_template': original.isTemplate,
   449→   449→        'template_id': original.id,
   450→   450→        'max_students': original.maxStudents,
   451→   451→        'ip_restriction': original.ipRestriction,
   452→   452→        'require_full_screen': original.requireFullScreen,
   453→   453→        'allow_resume': original.allowResume,
   454→   454→        'browser_lockdown': original.browserLockdown,
   455→   455→        'metadata': original.metadata,
   456→   456→      };
   457→   457→
   458→   458→      final clonedExam = await createExam(cloneData);
   459→   459→
   460→   460→      // Clone sections
   461→   461→      for (final section in original.sections) {
   462→   462→        await _supabaseClient.from(_examSectionsTable).insert({
   463→   463→          'exam_id': clonedExam.id,
   464→   464→          'title': section.title,
   465→   465→          'description': section.description,
   466→   466→          'instructions': section.instructions,
   467→   467→          'sort_order': section.sortOrder,
   468→   468→          'time_limit_minutes': section.timeLimitMinutes,
   469→   469→          'randomize_questions': section.randomizeQuestions,
   470→   470→        });
   471→   471→      }
   472→   472→
   473→   473→      // Clone questions
   474→   474→      for (final question in original.questions) {
   475→   475→        await _supabaseClient.from(_examQuestionsTable).insert({
   476→   476→          'exam_id': clonedExam.id,
   477→   477→          'section_id': question.sectionId,
   478→   478→          'question_id': question.questionId,
   479→   479→          'sort_order': question.sortOrder,
   480→   480→          'marks': question.marks,
   481→   481→          'negative_marks': question.negativeMarks,
   482→   482→          'is_compulsory': question.isCompulsory,
   483→   483→        });
   484→   484→      }
   485→   485→
   486→   486→      return clonedExam;
   487→   487→    } on ServerException {

   726→   726→  @override
   727→   727→  Future<ExamResultModel> submitAttempt(
   728→   728→    String attemptId, {
   729→   729→    String submissionType = 'manual',
   730→   730→  }) async {
   731→   731→    try {
   732→   732→      final now = DateTime.now();
   733→   733→
   734→   734→      // Fetch the attempt
   735→   735→      final attemptResponse = await _supabaseClient
   736→   736→          .from(_examAttemptsTable)
   737→   737→          .select()
   738→   738→          .eq('id', attemptId)
   739→   739→          .single();
   740→   740→
   741→   741→      final attempt = ExamAttemptModel.fromJson(attemptResponse);
   742→   742→
   743→   743→      // PERF: Select only needed columns instead of all columns
   744→   744→      // Note: Answer fetching for a single attempt is bounded by exam question count
   745→   745→      // (typically 20-100 questions), so no pagination needed.
   746→   746→      final answersResponse = await _supabaseClient
   747→   747→          .from(_studentAnswersTable)
   748→   748→          .select('id, attempt_id, question_id, answer_data, is_correct, marks_awarded, marks_deducted, is_flagged, answered_at, graded_at, graded_by, teacher_comment')
   749→   749→          .eq('attempt_id', attemptId);
   750→   750→
   751→   751→      final answers = answersResponse
   752→   752→          .map<StudentAnswerModel>((a) => StudentAnswerModel.fromJson(a))
   753→   753→          .toList();
   754→   754→
   755→   755→      // Calculate total marks
   756→   756→      double totalMarks = 0;
   757→   757→      for (final answer in answers) {
   758→   758→        totalMarks += answer.marksAwarded - answer.marksDeducted;
   759→   759→      }
   760→   760→
   761→   761→      // Calculate time spent
   762→   762→      final startedAt = attempt.startedAt;
   763→   763→      final timeSpent = now.difference(startedAt).inSeconds;
   764→   764→
   765→   765→      // Determine if all objective questions are graded
   766→   766→      final hasUngradedSubjective = answers.any(
   767→   767→        (a) => a.isCorrect == null && a.marksAwarded == 0,
   768→   768→      );
   769→   769→      final gradingStatus =
   770→   770→          hasUngradedSubjective ? 'partially_graded' : 'auto_graded';
   771→   771→
   772→   772→      // Update the attempt
   773→   773→      await _supabaseClient
   774→   774→          .from(_examAttemptsTable)
   775→   775→          .update({
   776→   776→            'status': submissionType == 'auto_submit' ? 'auto_submitted' : 'submitted',
   777→   777→            'submitted_at': now.toIso8601String(),
   778→   778→            'submission_type': submissionType,
   779→   779→            'time_spent_seconds': timeSpent,
   780→   780→            'total_marks': totalMarks,
   781→   781→            'grading_status': gradingStatus,
   782→   782→            'last_activity_at': now.toIso8601String(),
   783→   783→          })
   784→   784→          .eq('id', attemptId);
   785→   785→
   786→   786→      // Deactivate the session
   787→   787→      await _supabaseClient
   788→   788→          .from(_examSessionsTable)
   789→   789→          .update({
   790→   790→            'is_active': false,
   791→   791→            'connection_status': 'disconnected',
   792→   792→          })
   793→   793→          .eq('attempt_id', attemptId);
   794→   794→
   795→   795→      // Calculate score percentage
   796→   796→      // Fetch exam total marks
   797→   797→      final examResponse = await _supabaseClient
   798→   798→          .from(_examsTable)
   799→   799→          .select('total_marks, pass_mark, pass_mark_type')
   800→   800→          .eq('id', attempt.examId)
   801→   801→          .single();
   802→   802→
   803→   803→      final examTotalMarks =
   804→   804→          (examResponse['total_marks'] as num?)?.toDouble() ?? 0.0;
   805→   805→      final passMark =
   806→   806→          (examResponse['pass_mark'] as num?)?.toDouble() ?? 0.0;
   807→   807→      final passMarkType =
   808→   808→          examResponse['pass_mark_type'] as String? ?? 'percentage';
   809→   809→
   810→   810→      final scorePercentage = examTotalMarks > 0
   811→   811→          ? (totalMarks / examTotalMarks) * 100
   812→   812→          : 0.0;
   813→   813→
   814→   814→      final effectivePassMark = passMarkType == 'percentage'
   815→   815→          ? passMark
   816→   816→          : (examTotalMarks > 0 ? (passMark / examTotalMarks) * 100 : 0.0);
   817→   817→
   818→   818→      final isPassed = scorePercentage >= effectivePassMark;
   819→   819→
   820→   820→      // Create or update result
   821→   821→      final existingResult = await _supabaseClient
   822→   822→          .from(_examResultsTable)
   823→   823→          .select('id')
   824→   824→          .eq('attempt_id', attemptId)
   825→   825→          .maybeSingle();
   826→   826→
   827→   827→      final resultData = {
   828→   828→        'exam_id': attempt.examId,
   829→   829→        'student_id': attempt.studentId,
   830→   830→        'attempt_id': attemptId,
   831→   831→        'total_marks': totalMarks,
   832→   832→        'total_possible': examTotalMarks,
   833→   833→        'score_percentage': scorePercentage,
   834→   834→        'is_passed': isPassed,
   835→   835→        'time_spent_seconds': timeSpent,
   836→   836→        'grading_status': gradingStatus,
   837→   837→        'is_released': false,
   838→   838→      };
   839→   839→
   840→   840→      if (existingResult != null) {
   841→   841→        await _supabaseClient
   842→   842→            .from(_examResultsTable)
   843→   843→            .update(resultData)
   844→   844→            .eq('id', existingResult['id'] as String);
   845→   845→
   846→   846→        final updatedResult = await _supabaseClient
   847→   847→            .from(_examResultsTable)
   848→   848→            .select()
   849→   849→            .eq('id', existingResult['id'] as String)
   850→   850→            .single();
   851→   851→
   852→   852→        return ExamResultModel.fromJson(updatedResult);
   853→   853→      } else {
   854→   854→        final insertResponse = await _supabaseClient
   855→   855→            .from(_examResultsTable)
   856→   856→            .insert(resultData)
   857→   857→            .select()
   858→   858→            .single();
   859→   859→
   860→   860→        return ExamResultModel.fromJson(insertResponse);
   861→   861→      }
   862→   862→    } on sb.PostgrestException catch (e) {
   863→   863→      AppLogger.error('Submit attempt failed', error: e);
   864→   864→      throw ServerException(
   865→   865→        message: e.message,
   866→   866→        statusCode: int.tryParse(e.code ?? '') ?? 500,
   867→   867→        data: e.details,
   868→   868→      );
   869→   869→    } catch (e) {
   870→   870→      if (e is ServerException) rethrow;
   871→   871→      AppLogger.error('Submit attempt unexpected error', error: e);
   872→   872→      throw ServerException(
   873→   873→        message: 'Failed to submit attempt: $e',
   874→   874→        statusCode: 500,
   875→   875→      );
   876→   876→    }
   877→   877→  }

  1390→  1390→      final attempts = await _supabaseClient
  1391→  1391→          .from(_examAttemptsTable)

  1397→  1397→      final startedStudents = attempts

  1399→  1399→          .length;
  1400→  1400→      final completedStudents = attempts

  1402→  1402→              a['status'] == 'submitted' ||

  1407→  1407→

  1409→  1409→      final scores = results
  1410→  1410→          .map<double>((r) => (r['score_percentage'] as num?)?.toDouble() ?? 0.0)

  1412→  1412→

  1416→  1416→      final highestScore = scores.isNotEmpty

  1422→  1422→
  1423→  1423→      // Calculate median
