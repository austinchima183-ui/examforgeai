    13→    13→-- 1. CUSTOM ENUMERATION TYPES
    14→    14→-- ============================================================================
    15→    15→
    16→    16→DO $$
    17→    17→BEGIN
    18→    18→  -- user_role: Defines the hierarchy of user permissions
    19→    19→  IF NOT EXISTS (SELECT 1 FROM pg_type WHERE typname = 'user_role') THEN
    20→    20→    CREATE TYPE user_role AS ENUM (
    21→    21→      'super_admin',
    22→    22→      'school_admin',
    23→    23→      'teacher',
    24→    24→      'student'
    25→    25→    );
    26→    26→  END IF;
    27→    27→
    28→    28→  -- subscription_status: School subscription tier
    29→    29→  IF NOT EXISTS (SELECT 1 FROM pg_type WHERE typname = 'subscription_status') THEN
    30→    30→    CREATE TYPE subscription_status AS ENUM (
    31→    31→      'free',
    32→    32→      'basic',
    33→    33→      'premium',
    34→    34→      'enterprise'
    35→    35→    );
    36→    36→  END IF;
    37→    37→
    38→    38→  -- exam_status: Lifecycle states for an exam
    39→    39→  IF NOT EXISTS (SELECT 1 FROM pg_type WHERE typname = 'exam_status') THEN
    40→    40→    CREATE TYPE exam_status AS ENUM (
    41→    41→      'draft',
    42→    42→      'published',
    43→    43→      'active',
    44→    44→      'completed',
    45→    45→      'archived'
    46→    46→    );
    47→    47→  END IF;
    48→    48→
    49→    49→  -- notification_type: Category of notification
    50→    50→  IF NOT EXISTS (SELECT 1 FROM pg_type WHERE typname = 'notification_type') THEN
    51→    51→    CREATE TYPE notification_type AS ENUM (
    52→    52→      'exam',
    53→    53→      'system',
    54→    54→      'result',
    55→    55→      'reminder'
    56→    56→    );
    57→    57→  END IF;
    58→    58→
    59→    59→  -- question_type: Supported question formats
    60→    60→  IF NOT EXISTS (SELECT 1 FROM pg_type WHERE typname = 'question_type') THEN
    61→    61→    CREATE TYPE question_type AS ENUM (
    62→    62→      'multiple_choice',
    63→    63→      'true_false',
    64→    64→      'short_answer',
    65→    65→      'essay',
    66→    66→      'fill_in_blank'
    67→    67→    );
    68→    68→  END IF;
    69→    69→END
    70→    70→$$;

   294→   294→-- ---------------------------------------------------------------------------
   295→   295→-- update_updated_at_column()
   296→   296→-- Automatically sets updated_at = now() on every row update.
   297→   297→-- ---------------------------------------------------------------------------
   298→   298→CREATE OR REPLACE FUNCTION update_updated_at_column()
   299→   299→RETURNS TRIGGER AS $$
   300→   300→BEGIN
   301→   301→  NEW.updated_at = now();
   302→   302→  RETURN NEW;
   303→   303→END;
   304→   304→$$ LANGUAGE plpgsql;
   305→   305→
   306→   306→COMMENT ON FUNCTION update_updated_at_column() IS
   307→   307→  'Trigger function: automatically refreshes updated_at to the current timestamp';

   309→   309→-- ---------------------------------------------------------------------------
   310→   310→-- is_school_member(school_id)
   311→   311→-- Returns TRUE if the currently authenticated user belongs to the given school.
   312→   312→-- ---------------------------------------------------------------------------
   313→   313→CREATE OR REPLACE FUNCTION is_school_member(target_school_id UUID)
   314→   314→RETURNS BOOLEAN AS $$
   315→   315→BEGIN
   316→   316→  RETURN EXISTS (
   317→   317→    SELECT 1
   318→   318→    FROM users
   319→   319→    WHERE id = auth.uid()
   320→   320→      AND school_id = target_school_id
   321→   321→  );
   322→   322→END;
   323→   323→$$ LANGUAGE plpgsql SECURITY DEFINER STABLE;
   324→   324→
   325→   325→COMMENT ON FUNCTION is_school_member(UUID) IS
   326→   326→  'Checks whether the current auth user is a member of the specified school';
   327→   327→
   328→   328→-- ---------------------------------------------------------------------------
   329→   329→-- get_user_role()
   330→   330→-- Returns the user_role of the currently authenticated user.
   331→   331→-- ---------------------------------------------------------------------------
   332→   332→CREATE OR REPLACE FUNCTION get_user_role()
   333→   333→RETURNS user_role AS $$
   334→   334→DECLARE
   335→   335→  current_role user_role;
   336→   336→BEGIN
   337→   337→  SELECT role INTO current_role
   338→   338→  FROM users
   339→   339→  WHERE id = auth.uid();
   340→   340→
   341→   341→  RETURN current_role;
   342→   342→END;
   343→   343→$$ LANGUAGE plpgsql SECURITY DEFINER STABLE;
   344→   344→
   345→   345→COMMENT ON FUNCTION get_user_role() IS
   346→   346→  'Returns the role of the currently authenticated user';
   347→   347→
   348→   348→-- ---------------------------------------------------------------------------
   349→   349→-- handle_new_user()
   350→   350→-- Trigger function: automatically creates a user profile row in public.users
   351→   351→-- after a new row is inserted into auth.users.
   352→   352→-- ---------------------------------------------------------------------------
   353→   353→CREATE OR REPLACE FUNCTION handle_new_user()
   354→   354→RETURNS TRIGGER AS $$
   355→   355→BEGIN
   356→   356→  INSERT INTO public.users (id, email, full_name, role, is_email_verified)
   357→   357→  VALUES (
   358→   358→    NEW.id,
   359→   359→    NEW.email,
   360→   360→    COALESCE(NEW.raw_user_meta_data ->> 'full_name', NEW.email),
   361→   361→    COALESCE(
   362→   362→      (NEW.raw_user_meta_data ->> 'role')::user_role,
   363→   363→      'student'
   364→   364→    ),
   365→   365→    NEW.email_confirmed_at IS NOT NULL
   366→   366→  );
   367→   367→  RETURN NEW;
   368→   368→END;
   369→   369→$$ LANGUAGE plpgsql SECURITY DEFINER;
   370→   370→
   371→   371→COMMENT ON FUNCTION handle_new_user() IS
   372→   372→  'Trigger: auto-creates a public.users profile when a new auth.users row is inserted';

   438→   438→-- SCHOOLS RLS POLICIES
   439→   439→-- ===========================================================================
   440→   440→
   441→   441→-- Any authenticated user can read active schools
   442→   442→CREATE POLICY "Authenticated users can read active schools"
   443→   443→  ON schools FOR SELECT
   444→   444→  TO authenticated
   445→   445→  USING (is_active = true);
   446→   446→
   447→   447→-- Super admins can read all schools (including inactive)
   448→   448→CREATE POLICY "Super admins can read all schools"
   449→   449→  ON schools FOR SELECT
   450→   450→  TO authenticated
   451→   451→  USING (get_user_role() = 'super_admin');
   452→   452→
   453→   453→-- School admins can update their own school
   454→   454→CREATE POLICY "School admins can update own school"
   455→   455→  ON schools FOR UPDATE
   456→   456→  TO authenticated
   457→   457→  USING (
   458→   458→    role = 'school_admin'
   459→   459→    AND school_id = id
   460→   460→  )
   461→   461→  WITH CHECK (
   462→   462→    role = 'school_admin'
   463→   463→    AND school_id = id
   464→   464→  );
   465→   465→
   466→   466→-- Super admins can do everything on schools
   467→   467→CREATE POLICY "Super admins have full access to schools"
   468→   468→  ON schools FOR ALL
   469→   469→  TO authenticated
   470→   470→  USING (get_user_role() = 'super_admin')
   471→   471→  WITH CHECK (get_user_role() = 'super_admin');
   472→   472→
   473→   473→-- Super admins can insert schools
   474→   474→CREATE POLICY "Super admins can insert schools"
   475→   475→  ON schools FOR INSERT
   476→   476→  TO authenticated
   477→   477→  WITH CHECK (get_user_role() = 'super_admin');
