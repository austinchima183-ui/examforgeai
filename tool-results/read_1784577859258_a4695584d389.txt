   251→/// Provides the Supabase client singleton.
   252→///
   253→/// Depends on [SupabaseConfig.initialize] having been called in `main()`.
   254→final supabaseClientProvider = Provider<sb.SupabaseClient>((ref) {
   255→  try {
   256→    return SupabaseConfig.client;
   257→  } catch (e) {
   258→    AppLogger.error('Failed to resolve Supabase client', error: e);
   259→    rethrow;
   260→  }
   261→});
   262→
   263→/// Provides a configured [Dio] instance with base options pulled from
   264→/// [AppConfig] and [ApiConstants].
   265→final dioProvider = Provider<Dio>((ref) {
   266→  final dio = Dio(
   267→    BaseOptions(
   268→      baseUrl: AppConfig.baseUrl,
   269→      connectTimeout: AppConfig.connectTimeoutDuration,
   270→      receiveTimeout: AppConfig.receiveTimeoutDuration,
   271→      sendTimeout: AppConfig.sendTimeoutDuration,
   272→      headers: ApiConstants.baseHeaders(),
   273→      responseType: ResponseType.json,
   274→      validateStatus: (status) => status != null && status < 500,
   275→    ),
   276→  );
   277→  return dio;
   278→});
   279→
   280→/// Provides the [ApiClient] wrapper around [Dio].
   281→final apiClientProvider = Provider<ApiClient>((ref) {
   282→  final dio = ref.watch(dioProvider);
   283→  return ApiClient(dio);
   284→});
   285→
   286→/// Re-exports the [NetworkInfo] provider from the core layer so that
   287→/// it is accessible through the DI container alongside other services.
   288→final networkInfoProvider = Provider<NetworkInfo>((ref) {
   289→  // Delegate to the core-layer provider.
   290→  return ref.watch(coreNetworkInfoProvider);

   334→/// Stream provider that emits the current Supabase [sb.AuthState].
   335→///
   336→/// Automatically reconnects on transient failures and emits the latest
   337→/// state to any listener.
   338→final authStateProvider = StreamProvider<sb.AuthState>((ref) {
   339→  final supabaseClient = ref.watch(supabaseClientProvider);
   340→  return supabaseClient.auth.onAuthStateChange;
   341→});
   342→
   343→/// Convenience provider that yields the currently authenticated [sb.User],
   344→/// or `null` if not signed in.
   345→final currentUserProvider = Provider<sb.User?>((ref) {
   346→  final authState = ref.watch(authStateProvider);
   347→  return authState.when(
   348→    data: (state) => state.session?.user,
   349→    loading: () => SupabaseConfig.currentUser,
   350→    error: (_, __) => null,
   351→  );
   352→});
   353→
   354→/// Convenience provider that yields `true` when the user is authenticated.
   355→final isAuthenticatedProvider = Provider<bool>((ref) {
