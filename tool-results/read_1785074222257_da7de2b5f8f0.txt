   201→   201→    } on sb.PostgrestException catch (e) {

   204→   204→      rethrow;

   294→   294→  @override
   295→   295→  Future<Map<String, dynamic>> getQuestionWithDetails(
   296→   296→    String questionId,
   297→   297→  ) async {
   298→   298→    try {
   299→   299→      // Fetch the base question
   300→   300→      final questionResponse = await _supabase
   301→   301→          .from(_questionsTable)
   302→   302→          .select()
   303→   303→          .eq('id', questionId)
   304→   304→          .limit(1);
   305→   305→
   306→   306→      if (questionResponse.isEmpty) {
   307→   307→        throw const NotFoundException(message: 'Question not found.');
   308→   308→      }
   309→   309→
   310→   310→      final questionJson = Map<String, dynamic>.from(questionResponse.first);
   311→   311→
   312→   312→      // Fetch related data in parallel
   313→   313→      final results = await Future.wait([
   314→   314→        _supabase
   315→   315→            .from(_answerOptionsTable)
   316→   316→            .select()
   317→   317→            .eq('question_id', questionId)
   318→   318→            .order('sort_order'),
   319→   319→        _supabase
   320→   320→            .from(_matchingPairsTable)
   321→   321→            .select()
   322→   322→            .eq('question_id', questionId)
   323→   323→            .order('sort_order'),
   324→   324→        _supabase
   325→   325→            .from(_orderingItemsTable)
   326→   326→            .select()
   327→   327→            .eq('question_id', questionId)
   328→   328→            .order('correct_position'),
   329→   329→        _supabase
   330→   330→            .from(_fillInBlankTable)
   331→   331→            .select()
   332→   332→            .eq('question_id', questionId)
   333→   333→            .order('blank_index'),
   334→   334→        _supabase
   335→   335→            .from(_attachmentsTable)
   336→   336→            .select()
   337→   337→            .eq('question_id', questionId)
   338→   338→            .order('sort_order'),
   339→   339→        // Tags need a join through the relation table
   340→   340→        _supabase
   341→   341→            .from(_tagRelationsTable)
   342→   342→            .select('question_tags(*)')
   343→   343→            .eq('question_id', questionId),
   344→   344→      ]);
   345→   345→
   346→   346→      questionJson['answer_options'] = results[0] as List;
   347→   347→      questionJson['matching_pairs'] = results[1] as List;
   348→   348→      questionJson['ordering_items'] = results[2] as List;
   349→   349→      questionJson['fill_in_blank_answers'] = results[3] as List;
   350→   350→      questionJson['question_attachments'] = results[4] as List;
   351→   351→
   352→   352→      // Flatten tag join results
   353→   353→      final tagRows = results[5] as List;
   354→   354→      final tagList = tagRows
   355→   355→          .map((row) => row['question_tags'] as Map<String, dynamic>?)
   356→   356→          .whereType<Map<String, dynamic>>()
   357→   357→          .toList();
   358→   358→      questionJson['question_tags'] = tagList;
   359→   359→
   360→   360→      return questionJson;
   361→   361→    } on sb.PostgrestException catch (e) {
   362→   362→      throw _mapPostgrestException(e);
   363→   363→    } on NotFoundException {
   364→   364→      rethrow;
   365→   365→    } catch (e) {
   366→   366→      AppLogger.error('Unexpected getQuestionWithDetails error', error: e);
   367→   367→      throw const ServerException(
   368→   368→        message: 'Failed to retrieve question details.',
   369→   369→        statusCode: 500,
   370→   370→      );
   371→   371→    }
   372→   372→  }

   376→   376→    Map<String, dynamic> filters,
   377→   377→  ) async {
   378→   378→    try {
   379→   379→      final filterQuery = _applyFilters(_supabase.from(_questionsTable).select(), filters);
   380→   380→      var transformQuery = _applySorting(filterQuery, filters['sort_by'] as String?);
   381→   381→      transformQuery = _applyPagination(
   382→   382→        transformQuery,
   383→   383→        page: filters['page'] as int?,

   385→   385→      );
   386→   386→
   387→   387→      final response = await transformQuery;
   388→   388→      return response
   389→   389→          .map((json) => QuestionModel.fromJson(json as Map<String, dynamic>))
   390→   390→          .toList();
   391→   391→    } on sb.PostgrestException catch (e) {
   392→   392→      throw _mapPostgrestException(e);
   393→   393→    } catch (e) {
   394→   394→      AppLogger.error('Unexpected getQuestions error', error: e);

   498→   498→  @override
   499→   499→  Future<QuestionModel> duplicateQuestion(String questionId) async {
   500→   500→    try {
   501→   501→      // Fetch the original question
   502→   502→      final original = await getQuestion(questionId);
   503→   503→
   504→   504→      // Prepare the duplicate data (strip ID, reset version)
   505→   505→      final duplicateData = original.toJson()
   506→   506→        ..remove('id')
   507→   507→        ..remove('created_at')
   508→   508→        ..remove('updated_at')

  1330→  1330→  Future<QuestionBankStatsModel> getStats({String? schoolId}) async {
  1331→  1331→    try {

  1333→  1333→      var baseQuery = _supabase.from(_questionsTable).select();
  1334→  1334→
  1335→  1335→      if (schoolId != null) {
  1336→  1336→        baseQuery = baseQuery.eq('school_id', schoolId);
  1337→  1337→      }
  1338→  1338→
  1339→  1339→      // Fetch all questions for stats computation
  1340→  1340→      final allQuestions = await baseQuery;
  1341→  1341→
  1342→  1342→      final totalQuestions = allQuestions.length;
  1343→  1343→      final publishedQuestions = allQuestions
  1344→  1344→          .where((q) => q['is_published'] == true)
  1345→  1345→          .length;
  1346→  1346→      final draftQuestions = allQuestions
  1347→  1347→          .where((q) => q['is_published'] == false && q['is_archived'] == false)
  1348→  1348→          .length;
  1349→  1349→      final archivedQuestions = allQuestions
  1350→  1350→          .where((q) => q['is_archived'] == true)
  1351→  1351→          .length;
  1352→  1352→
  1353→  1353→      // Compute breakdowns
  1354→  1354→      final Map<String, int> bySubject = {};
  1355→  1355→      final Map<String, int> byDifficulty = {};
  1356→  1356→      final Map<String, int> byType = {};
  1357→  1357→      final Map<String, int> byExamType = {};
  1358→  1358→
  1359→  1359→      for (final q in allQuestions) {
  1360→  1360→        final subjectId = q['subject_id'] as String?;
  1361→  1361→        if (subjectId != null) {
  1362→  1362→          bySubject[subjectId] = (bySubject[subjectId] ?? 0) + 1;
  1363→  1363→        }
  1364→  1364→
  1365→  1365→        final diff = q['difficulty'] as String?;
  1366→  1366→        if (diff != null) {
  1367→  1367→          byDifficulty[diff] = (byDifficulty[diff] ?? 0) + 1;
  1368→  1368→        }
  1369→  1369→
  1370→  1370→        final type = q['question_type'] as String?;
  1371→  1371→        if (type != null) {
  1372→  1372→          byType[type] = (byType[type] ?? 0) + 1;
  1373→  1373→        }
  1374→  1374→
  1375→  1375→        final exam = q['exam_type'] as String?;
  1376→  1376→        if (exam != null) {
  1377→  1377→          byExamType[exam] = (byExamType[exam] ?? 0) + 1;
  1378→  1378→        }
  1379→  1379→      }
  1380→  1380→
  1381→  1381→      // Recent questions (last 7 days)
  1382→  1382→      final weekAgo = DateTime.now().subtract(const Duration(days: 7));
  1383→  1383→      final recentQuestions = allQuestions

  1385→  1385→            final createdAt = q['created_at'] as String?;

  1388→  1388→          })

  1391→  1391→      // Collection count

  1393→  1393→          _supabase.from(_collectionsTable).select('id');

  1395→  1395→        collectionQuery = collectionQuery.eq('school_id', schoolId);
  1396→  1396→      }
  1397→  1397→      final collections = await collectionQuery;
  1398→  1398→

  1400→  1400→      final userId = _currentUserId;
  1401→  1401→      int totalFavorites = 0;
  1402→  1402→      if (userId != null) {
  1403→  1403→        final favorites = await _supabase
  1404→  1404→            .from(_favoritesTable)
  1405→  1405→            .select('id')
  1406→  1406→            .eq('user_id', userId);
  1407→  1407→        totalFavorites = favorites.length;
  1408→  1408→      }
  1409→  1409→
  1410→  1410→      // Most used questions (top 5)
  1411→  1411→      final sortedByUsage = List<Map<String, dynamic>>.from(allQuestions)
  1412→  1412→        ..sort((a, b) => (b['usage_count'] as int? ?? 0)
  1413→  1413→            .compareTo(a['usage_count'] as int? ?? 0),);
  1414→  1414→      final mostUsed = sortedByUsage
  1415→  1415→          .take(5)
  1416→  1416→          .map((q) => QuestionModel.fromJson(q))
  1417→  1417→          .toList();
  1418→  1418→
  1419→  1419→      return QuestionBankStatsModel(
  1420→  1420→        totalQuestions: totalQuestions,
  1421→  1421→        publishedQuestions: publishedQuestions,
  1422→  1422→        draftQuestions: draftQuestions,
  1423→  1423→        archivedQuestions: archivedQuestions,
  1424→  1424→        questionsBySubject: bySubject,
  1425→  1425→        questionsByDifficulty: byDifficulty,
  1426→  1426→        questionsByType: byType,
  1427→  1427→        questionsByExamType: byExamType,
  1428→  1428→        recentQuestions: recentQuestions,
  1429→  1429→        totalCollections: collections.length,
  1430→  1430→        totalFavorites: totalFavorites,
  1431→  1431→        mostUsedQuestions: mostUsed,
  1432→  1432→      );
  1433→  1433→    } on sb.PostgrestException catch (e) {
  1434→  1434→      throw _mapPostgrestException(e);
  1435→  1435→    } catch (e) {
  1436→  1436→      AppLogger.error('Unexpected getStats error', error: e);
