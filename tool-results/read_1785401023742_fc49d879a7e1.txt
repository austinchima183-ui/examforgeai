    14→CREATE OR REPLACE FUNCTION public.create_enum_if_not_exists(
    15→  enum_name TEXT,
    16→  enum_values TEXT[]
    17→) RETURNS VOID AS $$
    18→BEGIN
    19→  IF NOT EXISTS (SELECT 1 FROM pg_type WHERE typname = enum_name) THEN
    20→    EXECUTE format('CREATE TYPE public.%I AS ENUM (%s)',
    21→      enum_name,
    22→      string_agg(quote_literal(v), ',' ) )
    23→    FROM unnest(enum_values) AS v;
    24→  END IF;
    25→END;
    26→$$ LANGUAGE plpgsql;

  1610→-- Get educational levels for a school
  1611→CREATE OR REPLACE FUNCTION public.get_school_levels(p_school_id UUID)
  1612→RETURNS TABLE (
  1613→  id UUID, code TEXT, name TEXT, level_category level_category_type,
  1614→  level_order INT, min_age INT, max_age INT, description TEXT,
  1615→  is_enabled BOOLEAN, custom_name TEXT
  1616→) AS $$
  1617→SELECT el.id, el.code, el.name, el.level_category, el.level_order,
  1618→       el.min_age, el.max_age, el.description,
  1619→       COALESCE(slc.is_enabled, false) AS is_enabled,
  1620→       slc.custom_name
  1621→FROM public.educational_levels el
  1622→LEFT JOIN public.school_level_configurations slc
  1623→  ON slc.educational_level_id = el.id AND slc.school_id = p_school_id
  1624→WHERE el.is_active = true
  1625→ORDER BY el.level_order;
  1626→$$ LANGUAGE sql STABLE;

  1648→-- Get content items with full details
  1649→CREATE OR REPLACE FUNCTION public.get_content_with_details(
  1650→  p_subject_id UUID DEFAULT NULL,
  1651→  p_educational_level_id UUID DEFAULT NULL,
  1652→  p_topic_id UUID DEFAULT NULL,
  1653→  p_content_type content_type_enum DEFAULT NULL,
  1654→  p_difficulty_level difficulty_level_type DEFAULT NULL,
  1655→  p_status content_status_type DEFAULT NULL,
  1656→  p_limit INT DEFAULT 50,
  1657→  p_offset INT DEFAULT 0
  1658→)
  1659→RETURNS TABLE (
  1660→  id UUID, title TEXT, content_type content_type_enum,
  1661→  subject_name TEXT, level_name TEXT, topic_title TEXT,
  1662→  difficulty_level difficulty_level_type, bloom_level bloom_taxonomy_type,
  1663→  status content_status_type, version INT, usage_count INT,
  1664→  average_quality_score DECIMAL, created_at TIMESTAMPTZ
  1665→) AS $$
  1666→SELECT ci.id, ci.title, ci.content_type,
  1667→       s.name AS subject_name,
  1668→       el.name AS level_name,
  1669→       t.title AS topic_title,
  1670→       ci.difficulty_level, ci.bloom_level,
  1671→       ci.status, ci.version, ci.usage_count,
  1672→       ci.average_quality_score, ci.created_at
  1673→FROM public.content_items ci
  1674→LEFT JOIN public.subjects s ON s.id = ci.subject_id
  1675→LEFT JOIN public.educational_levels el ON el.id = ci.educational_level_id
  1676→LEFT JOIN public.topics t ON t.id = ci.topic_id
  1677→WHERE (p_subject_id IS NULL OR ci.subject_id = p_subject_id)
  1678→  AND (p_educational_level_id IS NULL OR ci.educational_level_id = p_educational_level_id)
  1679→  AND (p_topic_id IS NULL OR ci.topic_id = p_topic_id)
  1680→  AND (p_content_type IS NULL OR ci.content_type = p_content_type)
  1681→  AND (p_difficulty_level IS NULL OR ci.difficulty_level = p_difficulty_level)
  1682→  AND (p_status IS NULL OR ci.status = p_status)
  1683→ORDER BY ci.created_at DESC
  1684→LIMIT p_limit OFFSET p_offset;
  1685→$$ LANGUAGE sql STABLE;

  1713→-- Update content quality score
  1714→CREATE OR REPLACE FUNCTION public.update_content_quality_score()
  1715→RETURNS TRIGGER AS $$
  1716→BEGIN
  1717→  UPDATE public.content_items
  1718→  SET average_quality_score = (
  1719→    SELECT ROUND(AVG((quality_score + accuracy_score + relevance_score + curriculum_alignment_score) / 4.0), 2)

  1823→-- Check rate limit
  1824→CREATE OR REPLACE FUNCTION public.check_rate_limit(
  1825→  p_scope rate_limit_scope_type,
  1826→  p_identifier TEXT,
  1827→  p_endpoint TEXT DEFAULT '*'
  1828→)
  1829→RETURNS TABLE (allowed BOOLEAN, remaining INT, reset_at TIMESTAMPTZ) AS $$
  1830→DECLARE
  1831→  v_config RECORD;
  1832→  v_count INT;
  1833→  v_window_start TIMESTAMPTZ;
  1834→BEGIN
  1835→  SELECT * INTO v_config FROM public.rate_limit_configs
  1836→  WHERE scope = p_scope AND identifier = p_identifier
  1837→    AND (endpoint_pattern = '*' OR endpoint_pattern = p_endpoint OR p_endpoint LIKE endpoint_pattern)
  1838→    AND is_active = true
  1839→  ORDER BY endpoint_pattern DESC LIMIT 1;
  1840→
  1841→  IF NOT FOUND THEN
  1842→    RETURN QUERY SELECT true AS allowed, 999999 AS remaining, now() + interval '1 minute' AS reset_at;
  1843→    RETURN;
  1844→  END IF;
  1845→
  1846→  v_window_start := date_trunc('second', now() - (v_config.window_seconds || ' seconds')::interval);
  1847→
  1848→  SELECT COALESCE(SUM(request_count), 0) INTO v_count
  1849→  FROM public.rate_limit_counters
  1850→  WHERE scope = p_scope AND identifier = p_identifier
  1851→    AND endpoint = p_endpoint
  1852→    AND window_start >= v_window_start;
  1853→
  1854→  IF v_count >= v_config.max_requests THEN
  1855→    RETURN QUERY SELECT false AS allowed, 0 AS remaining, v_window_start + (v_config.window_seconds || ' seconds')::interval AS reset_at;
  1856→  ELSE
  1857→    INSERT INTO public.rate_limit_counters (scope, identifier, endpoint, request_count, window_start)
  1858→    VALUES (p_scope, p_identifier, p_endpoint, 1, date_trunc('minute', now()))
  1859→    ON CONFLICT (scope, identifier, endpoint, window_start)
  1860→    DO UPDATE SET request_count = rate_limit_counters.request_count + 1;
  1861→
  1862→    RETURN QUERY SELECT true AS allowed, v_config.max_requests - v_count - 1 AS remaining, v_window_start + (v_config.window_seconds || ' seconds')::interval AS reset_at;
  1863→  END IF;
  1864→END;
  1865→$$ LANGUAGE plpgsql SECURITY DEFINER;
