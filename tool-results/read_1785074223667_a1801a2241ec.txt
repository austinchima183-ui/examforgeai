   184→   184→    switch (e.code) {

   188→   188→        throw const ServerException(
   189→   189→          message: 'A record with this data already exists.',
   190→   190→          statusCode: 409,
   191→   191→        );

   193→   193→        throw const ServerException(
   194→   194→          message: 'Referenced record not found.',
   195→   195→          statusCode: 404,
   196→   196→        );
   197→   197→      case '42501':
   198→   198→        throw const ForbiddenException(message: 'You do not have permission for this action.');
   199→   199→      default:

   201→   201→          message: e.message,

   309→   309→  ) async {

   311→   311→      AppLogger.info('Fetching lesson plans with filters: $filters');
   312→   312→      var query = _supabase.from(_lessonPlansTable).select();
   313→   313→
   314→   314→      // Search filter
   315→   315→      if (filters['searchQuery'] != null) {
   316→   316→        query = query.ilike('title', '%${filters['searchQuery']}%');
   317→   317→      }
   318→   318→
   319→   319→      // Subject filter
   320→   320→      if (filters['subject_id'] != null) {
   321→   321→        query = query.eq('subject_id', filters['subject_id']);
   322→   322→      }
   323→   323→
   324→   324→      // Class filter
   325→   325→      if (filters['class_id'] != null) {
   326→   326→        query = query.eq('class_id', filters['class_id']);
   327→   327→      }
   328→   328→
   329→   329→      // Published filter
   330→   330→      if (filters['is_published'] != null) {
   331→   331→        query = query.eq('is_published', filters['is_published']);
   332→   332→      }
   333→   333→
   334→   334→      // Archived filter
   335→   335→      if (filters['is_archived'] != null) {
   336→   336→        query = query.eq('is_archived', filters['is_archived']);
   337→   337→      }
   338→   338→
   339→   339→      // Tags overlap filter
   340→   340→      if (filters['tags'] != null) {
   341→   341→        query = query.overlaps('tags', filters['tags']);
   342→   342→      }
   343→   343→
   344→   344→      final page = filters['page'] as int? ?? 1;
   345→   345→      final perPage = filters['perPage'] as int? ?? 20;
   346→   346→      final response = await query.order('created_at', ascending: false).range((page - 1) * perPage, page * perPage - 1);
   347→   347→      AppLogger.info('Fetched ${response.length} lesson plans');
   348→   348→      return response.map(LessonPlanModel.fromJson).toList();
   349→   349→    } on sb.PostgrestException catch (e) {
   350→   350→      _mapPostgrestException(e);
   351→   351→    } catch (e) {
   352→   352→      AppLogger.error('Failed to fetch lesson plans', error: e);
   353→   353→      throw ServerException(message: e.toString(), statusCode: 500);
   354→   354→    }
