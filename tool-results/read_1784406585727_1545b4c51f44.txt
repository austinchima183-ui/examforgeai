    21→
    22→/// The enhanced Teacher Workspace Dashboard page.
    23→///
    24→/// This is a significantly upgraded version of the standard dashboard,
    25→/// presenting a modern SaaS-style layout with:
    26→/// - Welcome section with time-of-day greeting and quick stats
    27→/// - AI Quick Actions grid (12 action buttons)
    28→/// - Today's Timetable
    29→/// - Pending Tasks & Assignments
    30→/// - Teaching Statistics
    31→/// - Recent Documents
    32→/// - Saved Templates
    33→/// - Upcoming Events
    34→/// - Notifications
    35→/// - Recently Generated AI Content
    36→///
    37→/// Uses [ConsumerStatefulWidget] with a private [_State] class pattern,
    38→/// loads data in [initState] via [WidgetsBinding.instance.addPostFrameCallback],
    39→/// and renders a responsive layout using [CustomScrollView] with slivers.
    40→class EnhancedWorkspaceDashboardPage extends ConsumerStatefulWidget {
    41→  const EnhancedWorkspaceDashboardPage({super.key});
    42→
    43→  @override
    44→  ConsumerState<EnhancedWorkspaceDashboardPage> createState() => _State();
    45→}
    46→
    47→class _State extends ConsumerState<EnhancedWorkspaceDashboardPage> {
    48→  // ─── Lifecycle ──────────────────────────────────────────────────────
    49→
    50→  @override
    51→  void initState() {
    52→    super.initState();
    53→    WidgetsBinding.instance.addPostFrameCallback((_) {
    54→      ref.read(enhancedDashboardProvider.notifier).loadDashboard();
    55→    });
    56→  }
    57→
    58→  // ─── Build ──────────────────────────────────────────────────────────
    59→
    60→  @override
    61→  Widget build(BuildContext context) {
    62→    final dashboardState = ref.watch(enhancedDashboardProvider);
    63→
    64→    return Scaffold(
    65→      appBar: const AppAppBar(title: 'Workspace'),
    66→      body: _buildBody(context, dashboardState),
    67→    );
    68→  }
    69→
    70→  // ─── Body Router ────────────────────────────────────────────────────
    71→
    72→  Widget _buildBody(BuildContext context, EnhancedDashboardState state) {
    73→    // Loading state
    74→    if (state.isLoading && state.dashboard == null) {
    75→      return _buildShimmerLoading(context);
    76→    }
    77→
    78→    // Error state
    79→    if (state.error != null && state.dashboard == null) {
    80→      return AppErrorState.genericError(
    81→        message: state.error,
    82→        onRetry: () =>
    83→            ref.read(enhancedDashboardProvider.notifier).loadDashboard(),
    84→      );
    85→    }
    86→
    87→    // Empty state
    88→    final dashboard = state.dashboard;
    89→    if (dashboard == null) {
    90→      return AppEmptyState.noData(
    91→        title: 'No Workspace Data',
    92→        subtitle: 'Your workspace dashboard will appear here once loaded.',
    93→        actionLabel: 'Refresh',
    94→        onAction: () =>
    95→            ref.read(enhancedDashboardProvider.notifier).loadDashboard(),
    96→      );
    97→    }
    98→
    99→    // Success — render dashboard
   100→    return RefreshIndicator(
   101→      onRefresh: () =>
   102→          ref.read(enhancedDashboardProvider.notifier).refreshDashboard(),
   103→      child: CustomScrollView(
   104→        physics: const AlwaysScrollableScrollPhysics(),
   105→        slivers: [
   106→          // ─── Welcome Section ──────────────────────────────────────
   107→          SliverToBoxAdapter(child: _buildWelcomeSection(context, dashboard)),
   108→
   109→          // ─── AI Quick Actions ─────────────────────────────────────
   110→          SliverToBoxAdapter(child: _buildQuickActionsSection(context)),
   111→
   112→          // ─── Today's Timetable ────────────────────────────────────
   113→          SliverToBoxAdapter(
   114→            child: _buildTodayTimetable(context, dashboard),
   115→          ),
   116→
   117→          // ─── Pending Tasks & Assignments ──────────────────────────
   118→          SliverToBoxAdapter(
   119→            child: _buildPendingTasks(context, dashboard),
   120→          ),
   121→
   122→          // ─── Teaching Statistics ──────────────────────────────────
   123→          SliverToBoxAdapter(
   124→            child: _buildTeachingStatistics(context, dashboard),
   125→          ),
   126→
   127→          // ─── Recent Documents ─────────────────────────────────────
   128→          SliverToBoxAdapter(
   129→            child: _buildRecentDocuments(context, dashboard),
   130→          ),
   131→
   132→          // ─── Saved Templates ─────────────────────────────────────
   133→          SliverToBoxAdapter(
   134→            child: _buildSavedTemplates(context, dashboard),
   135→          ),
   136→
   137→          // ─── Upcoming Events ─────────────────────────────────────
   138→          SliverToBoxAdapter(
   139→            child: _buildUpcomingEvents(context, dashboard),
   140→          ),
   141→
   142→          // ─── Notifications ────────────────────────────────────────
   143→          SliverToBoxAdapter(
   144→            child: _buildNotifications(context, dashboard),
   145→          ),
   146→
   147→          // ─── Recently Generated AI Content ────────────────────────
   148→          SliverToBoxAdapter(
   149→            child: _buildRecentAiContent(context, dashboard),
   150→          ),
   151→
   152→          // Bottom padding
   153→          const SliverPadding(padding: EdgeInsets.only(bottom: Spacings.xxl)),
   154→        ],
   155→      ),
   156→    );
   157→  }

  1233→  /// Formats a [DateTime] as a short time string (e.g. "2:30 PM").
  1234→  String _formatTime(DateTime dt) {
  1235→    final hour = dt.hour > 12 ? dt.hour - 12 : dt.hour;
  1236→    final amPm = dt.hour >= 12 ? 'PM' : 'AM';
  1237→    final minute = dt.minute.toString().padLeft(2, '0');
  1238→    return '$hour:$minute $amPm';
  1239→  }
  1240→
  1241→  /// Formats a [DateTime] relative to now (e.g. "2h ago", "3d ago").
  1242→  String _formatRelativeTime(DateTime dt) {
  1243→    final diff = DateTime.now().difference(dt);
  1244→    if (diff.inMinutes < 1) return 'Just now';
  1245→    if (diff.inMinutes < 60) return '${diff.inMinutes}m ago';
  1246→    if (diff.inHours < 24) return '${diff.inHours}h ago';
  1247→    return '${diff.inDays}d ago';
  1248→  }
