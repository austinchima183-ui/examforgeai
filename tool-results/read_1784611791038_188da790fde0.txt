    35→    35→// PROMPT ENGINE
    36→    36→// ═══════════════════════════════════════════════════════════════════════
    37→    37→
    38→    38→/// The Prompt Engine is the brain of the AI question generation system.
    39→    39→///
    40→    40→/// It is responsible for:
    41→    41→/// - Resolving prompt templates with input variables
    42→    42→/// - Selecting the best template for a given generation request
    43→    43→/// - Building generation, improvement, validation, and extraction prompts
    44→    44→/// - Parsing AI responses into structured domain entities
    45→    45→class PromptEngine {
    46→    46→  // ─── Template Resolution ──────────────────────────────────────────
    47→    47→
    48→    48→  /// Resolve a prompt template by replacing `{{variable_name}}` placeholders
    49→    49→  /// with actual values from [variables].
    50→    50→  ///
    51→    51→  /// Required variables that are missing will produce a warning log but
    52→    52→  /// will be replaced with an empty string so generation can continue.
    53→    53→  String resolvePrompt(
    54→    54→    PromptTemplateEntity template,
    55→    55→    Map<String, dynamic> variables,
    56→    56→  ) {
    57→    57→    String resolved = template.userPromptTemplate;
    58→    58→
    59→    59→    // Replace all {{variable_name}} occurrences
    60→    60→    final variablePattern = RegExp(r'\{\{(\w+)\}\}');
    61→    61→    resolved = resolved.replaceAllMapped(variablePattern, (match) {
    62→    62→      final varName = match.group(1)!;
    63→    63→      if (variables.containsKey(varName)) {
    64→    64→        return variables[varName].toString();
    65→    65→      }
    66→    66→      // Check if template declares this variable with a default
    67→    67→      final templateVar = template.variables
    68→    68→          .where((v) => v.name == varName)
    69→    69→          .firstOrNull;
    70→    70→      if (templateVar?.defaultValue != null) {
    71→    71→        return templateVar!.defaultValue!;
    72→    72→      }
    73→    73→      if (templateVar?.isRequired ?? false) {
    74→    74→        AppLogger.warning('Missing required variable: $varName in template '
    75→    75→            '"${template.name}"');
    76→    76→      }
    77→    77→      return '';
    78→    78→    });
    79→    79→
    80→    80→    return resolved;
    81→    81→  }
    82→    82→
    83→    83→  /// Build a generation prompt from teacher inputs.
    84→    84→  ///
    85→    85→  /// Selects the best matching template from [templates] based on
    86→    86→  /// subject, question type, difficulty, curriculum, and language.
    87→    87→  /// If no suitable template is found, falls back to the built-in
    88→    88→  /// default generation prompt.
    89→    89→  PromptResolution resolveGenerationPrompt(
    90→    90→    GenerationInputEntity input,
    91→    91→    List<PromptTemplateEntity> templates,
    92→    92→  ) {
    93→    93→    // ─── AI SECURITY: Check user inputs for prompt injection ──────────
    94→    94→    // Validate the topic, custom instructions, and other user-supplied
    95→    95→    // fields before incorporating them into the prompt.
    96→    96→    final topicCheck = AiSecurityService.checkInput(input.topic);
    97→    97→    if (!topicCheck.isSafe) {
    98→    98→      AppLogger.error(
    99→    99→        'AI SECURITY: Prompt injection detected in topic. '
   100→   100→        'Blocking generation request.',
   101→   101→      );
   102→   102→      // Return a safe resolution that will result in an error message
   103→   103→      return PromptResolution(
   104→   104→        systemPrompt: 'You are a helpful educational assistant.',
   105→   105→        userPrompt: 'The provided topic contains potentially unsafe content. '
   106→   106→            'Please revise and try again.',
   107→   107→        templateUsed: null,
   108→   108→        resolvedVariables: {},
   109→   109→      );
   110→   110→    }
   111→   111→
   112→   112→    final instructionsCheck = AiSecurityService.checkInput(
   113→   113→      input.customInstructions ?? '',
   114→   114→    );
   115→   115→    if (!instructionsCheck.isSafe && instructionsCheck.sanitizedInput != null) {
   116→   116→      AppLogger.warning(
   117→   117→        'AI SECURITY: Sanitizing custom instructions due to injection patterns.',
   118→   118→      );
   119→   119→      // Use sanitized version instead of blocking
   120→   120→    }
   121→   121→
   122→   122→    final activeTemplates = templates
   123→   123→        .where((t) =>
   124→   124→            t.isActive &&
   125→   125→            t.promptType == PromptType.questionGeneration)
   126→   126→        .toList();
   127→   127→
   128→   128→    // Score each template based on how well it matches the input
   129→   129→    PromptTemplateEntity? bestMatch;
   130→   130→    int bestScore = -1;
   131→   131→
   132→   132→    for (final template in activeTemplates) {
   133→   133→      int score = 0;
   134→   134→
   135→   135→      // Match on question type (high weight)
   136→   136→      if (template.questionType == input.questionType) {
   137→   137→        score += 50;
   138→   138→      }
   139→   139→
   140→   140→      // Match on difficulty (medium weight)
   141→   141→      if (template.difficulty == input.difficulty) {
   142→   142→        score += 20;
   143→   143→      }
   144→   144→
   145→   145→      // Match on subject (high weight)
   146→   146→      if (template.subjectId == input.subjectId) {
   147→   147→        score += 30;
   148→   148→      }
   149→   149→
   150→   150→      // Match on curriculum (medium weight)
   151→   151→      if (template.curriculum == input.curriculum) {
   152→   152→        score += 15;
   153→   153→      }
   154→   154→
   155→   155→      // Match on Bloom's level (medium weight)
   156→   156→      if (template.bloomLevel == input.bloomLevel) {
   157→   157→        score += 15;
   158→   158→      }
   159→   159→
   160→   160→      // Match on language (low weight)
   161→   161→      if (template.language == input.language) {
   162→   162→        score += 10;
   163→   163→      }
   164→   164→
   165→   165→      // Match on provider preference (low weight)
   166→   166→      if (template.provider == input.provider) {
   167→   167→        score += 5;
   168→   168→      }
   169→   169→
   170→   170→      // Default templates get a small bonus
   171→   171→      if (template.isDefault) {
   172→   172→        score += 3;
   173→   173→      }
   174→   174→
   175→   175→      // Higher quality score gets a bonus
   176→   176→      if (template.qualityScore != null) {
   177→   177→        score += (template.qualityScore! * 10).round();
   178→   178→      }
   179→   179→
   180→   180→      // Higher success rate gets a bonus
   181→   181→      if (template.successRate != null) {
   182→   182→        score += (template.successRate! * 10).round();
   183→   183→      }
   184→   184→
   185→   185→      if (score > bestScore) {
   186→   186→        bestScore = score;
   187→   187→        bestMatch = template;
   188→   188→      }
   189→   189→    }
   190→   190→
   191→   191→    // Build resolved variables from input
   192→   192→    final resolvedVariables = _buildGenerationVariables(input);
   193→   193→
   194→   194→    if (bestMatch != null) {
   195→   195→      final systemPrompt = bestMatch.systemPrompt;
   196→   196→      final userPrompt = resolvePrompt(bestMatch, resolvedVariables);
   197→   197→
   198→   198→      // Append few-shot examples if available
   199→   199→      final fullUserPrompt = _appendFewShotExamples(userPrompt, bestMatch);
   200→   200→
   201→   201→      // Append chain-of-thought instruction if enabled
   202→   202→      final finalUserPrompt = bestMatch.chainOfThought
   203→   203→          ? '$fullUserPrompt\n\nThink through this step by step before generating the questions.'
   204→   204→          : fullUserPrompt;
   205→   205→
   206→   206→      return PromptResolution(
   207→   207→        systemPrompt: systemPrompt,
   208→   208→        userPrompt: finalUserPrompt,
   209→   209→        templateUsed: bestMatch,
   210→   210→        resolvedVariables: resolvedVariables,
   211→   211→      );
   212→   212→    }
   213→   213→
   214→   214→    // Fallback to built-in default prompt
   215→   215→    return PromptResolution(
   216→   216→      systemPrompt: _defaultSystemPrompt,
   217→   217→      userPrompt: _buildDefaultUserPrompt(input),
   218→   218→      templateUsed: null,
   219→   219→      resolvedVariables: resolvedVariables,
   220→   220→    );
   221→   221→  }

   323→   323→    buffer.writeln('--- END ORIGINAL QUESTION ---');

   325→   325→    buffer.writeln('Return a JSON object with the following structure:');

   472→   472→  /// Parse AI response into structured generated questions.
   473→   473→  ///
   474→   474→  /// The response should be a JSON object or a list of question objects.
   475→   475→  /// This method handles multiple response formats:
   476→   476→  /// 1. `{ "questions": [...] }` — standard format
   477→   477→  /// 2. `[...]` — direct list of questions
   478→   478→  /// 3. `{ "question": {...} }` — single question wrapper
   479→   479→  List<GeneratedQuestionEntity> parseGeneratedQuestions(
   480→   480→    Map<String, dynamic> response,
   481→   481→    GenerationRequestEntity request,
   482→   482→  ) {
   483→   483→    try {
   484→   484→      List<dynamic> questionsList;
   485→   485→
   486→   486→      if (response.containsKey('questions')) {
   487→   487→        questionsList = response['questions'] as List<dynamic>;
   488→   488→      } else if (response.containsKey('question')) {
   489→   489→        questionsList = [response['question']];
   490→   490→      } else if (response.values.any((v) => v is List)) {
   491→   491→        // Try to find the list of questions
   492→   492→        final listEntry = response.entries
   493→   493→            .firstWhere((e) => e.value is List && (e.value as List).isNotEmpty);
   494→   494→        questionsList = listEntry.value as List<dynamic>;
   495→   495→      } else {
   496→   496→        // Assume the entire response is a single question
   497→   497→        questionsList = [response];
   498→   498→      }
   499→   499→
   500→   500→      return questionsList
   501→   501→          .whereType<Map<String, dynamic>>()
   502→   502→          .map((q) => _mapToGeneratedQuestion(q, request))
   503→   503→          .toList();
   504→   504→    } catch (e) {
   505→   505→      AppLogger.error('Failed to parse generated questions from response', error: e);
   506→   506→      return [];
   507→   507→    }
   508→   508→  }

   542→   542→  List<ValidationResultEntity> parseValidationResults(
   543→   543→    Map<String, dynamic> response,
   544→   544→    String questionId,
   545→   545→  ) {
   546→   546→    try {
   547→   547→      final results = <ValidationResultEntity>[];
   548→   548→
   549→   549→      // If response has validation_results array
   550→   550→      if (response.containsKey('validation_results')) {
   551→   551→        final validationResults =
   552→   552→            response['validation_results'] as List<dynamic>;
   553→   553→        for (int i = 0; i < validationResults.length; i++) {
   554→   554→          final result = validationResults[i] as Map<String, dynamic>;
   555→   555→          results.add(ValidationResultEntity(
   556→   556→            id: '',
   557→   557→            generatedQuestionId: questionId,
   558→   558→            validationType: result['type'] as String? ?? 'general',
   559→   559→            severity: ValidationSeverity.fromString(
   560→   560→                    result['severity'] as String?) ??
   561→   561→                ValidationSeverity.info,
   562→   562→            message: result['message'] as String? ?? 'No message provided',
   563→   563→            suggestion: result['suggestion'] as String?,
   564→   564→            isResolved: false,
   565→   565→            resolvedBy: null,
   566→   566→            resolvedAt: null,
   567→   567→            createdAt: DateTime.now(),
   568→   568→          ));
   569→   569→        }
   570→   570→      }
   571→   571→
   572→   572→      // If no structured results, create one from the overall validity
   573→   573→      if (results.isEmpty) {
   574→   574→        final isValid = response['is_valid'] as bool? ?? true;
   575→   575→        results.add(ValidationResultEntity(
   576→   576→          id: '',
   577→   577→          generatedQuestionId: questionId,
   578→   578→          validationType: 'overall',
   579→   579→          severity: isValid ? ValidationSeverity.info : ValidationSeverity.error,
   580→   580→          message: isValid
   581→   581→              ? 'Question passed validation'
   582→   582→              : 'Question failed validation',
   583→   583→          suggestion: response['suggestion'] as String?,
   584→   584→          isResolved: false,
   585→   585→          resolvedBy: null,
   586→   586→          resolvedAt: null,
   587→   587→          createdAt: DateTime.now(),
   588→   588→        ));
   589→   589→      }
   590→   590→
   591→   591→      return results;
   592→   592→    } catch (e) {
   593→   593→      AppLogger.error('Failed to parse validation results', error: e);
   594→   594→      return [
   595→   595→        ValidationResultEntity(
   596→   596→          id: '',
   597→   597→          generatedQuestionId: questionId,
   598→   598→          validationType: 'parse_error',
   599→   599→          severity: ValidationSeverity.warning,
   600→   600→          message: 'Could not parse validation results from AI response',
   601→   601→          isResolved: false,
   602→   602→          resolvedBy: null,
   603→   603→          resolvedAt: null,
   604→   604→          createdAt: DateTime.now(),
   605→   605→        ),
   606→   606→      ];
   607→   607→    }
   608→   608→  }

   652→   652→  /// Maps a raw question JSON object to a [GeneratedQuestionEntity].
   653→   653→  GeneratedQuestionEntity _mapToGeneratedQuestion(
   654→   654→    Map<String, dynamic> q,
   655→   655→    GenerationRequestEntity request,
   656→   656→  ) {
   657→   657→    // Determine question type
   658→   658→    final questionTypeStr = q['question_type'] as String? ??
   659→   659→        q['type'] as String? ??
   660→   660→        request.inputParams['question_type'] as String? ??
   661→   661→        'multiple_choice';
   662→   662→    final questionType = QuestionType.values.cast<QuestionType?>().firstWhere(
   663→   663→          (t) => t?.value == questionTypeStr,
   664→   664→          orElse: () => QuestionType.multipleChoice,
   665→   665→        ) ?? QuestionType.multipleChoice;
   666→   666→
   667→   667→    // Determine difficulty
   668→   668→    final difficultyStr = q['difficulty'] as String? ??
   669→   669→        request.inputParams['difficulty'] as String? ??
   670→   670→        'medium';
   671→   671→    final difficulty = DifficultyLevel.values.cast<DifficultyLevel?>().firstWhere(
   672→   672→          (d) => d?.value == difficultyStr,
   673→   673→          orElse: () => DifficultyLevel.medium,
   674→   674→        ) ?? DifficultyLevel.medium;
   675→   675→
   676→   676→    // Determine Bloom's level
   677→   677→    final bloomStr = q['bloom_level'] as String? ??
   678→   678→        request.inputParams['bloom_level'] as String?;
   679→   679→    final bloomLevel = BloomTaxonomy.fromString(bloomStr);
   680→   680→
   681→   681→    // Parse answer options
   682→   682→    List<Map<String, dynamic>> answerOptions = [];
   683→   683→    if (q['answer_options'] != null) {
   684→   684→      answerOptions = (q['answer_options'] as List<dynamic>)
   685→   685→          .whereType<Map<String, dynamic>>()
   686→   686→          .toList();
   687→   687→    } else if (q['options'] != null) {
   688→   688→      answerOptions = (q['options'] as List<dynamic>)
   689→   689→          .whereType<Map<String, dynamic>>()
   690→   690→          .toList();
   691→   691→    }
   692→   692→
   693→   693→    // Parse matching pairs
   694→   694→    List<Map<String, dynamic>> matchingPairs = [];
   695→   695→    if (q['matching_pairs'] != null) {
   696→   696→      matchingPairs = (q['matching_pairs'] as List<dynamic>)
   697→   697→          .whereType<Map<String, dynamic>>()
   698→   698→          .toList();
   699→   699→    }
   700→   700→
   701→   701→    // Parse ordering items
   702→   702→    List<Map<String, dynamic>> orderingItems = [];
   703→   703→    if (q['ordering_items'] != null) {
   704→   704→      orderingItems = (q['ordering_items'] as List<dynamic>)
   705→   705→          .whereType<Map<String, dynamic>>()
   706→   706→          .toList();
   707→   707→    }
   708→   708→
   709→   709→    // Parse fill-in-blank answers
   710→   710→    List<Map<String, dynamic>> fillInBlankAnswers = [];
   711→   711→    if (q['fill_in_blank_answers'] != null) {
   712→   712→      fillInBlankAnswers = (q['fill_in_blank_answers'] as List<dynamic>)
   713→   713→          .whereType<Map<String, dynamic>>()
   714→   714→          .toList();
   715→   715→    }
   716→   716→
   717→   717→    // Parse suggested references
   718→   718→    List<String> references = [];
   719→   719→    if (q['references'] != null) {
   720→   720→      references = (q['references'] as List<dynamic>)
   721→   721→          .whereType<String>()
   722→   722→          .toList();
   723→   723→    } else if (q['suggested_references'] != null) {
   724→   724→      references = (q['suggested_references'] as List<dynamic>)
   725→   725→          .whereType<String>()
   726→   726→          .toList();
   727→   727→    }
   728→   728→
   729→   729→    return GeneratedQuestionEntity(
   730→   730→      id: '',
   731→   731→      generationRequestId: request.id,
   732→   732→      questionBankId: null,
   733→   733→      schoolId: request.schoolId,
   734→   734→      questionType: questionType,
   735→   735→      difficulty: difficulty,
   736→   736→      bloomLevel: bloomLevel,
   737→   737→      content: q['content'] as String? ?? q['question'] as String? ?? '',
   738→   738→      contentJson: q['content_json'] as Map<String, dynamic>?,
   739→   739→      answerOptions: answerOptions,
   740→   740→      matchingPairs: matchingPairs,
   741→   741→      orderingItems: orderingItems,
   742→   742→      fillInBlankAnswers: fillInBlankAnswers,
   743→   743→      explanation: q['explanation'] as String?,
   744→   744→      suggestedReferences: references,
   745→   745→      marks: q['marks'] as int? ?? 1,
   746→   746→      estimatedTimeSeconds: q['estimated_time_seconds'] as int?,
   747→   747→      confidenceScore: (q['confidence_score'] as num?)?.toDouble(),
   748→   748→      curriculumAlignment: q['curriculum_alignment'] as Map<String, dynamic>?,
   749→   749→      reviewStatus: ReviewStatus.pending,
   750→   750→      reviewedBy: null,
   751→   751→      reviewedAt: null,
   752→   752→      reviewNotes: null,
   753→   753→      teacherEdits: null,
   754→   754→      isEdited: false,
   755→   755→      isApproved: false,
   756→   756→      metadata: q['metadata'] as Map<String, dynamic>?,
   757→   757→      createdAt: DateTime.now(),
   758→   758→      updatedAt: DateTime.now(),
   759→   759→    );
   760→   760→  }
