    30→    this.error,

    33→
    34→  /// Grade scales available for the school.
    35→  final List<GradeScaleEntity> gradeScales;
    36→
    37→  /// Class-level performance summary.
    38→  final ClassPerformanceEntity? classPerformance;
    39→
    40→  /// School-level performance summary.
    41→  final SchoolPerformanceEntity? schoolPerformance;
    42→
    43→  /// Whether data is being loaded.
    44→  final bool isLoading;
    45→
    46→  /// The most recent error message, or `null`.
    47→  final String? error;
    48→
    49→  /// A transient success message, or `null`.
    50→  final String? successMessage;

    52→  /// Whether any async operation is in progress.

    54→

    57→      gradeScales.isNotEmpty ||

   109→  /// Loads all dashboard data for the given school and session.
   110→  Future<void> loadDashboard({
   111→    required String schoolId,
   112→    required String classId,
   113→    required String academicSessionId,
   114→  }) async {
   115→    state = state.copyWith(isLoading: true, error: null);
   116→
   117→    // Load grade scales
   118→    final scalesResult = await _getGradeScalesUseCase(schoolId, isActive: true);
   119→
   120→    scalesResult.fold(
   121→      onSuccess: (scales) {
   122→        state = state.copyWith(gradeScales: scales);
   123→      },
   124→      onFailure: (failure) {
   125→        AppLogger.warning('Failed to load grade scales: $failure');
   126→      },
   127→    );
   128→
   129→    // Load class performance
   130→    final classResult = await _getClassPerformanceUseCase(
   131→      classId: classId,
   132→      academicSessionId: academicSessionId,
   133→    );
   134→
   135→    classResult.fold(
   136→      onSuccess: (performance) {
   137→        state = state.copyWith(classPerformance: performance);
   138→      },
   139→      onFailure: (failure) {
   140→        AppLogger.warning('Failed to load class performance: $failure');
   141→      },
   142→    );
   143→
   144→    // Load school performance
   145→    final schoolResult = await _getSchoolPerformanceUseCase(
   146→      schoolId: schoolId,
   147→      academicSessionId: academicSessionId,
   148→    );
   149→
   150→    schoolResult.fold(
   151→      onSuccess: (performance) {
   152→        state = state.copyWith(
   153→          schoolPerformance: performance,
   154→          isLoading: false,
   155→          error: null,
   156→        );
   157→        AppLogger.info('Dashboard data loaded successfully');
   158→      },
   159→      onFailure: (failure) {
   160→        state = state.copyWith(
   161→          isLoading: false,
   162→          error: _mapFailureToMessage(failure),
   163→        );
   164→        AppLogger.warning('Failed to load school performance: $failure');
   165→      },
   166→    );

   436→  /// Updates an existing grade scale.

   456→        state = state.copyWith(
   457→          isLoading: false,
   458→          error: _mapFailureToMessage(failure),

   467→  /// Deletes a grade scale by its ID.

   521→          'Grade scale applied: $percentage% → ${entry?.grade ?? "no match"}',

   526→          isLoading: false,
