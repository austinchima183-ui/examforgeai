     1→-- ============================================================================
     2→-- ExamForge AI - Question Bank Module Schema
     3→-- ============================================================================
     4→-- Production-ready schema for the Question Bank module.
     5→-- Supports millions of questions across thousands of schools.
     6→--
     7→-- Prerequisites: Existing tables (schools, users, classes, subjects,
     8→--                class_subjects, class_students, notifications, audit_log)
     9→--                and existing enums (user_role, subscription_status,
    10→--                exam_status, notification_type, question_type)
    11→--
    12→-- Performance: Designed for high-throughput with composite indexes,
    13→--              GIN indexes for JSONB/full-text, and partitioning-ready layout.
    14→-- ============================================================================
    15→
    16→BEGIN;
    17→
    18→-- ============================================================================
    19→-- 1. EXTEND EXISTING ENUMS
    20→-- ============================================================================
    21→
    22→-- ---------------------------------------------------------------------------
    23→-- Extend question_type with additional types for richer question formats
    24→-- ---------------------------------------------------------------------------
    25→DO $$
    26→BEGIN
    27→  -- Add new values to the existing question_type enum
    28→  IF NOT EXISTS (SELECT 1 FROM pg_enum WHERE enumlabel = 'multiple_response' AND enumtypid = (SELECT oid FROM pg_type WHERE typname = 'question_type')) THEN
    29→    ALTER TYPE question_type ADD VALUE 'multiple_response';
    30→  END IF;
    31→  IF NOT EXISTS (SELECT 1 FROM pg_enum WHERE enumlabel = 'matching' AND enumtypid = (SELECT oid FROM pg_type WHERE typname = 'question_type')) THEN
    32→    ALTER TYPE question_type ADD VALUE 'matching';
    33→  END IF;
    34→  IF NOT EXISTS (SELECT 1 FROM pg_enum WHERE enumlabel = 'ordering' AND enumtypid = (SELECT oid FROM pg_type WHERE typname = 'question_type')) THEN
    35→    ALTER TYPE question_type ADD VALUE 'ordering';
    36→  END IF;
    37→  IF NOT EXISTS (SELECT 1 FROM pg_enum WHERE enumlabel = 'numerical' AND enumtypid = (SELECT oid FROM pg_type WHERE typname = 'question_type')) THEN
    38→    ALTER TYPE question_type ADD VALUE 'numerical';
    39→  END IF;
    40→  IF NOT EXISTS (SELECT 1 FROM pg_enum WHERE enumlabel = 'image_based' AND enumtypid = (SELECT oid FROM pg_type WHERE typname = 'question_type')) THEN
    41→    ALTER TYPE question_type ADD VALUE 'image_based';
    42→  END IF;
    43→  IF NOT EXISTS (SELECT 1 FROM pg_enum WHERE enumlabel = 'audio_based' AND enumtypid = (SELECT oid FROM pg_type WHERE typname = 'question_type')) THEN
    44→    ALTER TYPE question_type ADD VALUE 'audio_based';
    45→  END IF;
    46→  IF NOT EXISTS (SELECT 1 FROM pg_enum WHERE enumlabel = 'video_based' AND enumtypid = (SELECT oid FROM pg_type WHERE typname = 'question_type')) THEN
    47→    ALTER TYPE question_type ADD VALUE 'video_based';
    48→  END IF;
    49→  IF NOT EXISTS (SELECT 1 FROM pg_enum WHERE enumlabel = 'practical' AND enumtypid = (SELECT oid FROM pg_type WHERE typname = 'question_type')) THEN
    50→    ALTER TYPE question_type ADD VALUE 'practical';
    51→  END IF;
    52→  IF NOT EXISTS (SELECT 1 FROM pg_enum WHERE enumlabel = 'case_study' AND enumtypid = (SELECT oid FROM pg_type WHERE typname = 'question_type')) THEN
    53→    ALTER TYPE question_type ADD VALUE 'case_study';
    54→  END IF;
    55→END
    56→$$;
    57→
    58→COMMENT ON TYPE question_type IS
    59→  'Supported question formats: MC, MR, T/F, short answer, essay, fill-in-blank, matching, ordering, numerical, media-based, practical, case study';
    60→
    61→-- ============================================================================
    62→-- 2. NEW ENUMERATION TYPES
    63→-- ============================================================================
    64→
    65→DO $$
    66→BEGIN
    67→  -- difficulty_level: Question difficulty classification
    68→  IF NOT EXISTS (SELECT 1 FROM pg_type WHERE typname = 'difficulty_level') THEN
    69→    CREATE TYPE difficulty_level AS ENUM (
    70→      'easy',
    71→      'medium',
    72→      'hard',
    73→      'expert'
    74→    );
    75→  END IF;
    76→
    77→  -- exam_type: Nigerian and international examination bodies / contexts
    78→  IF NOT EXISTS (SELECT 1 FROM pg_type WHERE typname = 'exam_type') THEN
    79→    CREATE TYPE exam_type AS ENUM (
    80→      'waecc',
    81→      'neco',
    82→      'jamb',
    83→      'school_exam',
    84→      'mock',
    85→      'practice',
    86→      'custom'
    87→    );
    88→  END IF;
    89→
    90→  -- share_permission: Granular permissions for shared questions
    91→  IF NOT EXISTS (SELECT 1 FROM pg_type WHERE typname = 'share_permission') THEN
    92→    CREATE TYPE share_permission AS ENUM (
    93→      'read',
    94→      'comment',
    95→      'edit'
    96→    );
    97→  END IF;
    98→
    99→  -- import_status: Lifecycle states for bulk import jobs
   100→  IF NOT EXISTS (SELECT 1 FROM pg_type WHERE typname = 'import_status') THEN
   101→    CREATE TYPE import_status AS ENUM (
   102→      'pending',
   103→      'processing',
   104→      'completed',
   105→      'failed'
   106→    );
   107→  END IF;
   108→
   109→  -- content_type: Media / attachment content classification
   110→  IF NOT EXISTS (SELECT 1 FROM pg_type WHERE typname = 'content_type') THEN
   111→    CREATE TYPE content_type AS ENUM (
   112→      'text',
   113→      'image',
   114→      'audio',
   115→      'video',
   116→      'document'
   117→    );
   118→  END IF;
   119→
   120→  -- curriculum_standard_type: Nigerian curriculum framework types
   121→  IF NOT EXISTS (SELECT 1 FROM pg_type WHERE typname = 'curriculum_standard_type') THEN
   122→    CREATE TYPE curriculum_standard_type AS ENUM (
   123→      'nigeria_erc',
   124→      'waec',
   125→      'neco',
   126→      'jamb',
   127→      'custom'
   128→    );
   129→  END IF;
   130→END
   131→$$;
   132→
   133→COMMENT ON TYPE difficulty_level IS 'Question difficulty: easy → medium → hard → expert';
   134→COMMENT ON TYPE exam_type IS 'Examination body or context for which the question is designed';
   135→COMMENT ON TYPE share_permission IS 'Permission level when sharing a question: read, comment, or edit';
   136→COMMENT ON TYPE import_status IS 'Status of a bulk question import/export job';
   137→COMMENT ON TYPE content_type IS 'Type of media content attached to a question';
   138→COMMENT ON TYPE curriculum_standard_type IS 'Curriculum framework a standard belongs to';

   252→-- ============================================================================
   253→-- 8. QUESTION_BANK TABLE (THE MAIN TABLE)
   254→-- ============================================================================
   255→-- Central repository for all questions. Designed for millions of rows.
   256→-- - school_id NULL = platform-wide question (visible to all schools)
   257→-- - parent_id = previous version (versioning chain)
   258→-- - content_json = structured data for complex question types
   259→-- ============================================================================
   260→
   261→CREATE TABLE IF NOT EXISTS question_bank (
   262→  id                      UUID PRIMARY KEY DEFAULT gen_random_uuid(),
   263→  school_id               UUID REFERENCES schools(id) ON DELETE CASCADE,  -- NULL = platform-wide
   264→  subject_id              UUID NOT NULL REFERENCES subjects(id),
   265→  topic_id                UUID REFERENCES topics(id) ON DELETE SET NULL,
   266→  subtopic_id             UUID REFERENCES subtopics(id) ON DELETE SET NULL,
   267→  class_id                UUID REFERENCES classes(id) ON DELETE SET NULL,
   268→  category_id             UUID REFERENCES question_categories(id) ON DELETE SET NULL,
   269→  curriculum_standard_id  UUID REFERENCES curriculum_standards(id) ON DELETE SET NULL,
   270→  academic_session_id     UUID REFERENCES academic_sessions(id) ON DELETE SET NULL,
   271→  question_type           question_type NOT NULL DEFAULT 'multiple_choice',
   272→  difficulty              difficulty_level NOT NULL DEFAULT 'medium',
   273→  exam_type               exam_type DEFAULT 'school_exam',
   274→  content                 TEXT NOT NULL,                  -- Question text (HTML/LaTeX supported)
   275→  content_json            JSONB DEFAULT '{}',             -- Structured content for complex types
   276→  explanation             TEXT,                           -- Detailed answer explanation
   277→  teacher_notes           TEXT,                           -- Private notes for teachers
   278→  reference_materials     TEXT,                           -- Reference sources
   279→  marks                   NUMERIC(5,2) NOT NULL DEFAULT 1.00,
   280→  negative_marks          NUMERIC(5,2) DEFAULT 0.00,
   281→  time_allowed_seconds    INTEGER,                        -- Suggested time to answer
   282→  is_published            BOOLEAN DEFAULT false,
   283→  is_archived             BOOLEAN DEFAULT false,
   284→  is_featured             BOOLEAN DEFAULT false,
   285→  version                 INTEGER DEFAULT 1,
   286→  parent_id               UUID REFERENCES question_bank(id) ON DELETE SET NULL,  -- Version chain
   287→  created_by              UUID NOT NULL REFERENCES users(id),
   288→  updated_by              UUID REFERENCES users(id),
   289→  usage_count             INTEGER DEFAULT 0,
   290→  avg_score               NUMERIC(5,2) DEFAULT 0.00,
   291→  metadata                JSONB DEFAULT '{}',             -- Extensible metadata
   292→  created_at              TIMESTAMPTZ DEFAULT now(),
   293→  updated_at              TIMESTAMPTZ DEFAULT now(),
   294→
   295→  -- Ensure non-negative marks
   296→  CONSTRAINT question_bank_marks_positive CHECK (marks >= 0),
   297→  CONSTRAINT question_bank_negative_marks_non_negative CHECK (negative_marks >= 0),
   298→  CONSTRAINT question_bank_avg_score_range CHECK (avg_score >= 0 AND avg_score <= marks),
   299→  -- A question cannot be both published and archived
   300→  CONSTRAINT question_bank_not_published_and_archived CHECK (NOT (is_published = true AND is_archived = true))
   301→);
   302→
   303→COMMENT ON TABLE question_bank IS 'Central question repository supporting millions of questions';
   304→COMMENT ON COLUMN question_bank.school_id IS 'Owning school; NULL means platform-wide question visible to all';
   305→COMMENT ON COLUMN question_bank.content IS 'The question text — supports HTML and LaTeX formatting';
   306→COMMENT ON COLUMN question_bank.content_json IS 'Structured content payload for complex question types (matching, ordering, etc.)';
   307→COMMENT ON COLUMN question_bank.parent_id IS 'Previous version of this question for versioning chain';
   308→COMMENT ON COLUMN question_bank.metadata IS 'Extensible JSONB metadata for custom attributes and integrations';
   309→COMMENT ON COLUMN question_bank.usage_count IS 'Number of times this question has been used in exams';

   815→-- ---------------------------------------------------------------------------
   816→-- create_question_version(question_id UUID, change_type VARCHAR, change_desc TEXT)
   817→-- Creates a version snapshot of a question before it is modified.
   818→-- Returns the new version number.
   819→-- ---------------------------------------------------------------------------
   820→CREATE OR REPLACE FUNCTION create_question_version(
   821→  target_question_id UUID,
   822→  p_change_type     VARCHAR,
   823→  p_change_desc     TEXT DEFAULT NULL
   824→)
   825→RETURNS INTEGER AS $$
   826→DECLARE
   827→  v_question   RECORD;
   828→  v_version    INTEGER;
   829→  v_snapshot   JSONB;
   830→  v_changed_by UUID;
   831→BEGIN
   832→  -- Get current auth user as the changer
   833→  SELECT id INTO v_changed_by FROM users WHERE id = auth.uid();
   834→
   835→  -- Fetch the current question row
   836→  SELECT * INTO v_question FROM question_bank WHERE id = target_question_id;
   837→  IF NOT FOUND THEN
   838→    RAISE EXCEPTION 'Question % not found', target_question_id;
   839→  END IF;
   840→
   841→  -- Build the snapshot JSONB
   842→  v_snapshot := jsonb_build_object(
   843→    'id',                 v_question.id,
   844→    'school_id',          v_question.school_id,
   845→    'subject_id',         v_question.subject_id,
   846→    'topic_id',           v_question.topic_id,
   847→    'subtopic_id',        v_question.subtopic_id,
   848→    'class_id',           v_question.class_id,
   849→    'category_id',        v_question.category_id,
   850→    'curriculum_standard_id', v_question.curriculum_standard_id,
   851→    'academic_session_id',v_question.academic_session_id,
   852→    'question_type',      v_question.question_type,
   853→    'difficulty',         v_question.difficulty,
   854→    'exam_type',          v_question.exam_type,
   855→    'content',            v_question.content,
   856→    'content_json',       v_question.content_json,
   857→    'explanation',        v_question.explanation,
   858→    'teacher_notes',      v_question.teacher_notes,
   859→    'reference_materials',v_question.reference_materials,
   860→    'marks',              v_question.marks,
   861→    'negative_marks',     v_question.negative_marks,
   862→    'time_allowed_seconds', v_question.time_allowed_seconds,
   863→    'is_published',       v_question.is_published,
   864→    'is_archived',        v_question.is_archived,
   865→    'is_featured',        v_question.is_featured,
   866→    'version',            v_question.version,
   867→    'parent_id',          v_question.parent_id,
   868→    'created_by',         v_question.created_by,
   869→    'updated_by',         v_question.updated_by,
   870→    'usage_count',        v_question.usage_count,
   871→    'avg_score',          v_question.avg_score,
   872→    'metadata',           v_question.metadata,
   873→    'created_at',         v_question.created_at,
   874→    'updated_at',         v_question.updated_at
   875→  );
   876→
   877→  -- Compute the next version number
   878→  v_version := v_question.version + 1;
   879→
   880→  -- Insert the version history record
   881→  INSERT INTO question_version_history (
   882→    question_id, version, snapshot, change_description, changed_by, change_type
   883→  ) VALUES (
   884→    target_question_id, v_version, v_snapshot,
   885→    COALESCE(p_change_desc, 'Version ' || v_version),
   886→    v_changed_by, p_change_type
   887→  );
   888→
   889→  -- Update the question's version counter
   890→  UPDATE question_bank
   891→  SET version    = v_version,
   892→      updated_at = now()
   893→  WHERE id = target_question_id;
   894→
   895→  RETURN v_version;
   896→END;
   897→$$ LANGUAGE plpgsql SECURITY DEFINER;
   898→
   899→COMMENT ON FUNCTION create_question_version(UUID, VARCHAR, TEXT) IS
   900→  'Create a version snapshot of a question and increment its version number';
   901→
   902→-- ---------------------------------------------------------------------------
   903→-- get_question_with_details(target_question_id UUID)
   904→-- Returns a JSONB object containing the question + all related data
   905→-- (answer_options, matching_pairs, ordering_items, fill_in_blank_answers,
   906→--  question_attachments, tags) in a single call.
   907→-- ---------------------------------------------------------------------------
   908→CREATE OR REPLACE FUNCTION get_question_with_details(target_question_id UUID)
   909→RETURNS JSONB AS $$
   910→DECLARE
   911→  v_result JSONB;
   912→BEGIN
   913→  SELECT jsonb_build_object(
   914→    'question',    to_jsonb(q.*),
   915→    'options',     COALESCE(
   916→      (SELECT jsonb_agg(to_jsonb(o.*) ORDER BY o.sort_order)
   917→       FROM answer_options o WHERE o.question_id = target_question_id),
   918→      '[]'::jsonb
   919→    ),
   920→    'matching_pairs', COALESCE(
   921→      (SELECT jsonb_agg(to_jsonb(mp.*) ORDER BY mp.sort_order)
   922→       FROM matching_pairs mp WHERE mp.question_id = target_question_id),
   923→      '[]'::jsonb
   924→    ),
   925→    'ordering_items', COALESCE(
   926→      (SELECT jsonb_agg(to_jsonb(oi.*) ORDER BY oi.correct_position)
   927→       FROM ordering_items oi WHERE oi.question_id = target_question_id),
   928→      '[]'::jsonb
   929→    ),
   930→    'fill_in_blank_answers', COALESCE(
   931→      (SELECT jsonb_agg(to_jsonb(fib.*) ORDER BY fib.blank_index)
   932→       FROM fill_in_blank_answers fib WHERE fib.question_id = target_question_id),
   933→      '[]'::jsonb
   934→    ),
   935→    'attachments', COALESCE(
   936→      (SELECT jsonb_agg(to_jsonb(a.*) ORDER BY a.sort_order)
   937→       FROM question_attachments a WHERE a.question_id = target_question_id),
   938→      '[]'::jsonb
   939→    ),
   940→    'tags', COALESCE(
   941→      (SELECT jsonb_agg(jsonb_build_object('id', t.id, 'name', t.name))
   942→       FROM question_tag_relations qtr
   943→       JOIN question_tags t ON t.id = qtr.tag_id
   944→       WHERE qtr.question_id = target_question_id),
   945→      '[]'::jsonb
   946→    )
   947→  ) INTO v_result
   948→  FROM question_bank q
   949→  WHERE q.id = target_question_id;
   950→
   951→  RETURN v_result;
   952→END;
   953→$$ LANGUAGE plpgsql SECURITY DEFINER STABLE;

   956→  'Fetch a question with all its related data (options, pairs, items, blanks, attachments, tags) as a single JSONB';

   958→-- ---------------------------------------------------------------------------
   959→-- search_questions(search_query TEXT, filters JSONB)
   960→-- Full-text search function with optional filters.
   961→-- Filters JSONB can include:
   962→--   school_id, subject_id, topic_id, difficulty, question_type, exam_type,
   963→--   is_published, is_archived, class_id, category_id, created_by, limit, offset
   964→-- ---------------------------------------------------------------------------
   965→CREATE OR REPLACE FUNCTION search_questions(
   966→  p_search_query TEXT,
   967→  p_filters      JSONB DEFAULT '{}'
   968→)
   969→RETURNS TABLE (
   970→  id                UUID,
   971→  school_id         UUID,
   972→  subject_id        UUID,
   973→  topic_id          UUID,
   974→  question_type     question_type,
   975→  difficulty        difficulty_level,
   976→  exam_type         exam_type,
   977→  content           TEXT,
   978→  marks             NUMERIC,
   979→  is_published      BOOLEAN,
   980→  is_archived       BOOLEAN,
   981→  usage_count       INTEGER,
   982→  avg_score         NUMERIC,
   983→  created_by        UUID,
   984→  created_at        TIMESTAMPTZ,
   985→  rank              REAL
   986→) AS $$
   987→DECLARE
   988→  v_limit  INTEGER := COALESCE((p_filters->>'limit')::INTEGER, 50);
   989→  v_offset INTEGER := COALESCE((p_filters->>'offset')::INTEGER, 0);
   990→BEGIN
   991→  RETURN QUERY
   992→  SELECT
   993→    qb.id,
   994→    qb.school_id,
   995→    qb.subject_id,
   996→    qb.topic_id,
   997→    qb.question_type,
   998→    qb.difficulty,
   999→    qb.exam_type,
  1000→    qb.content,
  1001→    qb.marks,
  1002→    qb.is_published,
  1003→    qb.is_archived,
  1004→    qb.usage_count,
  1005→    qb.avg_score,
  1006→    qb.created_by,
  1007→    qb.created_at,
  1008→    ts_rank(to_tsvector('english', qb.content), plainto_tsquery('english', p_search_query)) AS rank
  1009→  FROM question_bank qb
  1010→  WHERE
  1011→    -- Full-text search filter
  1012→    (p_search_query IS NULL OR p_search_query = '' OR
  1013→     to_tsvector('english', qb.content) @@ plainto_tsquery('english', p_search_query))
  1014→    -- School filter (includes platform-wide NULL school_id)
  1015→    AND (p_filters->>'school_id' IS NULL OR qb.school_id::TEXT = p_filters->>'school_id' OR qb.school_id IS NULL)
  1016→    -- Subject filter
  1017→    AND (p_filters->>'subject_id' IS NULL OR qb.subject_id::TEXT = p_filters->>'subject_id')
  1018→    -- Topic filter
  1019→    AND (p_filters->>'topic_id' IS NULL OR qb.topic_id::TEXT = p_filters->>'topic_id')
  1020→    -- Difficulty filter
  1021→    AND (p_filters->>'difficulty' IS NULL OR qb.difficulty::TEXT = p_filters->>'difficulty')
  1022→    -- Question type filter
  1023→    AND (p_filters->>'question_type' IS NULL OR qb.question_type::TEXT = p_filters->>'question_type')
  1024→    -- Exam type filter
  1025→    AND (p_filters->>'exam_type' IS NULL OR qb.exam_type::TEXT = p_filters->>'exam_type')
  1026→    -- Published filter
  1027→    AND (p_filters->>'is_published' IS NULL OR qb.is_published = (p_filters->>'is_published')::BOOLEAN)
  1028→    -- Not archived (default: exclude archived)
  1029→    AND (p_filters->>'is_archived' IS NULL OR qb.is_archived = (p_filters->>'is_archived')::BOOLEAN)
  1030→    -- Class filter
  1031→    AND (p_filters->>'class_id' IS NULL OR qb.class_id::TEXT = p_filters->>'class_id')
  1032→    -- Category filter
  1033→    AND (p_filters->>'category_id' IS NULL OR qb.category_id::TEXT = p_filters->>'category_id')
  1034→    -- Created by filter
  1035→    AND (p_filters->>'created_by' IS NULL OR qb.created_by::TEXT = p_filters->>'created_by')
  1036→  ORDER BY
  1037→    rank DESC NULLS LAST,
  1038→    qb.created_at DESC
  1039→  LIMIT v_limit
  1040→  OFFSET v_offset;
  1041→END;
  1042→$$ LANGUAGE plpgsql SECURITY DEFINER STABLE;

  1157→-- Auto-create version history on question_bank update
  1158→-- ---------------------------------------------------------------------------
  1159→CREATE OR REPLACE FUNCTION trigger_create_question_version()
  1160→RETURNS TRIGGER AS $$
  1161→DECLARE
  1162→  v_change_type VARCHAR(20);
  1163→BEGIN
  1164→  -- Determine the change type from the transition
  1165→  IF TG_OP = 'INSERT' THEN
  1166→    v_change_type := 'create';
  1167→    -- Insert initial version history on creation
  1168→    INSERT INTO question_version_history (
  1169→      question_id, version, snapshot, change_description, changed_by, change_type
  1170→    ) VALUES (
  1171→      NEW.id,
  1172→      1,
  1173→      to_jsonb(NEW),
  1174→      'Initial creation',
  1175→      NEW.created_by,
  1176→      v_change_type
  1177→    );
  1178→    RETURN NEW;
  1179→  END IF;
  1180→
  1181→  -- For updates, determine the change type
  1182→  IF TG_OP = 'UPDATE' THEN
  1183→    -- Detect specific state transitions
  1184→    IF OLD.is_archived = false AND NEW.is_archived = true THEN
  1185→      v_change_type := 'archive';
  1186→    ELSIF OLD.is_published = false AND NEW.is_published = true THEN
  1187→      v_change_type := 'publish';
  1188→    ELSIF OLD.is_archived = true AND NEW.is_archived = false AND NEW.is_published = true THEN
  1189→      v_change_type := 'restore';
  1190→    ELSE
  1191→      v_change_type := 'update';
  1192→    END IF;
  1193→
  1194→    -- Only create a version snapshot if substantive fields changed
  1195→    -- (skip updated_at-only changes to avoid noise)
  1196→    IF OLD.content            IS DISTINCT FROM NEW.content OR
  1197→       OLD.content_json       IS DISTINCT FROM NEW.content_json OR
  1198→       OLD.explanation        IS DISTINCT FROM NEW.explanation OR
  1199→       OLD.marks              IS DISTINCT FROM NEW.marks OR
  1200→       OLD.negative_marks     IS DISTINCT FROM NEW.negative_marks OR
  1201→       OLD.difficulty         IS DISTINCT FROM NEW.difficulty OR
  1202→       OLD.question_type      IS DISTINCT FROM NEW.question_type OR
  1203→       OLD.exam_type          IS DISTINCT FROM NEW.exam_type OR
  1204→       OLD.is_published       IS DISTINCT FROM NEW.is_published OR
  1205→       OLD.is_archived        IS DISTINCT FROM NEW.is_archived OR
  1206→       OLD.is_featured        IS DISTINCT FROM NEW.is_featured OR
  1207→       OLD.topic_id           IS DISTINCT FROM NEW.topic_id OR
  1208→       OLD.subtopic_id        IS DISTINCT FROM NEW.subtopic_id OR
  1209→       OLD.category_id        IS DISTINCT FROM NEW.category_id OR
  1210→       OLD.metadata           IS DISTINCT FROM NEW.metadata OR
  1211→       OLD.teacher_notes      IS DISTINCT FROM NEW.teacher_notes OR
  1212→       OLD.reference_materials IS DISTINCT FROM NEW.reference_materials OR
  1213→       OLD.time_allowed_seconds IS DISTINCT FROM NEW.time_allowed_seconds
  1214→    THEN
  1215→      PERFORM create_question_version(NEW.id, v_change_type);
  1216→    END IF;
  1217→
  1218→    RETURN NEW;
  1219→  END IF;
  1220→
  1221→  RETURN COALESCE(NEW, OLD);
  1222→END;
  1223→$$ LANGUAGE plpgsql SECURITY DEFINER;
