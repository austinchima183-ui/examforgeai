   546→  void _showCreateApiKeyDialog() {
   547→    final nameCtrl = TextEditingController();
   548→    final scopes = <String>['read', 'write'];
   549→
   550→    showDialog(
   551→      context: context,
   552→      builder: (context) => StatefulBuilder(
   553→        builder: (context, setDialogState) => AlertDialog(
   554→          title: const Text('Create API Key'),
   555→          content: SingleChildScrollView(
   556→            child: Column(
   557→              mainAxisSize: MainAxisSize.min,
   558→              children: [
   559→                TextField(
   560→                    controller: nameCtrl,
   561→                    decoration: const InputDecoration(
   562→                        labelText: 'Key Name *',
   563→                        border: OutlineInputBorder())),
   564→                const SizedBox(height: Spacings.md),
   565→                Text('Scopes',
   566→                    style: AppTypography.labelMedium!.copyWith(
   567→                        fontWeight: AppTypography.wSemiBold)),
   568→                const SizedBox(height: Spacings.sm),
   569→                Wrap(
   570→                  spacing: Spacings.sm,
   571→                  children: ['read', 'write', 'admin', 'delete']
   572→                      .map((scope) => FilterChip(
   573→                            label: Text(scope),
   574→                            selected: scopes.contains(scope),
   575→                            onSelected: (selected) {
   576→                              setDialogState(() {
   577→                                if (selected) {
   578→                                  scopes.add(scope);
   579→                                } else {
   580→                                  scopes.remove(scope);
   581→                                }
   582→                              });
   583→                            },
   584→                          ))
   585→                      .toList(),
   586→                ),
   587→              ],
   588→            ),
   589→          ),
   590→          actions: [
   591→            TextButton(
   592→                onPressed: () => Navigator.pop(context),
   593→                child: const Text('Cancel')),
   594→            AppButton(
   595→              label: 'Create',
   596→              onPressed: () {
   597→                ref.read(enterpriseProvider.notifier).createApiKey(
   598→                      userId: 'current_user',
   599→                      name: nameCtrl.text,
   600→                    );
   601→                Navigator.pop(context);
   602→              },
   603→            ),
   604→          ],
   605→        ),
   606→      ),
   607→    );
