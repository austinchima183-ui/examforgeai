    26→/// The main Billing Dashboard page.
    27→///
    28→/// This is the entry point for the billing feature, presenting a modern
    29→/// SaaS-style billing dashboard with:
    30→/// - Current subscription overview (plan name, status, period end)
    31→/// - AI credit balance card
    32→/// - Quick actions (Upgrade, Manage, View Invoices, Buy Credits)
    33→/// - Recent transactions list
    34→/// - Billing alerts/notifications
    35→///
    36→/// Uses [ConsumerStatefulWidget] with a private [_State] class pattern,
    37→/// loads data in [initState] via [WidgetsBinding.instance.addPostFrameCallback],
    38→/// and renders a responsive layout using [SingleChildScrollView] with
    39→/// vertically arranged Card sections.
    40→class BillingDashboardPage extends ConsumerStatefulWidget {
    41→  const BillingDashboardPage({super.key});
    42→
    43→  @override
    44→  ConsumerState<BillingDashboardPage> createState() => _State();
    45→}
    46→
    47→class _State extends ConsumerState<BillingDashboardPage> {
    48→  // ─── Lifecycle ──────────────────────────────────────────────────────
    49→
    50→  @override
    51→  void initState() {
    52→    super.initState();
    53→    WidgetsBinding.instance.addPostFrameCallback((_) {
    54→      _loadData();
    55→    });
    56→  }
    57→
    58→  /// Loads all billing data in parallel.
    59→  void _loadData() {
    60→    final userId = ref.read(userIdProvider);
    61→    if (userId == null) return;
    62→
    63→    final subscriberType = BillingModel.teacherSaas;
    64→
    65→    ref.read(subscriptionProvider.notifier).loadCurrentSubscription(
    66→          subscriberId: userId,
    67→          subscriberType: subscriberType,
    68→        );
    69→
    70→    ref.read(aiCreditsProvider.notifier).loadCreditBalance(
    71→          ownerId: userId,
    72→          ownerType: subscriberType,
    73→        );
    74→
    75→    ref.read(paymentProvider.notifier).loadTransactions(
    76→          userId: userId,
    77→          page: 1,
    78→          perPage: 5,
    79→        );
    80→
    81→    ref.read(billingNotificationProvider.notifier).loadNotifications(
    82→          userId: userId,
    83→          unreadOnly: false,
    84→          page: 1,
    85→          perPage: 10,
    86→        );
    87→  }
    88→
    89→  // ─── Build ──────────────────────────────────────────────────────────
    90→
    91→  @override
    92→  Widget build(BuildContext context) {
    93→    final subscriptionState = ref.watch(subscriptionProvider);
    94→    final creditsState = ref.watch(aiCreditsProvider);
    95→    final paymentState = ref.watch(paymentProvider);
    96→    final notificationState = ref.watch(billingNotificationProvider);
    97→
    98→    final isLoading = subscriptionState.isLoading ||
    99→        creditsState.isLoading ||
   100→        (paymentState.isLoading && paymentState.transactions.isEmpty) ||
   101→        (notificationState.isLoading && notificationState.notifications.isEmpty);
   102→
   103→    final hasError = subscriptionState.error != null &&
   104→        subscriptionState.subscription == null;
   105→
   106→    return Scaffold(
   107→      appBar: AppAppBar(
   108→        title: 'Billing',
   109→        actions: [
   110→          if (notificationState.hasUnread)
   111→            IconButton(
   112→              icon: Badge(
   113→                label: Text('${notificationState.unreadCount}'),
   114→                child: const Icon(Icons.notifications_outlined),
   115→              ),
   116→              onPressed: () => context.go(RouteNames.billingNotifications),
   117→              tooltip: 'Notifications',
   118→            )
   119→          else
   120→            IconButton(
   121→              icon: const Icon(Icons.notifications_outlined),
   122→              onPressed: () => context.go(RouteNames.billingNotifications),
   123→              tooltip: 'Notifications',
   124→            ),
   125→        ],
   126→      ),
   127→      body: _buildBody(
   128→        context,
   129→        isLoading: isLoading,
   130→        hasError: hasError,
   131→        subscriptionState: subscriptionState,
   132→        creditsState: creditsState,
   133→        paymentState: paymentState,
   134→        notificationState: notificationState,
   135→      ),
   136→    );
   137→  }
   138→
   139→  // ─── Body Router ────────────────────────────────────────────────────
   140→
   141→  Widget _buildBody(
   142→    BuildContext context, {
   143→    required bool isLoading,
   144→    required bool hasError,
   145→    required SubscriptionState subscriptionState,
   146→    required AiCreditsState creditsState,
   147→    required PaymentState paymentState,
   148→    required BillingNotificationState notificationState,
   149→  }) {
   150→    // Loading state — initial load with no data
   151→    if (isLoading &&
   152→        subscriptionState.subscription == null &&
   153→        creditsState.creditBalance == null &&
   154→        paymentState.transactions.isEmpty) {
   155→      return const Center(child: AppLoadingSpinner());
   156→    }
   157→
   158→    // Error state — no subscription data available
   159→    if (hasError) {
   160→      return AppErrorState.genericError(
   161→        message: subscriptionState.error,
   162→        onRetry: _loadData,
   163→      );
   164→    }
   165→
   166→    // Success — render dashboard
   167→    return RefreshIndicator(
   168→      onRefresh: () async => _loadData(),
   169→      child: SingleChildScrollView(
   170→        physics: const AlwaysScrollableScrollPhysics(),
   171→        padding: const EdgeInsets.symmetric(
   172→          horizontal: Spacings.lg,
   173→          vertical: Spacings.md,
   174→        ),
   175→        child: Column(
   176→          crossAxisAlignment: CrossAxisAlignment.start,
   177→          children: [
   178→            // ─── Subscription Overview ────────────────────────────
   179→            _buildSubscriptionOverview(context, subscriptionState),
   180→
   181→            const SizedBox(height: Spacings.lg),
   182→
   183→            // ─── AI Credit Balance Card ───────────────────────────
   184→            _buildCreditBalanceCard(context, creditsState),
   185→
   186→            const SizedBox(height: Spacings.lg),
   187→
   188→            // ─── Quick Actions ────────────────────────────────────
   189→            _buildQuickActions(context),
   190→
   191→            const SizedBox(height: Spacings.lg),
   192→
   193→            // ─── Billing Alerts / Notifications ───────────────────
   194→            _buildBillingAlerts(context, notificationState),
   195→
   196→            const SizedBox(height: Spacings.lg),
   197→
   198→            // ─── Recent Transactions ──────────────────────────────
   199→            _buildRecentTransactions(context, paymentState),
   200→
   201→            // Bottom padding
   202→            const SizedBox(height: Spacings.xxl),
   203→          ],
   204→        ),
   205→      ),
   206→    );
   207→  }

  1203→  /// Formats a [DateTime] for display in the transaction list.
  1204→  String _formatTransactionDate(DateTime date) {
  1205→    final now = DateTime.now();
  1206→    final diff = now.difference(date);
  1207→
  1208→    if (diff.inDays == 0) {
  1209→      if (diff.inHours == 0) {
  1210→        if (diff.inMinutes == 0) return 'Just now';
  1211→        return '${diff.inMinutes}m ago';
  1212→      }
  1213→      return '${diff.inHours}h ago';
  1214→    } else if (diff.inDays == 1) {
  1215→      return 'Yesterday';
  1216→    } else if (diff.inDays < 7) {
  1217→      return '${diff.inDays}d ago';
  1218→    } else {
  1219→      const months = [
  1220→        '',
  1221→        'Jan',
  1222→        'Feb',
  1223→        'Mar',
  1224→        'Apr',
  1225→        'May',
  1226→        'Jun',
  1227→        'Jul',
  1228→        'Aug',
  1229→        'Sep',
  1230→        'Oct',
  1231→        'Nov',
  1232→        'Dec',
  1233→      ];
  1234→      return '${months[date.month]} ${date.day}, ${date.year}';
  1235→    }
  1236→  }
