    24→/// AI Rubric Generator page where teachers fill in parameters and AI

   106→  void _listenForMessages() {
   107→    final state = ref.read(rubricProvider);
   108→    if (state.successMessage != null) {
   109→      _showSnackBar(state.successMessage!, isError: false);
   110→      ref.read(rubricProvider.notifier).clearSuccessMessage();
   111→    }
   112→    if (state.error != null) {
   113→      _showSnackBar(state.error!, isError: true);
   114→      ref.read(rubricProvider.notifier).clearError();
   115→    }
   116→  }
   117→
   118→  void _showSnackBar(String message, {required bool isError}) {
   119→    if (!mounted) return;
   120→    final cs = context.colorScheme;
   121→    context.scaffoldMessenger.showSnackBar(
   122→      SnackBar(
   123→        content: Text(message),
   124→        backgroundColor: isError ? cs.error : cs.primary,
   125→        behavior: SnackBarBehavior.floating,
   126→        shape: RoundedRectangleBorder(
   127→          borderRadius: BorderRadius.circular(Spacings.smRadius),
   128→        ),
   129→        duration: const Duration(seconds: 3),
   130→      ),
   131→    );
   132→  }
   133→
   134→  void _initEditControllers(RubricEntity rubric) {
   135→    _rubricTitleCtrl.text = rubric.title;
   136→
   137→    // Clear old controllers
   138→    for (final c in _criterionNameControllers) {
   139→      c.dispose();
   140→    }
   141→    _criterionNameControllers.clear();
   142→    for (final c in _criterionWeightControllers) {
   143→      c.dispose();
   144→    }
   145→    _criterionWeightControllers.clear();
   146→    for (final row in _cellControllers.values) {
   147→      for (final ctrl in row.values) {
   148→        ctrl.dispose();
   149→      }
   150→    }
   151→    _cellControllers.clear();
   152→
   153→    // Create new controllers for each criterion
   154→    for (int i = 0; i < rubric.criteria.length; i++) {
   155→      final criterion = rubric.criteria[i];
   156→      _criterionNameControllers
   157→          .add(TextEditingController(text: criterion.criterion));
   158→      _criterionWeightControllers
   159→          .add(TextEditingController(text: criterion.weight.toString()));
   160→
   161→      final levelControllers = <RubricCriterionLevel, TextEditingController>{};
   162→      for (final level in criterion.levels) {
   163→        levelControllers[level.level] =
   164→            TextEditingController(text: level.description);
   165→      }
   166→      // Ensure all levels have controllers even if missing
   167→      for (final levelValue in RubricCriterionLevel.values) {
   168→        levelControllers.putIfAbsent(
   169→          levelValue,
   170→          () => TextEditingController(text: ''),
   171→        );
   172→      }
   173→      _cellControllers[criterion.criterion] = levelControllers;
   174→    }
   175→  }
   176→
   177→  Future<void> _handleGenerate() async {
   178→    if (!_formKey.currentState!.validate()) return;
   179→
   180→    await ref.read(rubricProvider.notifier).generateRubric(
   181→      GenerateRubricParams(
   182→        subjectId: _selectedSubject,
   183→        topic: _topicCtrl.text.trim(),
   184→        criteriaCount: _criteriaCount.round(),
   185→        totalPoints: double.tryParse(_totalPointsCtrl.text) ?? 100,
   186→      ),
   187→    );
   188→
   189→    _listenForMessages();
   190→
   191→    final rubric = ref.read(rubricProvider).currentRubric;
   192→    if (rubric != null) {
   193→      _initEditControllers(rubric);
   194→    }
   195→  }

   197→  void _handleSaveDraft() {
   198→    final rubric = ref.read(rubricProvider).currentRubric;
   199→    if (rubric == null) return;
   200→    _showSnackBar('Rubric saved as draft', isError: false);
   201→  }

   203→  void _handleSaveAndPublish() {
   204→    final rubric = ref.read(rubricProvider).currentRubric;
   205→    if (rubric == null) return;

   207→  }

   217→  void _handleAddCriterion() {
   218→    final rubric = ref.read(rubricProvider).currentRubric;
   219→    if (rubric == null) return;
   220→
   221→    final newCriterion = RubricCriterionEntity(
   222→      criterion: 'New Criterion',
   223→      weight: 1.0,
   224→      levels: RubricCriterionLevel.values
   225→          .map((level) => RubricLevelEntity(
   226→                level: level,
   227→                description: '',
   228→                score: 0,
   229→              ))
   230→          .toList(),
   231→    );
   232→
   233→    final updatedRubric = rubric.copyWith(
   234→      criteria: [...rubric.criteria, newCriterion],
   235→    );
   236→    ref.read(rubricProvider.notifier).setCurrentRubric(updatedRubric);
   237→    _initEditControllers(updatedRubric);
   238→    setState(() {});
   239→  }
   240→
   241→  void _handleRemoveCriterion(int index) {
   242→    final rubric = ref.read(rubricProvider).currentRubric;
   243→    if (rubric == null || rubric.criteria.length <= 1) return;
   244→
   245→    final updatedCriteria = List<RubricCriterionEntity>.from(rubric.criteria)
   246→      ..removeAt(index);
   247→    final updatedRubric = rubric.copyWith(criteria: updatedCriteria);
   248→    ref.read(rubricProvider.notifier).setCurrentRubric(updatedRubric);
   249→    _initEditControllers(updatedRubric);
   250→    setState(() {});
   251→  }
   252→
   253→  void _handleReset() {
   254→    ref.read(rubricProvider.notifier).setCurrentRubric(null);
   255→    _rubricTitleCtrl.clear();
   256→    for (final c in _criterionNameControllers) {
   257→      c.dispose();
   258→    }
   259→    _criterionNameControllers.clear();
   260→    for (final c in _criterionWeightControllers) {
   261→      c.dispose();
   262→    }
   263→    _criterionWeightControllers.clear();
   264→    for (final row in _cellControllers.values) {
   265→      for (final ctrl in row.values) {
   266→        ctrl.dispose();
   267→      }
   268→    }
   269→    _cellControllers.clear();
   270→    setState(() => _isEditing = false);
   271→  }
   272→
   273→  void _handleCellTap(String criterionName, RubricCriterionLevel level) {
   274→    setState(() => _isEditing = true);
   275→  }

   422→  Widget _buildDropdownField({
   423→    required String label,
   424→    required String hint,
   425→    required String? value,
   426→    required List<String> items,
   427→    required IconData prefixIcon,
   428→    bool isRequired = false,
   429→    required ValueChanged<String?> onChanged,
   430→  }) {
   431→    final cs = context.colorScheme;
   432→
   433→    return Column(
   434→      crossAxisAlignment: CrossAxisAlignment.start,
   435→      children: [
   436→        Row(
   437→          children: [
   438→            Icon(prefixIcon, size: Spacings.smIcon, color: cs.onSurfaceVariant),
   439→            const SizedBox(width: Spacings.xs),
   440→            Text(
   441→              label + (isRequired ? ' *' : ''),
   442→              style: context.textTheme.labelLarge?.copyWith(
   443→                color: cs.onSurfaceVariant,
   444→              ),
   445→            ),
   446→          ],
   447→        ),
   448→        const SizedBox(height: Spacings.xs),
   449→        DropdownButtonFormField<String>(
   450→          value: value,
   451→          hint: Text(hint),
   452→          decoration: InputDecoration(
   453→            border: OutlineInputBorder(
   454→              borderRadius: BorderRadius.circular(Spacings.smRadius),
   455→            ),
   456→            contentPadding: Spacings.paddingInput,
   457→          ),
   458→          items: items
   459→              .map((item) => DropdownMenuItem(
   460→                    value: item,
   461→                    child: Text(item),
   462→                  ))
   463→              .toList(),
   464→          onChanged: onChanged,
   465→          validator: isRequired
   466→              ? (v) => v == null ? '$label is required' : null
   467→              : null,
   468→        ),
   469→      ],
   470→    );
   471→  }

   749→  Widget _buildGridTable(
   750→    RubricEntity rubric,
   751→    List<RubricCriterionLevel> levels,
   752→  ) {
   753→    final cs = context.colorScheme;
   754→    const colWidth = 140.0;
   755→    const criterionWidth = 120.0;
   756→
   757→    return DataTable(
   758→      headingRowHeight: 48,
   759→      dataRowHeight: null,
   760→      columnSpacing: Spacings.sm,
   761→      horizontalMargin: Spacings.md,
   762→      columns: [
   763→        const DataColumn(
   764→          label: SizedBox(
   765→            width: criterionWidth,
   766→            child: Text('Criterion'),
   767→          ),
   768→        ),
   769→        const DataColumn(label: Text('Weight')),
   770→        ...levels.map((level) => DataColumn(
   771→              label: SizedBox(
   772→                width: colWidth,
   773→                child: Text(
   774→                  level.label,
   775→                  style: context.textTheme.labelSmall?.copyWith(
   776→                    fontWeight: AppTypography.wSemiBold,
   777→                  ),
   778→                ),
   779→              ),
   780→            )),
   781→        const DataColumn(label: Text('')),
   782→      ],
   783→      rows: rubric.criteria.asMap().entries.map((entry) {
   784→        final index = entry.key;
   785→        final criterion = entry.value;
   786→        final levelMap = <RubricCriterionLevel, String>{};
   787→        for (final l in criterion.levels) {
   788→          levelMap[l.level] = l.description;
   789→        }
   790→
   791→        return DataRow(
   792→          cells: [
   793→            // Criterion name
   794→            DataCell(
   795→              SizedBox(
   796→                width: criterionWidth,
   797→                child: _isEditing
   798→                    ? TextFormField(
   799→                        controller: _criterionNameControllers.isNotEmpty &&
   800→                                index < _criterionNameControllers.length
   801→                            ? _criterionNameControllers[index]
   802→                            : null,
   803→                        style: context.textTheme.bodySmall,
   804→                        decoration: const InputDecoration(
   805→                          isDense: true,
   806→                          border: InputBorder.none,
   807→                        ),
   808→                      )
   809→                    : Text(
   810→                        criterion.criterion,
   811→                        style: context.textTheme.bodySmall?.copyWith(
   812→                          fontWeight: AppTypography.wMedium,
   813→                        ),
   814→                      ),
   815→              ),
   816→              onTap: () => _handleCellTap(
   817→                criterion.criterion,
   818→                RubricCriterionLevel.beginning,
   819→              ),
   820→            ),
   821→            // Weight
   822→            DataCell(
   823→              _isEditing &&
   824→                      _criterionWeightControllers.isNotEmpty &&
   825→                      index < _criterionWeightControllers.length
   826→                  ? SizedBox(
   827→                      width: 50,
   828→                      child: TextFormField(
   829→                        controller: _criterionWeightControllers[index],
   830→                        style: context.textTheme.bodySmall,
   831→                        keyboardType: TextInputType.number,
   832→                        decoration: const InputDecoration(
   833→                          isDense: true,
   834→                          border: InputBorder.none,
   835→                        ),
   836→                      ),
   837→                    )
   838→                  : Text(
   839→                      '${criterion.weight}',
   840→                      style: context.textTheme.bodySmall,
   841→                    ),
   842→            ),
   843→            // Level cells
   844→            ...levels.map((level) {
   845→              final description = levelMap[level] ?? '';
   846→              final controller =
   847→                  _cellControllers[criterion.criterion]?[level];
   848→
   849→              return DataCell(
   850→                SizedBox(
   851→                  width: colWidth,
   852→                  child: _isEditing && controller != null
   853→                      ? TextFormField(
   854→                          controller: controller,
   855→                          style: context.textTheme.bodySmall?.copyWith(
   856→                            fontSize: 11,
   857→                          ),
   858→                          maxLines: 3,
   859→                          minLines: 1,
   860→                          decoration: const InputDecoration(
   861→                            isDense: true,
   862→                            border: InputBorder.none,
   863→                          ),
   864→                        )
   865→                      : Text(
   866→                          description,
   867→                          style: context.textTheme.bodySmall?.copyWith(
   868→                            fontSize: 11,
   869→                          ),
   870→                          maxLines: 3,
   871→                          overflow: TextOverflow.ellipsis,
   872→                        ),
   873→                ),
   874→                onTap: () =>
   875→                    _handleCellTap(criterion.criterion, level),
   876→              );
   877→            }),
   878→            // Delete button
   879→            DataCell(
   880→              IconButton(
   881→                icon: Icon(
   882→                  Icons.delete_outline_rounded,
   883→                  size: Spacings.smIcon,
   884→                  color: cs.error,
   885→                ),
   886→                onPressed: rubric.criteria.length > 1
   887→                    ? () => _handleRemoveCriterion(index)
   888→                    : null,
   889→                tooltip: 'Remove Criterion',
   890→                padding: EdgeInsets.zero,
   891→                constraints: const BoxConstraints(),
   892→              ),
   893→            ),
   894→          ],
   895→        );
   896→      }).toList(),
   897→    );
   898→  }
