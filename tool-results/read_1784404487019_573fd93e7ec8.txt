    17→    17→  /// Helper: Convert exceptions to Failures
    18→    18→  Failure _mapExceptionToFailure(Object e) {
    19→    19→    if (e is AuthException) {
    20→    20→      return Failure.auth(message: e.message, code: e.code);
    21→    21→    } else if (e is ServerException) {
    22→    22→      return Failure.server(message: e.message, statusCode: e.statusCode, data: e.data);
    23→    23→    } else if (e is CacheException) {
    24→    24→      return Failure.cache(message: e.message);
    25→    25→    } else if (e is NetworkException) {
    26→    26→      return Failure.network(message: e.message);
    27→    27→    } else if (e is ValidationException) {
    28→    28→      return Failure.validation(message: e.message, fieldErrors: e.fieldErrors);
    29→    29→    } else if (e is NotFoundException) {
    30→    30→      return Failure.notFound(message: e.message);
    31→    31→    } else if (e is UnauthorizedException) {
    32→    32→      return Failure.unauthorized(message: e.message);
    33→    33→    } else if (e is ForbiddenException) {
    34→    34→      return Failure.forbidden(message: e.message);
    35→    35→    } else {
    36→    36→      AppLogger.error('Unexpected exception in TeacherWorkspaceRepositoryImpl', error: e);
    37→    37→      return Failure.server(message: 'An unexpected error occurred: $e', statusCode: 500);
    38→    38→    }
    39→    39→  }
