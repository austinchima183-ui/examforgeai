   132→   132→// APP ROUTER PROVIDER
   133→   133→// ═══════════════════════════════════════════════════════════════════════
   134→   134→
   135→   135→/// Riverpod provider for the application's [GoRouter].
   136→   136→///
   137→   137→/// Watches [authStateProvider] so that the router automatically
   138→   138→/// re-evaluates its [redirect] callback whenever the authentication
   139→   139→/// state changes (sign-in, sign-out, token refresh).
   140→   140→///
   141→   141→/// Usage in `main.dart`:
   142→   142→/// ```dart
   143→   143→/// final router = ref.read(appRouterProvider);
   144→   144→/// MaterialApp.router(routerConfig: router);
   145→   145→/// ```
   146→   146→final appRouterProvider = Provider<GoRouter>((ref) {
   147→   147→  // Watch auth state so the router re-evaluates redirects on change.
   148→   148→  final authState = ref.watch(authStateProvider);
   149→   149→
   150→   150→  // Also watch onboarding state and role so those redirects stay live.
   151→   151→  ref.watch(onboardingCompleteProvider);
   152→   152→  ref.watch(userRoleProvider);
   153→   153→
   154→   154→  return GoRouter(
   155→   155→    navigatorKey: _rootNavigatorKey,
   156→   156→    initialLocation: RouteNames.splash,
   157→   157→    debugLogDiagnostics: true,
   158→   158→
   159→   159→    // ─── Global Redirect ──────────────────────────────────────────
   160→   160→    redirect: (context, state) {
   161→   161→      final currentPath = state.matchedLocation;
   162→   162→      final isAuthenticated = authState.when(
   163→   163→        data: (authData) => authData.session != null,
   164→   164→        loading: () {
   165→   165→          // While loading, only allow splash. Everything else redirects
   166→   166→          // to splash so we don't flash the login page briefly.
   167→   167→          if (currentPath != RouteNames.splash) {
   168→   168→            return RouteNames.splash;
   169→   169→          }
   170→   170→          return null;
   171→   171→        },
   172→   172→        error: (_, __) => false,
   173→   173→      );
   174→   174→
   175→   175→      // ── Auth Guard ────────────────────────────────────────────
   176→   176→      final authRedirect = AuthGuard.evaluate(
   177→   177→        isAuthenticated: isAuthenticated,
   178→   178→        currentPath: currentPath,
   179→   179→      );
   180→   180→      if (authRedirect != null) return authRedirect;
   181→   181→
   182→   182→      // ── Onboarding Guard ─────────────────────────────────────
   183→   183→      final onboardingAsync = ref.read(onboardingCompleteProvider);
   184→   184→      final isOnboardingComplete = onboardingAsync.when(
   185→   185→        data: (complete) => complete,
   186→   186→        loading: () => true, // Assume complete while loading to avoid flicker.
   187→   187→        error: (_, __) => true,
   188→   188→      );
   189→   189→
   190→   190→      final onboardingRedirect = OnboardingGuard.evaluate(
   191→   191→        isAuthenticated: isAuthenticated,
   192→   192→        isOnboardingComplete: isOnboardingComplete,
   193→   193→        currentPath: currentPath,
   194→   194→      );
   195→   195→      if (onboardingRedirect != null) return onboardingRedirect;
   196→   196→
   197→   197→      // ── Role-Based Guard ─────────────────────────────────────
   198→   198→      final roleAsync = ref.read(userRoleProvider);
   199→   199→      final userRole = roleAsync.when(
   200→   200→        data: UserRole.fromString,
   201→   201→        loading: () => null,
   202→   202→        error: (_, __) => null,
   203→   203→      );
   204→   204→
   205→   205→      final roleRedirect = RoleBasedGuard.evaluate(
   206→   206→        userRole: userRole,
   207→   207→        currentPath: currentPath,
   208→   208→      );
   209→   209→      if (roleRedirect != null) return roleRedirect;
   210→   210→
   211→   211→      // No redirect needed — proceed as requested.
   212→   212→      return null;
   213→   213→    },
   214→   214→
