   199→/// Convenience provider that yields the currently authenticated [sb.User],

   202→  final authState = ref.watch(authStateProvider);
   203→  return authState.when(
   204→    data: (state) => state.session?.user,
   205→    loading: () => SupabaseConfig.currentUser,
   206→    error: (_, __) => null,

   487→    openaiApiKey: openaiApiKey,
