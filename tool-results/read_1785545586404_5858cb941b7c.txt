   250→   250→// NAVIGATOR KEYS
   251→   251→// ═══════════════════════════════════════════════════════════════════════
   252→   252→
   253→   253→/// Root navigator key — owns the top-level navigation stack.
   254→   254→final _rootNavigatorKey = GlobalKey<NavigatorState>();
   255→   255→
   256→   256→/// Shell navigator key — owns the nested navigation stack inside the
   257→   257→/// [DashboardShell]. This allows the shell (sidebar / bottom nav) to
   258→   258→/// remain mounted while inner pages transition.
   259→   259→final _shellNavigatorKey = GlobalKey<NavigatorState>();
   260→   260→
   261→   261→// ═══════════════════════════════════════════════════════════════════════
   262→   262→// APP ROUTER PROVIDER
   263→   263→// ═══════════════════════════════════════════════════════════════════════
   264→   264→
   265→   265→/// Riverpod provider for the application's [GoRouter].
   266→   266→///
   267→   267→/// Watches [authStateProvider] so that the router automatically
   268→   268→/// re-evaluates its [redirect] callback whenever the authentication
   269→   269→/// state changes (sign-in, sign-out, token refresh).
   270→   270→///
   271→   271→/// Usage in `main.dart`:
   272→   272→/// ```dart
   273→   273→/// final router = ref.read(appRouterProvider);
   274→   274→/// MaterialApp.router(routerConfig: router);
   275→   275→/// ```
   276→   276→final appRouterProvider = Provider<GoRouter>((ref) {
   277→   277→  // Watch auth state so the router re-evaluates redirects on change.
   278→   278→  final authState = ref.watch(authStateProvider);
   279→   279→
   280→   280→  // Also watch onboarding state and role so those redirects stay live.
   281→   281→  ref.watch(onboardingCompleteProvider);
   282→   282→  ref.watch(userRoleProvider);
   283→   283→
   284→   284→  return GoRouter(
   285→   285→    navigatorKey: _rootNavigatorKey,
   286→   286→    initialLocation: RouteNames.splash,
   287→   287→    debugLogDiagnostics: true,
   288→   288→
   289→   289→    // ─── Global Redirect ──────────────────────────────────────────
   290→   290→    redirect: (context, state) {
   291→   291→      final currentPath = state.matchedLocation;
   292→   292→      // ── Loading Guard ───────────────────────────────────────
   293→   293→      // While loading, only allow splash to avoid flicker.
   294→   294→      if (authState.isLoading && currentPath != RouteNames.splash) {
   295→   295→        return RouteNames.splash;
   296→   296→      }
   297→   297→
   298→   298→      final isAuthenticated = authState.when(
   299→   299→        data: (authData) => authData.session != null,
   300→   300→        loading: () => false, // Handled by loading guard above.
   301→   301→        error: (_, __) => false,
   302→   302→      );
   303→   303→
   304→   304→      // ── Auth Guard ────────────────────────────────────────────
   305→   305→      final authRedirect = AuthGuard.evaluate(
   306→   306→        isAuthenticated: isAuthenticated,
   307→   307→        currentPath: currentPath,
   308→   308→      );
   309→   309→      if (authRedirect != null) return authRedirect;
   310→   310→
   311→   311→      // ── Onboarding Guard ─────────────────────────────────────
   312→   312→      final onboardingAsync = ref.read(onboardingCompleteProvider);
   313→   313→      final isOnboardingComplete = onboardingAsync.when(
   314→   314→        data: (complete) => complete,
   315→   315→        loading: () => true, // Assume complete while loading to avoid flicker.
   316→   316→        error: (_, __) => true,
   317→   317→      );
   318→   318→
   319→   319→      final onboardingRedirect = OnboardingGuard.evaluate(
   320→   320→        isAuthenticated: isAuthenticated,
   321→   321→        isOnboardingComplete: isOnboardingComplete,
   322→   322→        currentPath: currentPath,
   323→   323→      );
   324→   324→      if (onboardingRedirect != null) return onboardingRedirect;
   325→   325→
   326→   326→      // ── Role-Based Guard ─────────────────────────────────────
   327→   327→      final roleAsync = ref.read(userRoleProvider);
   328→   328→      final userRole = roleAsync.when(
   329→   329→        data: UserRole.fromString,
   330→   330→        loading: () => null,
   331→   331→        error: (_, __) => null,
   332→   332→      );
   333→   333→
   334→   334→      final roleRedirect = RoleBasedGuard.evaluate(
   335→   335→        userRole: userRole,
   336→   336→        currentPath: currentPath,
   337→   337→      );
   338→   338→      if (roleRedirect != null) return roleRedirect;
   339→   339→
   340→   340→      // No redirect needed — proceed as requested.
   341→   341→      return null;
