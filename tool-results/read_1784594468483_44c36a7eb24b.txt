   135→        appBar: AppAppBar(

   732→  Future<DateTime?> _pickDateTime(BuildContext context, DateTime? initial) async {
   733→    final date = await showDatePicker(
   734→      context: context,
   735→      initialDate: initial ?? DateTime.now(),
   736→      firstDate: DateTime(2024),
   737→      lastDate: DateTime(2030),
   738→    );
   739→    if (date == null) return null;
   740→
   741→    final time = await showTimePicker(
   742→      context: context,
   743→      initialTime: TimeOfDay.fromDateTime(initial ?? DateTime.now()),
   744→    );
   745→    if (time == null) return null;
   746→
   747→    return DateTime(
   748→      date.year, date.month, date.day,
   749→      time.hour, time.minute,
   750→    );

   758→  Future<void> _saveDraft() async {
   759→    if (_titleController.text.trim().isEmpty) {

   761→        context: context,

   768→    await ref.read(examBuilderProvider.notifier).saveDraft();

   776→  Future<void> _publishExam() async {
   777→    if (_titleController.text.trim().isEmpty) {
   778→      AppDialog.showError(
   779→        context: context,
   780→        title: 'Missing Information',
   781→        message: 'Please fill in all required fields before publishing.',
   782→      );

   784→    }
   785→
   786→    final confirmed = await AppDialog.showConfirm(
   787→      context: context,
   788→      title: 'Publish Exam?',
   789→      message: 'Once published, the exam cannot be edited. Students will be able to see it.',
   790→      confirmText: 'Publish',
   791→    );
   792→
   793→    if (confirmed == true) {
   794→      await ref.read(examBuilderProvider.notifier).publishExam();
   795→      if (mounted) {
   796→        ScaffoldMessenger.of(context).showSnackBar(
   797→          const SnackBar(content: Text('Exam published successfully')),
   798→        );
