    11→// ═══════════════════════════════════════════════════════════════════════
    12→// Purpose: Production notification system built on Supabase Realtime
    13→// Architecture:
    14→//   - Supabase Realtime for instant in-app notifications
    15→//   - Browser Notification API for web push notifications
    16→//   - Supabase Postgres changes for notification persistence
    17→//   - device_tokens table for web push subscription management
    18→//   - Realtime channels for role-based and school-based broadcasts
    19→//
    20→// This replaces the previous Firebase Cloud Messaging implementation.
    21→// Firebase is NOT part of the ExamForge AI architecture — the project
    22→// is Supabase-only. This service provides:
    23→//   1. Real-time notification delivery via Supabase Realtime
    24→//   2. Browser push notifications via the Web Notification API
    25→//   3. Notification persistence in the notifications table
    26→//   4. Read/unread tracking
    27→//   5. Role-based and school-based notification channels
    28→//   6. Admin broadcast notifications
    29→//   7. CBT exam notifications
    30→//   8. Payment/billing notifications
    31→//   9. Parent notifications

    52→    required this.body,
    53→    this.type,

    63→
    64→  /// Notification title.
    65→  final String title;
    66→
    67→  /// Notification body text.
    68→  final String body;

    71→  final String? type;
    72→

    88→  /// Create from a Supabase row.
    89→  factory ExamForgeNotification.fromMap(Map<String, dynamic> map) {
    90→    return ExamForgeNotification(
    91→      id: map['id']?.toString() ?? '',
    92→      title: map['title']?.toString() ?? '',
    93→      body: map['message']?.toString() ?? map['body']?.toString() ?? '',
    94→      type: map['type']?.toString(),
    95→      data: map['data'] as Map<String, dynamic>?,
    96→      referenceId: map['reference_id']?.toString(),
    97→      referenceType: map['reference_type']?.toString(),
    98→      isRead: map['is_read'] == true,
    99→      createdAt: map['created_at'] != null
   100→          ? DateTime.parse(map['created_at'].toString())

   102→    );
   103→  }

   109→      'message': body,
   110→      'type': type,
   111→      'data': data,
   112→      'reference_id': referenceId,
   113→      'reference_type': referenceType,
   114→      'is_read': isRead,
   115→    };
   116→  }

   192→  /// Initializes the notification service.
   193→  ///
   194→  /// Steps:
   195→  /// 1. Requests browser notification permissions (web).
   196→  /// 2. Registers a device token for push notifications.
   197→  /// 3. Subscribes to the user's personal notification channel.
   198→  /// 4. Subscribes to role-based and school-based channels.
   199→  /// 5. Loads the initial unread count.
   200→  Future<void> initialize({
   201→    ForegroundNotificationCallback? onForegroundNotification,
   202→    NotificationTapCallback? onNotificationTap,
   203→  }) async {
   204→    if (_initialized) {
   205→      AppLogger.warning('NotificationService already initialized — skipping.');
   206→      return;
   207→    }
   208→
   209→    _onForegroundNotification = onForegroundNotification;
   210→    _onNotificationTap = onNotificationTap;
   211→
   212→    try {
   213→      // 1. Request browser notification permissions.
   214→      final permissionGranted = await _requestPermissions();
   215→      if (!permissionGranted) {
   216→        AppLogger.warning('Notification permissions not granted.');
   217→      }
   218→
   219→      // 2. Register device token.
   220→      await _registerDeviceToken();
   221→
   222→      // 3. Subscribe to user's personal notification channel.
   223→      await _subscribeToUserChannel();
   224→
   225→      // 4. Load initial unread count.
   226→      await _loadUnreadCount();
   227→
   228→      _initialized = true;
   229→      AppLogger.info('NotificationService initialized successfully');
   230→    } catch (e) {
   231→      AppLogger.error('Failed to initialize NotificationService', error: e);
   232→      // Mark initialized anyway so the app can continue — push
   233→      // notifications are a non-critical feature.
   234→      _initialized = true;
   235→    }
   236→  }

   320→  // ─── Realtime Channel Subscriptions ─────────────────────────────
   321→
   322→  /// Subscribes to the user's personal notification channel.
   323→  ///
   324→  /// This channel receives notifications targeted at the specific user
   325→  /// (e.g., "Your exam has been graded", "Payment received").
   326→  Future<void> _subscribeToUserChannel() async {
   327→    final userId = _supabase.auth.currentUser?.id;
   328→    if (userId == null) return;
   329→
   330→    try {
   331→      _notificationChannel = _supabase.channel(
   332→        'notifications:$userId',
   333→        opts: const sb.RealtimeChannelConfig(self: true),
   334→      );
   335→
   336→      _notificationChannel!.onPostgresChanges(
   337→        event: sb.PostgresChangeEvent.insert,
   338→        schema: 'public',
   339→        table: 'notifications',
   340→        filter: sb.PostgresChangeFilter(
   341→          type: sb.PostgresChangeFilterType.eq,
   342→          column: 'user_id',
   343→          value: userId,
   344→        ),
   345→        callback: (payload) {
   346→          _handleIncomingNotification(payload.newRecord);
   347→        },
   348→      );
   349→
   350→      _notificationChannel!.subscribe((status, error) {
   351→        if (status == sb.RealtimeSubscribeStatus.subscribed) {
   352→          AppLogger.info('User notification channel subscribed');
   353→        } else if (status == sb.RealtimeSubscribeStatus.channelError) {
   354→          AppLogger.error('User notification channel error', error: error);
   355→        }
   356→      });
   357→
   358→      _subscribedChannelNames.add('notifications:$userId');
   359→    } catch (e) {
   360→      AppLogger.error('Failed to subscribe to user notification channel', error: e);
   361→    }
   362→  }

   493→  /// Handles an incoming notification from Supabase Realtime.
   494→  void _handleIncomingNotification(Map<String, dynamic> record) {
   495→    try {
   496→      final notification = ExamForgeNotification.fromMap(record);
   497→
   498→      AppLogger.info('Notification received: ${notification.id} — ${notification.title}');
   499→
   500→      // Emit to the stream for reactive UI updates.
   501→      if (!_notificationController.isClosed) {
   502→        _notificationController.add(notification);
   503→      }
   504→
   505→      // Invoke the foreground callback.
   506→      _onForegroundNotification?.call(notification);
   507→
   508→      // Update unread count.
   509→      if (!notification.isRead) {
   510→        _unreadCount++;
   511→        if (!_unreadCountController.isClosed) {
   512→          _unreadCountController.add(_unreadCount);
   513→        }
   514→      }
   515→
   516→      // Display browser notification on web.
   517→      _displayBrowserNotification(notification);
   518→    } catch (e) {
   519→      AppLogger.error('Failed to handle incoming notification', error: e);
   520→    }
   521→  }

   537→  /// Loads the initial unread count from Supabase.
   538→  Future<void> _loadUnreadCount() async {
   539→    try {
   540→      final userId = _supabase.auth.currentUser?.id;
   541→      if (userId == null) return;
   542→
   543→      final response = await _supabase
   544→          .from('notifications')
   545→          .select('id')
   546→          .eq('user_id', userId)
   547→          .eq('is_read', false);
   548→
   549→      _unreadCount = response.length;
   550→      if (!_unreadCountController.isClosed) {
   551→        _unreadCountController.add(_unreadCount);
   552→      }
   553→    } catch (e) {
   554→      AppLogger.error('Failed to load unread count', error: e);
   555→    }
   556→  }

   558→  /// Marks a notification as read.
   559→  Future<void> markAsRead(String notificationId) async {
   560→    try {
   561→      await _supabase
   562→          .from('notifications')
   563→          .update({'is_read': true}).eq('id', notificationId);
   564→
   565→      _unreadCount = (_unreadCount - 1).clamp(0, _unreadCount);
   566→      if (!_unreadCountController.isClosed) {
   567→        _unreadCountController.add(_unreadCount);
   568→      }
   569→    } catch (e) {
   570→      AppLogger.error('Failed to mark notification as read', error: e);
   571→    }

   595→  /// Gets notifications for the current user, optionally filtered.
   596→  Future<List<ExamForgeNotification>> getNotifications({
   597→    int limit = 50,
   598→    int offset = 0,
   599→    String? type,
   600→    bool? isRead,
   601→  }) async {
   602→    try {
   603→      final userId = _supabase.auth.currentUser?.id;
   604→      if (userId == null) return [];
   605→
   606→      // Build the query with all filters applied before ordering/ranging.
   607→      // Supabase query builder: .select() returns PostgrestFilterBuilder,
   608→      // and filters must be applied before .order() and .range().
   609→      var query = _supabase
   610→          .from('notifications')
   611→          .select()
   612→          .eq('user_id', userId);
   613→
   614→      if (type != null) {
   615→        query = query.eq('type', type);
   616→      }
   617→      if (isRead != null) {
   618→        query = query.eq('is_read', isRead);
   619→      }
   620→
   621→      final response = await query
   622→          .order('created_at', ascending: false)
   623→          .range(offset, offset + limit - 1);
   624→      return response.map<ExamForgeNotification>((row) {
   625→        return ExamForgeNotification.fromMap(row);
   626→      }).toList();
   627→    } catch (e) {
   628→      AppLogger.error('Failed to get notifications', error: e);
   629→      return [];
   630→    }
   631→  }

   635→  /// Sends a broadcast notification to all users in a role.
   636→  ///
   637→  /// This inserts a notification for each user with the given role.
   638→  /// The Supabase Realtime channel will then deliver it instantly.
   639→  Future<void> broadcastToRole({
   640→    required String role,
   641→    required String title,
   642→    required String body,
   643→    String? type,
   644→    Map<String, dynamic>? data,
   645→  }) async {
   646→    try {
   647→      // Get all users with the given role.
   648→      final users = await _supabase
   649→          .from('user_profiles')
   650→          .select('user_id')
   651→          .eq('role', role);
   652→
   653→      final notifications = users.map<Map<String, dynamic>>((user) {
   654→        return {
   655→          'user_id': user['user_id'],
   656→          'title': title,
   657→          'message': body,
   658→          'type': type ?? 'broadcast',
   659→          'data': data,
   660→        };
   661→      }).toList();
   662→
   663→      if (notifications.isNotEmpty) {
   664→        // Insert in batches to avoid payload limits.
   665→        const batchSize = 100;
   666→        for (var i = 0; i < notifications.length; i += batchSize) {
   667→          final batch = notifications.sublist(
   668→            i,
   669→            i + batchSize > notifications.length
   670→                ? notifications.length
   671→                : i + batchSize,
   672→          );
   673→          await _supabase.from('notifications').insert(batch);
   674→        }
   675→      }
   676→
   677→      AppLogger.info('Broadcast notification sent to ${users.length} users with role: $role');
   678→    } catch (e) {
   679→      AppLogger.error('Failed to broadcast to role: $role', error: e);
   680→    }
   681→  }

   748→        'marketplace': response['marketplace_enabled'] as bool? ?? true,
   749→        'communication': response['communication_enabled'] as bool? ?? true,

   753→      return {};

   782→  /// Persists the device token to the `device_tokens` table in Supabase.
   783→  Future<void> _syncTokenToSupabase(String token) async {
   784→    try {
   785→      final userId = _supabase.auth.currentUser?.id;
   786→      if (userId == null) {
   787→        AppLogger.warning('Cannot sync device token — no authenticated user.');
   788→        return;
   789→      }
   790→
   791→      await _supabase.from('device_tokens').upsert(
   792→        {
   793→          'user_id': userId,
   794→          'fcm_token': token,
   795→          'platform': kIsWeb ? 'web' : 'native',
   796→          'updated_at': DateTime.now().toIso8601String(),
   797→        },
   798→        onConflict: 'user_id,fcm_token',
   799→      );
   800→
   801→      AppLogger.info('Device token synced to Supabase');
   802→    } on sb.PostgrestException catch (e) {
   803→      AppLogger.error('Supabase device token sync failed', error: e);
   804→    } catch (e) {
   805→      AppLogger.error('Unexpected device token sync error', error: e);
   806→    }
   807→  }
   808→
   809→  /// Removes the device token from Supabase (e.g. on logout).
   810→  Future<void> removeTokenFromSupabase() async {
   811→    try {
   812→      final userId = _supabase.auth.currentUser?.id;
   813→      if (userId == null || _currentToken == null) return;
   814→
   815→      await _supabase
   816→          .from('device_tokens')
   817→          .delete()
   818→          .eq('user_id', userId)
   819→          .eq('fcm_token', _currentToken!);
   820→
   821→      AppLogger.info('Device token removed from Supabase');
   822→    } on sb.PostgrestException catch (e) {
   823→      AppLogger.error('Failed to remove device token from Supabase', error: e);
   824→    } catch (e) {
   825→      AppLogger.error('Unexpected error removing device token', error: e);
   826→    }
   827→  }

   857→  Future<void> _restoreSubscribedChannels() async {
   858→    try {
   859→      final channelsString =
   860→          await _storage.readPreference(_subscribedChannelsKey);
   861→      if (channelsString == null || channelsString.isEmpty) return;
   862→
   863→      final channels = channelsString.split(',');
   864→      for (final channel in channels) {
   865→        final trimmed = channel.trim();
   866→        if (trimmed.isNotEmpty) {
   867→          _subscribedChannelNames.add(trimmed);
   868→        }
   869→      }
   870→
   871→      AppLogger.info(
   872→        'Restored ${_subscribedChannelNames.length} channel subscriptions',
   873→      );
   874→    } catch (e) {
   875→      AppLogger.error('Failed to restore subscribed channels', error: e);
   876→    }
   877→  }

   881→  /// Disposes all stream subscriptions and controllers.
   882→  void dispose() {
   883→    _notificationController.close();
   884→    _unreadCountController.close();
   885→
   886→    // Unsubscribe from all Realtime channels.
   887→    if (_notificationChannel != null) {
   888→      _supabase.removeChannel(_notificationChannel!);
   889→    }
   890→    if (_roleChannel != null) {
   891→      _supabase.removeChannel(_roleChannel!);
   892→    }
   893→    if (_schoolChannel != null) {
   894→      _supabase.removeChannel(_schoolChannel!);
   895→    }
   896→    for (final channel in _customChannels.values) {
   897→      _supabase.removeChannel(channel);
   898→    }
   899→    _customChannels.clear();
   900→
   901→    AppLogger.info('NotificationService disposed');
   902→  }

   904→  /// Full cleanup for sign-out: removes the token from Supabase,
   905→  /// unsubscribes from all channels, and disposes resources.
   906→  Future<void> handleSignOut() async {
   907→    try {
   908→      await removeTokenFromSupabase();
   909→      await unsubscribeFromAllChannels();
   910→      await _storage.writeSecure(_deviceTokenKey, '');
   911→      await _storage.writePreference(_subscribedChannelsKey, '');
   912→      _currentToken = null;
   913→      dispose();
   914→      AppLogger.info('NotificationService cleaned up for sign-out');
   915→    } catch (e) {
   916→      AppLogger.error('Error during notification sign-out cleanup', error: e);
   917→    }
   918→  }
