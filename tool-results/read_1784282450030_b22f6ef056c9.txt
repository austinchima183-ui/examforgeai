   474→    );
   475→    return AiProvidersRegistry.createEmpty();
   476→  }
   477→
   478→  return AiProvidersRegistry.createDefault(
   479→    openaiApiKey: openaiApiKey,
   480→    geminiApiKey: geminiApiKey,
   481→  );
