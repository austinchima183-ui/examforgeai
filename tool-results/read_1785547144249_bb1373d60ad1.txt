    12→-- ============================================================================
    13→-- 1. CUSTOM ENUMERATION TYPES
    14→-- ============================================================================
    15→
    16→DO $$
    17→BEGIN
    18→  -- user_role: Defines the hierarchy of user permissions
    19→  IF NOT EXISTS (SELECT 1 FROM pg_type WHERE typname = 'user_role') THEN
    20→    CREATE TYPE user_role AS ENUM (
    21→      'super_admin',
    22→      'school_admin',
    23→      'teacher',
    24→      'student'
    25→    );
    26→  END IF;
    27→
    28→  -- subscription_status: School subscription tier
    29→  IF NOT EXISTS (SELECT 1 FROM pg_type WHERE typname = 'subscription_status') THEN
    30→    CREATE TYPE subscription_status AS ENUM (
    31→      'free',
    32→      'basic',
    33→      'premium',
    34→      'enterprise'
    35→    );
    36→  END IF;
    37→
    38→  -- exam_status: Lifecycle states for an exam
    39→  IF NOT EXISTS (SELECT 1 FROM pg_type WHERE typname = 'exam_status') THEN
    40→    CREATE TYPE exam_status AS ENUM (
    41→      'draft',
    42→      'published',
    43→      'active',
    44→      'completed',
    45→      'archived'
    46→    );
    47→  END IF;
    48→
    49→  -- notification_type: Category of notification
    50→  IF NOT EXISTS (SELECT 1 FROM pg_type WHERE typname = 'notification_type') THEN
    51→    CREATE TYPE notification_type AS ENUM (
    52→      'exam',
    53→      'system',
    54→      'result',
    55→      'reminder'
    56→    );
    57→  END IF;
    58→
    59→  -- question_type: Supported question formats
    60→  IF NOT EXISTS (SELECT 1 FROM pg_type WHERE typname = 'question_type') THEN
    61→    CREATE TYPE question_type AS ENUM (
    62→      'multiple_choice',
    63→      'true_false',
    64→      'short_answer',
    65→      'essay',
    66→      'fill_in_blank'
    67→    );
    68→  END IF;
    69→END
    70→$$;

   294→-- ---------------------------------------------------------------------------
   295→-- update_updated_at_column()
   296→-- Automatically sets updated_at = now() on every row update.
   297→-- ---------------------------------------------------------------------------
   298→CREATE OR REPLACE FUNCTION update_updated_at_column()
   299→RETURNS TRIGGER AS $$
   300→BEGIN
   301→  NEW.updated_at = now();
   302→  RETURN NEW;
   303→END;
   304→$$ LANGUAGE plpgsql;
   305→
   306→COMMENT ON FUNCTION update_updated_at_column() IS
   307→  'Trigger function: automatically refreshes updated_at to the current timestamp';
   308→
   309→-- ---------------------------------------------------------------------------
   310→-- is_school_member(school_id)
   311→-- Returns TRUE if the currently authenticated user belongs to the given school.
   312→-- ---------------------------------------------------------------------------
   313→CREATE OR REPLACE FUNCTION is_school_member(target_school_id UUID)
   314→RETURNS BOOLEAN AS $$
   315→BEGIN
   316→  RETURN EXISTS (
   317→    SELECT 1
   318→    FROM users
   319→    WHERE id = auth.uid()
   320→      AND school_id = target_school_id
   321→  );
   322→END;
   323→$$ LANGUAGE plpgsql SECURITY DEFINER STABLE;
   324→
   325→COMMENT ON FUNCTION is_school_member(UUID) IS
   326→  'Checks whether the current auth user is a member of the specified school';
   327→
   328→-- ---------------------------------------------------------------------------
   329→-- get_user_role()
   330→-- Returns the user_role of the currently authenticated user.
   331→-- ---------------------------------------------------------------------------
   332→CREATE OR REPLACE FUNCTION get_user_role()
   333→RETURNS user_role AS $$
   334→DECLARE
   335→  current_role user_role;
   336→BEGIN
   337→  SELECT role INTO current_role
   338→  FROM users
   339→  WHERE id = auth.uid();
   340→
   341→  RETURN current_role;
   342→END;
   343→$$ LANGUAGE plpgsql SECURITY DEFINER STABLE;
   344→
   345→COMMENT ON FUNCTION get_user_role() IS
   346→  'Returns the role of the currently authenticated user';
   347→
   348→-- ---------------------------------------------------------------------------
   349→-- handle_new_user()
   350→-- Trigger function: automatically creates a user profile row in public.users
   351→-- after a new row is inserted into auth.users.
   352→-- ---------------------------------------------------------------------------
   353→CREATE OR REPLACE FUNCTION handle_new_user()
   354→RETURNS TRIGGER AS $$
   355→BEGIN
   356→  INSERT INTO public.users (id, email, full_name, role, is_email_verified)
   357→  VALUES (
   358→    NEW.id,
   359→    NEW.email,
   360→    COALESCE(NEW.raw_user_meta_data ->> 'full_name', NEW.email),
   361→    COALESCE(
   362→      (NEW.raw_user_meta_data ->> 'role')::user_role,
   363→      'student'
   364→    ),
   365→    NEW.email_confirmed_at IS NOT NULL
   366→  );
   367→  RETURN NEW;
   368→END;
   369→$$ LANGUAGE plpgsql SECURITY DEFINER;
   370→
   371→COMMENT ON FUNCTION handle_new_user() IS
   372→  'Trigger: auto-creates a public.users profile when a new auth.users row is inserted';

   437→-- ===========================================================================
   438→-- SCHOOLS RLS POLICIES
   439→-- ===========================================================================
   440→
   441→-- Any authenticated user can read active schools
   442→CREATE POLICY "Authenticated users can read active schools"
   443→  ON schools FOR SELECT
   444→  TO authenticated
   445→  USING (is_active = true);
   446→
   447→-- Super admins can read all schools (including inactive)
   448→CREATE POLICY "Super admins can read all schools"
   449→  ON schools FOR SELECT
   450→  TO authenticated
   451→  USING (get_user_role() = 'super_admin');
   452→
   453→-- School admins can update their own school
   454→CREATE POLICY "School admins can update own school"
   455→  ON schools FOR UPDATE
   456→  TO authenticated
   457→  USING (
   458→    role = 'school_admin'
   459→    AND school_id = id
   460→  )
   461→  WITH CHECK (
   462→    role = 'school_admin'
   463→    AND school_id = id
   464→  );
   465→
   466→-- Super admins can do everything on schools
   467→CREATE POLICY "Super admins have full access to schools"
   468→  ON schools FOR ALL
   469→  TO authenticated
   470→  USING (get_user_role() = 'super_admin')
   471→  WITH CHECK (get_user_role() = 'super_admin');
   472→
   473→-- Super admins can insert schools
   474→CREATE POLICY "Super admins can insert schools"
   475→  ON schools FOR INSERT
   476→  TO authenticated
   477→  WITH CHECK (get_user_role() = 'super_admin');

   483→-- Users can read their own row

   486→  TO authenticated
   487→  USING (id = auth.uid());

   489→-- School admins can read users in their school

   493→  USING (
   494→    get_user_role() = 'school_admin'
   495→    AND school_id = (SELECT school_id FROM users WHERE id = auth.uid())
   496→    AND school_id IS NOT NULL
   497→  );
   498→
   499→-- Super admins can read all users
   500→CREATE POLICY "Super admins can read all users"

   502→  TO authenticated
   503→  USING (get_user_role() = 'super_admin');
   504→
   505→-- Users can update their own row (except role)

   508→  TO authenticated

   518→  TO authenticated
   519→  USING (get_user_role() = 'super_admin')
   520→  WITH CHECK (get_user_role() = 'super_admin');

   522→-- Super admins can delete users

   524→  ON users FOR DELETE
   525→  TO authenticated
   526→  USING (get_user_role() = 'super_admin');
