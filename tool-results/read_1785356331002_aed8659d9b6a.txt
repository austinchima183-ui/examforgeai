     7→bool _mapEquals(Map<String, dynamic>? a, Map<String, dynamic>? b) {
     8→  if (identical(a, b)) return true;
     9→  if (a == null || b == null) return false;
    10→  if (a.length != b.length) return false;
    11→  for (final key in a.keys) {
    12→    if (!b.containsKey(key)) return false;
    13→    final aVal = a[key];
    14→    final bVal = b[key];
    15→    if (aVal is Map<String, dynamic> && bVal is Map<String, dynamic>) {
    16→      if (!_mapEquals(aVal, bVal)) return false;
    17→    } else if (aVal is List && bVal is List) {
    18→      if (aVal.length != bVal.length) return false;
    19→      for (var i = 0; i < aVal.length; i++) {
    20→        if (aVal[i] is Map<String, dynamic> &&
    21→            bVal[i] is Map<String, dynamic>) {
    22→          if (!_mapEquals(aVal[i] as Map<String, dynamic>,
    23→              bVal[i] as Map<String, dynamic>,)) {
    24→            return false;
    25→          }
    26→        } else if (aVal[i] != bVal[i]) {
    27→          return false;
    28→        }
    29→      }
    30→    } else if (aVal != bVal) {
    31→      return false;
    32→    }
    33→  }
    34→  return true;
    35→}
    36→
    37→bool _listEquals<T>(List<T> a, List<T> b) {
    38→  if (a.length != b.length) return false;
    39→  for (var i = 0; i < a.length; i++) {
    40→    if (a[i] is Map<String, dynamic> && b[i] is Map<String, dynamic>) {
    41→      if (!_mapEquals(a[i] as Map<String, dynamic>,
    42→          b[i] as Map<String, dynamic>,)) {
    43→        return false;
    44→      }
    45→    } else if (a[i] != b[i]) {
    46→      return false;
    47→    }
    48→  }
    49→  return true;
    50→}
