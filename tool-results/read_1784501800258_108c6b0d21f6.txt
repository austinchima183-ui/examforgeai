   210→  // ─── Helper: Handle Supabase errors ────────────────────────────────
   211→
   212→  Never _handlePostgrestError(sb.PostgrestException e) {
   213→    AppLogger.error('Supabase error: ${e.message}', error: e);
   214→    if (e.code == '401') {
   215→      throw UnauthorizedException(e.message);
   216→    } else if (e.code == '403') {
   217→      throw ForbiddenException(e.message);
   218→    } else if (e.code == '404' || e.code == 'PGRST116') {
   219→      throw NotFoundException(e.message);
   220→    }
   221→    throw ServerException(
   222→      message: e.message,
   223→      statusCode: int.tryParse(e.code ?? '500') ?? 500,
   224→      data: e.details,
   225→    );
   226→  }

   515→        } else {
   516→          query = query.eq('parent_topic_id', parentId as String);
   517→        }
   518→      }
   519→
   520→      final response = await query.order('sort_order');
   521→      return List<Map<String, dynamic>>.from(response);
   522→    } on sb.PostgrestException catch (e) {
