   119→   119→      case '23505':

   126→   126→          message: 'Referenced record not found.',

   134→   134→        throw ServerException(
   135→   135→          message: e.message,

   240→   240→  @override
   241→   241→  Future<List<LessonPlanModel>> getLessonPlans(
   242→   242→    Map<String, dynamic> filters,
   243→   243→  ) async {
   244→   244→    try {
   245→   245→      AppLogger.info('Fetching lesson plans with filters: $filters');
   246→   246→      var query = _supabase.from(_lessonPlansTable).select();
   247→   247→
   248→   248→      // Search filter
   249→   249→      if (filters['searchQuery'] != null) {
   250→   250→        query = query.ilike('title', '%${filters['searchQuery']}%');
   251→   251→      }
   252→   252→
   253→   253→      // Subject filter
   254→   254→      if (filters['subject_id'] != null) {
   255→   255→        query = query.eq('subject_id', filters['subject_id']);
   256→   256→      }
   257→   257→
   258→   258→      // Class filter
   259→   259→      if (filters['class_id'] != null) {
   260→   260→        query = query.eq('class_id', filters['class_id']);
   261→   261→      }
   262→   262→
   263→   263→      // Published filter
   264→   264→      if (filters['is_published'] != null) {
   265→   265→        query = query.eq('is_published', filters['is_published']);
   266→   266→      }
   267→   267→
   268→   268→      // Archived filter
   269→   269→      if (filters['is_archived'] != null) {
   270→   270→        query = query.eq('is_archived', filters['is_archived']);
   271→   271→      }
   272→   272→
   273→   273→      // Tags overlap filter
   274→   274→      if (filters['tags'] != null) {
   275→   275→        query = query.overlaps('tags', filters['tags']);
   276→   276→      }
   277→   277→
   278→   278→      // Ordering
   279→   279→      query = query.order('created_at', ascending: false);
   280→   280→
   281→   281→      // Pagination
   282→   282→      final page = filters['page'] as int? ?? 1;
   283→   283→      final perPage = filters['perPage'] as int? ?? 20;
   284→   284→      query = query.range((page - 1) * perPage, page * perPage - 1);
   285→   285→
   286→   286→      final response = await query;
   287→   287→      AppLogger.info('Fetched ${response.length} lesson plans');
   288→   288→      return response.map(LessonPlanModel.fromJson).toList();
   289→   289→    } on sb.PostgrestException catch (e) {
   290→   290→      _mapPostgrestException(e);
   291→   291→    } catch (e) {
   292→   292→      AppLogger.error('Failed to fetch lesson plans', error: e);
   293→   293→      throw ServerException(message: e.toString(), statusCode: 500);
   294→   294→    }
   295→   295→  }
