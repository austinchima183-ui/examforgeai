    28→  @override
    29→  Future<Result<QuestionEntity>> createQuestion(
    30→    QuestionEntity question,
    31→  ) async {
    32→    try {
    33→      final model = QuestionModel.fromEntity(question);
    34→      final created = await _remoteDataSource.createQuestion(model.toJson());
    35→      return Success(created.toEntity());
    36→    } on AuthException catch (e) {
    37→      return FailureResult(Failure.auth(message: e.message, code: e.code));
    38→    } on ServerException catch (e) {
    39→      return FailureResult(Failure.server(
    40→        message: e.message,
    41→        statusCode: e.statusCode,
    42→        data: e.data,
    43→      ));
    44→    } on NetworkException catch (e) {
    45→      return FailureResult(Failure.network(message: e.message));
    46→    } on ValidationException catch (e) {
    47→      return FailureResult(Failure.validation(
    48→        message: e.message,
    49→        fieldErrors: e.fieldErrors,
    50→      ));
    51→    } on NotFoundException catch (e) {
    52→      return FailureResult(Failure.notFound(message: e.message));
    53→    } on UnauthorizedException catch (e) {
    54→      return FailureResult(Failure.unauthorized(message: e.message));
    55→    } on ForbiddenException catch (e) {
    56→      return FailureResult(Failure.forbidden(message: e.message));
    57→    } catch (e) {

    59→          error: e);
    60→      return const FailureResult(Failure.server(
    61→        message: 'An unexpected error occurred.',
    62→        statusCode: 500,
