  1200→-- Get conversations for a user
  1201→CREATE OR REPLACE FUNCTION get_user_conversations(
  1202→  p_user_id UUID,
  1203→  p_page INT DEFAULT 1,
  1204→  p_per_page INT DEFAULT 20
  1205→)
  1206→RETURNS JSONB AS $$
  1207→DECLARE
  1208→  v_result JSONB;
  1209→BEGIN
  1210→  SELECT jsonb_agg(conv_data) INTO v_result
  1211→  FROM (
  1212→    SELECT jsonb_build_object(
  1213→      'id', c.id,
  1214→      'type', c.type,
  1215→      'name', c.name,
  1216→      'avatar_url', c.avatar_url,
  1217→      'last_message_text', c.last_message_text,
  1218→      'last_message_at', c.last_message_at,
  1219→      'last_sender_name', c.last_sender_name,
  1220→      'unread_count', cp.unread_count,
  1221→      'is_muted', cp.is_muted,
  1222→      'is_pinned', cp.is_pinned,
  1223→      'is_archived', cp.is_archived,
  1224→      'is_typing', cp.is_typing,
  1225→      'participants', (
  1226→        SELECT COALESCE(jsonb_agg(
  1227→          jsonb_build_object(
  1228→            'user_id', cp2.user_id,
  1229→            'role', cp2.role,
  1230→            'is_online', cp2.is_online,
  1231→            'last_seen_at', cp2.last_seen_at
  1232→          )
  1233→        ), '[]'::jsonb)
  1234→        FROM conversation_participants cp2
  1235→        WHERE cp2.conversation_id = c.id AND cp2.left_at IS NULL
  1236→      )
  1237→    ) AS conv_data
  1238→    FROM conversations c
  1239→    JOIN conversation_participants cp ON cp.conversation_id = c.id
  1240→    WHERE cp.user_id = p_user_id AND cp.left_at IS NULL
  1241→    ORDER BY cp.is_pinned DESC, c.last_message_at DESC NULLS LAST
  1242→    LIMIT p_per_page OFFSET (p_page - 1) * p_per_page
  1243→  ) sub;
  1244→
  1245→  RETURN COALESCE(v_result, '[]'::jsonb);
  1246→END;
  1247→$$ LANGUAGE plpgsql SECURITY DEFINER;

  1249→-- Get messages for a conversation
  1250→CREATE OR REPLACE FUNCTION get_conversation_messages(
  1251→  p_conversation_id UUID,
  1252→  p_page INT DEFAULT 1,
  1253→  p_per_page INT DEFAULT 50,
  1254→  p_before TIMESTAMPTZ DEFAULT NULL
  1255→)
  1256→RETURNS JSONB AS $$
  1257→DECLARE
  1258→  v_result JSONB;
  1259→BEGIN
  1260→  SELECT jsonb_agg(msg_data) INTO v_result
  1261→  FROM (
  1262→    SELECT jsonb_build_object(
  1263→      'id', m.id,
  1264→      'conversation_id', m.conversation_id,
  1265→      'sender_id', m.sender_id,
  1266→      'sender_name', m.sender_name,
  1267→      'sender_avatar', m.sender_avatar,
  1268→      'sender_role', m.sender_role,
  1269→      'type', m.type,
  1270→      'body', m.body,
  1271→      'reply_to_id', m.reply_to_id,
  1272→      'is_pinned', m.is_pinned,
  1273→      'is_edited', m.is_edited,
  1274→      'is_deleted', m.is_deleted,
  1275→      'read_by', m.read_by,
  1276→      'reactions', (
  1277→        SELECT COALESCE(jsonb_agg(
  1278→          jsonb_build_object('emoji', mr.emoji, 'user_id', mr.user_id, 'created_at', mr.created_at)
  1279→        ), '[]'::jsonb)
  1280→        FROM message_reactions mr WHERE mr.message_id = m.id
  1281→      ),
  1282→      'attachments', (
  1283→        SELECT COALESCE(jsonb_agg(
  1284→          jsonb_build_object(
  1285→            'id', ma.id, 'file_name', ma.file_name, 'file_url', ma.file_url,
  1286→            'file_type', ma.file_type, 'file_size_bytes', ma.file_size_bytes,
  1287→            'thumbnail_url', ma.thumbnail_url, 'is_previewable', ma.is_previewable
  1288→          )
  1289→        ), '[]'::jsonb)
  1290→        FROM message_attachments ma WHERE ma.message_id = m.id
  1291→      ),
  1292→      'reply_to', CASE WHEN m.reply_to_id IS NOT NULL THEN (
  1293→        SELECT jsonb_build_object(
  1294→          'id', rm.id, 'sender_name', rm.sender_name, 'body', LEFT(rm.body, 80), 'type', rm.type
  1295→        )
  1296→        FROM messages rm WHERE rm.id = m.reply_to_id
  1297→      ) END,
  1298→      'created_at', m.created_at,
  1299→      'updated_at', m.updated_at
  1300→    ) AS msg_data
  1301→    FROM messages m
  1302→    WHERE m.conversation_id = p_conversation_id
  1303→    AND m.is_deleted = FALSE
  1304→    AND (p_before IS NULL OR m.created_at < p_before)
  1305→    ORDER BY m.created_at DESC
  1306→    LIMIT p_per_page OFFSET (p_page - 1) * p_per_page
  1307→  ) sub;
  1308→
  1309→  RETURN COALESCE(v_result, '[]'::jsonb);
  1310→END;
  1311→$$ LANGUAGE plpgsql SECURITY DEFINER;
