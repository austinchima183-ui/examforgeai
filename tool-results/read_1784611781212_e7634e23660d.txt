   201→  @override
   202→  Future<GenerationRequestModel> createGenerationRequest(
   203→    Map<String, dynamic> requestData,
   204→  ) async {
   205→    try {
   206→      final response = await _supabase
   207→          .from(_generationRequestsTable)
   208→          .insert(requestData)
   209→          .select();
   210→
   211→      if (response.isEmpty) {
   212→        throw const ServerException(
   213→          message: 'Generation request creation returned no data.',
   214→          statusCode: 500,
   215→        );
   216→      }
   217→
   218→      AppLogger.info('Generation request created: ${response.first['id']}');
   219→      return GenerationRequestModel.fromJson(response.first);
   220→    } on sb.PostgrestException catch (e) {
   221→      throw _mapPostgrestException(e);
   222→    } on ServerException {
   223→      rethrow;
   224→    } catch (e) {
   225→      AppLogger.error('Unexpected createGenerationRequest error', error: e);
   226→      throw const ServerException(
   227→        message: 'Failed to create generation request.',
   228→        statusCode: 500,
   229→      );
   230→    }
   231→  }

   292→  @override
   293→  Future<List<GenerationRequestModel>> getGenerationRequests(
   294→    Map<String, dynamic> filters,
   295→  ) async {
   296→    try {
   297→      var query = _supabase
   298→          .from(_generationRequestsTable)
   299→          .select()
   300→          .order('created_at', ascending: false);
   301→
   302→      // Apply filters
   303→      if (filters.containsKey('school_id')) {
   304→        query = query.eq('school_id', filters['school_id'] as String);
   305→      }
   306→      if (filters.containsKey('status')) {
   307→        query = query.eq('status', filters['status'] as String);
   308→      }
   309→      if (filters.containsKey('requested_by')) {
   310→        query = query.eq('requested_by', filters['requested_by'] as String);
   311→      }
   312→
   313→      // Pagination
   314→      final page = filters['page'] as int? ?? 1;
   315→      final perPage = filters['per_page'] as int? ?? filters['perPage'] as int? ?? 20;
   316→      final from = (page - 1) * perPage;
   317→      query = query.range(from, from + perPage - 1);
   318→
   319→      final response = await query;
   320→      return response
   321→          .map<GenerationRequestModel>(
   322→              (json) => GenerationRequestModel.fromJson(json))
   323→          .toList();
   324→    } on sb.PostgrestException catch (e) {
   325→      throw _mapPostgrestException(e);
   326→    } catch (e) {
   327→      AppLogger.error('Unexpected getGenerationRequests error', error: e);
   328→      throw const ServerException(
   329→        message: 'Failed to get generation requests.',
   330→        statusCode: 500,
   331→      );
   332→    }
   333→  }

  1494→      return GenerationQueueModel.fromJson(
  1495→          updated.isNotEmpty ? updated.first : entry);

  1611→  @override
  1612→  Future<Map<String, dynamic>> getDashboardStats(String? schoolId) async {
  1613→    try {
  1614→      // Use an RPC call if available, otherwise aggregate manually
  1615→      try {
  1616→        final response = await _supabase.rpc(
  1617→          'get_ai_dashboard_stats',
  1618→          params: schoolId != null ? {'p_school_id': schoolId} : {},
  1619→        );
  1620→        return response as Map<String, dynamic>;
  1621→      } on sb.PostgrestException catch (e) {
  1622→        if (e.code != '42883') rethrow; // not undefined function
  1623→
  1624→        // Fallback: manually aggregate stats
  1625→        final questionsQuery = _supabase
  1626→            .from(_generatedQuestionsTable)
  1627→            .select('review_status, question_type, difficulty, bloom_level, '
  1628→                'confidence_score, school_id');
  1629→
  1630→        var filteredQuery = questionsQuery;
  1631→        if (schoolId != null) {
  1632→          filteredQuery = filteredQuery.eq('school_id', schoolId);
  1633→        }
  1634→
  1635→        final questions = await filteredQuery;
  1636→
  1637→        int totalGenerated = questions.length;
  1638→        int totalApproved = 0;
  1639→        int totalRejected = 0;
  1640→        int pendingReview = 0;
  1641→        final questionsByType = <String, int>{};
  1642→        final questionsByDifficulty = <String, int>{};
  1643→        final questionsByBloomLevel = <String, int>{};
  1644→        double totalConfidence = 0;
  1645→        int confidenceCount = 0;
  1646→
  1647→        for (final q in questions) {
  1648→          final status = q['review_status'] as String? ?? 'pending';
  1649→          switch (status) {
  1650→            case 'approved':
  1651→              totalApproved++;
  1652→              break;
  1653→            case 'rejected':
  1654→              totalRejected++;
  1655→              break;
  1656→            default:
  1657→              pendingReview++;
  1658→          }
  1659→
  1660→          final type = q['question_type'] as String? ?? 'unknown';
  1661→          questionsByType[type] = (questionsByType[type] ?? 0) + 1;
  1662→
  1663→          final diff = q['difficulty'] as String? ?? 'unknown';
  1664→          questionsByDifficulty[diff] =
  1665→              (questionsByDifficulty[diff] ?? 0) + 1;
  1666→
  1667→          final bloom = q['bloom_level'] as String?;
  1668→          if (bloom != null) {
  1669→            questionsByBloomLevel[bloom] =
  1670→                (questionsByBloomLevel[bloom] ?? 0) + 1;
  1671→          }
  1672→
  1673→          final conf = q['confidence_score'] as num?;
  1674→          if (conf != null) {
  1675→            totalConfidence += conf.toDouble();
  1676→            confidenceCount++;
  1677→          }
  1678→        }
  1679→
  1680→        // Get cost info from usage stats
  1681→        var usageQuery = _supabase
  1682→            .from(_usageStatsTable)
  1683→            .select('total_cost, total_input_tokens, total_output_tokens, provider');
  1684→        if (schoolId != null) {
  1685→          usageQuery = usageQuery.eq('school_id', schoolId);
  1686→        }
  1687→
  1688→        final usage = await usageQuery;
  1689→        double totalCost = 0;
  1690→        int totalTokensUsed = 0;
  1691→        final costByProvider = <String, double>{};
  1692→
  1693→        for (final u in usage) {
  1694→          final cost = (u['total_cost'] as num?)?.toDouble() ?? 0;
  1695→          totalCost += cost;
  1696→          totalTokensUsed +=
  1697→              (u['total_input_tokens'] as int? ?? 0) +
  1698→                  (u['total_output_tokens'] as int? ?? 0);
  1699→          final prov = u['provider'] as String? ?? 'unknown';
  1700→          costByProvider[prov] = (costByProvider[prov] ?? 0) + cost;
  1701→        }
  1702→
  1703→        return {
  1704→          'total_generated': totalGenerated,
  1705→          'total_approved': totalApproved,
  1706→          'total_rejected': totalRejected,
  1707→          'pending_review': pendingReview,
  1708→          'total_cost': totalCost,
  1709→          'total_tokens_used': totalTokensUsed,
  1710→          'questions_by_type': questionsByType,
  1711→          'questions_by_difficulty': questionsByDifficulty,
  1712→          'questions_by_bloom_level': questionsByBloomLevel,
  1713→          'avg_confidence_score':
  1714→              confidenceCount > 0 ? totalConfidence / confidenceCount : null,
  1715→          'avg_generation_time_ms': null,
  1716→          'recent_generations': [],
  1717→          'cost_by_provider': costByProvider,
  1718→          'daily_usage': [],
  1719→        };
  1720→      }
  1721→    } on sb.PostgrestException catch (e) {
  1722→      throw _mapPostgrestException(e);
  1723→    } catch (e) {
  1724→      AppLogger.error('Unexpected getDashboardStats error', error: e);
  1725→      throw const ServerException(

  1727→        statusCode: 500,
  1728→      );

  1775→  /// Maps a Supabase PostgrestException to a domain-appropriate exception.
  1776→  ServerException _mapPostgrestException(sb.PostgrestException e) {
  1777→    AppLogger.error('Supabase error: ${e.message}', error: e);
  1778→
  1779→    switch (e.code) {
  1780→      case '23505':
  1781→        return ServerException(
  1782→          message: 'A record with this data already exists.',
  1783→          statusCode: 409,
  1784→          data: e.details,
  1785→        );
  1786→      case '23503':
  1787→        return ServerException(
  1788→          message: 'Referenced record not found.',
  1789→          statusCode: 400,
  1790→          data: e.details,
  1791→        );
  1792→      case '42501':
  1793→      case 'PGRST301':
  1794→        return ServerException(
  1795→          message: 'Insufficient permissions.',
  1796→          statusCode: 403,
  1797→          data: e.details,
  1798→        );
  1799→      case 'PGRST116':
  1800→        return ServerException(
  1801→          message: 'Resource not found.',
  1802→          statusCode: 404,
  1803→          data: e.details,
  1804→        );
  1805→      default:
  1806→        return ServerException(
  1807→          message: e.message ?? 'An unexpected database error occurred.',
  1808→          statusCode: int.tryParse(e.code ?? '') ?? 500,
  1809→          data: e.details,
  1810→        );
  1811→    }
  1812→  }
