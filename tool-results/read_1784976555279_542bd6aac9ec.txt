   131→-- Generic updated_at trigger function
   132→CREATE OR REPLACE FUNCTION update_updated_at_column()
   133→RETURNS TRIGGER AS $$
   134→BEGIN
   135→  NEW.updated_at = NOW();
   136→  RETURN NEW;
   137→END;

   140→-- Generate unique order numbers
   141→CREATE OR REPLACE FUNCTION generate_order_number()
   142→RETURNS TEXT AS $$
   143→BEGIN
   144→  RETURN 'EF-' || TO_CHAR(NOW(), 'YYYYMMDD') || '-' || UPPER(SUBSTRING(MD5(RANDOM()::TEXT) FROM 1 FOR 8));
   145→END;
   146→$$ LANGUAGE plpgsql VOLATILE;
   147→
   148→-- Generate unique license keys
   149→CREATE OR REPLACE FUNCTION generate_license_key()
   150→RETURNS TEXT AS $$
   151→BEGIN
   152→  RETURN 'EFL-' || UPPER(SUBSTRING(MD5(RANDOM()::TEXT || CLOCK_TIMESTAMP()::TEXT) FROM 1 FOR 4))
   153→       || '-' || UPPER(SUBSTRING(MD5(RANDOM()::TEXT || CLOCK_TIMESTAMP()::TEXT) FROM 1 FOR 4))
   154→       || '-' || UPPER(SUBSTRING(MD5(RANDOM()::TEXT || CLOCK_TIMESTAMP()::TEXT) FROM 1 FOR 4))
   155→       || '-' || UPPER(SUBSTRING(MD5(RANDOM()::TEXT || CLOCK_TIMESTAMP()::TEXT) FROM 1 FOR 4));
   156→END;
   157→$$ LANGUAGE plpgsql VOLATILE;

   858→-- ---- Auto-update updated_at on all tables with that column ----
   859→DO $$
   860→DECLARE
   861→  t TEXT;
   862→BEGIN
   863→  FOR t IN
   864→    SELECT table_name FROM information_schema.columns
   865→    WHERE table_schema = 'public'
   866→      AND column_name = 'updated_at'
   867→      AND table_name IN (
   868→        'marketplace_categories',
   869→        'seller_profiles',
   870→        'marketplace_products',
   871→        'marketplace_carts',
   872→        'marketplace_orders',
   873→        'marketplace_reviews',
   874→        'marketplace_promo_codes',
   875→        'marketplace_commission_rates',
   876→        'marketplace_disputes'
   877→      )
   878→    GROUP BY table_name
   879→  LOOP
   880→    EXECUTE format(
   881→      'CREATE OR REPLACE TRIGGER trg_%s_updated_at
   882→         BEFORE UPDATE ON %I
   883→         FOR EACH ROW
   884→         EXECUTE FUNCTION update_updated_at_column()',
   885→      t, t
   886→    );
   887→  END LOOP;
   888→END;
   889→$$;

  1539→-- ---- Full-text search function for marketplace products ----
  1540→CREATE OR REPLACE FUNCTION marketplace_search(
  1541→  search_query   TEXT,
  1542→  p_product_type marketplace_product_type DEFAULT NULL,
  1543→  p_category_id  UUID DEFAULT NULL,
  1544→  p_subject      TEXT DEFAULT NULL,
  1545→  p_class_level  TEXT DEFAULT NULL,
  1546→  p_curriculum   TEXT DEFAULT NULL,
  1547→  p_price_min    DECIMAL DEFAULT NULL,
  1548→  p_price_max    DECIMAL DEFAULT NULL,
  1549→  p_license_type marketplace_license_type DEFAULT NULL,
  1550→  p_tags         TEXT[] DEFAULT NULL,
  1551→  p_is_free      BOOLEAN DEFAULT NULL,
  1552→  p_sort_by      TEXT DEFAULT 'relevance',
  1553→  p_limit        INTEGER DEFAULT 20,
  1554→  p_offset       INTEGER DEFAULT 0
  1555→)
  1556→RETURNS TABLE (
  1557→  id                UUID,
  1558→  title             TEXT,
  1559→  slug              TEXT,
  1560→  description       TEXT,
  1561→  product_type      marketplace_product_type,
  1562→  subject           TEXT,
  1563→  class_level       TEXT,
  1564→  curriculum        TEXT,
  1565→  price             DECIMAL,
  1566→  original_price    DECIMAL,
  1567→  currency          TEXT,
  1568→  license_type      marketplace_license_type,
  1569→  tags              TEXT[],
  1570→  preview_images    JSONB,
  1571→  average_rating    DECIMAL,
  1572→  total_reviews     INTEGER,
  1573→  total_sales       INTEGER,
  1574→  quality_score     DECIMAL,
  1575→  is_free           BOOLEAN,
  1576→  is_featured       BOOLEAN,
  1577→  seller_id         UUID,
  1578→  seller_name       TEXT,
  1579→  published_at      TIMESTAMPTZ,
  1580→  search_rank       REAL
  1581→) AS $$
  1582→BEGIN
  1583→  RETURN QUERY
  1584→  SELECT
  1585→    p.id,
  1586→    p.title,
  1587→    p.slug,
  1588→    p.description,
  1589→    p.product_type,
  1590→    p.subject,
  1591→    p.class_level,
  1592→    p.curriculum,
  1593→    p.price,
  1594→    p.original_price,
  1595→    p.currency,
  1596→    p.license_type,
  1597→    p.tags,
  1598→    p.preview_images,
  1599→    p.average_rating,
  1600→    p.total_reviews,
  1601→    p.total_sales,
  1602→    p.quality_score,
  1603→    p.is_free,
  1604→    p.is_featured,
  1605→    p.seller_id,
  1606→    sp.display_name AS seller_name,
  1607→    p.published_at,
  1608→    ts_rank(p.fts_document, websearch_to_tsquery('english', search_query)) AS search_rank
  1609→  FROM marketplace_products p
  1610→  INNER JOIN seller_profiles sp ON p.seller_id = sp.id
  1611→  WHERE p.status = 'approved'
  1612→    AND p.deleted_at IS NULL
  1613→    -- Full-text search
  1614→    AND p.fts_document @@ websearch_to_tsquery('english', search_query)
  1615→    -- Optional filters
  1616→    AND (p_product_type IS NULL OR p.product_type = p_product_type)
  1617→    AND (p_category_id IS NULL OR p.category_id = p_category_id)
  1618→    AND (p_subject IS NULL OR p.subject ILIKE '%' || p_subject || '%')
  1619→    AND (p_class_level IS NULL OR p.class_level = p_class_level)
  1620→    AND (p_curriculum IS NULL OR p.curriculum ILIKE '%' || p_curriculum || '%')
  1621→    AND (p_price_min IS NULL OR p.price >= p_price_min)
  1622→    AND (p_price_max IS NULL OR p.price <= p_price_max)
  1623→    AND (p_license_type IS NULL OR p.license_type = p_license_type)
  1624→    AND (p_is_free IS NULL OR p.is_free = p_is_free)
  1625→    AND (p_tags IS NULL OR p.tags && p_tags)
  1626→  ORDER BY
  1627→    CASE p_sort_by
  1628→      WHEN 'relevance' THEN ts_rank(p.fts_document, websearch_to_tsquery('english', search_query))
  1629→      ELSE 0
  1630→    END DESC,
  1631→    CASE p_sort_by
  1632→      WHEN 'newest' THEN EXTRACT(EPOCH FROM p.published_at)
  1633→      WHEN 'price_low' THEN -p.price
  1634→      WHEN 'price_high' THEN p.price
  1635→      WHEN 'rating' THEN p.average_rating
  1636→      WHEN 'popular' THEN p.total_sales
  1637→      ELSE 0
  1638→    END DESC,
  1639→    p.is_featured DESC,
  1640→    p.published_at DESC
  1641→  LIMIT p_limit
  1642→  OFFSET p_offset;
  1643→END;
  1644→$$ LANGUAGE plpgsql STABLE;
  1645→
  1646→COMMENT ON FUNCTION marketplace_search IS 'Full-text search with weighted ranking and multi-faceted filtering for marketplace products';

  1648→-- ---- AI Recommendation scoring function ----
  1649→CREATE OR REPLACE FUNCTION calculate_recommendation_score(
  1650→  p_user_id    UUID,
  1651→  p_product_id UUID
  1652→)
  1653→RETURNS DECIMAL AS $$
  1654→DECLARE
  1655→  v_score            DECIMAL := 0.0;
  1656→  v_user_purchases   UUID[];
  1657→  v_user_tags        TEXT[];
  1658→  v_product_tags     TEXT[];
  1659→  v_tag_overlap      INTEGER;
  1660→  v_purchase_count   INTEGER;
  1661→  v_product_rating   DECIMAL;
  1662→  v_product_sales    INTEGER;
  1663→  v_quality_score    DECIMAL;
  1664→BEGIN
  1665→  -- Get user's purchase history product categories
  1666→  SELECT array_agg(DISTINCT product_id) INTO v_user_purchases
  1667→  FROM marketplace_purchases
  1668→  WHERE buyer_id = p_user_id AND is_active = TRUE;
  1669→
  1670→  -- Get tags from user's purchased products
  1671→  SELECT array_agg(DISTINCT unnest_tags) INTO v_user_tags
  1672→  FROM (
  1673→    SELECT unnest(tags) AS unnest_tags
  1674→    FROM marketplace_products
  1675→    WHERE id = ANY(v_user_purchases) AND tags IS NOT NULL
  1676→  ) sub;
  1677→
  1678→  -- Get product tags
  1679→  SELECT tags INTO v_product_tags
  1680→  FROM marketplace_products
  1681→  WHERE id = p_product_id;
  1682→
  1683→  -- 1. Tag overlap score (0-0.30): How many tags overlap with user's interests
  1684→  IF v_user_tags IS NOT NULL AND v_product_tags IS NOT NULL THEN
  1685→    SELECT COUNT(*) INTO v_tag_overlap
  1686→    FROM (SELECT unnest(v_user_tags) INTERSECT SELECT unnest(v_product_tags)) overlap;
  1687→    v_score := v_score + LEAST(v_tag_overlap::DECIMAL / 10.0, 0.30);
  1688→  END IF;
  1689→
  1690→  -- 2. Quality score component (0-0.25)
  1691→  SELECT COALESCE(quality_score, 0) INTO v_quality_score
  1692→  FROM marketplace_products WHERE id = p_product_id;
  1693→  v_score := v_score + (v_quality_score / 100.0) * 0.25;
  1694→
  1695→  -- 3. Rating component (0-0.20)
  1696→  SELECT COALESCE(average_rating, 0) INTO v_product_rating
  1697→  FROM marketplace_products WHERE id = p_product_id;
  1698→  v_score := v_score + (v_product_rating / 5.0) * 0.20;
  1699→
  1700→  -- 4. Popularity component (0-0.15): Normalized sales
  1701→  SELECT COALESCE(total_sales, 0) INTO v_product_sales
  1702→  FROM marketplace_products WHERE id = p_product_id;
  1703→  v_score := v_score + LEAST(v_product_sales::DECIMAL / 1000.0, 0.15);
  1704→
  1705→  -- 5. Same-subject bonus (0-0.10)
  1706→  IF v_user_purchases IS NOT NULL THEN
  1707→    SELECT COUNT(*) INTO v_purchase_count
  1708→    FROM marketplace_products
  1709→    WHERE id = ANY(v_user_purchases)
  1710→      AND subject = (SELECT subject FROM marketplace_products WHERE id = p_product_id)
  1711→      AND subject IS NOT NULL;
  1712→    v_score := v_score + LEAST(v_purchase_count::DECIMAL * 0.05, 0.10);
  1713→  END IF;
  1714→
  1715→  RETURN LEAST(v_score, 1.0000);
  1716→END;
  1717→$$ LANGUAGE plpgsql STABLE;
  1718→
  1719→COMMENT ON FUNCTION calculate_recommendation_score IS 'Calculates a 0-1 recommendation score for a user-product pair based on tag overlap, quality, rating, popularity, and subject affinity';

  1721→-- ---- Quality check function ----
  1722→CREATE OR REPLACE FUNCTION run_product_quality_check(
  1723→  p_product_id UUID
  1724→)
  1725→RETURNS UUID AS $$
  1726→DECLARE
  1727→  v_overall_score    DECIMAL(5,2);
  1728→  v_grammar_score    DECIMAL(5,2) := 0.00;
  1729→  v_spelling_score   DECIMAL(5,2) := 0.00;
  1730→  v_formatting_score DECIMAL(5,2) := 0.00;

  1734→  v_accuracy_flag    BOOLEAN := FALSE;
  1735→  v_suggestions      TEXT[] := '{}';
  1736→  v_flagged_issues   TEXT[] := '{}';

  1738→  v_check_id         UUID;
  1739→  v_description      TEXT;
  1740→  v_title            TEXT;
  1741→BEGIN
  1742→  -- Fetch product content
  1743→  SELECT title, description INTO v_title, v_description
  1744→  FROM marketplace_products WHERE id = p_product_id;
  1745→
  1746→  IF NOT FOUND THEN
  1747→    RAISE EXCEPTION 'Product % not found', p_product_id;

  1750→  -- ---- Grammar heuristic scoring ----
  1751→  -- This is a simplified heuristic; in production, integrate with AI/NLP service

  1753→    WHEN char_length(v_description) < 50 THEN 40.00
  1754→    WHEN char_length(v_description) < 200 THEN 60.00
  1755→    WHEN char_length(v_description) >= 200 THEN 75.00
  1756→    ELSE 50.00
  1757→  END;
  1758→
  1759→  -- Add bonus for structured content (headings, lists, etc.)
  1760→  IF v_description ~ '(^|\n)#{1,6}\s' THEN  -- Markdown headings

  1765→  v_spelling_score := 85.00;  -- Placeholder; integrate spell-check API in production
  1766→
  1767→  -- ---- Formatting heuristic ----
  1768→  v_formatting_score := CASE
  1769→    WHEN v_description ~ '\n{2,}' THEN 80.00   -- Has paragraph breaks
  1770→    WHEN v_description ~ '\n' THEN 65.00        -- Has line breaks
  1771→    ELSE 50.00                                    -- No line breaks
  1772→  END;
  1773→
  1774→  IF v_description ~ '\*\*|__|\*|_' THEN  -- Markdown formatting
  1775→    v_formatting_score := LEAST(v_formatting_score + 10, 100);

  1777→
  1778→  -- ---- Curriculum alignment ----

  1784→    WHEN v_reading_level <= 4 THEN 'Elementary'

  1787→    WHEN v_reading_level <= 10 THEN 'Senior Secondary'
  1788→    WHEN v_reading_level <= 12 THEN 'Advanced Secondary'
  1789→    ELSE 'University Level'
  1790→  END;
  1791→
  1792→  -- ---- Calculate overall score (weighted) ----
  1793→  v_overall_score := (
  1794→    v_grammar_score * 0.20 +
  1795→    v_spelling_score * 0.15 +
  1796→    v_formatting_score * 0.20 +
  1797→    v_curriculum_score * 0.45
  1798→  );
  1799→
  1800→  -- ---- Generate suggestions ----
  1801→  IF v_grammar_score < 70 THEN
  1802→    v_suggestions := array_append(v_suggestions, 'Consider improving grammar and sentence structure');
  1803→  END IF;
  1804→  IF v_formatting_score < 70 THEN
  1805→    v_suggestions := array_append(v_suggestions, 'Add formatting: headings, bullet points, or numbered lists');
  1806→  END IF;
  1807→  IF char_length(v_description) < 200 THEN
  1808→    v_suggestions := array_append(v_suggestions, 'Description is too short; add more detail for better discoverability');
  1809→  END IF;
  1810→
  1811→  -- ---- Flag issues ----
  1812→  IF v_grammar_score < 50 THEN
  1813→    v_flagged_issues := array_append(v_flagged_issues, 'Poor grammar quality detected');
  1814→  END IF;
  1815→  IF v_spelling_score < 50 THEN
  1816→    v_flagged_issues := array_append(v_flagged_issues, 'Potential spelling issues detected');
  1817→  END IF;
  1818→
  1819→  -- ---- Determine quality status ----
  1820→  v_quality_status := CASE
  1821→    WHEN v_overall_score >= 80 THEN 'passed'
  1822→    WHEN v_overall_score >= 60 THEN 'needs_improvement'
  1823→    ELSE 'failed'
  1824→  END;
  1825→
  1826→  -- ---- Insert quality check record ----
  1827→  INSERT INTO marketplace_quality_checks (
  1828→    product_id, overall_score, grammar_score, spelling_score,

  1832→  ) VALUES (
  1833→    p_product_id, v_overall_score, v_grammar_score, v_spelling_score,
  1834→    v_formatting_score, v_curriculum_score, v_reading_level,
  1835→    v_reading_label, v_accuracy_flag,
  1836→    jsonb_build_object('factual_errors', '[]', 'source_conflicts', '[]', 'confidence_score', 0.85),
  1837→    v_suggestions, v_flagged_issues
  1838→  ) RETURNING id INTO v_check_id;
  1839→
  1840→  -- ---- Update product quality fields ----
  1841→  UPDATE marketplace_products
  1842→  SET
  1843→    quality_score = v_overall_score,
  1844→    quality_check_status = v_quality_status,
  1845→    quality_check_details = jsonb_build_object(
  1846→      'check_id', v_check_id,
  1847→      'checked_at', NOW(),
  1848→      'overall_score', v_overall_score
  1849→    )
  1850→  WHERE id = p_product_id;
  1851→
  1852→  RETURN v_check_id;
  1853→END;
  1854→$$ LANGUAGE plpgsql;

  1858→-- ---- Get applicable commission rate ----
  1859→CREATE OR REPLACE FUNCTION get_commission_rate(
  1860→  p_product_type marketplace_product_type,
  1861→  p_license_type marketplace_license_type
  1862→)
  1863→RETURNS DECIMAL AS $$
  1864→DECLARE
  1865→  v_rate DECIMAL;
  1866→BEGIN
  1867→  -- Try specific product_type + license_type match first
  1868→  SELECT commission_rate INTO v_rate
  1869→  FROM marketplace_commission_rates
  1870→  WHERE (product_type = p_product_type OR product_type IS NULL)
  1871→    AND (license_type = p_license_type OR license_type IS NULL)
  1872→    AND is_active = TRUE
  1873→    AND effective_from <= NOW()
  1874→    AND (effective_to IS NULL OR effective_to > NOW())
  1875→  ORDER BY
  1876→    -- Most specific match first
  1877→    CASE WHEN product_type IS NOT NULL AND license_type IS NOT NULL THEN 0
  1878→         WHEN product_type IS NOT NULL AND license_type IS NULL THEN 1
  1879→         WHEN product_type IS NULL AND license_type IS NOT NULL THEN 2
  1880→         ELSE 3
  1881→    END,
  1882→    effective_from DESC
  1883→  LIMIT 1;
  1884→
  1885→  RETURN COALESCE(v_rate, 0.1500);  -- Default 15% commission
  1886→END;
  1887→$$ LANGUAGE plpgsql STABLE;

  1931→      'Minimum order amount is ' || v_promo.min_order_amount || ' ' || v_promo.currency;
  1932→    RETURN;

  1944→    END IF;
  1945→  ELSE
  1946→    v_applicable_total := p_subtotal;
  1947→  END IF;

  1950→  IF v_promo.discount_type = 'percentage' THEN

  1952→    -- Cap at max discount

  1954→      v_discount := LEAST(v_discount, v_promo.max_discount_amount);
  1955→    END IF;
  1956→  ELSE  -- fixed
  1957→    v_discount := LEAST(v_promo.discount_value, v_applicable_total);
  1958→  END IF;
  1959→
  1960→  RETURN QUERY SELECT v_discount, v_promo.id, TRUE, 'Valid';
  1961→END;
