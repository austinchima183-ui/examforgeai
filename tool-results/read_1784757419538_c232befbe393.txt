   214→  String _mapFailureToMessage(Failure failure) {
   215→    return failure.when(
   216→      server: (message, statusCode, data) => 'Server error: $message',
   217→      cache: (message) => 'Cache error: $message',
   218→      auth: (message, code) => 'Auth error: $message',
   219→      network: (message) => 'Network error: $message',
   220→      validation: (message, fieldErrors) => 'Validation error: $message',
   221→      notFound: (message) => 'Not found: $message',
   222→      unauthorized: (message) => 'Unauthorized: $message',
   223→      forbidden: (message) => 'Forbidden: $message',
   224→    );
   225→  }

   229→  Future<void> recordAudit(AuditEntry data) async {
   230→    state = state.copyWith(isLoading: true, error: null);
   231→    final result = await _recordAuditEventUseCase(
   232→      RecordAuditEventParams(entry: data),
   233→    );
   234→    result.fold(
   235→      onSuccess: (recorded) => state = state.copyWith(
   236→        auditTrail: [recorded, ...state.auditTrail],
   237→        isLoading: false,
   238→      ),
   239→      onFailure: (failure) => state = state.copyWith(
   240→        isLoading: false,
   241→        error: _mapFailureToMessage(failure),
   242→      ),
   243→    );
   244→  }
