   145→/// Supabase-backed implementation of [QuestionBankRemoteDataSource].

   184→  @override
   185→  Future<QuestionModel> createQuestion(
   186→    Map<String, dynamic> questionData,
   187→  ) async {
   188→    try {
   189→      final response =
   190→          await _supabase.from(_questionsTable).insert(questionData).select();
   191→
   192→      if (response.isEmpty) {
   193→        throw const ServerException(
   194→          message: 'Question creation returned no data.',
   195→          statusCode: 500,
   196→        );
   197→      }
   198→
   199→      AppLogger.info('Question created: ${response.first['id']}');
   200→      return QuestionModel.fromJson(response.first);
   201→    } on sb.PostgrestException catch (e) {
   202→      throw _mapPostgrestException(e);
   203→    } on ServerException {
   204→      rethrow;
   205→    } catch (e) {
   206→      AppLogger.error('Unexpected createQuestion error', error: e);
   207→      throw const ServerException(
   208→        message: 'Failed to create question.',
   209→        statusCode: 500,
   210→      );
   211→    }
   212→  }

   294→  @override
   295→  Future<Map<String, dynamic>> getQuestionWithDetails(
   296→    String questionId,
   297→  ) async {
   298→    try {
   299→      // Fetch the base question
   300→      final questionResponse = await _supabase
   301→          .from(_questionsTable)
   302→          .select()
   303→          .eq('id', questionId)
   304→          .limit(1);
   305→
   306→      if (questionResponse.isEmpty) {
   307→        throw const NotFoundException('Question not found.');
   308→      }
   309→
   310→      final questionJson = Map<String, dynamic>.from(questionResponse.first);
   311→
   312→      // Fetch related data in parallel
   313→      final results = await Future.wait([
   314→        _supabase
   315→            .from(_answerOptionsTable)
   316→            .select()
   317→            .eq('question_id', questionId)
   318→            .order('sort_order'),
   319→        _supabase
   320→            .from(_matchingPairsTable)
   321→            .select()
   322→            .eq('question_id', questionId)
   323→            .order('sort_order'),
   324→        _supabase
   325→            .from(_orderingItemsTable)
   326→            .select()
   327→            .eq('question_id', questionId)
   328→            .order('correct_position'),
   329→        _supabase
   330→            .from(_fillInBlankTable)
   331→            .select()
   332→            .eq('question_id', questionId)
   333→            .order('blank_index'),
   334→        _supabase
   335→            .from(_attachmentsTable)
   336→            .select()
   337→            .eq('question_id', questionId)
   338→            .order('sort_order'),
   339→        // Tags need a join through the relation table
   340→        _supabase
   341→            .from(_tagRelationsTable)
   342→            .select('question_tags(*)')
   343→            .eq('question_id', questionId),
   344→      ]);
   345→
   346→      questionJson['answer_options'] = results[0] as List;
   347→      questionJson['matching_pairs'] = results[1] as List;
   348→      questionJson['ordering_items'] = results[2] as List;
   349→      questionJson['fill_in_blank_answers'] = results[3] as List;
   350→      questionJson['question_attachments'] = results[4] as List;
   351→
   352→      // Flatten tag join results
   353→      final tagRows = results[5] as List;
   354→      final tagList = tagRows
   355→          .map((row) => row['question_tags'] as Map<String, dynamic>?)
   356→          .whereType<Map<String, dynamic>>()
   357→          .toList();
   358→      questionJson['question_tags'] = tagList;
   359→
   360→      return questionJson;
   361→    } on sb.PostgrestException catch (e) {
   362→      throw _mapPostgrestException(e);
   363→    } on NotFoundException {
   364→      rethrow;
   365→    } catch (e) {
   366→      AppLogger.error('Unexpected getQuestionWithDetails error', error: e);
   367→      throw const ServerException(
   368→        message: 'Failed to retrieve question details.',
   369→        statusCode: 500,
   370→      );
   371→    }
   372→  }

   374→  @override
   375→  Future<List<QuestionModel>> getQuestions(
   376→    Map<String, dynamic> filters,
   377→  ) async {
   378→    try {
   379→      var query = _supabase.from(_questionsTable).select();
   380→
   381→      query = _applyFilters(query, filters);
   382→      query = _applySorting(query, filters['sort_by'] as String?);
   383→      query = _applyPagination(
   384→        query,
   385→        page: filters['page'] as int?,
   386→        perPage: filters['per_page'] as int?,
   387→      );
   388→
   389→      final response = await query;
   390→      return response
   391→          .map((json) => QuestionModel.fromJson(json as Map<String, dynamic>))
   392→          .toList();
   393→    } on sb.PostgrestException catch (e) {
   394→      throw _mapPostgrestException(e);
   395→    } catch (e) {
   396→      AppLogger.error('Unexpected getQuestions error', error: e);
   397→      throw const ServerException(
   398→        message: 'Failed to retrieve questions.',
   399→        statusCode: 500,
   400→      );

   501→  @override
   502→  Future<QuestionModel> duplicateQuestion(String questionId) async {
   503→    try {
   504→      // Fetch the original question
   505→      final original = await getQuestion(questionId);
   506→
   507→      // Prepare the duplicate data (strip ID, reset version)
   508→      final duplicateData = original.toJson()
   509→        ..remove('id')
   510→        ..remove('created_at')
   511→        ..remove('updated_at')
   512→        ..['version'] = 1
   513→        ..['parent_id'] = original.id
   514→        ..['is_published'] = false
   515→        ..['is_archived'] = false
   516→        ..['usage_count'] = 0
   517→        ..['avg_score'] = 0.0;
   518→
   519→      return createQuestion(duplicateData);
   520→    } on sb.PostgrestException catch (e) {
   521→      throw _mapPostgrestException(e);
   522→    } on ServerException {
   523→      rethrow;
   524→    } on NotFoundException {
   525→      rethrow;
   526→    } catch (e) {
   527→      AppLogger.error('Unexpected duplicateQuestion error', error: e);
   528→      throw const ServerException(
   529→        message: 'Failed to duplicate question.',
   530→        statusCode: 500,
   531→      );
   532→    }
   533→  }

   535→  @override
   536→  Future<void> moveQuestions(
   537→    List<String> questionIds,
   538→    Map<String, dynamic> target,
   539→  ) async {
   540→    try {
   541→      final updateData = <String, dynamic>{
   542→        'updated_at': DateTime.now().toIso8601String(),
   543→      };
   544→
   545→      if (target.containsKey('topic_id')) {
   546→        updateData['topic_id'] = target['topic_id'];
   547→      }
   548→      if (target.containsKey('category_id')) {
   549→        updateData['category_id'] = target['category_id'];
   550→      }
   551→
   552→      for (final id in questionIds) {
   553→        await _supabase
   554→            .from(_questionsTable)
   555→            .update(updateData)
   556→            .eq('id', id);
   557→      }
   558→
   559→      AppLogger.info('Moved ${questionIds.length} questions');
   560→    } on sb.PostgrestException catch (e) {
   561→      throw _mapPostgrestException(e);
   562→    } catch (e) {
   563→      AppLogger.error('Unexpected moveQuestions error', error: e);

   601→  @override
   602→  Future<List<AnswerOptionModel>> updateAnswerOptions(
   603→    String questionId,
   604→    List<Map<String, dynamic>> options,
   605→  ) async {
   606→    try {
   607→      // Delete existing options
   608→      await _supabase
   609→          .from(_answerOptionsTable)
   610→          .delete()
   611→          .eq('question_id', questionId);
   612→
   613→      // Insert new options
   614→      if (options.isNotEmpty) {
   615→        final optionsData = options
   616→            .map((o) => {
   617→                  ...o,
   618→                  'question_id': questionId,
   619→                })
   620→            .toList();
   621→
   622→        await _supabase
   623→            .from(_answerOptionsTable)
   624→            .insert(optionsData);
   625→      }
   626→
   627→      // Fetch the newly inserted options
   628→      return getAnswerOptions(questionId);
   629→    } on sb.PostgrestException catch (e) {
   630→      throw _mapPostgrestException(e);
   631→    } catch (e) {
   632→      AppLogger.error('Unexpected updateAnswerOptions error', error: e);
   633→      throw const ServerException(
   634→        message: 'Failed to update answer options.',
   635→        statusCode: 500,
   636→      );
   637→    }
   638→  }

   769→
   770→      // Check if already favorited

   775→          .eq('question_id', questionId);
   776→
   777→      if (existing.isNotEmpty) {

   779→        await _supabase
   780→            .from(_favoritesTable)

   782→            .eq('user_id', userId)

   785→      } else {

   787→        await _supabase.from(_favoritesTable).insert({
   788→          'user_id': userId,
   789→          'question_id': questionId,
   790→        });
   791→        AppLogger.info('Added favorite: $questionId');
   792→      }
   793→    } on sb.PostgrestException catch (e) {
   794→      throw _mapPostgrestException(e);
   795→    } on UnauthorizedException {
   796→      rethrow;
   797→    } catch (e) {
   798→      AppLogger.error('Unexpected toggleFavorite error', error: e);
   799→      throw const ServerException(
   800→        message: 'Failed to toggle favorite.',
   801→        statusCode: 500,
   802→      );
   803→    }

  1333→  @override
  1334→  Future<QuestionBankStatsModel> getStats({String? schoolId}) async {
  1335→    try {
  1336→      // Base query builder
  1337→      var baseQuery = _supabase.from(_questionsTable).select();
  1338→
  1339→      if (schoolId != null) {
  1340→        baseQuery = baseQuery.eq('school_id', schoolId);
  1341→      }
  1342→
  1343→      // Fetch all questions for stats computation
  1344→      final allQuestions = await baseQuery;
  1345→
  1346→      final totalQuestions = allQuestions.length;
  1347→      final publishedQuestions = allQuestions
  1348→          .where((q) => q['is_published'] == true)
  1349→          .length;
  1350→      final draftQuestions = allQuestions
  1351→          .where((q) => q['is_published'] == false && q['is_archived'] == false)
  1352→          .length;
  1353→      final archivedQuestions = allQuestions
  1354→          .where((q) => q['is_archived'] == true)
  1355→          .length;
  1356→
  1357→      // Compute breakdowns
  1358→      final Map<String, int> bySubject = {};
  1359→      final Map<String, int> byDifficulty = {};
  1360→      final Map<String, int> byType = {};
  1361→      final Map<String, int> byExamType = {};
  1362→
  1363→      for (final q in allQuestions) {
  1364→        final subjectId = q['subject_id'] as String?;
  1365→        if (subjectId != null) {
  1366→          bySubject[subjectId] = (bySubject[subjectId] ?? 0) + 1;
  1367→        }
  1368→
  1369→        final diff = q['difficulty'] as String?;
  1370→        if (diff != null) {
  1371→          byDifficulty[diff] = (byDifficulty[diff] ?? 0) + 1;
  1372→        }
  1373→
  1374→        final type = q['question_type'] as String?;
  1375→        if (type != null) {
  1376→          byType[type] = (byType[type] ?? 0) + 1;
  1377→        }
  1378→
  1379→        final exam = q['exam_type'] as String?;
  1380→        if (exam != null) {
  1381→          byExamType[exam] = (byExamType[exam] ?? 0) + 1;
  1382→        }
  1383→      }
  1384→
  1385→      // Recent questions (last 7 days)
  1386→      final weekAgo = DateTime.now().subtract(const Duration(days: 7));
  1387→      final recentQuestions = allQuestions
  1388→          .where((q) {
  1389→            final createdAt = q['created_at'] as String?;
  1390→            if (createdAt == null) return false;
  1391→            return DateTime.parse(createdAt).isAfter(weekAgo);
  1392→          })
  1393→          .length;
  1394→
  1395→      // Collection count
  1396→      var collectionQuery =
  1397→          _supabase.from(_collectionsTable).select('id');
  1398→      if (schoolId != null) {
  1399→        collectionQuery = collectionQuery.eq('school_id', schoolId);
  1400→      }
  1401→      final collections = await collectionQuery;
  1402→
  1403→      // Favorites count for current user
  1404→      final userId = _currentUserId;
  1405→      int totalFavorites = 0;
  1406→      if (userId != null) {
  1407→        final favorites = await _supabase
  1408→            .from(_favoritesTable)
  1409→            .select('id')
  1410→            .eq('user_id', userId);
  1411→        totalFavorites = favorites.length;
  1412→      }
  1413→
  1414→      // Most used questions (top 5)
  1415→      final sortedByUsage = List<Map<String, dynamic>>.from(allQuestions)
  1416→        ..sort((a, b) => (b['usage_count'] as int? ?? 0)
  1417→            .compareTo(a['usage_count'] as int? ?? 0));
  1418→      final mostUsed = sortedByUsage
  1419→          .take(5)
  1420→          .map((q) => QuestionModel.fromJson(q))
  1421→          .toList();
  1422→
  1423→      return QuestionBankStatsModel(
  1424→        totalQuestions: totalQuestions,
  1425→        publishedQuestions: publishedQuestions,
  1426→        draftQuestions: draftQuestions,
  1427→        archivedQuestions: archivedQuestions,
  1428→        questionsBySubject: bySubject,
  1429→        questionsByDifficulty: byDifficulty,
  1430→        questionsByType: byType,
  1431→        questionsByExamType: byExamType,
  1432→        recentQuestions: recentQuestions,
  1433→        totalCollections: collections.length,
  1434→        totalFavorites: totalFavorites,
  1435→        mostUsedQuestions: mostUsed,
  1436→      );
  1437→    } on sb.PostgrestException catch (e) {
  1438→      throw _mapPostgrestException(e);
  1439→    } catch (e) {
  1440→      AppLogger.error('Unexpected getStats error', error: e);
  1441→      throw const ServerException(
  1442→        message: 'Failed to retrieve question bank stats.',
  1443→        statusCode: 500,
  1444→      );
  1445→    }
  1446→  }

  1599→    String questionId,

  1601→  ) async {

  1604→      final versionResponse = await _supabase

  1606→          .select()
  1607→          .eq('question_id', questionId)
  1608→          .eq('version', version)
  1609→          .limit(1);

  1611→      if (versionResponse.isEmpty) {

  1616→
  1617→      final versionData = versionResponse.first;
  1618→      final snapshot = versionData['snapshot'] as Map<String, dynamic>?;
  1619→
  1620→      if (snapshot == null) {
  1621→        throw const ServerException(
  1622→          message: 'Version snapshot is empty.',
  1623→          statusCode: 500,
  1624→        );
  1625→      }
  1626→
  1627→      // Update the question with the snapshot data
  1628→      final restoredData = Map<String, dynamic>.from(snapshot)
  1629→        ..remove('id')
  1630→        ..remove('created_at')
  1631→        ..['updated_at'] = DateTime.now().toIso8601String()
  1632→        ..['version'] = (await getQuestion(questionId)).version + 1;
  1633→
  1634→      return updateQuestion(questionId, restoredData);
  1635→    } on sb.PostgrestException catch (e) {
  1636→      throw _mapPostgrestException(e);
  1637→    } on ServerException {
  1638→      rethrow;
  1639→    } on NotFoundException {
  1640→      rethrow;
  1641→    } catch (e) {
  1642→      AppLogger.error('Unexpected restoreVersion error', error: e);
  1643→      throw const ServerException(
  1644→        message: 'Failed to restore question version.',
  1645→        statusCode: 500,
  1646→      );
  1647→    }
  1648→  }

  1713→  /// Applies sorting based on the [sortBy] value.
  1714→  sb.PostgrestTransformBuilder<dynamic> _applySorting(
  1715→    sb.PostgrestTransformBuilder<dynamic> query,
  1716→    String? sortBy,
  1717→  ) {
  1718→    switch (sortBy) {
  1719→      case 'oldest':
  1720→        return query.order('created_at', ascending: true);
  1721→      case 'most_used':
  1722→        return query.order('usage_count', ascending: false);
  1723→      case 'least_used':
  1724→        return query.order('usage_count', ascending: true);
  1725→      case 'highest_rated':
  1726→        return query.order('avg_score', ascending: false);
  1727→      case 'a_z':
  1728→        return query.order('content', ascending: true);
  1729→      case 'z_a':
  1730→        return query.order('content', ascending: false);
  1731→      case 'newest':
  1732→      default:
  1733→        return query.order('created_at', ascending: false);
  1734→    }
  1735→  }

  1737→  /// Applies pagination using Supabase range.
  1738→  sb.PostgrestTransformBuilder<dynamic> _applyPagination(
  1739→    sb.PostgrestTransformBuilder<dynamic> query, {
  1740→    int? page,
  1741→    int? perPage,
  1742→  }) {
  1743→    final p = page ?? 1;
  1744→    final pp = perPage ?? 20;
  1745→    final from = (p - 1) * pp;
  1746→    final to = from + pp - 1;
  1747→    return query.range(from, to);
  1748→  }

  1750→  /// Maps a Supabase [sb.PostgrestException] to a domain exception.
  1751→  Exception _mapPostgrestException(sb.PostgrestException e) {
  1752→    final statusCode = e.code != null ? int.tryParse(e.code!) ?? 0 : 0;
  1753→    final message = e.message ?? 'An unexpected database error occurred.';
  1754→
  1755→    AppLogger.warning(
  1756→      'Supabase PostgrestException — code: ${e.code}, message: $message',
  1757→    );
  1758→
  1759→    switch (statusCode) {
  1760→      case 401:
  1761→        return UnauthorizedException(message);
  1762→      case 403:
  1763→        return ForbiddenException(message);
  1764→      case 404:
  1765→        return NotFoundException(message);
  1766→      case 422:
  1767→        return ValidationException(
  1768→          message: message,
  1769→          fieldErrors: e.details is Map<String, dynamic>
  1770→              ? e.details as Map<String, String>
  1771→              : {},
  1772→        );
  1773→      default:
  1774→        return ServerException(
  1775→          message: message,
  1776→          statusCode: statusCode,
  1777→          data: e.details,
  1778→        );
  1779→    }
  1780→  }
