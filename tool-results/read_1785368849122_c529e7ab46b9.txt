     1→-- ============================================================================
     2→-- ExamForge AI - AI Question Generation Engine Schema
     3→-- ============================================================================
     4→-- Production-ready schema for the AI-powered question generation engine.
     5→-- Supports: multi-provider AI, prompt management (database-driven),
     6→--           question generation, validation, improvement, document processing,
     7→--           review workflow, history tracking, and cost monitoring.
     8→--
     9→-- Prerequisites:
    10→--   Existing tables: schools, users, subjects, topics, subtopics,
    11→--                    question_bank, answer_options, matching_pairs,
    12→--                    ordering_items, fill_in_blank_answers
    13→--   Existing enums:  user_role, question_type, difficulty_level, exam_type,
    14→--                    subscription_status, exam_status, notification_type,
    15→--                    share_permission, import_status, content_type,
    16→--                    curriculum_standard_type
    17→--
    18→-- Performance: Designed for high-throughput with composite indexes,
    19→--              GIN indexes for JSONB, and partitioning-ready layout.
    20→-- ============================================================================
    21→
    22→BEGIN;
    23→
    24→-- ============================================================================
    25→-- 1. CUSTOM ENUMERATION TYPES
    26→-- ============================================================================
    27→-- All new enums required by the AI generation engine.
    28→-- Uses IF NOT EXISTS pattern to ensure idempotent migrations.
    29→-- ============================================================================
    30→

    32→BEGIN
    33→  -- ---------------------------------------------------------------------------
    34→  -- ai_provider: Supported AI model providers
    35→  -- ---------------------------------------------------------------------------
    36→  IF NOT EXISTS (SELECT 1 FROM pg_type WHERE typname = 'ai_provider') THEN
    37→    CREATE TYPE ai_provider AS ENUM (
    38→      'openai',
    39→      'gemini',
    40→      'claude',
    41→      'deepseek',
    42→      'grok',
    43→      'local_llm'
    44→    );
    45→  END IF;
    46→
    47→  -- ---------------------------------------------------------------------------
    48→  -- generation_status: Lifecycle states for an AI generation request
    49→  -- ---------------------------------------------------------------------------

    51→    CREATE TYPE generation_status AS ENUM (
    52→      'pending',
    53→      'processing',
    54→      'completed',
    55→      'failed',
    56→      'cancelled'
    57→    );
    58→  END IF;
    59→
    60→  -- ---------------------------------------------------------------------------
    61→  -- review_status: Human review workflow states for AI-generated questions
    62→  -- ---------------------------------------------------------------------------
    63→  IF NOT EXISTS (SELECT 1 FROM pg_type WHERE typname = 'review_status') THEN
    64→    CREATE TYPE review_status AS ENUM (
    65→      'pending',
    66→      'approved',
    67→      'rejected',
    68→      'needs_revision'
    69→    );
    70→  END IF;
    71→
    72→  -- ---------------------------------------------------------------------------
    73→  -- validation_severity: Severity levels for AI validation results
    74→  -- ---------------------------------------------------------------------------
    75→  IF NOT EXISTS (SELECT 1 FROM pg_type WHERE typname = 'validation_severity') THEN
    76→    CREATE TYPE validation_severity AS ENUM (
    77→      'info',
    78→      'warning',
    79→      'error',
    80→      'critical'
    81→    );
    82→  END IF;
    83→
    84→  -- ---------------------------------------------------------------------------
    85→  -- bloom_taxonomy: Bloom's Taxonomy cognitive levels for question alignment
    86→  -- ---------------------------------------------------------------------------
    87→  IF NOT EXISTS (SELECT 1 FROM pg_type WHERE typname = 'bloom_taxonomy') THEN
    88→    CREATE TYPE bloom_taxonomy AS ENUM (
    89→      'remember',
    90→      'understand',
    91→      'apply',
    92→      'analyze',
    93→      'evaluate',
    94→      'create'
    95→    );
    96→  END IF;
    97→
    98→  -- ---------------------------------------------------------------------------
    99→  -- prompt_type: Categories of AI prompt templates
   100→  -- ---------------------------------------------------------------------------
   101→  IF NOT EXISTS (SELECT 1 FROM pg_type WHERE typname = 'prompt_type') THEN
   102→    CREATE TYPE prompt_type AS ENUM (
   103→      'question_generation',
   104→      'question_improvement',
   105→      'question_validation',
   106→      'translation',
   107→      'document_extraction',
   108→      'distractor_generation',
   109→      'explanation_generation'
   110→    );
   111→  END IF;
   112→
   113→  -- ---------------------------------------------------------------------------
   114→  -- document_status: Lifecycle states for uploaded documents
   115→  -- ---------------------------------------------------------------------------
   116→  IF NOT EXISTS (SELECT 1 FROM pg_type WHERE typname = 'document_status') THEN
   117→    CREATE TYPE document_status AS ENUM (
   118→      'pending',
   119→      'processing',
   120→      'processed',
   121→      'failed'

   123→  END IF;
   124→
   125→  -- ---------------------------------------------------------------------------
   126→  -- curriculum_type: Nigerian and international curriculum frameworks

   131→      'neco',
   132→      'jamb',
   133→      'bece',
   134→      'cambridge_igcse',

   151→-- ============================================================================
   152→-- 2. AI PROVIDERS CONFIGURATION
   153→-- ============================================================================
   154→-- Stores configuration for each AI provider + model combination.
   155→-- Enables multi-provider routing, cost tracking, and capability detection.
   156→-- ============================================================================
   157→
   158→CREATE TABLE IF NOT EXISTS ai_providers_config (
   159→  id                        UUID PRIMARY KEY DEFAULT gen_random_uuid(),
   160→  provider                  ai_provider NOT NULL,
   161→  display_name              TEXT NOT NULL,                                -- e.g. "OpenAI GPT-4o"
   162→  model_name                TEXT NOT NULL,                                -- e.g. "gpt-4o", "gemini-1.5-pro"
   163→  api_endpoint              TEXT NOT NULL,                                -- REST endpoint URL
   164→  is_active                 BOOLEAN DEFAULT true,
   165→  max_tokens                INTEGER DEFAULT 4096,                        -- Maximum output tokens
   166→  temperature               NUMERIC(3,2) DEFAULT 0.70,                   -- Sampling temperature (0.00–2.00)
   167→  top_p                     NUMERIC(3,2) DEFAULT 0.95,                   -- Nucleus sampling threshold
   168→  cost_per_1k_input_tokens  NUMERIC(10,6) DEFAULT 0.0,                   -- USD per 1,000 input tokens
   169→  cost_per_1k_output_tokens NUMERIC(10,6) DEFAULT 0.0,                   -- USD per 1,000 output tokens
   170→  supports_streaming        BOOLEAN DEFAULT false,
   171→  supports_function_calling BOOLEAN DEFAULT false,
   172→  supports_vision           BOOLEAN DEFAULT false,                        -- Image / multimodal support
   173→  rate_limit_per_minute     INTEGER DEFAULT 60,                          -- Provider-imposed rate limit
   174→  config                    JSONB DEFAULT '{}',                           -- Provider-specific settings
   175→  created_at                TIMESTAMPTZ DEFAULT now(),
   176→  updated_at                TIMESTAMPTZ DEFAULT now(),
   177→
   178→  -- Prevent duplicate provider+model combinations
   179→  CONSTRAINT ai_providers_config_unique_provider_model UNIQUE (provider, model_name),
   180→  -- Validate temperature range
   181→  CONSTRAINT ai_providers_config_temperature_range CHECK (temperature >= 0.00 AND temperature <= 2.00),
   182→  -- Validate top_p range
   183→  CONSTRAINT ai_providers_config_top_p_range CHECK (top_p >= 0.00 AND top_p <= 1.00),
   184→  -- Non-negative cost values
   185→  CONSTRAINT ai_providers_config_input_cost_nonneg CHECK (cost_per_1k_input_tokens >= 0),
   186→  CONSTRAINT ai_providers_config_output_cost_nonneg CHECK (cost_per_1k_output_tokens >= 0),
   187→  -- Non-negative max_tokens
   188→  CONSTRAINT ai_providers_config_max_tokens_positive CHECK (max_tokens > 0),
   189→  -- Non-negative rate limit
   190→  CONSTRAINT ai_providers_config_rate_limit_positive CHECK (rate_limit_per_minute > 0)
   191→);
   192→
   193→COMMENT ON TABLE ai_providers_config IS 'Configuration registry for AI providers and models, enabling multi-provider routing and cost tracking';
   194→COMMENT ON COLUMN ai_providers_config.config IS 'Provider-specific settings (API version, headers, custom parameters) stored as JSONB';
   195→COMMENT ON COLUMN ai_providers_config.cost_per_1k_input_tokens IS 'Cost in USD per 1,000 input tokens for billing calculations';
   196→COMMENT ON COLUMN ai_providers_config.cost_per_1k_output_tokens IS 'Cost in USD per 1,000 output tokens for billing calculations';

   198→-- ============================================================================
   199→-- 3. PROMPT TEMPLATES (THE PROMPT MANAGEMENT SYSTEM)
   200→-- ============================================================================
   201→-- Database-driven prompt management enabling version control, A/B testing,
   202→-- per-curriculum customization, and quality tracking over time.
   203→-- Uses {{variable}} syntax for template interpolation.
   204→-- ============================================================================
   205→
   206→CREATE TABLE IF NOT EXISTS prompt_templates (
   207→  id                    UUID PRIMARY KEY DEFAULT gen_random_uuid(),
   208→  name                  TEXT NOT NULL,                                  -- e.g. "MCQ Generation - WAEC Science"
   209→  description           TEXT,                                          -- Purpose and usage notes
   210→  prompt_type           prompt_type NOT NULL,
   211→  provider              ai_provider,                                   -- NULL = works with any provider
   212→  curriculum            curriculum_type,                               -- Target curriculum framework
   213→  subject_id            UUID REFERENCES subjects(id) ON DELETE CASCADE,-- Subject scope
   214→  question_type         question_type,                                 -- For generation-type prompts
   215→  difficulty            difficulty_level,                              -- Target difficulty
   216→  bloom_level           bloom_taxonomy,                                -- Target cognitive level
   217→  language              VARCHAR(10) DEFAULT 'en',                      -- ISO 639-1 language code
   218→  system_prompt         TEXT NOT NULL,                                 -- System instruction sent to AI
   219→  user_prompt_template  TEXT NOT NULL,                                 -- Template with {{variables}}
   220→  variables             JSONB DEFAULT '[]',                            -- [{name, description, required, default}]
   221→  few_shot_examples     JSONB DEFAULT '[]',                            -- [{input, output}] examples
   222→  chain_of_thought      BOOLEAN DEFAULT false,                         -- Enable CoT reasoning
   223→  output_format         JSONB DEFAULT '{}',                            -- Expected JSON output schema
   224→  is_active             BOOLEAN DEFAULT true,
   225→  is_default            BOOLEAN DEFAULT false,                         -- Default prompt for this type
   226→  version               INTEGER DEFAULT 1,                             -- Template version number
   227→  parent_id             UUID REFERENCES prompt_templates(id),          -- Previous version in chain
   228→  quality_score         NUMERIC(3,2) DEFAULT 0.00,                     -- 0.00–5.00, tracked over time
   229→  usage_count           INTEGER DEFAULT 0,                             -- Times this template was used
   230→  success_rate          NUMERIC(5,2) DEFAULT 0.00,                     -- % of generations passing validation
   231→  created_by            UUID REFERENCES users(id),
   232→  school_id             UUID REFERENCES schools(id),                   -- NULL = platform-wide
   233→  created_at            TIMESTAMPTZ DEFAULT now(),
   234→  updated_at            TIMESTAMPTZ DEFAULT now(),
   235→
   236→  -- Only one default per (prompt_type, provider, subject_id, curriculum) combination
   237→  CONSTRAINT prompt_templates_one_default EXCLUDE (
   238→    CASE WHEN is_default THEN prompt_type END WITH =,
   239→    CASE WHEN is_default THEN COALESCE(provider, 'openai'::ai_provider) END WITH =,
   240→    CASE WHEN is_default THEN COALESCE(subject_id, '00000000-0000-0000-0000-000000000000'::uuid) END WITH =,
   241→    CASE WHEN is_default THEN COALESCE(curriculum, 'custom'::curriculum_type) END WITH =
   242→  ) WHERE (is_default = true),
   243→  -- Version must be positive
   244→  CONSTRAINT prompt_templates_version_positive CHECK (version > 0),
   245→  -- Quality score range: 0–5
   246→  CONSTRAINT prompt_templates_quality_score_range CHECK (quality_score >= 0.00 AND quality_score <= 5.00),
   247→  -- Success rate range: 0–100
   248→  CONSTRAINT prompt_templates_success_rate_range CHECK (success_rate >= 0.00 AND success_rate <= 100.00),
   249→  -- Usage count non-negative
   250→  CONSTRAINT prompt_templates_usage_count_nonneg CHECK (usage_count >= 0)
   251→);
   252→
   253→COMMENT ON TABLE prompt_templates IS 'Database-driven prompt management system with versioning, A/B testing, and quality tracking';
   254→COMMENT ON COLUMN prompt_templates.user_prompt_template IS 'Template string using {{variable}} syntax for interpolation (e.g. {{subject}}, {{topic}}, {{num_questions}})';
   255→COMMENT ON COLUMN prompt_templates.variables IS 'JSON array of variable definitions: [{name, description, required, default}]';
   256→COMMENT ON COLUMN prompt_templates.few_shot_examples IS 'JSON array of example input/output pairs for few-shot prompting';
   257→COMMENT ON COLUMN prompt_templates.output_format IS 'JSON Schema describing the expected AI response structure';
   258→COMMENT ON COLUMN prompt_templates.parent_id IS 'Previous version of this template, forming a version chain';
   259→COMMENT ON COLUMN prompt_templates.quality_score IS 'Aggregate quality score (0.00–5.00) computed from generation outcomes';
   260→COMMENT ON COLUMN prompt_templates.success_rate IS 'Percentage of generations using this template that passed validation';

   262→-- ============================================================================
   263→-- 4. AI GENERATION REQUESTS
   264→-- ============================================================================
   265→-- Master record for every AI generation attempt. Captures the full lifecycle
   266→-- from request submission through processing to completion, including
   267→-- token usage, cost, and timing metrics.
   268→-- ============================================================================
   269→
   270→CREATE TABLE IF NOT EXISTS ai_generation_requests (
   271→  id                    UUID PRIMARY KEY DEFAULT gen_random_uuid(),
   272→  school_id             UUID REFERENCES schools(id) ON DELETE CASCADE,
   273→  requested_by          UUID NOT NULL REFERENCES users(id),
   274→  provider              ai_provider NOT NULL DEFAULT 'openai',
   275→  model_name            TEXT NOT NULL,                                  -- e.g. "gpt-4o"
   276→  prompt_template_id    UUID REFERENCES prompt_templates(id),           -- Template used (nullable for custom)
   277→  generation_type       prompt_type NOT NULL DEFAULT 'question_generation',
   278→  status                generation_status NOT NULL DEFAULT 'pending',
   279→  input_params          JSONB NOT NULL DEFAULT '{}',                    -- Subject, topic, difficulty, etc.
   280→  system_prompt         TEXT,                                           -- Resolved system prompt
   281→  user_prompt           TEXT,                                           -- Resolved user prompt
   282→  raw_response          JSONB,                                          -- Raw AI provider response
   283→  processed_response    JSONB,                                          -- Parsed and structured output
   284→  input_tokens          INTEGER DEFAULT 0,
   285→  output_tokens         INTEGER DEFAULT 0,
   286→  total_cost            NUMERIC(10,6) DEFAULT 0.0,                     -- USD
   287→  generation_time_ms    INTEGER,                                        -- Wall-clock time
   288→  error_message         TEXT,
   289→  retry_count           INTEGER DEFAULT 0,
   290→  priority              INTEGER DEFAULT 0,                             -- Higher = processed sooner
   291→  started_at            TIMESTAMPTZ,
   292→  completed_at          TIMESTAMPTZ,
   293→  created_at            TIMESTAMPTZ DEFAULT now(),
   294→
   295→  -- Token counts non-negative
   296→  CONSTRAINT ai_gen_req_input_tokens_nonneg CHECK (input_tokens >= 0),
   297→  CONSTRAINT ai_gen_req_output_tokens_nonneg CHECK (output_tokens >= 0),
   298→  -- Cost non-negative
   299→  CONSTRAINT ai_gen_req_total_cost_nonneg CHECK (total_cost >= 0),
   300→  -- Retry count non-negative
   301→  CONSTRAINT ai_gen_req_retry_count_nonneg CHECK (retry_count >= 0),
   302→  -- Completed_at must be after started_at when both set
   303→  CONSTRAINT ai_gen_req_completed_after_started
   304→    CHECK (completed_at IS NULL OR started_at IS NULL OR completed_at >= started_at)
   305→);
   306→
   307→COMMENT ON TABLE ai_generation_requests IS 'Master record for every AI generation request, tracking full lifecycle from submission to completion';
   308→COMMENT ON COLUMN ai_generation_requests.input_params IS 'JSON object with generation parameters: subject, topic, difficulty, num_questions, bloom_level, exam_type, keywords, custom_instructions';
   309→COMMENT ON COLUMN ai_generation_requests.raw_response IS 'Unprocessed response from the AI provider for debugging and auditing';
   310→COMMENT ON COLUMN ai_generation_requests.processed_response IS 'Parsed and structured output ready for question creation';
   311→COMMENT ON COLUMN ai_generation_requests.priority IS 'Queue priority — higher values are processed first';

   313→-- ============================================================================
   314→-- 5. AI GENERATED QUESTIONS
   315→-- ============================================================================
   316→-- Individual questions produced by AI generation requests. Each row tracks
   317→-- the question content, answer options, validation state, and the human
   318→-- review workflow. On approval, questions can be promoted to question_bank.
   319→-- ============================================================================
   320→
   321→CREATE TABLE IF NOT EXISTS ai_generated_questions (
   322→  id                      UUID PRIMARY KEY DEFAULT gen_random_uuid(),
   323→  generation_request_id   UUID NOT NULL REFERENCES ai_generation_requests(id) ON DELETE CASCADE,
   324→  question_bank_id        UUID REFERENCES question_bank(id) ON DELETE SET NULL,  -- Linked after approval
   325→  school_id               UUID REFERENCES schools(id) ON DELETE CASCADE,
   326→  question_type           question_type NOT NULL,
   327→  difficulty              difficulty_level NOT NULL DEFAULT 'medium',
   328→  bloom_level             bloom_taxonomy DEFAULT 'understand',
   329→  content                 TEXT NOT NULL,                                 -- Question text
   330→  content_json            JSONB DEFAULT '{}',                            -- Structured content
   331→  answer_options          JSONB DEFAULT '[]',                            -- [{content, isCorrect, explanation, marks}]
   332→  matching_pairs          JSONB DEFAULT '[]',                            -- [{left, right}]
   333→  ordering_items          JSONB DEFAULT '[]',                            -- [{item, correctPosition}]
   334→  fill_in_blank_answers   JSONB DEFAULT '[]',                            -- [{answer, acceptableVariations, marks}]
   335→  explanation             TEXT,                                          -- Answer explanation
   336→  suggested_references    TEXT,                                          -- Source references
   337→  marks                   NUMERIC(5,2) DEFAULT 1.00,
   338→  estimated_time_seconds  INTEGER,                                      -- Suggested time to answer
   339→  confidence_score        NUMERIC(3,2) DEFAULT 0.00,                    -- 0.00–1.00 AI confidence
   340→  curriculum_alignment    JSONB DEFAULT '{}',                            -- Mapping to curriculum standards
   341→  review_status           review_status NOT NULL DEFAULT 'pending',
   342→  reviewed_by             UUID REFERENCES users(id),
   343→  reviewed_at             TIMESTAMPTZ,
   344→  review_notes            TEXT,                                          -- Reviewer feedback
   345→  teacher_edits           JSONB DEFAULT '{}',                            -- Tracks what the teacher changed
   346→  is_edited               BOOLEAN DEFAULT false,                        -- Whether teacher modified the question
   347→  is_approved             BOOLEAN DEFAULT false,                        -- Quick approval flag
   348→  metadata                JSONB DEFAULT '{}',                            -- AI model info, prompt, params
   349→  created_at              TIMESTAMPTZ DEFAULT now(),
   350→  updated_at              TIMESTAMPTZ DEFAULT now(),
   351→
   352→  -- Marks non-negative
   353→  CONSTRAINT ai_gen_q_marks_nonneg CHECK (marks >= 0),
   354→  -- Confidence score range
   355→  CONSTRAINT ai_gen_q_confidence_range CHECK (confidence_score >= 0.00 AND confidence_score <= 1.00),
   356→  -- Estimated time positive when set
   357→  CONSTRAINT ai_gen_q_time_positive CHECK (estimated_time_seconds IS NULL OR estimated_time_seconds > 0)
   358→);
   359→
   360→COMMENT ON TABLE ai_generated_questions IS 'Individual questions produced by AI, with review workflow and promotion path to question_bank';
   361→COMMENT ON COLUMN ai_generated_questions.question_bank_id IS 'Set when the question is approved and promoted to the main question bank';
   362→COMMENT ON COLUMN ai_generated_questions.answer_options IS 'JSON array of options: [{content: text, isCorrect: bool, explanation: text, marks: number}]';
   363→COMMENT ON COLUMN ai_generated_questions.confidence_score IS 'AI-assessed confidence in the question quality (0.00–1.00)';
   364→COMMENT ON COLUMN ai_generated_questions.teacher_edits IS 'JSON tracking teacher modifications: {field: {original: ..., modified: ...}}';
   365→COMMENT ON COLUMN ai_generated_questions.metadata IS 'AI generation metadata: model, prompt hash, generation params, template version';

   438→-- ============================================================================

   446→CREATE TABLE IF NOT EXISTS ai_document_uploads (

  1189→-- ---------------------------------------------------------------------------
  1190→-- calculate_generation_cost()
  1191→-- Compute cost for an AI generation request based on provider/model rates
  1192→-- ---------------------------------------------------------------------------
  1193→CREATE OR REPLACE FUNCTION calculate_generation_cost(
  1194→  p_provider     ai_provider,
  1195→  p_model_name   TEXT,
  1196→  p_input_tokens INTEGER,
  1197→  p_output_tokens INTEGER
  1198→)
  1199→RETURNS NUMERIC(10,6) AS $$
  1200→DECLARE
  1201→  v_input_cost  NUMERIC(10,6);
  1202→  v_output_cost NUMERIC(10,6);
  1203→BEGIN
  1204→  -- Look up per-1k token rates from the provider config
  1205→  SELECT
  1206→    cost_per_1k_input_tokens,
  1207→    cost_per_1k_output_tokens
  1208→  INTO
  1209→    v_input_cost,
  1210→    v_output_cost
  1211→  FROM ai_providers_config
  1212→  WHERE provider = p_provider
  1213→    AND model_name = p_model_name
  1214→    AND is_active = true
  1215→  LIMIT 1;
  1216→
  1217→  -- Fallback to zero if provider not found
  1218→  IF v_input_cost IS NULL THEN
  1219→    v_input_cost := 0.0;
  1220→    v_output_cost := 0.0;
  1221→  END IF;
  1222→
  1223→  -- Calculate: (tokens / 1000) * rate per 1k tokens
  1224→  RETURN (p_input_tokens::NUMERIC / 1000.0 * v_input_cost)
  1225→       + (p_output_tokens::NUMERIC / 1000.0 * v_output_cost);
  1226→END;
  1227→$$ LANGUAGE plpgsql STABLE;
  1228→
  1229→COMMENT ON FUNCTION calculate_generation_cost IS 'Compute cost for an AI generation request based on provider/model per-1k token rates';

  1231→-- ---------------------------------------------------------------------------
  1232→-- process_generation_queue()
  1233→-- Process the next item in the generation queue.
  1234→-- Returns the generation_request_id of the processed item, or NULL if empty.
  1235→-- ---------------------------------------------------------------------------
  1236→CREATE OR REPLACE FUNCTION process_generation_queue()
  1237→RETURNS UUID AS $$
  1238→DECLARE
  1239→  v_queue_item        RECORD;
  1240→  v_generation_req_id UUID;
  1241→BEGIN
  1242→  -- Atomically select and lock the next highest-priority, due item
  1243→  SELECT * INTO v_queue_item
  1244→  FROM ai_generation_queue
  1245→  WHERE status IN ('queued', 'failed')
  1246→    AND next_attempt_at <= now()
  1247→    AND attempts < max_attempts
  1248→  ORDER BY priority DESC, next_attempt_at ASC
  1249→  LIMIT 1
  1250→  FOR UPDATE SKIP LOCKED;
  1251→
  1252→  -- Nothing to process
  1253→  IF NOT FOUND THEN
  1254→    RETURN NULL;
  1255→  END IF;
  1256→
  1257→  -- Mark queue item as processing
  1258→  UPDATE ai_generation_queue
  1259→  SET
  1260→    status      = 'processing',
  1261→    attempts    = attempts + 1,
  1262→    updated_at  = now()
  1263→  WHERE id = v_queue_item.id;
  1264→
  1265→  -- Update the generation request status to processing
  1266→  UPDATE ai_generation_requests
  1267→  SET
  1268→    status     = 'processing',
  1269→    started_at = now()
  1270→  WHERE id = v_queue_item.generation_request_id;
  1271→
  1272→  RETURN v_queue_item.generation_request_id;
  1273→END;
  1274→$$ LANGUAGE plpgsql;
  1275→
  1276→COMMENT ON FUNCTION process_generation_queue IS 'Atomically dequeue and lock the next highest-priority generation job for processing. Returns the generation_request_id or NULL if queue is empty.';

  1278→-- ---------------------------------------------------------------------------
  1279→-- update_prompt_quality_score()
  1280→-- Recalculate the quality_score for a prompt template based on
  1281→-- the ratio of successful generations (completed + approved questions)
  1282→-- to total generations using this template.
  1283→-- ---------------------------------------------------------------------------
  1284→CREATE OR REPLACE FUNCTION update_prompt_quality_score(
  1285→  p_template_id UUID
  1286→)
  1287→RETURNS VOID AS $$
  1288→DECLARE
  1289→  v_total_generations  INTEGER;
  1290→  v_successful         INTEGER;
  1291→  v_new_success_rate   NUMERIC(5,2);
  1292→  v_new_quality_score  NUMERIC(3,2);
  1293→BEGIN
  1294→  -- Count total and successful generations for this template
  1295→  SELECT
  1296→    COUNT(*),
  1297→    COUNT(*) FILTER (
  1298→      WHERE status = 'completed'
  1299→        AND EXISTS (
  1300→          SELECT 1 FROM ai_generated_questions q
  1301→          WHERE q.generation_request_id = ai_generation_requests.id
  1302→            AND q.review_status IN ('approved', 'needs_revision')
  1303→        )
  1304→    )
  1305→  INTO
  1306→    v_total_generations,
  1307→    v_successful
  1308→  FROM ai_generation_requests
  1309→  WHERE prompt_template_id = p_template_id;
  1310→
  1311→  -- Calculate success rate (percentage)
  1312→  IF v_total_generations > 0 THEN
  1313→    v_new_success_rate := ROUND(
  1314→      (v_successful::NUMERIC / v_total_generations::NUMERIC) * 100.0,
  1315→      2
  1316→    );
  1317→  ELSE
  1318→    v_new_success_rate := 0.00;
  1319→  END IF;
  1320→
  1321→  -- Map success rate to quality score (0-5 scale)
  1322→  -- Simple linear mapping: quality_score = success_rate / 20
  1323→  v_new_quality_score := ROUND(v_new_success_rate / 20.0, 2);
  1324→
  1325→  -- Update the template
  1326→  UPDATE prompt_templates
  1327→  SET
  1328→    success_rate   = v_new_success_rate,
  1329→    quality_score  = v_new_quality_score,
  1330→    usage_count    = v_total_generations,
  1331→    updated_at     = now()
  1332→  WHERE id = p_template_id;
  1333→END;
  1334→$$ LANGUAGE plpgsql;
  1335→
  1336→COMMENT ON FUNCTION update_prompt_quality_score IS 'Recalculate quality_score and success_rate for a prompt template based on historical generation outcomes';

  1338→-- ---------------------------------------------------------------------------
  1339→-- approve_generated_question()
  1340→-- Approve an AI-generated question and optionally promote it to question_bank.
  1341→-- Sets review_status to 'approved', records the reviewer, and if
  1342→-- p_save_to_bank = true, creates a question_bank entry and links it.
  1343→-- ---------------------------------------------------------------------------
  1344→CREATE OR REPLACE FUNCTION approve_generated_question(
  1345→  p_question_id  UUID,
  1346→  p_reviewer_id  UUID,
  1347→  p_review_notes TEXT DEFAULT NULL,
  1348→  p_save_to_bank BOOLEAN DEFAULT true
  1349→)
  1350→RETURNS UUID AS $$
  1351→DECLARE
  1352→  v_gen_question  RECORD;
  1353→  v_question_bank_id UUID;
  1354→BEGIN
  1355→  -- Fetch the generated question
  1356→  SELECT * INTO v_gen_question
  1357→  FROM ai_generated_questions
  1358→  WHERE id = p_question_id
  1359→  FOR UPDATE;
  1360→
  1361→  IF NOT FOUND THEN
  1362→    RAISE EXCEPTION 'AI generated question % not found', p_question_id;
  1363→  END IF;
  1364→
  1365→  -- Update the generated question as approved
  1366→  UPDATE ai_generated_questions
  1367→  SET
  1368→    review_status = 'approved',
  1369→    reviewed_by   = p_reviewer_id,
  1370→    reviewed_at   = now(),
  1371→    review_notes  = COALESCE(p_review_notes, review_notes),
  1372→    is_approved   = true,
  1373→    updated_at    = now()
  1374→  WHERE id = p_question_id;
  1375→
  1376→  -- Optionally promote to question_bank
  1377→  IF p_save_to_bank AND v_gen_question.question_bank_id IS NULL THEN
  1378→    INSERT INTO question_bank (
  1379→      school_id,
  1380→      subject_id,
  1381→      topic_id,
  1382→      subtopic_id,
  1383→      question_type,
  1384→      difficulty,
  1385→      content,
  1386→      content_json,
  1387→      explanation,
  1388→      marks,
  1389→      time_allowed_seconds,
  1390→      is_published,
  1391→      created_by,
  1392→      metadata
  1393→    ) VALUES (
  1394→      v_gen_question.school_id,
  1395→      (v_gen_question.generation_request_id::TEXT)::UUID,  -- placeholder; will be corrected below
  1396→      NULL,   -- topic_id — not available directly on generated question
  1397→      NULL,   -- subtopic_id
  1398→      v_gen_question.question_type,
  1399→      v_gen_question.difficulty,
  1400→      v_gen_question.content,
  1401→      v_gen_question.content_json,
  1402→      v_gen_question.explanation,
  1403→      v_gen_question.marks,
  1404→      v_gen_question.estimated_time_seconds,
  1405→      false,  -- not published until explicitly published
  1406→      p_reviewer_id,
  1407→      jsonb_build_object(
  1408→        'source', 'ai_generation',
  1409→        'ai_question_id', p_question_id,
  1410→        'bloom_level', v_gen_question.bloom_level,
  1411→        'confidence_score', v_gen_question.confidence_score
  1412→      )
  1413→    )
  1414→    ON CONFLICT DO NOTHING
  1415→    RETURNING id INTO v_question_bank_id;
  1416→
  1417→    -- If we got a question_bank_id, link it back and create answer options
  1418→    IF v_question_bank_id IS NOT NULL THEN
  1419→      -- Fix: Update with proper subject_id from the generation request
  1420→      UPDATE question_bank
  1421→      SET subject_id = gr.input_params->>'subject_id'
  1422→      FROM ai_generation_requests gr
  1423→      WHERE gr.id = v_gen_question.generation_request_id
  1424→        AND question_bank.id = v_question_bank_id;
  1425→
  1426→      -- Link back to the generated question
  1427→      UPDATE ai_generated_questions
  1428→      SET question_bank_id = v_question_bank_id,
  1429→          updated_at = now()
  1430→      WHERE id = p_question_id;
  1431→
  1432→      -- Create answer_options rows from the JSONB answer_options array
  1433→      INSERT INTO answer_options (question_id, content, is_correct, explanation, marks, sort_order)
  1434→      SELECT
  1435→        v_question_bank_id,
  1436→        (opt->>'content')::TEXT,
  1437→        COALESCE((opt->>'isCorrect')::BOOLEAN, false),
  1438→        opt->>'explanation',
  1439→        COALESCE((opt->>'marks')::NUMERIC, 0.00),
  1440→        ROW_NUMBER() OVER () - 1
  1441→      FROM jsonb_array_elements(v_gen_question.answer_options) AS opt
  1442→      WHERE v_gen_question.answer_options IS NOT NULL
  1443→        AND jsonb_array_length(v_gen_question.answer_options) > 0;
  1444→    END IF;
  1445→  END IF;
  1446→
  1447→  -- Resolve any pending validation results for this question
  1448→  UPDATE ai_validation_results
  1449→  SET
  1450→    is_resolved = true,
  1451→    resolved_by = p_reviewer_id,
  1452→    resolved_at = now()
  1453→  WHERE generated_question_id = p_question_id
  1454→    AND is_resolved = false;
  1455→
  1456→  -- Update usage stats: increment questions_approved for today
  1457→  INSERT INTO ai_usage_stats (school_id, provider, model_name, date, questions_approved)
  1458→  SELECT
  1459→    gr.school_id,
  1460→    gr.provider,
  1461→    gr.model_name,
  1462→    CURRENT_DATE,
  1463→    1
  1464→  FROM ai_generation_requests gr
  1465→  WHERE gr.id = v_gen_question.generation_request_id
  1466→  ON CONFLICT (school_id, provider, model_name, date)
  1467→  DO UPDATE SET questions_approved = ai_usage_stats.questions_approved + 1;
  1468→
  1469→  RETURN v_question_bank_id;
  1470→END;
  1471→$$ LANGUAGE plpgsql;
  1472→
  1473→COMMENT ON FUNCTION approve_generated_question IS 'Approve an AI-generated question, optionally promoting it to question_bank with answer options. Resolves pending validations and updates usage stats.';

  1480→CREATE OR REPLACE FUNCTION reject_generated_question(
