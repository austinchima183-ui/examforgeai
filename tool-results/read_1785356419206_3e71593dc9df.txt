     1→-- ============================================================================
     2→-- ExamForge AI — Billing & Subscription Schema
     3→-- ============================================================================
     4→-- Supports three independent billing models:
     5→--   1. Teacher SaaS  (individual teacher subscribes independently)
     6→--   2. School SaaS   (school subscribes, manages teachers/students/parents)
     7→--   3. Enterprise SaaS (organizations, ministries, exam bodies — custom pricing)
     8→--
     9→-- Payment provider: Flutterwave (Standard Checkout, Webhooks, Recurring)
    10→-- AI Credit System, Coupons, Referrals, Invoices, Licenses, Revenue Analytics
    11→-- ============================================================================

    15→-- ════════════════════════════════════════════════════════════════════════════
    16→-- CUSTOM ENUM TYPES
    17→-- ════════════════════════════════════════════════════════════════════════════
    18→
    19→DO $$ BEGIN
    20→  -- Billing model: who is the subscriber?
    21→  CREATE TYPE billing_model AS ENUM ('teacher_saas', 'school_saas', 'enterprise_saas');
    22→
    23→  -- Subscription plan tier
    24→  CREATE TYPE plan_tier AS ENUM ('free', 'starter', 'professional', 'enterprise');
    25→
    26→  -- Subscription status lifecycle
    27→  CREATE TYPE subscription_status AS ENUM (
    28→    'trial', 'active', 'past_due', 'paused',
    29→    'cancelled', 'expired', 'pending_activation'
    30→  );
    31→
    32→  -- Transaction/payment status
    33→  CREATE TYPE transaction_status AS ENUM (
    34→    'pending', 'processing', 'successful', 'failed',
    35→    'refunded', 'partially_refunded', 'disputed', 'voided'
    36→  );
    37→
    38→  -- Payment channel
    39→  CREATE TYPE payment_channel AS ENUM (
    40→    'card', 'bank_transfer', 'ussd', 'mobile_money',
    41→    'qr_code', 'credit', 'coupon', 'refund'
    42→  );
    43→
    44→  -- Coupon discount type
    45→  CREATE TYPE coupon_discount_type AS ENUM (
    46→    'percentage', 'fixed_amount', 'free_trial', 'fixed_per_seat'
    47→  );
    48→
    49→  -- Referral reward type
    50→  CREATE TYPE referral_reward_type AS ENUM (
    51→    'credit_days', 'percentage_discount', 'fixed_credit', 'ai_credits'
    52→  );
    53→
    54→  -- AI credit transaction type
    55→  CREATE TYPE credit_transaction_type AS ENUM (
    56→    'monthly_allocation', 'purchase', 'usage', 'expiration',
    57→    'bonus', 'referral_reward', 'admin_adjustment', 'refund'
    58→  );
    59→
    60→  -- License type
    61→  CREATE TYPE license_type AS ENUM (
    62→    'school', 'teacher', 'branch', 'seat', 'custom'
    63→  );
    64→
    65→  -- Invoice status
    66→  CREATE TYPE invoice_status AS ENUM (
    67→    'draft', 'issued', 'paid', 'partially_paid',
    68→    'overdue', 'cancelled', 'void', 'credit_note'
    69→  );
    70→
    71→  -- Webhook event processing status
    72→  CREATE TYPE webhook_status AS ENUM (
    73→    'received', 'processing', 'processed', 'failed', 'retrying'
    74→  );
    75→
    76→  -- Billing notification type
    77→  CREATE TYPE billing_notification_type AS ENUM (
    78→    'payment_success', 'payment_failed', 'trial_ending',
    79→    'subscription_renewal', 'plan_expiring', 'low_ai_credits',
    80→    'invoice_generated', 'refund_status', 'card_expiring',
    81→    'subscription_cancelled', 'upgrade_available'
    82→  );
    83→
    84→EXCEPTION WHEN duplicate_object THEN NULL;
    85→END $$;

   970→-- Auto-update updated_at timestamp
   971→CREATE OR REPLACE FUNCTION update_billing_updated_at()
   972→RETURNS TRIGGER AS $$
   973→BEGIN
   974→  NEW.updated_at = now();
   975→  RETURN NEW;
   976→END;
   977→$$ LANGUAGE plpgsql;
   978→
   979→-- Apply to all tables with updated_at
   980→DO $$ DECLARE t TEXT; BEGIN
   981→  FOR t IN SELECT table_name FROM information_schema.columns
   982→    WHERE column_name = 'updated_at'
   983→    AND table_schema = 'public'
   984→    AND table_name IN (
   985→      'subscription_plans', 'subscriptions', 'transactions', 'invoices',
   986→      'ai_credit_balances', 'coupons', 'referral_codes', 'referral_tracking',
   987→      'licenses', 'webhook_events', 'billing_notification_preferences',
   988→      'revenue_reports', 'school_billing_profiles', 'ai_credit_packs',
   989→      'billing_rate_limits'
   990→    )
   991→  LOOP
   992→    EXECUTE format(
   993→      'CREATE TRIGGER trg_%s_updated_at
   994→       BEFORE UPDATE ON %I
   995→       FOR EACH ROW EXECUTE FUNCTION update_billing_updated_at()',
   996→      t, t
   997→    );
   998→  EXCEPTION WHEN others THEN NULL;
   999→  END LOOP;
  1000→END $$;

  1002→-- Generate invoice number
  1003→CREATE OR REPLACE FUNCTION generate_invoice_number()
  1004→RETURNS TEXT AS $$
  1005→DECLARE
  1006→  seq_val INT;
  1007→  year_part TEXT := to_char(now(), 'YY');
  1008→  month_part TEXT := to_char(now(), 'MM');
  1009→BEGIN
  1010→  SELECT nextval('invoice_number_seq') INTO seq_val;
  1011→  RETURN 'INV-' || year_part || month_part || '-' || lpad(seq_val::text, 6, '0');
  1012→END;
  1013→$$ LANGUAGE plpgsql;

  1041→-- Check subscription limits (used by RLS and API)
  1042→CREATE OR REPLACE FUNCTION check_subscription_limit(
  1043→  p_school_id UUID,
  1044→  p_limit_type TEXT  -- 'students', 'teachers', 'storage', 'exams'
  1045→)
  1046→RETURNS BOOLEAN AS $$
  1047→DECLARE
  1048→  v_plan RECORD;
  1049→  v_current_count INT;
  1050→BEGIN
  1051→  SELECT sp.max_students, sp.max_teachers, sp.max_storage_mb, sp.max_exams_per_month
  1052→  INTO v_plan
  1053→  FROM subscriptions s
  1054→  JOIN subscription_plans sp ON s.plan_id = sp.id
  1055→  WHERE s.school_id = p_school_id
  1056→    AND s.status IN ('active', 'trial')
  1057→  ORDER BY s.created_at DESC LIMIT 1;
  1058→
  1059→  IF NOT FOUND THEN RETURN false; END IF;
  1060→
  1061→  CASE p_limit_type
  1062→    WHEN 'students' THEN
  1063→      SELECT count(*) INTO v_current_count FROM students WHERE school_id = p_school_id AND is_active = true;
  1064→      RETURN v_plan.max_students = 0 OR v_current_count < v_plan.max_students;
  1065→    WHEN 'teachers' THEN
  1066→      SELECT count(*) INTO v_current_count FROM teachers WHERE school_id = p_school_id AND is_active = true;
  1067→      RETURN v_plan.max_teachers = 0 OR v_current_count < v_plan.max_teachers;
  1068→    WHEN 'exams' THEN
  1069→      RETURN v_plan.max_exams_per_month = 0; -- simplified; real impl counts monthly exams
  1070→    ELSE
  1071→      RETURN true;
  1072→  END CASE;
  1073→END;
  1074→$$ LANGUAGE plpgsql SECURITY DEFINER;

  1076→-- Consume AI credits (called by feature modules)
  1077→CREATE OR REPLACE FUNCTION consume_ai_credits(
  1078→  p_owner_id UUID,
  1079→  p_owner_type billing_model,
  1080→  p_credits INT,
  1081→  p_feature_name TEXT,
  1082→  p_reference_id UUID DEFAULT NULL,
  1083→  p_estimated_cost_usd NUMERIC DEFAULT 0
  1084→)
  1085→RETURNS BOOLEAN AS $$
  1086→DECLARE
  1087→  v_balance RECORD;
  1088→  v_balance_after INT;
  1089→BEGIN
  1090→  -- Lock the balance row for atomic update
  1091→  SELECT * INTO v_balance
  1092→  FROM ai_credit_balances
  1093→  WHERE owner_id = p_owner_id AND owner_type = p_owner_type
  1094→  FOR UPDATE;
  1095→
  1096→  IF NOT FOUND THEN RETURN false; END IF;
  1097→
  1098→  IF v_balance.remaining_credits < p_credits THEN
  1099→    RETURN false;  -- Insufficient credits
  1100→  END IF;
  1101→
  1102→  v_balance_after := v_balance.remaining_credits - p_credits;
  1103→
  1104→  -- Update balance
  1105→  UPDATE ai_credit_balances SET
  1106→    used_credits = used_credits + p_credits,
  1107→    remaining_credits = v_balance_after
  1108→  WHERE id = v_balance.id;
  1109→
  1110→  -- Record transaction
  1111→  INSERT INTO ai_credit_transactions (
  1112→    balance_id, owner_id, owner_type, school_id,
  1113→    transaction_type, credits, balance_before, balance_after,
  1114→    feature_name, reference_id, estimated_cost_usd
  1115→  ) VALUES (
  1116→    v_balance.id, p_owner_id, p_owner_type, v_balance.school_id,
  1117→    'usage', -p_credits, v_balance.remaining_credits, v_balance_after,
  1118→    p_feature_name, p_reference_id, p_estimated_cost_usd
  1119→  );
  1120→
  1121→  RETURN true;
  1122→END;
  1123→$$ LANGUAGE plpgsql SECURITY DEFINER;

  1125→-- Expire old credits (run daily via cron/supabase function)
  1126→CREATE OR REPLACE FUNCTION expire_ai_credits()
  1127→RETURNS INT AS $$
  1128→DECLARE
  1129→  v_expired_count INT := 0;
  1130→  v_balance RECORD;
  1131→BEGIN
  1132→  FOR v_balance IN
  1133→    SELECT * FROM ai_credit_balances
  1134→    WHERE credits_expire = true
  1135→      AND expiration_date IS NOT NULL
  1136→      AND expiration_date < now()
  1137→      AND remaining_credits > 0
  1138→  LOOP
  1139→    -- Record expiration transaction
  1140→    INSERT INTO ai_credit_transactions (
  1141→      balance_id, owner_id, owner_type, school_id,
  1142→      transaction_type, credits, balance_before, balance_after,
  1143→      feature_name
  1144→    ) VALUES (
  1145→      v_balance.id, v_balance.owner_id, v_balance.owner_type, v_balance.school_id,
  1146→      'expiration', -v_balance.remaining_credits, v_balance.remaining_credits, 0,
  1147→      'credit_expiration'
  1148→    );
  1149→
  1150→    -- Reset balance
  1151→    UPDATE ai_credit_balances SET
  1152→      remaining_credits = 0,
  1153→      used_credits = used_credits + remaining_credits
  1154→    WHERE id = v_balance.id;
  1155→
  1156→    v_expired_count := v_expired_count + 1;
  1157→  END LOOP;
  1158→
  1159→  RETURN v_expired_count;
  1160→END;
  1161→$$ LANGUAGE plpgsql SECURITY DEFINER;

  1517→-- Revenue summary by month
  1518→CREATE OR REPLACE VIEW v_monthly_revenue AS
  1519→SELECT
  1520→  date_trunc('month', t.initiated_at) AS month,
  1521→  t.currency,
  1522→  count(*) AS transaction_count,
  1523→  sum(t.amount) FILTER (WHERE t.status = 'successful') AS successful_volume,
  1524→  sum(t.net_amount) FILTER (WHERE t.status = 'successful') AS net_revenue,
  1525→  sum(t.amount) FILTER (WHERE t.status = 'refunded') AS refund_volume,
  1526→  count(*) FILTER (WHERE t.status = 'failed') AS failed_count
  1527→FROM transactions t
  1528→GROUP BY 1, 2
  1529→ORDER BY 1 DESC;
