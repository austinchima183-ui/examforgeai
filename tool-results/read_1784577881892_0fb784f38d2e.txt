   241→  @override
   242→  Future<List<SubscriptionPlanModel>> getSubscriptionPlans({
   243→    String? billingModel,
   244→    bool activeOnly = true,
   245→  }) async {
   246→    try {
   247→      var query = _supabase.from(_plansTable).select();
   248→
   249→      if (billingModel != null) {
   250→        query = query.eq('billing_model', billingModel);
   251→      }
   252→      if (activeOnly) {
   253→        query = query.eq('is_active', true);
   254→      }
   255→
   256→      final response = query.order('sort_order', ascending: true);
   257→
   258→      final list = await response;
   259→      AppLogger.info('Fetched ${list.length} subscription plans');
   260→      return list
   261→          .map<SubscriptionPlanModel>(
   262→            (row) => SubscriptionPlanModel.fromJson(row),
   263→          )
   264→          .toList();
   265→    } on sb.PostgrestException catch (e) {
   266→      throw _mapPostgrestException(e);
   267→    } catch (e) {
   268→      AppLogger.error('Unexpected getSubscriptionPlans error', error: e);
   269→      throw const ServerException(
   270→        message: 'Failed to fetch subscription plans.',
   271→        statusCode: 500,
   272→      );
   273→    }
   274→  }
   275→
   276→  @override
   277→  Future<SubscriptionPlanModel> getSubscriptionPlan(String planId) async {
   278→    try {
   279→      final response = await _supabase
   280→          .from(_plansTable)
   281→          .select()
   282→          .eq('id', planId)
   283→          .limit(1);
   284→
   285→      if (response.isEmpty) {
   286→        throw const NotFoundException('Subscription plan not found.');
   287→      }
   288→
   289→      return SubscriptionPlanModel.fromJson(response.first);
   290→    } on sb.PostgrestException catch (e) {
   291→      throw _mapPostgrestException(e);
   292→    } on NotFoundException {
   293→      rethrow;
   294→    } catch (e) {
   295→      AppLogger.error('Unexpected getSubscriptionPlan error', error: e);
   296→      throw const ServerException(
   297→        message: 'Failed to fetch subscription plan.',
   298→        statusCode: 500,
   299→      );
   300→    }
   301→  }

   303→  @override
   304→  Future<SubscriptionPlanModel> upsertSubscriptionPlan(
   305→    Map<String, dynamic> planData,
   306→  ) async {
   307→    try {
   308→      planData['updated_at'] = DateTime.now().toIso8601String();
   309→
   310→      final response = await _supabase
   311→          .from(_plansTable)
   312→          .upsert(planData)
   313→          .select();
   314→
   315→      if (response.isEmpty) {
   316→        throw const ServerException(
   317→          message: 'Subscription plan upsert returned no data.',
   318→          statusCode: 500,
   319→        );
   320→      }
   321→
   322→      AppLogger.info('Subscription plan upserted: ${response.first['id']}');
   323→      return SubscriptionPlanModel.fromJson(response.first);
   324→    } on sb.PostgrestException catch (e) {
   325→      throw _mapPostgrestException(e);
   326→    } on ServerException {
   327→      rethrow;
   328→    } catch (e) {
   329→      AppLogger.error('Unexpected upsertSubscriptionPlan error', error: e);
   330→      throw const ServerException(
   331→        message: 'Failed to upsert subscription plan.',

   374→  @override
   375→  Future<List<SubscriptionModel>> getSubscriptions({
   376→    String? schoolId,
   377→    String? subscriberType,
   378→    String? status,
   379→    int page = 1,
   380→    int perPage = 20,
   381→  }) async {
   382→    try {
   383→      var query = _supabase.from(_subscriptionsTable).select();
   384→
   385→      if (schoolId != null) {
   386→        query = query.eq('school_id', schoolId);
   387→      }
   388→      if (subscriberType != null) {
   389→        query = query.eq('subscriber_type', subscriberType);
   390→      }
   391→      if (status != null) {
   392→        query = query.eq('status', status);
   393→      }
   394→
   395→      final filtered = _applyFilters(query, {
   396→        if (schoolId != null) 'school_id': schoolId,
   397→        if (subscriberType != null) 'subscriber_type': subscriberType,
   398→        if (status != null) 'status': status,
   399→      });
   400→
   401→      var transformed =
   402→          filtered.order('created_at', ascending: false) as sb.PostgrestTransformBuilder<dynamic>;
   403→      transformed = _applyPagination(transformed, page: page, perPage: perPage);
   404→
   405→      final list = await transformed;
   406→      AppLogger.info('Fetched ${list.length} subscriptions');
   407→      return list
   408→          .map<SubscriptionModel>(
   409→            (row) => SubscriptionModel.fromJson(row),
   410→          )
   411→          .toList();
   412→    } on sb.PostgrestException catch (e) {
   413→      throw _mapPostgrestException(e);
   414→    } catch (e) {
   415→      AppLogger.error('Unexpected getSubscriptions error', error: e);
   416→      throw const ServerException(

  1059→  @override
  1060→  Future<CouponModel> validateCoupon({
  1061→    required String code,
  1062→    required String billingModel,
  1063→    String? planId,
  1064→  }) async {
  1065→    try {
  1066→      var query = _supabase
  1067→          .from(_couponsTable)
  1068→          .select()
  1069→          .eq('code', code)
  1070→          .eq('billing_model', billingModel)
  1071→          .eq('is_active', true) as sb.PostgrestFilterBuilder<dynamic>;
  1072→
  1073→      if (planId != null) {
  1074→        query = query.eq('plan_id', planId);
  1075→      }
  1076→
  1077→      final response = await query.limit(1);
  1078→
  1079→      if (response.isEmpty) {
  1080→        throw const NotFoundException('Invalid or expired coupon code.');
  1081→      }
  1082→
  1083→      final couponData = response.first;
  1084→
  1085→      // Validate expiry
  1086→      final validUntil = couponData['valid_until'] != null
  1087→          ? DateTime.tryParse(couponData['valid_until'] as String)
  1088→          : null;
  1089→      if (validUntil != null && validUntil.isBefore(DateTime.now())) {
  1090→        throw const ValidationException(
  1091→          message: 'This coupon has expired.',
  1092→          fieldErrors: {'code': 'Coupon has expired.'},
  1093→        );
  1094→      }
  1095→
  1096→      final validFrom = couponData['valid_from'] != null
  1097→          ? DateTime.tryParse(couponData['valid_from'] as String)
  1098→          : null;
  1099→      if (validFrom != null && validFrom.isAfter(DateTime.now())) {
  1100→        throw const ValidationException(
  1101→          message: 'This coupon is not yet valid.',
  1102→          fieldErrors: {'code': 'Coupon is not yet valid.'},
  1103→        );
  1104→      }
  1105→
  1106→      // Validate redemption count
  1107→      final maxRedemptions = couponData['max_redemptions'] as int?;
  1108→      final currentRedemptions =
  1109→          couponData['current_redemptions'] as int? ?? 0;
  1110→      if (maxRedemptions != null &&
  1111→          currentRedemptions >= maxRedemptions) {
  1112→        throw const ValidationException(
  1113→          message: 'This coupon has reached its maximum redemptions.',
  1114→          fieldErrors: {'code': 'Coupon redemption limit reached.'},
  1115→        );
  1116→      }
  1117→
  1118→      AppLogger.info('Coupon validated: $code');
  1119→      return CouponModel.fromJson(couponData);
  1120→    } on sb.PostgrestException catch (e) {
  1121→      throw _mapPostgrestException(e);
  1122→    } on NotFoundException {
  1123→      rethrow;
  1124→    } on ValidationException {
  1125→      rethrow;
  1126→    } catch (e) {
  1127→      AppLogger.error('Unexpected validateCoupon error', error: e);
  1128→      throw const ServerException(
  1129→        message: 'Failed to validate coupon.',
  1130→        statusCode: 500,
  1131→      );
  1132→    }
  1133→  }

  1654→

  1872→  sb.PostgrestTransformBuilder<dynamic> _applyPagination(
  1873→    sb.PostgrestTransformBuilder<dynamic> query, {
  1874→    int? page,
  1875→    int? perPage,
  1876→  }) {
  1877→    final p = page ?? 1;
  1878→    final pp = perPage ?? 20;
  1879→    final from = (p - 1) * pp;
  1880→    final to = from + pp - 1;
  1881→    return query.range(from, to);
  1882→  }

  1884→  /// Maps a Supabase [sb.PostgrestException] to a domain exception.

  1887→    final message = e.message ?? 'An unexpected database error occurred.';

  1889→    AppLogger.warning(
  1890→      'Supabase PostgrestException — code: ${e.code}, message: $message',

  1892→
  1893→    switch (statusCode) {
  1894→      case 401:
  1895→        return UnauthorizedException(message);
  1896→      case 403:
  1897→        return ForbiddenException(message);
  1898→      case 404:
  1899→        return NotFoundException(message);
  1900→      case 422:
  1901→        return ValidationException(
  1902→          message: message,
  1903→          fieldErrors: e.details is Map<String, dynamic>
  1904→              ? e.details as Map<String, String>

  1906→        );
  1907→      default:
  1908→        return ServerException(
  1909→          message: message,
  1910→          statusCode: statusCode,
  1911→          data: e.details,

  1913→    }
