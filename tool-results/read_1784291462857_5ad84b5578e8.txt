    14→///
    15→/// Responsibilities:
    16→/// - Converts domain entities to/from data models

    18→/// - Returns [Result] to force callers to handle both success and failure

    29→  // ═══════════════════════════════════════════════════════════════════════
    30→  // Helper: Safe call with exception → Failure mapping
    31→  // ═══════════════════════════════════════════════════════════════════════
    32→
    33→  /// Executes [call] and wraps the result in [Result<T>].
    34→  ///
    35→  /// Catches all known exception types and maps them to the appropriate
    36→  /// [Failure] variant. Unknown exceptions become [ServerFailure] with
    37→  /// status code 500.
    38→  Future<Result<T>> _safeCall<T>(Future<T> Function() call) async {
    39→    try {
    40→      final result = await call();
    41→      return Success(result);
    42→    } on ServerException catch (e) {
    43→      return FailureResult(Failure.server(
    44→        message: e.message,
    45→        statusCode: e.statusCode,
    46→        data: e.data,
    47→      ));
    48→    } on AuthException catch (e) {
    49→      return FailureResult(Failure.auth(message: e.message, code: e.code));
    50→    } on NetworkException catch (e) {
    51→      return FailureResult(Failure.network(message: e.message));
    52→    } on NotFoundException catch (e) {
    53→      return FailureResult(Failure.notFound(message: e.message));
    54→    } on UnauthorizedException catch (e) {
    55→      return FailureResult(Failure.unauthorized(message: e.message));
    56→    } on ForbiddenException catch (e) {
    57→      return FailureResult(Failure.forbidden(message: e.message));
    58→    } on ValidationException catch (e) {
    59→      return FailureResult(Failure.validation(
    60→        message: e.message,
    61→        fieldErrors: e.fieldErrors,
    62→      ));
    63→    } on CacheException catch (e) {
    64→      return FailureResult(Failure.cache(message: e.message));
    65→    } catch (e) {
    66→      AppLogger.error('Unexpected exception in StudentPortalRepositoryImpl',
    67→          error: e);
    68→      return FailureResult(Failure.server(
    69→        message: e.toString(),
    70→        statusCode: 500,
    71→      ));
    72→    }
    73→  }

   423→  @override
   424→  Future<Result<DocumentChatEntity>> uploadDocument({
   425→    required String studentId,
   426→    String? schoolId,
   427→    required String fileName,
   428→    required String fileUrl,
   429→    required String fileFormat,
   430→    int? fileSize,
   431→  }) =>
   432→      _safeCall(() async {
   433→        // The repository receives a pre-uploaded file URL, so we create
   434→        // the document_chats record directly and trigger processing.
   435→        final response = await _supabaseClient
   436→            .from('document_chats')
   437→            .insert({
   438→              'student_id': studentId,
   439→              'school_id': schoolId,
   440→              'file_name': fileName,
   441→              'file_url': fileUrl,
   442→              'file_size': fileSize,
   443→              'file_format': fileFormat,
   444→              'status': 'processing',
   445→            })
   446→            .select()
   447→            .single();
   448→
   449→        final documentId = response['id'] as String;
   450→
   451→        // Trigger the process-document edge function asynchronously
   452→        _supabaseClient.functions
   453→            .invoke(
   454→          'process-document',
   455→          body: {
   456→            'document_id': documentId,
   457→            'file_url': fileUrl,
   458→            'file_format': fileFormat,
   459→          },
   460→        )
   461→            .catchError((error) {
   462→          AppLogger.error('Process document edge function failed',
   463→              error: error);
   464→        });
   465→
   466→        return DocumentChatModel.fromJson(response).toEntity();
   467→      });
