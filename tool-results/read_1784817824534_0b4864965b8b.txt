   184→  @override
   185→  Future<QuestionModel> createQuestion(
   186→    Map<String, dynamic> questionData,
   187→  ) async {
   188→    try {
   189→      final response =
   190→          await _supabase.from(_questionsTable).insert(questionData).select();
   191→
   192→      if (response.isEmpty) {
   193→        throw const ServerException(
   194→          message: 'Question creation returned no data.',
   195→          statusCode: 500,
   196→        );
   197→      }
   198→
   199→      AppLogger.info('Question created: ${response.first['id']}');
   200→      return QuestionModel.fromJson(response.first);
   201→    } on sb.PostgrestException catch (e) {
   202→      throw _mapPostgrestException(e);
   203→    } on ServerException {
   204→      rethrow;
   205→    } catch (e) {
   206→      AppLogger.error('Unexpected createQuestion error', error: e);
   207→      throw const ServerException(
   208→        message: 'Failed to create question.',
   209→        statusCode: 500,
   210→      );
   211→    }

   294→  @override
   295→  Future<Map<String, dynamic>> getQuestionWithDetails(
   296→    String questionId,
   297→  ) async {
   298→    try {
   299→      // Fetch the base question
   300→      final questionResponse = await _supabase
   301→          .from(_questionsTable)
   302→          .select()
   303→          .eq('id', questionId)
   304→          .limit(1);
   305→
   306→      if (questionResponse.isEmpty) {
   307→        throw const NotFoundException(message: 'Question not found.');
   308→      }
   309→
   310→      final questionJson = Map<String, dynamic>.from(questionResponse.first);
   311→
   312→      // Fetch related data in parallel
   313→      final results = await Future.wait([
   314→        _supabase
   315→            .from(_answerOptionsTable)
   316→            .select()
   317→            .eq('question_id', questionId)
   318→            .order('sort_order'),
   319→        _supabase
   320→            .from(_matchingPairsTable)
   321→            .select()
   322→            .eq('question_id', questionId)
   323→            .order('sort_order'),
   324→        _supabase
   325→            .from(_orderingItemsTable)
   326→            .select()
   327→            .eq('question_id', questionId)
   328→            .order('correct_position'),
   329→        _supabase
   330→            .from(_fillInBlankTable)
   331→            .select()
   332→            .eq('question_id', questionId)
   333→            .order('blank_index'),
   334→        _supabase
   335→            .from(_attachmentsTable)
   336→            .select()
   337→            .eq('question_id', questionId)
   338→            .order('sort_order'),
   339→        // Tags need a join through the relation table
   340→        _supabase
   341→            .from(_tagRelationsTable)
   342→            .select('question_tags(*)')
   343→            .eq('question_id', questionId),
   344→      ]);
   345→
   346→      questionJson['answer_options'] = results[0] as List;
   347→      questionJson['matching_pairs'] = results[1] as List;
   348→      questionJson['ordering_items'] = results[2] as List;
   349→      questionJson['fill_in_blank_answers'] = results[3] as List;
   350→      questionJson['question_attachments'] = results[4] as List;
   351→
   352→      // Flatten tag join results
   353→      final tagRows = results[5] as List;
   354→      final tagList = tagRows
   355→          .map((row) => row['question_tags'] as Map<String, dynamic>?)
   356→          .whereType<Map<String, dynamic>>()
   357→          .toList();
   358→      questionJson['question_tags'] = tagList;
   359→
   360→      return questionJson;
   361→    } on sb.PostgrestException catch (e) {
   362→      throw _mapPostgrestException(e);
   363→    } on NotFoundException {
   364→      rethrow;
   365→    } catch (e) {
   366→      AppLogger.error('Unexpected getQuestionWithDetails error', error: e);
   367→      throw const ServerException(
   368→        message: 'Failed to retrieve question details.',
   369→        statusCode: 500,
   370→      );
   371→    }
   372→  }

   376→    Map<String, dynamic> filters,
   377→  ) async {

   381→      transformQuery = _applyPagination(
   382→        transformQuery,
   383→        page: filters['page'] as int?,
   384→        perPage: filters['per_page'] as int?,
   385→      );
   386→
   387→      final response = await transformQuery;
   388→      return response
   389→          .map((json) => QuestionModel.fromJson(json as Map<String, dynamic>))
   390→          .toList();
   391→    } on sb.PostgrestException catch (e) {
   392→      throw _mapPostgrestException(e);
   393→    } catch (e) {

   498→  @override
   499→  Future<QuestionModel> duplicateQuestion(String questionId) async {
   500→    try {
   501→      // Fetch the original question
   502→      final original = await getQuestion(questionId);
   503→
   504→      // Prepare the duplicate data (strip ID, reset version)
   505→      final duplicateData = original.toJson()
   506→        ..remove('id')
   507→        ..remove('created_at')
   508→        ..remove('updated_at')
   509→        ..['version'] = 1
   510→        ..['parent_id'] = original.id
   511→        ..['is_published'] = false
   512→        ..['is_archived'] = false
   513→        ..['usage_count'] = 0
   514→        ..['avg_score'] = 0.0;
   515→
   516→      return createQuestion(duplicateData);
   517→    } on sb.PostgrestException catch (e) {
   518→      throw _mapPostgrestException(e);
   519→    } on ServerException {
   520→      rethrow;
   521→    } on NotFoundException {
   522→      rethrow;
   523→    } catch (e) {
   524→      AppLogger.error('Unexpected duplicateQuestion error', error: e);
   525→      throw const ServerException(
   526→        message: 'Failed to duplicate question.',
   527→        statusCode: 500,
   528→      );
   529→    }

   551→            .from(_questionsTable)

   554→      }

   598→  @override
   599→  Future<List<AnswerOptionModel>> updateAnswerOptions(
   600→    String questionId,
   601→    List<Map<String, dynamic>> options,
   602→  ) async {
   603→    try {
   604→      // Delete existing options
   605→      await _supabase
   606→          .from(_answerOptionsTable)
   607→          .delete()
   608→          .eq('question_id', questionId);
   609→
   610→      // Insert new options
   611→      if (options.isNotEmpty) {
   612→        final optionsData = options
   613→            .map((o) => {
   614→                  ...o,
   615→                  'question_id': questionId,
   616→                },)
   617→            .toList();
   618→
   619→        await _supabase
   620→            .from(_answerOptionsTable)
   621→            .insert(optionsData);
   622→      }
   623→
   624→      // Fetch the newly inserted options
   625→      return getAnswerOptions(questionId);
   626→    } on sb.PostgrestException catch (e) {
   627→      throw _mapPostgrestException(e);
   628→    } catch (e) {
   629→      AppLogger.error('Unexpected updateAnswerOptions error', error: e);
   630→      throw const ServerException(
   631→        message: 'Failed to update answer options.',
   632→        statusCode: 500,
   633→      );
   634→    }
   635→  }

   787→        });

   794→    } catch (e) {
   795→      AppLogger.error('Unexpected toggleFavorite error', error: e);

   797→        message: 'Failed to toggle favorite.',
   798→        statusCode: 500,
   799→      );

  1329→  @override
  1330→  Future<QuestionBankStatsModel> getStats({String? schoolId}) async {
  1331→    try {
  1332→      // Base query builder
  1333→      var baseQuery = _supabase.from(_questionsTable).select();
  1334→
  1335→      if (schoolId != null) {
  1336→        baseQuery = baseQuery.eq('school_id', schoolId);
  1337→      }
  1338→
  1339→      // Fetch all questions for stats computation
  1340→      final allQuestions = await baseQuery;
  1341→
  1342→      final totalQuestions = allQuestions.length;
  1343→      final publishedQuestions = allQuestions
  1344→          .where((q) => q['is_published'] == true)
  1345→          .length;
  1346→      final draftQuestions = allQuestions
  1347→          .where((q) => q['is_published'] == false && q['is_archived'] == false)
  1348→          .length;
  1349→      final archivedQuestions = allQuestions
  1350→          .where((q) => q['is_archived'] == true)
  1351→          .length;
  1352→
  1353→      // Compute breakdowns
  1354→      final Map<String, int> bySubject = {};
  1355→      final Map<String, int> byDifficulty = {};
  1356→      final Map<String, int> byType = {};
  1357→      final Map<String, int> byExamType = {};
  1358→
  1359→      for (final q in allQuestions) {
  1360→        final subjectId = q['subject_id'] as String?;
  1361→        if (subjectId != null) {
  1362→          bySubject[subjectId] = (bySubject[subjectId] ?? 0) + 1;
  1363→        }
  1364→
  1365→        final diff = q['difficulty'] as String?;
  1366→        if (diff != null) {
  1367→          byDifficulty[diff] = (byDifficulty[diff] ?? 0) + 1;
  1368→        }
  1369→
  1370→        final type = q['question_type'] as String?;
  1371→        if (type != null) {
  1372→          byType[type] = (byType[type] ?? 0) + 1;
  1373→        }
  1374→
  1375→        final exam = q['exam_type'] as String?;
  1376→        if (exam != null) {
  1377→          byExamType[exam] = (byExamType[exam] ?? 0) + 1;
  1378→        }
  1379→      }
  1380→
  1381→      // Recent questions (last 7 days)
  1382→      final weekAgo = DateTime.now().subtract(const Duration(days: 7));
  1383→      final recentQuestions = allQuestions
  1384→          .where((q) {
  1385→            final createdAt = q['created_at'] as String?;
  1386→            if (createdAt == null) return false;
  1387→            return DateTime.parse(createdAt).isAfter(weekAgo);
  1388→          })
  1389→          .length;
  1390→
  1391→      // Collection count
  1392→      var collectionQuery =
  1393→          _supabase.from(_collectionsTable).select('id');
  1394→      if (schoolId != null) {
  1395→        collectionQuery = collectionQuery.eq('school_id', schoolId);
  1396→      }
  1397→      final collections = await collectionQuery;
  1398→
  1399→      // Favorites count for current user
  1400→      final userId = _currentUserId;
  1401→      int totalFavorites = 0;
  1402→      if (userId != null) {
  1403→        final favorites = await _supabase
  1404→            .from(_favoritesTable)
  1405→            .select('id')
  1406→            .eq('user_id', userId);
  1407→        totalFavorites = favorites.length;
  1408→      }
  1409→
  1410→      // Most used questions (top 5)
  1411→      final sortedByUsage = List<Map<String, dynamic>>.from(allQuestions)
  1412→        ..sort((a, b) => (b['usage_count'] as int? ?? 0)
  1413→            .compareTo(a['usage_count'] as int? ?? 0),);
  1414→      final mostUsed = sortedByUsage
  1415→          .take(5)
  1416→          .map((q) => QuestionModel.fromJson(q))
  1417→          .toList();
  1418→
  1419→      return QuestionBankStatsModel(
  1420→        totalQuestions: totalQuestions,
  1421→        publishedQuestions: publishedQuestions,
  1422→        draftQuestions: draftQuestions,
  1423→        archivedQuestions: archivedQuestions,
  1424→        questionsBySubject: bySubject,
  1425→        questionsByDifficulty: byDifficulty,
  1426→        questionsByType: byType,
  1427→        questionsByExamType: byExamType,
  1428→        recentQuestions: recentQuestions,
  1429→        totalCollections: collections.length,
  1430→        totalFavorites: totalFavorites,
  1431→        mostUsedQuestions: mostUsed,
  1432→      );
  1433→    } on sb.PostgrestException catch (e) {
  1434→      throw _mapPostgrestException(e);
  1435→    } catch (e) {
  1436→      AppLogger.error('Unexpected getStats error', error: e);
  1437→      throw const ServerException(
  1438→        message: 'Failed to retrieve question bank stats.',
  1439→        statusCode: 500,
  1440→      );
  1441→    }
  1442→  }

  1617→
  1618→      if (snapshot == null) {

  1620→          message: 'Version snapshot is empty.',
  1621→          statusCode: 500,

  1624→
  1625→      // Update the question with the snapshot data
  1626→      final restoredData = Map<String, dynamic>.from(snapshot)
  1627→        ..remove('id')
  1628→        ..remove('created_at')
  1629→        ..['updated_at'] = DateTime.now().toIso8601String()
  1630→        ..['version'] = (await getQuestion(questionId)).version + 1;
  1631→
  1632→      return updateQuestion(questionId, restoredData);

  1711→  /// Applies sorting based on the [sortBy] value.
  1712→  sb.PostgrestTransformBuilder<dynamic> _applySorting(
  1713→    sb.PostgrestFilterBuilder<dynamic> query,
  1714→    String? sortBy,
  1715→  ) {
  1716→    switch (sortBy) {
  1717→      case 'oldest':
  1718→        return query.order('created_at', ascending: true);
  1719→      case 'most_used':
  1720→        return query.order('usage_count', ascending: false);
  1721→      case 'least_used':
  1722→        return query.order('usage_count', ascending: true);
  1723→      case 'highest_rated':
  1724→        return query.order('avg_score', ascending: false);
  1725→      case 'a_z':
  1726→        return query.order('content', ascending: true);
  1727→      case 'z_a':
  1728→        return query.order('content', ascending: false);
  1729→      case 'newest':
  1730→      default:
  1731→        return query.order('created_at', ascending: false);
  1732→    }
  1733→  }

  1735→  /// Applies pagination using Supabase range.
  1736→  sb.PostgrestTransformBuilder<dynamic> _applyPagination(
  1737→    sb.PostgrestTransformBuilder<dynamic> query, {
  1738→    int? page,
  1739→    int? perPage,
  1740→  }) {
  1741→    final p = page ?? 1;
  1742→    final pp = perPage ?? 20;
  1743→    final from = (p - 1) * pp;
  1744→    final to = from + pp - 1;
  1745→    return query.range(from, to);
  1746→  }

  1748→  /// Maps a Supabase [sb.PostgrestException] to a domain exception.
  1749→  Exception _mapPostgrestException(sb.PostgrestException e) {
  1750→    final statusCode = e.code != null ? int.tryParse(e.code!) ?? 0 : 0;
  1751→    final message = e.message ?? 'An unexpected database error occurred.';
  1752→
  1753→    AppLogger.warning(
  1754→      'Supabase PostgrestException — code: ${e.code}, message: $message',
  1755→    );
  1756→
  1757→    switch (statusCode) {
  1758→      case 401:
  1759→        return UnauthorizedException(message: message);
  1760→      case 403:
  1761→        return ForbiddenException(message: message);
  1762→      case 404:
  1763→        return NotFoundException(message: message);
  1764→      case 422:
  1765→        return ValidationException(
  1766→          message: message,
  1767→          fieldErrors: e.details is Map<String, dynamic>
  1768→              ? e.details as Map<String, String>
  1769→              : {},
  1770→        );
  1771→      default:
  1772→        return ServerException(
  1773→          message: message,
  1774→          statusCode: statusCode,
  1775→          data: e.details,
  1776→        );
  1777→    }
  1778→  }
