    25→/// AI-powered Scheme of Work Generator page.
    26→///
    27→/// Teachers specify subject, class, curriculum, duration, and term
    28→/// parameters and the AI generates a complete scheme of work with
    29→/// weekly/monthly breakdowns including topics, objectives, activities,
    30→/// resources, and assessments.
    31→class SchemeOfWorkGeneratorPage extends ConsumerStatefulWidget {
    32→  const SchemeOfWorkGeneratorPage({super.key});
    33→
    34→  @override
    35→  ConsumerState<SchemeOfWorkGeneratorPage> createState() =>
    36→      _SchemeOfWorkGeneratorPageState();
    37→}

    39→class _SchemeOfWorkGeneratorPageState
    40→    extends ConsumerState<SchemeOfWorkGeneratorPage> {
    41→  final _formKey = GlobalKey<FormState>();
    42→  final _subjectController = TextEditingController();
    43→  final _classNameController = TextEditingController();
    44→  final _termController = TextEditingController();
    45→
    46→  CurriculumType _curriculum = CurriculumType.nigerian;
    47→  PlanDuration _durationType = PlanDuration.term;
    48→  DateTime? _startDate;
    49→  DateTime? _endDate;
    50→  bool _showResult = false;

    52→  @override
    53→  void dispose() {
    54→    _subjectController.dispose();
    55→    _classNameController.dispose();
    56→    _termController.dispose();
    57→    super.dispose();
    58→  }
    59→
    60→  // ─── Form Validation ──────────────────────────────────────────────
    61→
    62→  bool get _isFormValid {
    63→    return _subjectController.text.trim().isNotEmpty;
    64→  }
    65→
    66→  // ─── Generate Scheme ──────────────────────────────────────────────
    67→
    68→  void _generateScheme() {
    69→    if (!_formKey.currentState!.validate()) return;
    70→
    71→    ref.read(schemeOfWorkProvider.notifier).generateScheme(
    72→          GenerateSchemeOfWorkParams(
    73→            subject: _subjectController.text.trim(),
    74→            className: _classNameController.text.trim(),
    75→            curriculum: _curriculum,
    76→            durationType: _durationType,
    77→            term: _termController.text.trim().isNotEmpty
    78→                ? _termController.text.trim()
    79→                : null,
    80→          ),
    81→        );
    82→    setState(() => _showResult = true);
    83→  }
    84→
    85→  // ─── Save Scheme ──────────────────────────────────────────────────
    86→
    87→  void _saveScheme() {
    88→    final scheme = ref.read(schemeOfWorkProvider).currentScheme;
    89→    if (scheme == null) return;
    90→
    91→    ref.read(schemeOfWorkProvider.notifier).createScheme(
    92→          CreateSchemeOfWorkParams(scheme: scheme),
    93→        );
    94→  }
    95→
    96→  // ─── Date Picker ──────────────────────────────────────────────────
    97→
    98→  Future<void> _pickStartDate() async {
    99→    final picked = await showDatePicker(
   100→      context: context,
   101→      initialDate: _startDate ?? DateTime.now(),
   102→      firstDate: DateTime(2020),
   103→      lastDate: DateTime(2100),
   104→    );
   105→    if (picked != null) {
   106→      setState(() => _startDate = picked);
   107→    }
   108→  }
   109→
   110→  Future<void> _pickEndDate() async {
   111→    final picked = await showDatePicker(
   112→      context: context,
   113→      initialDate: _endDate ?? _startDate ?? DateTime.now(),
   114→      firstDate: _startDate ?? DateTime(2020),
   115→      lastDate: DateTime(2100),
   116→    );
   117→    if (picked != null) {
   118→      setState(() => _endDate = picked);
   119→    }
   120→  }
   121→
   122→  String _formatDate(DateTime? date) {
   123→    if (date == null) return 'Select date';
   124→    const months = [
   125→      '', 'Jan', 'Feb', 'Mar', 'Apr', 'May', 'Jun',
   126→      'Jul', 'Aug', 'Sep', 'Oct', 'Nov', 'Dec',
   127→    ];
   128→    return '${months[date.month]} ${date.day.toString().padLeft(2, '0')}, ${date.year}';
   129→  }

   132→

   136→    final tt = context.textTheme;
   137→    final state = ref.watch(schemeOfWorkProvider);
   138→
   139→    ref.listen<SchemeOfWorkState>(schemeOfWorkProvider, (prev, next) {
   140→      if (next.error != null && prev?.error != next.error) {
   141→        ScaffoldMessenger.of(context).showSnackBar(
   142→          SnackBar(
   143→            content: Text(next.error!),
   144→            backgroundColor: cs.error,
   145→          ),
   146→        );
   147→        ref.read(schemeOfWorkProvider.notifier).clearError();
   148→      }
   149→      if (next.successMessage != null &&
   150→          prev?.successMessage != next.successMessage) {
   151→        ScaffoldMessenger.of(context).showSnackBar(
   152→          SnackBar(
   153→            content: Text(next.successMessage!),
   154→            backgroundColor: cs.primary,
   155→          ),
   156→        );
   157→        ref.read(schemeOfWorkProvider.notifier).clearSuccessMessage();
   158→      }
   159→    });
   160→
   161→    return Scaffold(
   162→      appBar: AppAppBar(
   163→        title: 'AI Scheme of Work',
   164→      ),
   165→      body: state.isGenerating
   166→          ? Center(
   167→              child: Column(
   168→                mainAxisSize: MainAxisSize.min,
   169→                children: [
   170→                  const AppLoadingSpinner(size: AppLoadingSpinnerSize.large),
   171→                  const SizedBox(height: Spacings.lg),
   172→                  Text(
   173→                    'Generating scheme of work...',
   174→                    style: tt.bodyLarge?.copyWith(
   175→                      color: cs.onSurfaceVariant,
   176→                    ),
   177→                  ),
   178→                  const SizedBox(height: Spacings.sm),
   179→                  Text(
   180→                    'This may take a moment',
   181→                    style: tt.bodySmall?.copyWith(
   182→                      color: cs.onSurfaceVariant,
   183→                    ),
   184→                  ),
   185→                ],
   186→              ),
   187→            )
   188→          : SingleChildScrollView(
   189→              padding: Spacings.paddingScreen,
   190→              child: Column(
   191→                crossAxisAlignment: CrossAxisAlignment.stretch,
   192→                children: [
   193→                  // ── Input Form ─────────────────────────────────────
   194→                  if (!_showResult || state.currentScheme == null) ...[
   195→                    _buildFormSection(cs, tt, state),
   196→                  ],
   197→
   198→                  // ── Generated Result ──────────────────────────────
   199→                  if (_showResult && state.currentScheme != null) ...[
   200→                    _buildResultHeader(cs, tt, state.currentScheme!),
   201→                    const SizedBox(height: Spacings.lg),
   202→                    _buildBreakdownList(cs, tt, state.currentScheme!),
   203→                    const SizedBox(height: Spacings.xl),
   204→                    _buildActionButtons(cs, state),
   205→                  ],
   206→                ],
   207→              ),
   208→            ),
   209→    );
   210→  }

   500→      children: [
   501→        Text(
   502→          '${scheme.durationType.label} Breakdown',
   503→          style: tt.titleMedium?.copyWith(

   505→            color: cs.onSurface,

   513→            padding: const EdgeInsets.only(bottom: Spacings.md),
   514→            child: _buildBreakdownCard(cs, tt, index + 1, plan),
   515→          );
   516→        }),
   517→      ],
   518→    );

   649→  Widget _buildSubSection(ColorScheme cs, TextTheme tt, IconData icon,
