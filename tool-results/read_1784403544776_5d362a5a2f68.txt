   155→  String _formatDateTime(DateTime dt) {
   156→    const months = [
   157→      '',
   158→      'Jan',

   160→      'Mar',
   161→      'Apr',

   165→      'Aug',
   166→      'Sep',

   168→      'Nov',

   173→  }

   175→  String _formatMonthYear(DateTime d) {
   176→    const months = [
   177→      '',
   178→      'January',
   179→      'February',
   180→      'March',
   181→      'April',
   182→      'May',
   183→      'June',
   184→      'July',
   185→      'August',
   186→      'September',
   187→      'October',
   188→      'November',
   189→      'December',
   190→    ];

   192→  }

   194→  String _formatWeekRange(DateTime focused) {
   195→    // Week starts on Monday
   196→    final weekday = focused.weekday;
   197→    final monday = focused.subtract(Duration(days: weekday - 1));
   198→    final sunday = monday.add(const Duration(days: 6));
   199→    const months = [
   200→      '',
   201→      'Jan',
   202→      'Feb',
   203→      'Mar',
   204→      'Apr',
   205→      'May',
   206→      'Jun',
   207→      'Jul',
   208→      'Aug',
   209→      'Sep',
   210→      'Oct',
   211→      'Nov',
   212→      'Dec',
   213→    ];
   214→    return '${months[monday.month]} ${monday.day} – ${months[sunday.month]} ${sunday.day}, ${sunday.year}';
   215→  }

   217→  String _colorToHex(Color c) =>
   218→      '#${(c.value & 0xFFFFFF).toRadixString(16).padLeft(6, '0')}';

   220→  Color _hexToColor(String? hex) {
   221→    if (hex == null || hex.isEmpty) return _kPresetColors.first;
   222→    final code = hex.replaceFirst('#', '');
   223→    if (code.length != 6) return _kPresetColors.first;
   224→    return Color(int.parse('FF$code', radix: 16));
   225→  }

   279→  void _goToPrevious() {
   280→    setState(() {
   281→      switch (_viewMode) {
   282→        case CalendarViewMode.monthly:
   283→          _focusedDate = DateTime(_focusedDate.year, _focusedDate.month - 1);
   284→        case CalendarViewMode.weekly:
   285→          _focusedDate = _focusedDate.subtract(const Duration(days: 7));
   286→        case CalendarViewMode.daily:
   287→          _focusedDate = _focusedDate.subtract(const Duration(days: 1));
   288→      }
   289→      _selectedDate = _normalizeDate(_focusedDate);
   290→    });
   291→    ref.read(calendarPlannerProvider.notifier).selectDate(_selectedDate);
   292→  }

   294→  void _goToNext() {
   295→    setState(() {
   296→      switch (_viewMode) {
   297→        case CalendarViewMode.monthly:
   298→          _focusedDate = DateTime(_focusedDate.year, _focusedDate.month + 1);
   299→        case CalendarViewMode.weekly:
   300→          _focusedDate = _focusedDate.add(const Duration(days: 7));
   301→        case CalendarViewMode.daily:
   302→          _focusedDate = _focusedDate.add(const Duration(days: 1));
   303→      }
   304→      _selectedDate = _normalizeDate(_focusedDate);
   305→    });
   306→    ref.read(calendarPlannerProvider.notifier).selectDate(_selectedDate);
   307→  }

   350→  void _resetForm() {
   351→    _titleCtrl.clear();
   352→    _subjectCtrl.clear();
   353→    _locationCtrl.clear();
   354→    _eventType = EventType.class_;
   355→    _startTime = TimeOfDay.now();
   356→    _endTime = TimeOfDay(
   357→      hour: (TimeOfDay.now().hour + 1) % 24,
   358→      minute: 0,
   359→    );
   360→    _isAllDay = false;
   361→    _selectedColor = _kPresetColors.first;
   362→    _reminder = ReminderOption.none;

   365→  void _handleSaveEvent() {
   366→    if (_titleCtrl.text.trim().isEmpty) {
   367→      _showSnackBar('Event title is required', isError: true);
   368→      return;
   369→    }
   370→
   371→    final now = DateTime.now();
   372→    final eventDate = _selectedDate;
   373→    final startDateTime = DateTime(
   374→      eventDate.year,
   375→      eventDate.month,
   376→      eventDate.day,
   377→      _startTime.hour,
   378→      _startTime.minute,
   379→    );
   380→    var endDateTime = DateTime(
   381→      eventDate.year,
   382→      eventDate.month,
   383→      eventDate.day,
   384→      _endTime.hour,
   385→      _endTime.minute,
   386→    );
   387→    if (!endDateTime.isAfter(startDateTime)) {
   388→      endDateTime = startDateTime.add(const Duration(hours: 1));
   389→    }
   390→
   391→    final event = CalendarEventEntity(
   392→      id: _editingEvent?.id ?? '',
   393→      teacherId: _editingEvent?.teacherId ?? '',
   394→      title: _titleCtrl.text.trim(),
   395→      description: _editingEvent?.description,
   396→      eventType: _eventType,
   397→      subject: _subjectCtrl.text.trim().isEmpty ? null : _subjectCtrl.text.trim(),
   398→      subjectId: _editingEvent?.subjectId,
   399→      classId: _editingEvent?.classId,
   400→      location: _locationCtrl.text.trim().isEmpty ? null : _locationCtrl.text.trim(),
   401→      startTime: startDateTime,
   402→      endTime: endDateTime,
   403→      isAllDay: _isAllDay,
   404→      isRecurring: _editingEvent?.isRecurring ?? false,
   405→      recurrenceRule: _editingEvent?.recurrenceRule,
   406→      color: _colorToHex(_selectedColor),
   407→      reminderMinutesBefore: _reminder.minutes,
   408→      relatedResourceType: _editingEvent?.relatedResourceType,
   409→      relatedResourceId: _editingEvent?.relatedResourceId,
   410→      metadata: _editingEvent?.metadata ?? {},
   411→      createdAt: _editingEvent?.createdAt ?? now,
   412→      updatedAt: now,
   413→    );
   414→
   415→    Navigator.pop(context); // Close bottom sheet first
   416→
   417→    if (_editingEvent != null) {
   418→      ref.read(calendarPlannerProvider.notifier).updateEvent(event);
   419→    } else {
   420→      ref.read(calendarPlannerProvider.notifier).createEvent(
   421→            CreateEventParams(event: event),
   422→          );
   423→    }
   424→    _listenForMessages();
   425→  }

   427→  void _handleDeleteEvent(String eventId) {
   428→    AppDialog.showConfirm(

   430→      title: 'Delete Event',
   431→      message: 'Are you sure you want to delete this event? This action cannot be undone.',

   438→      }
   439→    });
   440→  }

   657→  Widget _buildMonthlyView(CalendarPlannerState state) {
   658→    final cs = context.colorScheme;
   659→    final tt = context.textTheme;
   660→
   661→    final firstDayOfMonth = DateTime(_focusedDate.year, _focusedDate.month, 1);
   662→    final daysInMonth =
   663→        DateTime(_focusedDate.year, _focusedDate.month + 1, 0).day;
   664→    final startWeekday = firstDayOfMonth.weekday % 7; // Sun = 0
   665→
   666→    // Build a set of dates that have events
   667→    final eventDates = <DateTime>{};
   668→    for (final e in state.events) {
   669→      eventDates.add(_normalizeDate(e.startTime));
   670→    }
   671→
   672→    // Build grid cells
   673→    final cells = <Widget>[];
   674→
   675→    // Day-of-week headers
   676→    const dayLabels = ['Sun', 'Mon', 'Tue', 'Wed', 'Thu', 'Fri', 'Sat'];
   677→    for (final label in dayLabels) {
   678→      cells.add(
   679→        Center(
   680→          child: Text(
   681→            label,
   682→            style: tt.labelSmall?.copyWith(
   683→              color: cs.onSurfaceVariant,
   684→              fontWeight: AppTypography.wSemiBold,
   685→            ),
   686→          ),
   687→        ),
   688→      );
   689→    }
   690→
   691→    // Empty cells before the first day
   692→    for (var i = 0; i < startWeekday; i++) {
   693→      cells.add(const SizedBox.shrink());
   694→    }
   695→
   696→    // Day cells
   697→    final today = _normalizeDate(DateTime.now());
   698→    for (var day = 1; day <= daysInMonth; day++) {
   699→      final date = DateTime(_focusedDate.year, _focusedDate.month, day);
   700→      final isToday = date == today;
   701→      final isSelected = date == _selectedDate;
   702→      final hasEvents = eventDates.contains(date);
   703→      final dayEvents = state.events
   704→          .where((e) => _normalizeDate(e.startTime) == date)
   705→          .toList();
   706→      final dotColors = dayEvents
   707→          .take(3)
   708→          .map((e) => _hexToColor(e.color))
   709→          .toList();
   710→
   711→      cells.add(
   712→        GestureDetector(
   713→          onTap: () => _selectDate(date),
   714→          child: Container(
   715→            margin: const EdgeInsets.all(1),
   716→            decoration: BoxDecoration(
   717→              color: isSelected
   718→                  ? cs.primary
   719→                  : isToday
   720→                      ? cs.primary.withValues(alpha: 0.08)
   721→                      : Colors.transparent,
   722→              borderRadius: BorderRadius.circular(Spacings.smRadius),
   723→              border: isToday && !isSelected
   724→                  ? Border.all(color: cs.primary, width: 1.5)
   725→                  : null,
   726→            ),
   727→            child: Column(
   728→              mainAxisAlignment: MainAxisAlignment.center,
   729→              children: [
   730→                Text(
   731→                  '$day',
   732→                  style: tt.bodySmall?.copyWith(
   733→                    color: isSelected
   734→                        ? cs.onPrimary
   735→                        : isToday
   736→                            ? cs.primary
   737→                            : cs.onSurface,
   738→                    fontWeight: isToday || isSelected
   739→                        ? AppTypography.wBold
   740→                        : AppTypography.wRegular,
   741→                  ),
   742→                ),
   743→                if (hasEvents)
   744→                  Padding(
   745→                    padding: const EdgeInsets.only(top: 2),
   746→                    child: Row(
   747→                      mainAxisAlignment: MainAxisAlignment.center,
   748→                      mainAxisSize: MainAxisSize.min,
   749→                      children: List.generate(
   750→                        dotColors.length.clamp(0, 3),
   751→                        (i) => Container(
   752→                          width: 4,
   753→                          height: 4,
   754→                          margin: const EdgeInsets.symmetric(horizontal: 0.5),
   755→                          decoration: BoxDecoration(
   756→                            color: isSelected
   757→                                ? cs.onPrimary
   758→                                : dotColors[i],
   759→                            shape: BoxShape.circle,
   760→                          ),
   761→                        ),
   762→                      ),
   763→                    ),
   764→                  ),
   765→              ],
   766→            ),
   767→          ),
   768→        ),
   769→      );
   770→    }
   771→
   772→    // Pad remaining cells to fill the grid
   773→    final totalCells = 7 + startWeekday + daysInMonth;
   774→    final remainder = totalCells % 7;
   775→    if (remainder > 0) {
   776→      for (var i = 0; i < 7 - remainder; i++) {
   777→        cells.add(const SizedBox.shrink());
   778→      }
   779→    }
   780→
   781→    return Padding(
   782→      padding: const EdgeInsets.symmetric(horizontal: Spacings.lg),
   783→      child: GridView.count(
   784→        crossAxisCount: 7,
   785→        childAspectRatio: 1.0,
   786→        shrinkWrap: true,
   787→        physics: const NeverScrollableScrollPhysics(),
   788→        children: cells,
   789→      ),
   790→    );
   791→  }
