   131→// ═══════════════════════════════════════════════════════════════════════
   132→// APP ROUTER PROVIDER
   133→// ═══════════════════════════════════════════════════════════════════════

   135→/// Riverpod provider for the application's [GoRouter].
   136→///
   137→/// Watches [authStateProvider] so that the router automatically
   138→/// re-evaluates its [redirect] callback whenever the authentication
   139→/// state changes (sign-in, sign-out, token refresh).
   140→///
   141→/// Usage in `main.dart`:
   142→/// ```dart
   143→/// final router = ref.read(appRouterProvider);
   144→/// MaterialApp.router(routerConfig: router);
   145→/// ```
   146→final appRouterProvider = Provider<GoRouter>((ref) {
   147→  // Watch auth state so the router re-evaluates redirects on change.
   148→  final authState = ref.watch(authStateProvider);
   149→
   150→  // Also watch onboarding state and role so those redirects stay live.
   151→  ref.watch(onboardingCompleteProvider);
   152→  ref.watch(userRoleProvider);
   153→
   154→  return GoRouter(
   155→    navigatorKey: _rootNavigatorKey,
   156→    initialLocation: RouteNames.splash,
   157→    debugLogDiagnostics: true,
   158→
   159→    // ─── Global Redirect ──────────────────────────────────────────
   160→    redirect: (context, state) {
   161→      final currentPath = state.matchedLocation;
   162→      final isAuthenticated = authState.when(
   163→        data: (authData) => authData.session != null,
   164→        loading: () {
   165→          // While loading, only allow splash. Everything else redirects
   166→          // to splash so we don't flash the login page briefly.
   167→          if (currentPath != RouteNames.splash) {
   168→            return RouteNames.splash;
   169→          }
   170→          return null;
   171→        },
   172→        error: (_, __) => false,
   173→      );
   174→
   175→      // ── Auth Guard ────────────────────────────────────────────
   176→      final authRedirect = AuthGuard.evaluate(
   177→        isAuthenticated: isAuthenticated,
   178→        currentPath: currentPath,
   179→      );
   180→      if (authRedirect != null) return authRedirect;
   181→
   182→      // ── Onboarding Guard ─────────────────────────────────────
   183→      final onboardingAsync = ref.read(onboardingCompleteProvider);
   184→      final isOnboardingComplete = onboardingAsync.when(
   185→        data: (complete) => complete,
   186→        loading: () => true, // Assume complete while loading to avoid flicker.
   187→        error: (_, __) => true,
   188→      );
   189→
   190→      final onboardingRedirect = OnboardingGuard.evaluate(
   191→        isAuthenticated: isAuthenticated,
   192→        isOnboardingComplete: isOnboardingComplete,
   193→        currentPath: currentPath,
   194→      );
   195→      if (onboardingRedirect != null) return onboardingRedirect;
   196→
   197→      // ── Role-Based Guard ─────────────────────────────────────
   198→      final roleAsync = ref.read(userRoleProvider);
   199→      final userRole = roleAsync.when(
   200→        data: UserRole.fromString,
   201→        loading: () => null,
   202→        error: (_, __) => null,
   203→      );
   204→
   205→      final roleRedirect = RoleBasedGuard.evaluate(
   206→        userRole: userRole,
   207→        currentPath: currentPath,
   208→      );
   209→      if (roleRedirect != null) return roleRedirect;
   210→
   211→      // No redirect needed — proceed as requested.
   212→      return null;
   213→    },

  1046→// NOT FOUND PAGE (404)
  1047→// ═══════════════════════════════════════════════════════════════════════
  1048→
  1049→/// Shown when the user navigates to a route that does not exist in
  1050→/// the route tree.
  1051→///
  1052→/// Provides a friendly message and a button to navigate back to the
  1053→/// dashboard (or login if unauthenticated).
  1054→class NotFoundPage extends ConsumerWidget {
  1055→  const NotFoundPage({
  1056→    required this.error,
  1057→    this.path,
  1058→    super.key,
  1059→  });
  1060→
  1061→  /// The [GoException] from GoRouter describing the error.
  1062→  final GoException? error;
  1063→
  1064→  /// The unmatched path the user attempted to visit.
  1065→  final String? path;
  1066→
  1067→  @override
  1068→  Widget build(BuildContext context, WidgetRef ref) {
  1069→    final theme = Theme.of(context);
  1070→    final isAuthenticated = ref.watch(isAuthenticatedProvider);
  1071→
  1072→    return Scaffold(
  1073→      body: Center(
  1074→        child: Padding(
  1075→          padding: const EdgeInsets.all(32),
  1076→          child: Column(
  1077→            mainAxisSize: MainAxisSize.min,
  1078→            children: [
  1079→              Icon(
  1080→                Icons.search_off_rounded,
  1081→                size: 80,
  1082→                color: theme.colorScheme.error,
  1083→              ),
  1084→              const SizedBox(height: 24),
  1085→              Text(
  1086→                'Page Not Found',
  1087→                style: theme.textTheme.headlineMedium?.copyWith(
  1088→                  fontWeight: FontWeight.bold,
  1089→                ),
  1090→                textAlign: TextAlign.center,
  1091→              ),
  1092→              const SizedBox(height: 12),
  1093→              if (path != null)
  1094→                Text(
  1095→                  'The page "$path" does not exist.',
  1096→                  style: theme.textTheme.bodyLarge?.copyWith(
  1097→                    color: theme.colorScheme.onSurfaceVariant,
  1098→                  ),
  1099→                  textAlign: TextAlign.center,
  1100→                ),
  1101→              if (error != null) ...[
  1102→                const SizedBox(height: 8),
  1103→                Text(
  1104→                  error!.message,
  1105→                  style: theme.textTheme.bodySmall?.copyWith(
  1106→                    color: theme.colorScheme.outline,
  1107→                  ),
  1108→                  textAlign: TextAlign.center,
  1109→                ),
  1110→              ],
  1111→              const SizedBox(height: 32),
  1112→              FilledButton.icon(
  1113→                onPressed: () => context.go(
  1114→                  isAuthenticated
  1115→                      ? RouteNames.dashboard
  1116→                      : RouteNames.login,
  1117→                ),
  1118→                icon: const Icon(Icons.home),
  1119→                label: Text(
  1120→                  isAuthenticated ? 'Go to Dashboard' : 'Go to Login',
  1121→                ),
  1122→              ),
