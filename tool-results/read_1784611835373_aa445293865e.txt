   240→-- 6. EXAM_NOTIFICATIONS TABLE (ENHANCED)
   241→-- ============================================================================
   242→-- Enhanced exam-specific notifications that extend the general notifications
   243→-- system. Uses user_id (instead of student_id) to support notifications for
   244→-- both students and teachers. notification_type provides fine-grained event
   245→-- categorization for the exam lifecycle.
   246→--
   247→-- NOTE: This replaces the simpler exam_notifications table from the base
   248→-- cbt_engine_schema.sql. The old table used student_id + category enum;
   249→-- this version uses user_id + notification_type TEXT for broader support.
   250→-- ============================================================================
   251→
   252→-- Safely drop the old exam_notifications table if it exists with the old schema
   253→-- (has 'category' column instead of 'notification_type')
   254→DO $$
   255→BEGIN
   256→  IF EXISTS (
   257→    SELECT 1 FROM information_schema.tables
   258→    WHERE table_name = 'exam_notifications'
   259→      AND table_schema = 'public'
   260→  ) THEN
   261→    -- Check if it's the old schema (has 'category' column)
   262→    IF EXISTS (
   263→      SELECT 1 FROM information_schema.columns
   264→      WHERE table_name = 'exam_notifications'
   265→        AND table_schema = 'public'
   266→        AND column_name = 'category'
   267→    ) THEN
   268→      -- Drop old table with cascade to remove dependent indexes/policies
   269→      DROP TABLE IF EXISTS exam_notifications CASCADE;
   270→      RAISE NOTICE 'Dropped old exam_notifications table (category-based schema) for enhancement migration';
   271→    END IF;
   272→  END IF;
   273→END
   274→$$;
   275→
   276→CREATE TABLE IF NOT EXISTS exam_notifications (
   277→  id                    UUID PRIMARY KEY DEFAULT gen_random_uuid(),
   278→  exam_id               UUID NOT NULL REFERENCES exams(id) ON DELETE CASCADE,
   279→  user_id               UUID NOT NULL REFERENCES users(id) ON DELETE CASCADE,
   280→  notification_type     TEXT NOT NULL CHECK (notification_type IN (
   281→    'exam_published', 'exam_started', 'time_warning_5min', 'time_warning_1min',
   282→    'exam_submitted', 'results_released', 'manual_grading_required',
   283→    'all_submissions_complete', 'suspicious_activity', 'exam_cancelled',
   284→    'exam_rescheduled', 'grading_complete'
   285→  )),
   286→  title                 TEXT NOT NULL,
   287→  message               TEXT NOT NULL,
   288→  data                  JSONB DEFAULT '{}',
   289→  is_read               BOOLEAN NOT NULL DEFAULT false,
   290→  read_at               TIMESTAMPTZ,
   291→  created_at            TIMESTAMPTZ NOT NULL DEFAULT now()
   292→);
   293→
   294→COMMENT ON TABLE exam_notifications IS 'Enhanced exam-specific notifications extending the general notifications system with fine-grained event types';
   295→COMMENT ON COLUMN exam_notifications.exam_id IS 'The exam this notification relates to';
   296→COMMENT ON COLUMN exam_notifications.user_id IS 'The user (student or teacher) who receives this notification';
   297→COMMENT ON COLUMN exam_notifications.notification_type IS 'Specific event type: exam_published, exam_started, time_warning_*, exam_submitted, results_released, etc.';
   298→COMMENT ON COLUMN exam_notifications.title IS 'Notification title for quick scanning';
   299→COMMENT ON COLUMN exam_notifications.message IS 'Full notification message body';
   300→COMMENT ON COLUMN exam_notifications.data IS 'Structured payload (e.g., exam_id, time_remaining, result summary, attempt details)';
   301→COMMENT ON COLUMN exam_notifications.is_read IS 'Whether the user has read this notification';
   302→COMMENT ON COLUMN exam_notifications.read_at IS 'Timestamp when the user read the notification';

   421→-- Increments the usage_count of an exam template when it is used to create
   422→-- an exam. Called by application code or trigger after exam creation.

   455→-- create_exam_from_template(p_template_id UUID, p_created_by UUID,
   456→--   p_start_time TIMESTAMPTZ, p_end_time TIMESTAMPTZ,
   457→--   p_school_id UUID, p_title TEXT DEFAULT NULL)
   458→-- Creates a new exam from a template, copying all configuration and sections.
   459→-- Increments template usage_count. Returns the new exam id.
   460→-- ---------------------------------------------------------------------------
   461→CREATE OR REPLACE FUNCTION create_exam_from_template(
   462→  p_template_id  UUID,
   463→  p_created_by   UUID,
   464→  p_start_time   TIMESTAMPTZ,
   465→  p_end_time     TIMESTAMPTZ,
   466→  p_school_id    UUID,
   467→  p_title        TEXT DEFAULT NULL
   468→)
   469→RETURNS UUID
   470→LANGUAGE plpgsql
   471→SECURITY DEFINER
   472→AS $$
   473→DECLARE
   474→  v_exam_id       UUID;
   475→  v_template      exam_templates%ROWTYPE;
   476→  v_section       exam_template_sections%ROWTYPE;
   477→  v_new_section_id UUID;
   478→BEGIN
   479→  -- Fetch the template
   480→  SELECT * INTO v_template FROM exam_templates WHERE id = p_template_id;
   481→  IF NOT FOUND THEN
   482→    RAISE EXCEPTION 'Template not found: %', p_template_id;
   483→  END IF;
   484→
   485→  -- Verify the template belongs to the specified school or is public
   486→  IF v_template.school_id != p_school_id AND v_template.is_public = false THEN
   487→    RAISE EXCEPTION 'Template % does not belong to school % and is not public', p_template_id, p_school_id;
   488→  END IF;
   489→
   490→  -- Create the exam from template configuration
   491→  INSERT INTO exams (
   492→    school_id, created_by, title, description, subject_id, class_id,
   493→    exam_type, status, start_time, end_time, time_limit_minutes,
   494→    pass_mark, pass_mark_type, instructions, allowed_attempts,
   495→    negative_marking_enabled, negative_mark_value, grace_period_minutes,
   496→    auto_submit, randomize_questions, randomize_options,
   497→    show_results, show_correct_answers, show_explanations,
   498→    is_template, template_id, require_full_screen, allow_resume,
   499→    browser_lockdown, metadata
   500→  ) VALUES (
   501→    p_school_id, p_created_by,
   502→    COALESCE(p_title, v_template.name), v_template.description,
   503→    v_template.subject_id, v_template.class_id,
   504→    v_template.exam_type, 'draft', p_start_time, p_end_time,
   505→    v_template.time_limit_minutes, v_template.pass_mark,
   506→    v_template.pass_mark_type, v_template.instructions,
   507→    v_template.allowed_attempts, v_template.negative_marking_enabled,
   508→    v_template.negative_mark_value, v_template.grace_period_minutes,
   509→    v_template.auto_submit, v_template.randomize_questions,
   510→    v_template.randomize_options, v_template.show_results,
   511→    v_template.show_correct_answers, v_template.show_explanations,
   512→    false, p_template_id, v_template.require_full_screen,
   513→    v_template.allow_resume, v_template.browser_lockdown,
   514→    v_template.metadata
   515→  )
   516→  RETURNING id INTO v_exam_id;
   517→
   518→  -- Copy template sections to exam sections
   519→  FOR v_section IN
   520→    SELECT * FROM exam_template_sections
   521→    WHERE template_id = p_template_id
   522→    ORDER BY sort_order
   523→  LOOP
   524→    INSERT INTO exam_sections (
   525→      exam_id, title, description, instructions, sort_order,
   526→      time_limit_minutes, randomize_questions
   527→    ) VALUES (
   528→      v_exam_id, v_section.title, v_section.description,
   529→      v_section.instructions, v_section.sort_order,
   530→      v_section.time_limit_minutes, v_section.randomize_questions
   531→    )
   532→    RETURNING id INTO v_new_section_id;
   533→
   534→    -- Note: Question selection rules are NOT automatically resolved here.
   535→    -- The application layer should use question_selection_rules to select
   536→    -- questions from the question bank and populate exam_questions.
   537→    -- This function only clones the structural template.
   538→  END LOOP;
   539→
   540→  -- Increment template usage count
   541→  PERFORM increment_template_usage(p_template_id);
   542→
   543→  RETURN v_exam_id;
   544→END;
   545→$$;

   550→-- ---------------------------------------------------------------------------
   551→-- generate_submission_receipt(p_attempt_id UUID)
   552→-- Generates a submission receipt for an exam attempt. Creates a verifiable
   553→-- digital receipt with answer statistics, device info, and a unique receipt
   554→-- number. Returns the receipt id.
   555→-- ---------------------------------------------------------------------------
   556→CREATE OR REPLACE FUNCTION generate_submission_receipt(p_attempt_id UUID)
   557→RETURNS UUID
   558→LANGUAGE plpgsql
   559→SECURITY DEFINER
   560→AS $$
   561→DECLARE
   562→  v_receipt_id     UUID;
   563→  v_attempt        exam_attempts%ROWTYPE;
   564→  v_exam           exams%ROWTYPE;
   565→  v_total_questions  INTEGER := 0;
   566→  v_answered        INTEGER := 0;
   567→  v_unanswered      INTEGER := 0;
   568→  v_flagged         INTEGER := 0;
   569→  v_time_spent_min  INTEGER := 0;
   570→BEGIN
   571→  -- Fetch the attempt
   572→  SELECT * INTO v_attempt FROM exam_attempts WHERE id = p_attempt_id;
   573→  IF NOT FOUND THEN
   574→    RAISE EXCEPTION 'Exam attempt not found: %', p_attempt_id;
   575→  END IF;
   576→
   577→  -- Fetch the exam
   578→  SELECT * INTO v_exam FROM exams WHERE id = v_attempt.exam_id;
   579→
   580→  -- Calculate answer statistics
   581→  SELECT
   582→    COUNT(*),
   583→    COUNT(*) FILTER (WHERE answer_data != '{}'),
   584→    COUNT(*) FILTER (WHERE answer_data = '{}'),
   585→    COUNT(*) FILTER (WHERE is_flagged = true)
   586→  INTO v_total_questions, v_answered, v_unanswered, v_flagged
   587→  FROM student_answers
   588→  WHERE attempt_id = p_attempt_id;
   589→
   590→  -- If no student_answers exist yet, count from exam_questions
   591→  IF v_total_questions = 0 THEN
   592→    SELECT COUNT(*) INTO v_total_questions
   593→    FROM exam_questions
   594→    WHERE exam_id = v_attempt.exam_id;
   595→
   596→    v_unanswered := v_total_questions;
   597→  END IF;
   598→
   599→  -- Calculate time spent in minutes
   600→  IF v_attempt.started_at IS NOT NULL AND v_attempt.submitted_at IS NOT NULL THEN
   601→    v_time_spent_min := EXTRACT(EPOCH FROM (v_attempt.submitted_at - v_attempt.started_at))::INTEGER / 60;
   602→  END IF;
   603→
   604→  -- Create the receipt
   605→  INSERT INTO submission_receipts (
   606→    attempt_id, exam_id, student_id, exam_title,
   607→    submitted_at, submission_type, total_questions, answered_questions,
   608→    unanswered_questions, flagged_questions, time_spent_minutes,
   609→    ip_address, device_info
   610→  ) VALUES (
   611→    p_attempt_id, v_attempt.exam_id, v_attempt.student_id, v_exam.title,
   612→    COALESCE(v_attempt.submitted_at, now()), COALESCE(v_attempt.submission_type, 'manual'),
   613→    v_total_questions, v_answered, v_unanswered, v_flagged,
   614→    v_time_spent_min, v_attempt.ip_address, v_attempt.device_info
   615→  )
   616→  RETURNING id INTO v_receipt_id;
   617→
   618→  RETURN v_receipt_id;
   619→END;
   620→$$;
   621→
   622→COMMENT ON FUNCTION generate_submission_receipt(UUID) IS
   623→  'Generate a verifiable submission receipt for an exam attempt with answer statistics and unique receipt number';
