   100→  Future<void> _generateComments() async {
   101→    if (!_formKey.currentState!.validate()) return;
   102→    if (_selectedClass == null) {
   103→      ScaffoldMessenger.of(context).showSnackBar(
   104→        const SnackBar(
   105→          content: Text('Please select a class'),
   106→          behavior: SnackBarBehavior.floating,
   107→        ),
   108→      );
   109→      return;
   110→    }
   111→
   112→    await ref.read(reportCommentProvider.notifier).generateComments(
   113→          GenerateReportCommentsParams(
   114→            classId: _selectedClass!.id,
   115→            subjectId: _selectedSubject?.id,
   116→            academicSessionId: _selectedSession?.id,
   117→          ),
   118→        );
   119→
   120→    if (!mounted) return;
   121→    final state = ref.read(reportCommentProvider);
   122→    if (state.error != null) return;
   123→
   124→    // Initialize controllers for generated comments.
   125→    _initControllers(state.comments);
   126→  }
   127→
   128→  void _initControllers(List<ReportCommentEntity> comments) {
   129→    for (final comment in comments) {
   130→      _overallCommentControllers.putIfAbsent(
   131→        comment.id,
   132→        () => TextEditingController(text: comment.commentText),
   133→      );
   134→      _academicControllers.putIfAbsent(
   135→        comment.id,
   136→        () => TextEditingController(text: comment.academicPerformance ?? ''),
   137→      );
   138→      _attendanceControllers.putIfAbsent(
   139→        comment.id,
   140→        () => TextEditingController(text: comment.attendanceComment ?? ''),
   141→      );
   142→      _behaviourControllers.putIfAbsent(
   143→        comment.id,
   144→        () => TextEditingController(text: comment.behaviourComment ?? ''),
   145→      );
   146→      _participationControllers.putIfAbsent(
   147→        comment.id,
   148→        () => TextEditingController(text: comment.participationComment ?? ''),
   149→      );
   150→    }
   151→  }
   152→
   153→  Future<void> _publishComment(String commentId) async {
   154→    await ref.read(reportCommentProvider.notifier).publishComment(commentId);
   155→
   156→    if (!mounted) return;
   157→    final state = ref.read(reportCommentProvider);
   158→    if (state.successMessage != null) {
   159→      ScaffoldMessenger.of(context).showSnackBar(
   160→        SnackBar(
   161→          content: Text(state.successMessage!),
   162→          behavior: SnackBarBehavior.floating,
   163→        ),
   164→      );
   165→      ref.read(reportCommentProvider.notifier).clearSuccessMessage();
   166→    }
   167→  }
   168→
   169→  Future<void> _publishAll() async {
   170→    final state = ref.read(reportCommentProvider);
   171→    for (final comment in state.comments) {
   172→      if (!comment.isPublished) {
   173→        await ref.read(reportCommentProvider.notifier).publishComment(comment.id);
   174→      }
   175→    }
   176→
   177→    if (!mounted) return;
   178→    ScaffoldMessenger.of(context).showSnackBar(
   179→      const SnackBar(
   180→        content: Text('All comments published successfully'),
   181→        behavior: SnackBarBehavior.floating,
   182→      ),
   183→    );
   184→  }
   185→
   186→  void _editComment(ReportCommentEntity comment) {
   187→    // Build updated comment from the controllers.
   188→    final updated = comment.copyWith(
   189→      commentText: _overallCommentControllers[comment.id]?.text ?? comment.commentText,
   190→      academicPerformance: _academicControllers[comment.id]?.text ?? comment.academicPerformance,
   191→      attendanceComment: _attendanceControllers[comment.id]?.text ?? comment.attendanceComment,
   192→      behaviourComment: _behaviourControllers[comment.id]?.text ?? comment.behaviourComment,
   193→      participationComment: _participationControllers[comment.id]?.text ?? comment.participationComment,
   194→      isEdited: true,
   195→    );
   196→    ref.read(reportCommentProvider.notifier).updateComment(updated);
   197→
   198→    ScaffoldMessenger.of(context).showSnackBar(
   199→      const SnackBar(
   200→        content: Text('Comment updated'),
   201→        behavior: SnackBarBehavior.floating,
   202→      ),
   203→    );
   204→  }

   474→  Widget _buildBulkPublishBar(BuildContext context, ReportCommentState state) {

   478→        state.comments.where((c) => c.isPublished).length;

   482→      child: Row(
   483→        children: [
   484→          Icon(Icons.rate_review_rounded, size: Spacings.mdIcon, color: cs.primary),

   490→                Text(
   491→                  '$totalCount Comments Generated',

   493→                    fontWeight: AppTypography.wSemiBold,
   494→                    color: cs.onSurface,
   495→                  ),
   496→                ),
   497→                Text(
   498→                  '$publishedCount of $totalCount published',
   499→                  style: tt.bodySmall?.copyWith(
   500→                    color: cs.onSurfaceVariant,
   501→                  ),
   502→                ),
   503→              ],
   504→            ),
   505→          ),
   506→          AppButton(
   507→            label: 'Publish All',
   508→            onPressed: _publishAll,
   509→            variant: AppButtonVariant.elevated,

   512→            isLoading: state.isPublishing,
   513→            isDisabled: state.isPublishing || publishedCount == totalCount,
   514→          ),
   515→        ],
   516→      ),
   517→    );
   518→  }

   525→/// A card displaying a single student's AI-generated report comment with
   526→/// editable fields for each section.
   527→class _StudentCommentCard extends StatelessWidget {
   528→  const _StudentCommentCard({
   529→    required this.comment,
   530→    this.overallController,
   531→    this.academicController,
   532→    this.attendanceController,
   533→    this.behaviourController,
   534→    this.participationController,
   535→    this.onEdit,
   536→    this.onPublish,
   537→  });
   538→
   539→  final ReportCommentEntity comment;
   540→  final TextEditingController? overallController;
   541→  final TextEditingController? academicController;
   542→  final TextEditingController? attendanceController;
   543→  final TextEditingController? behaviourController;
   544→  final TextEditingController? participationController;
   545→  final VoidCallback? onEdit;
   546→  final VoidCallback? onPublish;
   547→
   548→  @override
   549→  Widget build(BuildContext context) {
   550→    final cs = context.colorScheme;
   551→    final tt = context.textTheme;
   552→
   553→    // Student name from metadata or fallback
   554→    final studentName = comment.metadata['studentName'] as String? ??
   555→        'Student ${comment.studentId?.substring(0, 8) ?? 'Unknown'}';
   556→
   557→    return AppCard(
   558→      child: Column(
   559→        crossAxisAlignment: CrossAxisAlignment.start,
   560→        children: [
   561→          // ── Header: Student name + AI badge ──────────────────────────
   562→          Row(
   563→            children: [
   564→              // Avatar
   565→              CircleAvatar(
   566→                radius: 20,
   567→                backgroundColor:
   568→                    cs.primary.withValues(alpha: context.isDarkMode ? 0.20 : 0.12),
   569→                child: Text(
   570→                  studentName.isNotEmpty ? studentName[0].toUpperCase() : '?',
   571→                  style: tt.titleMedium?.copyWith(
   572→                    color: cs.primary,
   573→                    fontWeight: AppTypography.wBold,
   574→                  ),
   575→                ),
   576→              ),
   577→              const SizedBox(width: Spacings.md),
   578→              Expanded(
   579→                child: Column(
   580→                  crossAxisAlignment: CrossAxisAlignment.start,
   581→                  children: [
   582→                    Text(
   583→                      studentName,
   584→                      style: tt.titleSmall?.copyWith(
   585→                        fontWeight: AppTypography.wSemiBold,
   586→                        color: cs.onSurface,
   587→                      ),
   588→                    ),
   589→                    if (comment.isEdited)
   590→                      Text(
   591→                        'Edited',
   592→                        style: tt.bodySmall?.copyWith(
   593→                          color: cs.onSurfaceVariant,
   594→                          fontStyle: FontStyle.italic,
   595→                        ),
   596→                      ),
   597→                  ],
   598→                ),
   599→              ),
   600→              // AI-generated indicator
   601→              if (comment.isAiGenerated)
   602→                Container(
   603→                  padding: const EdgeInsets.symmetric(
   604→                    horizontal: Spacings.sm,
   605→                    vertical: Spacings.xs,
   606→                  ),
   607→                  decoration: BoxDecoration(
   608→                    color: cs.tertiary.withValues(
   609→                        alpha: context.isDarkMode ? 0.20 : 0.12),
   610→                    borderRadius: BorderRadius.circular(Spacings.fullRadius),
   611→                  ),
   612→                  child: Row(
