   305→  SuperAdminRemoteDataSourceImpl({

   363→  @override
   364→  Future<List<PlatformSettingModel>> getPlatformSettings({
   365→    String? scope,
   366→    int limit = PaginatedQueryMixin.defaultPageSize,
   367→    int offset = 0,
   368→  }) async {
   369→    try {
   370→      // PERF: Replaced bare .select() with explicit columns + pagination to avoid unbounded result sets
   371→      var query = _supabase.from(_platformSettingsTable).select('id, key, value, scope, description, is_active, updated_at, created_at');
   372→
   373→      if (scope != null) {
   374→        query = query.eq('scope', scope);
   375→      }
   376→
   377→      final list = await query.order('key', ascending: true).range(offset, offset + limit - 1);
   378→      AppLogger.info('Fetched ${list.length} platform settings');
   379→      return list
   380→          .map<PlatformSettingModel>(
   381→            (row) => PlatformSettingModel.fromJson(row),
   382→          )
   383→          .toList();
   384→    } on sb.PostgrestException catch (e) {
   385→      throw _mapPostgrestException(e);
   386→    } catch (e) {
   387→      AppLogger.error('Unexpected getPlatformSettings error', error: e);
   388→      throw const ServerException(
   389→        message: 'Failed to fetch platform settings.',
   390→        statusCode: 500,
   391→      );
   392→    }
   393→  }
   394→
   395→  @override
   396→  Future<PlatformSettingModel> getPlatformSetting(
   397→    String key, {
   398→    String? scope,
   399→  }) async {
   400→    try {
   401→      var query = _supabase
   402→          .from(_platformSettingsTable)
   403→          .select()
   404→          .eq('key', key);
   405→
   406→      if (scope != null) {
   407→        query = query.eq('scope', scope);
   408→      }
   409→
   410→      final response = await query.limit(1);
   411→
   412→      if (response.isEmpty) {
   413→        throw const NotFoundException('Platform setting not found.');
   414→      }
   415→
   416→      return PlatformSettingModel.fromJson(response.first);
   417→    } on sb.PostgrestException catch (e) {
   418→      throw _mapPostgrestException(e);
   419→    } on NotFoundException {
   420→      rethrow;
   421→    } catch (e) {
   422→      AppLogger.error('Unexpected getPlatformSetting error', error: e);
   423→      throw const ServerException(
   424→        message: 'Failed to fetch platform setting.',
   425→        statusCode: 500,
   426→      );
   427→    }
   428→  }

   430→  @override
   431→  Future<PlatformSettingModel> updatePlatformSetting(
   432→    Map<String, dynamic> settingData,
   433→  ) async {
   434→    try {
   435→      settingData['updated_at'] = DateTime.now().toIso8601String();
   436→
   437→      final response = await _supabase
   438→          .from(_platformSettingsTable)
   439→          .update(settingData)
   440→          .eq('id', settingData['id'] as String)
   441→          .select();
   442→
   443→      if (response.isEmpty) {
   444→        throw const NotFoundException('Platform setting not found for update.');
   445→      }
   446→
   447→      AppLogger.info('Platform setting updated: ${response.first['key']}');
   448→      return PlatformSettingModel.fromJson(response.first);
   449→    } on sb.PostgrestException catch (e) {
   450→      throw _mapPostgrestException(e);
   451→    } on NotFoundException {
   452→      rethrow;
   453→    } catch (e) {
   454→      AppLogger.error('Unexpected updatePlatformSetting error', error: e);
   455→      throw const ServerException(
   456→        message: 'Failed to update platform setting.',
   457→        statusCode: 500,
   458→      );
   459→    }
   460→  }

   462→  @override
   463→  Future<List<PlatformSettingModel>> bulkUpdateSettings(
   464→    List<Map<String, dynamic>> settingsData,
   465→  ) async {
   466→    try {
   467→      final results = <PlatformSettingModel>[];
   468→
   469→      for (final data in settingsData) {
   470→        data['updated_at'] = DateTime.now().toIso8601String();
   471→
   472→        final response = await _supabase
   473→            .from(_platformSettingsTable)
   474→            .update(data)
   475→            .eq('id', data['id'] as String)
   476→            .select();
   477→
   478→        if (response.isNotEmpty) {
   479→          results.add(PlatformSettingModel.fromJson(response.first));
   480→        }
   481→      }
   482→
   483→      AppLogger.info('Bulk updated ${results.length} platform settings');
   484→      return results;
   485→    } on sb.PostgrestException catch (e) {
   486→      throw _mapPostgrestException(e);
   487→    } catch (e) {
   488→      AppLogger.error('Unexpected bulkUpdateSettings error', error: e);
   489→      throw const ServerException(
   490→        message: 'Failed to bulk update platform settings.',
   491→        statusCode: 500,
   492→      );
   493→    }
   494→  }
