   241→   241→  @override
   242→   242→  Future<List<SubscriptionPlanModel>> getSubscriptionPlans({
   243→   243→    String? billingModel,
   244→   244→    bool activeOnly = true,
   245→   245→  }) async {
   246→   246→    try {
   247→   247→      var query = _supabase.from(_plansTable).select();
   248→   248→
   249→   249→      if (billingModel != null) {
   250→   250→        query = query.eq('billing_model', billingModel);
   251→   251→      }
   252→   252→      if (activeOnly) {
   253→   253→        query = query.eq('is_active', true);
   254→   254→      }
   255→   255→
   256→   256→      final response = query.order('sort_order', ascending: true);
   257→   257→
   258→   258→      final list = await response;
   259→   259→      AppLogger.info('Fetched ${list.length} subscription plans');
   260→   260→      return list
   261→   261→          .map<SubscriptionPlanModel>(
   262→   262→            (row) => SubscriptionPlanModel.fromJson(row),
   263→   263→          )
   264→   264→          .toList();
   265→   265→    } on sb.PostgrestException catch (e) {
   266→   266→      throw _mapPostgrestException(e);
   267→   267→    } catch (e) {
   268→   268→      AppLogger.error('Unexpected getSubscriptionPlans error', error: e);
   269→   269→      throw const ServerException(
   270→   270→        message: 'Failed to fetch subscription plans.',
   271→   271→        statusCode: 500,
   272→   272→      );
   273→   273→    }
   274→   274→  }

   276→   276→  @override

   279→   279→      final response = await _supabase
   280→   280→          .from(_plansTable)
   281→   281→          .select()
   282→   282→          .eq('id', planId)
   283→   283→          .limit(1);
   284→   284→
   285→   285→      if (response.isEmpty) {
   286→   286→        throw const NotFoundException('Subscription plan not found.');
   287→   287→      }
   288→   288→
   289→   289→      return SubscriptionPlanModel.fromJson(response.first);
   290→   290→    } on sb.PostgrestException catch (e) {
   291→   291→      throw _mapPostgrestException(e);
   292→   292→    } on NotFoundException {
   293→   293→      rethrow;
   294→   294→    } catch (e) {
   295→   295→      AppLogger.error('Unexpected getSubscriptionPlan error', error: e);
   296→   296→      throw const ServerException(
   297→   297→        message: 'Failed to fetch subscription plan.',
   298→   298→        statusCode: 500,

   361→   361→    } on sb.PostgrestException catch (e) {

   363→   363→    } on NotFoundException {
   364→   364→      rethrow;
   365→   365→    } catch (e) {
   366→   366→      AppLogger.error('Unexpected getCurrentSubscription error', error: e);

   374→   374→  @override
   375→   375→  Future<List<SubscriptionModel>> getSubscriptions({
   376→   376→    String? schoolId,
   377→   377→    String? subscriberType,
   378→   378→    String? status,
   379→   379→    int page = 1,
   380→   380→    int perPage = 20,
   381→   381→  }) async {
   382→   382→    try {
   383→   383→      var query = _supabase.from(_subscriptionsTable).select();
   384→   384→
   385→   385→      if (schoolId != null) {
   386→   386→        query = query.eq('school_id', schoolId);
   387→   387→      }
   388→   388→      if (subscriberType != null) {
   389→   389→        query = query.eq('subscriber_type', subscriberType);
   390→   390→      }
   391→   391→      if (status != null) {
   392→   392→        query = query.eq('status', status);
   393→   393→      }
   394→   394→
   395→   395→      final filtered = _applyFilters(query, {
   396→   396→        if (schoolId != null) 'school_id': schoolId,
   397→   397→        if (subscriberType != null) 'subscriber_type': subscriberType,
   398→   398→        if (status != null) 'status': status,
   399→   399→      });
   400→   400→
   401→   401→      var transformed =
   402→   402→          filtered.order('created_at', ascending: false) as sb.PostgrestTransformBuilder<dynamic>;
   403→   403→      transformed = _applyPagination(transformed, page: page, perPage: perPage);
   404→   404→
   405→   405→      final list = await transformed;
   406→   406→      AppLogger.info('Fetched ${list.length} subscriptions');
   407→   407→      return list
   408→   408→          .map<SubscriptionModel>(
   409→   409→            (row) => SubscriptionModel.fromJson(row),
   410→   410→          )
   411→   411→          .toList();
   412→   412→    } on sb.PostgrestException catch (e) {
   413→   413→      throw _mapPostgrestException(e);
   414→   414→    } catch (e) {
   415→   415→      AppLogger.error('Unexpected getSubscriptions error', error: e);
   416→   416→      throw const ServerException(
   417→   417→        message: 'Failed to fetch subscriptions.',
   418→   418→        statusCode: 500,
   419→   419→      );
