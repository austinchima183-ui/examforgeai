   136→   136→   136→/// Root navigator key — owns the top-level navigation stack.
   137→   137→   137→final _rootNavigatorKey = GlobalKey<NavigatorState>();
   138→   138→   138→
   139→   139→   139→/// Shell navigator key — owns the nested navigation stack inside the
   140→   140→   140→/// [DashboardShell]. This allows the shell (sidebar / bottom nav) to
   141→   141→   141→/// remain mounted while inner pages transition.

   145→   145→   145→// APP ROUTER PROVIDER
   146→   146→   146→// ═══════════════════════════════════════════════════════════════════════
   147→   147→   147→
   148→   148→   148→/// Riverpod provider for the application's [GoRouter].
   149→   149→   149→///
   150→   150→   150→/// Watches [authStateProvider] so that the router automatically
   151→   151→   151→/// re-evaluates its [redirect] callback whenever the authentication
   152→   152→   152→/// state changes (sign-in, sign-out, token refresh).
   153→   153→   153→///
   154→   154→   154→/// Usage in `main.dart`:
   155→   155→   155→/// ```dart
   156→   156→   156→/// final router = ref.read(appRouterProvider);
   157→   157→   157→/// MaterialApp.router(routerConfig: router);
   158→   158→   158→/// ```
   159→   159→   159→final appRouterProvider = Provider<GoRouter>((ref) {
   160→   160→   160→  // Watch auth state so the router re-evaluates redirects on change.
   161→   161→   161→  final authState = ref.watch(authStateProvider);
   162→   162→   162→
   163→   163→   163→  // Also watch onboarding state and role so those redirects stay live.
   164→   164→   164→  ref.watch(onboardingCompleteProvider);
   165→   165→   165→  ref.watch(userRoleProvider);
   166→   166→   166→
   167→   167→   167→  return GoRouter(
   168→   168→   168→    navigatorKey: _rootNavigatorKey,
   169→   169→   169→    initialLocation: RouteNames.splash,
   170→   170→   170→    debugLogDiagnostics: true,
   171→   171→   171→
   172→   172→   172→    // ─── Global Redirect ──────────────────────────────────────────
   173→   173→   173→    redirect: (context, state) {
   174→   174→   174→      final currentPath = state.matchedLocation;
   175→   175→   175→      final isAuthenticated = authState.when(
   176→   176→   176→        data: (authData) => authData.session != null,
   177→   177→   177→        loading: () {
   178→   178→   178→          // While loading, only allow splash. Everything else redirects
   179→   179→   179→          // to splash so we don't flash the login page briefly.
   180→   180→   180→          if (currentPath != RouteNames.splash) {
   181→   181→   181→            return RouteNames.splash;
   182→   182→   182→          }
   183→   183→   183→          return null;
   184→   184→   184→        },
   185→   185→   185→        error: (_, __) => false,
   186→   186→   186→      );
   187→   187→   187→
   188→   188→   188→      // ── Auth Guard ────────────────────────────────────────────
   189→   189→   189→      final authRedirect = AuthGuard.evaluate(
   190→   190→   190→        isAuthenticated: isAuthenticated,
   191→   191→   191→        currentPath: currentPath,
   192→   192→   192→      );
   193→   193→   193→      if (authRedirect != null) return authRedirect;
   194→   194→   194→
   195→   195→   195→      // ── Onboarding Guard ─────────────────────────────────────
   196→   196→   196→      final onboardingAsync = ref.read(onboardingCompleteProvider);
   197→   197→   197→      final isOnboardingComplete = onboardingAsync.when(
   198→   198→   198→        data: (complete) => complete,
   199→   199→   199→        loading: () => true, // Assume complete while loading to avoid flicker.
   200→   200→   200→        error: (_, __) => true,
   201→   201→   201→      );
   202→   202→   202→
   203→   203→   203→      final onboardingRedirect = OnboardingGuard.evaluate(
   204→   204→   204→        isAuthenticated: isAuthenticated,
   205→   205→   205→        isOnboardingComplete: isOnboardingComplete,
   206→   206→   206→        currentPath: currentPath,
   207→   207→   207→      );
   208→   208→   208→      if (onboardingRedirect != null) return onboardingRedirect;
   209→   209→   209→
   210→   210→   210→      // ── Role-Based Guard ─────────────────────────────────────
   211→   211→   211→      final roleAsync = ref.read(userRoleProvider);
   212→   212→   212→      final userRole = roleAsync.when(
   213→   213→   213→        data: UserRole.fromString,
   214→   214→   214→        loading: () => null,
   215→   215→   215→        error: (_, __) => null,
   216→   216→   216→      );
   217→   217→   217→
   218→   218→   218→      final roleRedirect = RoleBasedGuard.evaluate(
   219→   219→   219→        userRole: userRole,
   220→   220→   220→        currentPath: currentPath,
   221→   221→   221→      );
   222→   222→   222→      if (roleRedirect != null) return roleRedirect;
   223→   223→   223→
   224→   224→   224→      // No redirect needed — proceed as requested.
   225→   225→   225→      return null;
   226→   226→   226→    },
