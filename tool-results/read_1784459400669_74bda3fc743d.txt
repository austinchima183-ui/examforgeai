    71→    71→String _formatLastLogin(DateTime? lastLoginAt) {
    72→    72→  if (lastLoginAt == null) return 'Never';
    73→    73→  final now = DateTime.now();
    74→    74→  final diff = now.difference(lastLoginAt);
    75→    75→  if (diff.inMinutes < 1) return 'Just now';
    76→    76→  if (diff.inHours < 1) return '${diff.inMinutes}m ago';
    77→    77→  if (diff.inDays < 1) return '${diff.inHours}h ago';
    78→    78→  if (diff.inDays < 30) return '${diff.inDays}d ago';
    79→    79→  return '${lastLoginAt.day}/${lastLoginAt.month}/${lastLoginAt.year}';
    80→    80→}
