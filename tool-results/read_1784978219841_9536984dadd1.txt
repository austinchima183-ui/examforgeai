    46→  /// Stores [data] in the cache under [key] for [userId].
    47→  ///
    48→  /// If [ttl] is provided the entry will be considered expired after
    49→  /// the given duration; otherwise it never expires.
    50→  Future<void> cacheData({
    51→    required String key,
    52→    required String userId,
    53→    required String resourceType,
    54→    required String resourceId,
    55→    required Map<String, dynamic> data,
    56→    Duration? ttl,
    57→  }) async {
    58→    try {
    59→      final now = DateTime.now();
    60→      final id = _uuid.v4();
    61→      final json = jsonEncode(data);
    62→      final expiresAt = ttl != null ? now.add(ttl) : null;
    63→
    64→      await database.into(database.cache).insertOnConflictUpdate(
    65→            LocalCacheTableCompanion.insert(
    66→              id: id,
    67→              userId: userId,
    68→              cacheKey: key,
    69→              resourceType: resourceType,
    70→              resourceId: resourceId,
    71→              data: json,
    72→              expiresAt: Value(expiresAt),
    73→              fileSizeBytes: Value(json.length),
    74→              createdAt: now,
    75→              updatedAt: now,
    76→            ),
    77→          );
    78→      AppLogger.debug('CacheManager: cached data for key=$key');
    79→    } catch (e, st) {
    80→      AppLogger.error('CacheManager: failed to cache data', error: e, stackTrace: st);
    81→      rethrow;
    82→    }
    83→  }
    84→
    85→  /// Retrieves cached data for [key] belonging to [userId].
    86→  ///
    87→  /// Returns `null` if no entry exists or the entry has expired.
    88→  /// On a successful hit the access count and last-accessed timestamp
    89→  /// are updated.
    90→  Future<Map<String, dynamic>?> getCachedData({
    91→    required String key,
    92→    required String userId,
    93→  }) async {
    94→    try {
    95→      final query = database.select(database.cache)
    96→        ..where((t) => t.cacheKey.equals(key) & t.userId.equals(userId));
    97→
    98→      final row = await query.getSingleOrNull();
    99→      if (row == null) return null;
   100→
   101→      // Check expiry.
   102→      if (row.expiresAt != null &&
   103→          row.expiresAt!.isBefore(DateTime.now())) {
   104→        await invalidateCache(key: key, userId: userId);
   105→        return null;
   106→      }
   107→
   108→      // Update access metadata.
   109→      await (database.update(database.cache)
   110→            ..where((t) => t.id.equals(row.id)))
   111→          .write(LocalCacheTableCompanion(
   112→        accessCount: Value(row.accessCount + 1),
   113→        lastAccessedAt: Value(DateTime.now()),
   114→      ),);
   115→
   116→      return jsonDecode(row.data) as Map<String, dynamic>;
   117→    } catch (e, st) {
   118→      AppLogger.error('CacheManager: failed to get cached data', error: e, stackTrace: st);
   119→      return null;
   120→    }
   121→  }
   122→
   123→  /// Removes a single cache entry identified by [key] and [userId].
   124→  Future<void> invalidateCache({
   125→    required String key,
   126→    required String userId,
   127→  }) async {
   128→    try {
   129→      await (database.delete(database.cache)
   130→            ..where((t) => t.cacheKey.equals(key) & t.userId.equals(userId)))
   131→          .go();
   132→      AppLogger.debug('CacheManager: invalidated cache key=$key');
   133→    } catch (e, st) {
   134→      AppLogger.error('CacheManager: failed to invalidate cache', error: e, stackTrace: st);
   135→      rethrow;
   136→    }
   137→  }
   138→
   139→  /// Removes **all** cache entries for [resourceType] belonging to [userId].
   140→  Future<void> invalidateCacheByType({
   141→    required String resourceType,
   142→    required String userId,
   143→  }) async {
   144→    try {
   145→      final count = await (database.delete(database.cache)
   146→            ..where(
   147→                (t) => t.resourceType.equals(resourceType) & t.userId.equals(userId),))
   148→          .go();
   149→      AppLogger.debug(
   150→        'CacheManager: invalidated $count cache entries of type=$resourceType',
   151→      );
   152→    } catch (e, st) {
   153→      AppLogger.error('CacheManager: failed to invalidate cache by type', error: e, stackTrace: st);
   154→      rethrow;
   155→    }
   156→  }
   157→
   158→  /// Removes all expired cache entries across all users.
   159→  Future<void> clearExpiredCache() async {
   160→    try {
   161→      final count = await database.clearExpiredCache();
   162→      AppLogger.debug('CacheManager: cleared $count expired cache entries');
   163→    } catch (e, st) {
   164→      AppLogger.error('CacheManager: failed to clear expired cache', error: e, stackTrace: st);
   165→      rethrow;
   166→    }
   167→  }

   169→  /// Returns `true` if a non-expired cache entry exists for [key].
   170→  Future<bool> isCacheValid({
   171→    required String key,
   172→    required String userId,
   173→  }) async {
   174→    try {
   175→      final data = await getCachedData(key: key, userId: userId);
   176→      return data != null;
   177→    } catch (e, st) {
   178→      AppLogger.error('CacheManager: cache validity check failed', error: e, stackTrace: st);
   179→      return false;
   180→    }
   181→  }

   200→  /// Caches a user profile locally.
   201→  ///
   202→  /// If a profile for [userId] already exists it will be updated.
   203→  Future<void> cacheUserProfile({
   204→    required String userId,
   205→    required Map<String, dynamic> profileData,
   206→    required String role,
   207→    String? schoolId,
   208→  }) async {
   209→    try {
   210→      final now = DateTime.now();
   211→
   212→      // Check if a profile already exists for this user.
   213→      final existing = await (database.select(database.userData)
   214→            ..where((t) => t.userId.equals(userId)))
   215→          .getSingleOrNull();
   216→
   217→      if (existing != null) {
   218→        await (database.update(database.userData)
   219→              ..where((t) => t.userId.equals(userId)))
   220→            .write(LocalUserDataTableCompanion(
   221→          profileData: Value(jsonEncode(profileData)),
   222→          role: Value(role),
   223→          schoolId: Value(schoolId),
   224→          lastSyncedAt: Value(now),
   225→          updatedAt: Value(now),
   226→        ),);
   227→      } else {
   228→        await database.into(database.userData).insert(
   229→              LocalUserDataTableCompanion.insert(
   230→                id: _uuid.v4(),
   231→                userId: userId,
   232→                profileData: jsonEncode(profileData),
   233→                role: role,
   234→                schoolId: Value(schoolId),
   235→                lastSyncedAt: now,
   236→                createdAt: now,
   237→                updatedAt: now,
   238→              ),
   239→            );
   240→      }
   241→      AppLogger.debug('CacheManager: cached user profile for $userId');
   242→    } catch (e, st) {
   243→      AppLogger.error('CacheManager: failed to cache user profile', error: e, stackTrace: st);
   244→      rethrow;
   245→    }
   246→  }

   305→  /// Returns all drafts for [userId], optionally filtered by [draftType].
   306→  Future<List<Map<String, dynamic>>> getDrafts({
   307→    required String userId,
   308→    String? draftType,
   309→  }) async {
   310→    try {
   311→      final query = database.select(database.drafts)
   312→        ..where((t) => t.userId.equals(userId));
   313→
   314→      if (draftType != null) {
   315→        query.where((t) => t.draftType.equals(draftType));
   316→      }
   317→
   318→      // Most recently edited first.
   319→      query.orderBy([
   320→        (t) => OrderingTerm.desc(t.lastEditedAt),
   321→      ]);
   322→
   323→      final rows = await query.get();
   324→      return rows.map((row) => {
   325→            'id': row.id,
   326→            'userId': row.userId,
   327→            'draftType': row.draftType,
   328→            'title': row.title,
   329→            'content': jsonDecode(row.content),
   330→            'schoolId': row.schoolId,
   331→            'subjectId': row.subjectId,
   332→            'isSynced': row.isSynced,
   333→            'lastEditedAt': row.lastEditedAt.toIso8601String(),
   334→            'createdAt': row.createdAt.toIso8601String(),
   335→            'updatedAt': row.updatedAt.toIso8601String(),
   336→          },).toList();
   337→    } catch (e, st) {
   338→      AppLogger.error('CacheManager: failed to get drafts', error: e, stackTrace: st);
   339→      return [];
   340→    }
   341→  }

   405→  /// Returns up to [limit] pending sync items for [userId],
   406→  /// ordered by priority (ascending) then creation time (ascending).
   407→  Future<List<Map<String, dynamic>>> getPendingSyncItems({
   408→    required String userId,
   409→    int limit = 50,
   410→  }) async {
   411→    try {
   412→      final query = database.select(database.syncQueue)

   417→          (t) => OrderingTerm.asc(t.priority),
   418→          (t) => OrderingTerm.asc(t.createdAt),
   419→        ])
   420→        ..limit(limit);
   421→
   422→      final rows = await query.get();
   423→      return rows.map((row) => {
   424→            'id': row.id,
   425→            'userId': row.userId,
   426→            'targetTable': row.targetTable,
   427→            'recordId': row.recordId,
   428→            'operation': row.operation,
   429→            'payload': jsonDecode(row.payload),
   430→            'priority': row.priority,
   431→            'attempts': row.attempts,
   432→            'maxAttempts': row.maxAttempts,
   433→            'lastAttemptAt': row.lastAttemptAt?.toIso8601String(),
   434→            'nextRetryAt': row.nextRetryAt?.toIso8601String(),
   435→            'status': row.status,
   436→            'errorMessage': row.errorMessage,
   437→            'createdAt': row.createdAt.toIso8601String(),
   438→            'updatedAt': row.updatedAt.toIso8601String(),

   462→  /// Marks a sync queue item as failed with [errorMessage].
   463→  ///
   464→  /// If the number of attempts exceeds `maxAttempts` the item is
   465→  /// automatically moved to the `dead` status.
   466→  Future<void> markSyncItemFailed({
   467→    required String id,
   468→    required String errorMessage,
   469→  }) async {
   470→    try {
   471→      final now = DateTime.now();
   472→
   473→      // Fetch current row to check attempt count.
   474→      final row = await (database.select(database.syncQueue)
   475→            ..where((t) => t.id.equals(id)))
   476→          .getSingleOrNull();
   477→
   478→      if (row == null) {
   479→        AppLogger.warning('CacheManager: sync item $id not found');
   480→        return;
   481→      }
   482→
   483→      final newAttempts = row.attempts + 1;
   484→      final isDead = newAttempts >= row.maxAttempts;
   485→
   486→      // Exponential backoff: 2^attempts seconds, capped at 1 hour.
   487→      final backoffSeconds = (1 << newAttempts).clamp(1, 3600);
   488→
   489→      await (database.update(database.syncQueue)..where((t) => t.id.equals(id)))
   490→          .write(LocalSyncQueueTableCompanion(
   491→        attempts: Value(newAttempts),
   492→        status: Value(isDead ? 'dead' : 'failed'),
   493→        errorMessage: Value(errorMessage),
   494→        lastAttemptAt: Value(now),
   495→        nextRetryAt: Value(isDead ? null : now.add(Duration(seconds: backoffSeconds))),
   496→        updatedAt: Value(now),
   497→      ),);
   498→
   499→      AppLogger.debug(
   500→        'CacheManager: sync item $id failed (attempt $newAttempts/${row.maxAttempts})'
   501→        '${isDead ? ' → dead' : ''}',
   502→      );
   503→    } catch (e, st) {
   504→      AppLogger.error('CacheManager: failed to mark sync item failed', error: e, stackTrace: st);
   505→      rethrow;
   506→    }
   507→  }

   759→  /// Deletes all local data associated with [userId] across all tables.
   760→  ///
   761→  /// Typically called on logout. Does **not** clear connectivity logs
   762→  /// or sync metadata (which may be shared across sessions).
   763→  Future<void> clearAllUserData({required String userId}) async {
   764→    try {
   765→      await database.transaction(() async {
   766→        await (database.delete(database.syncQueue)
   767→              ..where((t) => t.userId.equals(userId)))
   768→            .go();
   769→        await (database.delete(database.cache)
   770→              ..where((t) => t.userId.equals(userId)))
   771→            .go();
   772→        await (database.delete(database.drafts)
   773→              ..where((t) => t.userId.equals(userId)))
   774→            .go();
   775→        await (database.delete(database.userData)
   776→              ..where((t) => t.userId.equals(userId)))
   777→            .go();
   778→        await (database.delete(database.resources)
   779→              ..where((t) => t.userId.equals(userId)))
   780→            .go();
   781→        await (database.delete(database.notifications)
   782→              ..where((t) => t.userId.equals(userId)))
   783→            .go();
   784→        await (database.delete(database.syncMetadata)
   785→              ..where((t) => t.userId.equals(userId)))
   786→            .go();
   787→        // Exam attempts are keyed by studentId.
   788→        await (database.delete(database.examAttempts)
   789→              ..where((t) => t.studentId.equals(userId)))
   790→            .go();
   791→      });
   792→      AppLogger.info('CacheManager: cleared all data for user $userId');
   793→    } catch (e, st) {
   794→      AppLogger.error('CacheManager: failed to clear user data', error: e, stackTrace: st);
   795→      rethrow;
   796→    }
   797→  }
   798→}

   804→/// Provides the singleton [AppDatabase] instance.
   805→///
   806→/// The database is lazily opened on first access and automatically
   807→/// closed when the provider is disposed.
   808→final appDatabaseProvider = Provider<AppDatabase>((ref) {
   809→  final db = AppDatabase();
   810→  ref.onDispose(() {
   811→    db.close();
   812→    AppLogger.info('AppDatabase: connection closed');
   813→  });
   814→  return db;
   815→});
   816→
   817→/// Provides the singleton [CacheManager] instance.
   818→///
   819→/// Depends on [appDatabaseProvider] and [storageServiceProvider].
   820→final cacheManagerProvider = Provider<CacheManager>((ref) {
   821→  final database = ref.watch(appDatabaseProvider);
   822→  final storageService = ref.watch(storageServiceProvider);
   823→  return CacheManager(
   824→    database: database,
   825→    storageService: storageService,
   826→  );
   827→});
