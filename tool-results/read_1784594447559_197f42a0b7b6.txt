   773→   773→  String _formatDateTime(DateTime dt) {
   774→   774→    const months = [
   775→   775→      '', 'Jan', 'Feb', 'Mar', 'Apr', 'May', 'Jun',
   776→   776→      'Jul', 'Aug', 'Sep', 'Oct', 'Nov', 'Dec',
   777→   777→    ];
   778→   778→    final hour = dt.hour.toString().padLeft(2, '0');
   779→   779→    final minute = dt.minute.toString().padLeft(2, '0');
   780→   780→    return '${months[dt.month]} ${dt.day}, ${dt.year} at $hour:$minute';
   781→   781→  }

   783→   783→  String _formatTimeSpent(int minutes) {
   784→   784→    if (minutes < 60) return '$minutes min';
   785→   785→    final hours = minutes ~/ 60;
   786→   786→    final mins = minutes % 60;
   787→   787→    return mins > 0 ? '${hours}h ${mins}m' : '${hours}h';
   788→   788→  }
   789→   789→}
