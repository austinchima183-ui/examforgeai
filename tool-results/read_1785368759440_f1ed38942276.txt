    10→    10→// SUPABASE REALTIME NOTIFICATION SERVICE
    11→    11→// ═══════════════════════════════════════════════════════════════════════
    12→    12→// Purpose: Production notification system built on Supabase Realtime
    13→    13→// Architecture:
    14→    14→//   - Supabase Realtime for instant in-app notifications
    15→    15→//   - Browser Notification API for web push notifications
    16→    16→//   - Supabase Postgres changes for notification persistence
    17→    17→//   - device_tokens table for web push subscription management
    18→    18→//   - Realtime channels for role-based and school-based broadcasts
    19→    19→//
    20→    20→// This replaces the previous Firebase Cloud Messaging implementation.
    21→    21→// Firebase is NOT part of the ExamForge AI architecture — the project
    22→    22→// is Supabase-only. This service provides:
    23→    23→//   1. Real-time notification delivery via Supabase Realtime
    24→    24→//   2. Browser push notifications via the Web Notification API
    25→    25→//   3. Notification persistence in the notifications table
    26→    26→//   4. Read/unread tracking
    27→    27→//   5. Role-based and school-based notification channels
    28→    28→//   6. Admin broadcast notifications
    29→    29→//   7. CBT exam notifications
    30→    30→//   8. Payment/billing notifications
    31→    31→//   9. Parent notifications

    88→    88→  /// Create from a Supabase row.

    92→    92→      title: map['title']?.toString() ?? '',

   100→   100→          ? DateTime.parse(map['created_at'].toString())

   192→   192→  /// Initializes the notification service.
   193→   193→  ///
   194→   194→  /// Steps:
   195→   195→  /// 1. Requests browser notification permissions (web).
   196→   196→  /// 2. Registers a device token for push notifications.
   197→   197→  /// 3. Subscribes to the user's personal notification channel.
   198→   198→  /// 4. Subscribes to role-based and school-based channels.

   200→   200→  Future<void> initialize({
   201→   201→    ForegroundNotificationCallback? onForegroundNotification,
   202→   202→    NotificationTapCallback? onNotificationTap,
   203→   203→  }) async {
   204→   204→    if (_initialized) {
   205→   205→      AppLogger.warning('NotificationService already initialized — skipping.');
   206→   206→      return;
   207→   207→    }
   208→   208→
   209→   209→    _onForegroundNotification = onForegroundNotification;
   210→   210→    _onNotificationTap = onNotificationTap;
   211→   211→
   212→   212→    try {
   213→   213→      // 1. Request browser notification permissions.
   214→   214→      final permissionGranted = await _requestPermissions();
   215→   215→      if (!permissionGranted) {
   216→   216→        AppLogger.warning('Notification permissions not granted.');
   217→   217→      }
   218→   218→
   219→   219→      // 2. Register device token.
   220→   220→      await _registerDeviceToken();
   221→   221→
   222→   222→      // 3. Subscribe to user's personal notification channel.
   223→   223→      await _subscribeToUserChannel();
   224→   224→
   225→   225→      // 4. Load initial unread count.
   226→   226→      await _loadUnreadCount();
   227→   227→
   228→   228→      _initialized = true;
   229→   229→      AppLogger.info('NotificationService initialized successfully');
   230→   230→    } catch (e) {
   231→   231→      AppLogger.error('Failed to initialize NotificationService', error: e);
   232→   232→      // Mark initialized anyway so the app can continue — push
   233→   233→      // notifications are a non-critical feature.
   234→   234→      _initialized = true;
   235→   235→    }
   236→   236→  }

   245→   245→  Future<bool> _requestPermissions() async {

   254→   254→      // Native platforms: notification permissions are handled by the OS.
   255→   255→      // For now, we return true and rely on Supabase Realtime for

   282→   282→  /// Registers a device token with Supabase for push notifications.
   283→   283→  ///
   284→   284→  /// On web, this creates a browser push subscription.
   285→   285→  /// On native, this would use the platform's push notification service.
   286→   286→  Future<void> _registerDeviceToken() async {
   287→   287→    try {
   288→   288→      final userId = _supabase.auth.currentUser?.id;
   289→   289→      if (userId == null) {
   290→   290→        AppLogger.warning('Cannot register device token — no authenticated user.');
   291→   291→        return;
   292→   292→      }
   293→   293→
   294→   294→      // Generate a unique token for this device/session.
   295→   295→      // On web, this is a browser-generated push subscription endpoint.
   296→   296→      // For now, we use a deterministic token based on user + session.
   297→   297→      _currentToken = 'web_${userId}_${DateTime.now().millisecondsSinceEpoch}';
   298→   298→
   299→   299→      await _persistTokenLocally(_currentToken!);
   300→   300→      await _syncTokenToSupabase(_currentToken!);
   301→   301→
   302→   302→      AppLogger.info('Device token registered: [REDACTED]');
   303→   303→    } catch (e) {
   304→   304→      AppLogger.error('Failed to register device token', error: e);
   305→   305→    }
   306→   306→  }

   320→   320→  // ─── Realtime Channel Subscriptions ─────────────────────────────
   321→   321→
   322→   322→  /// Subscribes to the user's personal notification channel.
   323→   323→  ///
   324→   324→  /// This channel receives notifications targeted at the specific user
   325→   325→  /// (e.g., "Your exam has been graded", "Payment received").
   326→   326→  Future<void> _subscribeToUserChannel() async {
   327→   327→    final userId = _supabase.auth.currentUser?.id;
   328→   328→    if (userId == null) return;
   329→   329→
   330→   330→    try {
   331→   331→      _notificationChannel = _supabase.channel(
   332→   332→        'notifications:$userId',
   333→   333→        opts: const sb.RealtimeChannelConfig(self: true),
   334→   334→      );
   335→   335→
   336→   336→      _notificationChannel!.onPostgresChanges(
   337→   337→        event: sb.PostgresChangeEvent.insert,
   338→   338→        schema: 'public',
   339→   339→        table: 'notifications',
   340→   340→        filter: sb.PostgresChangeFilter(
   341→   341→          type: sb.PostgresChangeFilterType.eq,
   342→   342→          column: 'user_id',
   343→   343→          value: userId,
   344→   344→        ),
   345→   345→        callback: (payload) {
   346→   346→          _handleIncomingNotification(payload.newRecord);
   347→   347→        },
   348→   348→      );
   349→   349→
   350→   350→      _notificationChannel!.subscribe((status, error) {
   351→   351→        if (status == sb.RealtimeSubscribeStatus.subscribed) {
   352→   352→          AppLogger.info('User notification channel subscribed');
   353→   353→        } else if (status == sb.RealtimeSubscribeStatus.channelError) {
   354→   354→          AppLogger.error('User notification channel error', error: error);
   355→   355→        }
   356→   356→      });
   357→   357→
   358→   358→      _subscribedChannelNames.add('notifications:$userId');
   359→   359→    } catch (e) {
   360→   360→      AppLogger.error('Failed to subscribe to user notification channel', error: e);
   361→   361→    }
   362→   362→  }

   493→   493→  /// Handles an incoming notification from Supabase Realtime.
   494→   494→  void _handleIncomingNotification(Map<String, dynamic> record) {
   495→   495→    try {
   496→   496→      final notification = ExamForgeNotification.fromMap(record);
   497→   497→
   498→   498→      AppLogger.info('Notification received: ${notification.id} — ${notification.title}');
   499→   499→
   500→   500→      // Emit to the stream for reactive UI updates.
   501→   501→      if (!_notificationController.isClosed) {
   502→   502→        _notificationController.add(notification);
   503→   503→      }
   504→   504→
   505→   505→      // Invoke the foreground callback.
   506→   506→      _onForegroundNotification?.call(notification);
   507→   507→

   595→   595→  /// Gets notifications for the current user, optionally filtered.
   596→   596→  Future<List<ExamForgeNotification>> getNotifications({
   597→   597→    int limit = 50,
   598→   598→    int offset = 0,
   599→   599→    String? type,
   600→   600→    bool? isRead,
   601→   601→  }) async {
   602→   602→    try {
   603→   603→      final userId = _supabase.auth.currentUser?.id;
   604→   604→      if (userId == null) return [];
   605→   605→
   606→   606→      // Build the query with all filters applied before ordering/ranging.
   607→   607→      // Supabase query builder: .select() returns PostgrestFilterBuilder,
   608→   608→      // and filters must be applied before .order() and .range().
   609→   609→      var query = _supabase
   610→   610→          .from('notifications')
   611→   611→          .select()
   612→   612→          .eq('user_id', userId);
   613→   613→
   614→   614→      if (type != null) {
   615→   615→        query = query.eq('type', type);
   616→   616→      }
   617→   617→      if (isRead != null) {
   618→   618→        query = query.eq('is_read', isRead);
   619→   619→      }
   620→   620→
   621→   621→      final response = await query
   622→   622→          .order('created_at', ascending: false)
   623→   623→          .range(offset, offset + limit - 1);
   624→   624→      return response.map<ExamForgeNotification>((row) {
   625→   625→        return ExamForgeNotification.fromMap(row);
   626→   626→      }).toList();
   627→   627→    } catch (e) {
   628→   628→      AppLogger.error('Failed to get notifications', error: e);
   629→   629→      return [];
   630→   630→    }
   631→   631→  }

   635→   635→  /// Sends a broadcast notification to all users in a role.
   636→   636→  ///
   637→   637→  /// This inserts a notification for each user with the given role.
   638→   638→  /// The Supabase Realtime channel will then deliver it instantly.
   639→   639→  Future<void> broadcastToRole({
   640→   640→    required String role,
   641→   641→    required String title,
   642→   642→    required String body,
   643→   643→    String? type,
   644→   644→    Map<String, dynamic>? data,
   645→   645→  }) async {
   646→   646→    try {
   647→   647→      // Get all users with the given role.
   648→   648→      final users = await _supabase
   649→   649→          .from('user_profiles')
   650→   650→          .select('user_id')
   651→   651→          .eq('role', role);
   652→   652→
   653→   653→      final notifications = users.map<Map<String, dynamic>>((user) {
   654→   654→        return {
   655→   655→          'user_id': user['user_id'],
   656→   656→          'title': title,
   657→   657→          'message': body,
   658→   658→          'type': type ?? 'broadcast',
   659→   659→          'data': data,
   660→   660→        };
   661→   661→      }).toList();
   662→   662→
   663→   663→      if (notifications.isNotEmpty) {
   664→   664→        // Insert in batches to avoid payload limits.
   665→   665→        const batchSize = 100;
   666→   666→        for (var i = 0; i < notifications.length; i += batchSize) {
   667→   667→          final batch = notifications.sublist(
   668→   668→            i,
   669→   669→            i + batchSize > notifications.length
   670→   670→                ? notifications.length
   671→   671→                : i + batchSize,
   672→   672→          );
   673→   673→          await _supabase.from('notifications').insert(batch);
   674→   674→        }
   675→   675→      }
   676→   676→
   677→   677→      AppLogger.info('Broadcast notification sent to ${users.length} users with role: $role');
   678→   678→    } catch (e) {
   679→   679→      AppLogger.error('Failed to broadcast to role: $role', error: e);
   680→   680→    }
   681→   681→  }

   782→   782→  /// Persists the device token to the `device_tokens` table in Supabase.
   783→   783→  Future<void> _syncTokenToSupabase(String token) async {
   784→   784→    try {
   785→   785→      final userId = _supabase.auth.currentUser?.id;
   786→   786→      if (userId == null) {
   787→   787→        AppLogger.warning('Cannot sync device token — no authenticated user.');
   788→   788→        return;
   789→   789→      }
   790→   790→
   791→   791→      await _supabase.from('device_tokens').upsert(
   792→   792→        {
   793→   793→          'user_id': userId,
   794→   794→          'fcm_token': token,
   795→   795→          'platform': kIsWeb ? 'web' : 'native',
   796→   796→          'updated_at': DateTime.now().toIso8601String(),
   797→   797→        },
   798→   798→        onConflict: 'user_id,fcm_token',
   799→   799→      );
   800→   800→
   801→   801→      AppLogger.info('Device token synced to Supabase');
   802→   802→    } on sb.PostgrestException catch (e) {
   803→   803→      AppLogger.error('Supabase device token sync failed', error: e);
   804→   804→    } catch (e) {
   805→   805→      AppLogger.error('Unexpected device token sync error', error: e);
   806→   806→    }
   807→   807→  }
