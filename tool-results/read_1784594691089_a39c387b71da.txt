   104→  Future<void> _saveSettings() async {
   105→    setState(() => _isSaving = true);
   106→
   107→    final detailState = ref.read(schoolDetailProvider);
   108→    final school = detailState.school;
   109→
   110→    if (school == null) {
   111→      setState(() => _isSaving = false);
   112→      return;
   113→    }
   114→
   115→    final updatedSchool = school.copyWith(
   116→      primaryColor: _primaryColorCtrl.text.trim(),
   117→      secondaryColor: _secondaryColorCtrl.text.trim(),
   118→      motto: _mottoCtrl.text.trim().isEmpty ? null : _mottoCtrl.text.trim(),
   119→      logoUrl: _logoUrlCtrl.text.trim().isEmpty ? null : _logoUrlCtrl.text.trim(),
   120→      maxStudents: int.tryParse(_maxStudentsCtrl.text.trim()) ?? school.maxStudents,
   121→      maxTeachers: int.tryParse(_maxTeachersCtrl.text.trim()) ?? school.maxTeachers,
   122→      settings: {
   123→        ...school.settings,
   124→        'enable_notifications': _enableNotifications,
   125→        'enable_attendance': _enableAttendance,
   126→        'enable_homework': _enableHomework,
   127→        'enable_announcements': _enableAnnouncements,
   128→        'enable_document_upload': _enableDocumentUpload,
   129→        'enable_parent_portal': _enableParentPortal,
   130→        'enable_student_portal': _enableStudentPortal,
   131→        'maintenance_mode': _maintenanceMode,
   132→      },
   133→    );
   134→
   135→    await ref.read(schoolDetailProvider.notifier).updateSchool(updatedSchool);
   136→
   137→    if (mounted) {
   138→      setState(() => _isSaving = false);
   139→      ScaffoldMessenger.of(context).showSnackBar(
   140→        const SnackBar(
   141→          content: Text('Settings saved successfully'),
   142→          backgroundColor: AppColors.success,
   143→        ),
   144→      );
   145→    }
   146→  }

   600→  Widget _buildSubscriptionInfo(BuildContext context, SchoolEntity school) {
   601→    final cs = context.colorScheme;
   602→    final tt = context.textTheme;
   603→    final isDark = context.isDarkMode;
   604→
   605→    final planColor = _subscriptionColor(school.subscriptionStatus);
   606→    final planLabel = _subscriptionLabel(school.subscriptionStatus);
   607→    final hasExpiry = school.subscriptionExpiresAt != null;
   608→    final isExpiringSoon = hasExpiry &&
   609→        school.subscriptionExpiresAt!.difference(DateTime.now()).inDays < 30;
   610→
   611→    return Column(
   612→      children: [
   613→        Row(
   614→          children: [
   615→            Container(
   616→              padding: const EdgeInsets.all(Spacings.md),
   617→              decoration: BoxDecoration(

   831→
