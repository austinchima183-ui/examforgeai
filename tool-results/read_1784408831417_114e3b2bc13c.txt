   132→/// Provides the Supabase client singleton.

   136→  try {

   139→    AppLogger.error('Failed to resolve Supabase client', error: e);
   140→    rethrow;
   141→  }

   144→/// Provides a configured [Dio] instance with base options pulled from
   145→/// [AppConfig] and [ApiConstants].
   146→final dioProvider = Provider<Dio>((ref) {
   147→  final dio = Dio(
   148→    BaseOptions(
   149→      baseUrl: AppConfig.baseUrl,
   150→      connectTimeout: AppConfig.connectTimeoutDuration,
   151→      receiveTimeout: AppConfig.receiveTimeoutDuration,
   152→      sendTimeout: AppConfig.sendTimeoutDuration,
   153→      headers: ApiConstants.baseHeaders(),
   154→      responseType: ResponseType.json,
   155→      validateStatus: (status) => status != null && status < 500,
   156→    ),
   157→  );
   158→  return dio;
   159→});

   161→/// Provides the [ApiClient] wrapper around [Dio].

   215→/// Stream provider that emits the current Supabase [sb.AuthState].

   217→/// Automatically reconnects on transient failures and emits the latest
   218→/// state to any listener.
   219→final authStateProvider = StreamProvider<sb.AuthState>((ref) {
   220→  final supabaseClient = ref.watch(supabaseClientProvider);
   221→  return supabaseClient.auth.onAuthStateChange;

   223→
   224→/// Convenience provider that yields the currently authenticated [sb.User],
   225→/// or `null` if not signed in.
   226→final currentUserProvider = Provider<sb.User?>((ref) {
   227→  final authState = ref.watch(authStateProvider);
   228→  return authState.when(
   229→    data: (state) => state.session?.user,
   230→    loading: () => SupabaseConfig.currentUser,
   231→    error: (_, __) => null,
   232→  );
   233→});

   235→/// Convenience provider that yields `true` when the user is authenticated.
   236→final isAuthenticatedProvider = Provider<bool>((ref) {
   237→  final user = ref.watch(currentUserProvider);
   238→  return user != null;
