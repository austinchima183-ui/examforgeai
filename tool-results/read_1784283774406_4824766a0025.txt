     1→/// Report Generator — pure computation service for structured report outputs.
     2→///
     3→/// Transforms domain entities (results, exam, and CBT data) into structured
     4→/// [ReportData] objects ready for export to PDF, Excel, or CSV by the
     5→/// datasource layer (Supabase Edge Functions or local generators).
     6→///
     7→/// ## Design Principles
     8→///
     9→/// - **Pure transformation**: Every public method accepts entities and returns
    10→///   a [ReportData] with no I/O, no database access, no side effects.
    11→/// - **Self-contained DTOs**: Output content types are defined in this file to
    12→///   avoid coupling to persistence-layer models.
    13→/// - **Null-safe**: All methods handle missing or partial data gracefully,
    14→///   providing sensible defaults rather than throwing.
    15→/// - **Format-agnostic**: This service produces structured data only; the
    16→///   actual file generation is handled downstream.
    17→///
    18→/// ## Usage
    19→///
    20→/// ```dart
    21→/// final generator = ReportGenerator();
    22→/// final report = generator.generateStudentReport(
    23→///   overallResult: studentOverall,
    24→///   subjectResults: subjectList,
    25→///   studentName: 'John Doe',
    26→///   className: 'SS2A',
    27→///   sessionName: '2025/2026',
    28→///   trendData: trendPoints,
    29→/// );
    30→/// // Pass report to datasource layer for PDF/Excel/CSV generation
    31→/// ```

   573→// REPORT GENERATOR SERVICE
   574→// ═══════════════════════════════════════════════════════════════════════
   575→
   576→/// Generates structured report data from domain entities.
   577→///
   578→/// This service is a pure computation layer — it transforms raw entity data
   579→/// into [ReportData] structures that the datasource layer can then render
   580→/// into PDF, Excel, or CSV files. No file I/O is performed here.
   581→class ReportGenerator {
   582→  // ── Student Report ─────────────────────────────────────────────────
   583→
   584→  /// Creates a comprehensive report for a single student.
   585→  ///
   586→  /// Requires the student's [overallResult] entity, their [subjectResults],
   587→  /// and contextual information like [studentName], [className], and
   588→  /// [sessionName]. Optional [trendData] provides historical performance
   589→  /// for chart rendering.
   590→  ///
   591→  /// The optional [gradeScale] is used to derive the overall grade from the
   592→  /// overall percentage if [StudentOverallResultEntity.overallGrade] is null.
   593→  ReportData generateStudentReport({
   594→    required StudentOverallResultEntity overallResult,
   595→    required List<StudentSubjectResultEntity> subjectResults,
   596→    required String studentName,
   597→    required String className,
   598→    required String sessionName,
   599→    List<TrendDataPoint>? trendData,
   600→    GradeScaleEntity? gradeScale,
   601→  }) {
   602→    // Resolve the overall grade — use the entity's grade first, then
   603→    // fall back to the grade scale if available.
   604→    final overallGrade = overallResult.overallGrade ??
   605→        gradeScale?.applyToPercentage(overallResult.overallPercentage)?.grade ??
   606→        _defaultGrade(overallResult.overallPercentage);
   607→
   608→    // Build subject-by-subject breakdown rows.
   609→    final subjectRows = subjectResults.map((sr) {
   610→      final grade = sr.grade ??
   611→          gradeScale?.applyToPercentage(sr.percentage)?.grade ??
   612→          _defaultGrade(sr.percentage);
   613→      return SubjectResultRow(
   614→        subjectName: sr.subjectId, // Caller should map ID → name upstream
   615→        score: sr.totalMarksObtained,
   616→        percentage: sr.percentage,
   617→        grade: grade,
   618→        classAverage: sr.classAverage,
   619→        position: sr.classPosition,
   620→        trend: sr.performanceTrend.label,
   621→      );
   622→    }).toList();
   623→
   624→    // Collect strengths and weaknesses from subject results.
   625→    final strengths = <String>[
   626→      ...overallResult.metadata['strengths'] as List<dynamic>? ?? [],
   627→      ...subjectResults
   628→          .where((sr) => sr.performanceTrend == PerformanceTrend.improving)
   629→          .map((sr) => 'Strong performance in ${sr.subjectId}'),
   630→    ];
   631→
   632→    final weaknesses = <String>[
   633→      ...overallResult.metadata['weaknesses'] as List<dynamic>? ?? [],
   634→      ...subjectResults
   635→          .where((sr) => sr.performanceTrend == PerformanceTrend.declining)
   636→          .map((sr) => 'Needs improvement in ${sr.subjectId}'),
   637→    ];
   638→
   639→    // AI study recommendations from the overall result entity.
   640→    final recommendations = overallResult.aiStudyRecommendations;
   641→
   642→    // Build trend data from supplied points, or default to a single point.
   643→    final trend = trendData ??
   644→        [
   645→          TrendDataPoint(
   646→            period: sessionName,
   647→            score: overallResult.overallPercentage,
   648→          ),
   649→        ];
   650→
   651→    final content = StudentReportContent(
   652→      studentName: studentName,
   653→      className: className,
   654→      sessionName: sessionName,
   655→      overallScore: overallResult.overallPercentage,
   656→      overallGrade: overallGrade,
   657→      overallGpa: overallResult.overallGpa,
   658→      classPosition: overallResult.classPosition,
   659→      classSize: overallResult.classSize,
   660→      subjectResults: subjectRows,
   661→      strengths: strengths,
   662→      weaknesses: weaknesses,
   663→      recommendations: recommendations,
   664→      teacherComment: overallResult.teacherComment,
   665→      trendData: trend,
   666→    );
   667→
   668→    return ReportData(
   669→      type: ReportType.student,
   670→      title: 'Student Report — $studentName',
   671→      content: content.toJson(),

   678→  /// Creates a class-wide performance report.
   679→  ///
   680→  /// Accepts a [ClassPerformanceEntity] for the class statistics,

   682→  /// [subjectResults] for subject performance comparisons.

   700→      final sr = entry.value;

   702→          gradeScale?.applyToPercentage(sr.overallPercentage)?.grade ??

   718→      final bySubject = <String, List<StudentSubjectResultEntity>>{};

   720→        bySubject.putIfAbsent(sr.subjectId, () => []).add(sr);

   729→          subjectName: subjectNameMap?[entry.key] ?? entry.key,
   730→          classAverage: avg,
   731→          highestScore: scores.last,
   732→          lowestScore: scores.first,

   734→              ? (passed / scores.length) * 100
   735→              : 0,

  1072→  /// Formats a position as an ordinal string (e.g., `1st`, `2nd`, `3rd`,
  1073→  /// `4th`, `11th`, `21st`).
  1074→  ///
  1075→  /// Returns `N/A` if [position] is null.
  1076→  static String formatPosition(int? position) {
  1077→    if (position == null) return 'N/A';
  1078→    if (position < 1) return position.toString();
  1079→
  1080→    final lastTwo = position % 100;
  1081→    if (lastTwo >= 11 && lastTwo <= 13) return '${position}th';
  1082→
  1083→    switch (position % 10) {
  1084→      case 1:
  1085→        return '${position}st';
  1086→      case 2:
  1087→        return '${position}nd';
  1088→      case 3:
  1089→        return '${position}rd';
  1090→      default:
  1091→        return '${position}th';
  1092→    }
  1093→  }

  1095→  /// Formats a duration in seconds to a human-readable string.
  1096→  ///
  1097→  /// Examples:
  1098→  /// - `3612` → `"1h 0m"`
  1099→  /// - `2712` → `"45m 12s"`
  1100→  /// - `60` → `"1m 0s"`
  1101→  /// - `30` → `"0m 30s"`
  1102→  static String formatDuration(int seconds) {
  1103→    if (seconds < 0) seconds = 0;
  1104→    final hours = seconds ~/ 3600;
  1105→    final remainingSeconds = seconds % 3600;
  1106→    final minutes = remainingSeconds ~/ 60;
  1107→    final secs = remainingSeconds % 60;
  1108→
  1109→    if (hours > 0) {
  1110→      return '${hours}h ${minutes}m';
  1111→    }
  1112→    return '${minutes}m $secs\ s';
  1113→  }

  1121→    return '${months[date.month]} ${date.day}, ${date.year}';
  1122→  }

  1147→  /// Computes score distribution buckets from a list of subject results.
  1148→  ///
  1149→  /// Standard ranges: 0–19, 20–39, 40–49, 50–59, 60–69, 70–79, 80–89,
  1150→  /// 90–100.
  1151→  static List<ScoreBucket> _computeScoreBuckets(
  1152→    List<StudentSubjectResultEntity> results,
  1153→  ) {
  1154→    const ranges = [
  1155→      (label: '0–19', min: 0.0, max: 19.99),
  1156→      (label: '20–39', min: 20.0, max: 39.99),
  1157→      (label: '40–49', min: 40.0, max: 49.99),
  1158→      (label: '50–59', min: 50.0, max: 59.99),
  1159→      (label: '60–69', min: 60.0, max: 69.99),
  1160→      (label: '70–79', min: 70.0, max: 79.99),
  1161→      (label: '80–89', min: 80.0, max: 89.99),
  1162→      (label: '90–100', min: 90.0, max: 100.0),
  1163→    ];
  1164→
  1165→    final total = results.length.toDouble();
  1166→    return ranges.map((range) {
  1167→      final count = results
  1168→          .where((r) => r.percentage >= range.min && r.percentage <= range.max)
  1169→          .length;
  1170→      return ScoreBucket(
  1171→        range: range.label,
  1172→        count: count,
  1173→        percentage: total > 0 ? (count / total) * 100 : 0,
  1174→      );
  1175→    }).toList();
  1176→  }

  1178→  /// Determines the most common mastery level among a list of topic
  1179→  /// mastery entries.
  1180→  static String _dominantMasteryLevel(List<TopicMasteryEntity> items) {
  1181→    if (items.isEmpty) return MasteryLevel.notStarted.label;
  1182→    final counts = <MasteryLevel, int>{};
  1183→    for (final item in items) {
  1184→      counts[item.masteryLevel] = (counts[item.masteryLevel] ?? 0) + 1;
  1185→    }
  1186→    final dominant = counts.entries.reduce(
  1187→      (a, b) => a.value >= b.value ? a : b,
  1188→    );
  1189→    return dominant.key.label;
  1190→  }
