    30→/// Full-featured question editor supporting create and edit modes.
    31→///
    32→/// The form is divided into sections:
    33→/// 1. Question Type selection (grid of selectable type cards)
    34→/// 2. Question Content (multiline text field)
    35→/// 3. Answer Configuration (dynamic based on selected type)
    36→/// 4. Classification (subject, topic, subtopic, class, category)
    37→/// 5. Settings (difficulty, exam type, marks, negative marks, time)
    38→/// 6. Explanation & Notes
    39→/// 7. Attachments
    40→/// 8. Tags
    41→///
    42→/// Supports:
    43→/// - Preview mode toggle
    44→/// - Save Draft / Publish actions
    45→/// - Form validation
    46→/// - Unsaved changes dialog on back
    47→/// - Loading state while saving
    48→class QuestionEditorPage extends ConsumerStatefulWidget {
    49→  const QuestionEditorPage({
    50→    super.key,
    51→    this.questionId,
    52→  });
    53→
    54→  /// If provided, the editor loads this question for editing.
    55→  final String? questionId;
    56→
    57→  @override
    58→  ConsumerState<QuestionEditorPage> createState() =>
    59→      _QuestionEditorPageState();
    60→}
    61→
    62→class _QuestionEditorPageState extends ConsumerState<QuestionEditorPage> {
    63→  final _formKey = GlobalKey<FormState>();
    64→  final _contentController = TextEditingController();
    65→  final _explanationController = TextEditingController();
    66→  final _teacherNotesController = TextEditingController();
    67→  final _referenceMaterialsController = TextEditingController();
    68→  final _marksController = TextEditingController();
    69→  final _negativeMarksController = TextEditingController();
    70→  final _timeAllowedController = TextEditingController();
    71→  final _tagSearchController = TextEditingController();
    72→
    73→  bool _hasInitialized = false;
    74→  bool _hasUnsavedChanges = false;
    75→
    76→  @override
    77→  void initState() {
    78→    super.initState();
    79→    WidgetsBinding.instance.addPostFrameCallback((_) {
    80→      _initializeEditor();
    81→    });
    82→  }
    83→
    84→  @override
    85→  void dispose() {
    86→    _contentController.dispose();
    87→    _explanationController.dispose();
    88→    _teacherNotesController.dispose();
    89→    _referenceMaterialsController.dispose();
    90→    _marksController.dispose();
    91→    _negativeMarksController.dispose();
    92→    _timeAllowedController.dispose();
    93→    _tagSearchController.dispose();
    94→    super.dispose();
    95→  }
    96→
    97→  Future<void> _initializeEditor() async {
    98→    if (widget.questionId != null) {
    99→      await ref
   100→          .read(questionEditorProvider.notifier)
   101→          .loadQuestionForEdit(widget.questionId!);
   102→    }
   103→    _syncControllersFromState();
   104→    _hasInitialized = true;
   105→  }
   106→
   107→  void _syncControllersFromState() {
   108→    final state = ref.read(questionEditorProvider);
   109→    _contentController.text = state.content;
   110→    _explanationController.text = state.explanation;
   111→    _teacherNotesController.text = state.teacherNotes;
   112→    _referenceMaterialsController.text = state.referenceMaterials;
   113→    _marksController.text = state.marks.toString();
   114→    _negativeMarksController.text = state.negativeMarks.toString();
   115→    if (state.timeAllowedSeconds != null) {
   116→      _timeAllowedController.text = (state.timeAllowedSeconds! ~/ 60).toString();
   117→    }
   118→  }
   119→
   120→  void _markUnsaved() {
   121→    if (!_hasUnsavedChanges) {
   122→      setState(() => _hasUnsavedChanges = true);
   123→    }
   124→  }
   125→
   126→  // ─── Build ──────────────────────────────────────────────────────────
   127→
   128→  @override
   129→  Widget build(BuildContext context) {
   130→    final editorState = ref.watch(questionEditorProvider);
   131→    final cs = context.colorScheme;
   132→    final tt = context.textTheme;
   133→
   134→    return PopScope(
   135→      canPop: !_hasUnsavedChanges,
   136→      onPopInvokedWithResult: (didPop, _) async {
   137→        if (didPop) return;
   138→        if (_hasUnsavedChanges) {
   139→          final shouldLeave = await _showUnsavedChangesDialog();
   140→          if (shouldLeave == true && mounted) {
   141→            context.pop();
   142→          }
   143→        }
   144→      },
   145→      child: Scaffold(
   146→        appBar: AppAppBar(
   147→          title: editorState.isEditMode ? 'Edit Question' : 'Create Question',
   148→          actions: [
   149→            // Preview toggle
   150→            AppIconButton(
   151→              icon: editorState.isPreviewMode
   152→                  ? Icons.edit_rounded
   153→                  : Icons.visibility_rounded,
   154→              onPressed: () {
   155→                ref.read(questionEditorProvider.notifier).togglePreviewMode();
   156→              },
   157→              tooltip: editorState.isPreviewMode ? 'Edit' : 'Preview',
   158→            ),
   159→            // Save Draft
   160→            if (!editorState.isPreviewMode)
   161→              AppButton(
   162→                label: 'Save Draft',
   163→                onPressed: editorState.isSaving ? null : _saveDraft,
   164→                variant: AppButtonVariant.outlined,
   165→                size: AppButtonSize.small,
   166→                isLoading: editorState.isSaving,
   167→              ),
   168→            const SizedBox(width: Spacings.sm),
   169→            // Publish
   170→            AppButton(
   171→              label: 'Publish',
   172→              onPressed: editorState.isSaving ? null : _publish,
   173→              variant: AppButtonVariant.elevated,
   174→              size: AppButtonSize.small,
   175→              isLoading: editorState.isSaving,
   176→            ),
   177→            const SizedBox(width: Spacings.sm),
   178→          ],
   179→        ),
   180→        body: editorState.isLoading
   181→            ? const Center(
   182→                child: AppLoadingSpinner(size: AppLoadingSpinnerSize.large),
   183→              )
   184→            : editorState.isPreviewMode
   185→                ? _buildPreviewMode(editorState)

   187→      ),
   188→    );
   189→  }

   334→  // ─── 1. Question Type Grid ──────────────────────────────────────────
   335→
   336→  Widget _buildQuestionTypeGrid(QuestionEditorState editorState) {
   337→    final cs = context.colorScheme;
   338→    final isDesktop = context.isDesktop;
   339→    final crossAxisCount = isDesktop ? 4 : (context.isTablet ? 3 : 2);
   340→
   341→    return GridView.count(
   342→      shrinkWrap: true,
   343→      physics: const NeverScrollableScrollPhysics(),
   344→      crossAxisCount: crossAxisCount,
   345→      childAspectRatio: 2.8,
   346→      crossAxisSpacing: Spacings.sm,
   347→      mainAxisSpacing: Spacings.sm,
   348→      children: QuestionType.values.map((type) {
   349→        final isSelected = editorState.selectedQuestionType == type;
   350→        return InkWell(
   351→          onTap: () {
   352→            ref.read(questionEditorProvider.notifier).setQuestionType(type);
   353→            _markUnsaved();
   354→          },
   355→          borderRadius: BorderRadius.circular(Spacings.mdRadius),
   356→          child: Container(
   357→            padding: const EdgeInsets.symmetric(
   358→              horizontal: Spacings.md,
   359→              vertical: Spacings.sm,
   360→            ),
   361→            decoration: BoxDecoration(
   362→              color: isSelected
   363→                  ? cs.primary.withValues(alpha: context.isDarkMode ? 0.20 : 0.08)
   364→                  : cs.surfaceContainerLow,
   365→              borderRadius: BorderRadius.circular(Spacings.mdRadius),
   366→              border: Border.all(
   367→                color: isSelected ? cs.primary : cs.outlineVariant.withValues(alpha: 0.5),
   368→                width: isSelected ? 2.0 : 1.0,
   369→              ),
   370→            ),
   371→            child: Row(
   372→              children: [
   373→                QuestionTypeBadge(
   374→                  type: type,
   375→                  variant: QuestionTypeBadgeVariant.iconOnly,
   376→                  size: QuestionTypeBadgeSize.small,
   377→                ),
   378→                const SizedBox(width: Spacings.sm),
   379→                Flexible(
   380→                  child: Text(
   381→                    type.label,
   382→                    style: context.textTheme.bodySmall?.copyWith(
   383→                      color: isSelected ? cs.primary : cs.onSurface,
   384→                      fontWeight: isSelected
   385→                          ? AppTypography.wSemiBold
   386→                          : null,
   387→                    ),
   388→                    maxLines: 1,
   389→                    overflow: TextOverflow.ellipsis,
   390→                  ),
   391→                ),
   392→              ],
   393→            ),
   394→          ),
   395→        );
   396→      }).toList(),
   397→    );
   398→  }

   400→  // ─── 3. Answer Configuration ────────────────────────────────────────
   401→
   402→  Widget _buildAnswerConfiguration(QuestionEditorState editorState) {
   403→    final type = editorState.selectedQuestionType;
   404→    final cs = context.colorScheme;
   405→
   406→    switch (type) {
   407→      case QuestionType.multipleChoice:
   408→      case QuestionType.multipleResponse:
   409→      case QuestionType.imageBased:
   410→      case QuestionType.audioBased:
   411→      case QuestionType.videoBased:
   412→        return AnswerOptionsEditor(
   413→          options: editorState.answerOptions,
   414→          isMultipleChoice: type == QuestionType.multipleChoice,
   415→          onAddOption: () {
   416→            ref.read(questionEditorProvider.notifier).addAnswerOption();
   417→            _markUnsaved();
   418→          },
   419→          onRemoveOption: (index) {
   420→            ref.read(questionEditorProvider.notifier).removeAnswerOption(index);
   421→            _markUnsaved();
   422→          },
   423→          onUpdateOption: (index, option) {
   424→            ref.read(questionEditorProvider.notifier).updateAnswerOption(index, option);
   425→            _markUnsaved();
   426→          },
   427→          onSetCorrectAnswer: (index) {
   428→            ref.read(questionEditorProvider.notifier).setCorrectAnswer(index);
   429→            _markUnsaved();
   430→          },
   431→          onSetCorrectAnswers: (indices) {
   432→            ref.read(questionEditorProvider.notifier).setCorrectAnswers(indices);
   433→            _markUnsaved();
   434→          },
   435→        );
   436→
   437→      case QuestionType.trueFalse:
   438→        return _buildTrueFalseEditor(editorState);
   439→
   440→      case QuestionType.matching:
   441→        return MatchingPairsEditor(
   442→          pairs: editorState.matchingPairs,
   443→          onAddPair: () {
   444→            ref.read(questionEditorProvider.notifier).addMatchingPair();
   445→            _markUnsaved();
   446→          },
   447→          onRemovePair: (index) {
   448→            ref.read(questionEditorProvider.notifier).removeMatchingPair(index);
   449→            _markUnsaved();
   450→          },
   451→          onUpdatePair: (index, pair) {
   452→            ref.read(questionEditorProvider.notifier).updateMatchingPair(index, pair);
   453→            _markUnsaved();
   454→          },
   455→        );
   456→
   457→      case QuestionType.ordering:
   458→        return OrderingItemsEditor(
   459→          items: editorState.orderingItems,
   460→          onAddItem: () {
   461→            ref.read(questionEditorProvider.notifier).addOrderingItem();
   462→            _markUnsaved();
   463→          },
   464→          onRemoveItem: (index) {
   465→            ref.read(questionEditorProvider.notifier).removeOrderingItem(index);
   466→            _markUnsaved();
   467→          },
   468→          onUpdateItem: (index, item) {
   469→            ref.read(questionEditorProvider.notifier).updateOrderingItem(index, item);
   470→            _markUnsaved();
   471→          },
   472→          onReorder: (oldIndex, newIndex) {
   473→            ref.read(questionEditorProvider.notifier).reorderOrderingItems(oldIndex, newIndex);
   474→            _markUnsaved();
   475→          },
   476→        );
   477→
   478→      case QuestionType.fillInBlank:
   479→        return FillInBlankEditor(
   480→          answers: editorState.fillInBlankAnswers,
   481→          onAddBlank: () {
   482→            ref.read(questionEditorProvider.notifier).addFillInBlankAnswer();
   483→            _markUnsaved();
   484→          },
   485→          onRemoveBlank: (index) {
   486→            ref.read(questionEditorProvider.notifier).removeFillInBlankAnswer(index);
   487→            _markUnsaved();
   488→          },
   489→          onUpdateBlank: (index, answer) {
   490→            ref.read(questionEditorProvider.notifier).updateFillInBlankAnswer(index, answer);
   491→            _markUnsaved();
   492→          },
   493→        );
   494→
   495→      case QuestionType.shortAnswer:
   496→      case QuestionType.essay:
   497→      case QuestionType.numerical:
   498→      case QuestionType.practical:
   499→      case QuestionType.caseStudy:
   500→        return AppTextField(
   501→          label: 'Model Answer',
   502→          hint: 'Enter the model answer or expected response…',
   503→          controller: _explanationController, // Reuse for model answer
   504→          maxLines: 8,
   505→          minLines: 3,
   506→          onChanged: (v) {
   507→            _markUnsaved();
   508→          },
   509→        );
   510→    }
   511→  }

   513→  // ─── True/False Editor ──────────────────────────────────────────────
   514→
   515→  Widget _buildTrueFalseEditor(QuestionEditorState editorState) {
   516→    final cs = context.colorScheme;
   517→    final tt = context.textTheme;
   518→
   519→    // Check if "True" is the correct answer
   520→    final isTrueCorrect = editorState.answerOptions.any(
   521→      (o) => o.isCorrect && o.content.toLowerCase() == 'true',
   522→    );
   523→
   524→    return Column(
   525→      crossAxisAlignment: CrossAxisAlignment.start,
   526→      children: [
   527→        Text(
   528→          'Select the correct answer',
   529→          style: tt.bodySmall?.copyWith(color: cs.onSurfaceVariant),
   530→        ),
   531→        const SizedBox(height: Spacings.md),
   532→        SegmentedButton<bool>(
   533→          segments: const [
   534→            ButtonSegment(value: true, label: Text('True'), icon: Icon(Icons.check_rounded)),
   535→            ButtonSegment(value: false, label: Text('False'), icon: Icon(Icons.close_rounded)),
   536→          ],
   537→          selected: {isTrueCorrect},
   538→          onSelectionChanged: (selection) {
   539→            final isTrue = selection.first;
   540→            // Update the state with True/False options
   541→            ref.read(questionEditorProvider.notifier).setCorrectAnswer(isTrue ? 0 : 1);
   542→            _markUnsaved();
   543→          },
   544→        ),
   545→      ],
   546→    );
   547→  }

   951→  Future<void> _saveDraft() async {
   952→    if (!_formKey.currentState!.validate()) return;
   953→

   955→    await notifier.saveAsDraft();

   960→  // ─── Publish ────────────────────────────────────────────────────────
   961→
   962→  Future<void> _publish() async {
   963→    if (!_formKey.currentState!.validate()) return;
   964→
   965→    final notifier = ref.read(questionEditorProvider.notifier);
   966→    await notifier.saveAndPublish();
   967→
   968→    _handleSaveResult();
   969→  }

   971→  void _handleSaveResult() {
   972→    final state = ref.read(questionEditorProvider);
   973→    if (state.error != null && mounted) {
   974→      AppDialog.showError(
   975→        context: context,
   976→        title: 'Save Failed',
   977→        message: state.error!,
   978→      );
   979→    } else if (state.successMessage != null && mounted) {
   980→      setState(() => _hasUnsavedChanges = false);
   981→      ScaffoldMessenger.of(context).showSnackBar(
   982→        SnackBar(
   983→          content: Text(state.successMessage!),
   984→          behavior: SnackBarBehavior.floating,
   985→        ),
   986→      );
   987→      if (mounted) context.pop();
   988→    }
   989→  }

  1071→  String _formatFileSize(int bytes) {
  1072→    if (bytes < 1024) return '$bytes B';
  1073→    if (bytes < 1024 * 1024) return '${(bytes / 1024).toStringAsFixed(1)} KB';
