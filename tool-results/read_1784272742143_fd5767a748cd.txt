   253→   253→-- 8. QUESTION_BANK TABLE (THE MAIN TABLE)
   254→   254→-- ============================================================================
   255→   255→-- Central repository for all questions. Designed for millions of rows.
   256→   256→-- - school_id NULL = platform-wide question (visible to all schools)
   257→   257→-- - parent_id = previous version (versioning chain)
   258→   258→-- - content_json = structured data for complex question types
   259→   259→-- ============================================================================
   260→   260→
   261→   261→CREATE TABLE IF NOT EXISTS question_bank (
   262→   262→  id                      UUID PRIMARY KEY DEFAULT gen_random_uuid(),
   263→   263→  school_id               UUID REFERENCES schools(id) ON DELETE CASCADE,  -- NULL = platform-wide
   264→   264→  subject_id              UUID NOT NULL REFERENCES subjects(id),
   265→   265→  topic_id                UUID REFERENCES topics(id) ON DELETE SET NULL,
   266→   266→  subtopic_id             UUID REFERENCES subtopics(id) ON DELETE SET NULL,
   267→   267→  class_id                UUID REFERENCES classes(id) ON DELETE SET NULL,
   268→   268→  category_id             UUID REFERENCES question_categories(id) ON DELETE SET NULL,
   269→   269→  curriculum_standard_id  UUID REFERENCES curriculum_standards(id) ON DELETE SET NULL,
   270→   270→  academic_session_id     UUID REFERENCES academic_sessions(id) ON DELETE SET NULL,
   271→   271→  question_type           question_type NOT NULL DEFAULT 'multiple_choice',
   272→   272→  difficulty              difficulty_level NOT NULL DEFAULT 'medium',
   273→   273→  exam_type               exam_type DEFAULT 'school_exam',
   274→   274→  content                 TEXT NOT NULL,                  -- Question text (HTML/LaTeX supported)
   275→   275→  content_json            JSONB DEFAULT '{}',             -- Structured content for complex types
   276→   276→  explanation             TEXT,                           -- Detailed answer explanation
   277→   277→  teacher_notes           TEXT,                           -- Private notes for teachers
   278→   278→  reference_materials     TEXT,                           -- Reference sources
   279→   279→  marks                   NUMERIC(5,2) NOT NULL DEFAULT 1.00,
   280→   280→  negative_marks          NUMERIC(5,2) DEFAULT 0.00,
   281→   281→  time_allowed_seconds    INTEGER,                        -- Suggested time to answer
   282→   282→  is_published            BOOLEAN DEFAULT false,
   283→   283→  is_archived             BOOLEAN DEFAULT false,
   284→   284→  is_featured             BOOLEAN DEFAULT false,
   285→   285→  version                 INTEGER DEFAULT 1,
   286→   286→  parent_id               UUID REFERENCES question_bank(id) ON DELETE SET NULL,  -- Version chain
   287→   287→  created_by              UUID NOT NULL REFERENCES users(id),
   288→   288→  updated_by              UUID REFERENCES users(id),
   289→   289→  usage_count             INTEGER DEFAULT 0,
   290→   290→  avg_score               NUMERIC(5,2) DEFAULT 0.00,
   291→   291→  metadata                JSONB DEFAULT '{}',             -- Extensible metadata
   292→   292→  created_at              TIMESTAMPTZ DEFAULT now(),
   293→   293→  updated_at              TIMESTAMPTZ DEFAULT now(),
   294→   294→
   295→   295→  -- Ensure non-negative marks
   296→   296→  CONSTRAINT question_bank_marks_positive CHECK (marks >= 0),
   297→   297→  CONSTRAINT question_bank_negative_marks_non_negative CHECK (negative_marks >= 0),
   298→   298→  CONSTRAINT question_bank_avg_score_range CHECK (avg_score >= 0 AND avg_score <= marks),
   299→   299→  -- A question cannot be both published and archived
   300→   300→  CONSTRAINT question_bank_not_published_and_archived CHECK (NOT (is_published = true AND is_archived = true))
   301→   301→);
   302→   302→
   303→   303→COMMENT ON TABLE question_bank IS 'Central question repository supporting millions of questions';
   304→   304→COMMENT ON COLUMN question_bank.school_id IS 'Owning school; NULL means platform-wide question visible to all';
   305→   305→COMMENT ON COLUMN question_bank.content IS 'The question text — supports HTML and LaTeX formatting';
   306→   306→COMMENT ON COLUMN question_bank.content_json IS 'Structured content payload for complex question types (matching, ordering, etc.)';
   307→   307→COMMENT ON COLUMN question_bank.parent_id IS 'Previous version of this question for versioning chain';
   308→   308→COMMENT ON COLUMN question_bank.metadata IS 'Extensible JSONB metadata for custom attributes and integrations';
