    13→/// A colored banner shown at the top of the screen when offline or limited.
    14→///
    15→/// - **Red**: Offline — no internet connection
    16→/// - **Orange**: Limited — poor connection quality
    17→///
    18→/// Tapping the banner can navigate to the connectivity details page.
    19→class ConnectivityBanner extends StatelessWidget {
    20→  const ConnectivityBanner({
    21→    super.key,
    22→    required this.connectivityInfo,
    23→    this.onTap,
    24→  });
    25→
    26→  final ConnectivityInfo connectivityInfo;
    27→  final VoidCallback? onTap;
    28→
    29→  /// Whether the banner should be shown.
    30→  bool get shouldShow =>
    31→      !connectivityInfo.isOnline ||
    32→      connectivityInfo.connectionQuality == ConnectionQuality.limited;
    33→
    34→  @override
    35→  Widget build(BuildContext context) {
    36→    if (!shouldShow) return const SizedBox.shrink();
    37→
    38→    final isOffline = !connectivityInfo.isOnline;
    39→    final backgroundColor = isOffline ? AppColors.error : AppColors.warning;
    40→    const textColor = Colors.white;
    41→    final icon = isOffline ? Icons.cloud_off : Icons.signal_cellular_alt;
    42→
    43→    return Material(
    44→      color: backgroundColor,
    45→      child: InkWell(
    46→        onTap: onTap,
    47→        child: SafeArea(
    48→          bottom: false,
    49→          child: Padding(
    50→            padding: const EdgeInsets.symmetric(
    51→              horizontal: Spacings.lg,
    52→              vertical: Spacings.sm,
    53→            ),
    54→            child: Row(
    55→              children: [
    56→                Icon(icon, size: Spacings.mdIcon, color: textColor),
    57→                const SizedBox(width: Spacings.sm),
    58→                Expanded(
    59→                  child: Text(
    60→                    isOffline
    61→                        ? 'You are offline. Some features may be limited.'
    62→                        : 'Limited connection. Syncing may be delayed.',
    63→                    style: AppTypography.buttonSmall.copyWith(color: textColor),
    64→                  ),
    65→                ),
    66→                Icon(
    67→                  Icons.chevron_right,
    68→                  size: Spacings.mdIcon,
    69→                  color: textColor.withValues(alpha: 0.7),
    70→                ),
    71→              ],
    72→            ),
    73→          ),
    74→        ),
    75→      ),
    76→    );
    77→  }
    78→}

   552→
   553→/// A progress bar showing local storage usage.
   554→///
   555→/// Displays a linear bar with a label underneath showing the used/total
   556→/// amount, and a percentage indicator.
   557→class StorageUsageBar extends StatelessWidget {
   558→  const StorageUsageBar({
   559→    super.key,
   560→    required this.usedBytes,
   561→    required this.totalBytes,
   562→    this.label,
   563→  });
   564→
   565→  final int usedBytes;
   566→  final int totalBytes;
   567→  final String? label;
   568→
   569→  @override
   570→  Widget build(BuildContext context) {
   571→    final cs = context.colorScheme;
   572→    final tt = context.textTheme;
   573→    final ratio = totalBytes > 0 ? usedBytes / totalBytes : 0.0;
   574→    final clampedRatio = ratio.clamp(0.0, 1.0);
   575→    final percent = (clampedRatio * 100).toStringAsFixed(0);
   576→
   577→    final barColor = ratio > 0.9
   578→        ? AppColors.error
   579→        : ratio > 0.7
   580→            ? AppColors.warning
   581→            : AppColors.success;
   582→
   583→    return Card(
   584→      elevation: Spacings.elevationSm,
   585→      shadowColor: cs.shadow.withValues(alpha: 0.08),
   586→      shape: RoundedRectangleBorder(
   587→        borderRadius: Spacings.borderRadiusLg,
   588→        side: BorderSide(
   589→          color: cs.outlineVariant.withValues(alpha: 0.5),
   590→        ),
   591→      ),
   592→      child: Padding(
   593→        padding: const EdgeInsets.all(Spacings.lg),
   594→        child: Column(
   595→          crossAxisAlignment: CrossAxisAlignment.start,
   596→          children: [
   597→            Row(
   598→              mainAxisAlignment: MainAxisAlignment.spaceBetween,
   599→              children: [
   600→                Text(
   601→                  label ?? '${_formatBytes(usedBytes)} / ${_formatBytes(totalBytes)}',
   602→                  style: tt.bodyMedium?.copyWith(
   603→                    fontWeight: AppTypography.wMedium,
   604→                    color: cs.onSurface,
   605→                  ),
   606→                ),
   607→                Text(
   608→                  '$percent%',
   609→                  style: tt.labelLarge?.copyWith(
   610→                    fontWeight: AppTypography.wSemiBold,
   611→                    color: barColor,
   612→                  ),
   613→                ),
   614→              ],
   615→            ),
   616→            const SizedBox(height: Spacings.sm),
   617→            ClipRRect(
   618→              borderRadius: Spacings.borderRadiusFull,
   619→              child: LinearProgressIndicator(
   620→                value: clampedRatio,
   621→                minHeight: 10,
   622→                backgroundColor: cs.surfaceContainerHighest,
   623→                valueColor: AlwaysStoppedAnimation<Color>(barColor),
   624→              ),
   625→            ),
   626→          ],
   627→        ),
   628→      ),
   629→    );
   630→  }
   631→
   632→  String _formatBytes(int bytes) {
   633→    if (bytes < 1024) return '$bytes B';
   634→    if (bytes < 1024 * 1024) return '${(bytes / 1024).toStringAsFixed(1)} KB';
   635→    if (bytes < 1024 * 1024 * 1024) {
   636→      return '${(bytes / (1024 * 1024)).toStringAsFixed(1)} MB';
   637→    }
   638→    return '${(bytes / (1024 * 1024 * 1024)).toStringAsFixed(1)} GB';
   639→  }
   640→}

   646→/// An animated indicator showing the current connection quality.
   647→///
   648→/// Displays concentric arcs (like a Wi-Fi signal) that fill based on
   649→/// the connection quality level:
   650→/// - **Excellent**: 4 arcs, green
   651→/// - **Good**: 3 arcs, blue
   652→/// - **Limited**: 2 arcs, orange
   653→/// - **Offline**: 0 arcs, red with a cross
   654→class ConnectionQualityIndicator extends StatefulWidget {
   655→  const ConnectionQualityIndicator({
   656→    super.key,
   657→    required this.quality,
   658→    this.size = 40,
   659→  });
   660→
   661→  final ConnectionQuality? quality;
   662→  final double size;
   663→
   664→  @override
   665→  State<ConnectionQualityIndicator> createState() =>
   666→      _ConnectionQualityIndicatorState();
   667→}
   668→
   669→class _ConnectionQualityIndicatorState
   670→    extends State<ConnectionQualityIndicator>
   671→    with SingleTickerProviderStateMixin {
   672→  late final AnimationController _controller;
   673→
   674→  @override
   675→  void initState() {
   676→    super.initState();
   677→    _controller = AnimationController(
   678→      vsync: this,
   679→      duration: const Duration(milliseconds: 1200),
   680→    )..repeat(reverse: true);
   681→  }
   682→
   683→  @override
   684→  void didUpdateWidget(ConnectionQualityIndicator oldWidget) {
   685→    super.didUpdateWidget(oldWidget);
   686→    if (oldWidget.quality != widget.quality) {
   687→      _controller.reset();
   688→      if (widget.quality != ConnectionQuality.offline) {
   689→        _controller.repeat(reverse: true);
   690→      }
   691→    }
   692→  }
   693→
   694→  @override
   695→  void dispose() {
   696→    _controller.dispose();
   697→    super.dispose();
   698→  }
   699→
   700→  @override
   701→  Widget build(BuildContext context) {
   702→    final quality = widget.quality ?? ConnectionQuality.offline;
   703→    final color = _qualityColor(quality);
   704→    final arcCount = _arcCount(quality);
   705→
   706→    return AnimatedBuilder(
   707→      animation: _controller,
   708→      builder: (context, child) {
   709→        final pulseValue = quality == ConnectionQuality.offline
   710→            ? 1.0
   711→            : 0.85 + (_controller.value * 0.15);
   712→
   713→        return Transform.scale(
   714→          scale: pulseValue,
   715→          child: CustomPaint(
   716→            size: Size(widget.size, widget.size),
   717→            painter: _SignalArcPainter(
   718→              arcCount: arcCount,
   719→              color: color,
   720→              isOffline: quality == ConnectionQuality.offline,
   721→            ),
   722→          ),
   723→        );
   724→      },
   725→    );
   726→  }
   727→
   728→  int _arcCount(ConnectionQuality quality) {
   729→    switch (quality) {
   730→      case ConnectionQuality.excellent:
   731→        return 4;
   732→      case ConnectionQuality.good:
   733→        return 3;
   734→      case ConnectionQuality.limited:
   735→        return 2;
   736→      case ConnectionQuality.offline:
   737→        return 0;
   738→    }
   739→  }
   740→
   741→  Color _qualityColor(ConnectionQuality quality) {
   742→    switch (quality) {
   743→      case ConnectionQuality.excellent:
   744→        return AppColors.success;
   745→      case ConnectionQuality.good:
   746→        return AppColors.info;
   747→      case ConnectionQuality.limited:
   748→        return AppColors.warning;
   749→      case ConnectionQuality.offline:
   750→        return AppColors.error;
   751→    }
   752→  }
   753→}

   755→/// Custom painter for the signal arc indicator.
   756→class _SignalArcPainter extends CustomPainter {
   757→  _SignalArcPainter({
   758→    required this.arcCount,
   759→    required this.color,
   760→    required this.isOffline,
   761→  });
   762→
   763→  final int arcCount;
   764→  final Color color;
   765→  final bool isOffline;
   766→
   767→  @override
   768→  void paint(Canvas canvas, Size size) {
   769→    final center = Offset(size.width / 2, size.height * 0.85);
   770→    final baseRadius = size.width * 0.15;
   771→
   772→    if (isOffline) {
   773→      // Draw an X for offline
   774→      final paint = Paint()
   775→        ..color = color
   776→        ..strokeWidth = 2.5
   777→        ..style = PaintingStyle.stroke
   778→        ..strokeCap = StrokeCap.round;
   779→
   780→      final crossSize = size.width * 0.25;
   781→      canvas.drawLine(
   782→        Offset(center.dx - crossSize, center.dy - crossSize),
   783→        Offset(center.dx + crossSize, center.dy + crossSize),
   784→        paint,
   785→      );
   786→      canvas.drawLine(
   787→        Offset(center.dx + crossSize, center.dy - crossSize),
   788→        Offset(center.dx - crossSize, center.dy + crossSize),
   789→        paint,
   790→      );
   791→      return;
   792→    }
   793→
   794→    // Draw signal arcs
   795→    for (int i = 0; i < 4; i++) {
   796→      final radius = baseRadius + (i * size.width * 0.12);
   797→      final isActive = i < arcCount;
   798→      final paint = Paint()
   799→        ..color = isActive ? color : color.withValues(alpha: 0.15)
   800→        ..strokeWidth = 2.5
   801→        ..style = PaintingStyle.stroke
   802→        ..strokeCap = StrokeCap.round;
   803→
   804→      canvas.drawArc(
   805→        Rect.fromCircle(center: center, radius: radius),
   806→        -2.2, // ~-126 degrees
   807→        1.4, // ~80 degrees sweep
   808→        false,
   809→        paint,
   810→      );
   811→    }
   812→
   813→    // Draw center dot
   814→    final dotPaint = Paint()
   815→      ..color = color
   816→      ..style = PaintingStyle.fill;
   817→    canvas.drawCircle(center, 2.5, dotPaint);
   818→  }
   819→
   820→  @override
   821→  bool shouldRepaint(_SignalArcPainter oldDelegate) {
   822→    return arcCount != oldDelegate.arcCount ||
   823→        color != oldDelegate.color ||
   824→        isOffline != oldDelegate.isOffline;
   825→  }
   826→}
