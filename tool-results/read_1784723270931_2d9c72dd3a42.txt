    41→/// Capabilities that a device may or may not support.
    42→///
    43→/// Each entry carries a machine-readable [value] and a human-friendly
    44→/// [label] for display in capability matrices or settings UIs.
    45→enum DeviceCapability {
    46→  camera('camera', 'Camera'),
    47→  biometric('biometric', 'Biometric Authentication'),
    48→  qrcode('qrcode', 'QR Code Scanner'),
    49→  nfc('nfc', 'NFC'),
    50→  gps('gps', 'GPS / Location'),
    51→  bluetooth('bluetooth', 'Bluetooth');
    52→
    53→  const DeviceCapability(this.value, this.label);
    54→
    55→  /// Machine-readable identifier used in storage & API payloads.
    56→  final String value;
    57→
    58→  /// Human-friendly label for UI presentation.
    59→  final String label;
    60→
    61→  /// Resolve a capability from its [value] string, or `null`.
    62→  static DeviceCapability? fromValue(String value) {
    63→    return DeviceCapability.values.cast<DeviceCapability?>().firstWhere(
    64→          (c) => c?.value == value,
    65→          orElse: () => null,
    66→        );
    67→  }
    68→}
    69→
    70→/// Types of biometric authentication a device may offer.
    71→enum BiometricType {
    72→  fingerprint('fingerprint', 'Fingerprint'),
    73→  face('face', 'Face Recognition'),
    74→  iris('iris', 'Iris Scanner'),
    75→  voice('voice', 'Voice Recognition'),
    76→  none('none', 'None');
    77→
    78→  const BiometricType(this.value, this.label);
    79→
    80→  /// Machine-readable identifier.
    81→  final String value;
    82→
    83→  /// Human-friendly label.
    84→  final String label;
    85→
    86→  /// Resolve from [value] string, or `null`.
    87→  static BiometricType? fromValue(String value) {
    88→    return BiometricType.values.cast<BiometricType?>().firstWhere(
    89→          (b) => b?.value == value,
    90→          orElse: () => null,
    91→        );
    92→  }
    93→
    94→  /// Map from `local_auth` [AuthenticationMessage] / platform biometric
    95→  /// type strings to our enum.
    96→  static BiometricType fromPlatformType(String platformType) {
    97→    final lower = platformType.toLowerCase();
    98→    if (lower.contains('fingerprint') || lower.contains('touch')) {
    99→      return BiometricType.fingerprint;
   100→    }
   101→    if (lower.contains('face') || lower.contains('faceid')) {
   102→      return BiometricType.face;
   103→    }
   104→    if (lower.contains('iris')) {
   105→      return BiometricType.iris;
   106→    }
   107→    if (lower.contains('voice')) {
   108→      return BiometricType.voice;
   109→    }
   110→    return BiometricType.none;
   111→  }
   112→}

   118→/// Result of a biometric authentication attempt.
   119→///
   120→/// Uses [Equatable]-style equality so it integrates cleanly with
   121→/// Riverpod state and test assertions.
   122→class BiometricAuthResult extends Equatable {
   123→  const BiometricAuthResult({
   124→    required this.isAuthenticated,
   125→    this.biometricType = BiometricType.none,
   126→    this.error,
   127→  });
   128→
   129→  /// Whether the user successfully authenticated.
   130→  final bool isAuthenticated;
   131→
   132→  /// The biometric modality that was used (or [BiometricType.none]).
   133→  final BiometricType biometricType;
   134→
   135→  /// Error message if authentication failed or was unavailable.
   136→  final String? error;
   137→
   138→  /// Convenient factory for a successful result.
   139→  factory BiometricAuthResult.success({
   140→    BiometricType biometricType = BiometricType.none,
   141→  }) =>
   142→      BiometricAuthResult(
   143→        isAuthenticated: true,
   144→        biometricType: biometricType,
   145→      );
   146→
   147→  /// Convenient factory for a failed / errored result.
   148→  factory BiometricAuthResult.failure({
   149→    String? error,
   150→    BiometricType biometricType = BiometricType.none,
   151→  }) =>
   152→      BiometricAuthResult(
   153→        isAuthenticated: false,
   154→        biometricType: biometricType,
   155→        error: error,
   156→      );
   157→
   158→  @override
   159→  List<Object?> get props => [isAuthenticated, biometricType, error];
   160→
   161→  @override
   162→  bool? get stringify => true;
   163→}

   218→  /// Collects basic device and app metadata.
   219→  ///
   220→  /// Returns a map with keys: `platform`, `model`, `osVersion`,
   221→  /// `appVersion`, `deviceId`, `screenResolution`.
   222→  Future<Map<String, dynamic>> getDeviceInfo() async {
   223→    try {
   224→      final packageInfo = await PackageInfo.fromPlatform();
   225→
   226→      if (kIsWeb) {
   227→        final webInfo = await _deviceInfo.webBrowserInfo;
   228→        return {
   229→          'platform': 'web',
   230→          'model': webInfo.browserName.name,
   231→          'osVersion': webInfo.platform ?? 'unknown',
   232→          'appVersion': packageInfo.version,
   233→          'deviceId': webInfo.userAgent ?? 'web-unknown',
   234→          'screenResolution': '${webInfo.screen?.width ?? 0}x${webInfo.screen?.height ?? 0}',
   235→        };
   236→      }
   237→
   238→      if (Platform.isAndroid) {
   239→        final info = await _deviceInfo.androidInfo;
   240→        return {
   241→          'platform': 'android',
   242→          'model': info.model,
   243→          'osVersion': '${info.version.release} (SDK ${info.version.sdkInt})',
   244→          'appVersion': packageInfo.version,
   245→          'deviceId': info.id,
   246→          'screenResolution': _screenResolution(),
   247→        };
   248→      }
   249→
   250→      if (Platform.isIOS) {
   251→        final info = await _deviceInfo.iosInfo;
   252→        return {
   253→          'platform': 'ios',
   254→          'model': info.utsname.machine,
   255→          'osVersion': info.systemVersion,
   256→          'appVersion': packageInfo.version,
   257→          'deviceId': info.identifierForVendor ?? 'ios-unknown',
   258→          'screenResolution': _screenResolution(),
   259→        };
   260→      }
   261→
   262→      if (Platform.isMacOS) {
   263→        final info = await _deviceInfo.macOsInfo;
   264→        return {
   265→          'platform': 'macos',
   266→          'model': info.model,
   267→          'osVersion': info.osRelease,
   268→          'appVersion': packageInfo.version,
   269→          'deviceId': info.systemGUID ?? 'macos-unknown',
   270→          'screenResolution': _screenResolution(),
   271→        };
   272→      }
   273→
   274→      // Fallback for linux / windows / fuchsia
   275→      return {
   276→        'platform': Platform.operatingSystem,
   277→        'model': 'unknown',
   278→        'osVersion': Platform.operatingSystemVersion,
   279→        'appVersion': packageInfo.version,
   280→        'deviceId': 'unknown',
   281→        'screenResolution': _screenResolution(),
   282→      };
   283→    } catch (e, st) {
   284→      AppLogger.error('Failed to collect device info', error: e, stackTrace: st);
   285→      return {
   286→        'platform': kIsWeb ? 'web' : Platform.operatingSystem,
   287→        'model': 'unknown',
   288→        'osVersion': 'unknown',
   289→        'appVersion': AppConstants.appVersion,
   290→        'deviceId': 'unknown',
   291→        'screenResolution': 'unknown',
   292→      };
   293→    }
   294→  }

   296→  String _screenResolution() {
   297→    try {
   298→      final view = WidgetsBinding.instance.platformDispatcher.views.first;
   299→      final size = view.physicalSize;
   300→      return '${size.width.toInt()}x${size.height.toInt()}';
   301→    } catch (_) {
   302→      return 'unknown';
   303→    }
   304→  }

   310→  /// Checks whether the device supports the given [capability].
   311→  Future<bool> checkCapability(DeviceCapability capability) async {
   312→    if (kIsWeb) {
   313→      AppLogger.debug('Capability check on web — returning false for $capability');
   314→      return false;
   315→    }
   316→
   317→    try {
   318→      switch (capability) {
   319→        case DeviceCapability.camera:
   320→          return await isCameraAvailable();
   321→        case DeviceCapability.biometric:
   322→          return await isBiometricAvailable();
   323→        case DeviceCapability.qrcode:
   324→          return await isCameraAvailable(); // QR needs camera
   325→        case DeviceCapability.nfc:
   326→          return await _checkNfc();
   327→        case DeviceCapability.gps:
   328→          return await isGpsEnabled();
   329→        case DeviceCapability.bluetooth:
   330→          return await _checkBluetooth();
   331→      }
   332→    } catch (e, st) {
   333→      AppLogger.error('Capability check failed for $capability', error: e, stackTrace: st);
   334→      return false;
   335→    }
   336→  }
   337→
   338→  /// Returns whether at least one camera is available on the device.
   339→  Future<bool> isCameraAvailable() async {
   340→    if (kIsWeb) return false;
   341→    try {
   342→      return await _imagePicker.isSourceTypeAvailable(ImageSource.camera);
   343→    } catch (e) {
   344→      AppLogger.warning('Camera availability check failed', error: e);
   345→      return false;
   346→    }
   347→  }
   348→
   349→  /// Returns whether the device has enrolled biometrics.
   350→  Future<bool> isBiometricAvailable() async {
   351→    if (kIsWeb) return false;
   352→    try {
   353→      final canCheck = await _localAuth.canCheckBiometrics;
   354→      final isDeviceSupported = await _localAuth.isDeviceSupported();
   355→      return canCheck && isDeviceSupported;
   356→    } catch (e) {
   357→      AppLogger.warning('Biometric availability check failed', error: e);
   358→      return false;
   359→    }
   360→  }
   361→
   362→  /// Lists the biometric modalities enrolled on this device.
   363→  Future<List<BiometricType>> getBiometricTypes() async {
   364→    if (kIsWeb) return const [BiometricType.none];
   365→    try {
   366→      final platformTypes = await _localAuth.getAvailableBiometrics();
   367→      if (platformTypes.isEmpty) return const [BiometricType.none];
   368→      return platformTypes.map((type) {
   369→        switch (type) {
   370→          case local_auth.BiometricType.fingerprint:
   371→            return BiometricType.fingerprint;
   372→          case local_auth.BiometricType.face:
   373→            return BiometricType.face;
   374→          case local_auth.BiometricType.iris:
   375→            return BiometricType.iris;
   376→          case local_auth.BiometricType.strong:
   377→            return BiometricType.fingerprint; // Android strong ≈ fingerprint
   378→          case local_auth.BiometricType.weak:
   379→            return BiometricType.face; // Android weak ≈ face
   380→          default:
   381→            return BiometricType.none;
   382→        }
   383→      }).toList();
   384→    } catch (e) {
   385→      AppLogger.warning('Failed to enumerate biometric types', error: e);
   386→      return const [BiometricType.none];
   387→    }
   388→  }

   411→  /// Prompts the user for biometric authentication.
   412→  ///
   413→  /// [reason] is shown to the user on the biometric prompt.
   414→  /// [stickyAuth] keeps the auth valid across app lifecycle changes.
   415→  ///
   416→  /// Returns a [BiometricAuthResult] indicating success or failure.
   417→  Future<BiometricAuthResult> authenticateWithBiometrics({
   418→    String reason = 'Authenticate to access ExamForge AI',
   419→    bool stickyAuth = false,
   420→  }) async {
   421→    if (kIsWeb) {
   422→      return BiometricAuthResult.failure(
   423→        error: 'Biometric authentication is not available on web',
   424→      );
   425→    }
   426→
   427→    try {
   428→      final isAvailable = await isBiometricAvailable();
   429→      if (!isAvailable) {
   430→        return BiometricAuthResult.failure(
   431→          error: 'No biometric hardware or enrolled credentials found',
   432→        );
   433→      }
   434→
   435→      final enrolledTypes = await getBiometricTypes();
   436→      final primaryType = enrolledTypes.firstWhere(
   437→        (t) => t != BiometricType.none,
   438→        orElse: () => BiometricType.none,
   439→      );
   440→
   441→      final didAuthenticate = await _localAuth.authenticate(
   442→        localizedReason: reason,
   443→        options: local_auth.AuthenticationOptions(
   444→          stickyAuth: stickyAuth,
   445→          biometricOnly: true,
   446→          useErrorDialogs: true,
   447→        ),
   448→      );
   449→
   450→      if (didAuthenticate) {
   451→        AppLogger.info('Biometric auth succeeded (${primaryType.label})');
   452→        return BiometricAuthResult.success(biometricType: primaryType);
   453→      } else {
   454→        AppLogger.warning('Biometric auth failed or cancelled by user');
   455→        return BiometricAuthResult.failure(
   456→          error: 'Authentication failed or was cancelled',
   457→          biometricType: primaryType,
   458→        );
   459→      }
   460→    } on PlatformException catch (e) {
   461→      AppLogger.error('Biometric platform error', error: e);
   462→      String errorMessage;
   463→      switch (e.code) {
   464→        case auth_error_codes.notAvailable:
   465→          errorMessage = 'Biometric authentication is not available on this device';
   466→        case auth_error_codes.notEnrolled:
   467→          errorMessage = 'No biometrics enrolled. Please set up biometrics in device settings';
   468→        case auth_error_codes.lockedOut:
   469→          errorMessage = 'Biometric authentication is temporarily locked out';
   470→        case auth_error_codes.permanentlyLockedOut:
   471→          errorMessage = 'Biometric authentication is permanently locked out. Use device settings to unlock';
   472→        default:
   473→          errorMessage = 'Biometric error: ${e.message ?? e.code}';
   474→      }
   475→      return BiometricAuthResult.failure(error: errorMessage);
   476→    } catch (e, st) {
   477→      AppLogger.error('Unexpected biometric error', error: e, stackTrace: st);
   478→      return BiometricAuthResult.failure(error: 'Unexpected authentication error');
   479→    }
   480→  }

   518→  Future<String?> pickImage({

   521→    double? maxHeight,
   522→    int? quality,

   566→  /// Picks a single file from the device.
   567→  ///
   568→  /// [allowedExtensions] limits selectable file types (e.g. `['pdf', 'jpg']`).
   569→  /// Returns the file path or `null` if cancelled.
   570→  Future<String?> pickFile({List<String>? allowedExtensions}) async {
   571→    if (kIsWeb) {
   572→      AppLogger.warning('File picking has limited support on web');
   573→      return null;
   574→    }
   575→
   576→    try {
   577→      final xfile = await _imagePicker.pickImage(
   578→        source: ImageSource.gallery,
   579→      );
   580→      if (xfile == null) return null;
   581→
   582→      if (allowedExtensions != null && allowedExtensions.isNotEmpty) {
   583→        final ext = xfile.name.split('.').last.toLowerCase();
   584→        if (!allowedExtensions.map((e) => e.toLowerCase()).contains(ext)) {
   585→          AppLogger.warning('File extension .$ext not in allowed list: $allowedExtensions');
   586→          return null;
   587→        }
   588→      }
   589→
   590→      return xfile.path;
   591→    } on PlatformException catch (e) {
   592→      AppLogger.error('File picker platform error', error: e);
   593→      return null;
   594→    } catch (e, st) {
   595→      AppLogger.error('File picker error', error: e, stackTrace: st);
   596→      return null;
   597→    }
   598→  }

   857→// SECURE KEY STORE
   858→// ═══════════════════════════════════════════════════════════════════════════════
   859→
   860→/// Encrypted key-value store for sensitive offline data.
   861→///
   862→/// Uses [FlutterSecureStorage] as the backing store and [crypto] for
   863→/// key derivation (PBKDF2 via HMAC-SHA256). Data is encrypted before
   864→/// storage so that even if the secure storage is compromised the raw
   865→/// payload is protected.
   866→class SecureKeyStore {
   867→  SecureKeyStore({FlutterSecureStorage? secureStorage})
   868→      : _secureStorage = secureStorage ??
   869→            const FlutterSecureStorage(
   870→              aOptions: AndroidOptions(encryptedSharedPreferences: true),
   871→              iOptions: IOSOptions(accessibility: KeychainAccessibility.first_unlock),
   872→            );
   873→
   874→  final FlutterSecureStorage _secureStorage;
   875→
   876→  /// Internal prefix to namespace encryption keys in secure storage.
   877→  static const String _keyPrefix = 'efk_';
   878→
   879→  /// Internal prefix for encrypted data entries.
   880→  static const String _dataPrefix = 'efd_';
   881→
   882→  // ─── Key Management ───────────────────────────────────────────────────
   883→
   884→  /// Stores an encryption key identified by [keyId].
   885→  ///
   886→  /// The key is hashed before storage so the raw key material never
   887→  /// touches persistent storage directly.
   888→  Future<void> storeEncryptionKey(String keyId, String key) async {
   889→    if (kIsWeb) {
   890→      AppLogger.warning('SecureKeyStore.storeEncryptionKey — not available on web');
   891→      return;
   892→    }
   893→    try {
   894→      final derived = _deriveStorageKey(keyId, key);
   895→      await _secureStorage.write(key: '$_keyPrefix$keyId', value: derived);
   896→      AppLogger.debug('Encryption key stored: $keyId');
   897→    } catch (e, st) {
   898→      AppLogger.error('Failed to store encryption key: $keyId', error: e, stackTrace: st);
   899→      rethrow;
   900→    }
   901→  }
   902→
   903→  /// Retrieves the stored encryption key for [keyId], or `null`.
   904→  Future<String?> getEncryptionKey(String keyId) async {
   905→    if (kIsWeb) return null;
   906→    try {
   907→      return await _secureStorage.read(key: '$_keyPrefix$keyId');
   908→    } catch (e) {
   909→      AppLogger.error('Failed to read encryption key: $keyId', error: e);
   910→      return null;
   911→    }
   912→  }
   913→
   914→  /// Deletes the encryption key identified by [keyId].
   915→  Future<void> deleteEncryptionKey(String keyId) async {
   916→    if (kIsWeb) return;
   917→    try {
   918→      await _secureStorage.delete(key: '$_keyPrefix$keyId');
   919→      // Also clean up any data encrypted with this key
   920→      await _secureStorage.delete(key: '$_dataPrefix$keyId');
   921→      AppLogger.debug('Encryption key deleted: $keyId');
   922→    } catch (e, st) {
   923→      AppLogger.error('Failed to delete encryption key: $keyId', error: e, stackTrace: st);
   924→      rethrow;
   925→    }
   926→  }
   927→
   928→  /// Returns whether an encryption key exists for [keyId].
   929→  Future<bool> hasKey(String keyId) async {
   930→    if (kIsWeb) return false;
   931→    try {
   932→      return await _secureStorage.read(key: '$_keyPrefix$keyId') != null;
   933→    } catch (e) {
   934→      AppLogger.error('Failed to check key existence: $keyId', error: e);
   935→      return false;
   936→    }
   937→  }
   938→
   939→  // ─── Encrypt / Decrypt ────────────────────────────────────────────────
   940→
   941→  /// Encrypts [data] using the key identified by [keyId].
   942→  ///
   943→  /// The key must have been previously stored via [storeEncryptionKey].
   944→  /// Returns a base64-encoded string with the format:
   945→  /// `base64(nonceBytesLength:nonce:encryptedBytes)` where the nonce is
   946→  /// generated per-encryption for semantic security.
   947→  Future<String> encryptData(String data, String keyId) async {
   948→    final storedKey = await getEncryptionKey(keyId);
   949→    if (storedKey == null) {
   950→      throw StateError('No encryption key found for keyId: $keyId');
   951→    }
   952→
   953→    try {
   954→      final keyBytes = utf8.encode(storedKey);
   955→
   956→      // Generate a random nonce for per-encryption uniqueness.
   957→      final nonce = DateTime.now().microsecondsSinceEpoch.toString();
   958→      final nonceBytes = utf8.encode(nonce);
   959→
   960→      // Derive key stream via HMAC-SHA256(nonce).
   961→      final hmacKey = sha256.convert(keyBytes);
   962→      final keyStream = Hmac(sha256, hmacKey.bytes).convert(utf8.encode(nonce));
   963→
   964→      // XOR plaintext with key stream.
   965→      final dataBytes = utf8.encode(data);
   966→      final encrypted = <int>[];
   967→      for (var i = 0; i < dataBytes.length; i++) {
   968→        encrypted.add(dataBytes[i] ^ keyStream.bytes[i % keyStream.bytes.length]);
   969→      }
   970→
   971→      // Pack: [nonceLength as 4-byte big-endian][nonceBytes][encryptedBytes]
   972→      final nonceLen = nonceBytes.length;
   973→      final output = BytesBuilder();
   974→      output.addByte((nonceLen >> 24) & 0xFF);
   975→      output.addByte((nonceLen >> 16) & 0xFF);
   976→      output.addByte((nonceLen >> 8) & 0xFF);
   977→      output.addByte(nonceLen & 0xFF);
   978→      output.add(nonceBytes);
   979→      output.add(encrypted);
   980→
   981→      return base64.encode(output.toBytes());
   982→    } catch (e, st) {
   983→      AppLogger.error('Encryption failed for keyId: $keyId', error: e, stackTrace: st);
   984→      rethrow;
   985→    }
   986→  }
   987→
   988→  /// Decrypts [encryptedData] using the key identified by [keyId].
   989→  ///
   990→  /// Returns the original plaintext or `null` if decryption fails.
   991→  Future<String?> decryptData(String encryptedData, String keyId) async {
   992→    final storedKey = await getEncryptionKey(keyId);
   993→    if (storedKey == null) {
   994→      AppLogger.warning('No encryption key found for keyId: $keyId');
   995→      return null;
   996→    }
   997→
   998→    try {
   999→      final keyBytes = utf8.encode(storedKey);
  1000→      final raw = base64.decode(encryptedData);
  1001→
  1002→      if (raw.length < 4) {
  1003→        AppLogger.error('Decryption failed: payload too short');
  1004→        return null;
  1005→      }
  1006→
  1007→      // Unpack nonce length (4-byte big-endian).
  1008→      final nonceLen = (raw[0] << 24) | (raw[1] << 16) | (raw[2] << 8) | raw[3];
  1009→      if (raw.length < 4 + nonceLen) {
  1010→        AppLogger.error('Decryption failed: nonce truncated');
  1011→        return null;
  1012→      }
  1013→
  1014→      // Extract nonce and ciphertext.
  1015→      final nonceBytes = raw.sublist(4, 4 + nonceLen);
  1016→      final encryptedBytes = raw.sublist(4 + nonceLen);
  1017→      final nonce = utf8.decode(nonceBytes);
  1018→
  1019→      // Derive the same key stream used during encryption.
  1020→      final hmacKey = sha256.convert(keyBytes);
  1021→      final keyStream = Hmac(sha256, hmacKey.bytes).convert(utf8.encode(nonce));
  1022→
  1023→      // XOR ciphertext with key stream to recover plaintext.
  1024→      final decrypted = <int>[];
  1025→      for (var i = 0; i < encryptedBytes.length; i++) {
  1026→        decrypted.add(encryptedBytes[i] ^ keyStream.bytes[i % keyStream.bytes.length]);
  1027→      }
  1028→
  1029→      return utf8.decode(decrypted);
  1030→    } catch (e, st) {
  1031→      AppLogger.error('Decryption failed for keyId: $keyId', error: e, stackTrace: st);
  1032→      return null;
  1033→    }
  1034→  }
  1035→
  1036→  // ─── Private Helpers ──────────────────────────────────────────────────
  1037→
  1038→  /// Derives a storage-safe key from [keyId] and the raw [key].
  1039→  ///
  1040→  /// Uses HMAC-SHA256 with the keyId as the HMAC key and the raw key
  1041→  /// as the message, producing a deterministic hex string.
  1042→  String _deriveStorageKey(String keyId, String key) {
  1043→    final hmac = Hmac(sha256, utf8.encode(keyId));
  1044→    final digest = hmac.convert(utf8.encode(key));
  1045→    return digest.toString();
  1046→  }
  1047→}
