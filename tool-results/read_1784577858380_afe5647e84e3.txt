   787→-- ─── FUNCTION: Apply grade scale to a percentage ─────────────────────
   788→
   789→CREATE OR REPLACE FUNCTION public.apply_grade_scale(
   790→  p_percentage DOUBLE PRECISION,
   791→  p_grade_scale_id UUID
   792→)
   793→RETURNS TABLE(grade TEXT, gpa_value DOUBLE PRECISION, is_passing BOOLEAN, description TEXT) AS $$
   794→DECLARE
   795→  v_school_id UUID;
   796→BEGIN
   797→  SELECT school_id INTO v_school_id FROM public.grade_scales WHERE id = p_grade_scale_id;
   798→  
   799→  RETURN QUERY
   800→  SELECT gse.grade, gse.gpa_value, gse.is_passing, gse.description
   801→  FROM public.grade_scale_entries gse
   802→  WHERE gse.grade_scale_id = p_grade_scale_id
   803→    AND p_percentage >= gse.min_percentage
   804→    AND p_percentage <= gse.max_percentage
   805→  ORDER BY gse.sort_order
   806→  LIMIT 1;
   807→END;
   808→$$ LANGUAGE plpgsql STABLE;

   810→-- ─── FUNCTION: Compute student subject result ────────────────────────
   811→
   812→CREATE OR REPLACE FUNCTION public.compute_student_subject_result(
   813→  p_student_id UUID,
   814→  p_subject_id UUID,
   815→  p_class_id UUID,
   816→  p_academic_session_id UUID
   817→)
   818→RETURNS public.student_subject_results AS $$
   819→DECLARE
   820→  v_result public.student_subject_results;
   821→  v_total_obtained DOUBLE PRECISION;
   822→  v_total_possible DOUBLE PRECISION;
   823→  v_exam_count INT;
   824→  v_percentage DOUBLE PRECISION;
   825→  v_grade TEXT;
   826→  v_class_avg DOUBLE PRECISION;
   827→  v_school_id UUID;
   828→BEGIN
   829→  -- Get school_id from class
   830→  SELECT school_id INTO v_school_id FROM public.classes WHERE id = p_class_id;
   831→  
   832→  -- Aggregate exam results
   833→  SELECT 
   834→    COALESCE(SUM(er.total_marks), 0),
   835→    COALESCE(SUM(er.total_possible), 0),
   836→    COUNT(*)
   837→  INTO v_total_obtained, v_total_possible, v_exam_count
   838→  FROM public.exam_results er
   839→  JOIN public.exams e ON e.id = er.exam_id
   840→  WHERE er.student_id = p_student_id
   841→    AND e.subject_id = p_subject_id
   842→    AND e.class_id = p_class_id
   843→    AND e.academic_session_id = p_academic_session_id
   844→    AND er.is_released = true;
   845→  
   846→  v_percentage := CASE WHEN v_total_possible > 0 THEN (v_total_obtained / v_total_possible) * 100 ELSE 0 END;
   847→  
   848→  -- Get class average
   849→  SELECT COALESCE(AVG(ssr.percentage), 0) INTO v_class_avg
   850→  FROM public.student_subject_results ssr
   851→  WHERE ssr.subject_id = p_subject_id
   852→    AND ssr.class_id = p_class_id
   853→    AND ssr.academic_session_id = p_academic_session_id;
   854→  
   855→  -- Upsert
   856→  INSERT INTO public.student_subject_results (
   857→    student_id, school_id, subject_id, class_id, academic_session_id,
   858→    exam_count, total_marks_obtained, total_marks_possible, percentage,
   859→    class_average, is_passed
   860→  ) VALUES (
   861→    p_student_id, v_school_id, p_subject_id, p_class_id, p_academic_session_id,
   862→    v_exam_count, v_total_obtained, v_total_possible, v_percentage,
   863→    v_class_avg, v_percentage >= 50
   864→  )
   865→  ON CONFLICT (student_id, subject_id, class_id, academic_session_id)
   866→  DO UPDATE SET
   867→    exam_count = EXCLUDED.exam_count,
   868→    total_marks_obtained = EXCLUDED.total_marks_obtained,
   869→    total_marks_possible = EXCLUDED.total_marks_possible,
   870→    percentage = EXCLUDED.percentage,
   871→    class_average = EXCLUDED.class_average,
   872→    is_passed = EXCLUDED.is_passed,
   873→    updated_at = now()
   874→  RETURNING * INTO v_result;
   875→  
   876→  RETURN v_result;
   877→END;
   878→$$ LANGUAGE plpgsql;
