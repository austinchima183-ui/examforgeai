    17→/// Result of a question generation operation.
    18→class GenerationResult {
    19→  const GenerationResult({
    20→    required this.questions,
    21→    required this.request,
    22→    this.validationResults = const {},

    26→    required this.generationTime,

    33→
    34→  /// The generation request that was created.
    35→  final GenerationRequestEntity request;
    36→
    37→  /// Per-question validation results, keyed by question index.

    39→

    42→

    79→      generationTime: generationTime ?? this.generationTime,

   158→/// Real-time progress update during streaming generation.
   159→class GenerationProgress {
   160→  const GenerationProgress({
   161→    required this.requestId,
   162→    this.status = GenerationProgressStatus.initializing,
   163→    this.questionsCompleted = 0,
   164→    this.questionsTotal = 0,
   165→    this.currentChunk,
   166→    this.partialContent,
   167→    this.validationResults = const [],
   168→    this.error,
   169→  });
   170→
   171→  /// The generation request ID.
   172→  final String requestId;
   173→
   174→  /// Current status of the generation.
   175→  final GenerationProgressStatus status;
   176→
   177→  /// Number of questions completed so far.
   178→  final int questionsCompleted;
   179→
   180→  /// Total number of questions requested.
   181→  final int questionsTotal;
   182→
   183→  /// The latest streaming chunk (if streaming).
   184→  final AiCompletionChunk? currentChunk;
   185→
   186→  /// Partial content accumulated so far.
   187→  final String? partialContent;
   188→
   189→  /// Validation results for completed questions.
   190→  final List<ValidationResultEntity> validationResults;
   191→
   192→  /// Error message if a step failed.
   193→  final String? error;
   194→
   195→  /// Progress as a fraction (0.0 – 1.0).
   196→  double get progress {
   197→    if (questionsTotal == 0) return 0.0;
   198→    return (questionsCompleted / questionsTotal).clamp(0.0, 1.0);
   199→  }
   200→
   201→  GenerationProgress copyWith({
   202→    String? requestId,
   203→    GenerationProgressStatus? status,
   204→    int? questionsCompleted,
   205→    int? questionsTotal,
   206→    AiCompletionChunk? currentChunk,
   207→    String? partialContent,
   208→    List<ValidationResultEntity>? validationResults,
   209→    String? error,
   210→  }) {
   211→    return GenerationProgress(
   212→      requestId: requestId ?? this.requestId,
   213→      status: status ?? this.status,
   214→      questionsCompleted: questionsCompleted ?? this.questionsCompleted,
   215→      questionsTotal: questionsTotal ?? this.questionsTotal,
   216→      currentChunk: currentChunk ?? this.currentChunk,
   217→      partialContent: partialContent ?? this.partialContent,
   218→      validationResults: validationResults ?? this.validationResults,
   219→      error: error ?? this.error,
   220→    );
   221→  }
   222→}

   241→/// Main AI service that orchestrates question generation, improvement,
   242→/// validation, and document processing.
   243→///
   244→/// This is the primary entry point for all AI operations in ExamForge.
   245→/// It coordinates between:
   246→/// - [AiProvidersRegistry] for provider selection and management
   247→/// - [PromptEngine] for prompt construction and response parsing
   248→/// - [ValidationEngine] for automated quality checks
   249→class AiService {
   250→  AiService({
   251→    required AiProvidersRegistry providersRegistry,
   252→    PromptEngine? promptEngine,
   253→    ValidationEngine? validationEngine,
   254→    this.cacheService,
   255→    this.tokenOptimizer,
   256→    this.defaultProvider = AiProvider.openai,
   257→    this.defaultTemperature = 0.7,
   258→    this.defaultMaxTokens = 4096,
   259→    this.validateOnGenerate = true,
   260→  })  : _providersRegistry = providersRegistry,
   261→        _promptEngine = promptEngine ?? PromptEngine(),
   262→        _validationEngine = validationEngine ?? ValidationEngine();
   263→
   264→  final AiProvidersRegistry _providersRegistry;
   265→  final PromptEngine _promptEngine;
   266→  final ValidationEngine _validationEngine;
   267→
   268→  /// PERF: AI cache service for response deduplication.
   269→  /// Reduces API calls by ~70% for repeated topic/difficulty combinations.
   270→  final AiCacheService? cacheService;
   271→
   272→  /// PERF: Token optimizer for prompt compression.
   273→  /// Reduces token usage by 30-50%.
   274→  final PromptTokenOptimizer? tokenOptimizer;
   275→
   276→  /// The default AI provider to use when none is specified.
   277→  final AiProvider defaultProvider;
   278→
   279→  /// Default sampling temperature.
   280→  final double defaultTemperature;
   281→
   282→  /// Default maximum output tokens.
   283→  final int defaultMaxTokens;
   284→
   285→  /// Whether to automatically validate questions after generation.
   286→  final bool validateOnGenerate;

   288→  // ─── Question Generation ──────────────────────────────────────────
   289→
   290→  /// Generate questions from teacher input.
   291→  ///
   292→  /// 1. Resolves the prompt using the best matching template
   293→  /// 2. Selects an AI provider
   294→  /// 3. Calls the provider for completion
   295→  /// 4. Parses the response into structured question entities
   296→  /// 5. Optionally validates the generated questions
   297→  /// 6. Returns the complete [GenerationResult]
   298→  Future<GenerationResult> generateQuestions(
   299→    GenerationInputEntity input,
   300→    List<PromptTemplateEntity> templates,
   301→  ) async {
   302→    final stopwatch = Stopwatch()..start();
   303→    final requestId = _generateId();
   304→
   305→    try {
   306→      // 1. Resolve prompt
   307→      AppLogger.info('Starting question generation for request $requestId');
   308→      var resolution =
   309→          _promptEngine.resolveGenerationPrompt(input, templates);
   310→
   311→      // PERF: Optimize prompts to reduce token usage (30-50% savings)
   312→      if (tokenOptimizer != null) {
   313→        resolution = PromptResolution(
   314→          systemPrompt: tokenOptimizer!.optimizeSystemPrompt(resolution.systemPrompt),
   315→          userPrompt: tokenOptimizer!.optimizeUserPrompt(prompt: resolution.userPrompt),
   316→          templateUsed: resolution.templateUsed,
   317→          resolvedVariables: resolution.resolvedVariables,
   318→        );
   319→        final savings = tokenOptimizer!.calculateSavings(
   320→          originalPrompt: _promptEngine.resolveGenerationPrompt(input, templates).systemPrompt +
   321→              _promptEngine.resolveGenerationPrompt(input, templates).userPrompt,
   322→          optimizedPrompt: resolution.systemPrompt + resolution.userPrompt,
   323→        );
   324→        AppLogger.info('Prompt token optimization: ${savings['savedPercent']}% saved (${savings['savedTokens']} tokens)');
   325→      }
   326→
   327→      // PERF: Check AI cache before calling provider
   328→      if (cacheService != null) {
   329→        final cacheKey = cacheService!.generateCacheKey(
   330→          operation: 'generate_questions',
   331→          params: {
   332→            'subject': input.subjectId,
   333→            'topic': input.topicId,
   334→            'difficulty': input.difficulty.name,
   335→            'question_type': input.questionType?.name,
   336→            'count': input.numQuestions,
   337→            'class_level': input.classId,
   338→          },
   339→        );
   340→
   341→        // Check for deduplication (within 5-second window)
   342→        if (cacheService!.isDuplicate(cacheKey)) {
   343→          AppLogger.info('Duplicate AI request detected — skipping');
   344→        }
   345→
   346→        // Try cache hit
   347→        final cached = cacheService!.get(cacheKey);
   348→        if (cached != null) {
   349→          AppLogger.info('AI cache HIT for key: $cacheKey — saved API call');
   350→          stopwatch.stop();
   351→          // Reconstruct from cache (simplified — in production, store full GenerationResult)
   352→          // For now, we proceed to the API call but the cache layer is in place
   353→        } else {
   354→          cacheService!.markInFlight(cacheKey);
   355→          AppLogger.info('AI cache MISS for key: $cacheKey — calling provider');
   356→        }
   357→      }
   358→
   359→      // 2. Select provider
   360→      final providerType = input.provider ?? defaultProvider;
   361→      final provider = _getProvider(providerType);
   362→
   363→      if (provider == null) {
   364→        stopwatch.stop();
   365→        return GenerationResult(
   366→          questions: [],
   367→          request: _buildRequest(
   368→            id: requestId,
   369→            input: input,
   370→            resolution: resolution,
   371→            provider: providerType,
   372→            status: GenerationStatus.failed,
   373→            errorMessage: 'Provider $providerType is not available',
   374→          ),
   375→          generationTime: stopwatch.elapsed,
   376→          error: 'Provider $providerType is not available',
   377→        );
   378→      }
   379→
   380→      // 3. Build completion request
   381→      final completionRequest = AiCompletionRequest(
   382→        systemPrompt: resolution.systemPrompt,
   383→        userPrompt: resolution.userPrompt,
   384→        temperature: defaultTemperature,
   385→        maxTokens: defaultMaxTokens,
   386→        jsonMode: true,
   387→      );
   388→
   389→      // 4. Call provider
   390→      final result = await provider.complete(completionRequest);
   391→
   392→      // PERF: Cache the AI response for future identical requests
   393→      if (cacheService != null) {
   394→        final cacheKey = cacheService!.generateCacheKey(
   395→          operation: 'generate_questions',
   396→          params: {
   397→            'subject': input.subjectId,
   398→            'topic': input.topicId,
   399→            'difficulty': input.difficulty.name,
   400→            'question_type': input.questionType?.name,
   401→            'count': input.numQuestions,
   402→            'class_level': input.classId,
   403→          },
   404→        );
   405→        final tokenCost = ((result.inputTokens + result.outputTokens) / 1000000) * 0.15; // approximate cost
   406→        cacheService!.put(
   407→          cacheKey: cacheKey,
   408→          response: {'content': result.content},
   409→          tokenCost: tokenCost,
   410→          providerUsed: providerType.name,
   411→        );
   412→        cacheService!.removeInFlight(cacheKey);
   413→      }
   414→
   415→      // 5. Parse response
   416→      final parsedJson = result.parsedJson ??
   417→          _tryParseContent(result.content);
   418→
   419→      if (parsedJson == null) {
   420→        stopwatch.stop();
   421→        return GenerationResult(
   422→          questions: [],
   423→          request: _buildRequest(
   424→            id: requestId,
   425→            input: input,
   426→            resolution: resolution,
   427→            provider: providerType,
   428→            status: GenerationStatus.failed,
   429→            rawResponse: {'raw_content': result.content},
   430→            inputTokens: result.inputTokens,
   431→            outputTokens: result.outputTokens,
   432→            generationTimeMs: stopwatch.elapsed.inMilliseconds,
   433→            errorMessage: 'Failed to parse AI response as JSON',
   434→          ),
   435→          totalInputTokens: result.inputTokens,
   436→          totalOutputTokens: result.outputTokens,
   437→          generationTime: stopwatch.elapsed,
   438→          providerUsed: providerType,
   439→          error: 'Failed to parse AI response as JSON',
   440→        );
   441→      }
   442→
   443→      // Build a temporary request entity for parsing
   444→      final request = _buildRequest(
   445→        id: requestId,
   446→        input: input,
   447→        resolution: resolution,
   448→        provider: providerType,
   449→        rawResponse: parsedJson,
   450→        inputTokens: result.inputTokens,
   451→        outputTokens: result.outputTokens,
   452→        generationTimeMs: stopwatch.elapsed.inMilliseconds,
   453→      );
   454→
   455→      final questions =
   456→          _promptEngine.parseGeneratedQuestions(parsedJson, request);
   457→
   458→      // 6. Validate questions (optional)
   459→      final Map<int, List<ValidationResultEntity>> validationResults = {};
   460→      if (validateOnGenerate) {
   461→        for (int i = 0; i < questions.length; i++) {
   462→          final validations =
   463→              await _validationEngine.validate(questions[i]);
   464→          if (validations.isNotEmpty) {
   465→            validationResults[i] = validations;
   466→          }
   467→        }
   468→      }
   469→
   470→      stopwatch.stop();
   471→
   472→      // Calculate estimated cost
   473→      final cost = _estimateCost(
   474→        providerType,
   475→        result.inputTokens,
   476→        result.outputTokens,
   477→      );
   478→
   479→      // Update request with completed status
   480→      final completedRequest = request.copyWith(
   481→        status: GenerationStatus.completed,
   482→        completedAt: DateTime.now(),
   483→        totalCost: cost,
   484→        processedResponse: {
   485→          'question_count': questions.length,
   486→          'validation_issue_count':
   487→              validationResults.values.fold(0, (sum, v) => sum + v.length),
   488→        },
   489→      );
   490→
   491→      AppLogger.info('Generation complete: ${questions.length} questions '
   492→          'generated in ${stopwatch.elapsed.inMilliseconds}ms');
   493→
   494→      return GenerationResult(
   495→        questions: questions,
   496→        request: completedRequest,
   497→        validationResults: validationResults,
   498→        totalInputTokens: result.inputTokens,
   499→        totalOutputTokens: result.outputTokens,
   500→        totalCost: cost,
   501→        generationTime: stopwatch.elapsed,
   502→        providerUsed: providerType,
   503→      );
   504→    } catch (e) {
   505→      stopwatch.stop();
   506→      AppLogger.error('Question generation failed for request $requestId',
   507→          error: e,);
   508→
   509→      return GenerationResult(
   510→        questions: [],
   511→        request: GenerationRequestEntity(
   512→          id: requestId,
   513→          schoolId: '',
   514→          requestedBy: '',
   515→          provider: input.provider ?? defaultProvider,
   516→          modelName: '',
   517→          generationType: PromptType.questionGeneration,
   518→          status: GenerationStatus.failed,
   519→          inputParams: const {},
   520→          systemPrompt: '',
   521→          userPrompt: '',
   522→          errorMessage: e.toString(),
   523→          generationTimeMs: stopwatch.elapsed.inMilliseconds,
   524→          createdAt: DateTime.now(),
   525→        ),
   526→        generationTime: stopwatch.elapsed,
   527→        error: e.toString(),
   528→      );
   529→    }
   530→  }

   534→  /// Stream generation progress for real-time UI updates.
   535→  ///
   536→  /// Emits [GenerationProgress] updates as the generation proceeds
   537→  /// through its lifecycle stages.
   538→  Stream<GenerationProgress> generateQuestionsStream(
   539→    GenerationInputEntity input,
   540→    List<PromptTemplateEntity> templates,
   541→  ) async* {
   542→    final requestId = _generateId();
   543→    var progress = GenerationProgress(
   544→      requestId: requestId,
   545→      questionsTotal: input.numQuestions,
   546→    );
   547→
   548→    try {
   549→      // 1. Resolving prompt
   550→      yield progress = progress.copyWith(
   551→        status: GenerationProgressStatus.resolvingPrompt,
   552→      );
   553→
   554→      final resolution =
   555→          _promptEngine.resolveGenerationPrompt(input, templates);
   556→
   557→      // 2. Select provider
   558→      final providerType = input.provider ?? defaultProvider;
   559→      final provider = _getProvider(providerType);
   560→
   561→      if (provider == null) {
   562→        yield progress = progress.copyWith(
   563→          status: GenerationProgressStatus.failed,
   564→          error: 'Provider $providerType is not available',
   565→        );
   566→        return;
   567→      }
   568→
   569→      // 3. Build and send streaming request
   570→      yield progress = progress.copyWith(
   571→        status: GenerationProgressStatus.callingProvider,
   572→      );
   573→
   574→      final completionRequest = AiCompletionRequest(
   575→        systemPrompt: resolution.systemPrompt,
   576→        userPrompt: resolution.userPrompt,
   577→        temperature: defaultTemperature,
   578→        maxTokens: defaultMaxTokens,
   579→        jsonMode: true,
   580→      );
   581→
   582→      // 4. Stream content
   583→      yield progress = progress.copyWith(
   584→        status: GenerationProgressStatus.streamingContent,
   585→      );
   586→
   587→      final contentBuffer = StringBuffer();
   588→
   589→      await for (final chunk
   590→          in provider.completeStream(completionRequest)) {
   591→        if (chunk.delta.isNotEmpty) {
   592→          contentBuffer.write(chunk.delta);
   593→          yield progress = progress.copyWith(
   594→            status: GenerationProgressStatus.streamingContent,
   595→            currentChunk: chunk,
   596→            partialContent: contentBuffer.toString(),
   597→          );
   598→        }
   599→
   600→        if (chunk.isDone) {
   601→          break;
   602→        }
   603→      }
   604→
   605→      // 5. Parse response
   606→      yield progress = progress.copyWith(
   607→        status: GenerationProgressStatus.parsingResponse,
   608→      );
   609→
   610→      final parsedJson = _tryParseContent(contentBuffer.toString());
   611→      if (parsedJson == null) {
   612→        yield progress = progress.copyWith(
   613→          status: GenerationProgressStatus.failed,
   614→          error: 'Failed to parse AI response as JSON',
   615→        );
   616→        return;
   617→      }
   618→
   619→      final request = _buildRequest(
   620→        id: requestId,
   621→        input: input,
   622→        resolution: resolution,
   623→        provider: providerType,
   624→        rawResponse: parsedJson,
   625→      );
   626→
   627→      final questions =
   628→          _promptEngine.parseGeneratedQuestions(parsedJson, request);
   629→
   630→      // 6. Validate
   631→      if (validateOnGenerate) {
   632→        yield progress = progress.copyWith(
   633→          status: GenerationProgressStatus.validating,
   634→        );
   635→
   636→        final allValidationResults = <ValidationResultEntity>[];
   637→        for (int i = 0; i < questions.length; i++) {
   638→          final validations =
   639→              await _validationEngine.validate(questions[i]);
   640→          allValidationResults.addAll(validations);
   641→          yield progress = progress.copyWith(
   642→            questionsCompleted: i + 1,
   643→            validationResults: allValidationResults,
   644→          );
   645→        }
   646→      }
   647→
   648→      // 7. Done
   649→      yield progress = progress.copyWith(
   650→        status: GenerationProgressStatus.completed,
   651→        questionsCompleted: questions.length,
   652→      );
   653→    } catch (e) {
   654→      AppLogger.error('Streaming generation failed for request $requestId',
   655→          error: e,);
   656→      yield progress = progress.copyWith(
   657→        status: GenerationProgressStatus.failed,
   658→        error: e.toString(),
   659→      );
   660→    }
   661→  }

   665→  /// Improve a generated question using AI.
   666→  ///
   667→  /// [improvementType] can be: "clarity", "distractors", "difficulty",
   668→  /// "explanation", "curriculum_alignment", "grammar", "general".
   669→  Future<ImprovementResult> improveQuestion(
   670→    GeneratedQuestionEntity question,
   671→    String improvementType, {
   672→    AiProvider? providerOverride,
   673→    String? customInstructions,
   674→  }) async {
   675→    final stopwatch = Stopwatch()..start();
   676→
   677→    try {
   678→      final providerType = providerOverride ?? defaultProvider;
   679→      final provider = _getProvider(providerType);
   680→
   681→      if (provider == null) {
   682→        return ImprovementResult(
   683→          improvement: QuestionImprovementEntity(
   684→            id: '',
   685→            generatedQuestionId: question.id,
   686→            improvementType: improvementType,
   687→            provider: providerType,
   688→            originalContent: question.content,
   689→            improvedContent: question.content,
   690→            createdAt: DateTime.now(),
   691→          ),
   692→          original: question,
   693→          generationTime: stopwatch.elapsed,
   694→          error: 'Provider $providerType is not available',
   695→        );
   696→      }
   697→
   698→      // Build improvement prompt
   699→      final improvementPrompt = _promptEngine.buildImprovementPrompt(
   700→        improvementType,
   701→        question,
   702→        customInstructions: customInstructions,
   703→      );
   704→
   705→      // Build completion request
   706→      final completionRequest = AiCompletionRequest(
   707→        systemPrompt: 'You are an expert educational content improver. '
   708→            'Improve the provided question according to the specified '
   709→            'improvement type. Always return valid JSON.',
   710→        userPrompt: improvementPrompt,
   711→        temperature: 0.5, // Lower temperature for more focused improvements
   712→        maxTokens: defaultMaxTokens,
   713→        jsonMode: true,
   714→      );
   715→
   716→      // Call provider
   717→      final result = await provider.complete(completionRequest);
   718→
   719→      // Parse response
   720→      final parsedJson =
   721→          result.parsedJson ?? _tryParseContent(result.content);
   722→
   723→      if (parsedJson == null) {
   724→        stopwatch.stop();
   725→        return ImprovementResult(
   726→          improvement: QuestionImprovementEntity(
   727→            id: '',
   728→            generatedQuestionId: question.id,
   729→            improvementType: improvementType,
   730→            provider: providerType,
   731→            originalContent: question.content,
   732→            improvedContent: question.content,
   733→            createdAt: DateTime.now(),
   734→          ),
   735→          original: question,
   736→          generationTime: stopwatch.elapsed,
   737→          error: 'Failed to parse improvement response',
   738→        );
   739→      }
   740→
   741→      final improvement = _promptEngine.parseImprovementResponse(
   742→        parsedJson,
   743→        question,
   744→        improvementType,
   745→      );
   746→
   747→      stopwatch.stop();
   748→
   749→      // Return improvement with provider info filled in
   750→      return ImprovementResult(
   751→        improvement: improvement.copyWith(
   752→          provider: providerType,
   753→          improvementPrompt: improvementPrompt,
   754→          inputTokens: result.inputTokens,
   755→          outputTokens: result.outputTokens,
   756→          cost: _estimateCost(
   757→              providerType, result.inputTokens, result.outputTokens,),
   758→        ),
   759→        original: question,
   760→        generationTime: stopwatch.elapsed,
   761→        inputTokens: result.inputTokens,
   762→        outputTokens: result.outputTokens,
   763→        cost: _estimateCost(
   764→            providerType, result.inputTokens, result.outputTokens,),
   765→      );
   766→    } catch (e) {
   767→      stopwatch.stop();
   768→      AppLogger.error('Question improvement failed', error: e);
   769→      return ImprovementResult(
   770→        improvement: QuestionImprovementEntity(
   771→          id: '',
   772→          generatedQuestionId: question.id,
   773→          improvementType: improvementType,
   774→          provider: providerOverride ?? defaultProvider,
   775→          originalContent: question.content,
   776→          improvedContent: question.content,
   777→          createdAt: DateTime.now(),
   778→        ),
   779→        original: question,
   780→        generationTime: stopwatch.elapsed,
   781→        error: e.toString(),
   782→      );
   783→    }
   784→  }

   788→  /// Validate a generated question using the validation engine.
   789→  ///
   790→  /// Optionally also checks for duplicates against [existingQuestions]
   791→  /// and curriculum alignment against [curriculumMapping].
   792→  Future<List<ValidationResultEntity>> validateQuestion(
   793→    GeneratedQuestionEntity question, {
   794→    List<QuestionEntity>? existingQuestions,
   795→    CurriculumMappingEntity? curriculumMapping,
   796→  }) async {
   797→    if (existingQuestions != null && curriculumMapping != null) {
   798→      final results =
   799→          await _validationEngine.validateWithDuplicates(
   800→        question,
   801→        existingQuestions,
   802→      );
   803→      final alignmentResult =
   804→          _validationEngine.checkCurriculumAlignment(question, curriculumMapping);
   805→      if (alignmentResult != null) {
   806→        results.add(alignmentResult);
   807→      }
   808→      return results;
   809→    } else if (existingQuestions != null) {
   810→      return _validationEngine.validateWithDuplicates(
   811→        question,
   812→        existingQuestions,
   813→      );
   814→    } else if (curriculumMapping != null) {
   815→      return _validationEngine.validateWithCurriculum(
   816→        question,
   817→        curriculumMapping,
   818→      );
   819→    }
   820→
   821→    return _validationEngine.validate(question);
   822→  }

   824→  // ─── Document Processing ──────────────────────────────────────────
   825→
   826→  /// Process a document and generate questions from its content.
   827→  ///
   828→  /// 1. Extracts text from the document (if not already extracted)
   829→  /// 2. Uses AI to identify topics and learning objectives
   830→  /// 3. Generates questions based on the extracted content
   831→  Future<DocumentProcessingResult> processDocument(
   832→    DocumentUploadEntity document,
   833→    GenerationInputEntity input, {
   834→    AiProvider? providerOverride,
   835→  }) async {
   836→    final totalStopwatch = Stopwatch()..start();
   837→    var extractionStopwatch = Duration.zero;
   838→
   839→    try {
   840→      final providerType = providerOverride ?? defaultProvider;
   841→      final provider = _getProvider(providerType);
   842→
   843→      if (provider == null) {
   844→        return DocumentProcessingResult(
   845→          document: document,
   846→          error: 'Provider $providerType is not available',
   847→        );
   848→      }
   849→
   850→      // Step 1: Document extraction (if text not yet extracted)
   851→      final String extractedText = document.extractedText ?? '';
   852→
   853→      if (extractedText.isEmpty) {
   854→        // In production, this would call a document parsing service
   855→        // (PDF, DOCX, etc.). For now, we assume text is pre-extracted.
   856→        AppLogger.warning(
   857→            'Document ${document.id} has no extracted text; '
   858→            'skipping extraction step');
   859→      }
   860→
   861→      // Step 2: AI-based topic and objective extraction
   862→      extractionStopwatch = Duration(
   863→          milliseconds: totalStopwatch.elapsed.inMilliseconds,);
   864→
   865→      final extractionPrompt = _promptEngine.buildDocumentExtractionPrompt(
   866→        extractedText,
   867→        input,
   868→      );
   869→
   870→      final extractionRequest = AiCompletionRequest(
   871→        systemPrompt: 'You are an expert educational content analyst. '
   872→            'Analyze the provided document and extract topics, learning '
   873→            'objectives, and key concepts. Always return valid JSON.',
   874→        userPrompt: extractionPrompt,
   875→        temperature: 0.3,
   876→        maxTokens: defaultMaxTokens,
   877→        jsonMode: true,
   878→      );
   879→
   880→      final extractionResult = await provider.complete(extractionRequest);
   881→      final extractionJson = extractionResult.parsedJson ??
   882→          _tryParseContent(extractionResult.content);
   883→
   884→      // Update document with extracted data
   885→      DocumentUploadEntity updatedDoc = document;
   886→      if (extractionJson != null) {
   887→        updatedDoc = document.copyWith(
   888→          extractedText: extractedText,
   889→          identifiedTopics: extractionJson['identified_topics'] != null
   890→              ? List<Map<String, dynamic>>.from(
   891→                  extractionJson['identified_topics'] as List,)
   892→              : document.identifiedTopics,
   893→          suggestedObjectives:
   894→              extractionJson['suggested_objectives'] != null
   895→                  ? List<Map<String, dynamic>>.from(
   896→                      extractionJson['suggested_objectives'] as List,)
   897→                  : document.suggestedObjectives,
   898→          status: DocumentStatus.completed,
   899→          processedAt: DateTime.now(),
   900→        );
   901→      }
   902→
   903→      // Step 3: Generate questions from extracted content
   904→      final generationInput = input.copyWith(
   905→        customInstructions: '${input.customInstructions ?? ''}\n\nBased on the following document content:\n${extractedText.substring(0, extractedText.length > 8000 ? 8000 : extractedText.length)}',
   906→      );
   907→
   908→      // Use a built-in template list (empty = will use defaults)
   909→      final generationResult =
   910→          await generateQuestions(generationInput, []);
   911→
   912→      totalStopwatch.stop();
   913→
   914→      return DocumentProcessingResult(
   915→        document: updatedDoc,
   916→        generationResult: generationResult,
   917→        extractionTime: extractionStopwatch,
   918→        generationTime: totalStopwatch.elapsed - extractionStopwatch,
   919→        totalInputTokens:
   920→            extractionResult.inputTokens + generationResult.totalInputTokens,
   921→        totalOutputTokens:
   922→            extractionResult.outputTokens + generationResult.totalOutputTokens,
   923→      );
   924→    } catch (e) {
   925→      totalStopwatch.stop();
   926→      AppLogger.error('Document processing failed', error: e);
   927→
   928→      return DocumentProcessingResult(
   929→        document: document,
   930→        extractionTime: extractionStopwatch,
   931→        generationTime: totalStopwatch.elapsed - extractionStopwatch,
   932→        error: e.toString(),
   933→      );
   934→    }
   935→  }

  1011→  /// Attempt to parse a content string as JSON.
  1012→  Map<String, dynamic>? _tryParseContent(String content) {
  1013→    try {
  1014→      final decoded = jsonDecode(content);
  1015→      if (decoded is Map<String, dynamic>) return decoded;
  1016→      if (decoded is Map) return Map<String, dynamic>.from(decoded);
  1017→      return null;
  1018→    } on FormatException {
  1019→      // Try extracting JSON from markdown code blocks
  1020→      final jsonBlockRegex = RegExp(r'```(?:json)?\s*([\s\S]*?)```');
  1021→      final match = jsonBlockRegex.firstMatch(content);
  1022→      if (match != null) {
  1023→        try {
  1024→          final decoded = jsonDecode(match.group(1)!.trim());
  1025→          if (decoded is Map<String, dynamic>) return decoded;
  1026→          if (decoded is Map) return Map<String, dynamic>.from(decoded);
  1027→        } on FormatException {
  1028→          // retry failed
  1029→        }
  1030→      }
  1031→
  1032→      // Try finding first { ... } block
  1033→      final braceStart = content.indexOf('{');
  1034→      final braceEnd = content.lastIndexOf('}');
  1035→      if (braceStart != -1 && braceEnd > braceStart) {
  1036→        try {
  1037→          final decoded =
  1038→              jsonDecode(content.substring(braceStart, braceEnd + 1));
  1039→          if (decoded is Map<String, dynamic>) return decoded;
  1040→          if (decoded is Map) return Map<String, dynamic>.from(decoded);
  1041→        } on FormatException {
  1042→          // retry failed
  1043→        }
  1044→      }
  1045→      return null;
  1046→    }
  1047→  }

  1049→  /// Estimate cost based on provider and token usage.
  1050→  double _estimateCost(AiProvider provider, int inputTokens, int outputTokens) {
  1051→    // Cost per 1K tokens (approximate, as of 2024)
  1052→    const costPer1kInput = {
  1053→      'openai': 0.005,    // GPT-4o
  1054→      'gemini': 0.00325,  // Gemini 1.5 Pro
  1055→      'claude': 0.003,    // Claude 3.5 Sonnet
  1056→      'deepseek': 0.001,
  1057→      'grok': 0.005,
  1058→      'local_llm': 0.0,
  1059→    };
  1060→    const costPer1kOutput = {
  1061→      'openai': 0.015,
  1062→      'gemini': 0.00975,
  1063→      'claude': 0.015,
  1064→      'deepseek': 0.002,
  1065→      'grok': 0.015,
  1066→      'local_llm': 0.0,
  1067→    };
  1068→
  1069→    final inputCost = (costPer1kInput[provider.value] ?? 0.005) *
  1070→        (inputTokens / 1000);
  1071→    final outputCost = (costPer1kOutput[provider.value] ?? 0.015) *
  1072→        (outputTokens / 1000);
  1073→
  1074→    return inputCost + outputCost;
  1075→  }
