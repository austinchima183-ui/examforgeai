    25→
    26→/// AI-powered Worksheet Generator page.
    27→///
    28→/// Teachers specify subject, class, topic, worksheet type, difficulty,
    29→/// and question count; the AI generates a complete worksheet with
    30→/// questions, answer key, total marks, and duration.
    31→class WorksheetGeneratorPage extends ConsumerStatefulWidget {
    32→  const WorksheetGeneratorPage({super.key});
    33→
    34→  @override
    35→  ConsumerState<WorksheetGeneratorPage> createState() =>
    36→      _WorksheetGeneratorPageState();
    37→}
    38→
    39→class _WorksheetGeneratorPageState
    40→    extends ConsumerState<WorksheetGeneratorPage> {
    41→  final _formKey = GlobalKey<FormState>();
    42→  final _subjectController = TextEditingController();
    43→  final _classNameController = TextEditingController();
    44→  final _topicController = TextEditingController();
    45→  final _questionCountController = TextEditingController(text: '10');
    46→
    47→  WorksheetType _worksheetType = WorksheetType.classwork;
    48→  String _difficulty = 'medium';
    49→  bool _answerKeyExpanded = false;
    50→  bool _showResult = false;
    51→
    52→  @override
    53→  void dispose() {
    54→    _subjectController.dispose();
    55→    _classNameController.dispose();
    56→    _topicController.dispose();
    57→    _questionCountController.dispose();
    58→    super.dispose();
    59→  }
    60→
    61→  // ─── Form Validation ──────────────────────────────────────────────
    62→
    63→  bool get _isFormValid {
    64→    return _subjectController.text.trim().isNotEmpty;
    65→  }
    66→
    67→  // ─── Generate Worksheet ───────────────────────────────────────────
    68→
    69→  void _generateWorksheet() {
    70→    if (!_formKey.currentState!.validate()) return;
    71→
    72→    final questionCount = int.tryParse(_questionCountController.text.trim()) ?? 10;
    73→
    74→    ref.read(worksheetProvider.notifier).generateWorksheet(
    75→          GenerateWorksheetParams(
    76→            subject: _subjectController.text.trim(),
    77→            className: _classNameController.text.trim().isNotEmpty
    78→                ? _classNameController.text.trim()
    79→                : null,
    80→            topic: _topicController.text.trim().isNotEmpty
    81→                ? _topicController.text.trim()
    82→                : null,
    83→            worksheetType: _worksheetType,
    84→            difficulty: _difficulty,
    85→            questionCount: questionCount,
    86→          ),
    87→        );
    88→    setState(() => _showResult = true);
    89→  }
    90→
    91→  // ─── Save Worksheet ───────────────────────────────────────────────
    92→
    93→  void _saveWorksheet() {
    94→    final worksheet = ref.read(worksheetProvider).currentWorksheet;
    95→    if (worksheet == null) return;
    96→
    97→    ref.read(worksheetProvider.notifier).createWorksheet(
    98→          CreateWorksheetParams(worksheet: worksheet),
    99→        );
   100→  }
   101→
   102→  // ─── Export Worksheet ─────────────────────────────────────────────
   103→
   104→  void _exportWorksheet(String format) {
   105→    final worksheet = ref.read(worksheetProvider).currentWorksheet;
   106→    if (worksheet == null) return;
   107→
   108→    ref.read(worksheetProvider.notifier).exportWorksheet(
   109→          ExportWorksheetParams(
   110→            worksheetId: worksheet.id,
   111→            format: format,
   112→          ),
   113→        );
   114→  }
   115→
   116→  // ─── Build ────────────────────────────────────────────────────────
   117→
   118→  @override
   119→  Widget build(BuildContext context) {
   120→    final cs = context.colorScheme;
   121→    final tt = context.textTheme;
   122→    final state = ref.watch(worksheetProvider);
   123→
   124→    ref.listen<WorksheetState>(worksheetProvider, (prev, next) {
   125→      if (next.error != null && prev?.error != next.error) {
   126→        ScaffoldMessenger.of(context).showSnackBar(
   127→          SnackBar(content: Text(next.error!), backgroundColor: cs.error),
   128→        );
   129→        ref.read(worksheetProvider.notifier).clearError();
   130→      }
   131→      if (next.successMessage != null &&
   132→          prev?.successMessage != next.successMessage) {
   133→        ScaffoldMessenger.of(context).showSnackBar(
   134→          SnackBar(
   135→              content: Text(next.successMessage!), backgroundColor: cs.primary),
   136→        );
   137→        ref.read(worksheetProvider.notifier).clearSuccessMessage();
   138→      }
   139→    });
   140→
   141→    return Scaffold(
   142→      appBar: AppAppBar(
   143→        title: 'AI Worksheet',
   144→      ),
   145→      body: state.isGenerating
   146→          ? Center(
   147→              child: Column(
   148→                mainAxisSize: MainAxisSize.min,
   149→                children: [
   150→                  const AppLoadingSpinner(size: AppLoadingSpinnerSize.large),
   151→                  const SizedBox(height: Spacings.lg),
   152→                  Text(
   153→                    'Generating worksheet...',
   154→                    style: tt.bodyLarge?.copyWith(
   155→                      color: cs.onSurfaceVariant,
   156→                    ),
   157→                  ),
   158→                  const SizedBox(height: Spacings.sm),
   159→                  Text(
   160→                    'This may take a moment',
   161→                    style: tt.bodySmall?.copyWith(
   162→                      color: cs.onSurfaceVariant,
   163→                    ),
   164→                  ),
   165→                ],
   166→              ),
   167→            )
   168→          : SingleChildScrollView(
   169→              padding: Spacings.paddingScreen,
   170→              child: Column(
   171→                crossAxisAlignment: CrossAxisAlignment.stretch,
   172→                children: [
   173→                  // ── Input Form ─────────────────────────────────────
   174→                  if (!_showResult || state.currentWorksheet == null) ...[
   175→                    _buildFormSection(cs, tt, state),
   176→                  ],
   177→
   178→                  // ── Generated Result ──────────────────────────────
   179→                  if (_showResult && state.currentWorksheet != null) ...[
   180→                    _buildResultHeader(cs, tt, state.currentWorksheet!),
   181→                    const SizedBox(height: Spacings.lg),
   182→                    _buildInstructions(cs, tt, state.currentWorksheet!),
   183→                    const SizedBox(height: Spacings.lg),
   184→                    _buildQuestionsList(cs, tt, state.currentWorksheet!),
   185→                    const SizedBox(height: Spacings.lg),
   186→                    _buildAnswerKey(cs, tt, state.currentWorksheet!),
   187→                    const SizedBox(height: Spacings.lg),
   188→                    _buildMarksAndDuration(cs, tt, state.currentWorksheet!),
   189→                    const SizedBox(height: Spacings.xl),
   190→                    _buildActionButtons(cs, state),
   191→                  ],
   192→                ],
   193→              ),
   194→            ),
   195→    );
   196→  }

   468→  // ─── Questions List ───────────────────────────────────────────────
   469→
   470→  Widget _buildQuestionsList(
   471→      ColorScheme cs, TextTheme tt, WorksheetEntity worksheet) {

   473→    if (questions.isEmpty) {
   474→      return AppCard(
   475→        child: Center(

   485→    }
   486→
   487→    return Column(
   488→      crossAxisAlignment: CrossAxisAlignment.start,
   489→      children: [
   490→        Text(
   491→          'Questions (${questions.length})',
   492→          style: tt.titleMedium?.copyWith(

   494→            color: cs.onSurface,
   495→          ),
   496→        ),

   499→          final index = entry.key;
   500→          final question = entry.value;

   502→            padding: const EdgeInsets.only(bottom: Spacings.md),

   505→        }),
   506→      ],
   507→    );

   515→    final options = question['options'] as List<dynamic>?;
   516→
   517→    return AppCard(
   518→      child: Column(
   519→        crossAxisAlignment: CrossAxisAlignment.start,
   520→        children: [
   521→          // Question header row
   522→          Row(
   523→            crossAxisAlignment: CrossAxisAlignment.start,
   524→            children: [

   537→                      fontWeight: AppTypography.wBold,
   538→                    ),
   539→                  ),
   540→                ),

   545→                  text,

   547→                ),
   548→              ),
   549→              if (marks != '') ...[
   550→                const SizedBox(width: Spacings.sm),

   553→                    horizontal: Spacings.sm,
   554→                    vertical: Spacings.xs,

   557→                    color: cs.secondaryContainer,
   558→                    borderRadius: BorderRadius.circular(Spacings.smRadius),

   566→                  ),
   567→                ),
   568→              ],
   569→            ],

   583→                  crossAxisAlignment: CrossAxisAlignment.start,

   586→                      '\u2022 ',
   587→                      style: tt.bodyMedium?.copyWith(
   588→                        color: cs.onSurfaceVariant,
   589→                      ),
   590→                    ),
   591→                    Expanded(
   592→                      child: Text(
   593→                        optionText,
   594→                        style: tt.bodyMedium?.copyWith(
   595→                          color: cs.onSurfaceVariant,
   596→                        ),
   597→                      ),
   598→                    ),
   599→                  ],
   600→                ),
   601→              );
   602→            }),
   603→          ],
   604→        ],
   605→      ),
   606→    );
   607→  }
