    24→  /// The fully-resolved user prompt.

    26→
    27→  /// The template that was used (null if using a built-in default).

    29→
    30→  /// The variables that were resolved and substituted.

    35→// PROMPT ENGINE

    38→/// The Prompt Engine is the brain of the AI question generation system.
    39→///
    40→/// It is responsible for:
    41→/// - Resolving prompt templates with input variables
    42→/// - Selecting the best template for a given generation request
    43→/// - Building generation, improvement, validation, and extraction prompts
    44→/// - Parsing AI responses into structured domain entities
    45→class PromptEngine {
    46→  // ─── Template Resolution ──────────────────────────────────────────
    47→
    48→  /// Resolve a prompt template by replacing `{{variable_name}}` placeholders
    49→  /// with actual values from [variables].
    50→  ///
    51→  /// Required variables that are missing will produce a warning log but
    52→  /// will be replaced with an empty string so generation can continue.
    53→  String resolvePrompt(
    54→    PromptTemplateEntity template,
    55→    Map<String, dynamic> variables,
    56→  ) {
    57→    String resolved = template.userPromptTemplate;
    58→
    59→    // Replace all {{variable_name}} occurrences
    60→    final variablePattern = RegExp(r'\{\{(\w+)\}\}');
    61→    resolved = resolved.replaceAllMapped(variablePattern, (match) {
    62→      final varName = match.group(1)!;
    63→      if (variables.containsKey(varName)) {
    64→        return variables[varName].toString();
    65→      }
    66→      // Check if template declares this variable with a default
    67→      final templateVar = template.variables
    68→          .where((v) => v.name == varName)
    69→          .firstOrNull;
    70→      if (templateVar?.defaultValue != null) {
    71→        return templateVar!.defaultValue!;
    72→      }
    73→      if (templateVar?.isRequired ?? false) {
    74→        AppLogger.warning('Missing required variable: $varName in template '
    75→            '"${template.name}"');
    76→      }
    77→      return '';
    78→    });
    79→
    80→    return resolved;
    81→  }
    82→
    83→  /// Build a generation prompt from teacher inputs.
    84→  ///
    85→  /// Selects the best matching template from [templates] based on
    86→  /// subject, question type, difficulty, curriculum, and language.
    87→  /// If no suitable template is found, falls back to the built-in
    88→  /// default generation prompt.
    89→  PromptResolution resolveGenerationPrompt(
    90→    GenerationInputEntity input,
    91→    List<PromptTemplateEntity> templates,
    92→  ) {
    93→    // ─── AI SECURITY: Check user inputs for prompt injection ──────────
    94→    // Validate the topic, custom instructions, and other user-supplied
    95→    // fields before incorporating them into the prompt.
    96→    final topicCheck = AiSecurityService.checkInput(input.topicId);
    97→    if (!topicCheck.isSafe) {
    98→      AppLogger.error(
    99→        'AI SECURITY: Prompt injection detected in topic. '
   100→        'Blocking generation request.',
   101→      );
   102→      // Return a safe resolution that will result in an error message
   103→      return const PromptResolution(
   104→        systemPrompt: 'You are a helpful educational assistant.',
   105→        userPrompt: 'The provided topic contains potentially unsafe content. '
   106→            'Please revise and try again.',
   107→        templateUsed: null,
   108→        resolvedVariables: {},
   109→      );
   110→    }
   111→
   112→    final instructionsCheck = AiSecurityService.checkInput(
   113→      input.customInstructions ?? '',
   114→    );
   115→    if (!instructionsCheck.isSafe && instructionsCheck.sanitizedInput != null) {
   116→      AppLogger.warning(
   117→        'AI SECURITY: Sanitizing custom instructions due to injection patterns.',
   118→      );
   119→      // Use sanitized version instead of blocking
   120→    }
   121→
   122→    final activeTemplates = templates
   123→        .where((t) =>
   124→            t.isActive &&
   125→            t.promptType == PromptType.questionGeneration,)
   126→        .toList();
   127→
   128→    // Score each template based on how well it matches the input
   129→    PromptTemplateEntity? bestMatch;
   130→    int bestScore = -1;
   131→
   132→    for (final template in activeTemplates) {
   133→      int score = 0;
   134→
   135→      // Match on question type (high weight)
   136→      if (template.questionType == input.questionType) {
   137→        score += 50;
   138→      }
   139→
   140→      // Match on difficulty (medium weight)
   141→      if (template.difficulty == input.difficulty) {
   142→        score += 20;
   143→      }
   144→
   145→      // Match on subject (high weight)
   146→      if (template.subjectId == input.subjectId) {
   147→        score += 30;
   148→      }
   149→
   150→      // Match on curriculum (medium weight)
   151→      if (template.curriculum == input.curriculum) {
   152→        score += 15;
   153→      }
   154→
   155→      // Match on Bloom's level (medium weight)
   156→      if (template.bloomLevel == input.bloomLevel) {
   157→        score += 15;
   158→      }
   159→
   160→      // Match on language (low weight)
   161→      if (template.language == input.language) {
   162→        score += 10;
   163→      }
   164→
   165→      // Match on provider preference (low weight)
   166→      if (template.provider == input.provider) {
   167→        score += 5;
   168→      }
   169→
   170→      // Default templates get a small bonus
   171→      if (template.isDefault) {
   172→        score += 3;
   173→      }
   174→
   175→      // Higher quality score gets a bonus
   176→      if (template.qualityScore != null) {
   177→        score += (template.qualityScore! * 10).round();
   178→      }
   179→
   180→      // Higher success rate gets a bonus
   181→      if (template.successRate != null) {
   182→        score += (template.successRate! * 10).round();
   183→      }
   184→
   185→      if (score > bestScore) {
   186→        bestScore = score;
   187→        bestMatch = template;
   188→      }
   189→    }
   190→
   191→    // Build resolved variables from input
   192→    final resolvedVariables = _buildGenerationVariables(input);
   193→
   194→    if (bestMatch != null) {
   195→      final systemPrompt = bestMatch.systemPrompt;
   196→      final userPrompt = resolvePrompt(bestMatch, resolvedVariables);
   197→
   198→      // Append few-shot examples if available
   199→      final fullUserPrompt = _appendFewShotExamples(userPrompt, bestMatch);
   200→
   201→      // Append chain-of-thought instruction if enabled
   202→      final finalUserPrompt = bestMatch.chainOfThought
   203→          ? '$fullUserPrompt\n\nThink through this step by step before generating the questions.'
   204→          : fullUserPrompt;
   205→
   206→      return PromptResolution(
   207→        systemPrompt: systemPrompt,
   208→        userPrompt: finalUserPrompt,
   209→        templateUsed: bestMatch,
   210→        resolvedVariables: resolvedVariables,
   211→      );
   212→    }
   213→
   214→    // Fallback to built-in default prompt
   215→    return PromptResolution(
   216→      systemPrompt: _defaultSystemPrompt,
   217→      userPrompt: _buildDefaultUserPrompt(input),
   218→      templateUsed: null,
   219→      resolvedVariables: resolvedVariables,
   220→    );
   221→  }

   228→  /// "explanation", "curriculum_alignment", "grammar", "general".

   240→    switch (improvementType) {

   242→        buffer.writeln(
   243→            'Focus on making the question clearer and more unambiguous.',);

   245→        buffer.writeln('- Ensure the question has a single correct interpretation');
   246→        buffer.writeln('- Simplify complex sentence structures while preserving meaning');
   247→      case 'distractors':
   248→        buffer.writeln(
   249→            'Focus on improving the answer options/distractors.',);
   250→        buffer.writeln('- Make incorrect options (distractors) more plausible');
   251→        buffer.writeln('- Ensure distractors represent common misconceptions');
   252→        buffer.writeln('- Make all options roughly the same length');
   253→        buffer.writeln('- Avoid obviously wrong distractors');

   258→            '- Current difficulty: ${question.difficulty.label}',);
   259→        buffer.writeln(
   260→            '- Adjust cognitive demand appropriately',);
   261→        buffer.writeln(
   262→            '- Align with Bloom\'s taxonomy level if specified',);
   263→      case 'explanation':
   264→        buffer.writeln(
   265→            'Focus on creating or improving the explanation for this question.',);
   266→        buffer.writeln('- Provide a clear, step-by-step explanation');
   267→        buffer.writeln('- Explain why the correct answer is right');
   268→        buffer.writeln('- Explain why each distractor is wrong');
   269→        buffer.writeln(
   270→            '- Include relevant references or learning resources',);
   271→      case 'curriculum_alignment':
   272→        buffer.writeln(
   273→            'Focus on improving curriculum alignment.',);
   274→        buffer.writeln(
   275→            '- Ensure the question directly assesses a learning objective',);
   276→        buffer.writeln('- Use curriculum-appropriate terminology');
   277→        buffer.writeln(
   278→            '- Align the difficulty with the expected level for this class',);

   281→            'Focus on correcting grammar and spelling errors.',);
   282→        buffer.writeln('- Fix any grammatical errors');
   283→        buffer.writeln('- Correct spelling mistakes');
   284→        buffer.writeln(
   285→            '- Improve sentence structure while preserving meaning',);
   286→      default:
   287→        buffer.writeln(
   288→            'Improve this question in any way that enhances its quality.',);
   289→    }
   290→
   291→    if (customInstructions != null && customInstructions.isNotEmpty) {
   292→      buffer.writeln();
   293→      buffer.writeln('Additional instructions from the teacher:');
   294→      buffer.writeln(customInstructions);
   295→    }
   296→
   297→    buffer.writeln();
   298→    buffer.writeln('--- ORIGINAL QUESTION ---');
   299→    buffer.writeln('Question Type: ${question.questionType.label}');
   300→    buffer.writeln('Difficulty: ${question.difficulty.label}');
   301→    buffer.writeln('Bloom\'s Level: ${question.bloomLevel?.label ?? "Not specified"}');
   302→    buffer.writeln();
   303→    buffer.writeln('Content: ${question.content}');
   304→
   305→    if (question.answerOptions.isNotEmpty) {
   306→      buffer.writeln();
   307→      buffer.writeln('Answer Options:');
   308→      for (int i = 0; i < question.answerOptions.length; i++) {
   309→        final option = question.answerOptions[i];
   310→        final label = option['label'] as String? ?? String.fromCharCode(65 + i);
   311→        final text = option['text'] as String? ?? option['content'] as String? ?? '';
   312→        final isCorrect = option['is_correct'] as bool? ?? option['isCorrect'] as bool? ?? false;
   313→        buffer.writeln('  $label. $text ${isCorrect ? "✓ (correct)" : ""}');
   314→      }
   315→    }
   316→
   317→    if (question.explanation != null) {
   318→      buffer.writeln();
   319→      buffer.writeln('Current Explanation: ${question.explanation}');
   320→    }
   321→
   322→    buffer.writeln();
   323→    buffer.writeln('--- END ORIGINAL QUESTION ---');
   324→    buffer.writeln();
   325→    buffer.writeln('Return a JSON object with the following structure:');
   326→    buffer.writeln('{');
   327→    buffer.writeln('  "improved_content": "The improved question text",');
   328→    buffer.writeln('  "improved_answer_options": [');
   329→    buffer.writeln(
   330→        '    {"label": "A", "text": "option text", "is_correct": false},',);

   334→    buffer.writeln(
   335→        '  "improvement_notes": "Description of changes made"',);
   336→    buffer.writeln('}');
   337→
   338→    return buffer.toString();
   339→  }

   470→  // ─── Response Parsing ─────────────────────────────────────────────
   471→
   472→  /// Parse AI response into structured generated questions.
   473→  ///
   474→  /// The response should be a JSON object or a list of question objects.
   475→  /// This method handles multiple response formats:
   476→  /// 1. `{ "questions": [...] }` — standard format
   477→  /// 2. `[...]` — direct list of questions
   478→  /// 3. `{ "question": {...} }` — single question wrapper
   479→  List<GeneratedQuestionEntity> parseGeneratedQuestions(
   480→    Map<String, dynamic> response,
   481→    GenerationRequestEntity request,
   482→  ) {
   483→    try {
   484→      List<dynamic> questionsList;
   485→
   486→      if (response.containsKey('questions')) {
   487→        questionsList = response['questions'] as List<dynamic>;
   488→      } else if (response.containsKey('question')) {
   489→        questionsList = [response['question']];
   490→      } else if (response.values.any((v) => v is List)) {
   491→        // Try to find the list of questions
   492→        final listEntry = response.entries
   493→            .firstWhere((e) => e.value is List && (e.value as List).isNotEmpty);
   494→        questionsList = listEntry.value as List<dynamic>;
   495→      } else {
   496→        // Assume the entire response is a single question
   497→        questionsList = [response];
   498→      }
   499→
   500→      return questionsList
   501→          .whereType<Map<String, dynamic>>()
   502→          .map((q) => _mapToGeneratedQuestion(q, request))
   503→          .toList();
   504→    } catch (e) {
   505→      AppLogger.error('Failed to parse generated questions from response', error: e);
   506→      return [];
   507→    }
   508→  }

   541→  /// Parse validation results from AI response.
   542→  List<ValidationResultEntity> parseValidationResults(
   543→    Map<String, dynamic> response,
   544→    String questionId,
   545→  ) {
   546→    try {
   547→      final results = <ValidationResultEntity>[];
   548→
   549→      // If response has validation_results array
   550→      if (response.containsKey('validation_results')) {
   551→        final validationResults =
   552→            response['validation_results'] as List<dynamic>;
   553→        for (int i = 0; i < validationResults.length; i++) {
   554→          final result = validationResults[i] as Map<String, dynamic>;
   555→          results.add(ValidationResultEntity(
   556→            id: '',
   557→            generatedQuestionId: questionId,
   558→            validationType: result['type'] as String? ?? 'general',
   559→            severity: ValidationSeverity.fromString(
   560→                    result['severity'] as String?,) ??
   561→                ValidationSeverity.info,
   562→            message: result['message'] as String? ?? 'No message provided',
   563→            suggestion: result['suggestion'] as String?,
   564→            isResolved: false,
   565→            resolvedBy: null,
   566→            resolvedAt: null,
   567→            createdAt: DateTime.now(),
   568→          ),);
   569→        }
   570→      }
   571→
   572→      // If no structured results, create one from the overall validity
   573→      if (results.isEmpty) {
   574→        final isValid = response['is_valid'] as bool? ?? true;
   575→        results.add(ValidationResultEntity(
   576→          id: '',
   577→          generatedQuestionId: questionId,
   578→          validationType: 'overall',
   579→          severity: isValid ? ValidationSeverity.info : ValidationSeverity.error,
   580→          message: isValid
   581→              ? 'Question passed validation'
   582→              : 'Question failed validation',
   583→          suggestion: response['suggestion'] as String?,
   584→          isResolved: false,
   585→          resolvedBy: null,
   586→          resolvedAt: null,
   587→          createdAt: DateTime.now(),
   588→        ),);
   589→      }
   590→
   591→      return results;
   592→    } catch (e) {
   593→      AppLogger.error('Failed to parse validation results', error: e);
   594→      return [
   595→        ValidationResultEntity(
   596→          id: '',
   597→          generatedQuestionId: questionId,
   598→          validationType: 'parse_error',
   599→          severity: ValidationSeverity.warning,
   600→          message: 'Could not parse validation results from AI response',
   601→          isResolved: false,
   602→          resolvedBy: null,
   603→          resolvedAt: null,
   604→          createdAt: DateTime.now(),
   605→        ),
   606→      ];
   607→    }
   608→  }

   652→  /// Maps a raw question JSON object to a [GeneratedQuestionEntity].
   653→  GeneratedQuestionEntity _mapToGeneratedQuestion(
   654→    Map<String, dynamic> q,
   655→    GenerationRequestEntity request,
   656→  ) {
   657→    // Determine question type
   658→    final questionTypeStr = q['question_type'] as String? ??
   659→        q['type'] as String? ??
   660→        request.inputParams['question_type'] as String? ??
   661→        'multiple_choice';
   662→    final questionType = QuestionType.values.cast<QuestionType?>().firstWhere(
   663→          (t) => t?.value == questionTypeStr,
   664→          orElse: () => QuestionType.multipleChoice,
   665→        ) ?? QuestionType.multipleChoice;
   666→
   667→    // Determine difficulty
   668→    final difficultyStr = q['difficulty'] as String? ??
   669→        request.inputParams['difficulty'] as String? ??
   670→        'medium';
   671→    final difficulty = DifficultyLevel.values.cast<DifficultyLevel?>().firstWhere(
   672→          (d) => d?.value == difficultyStr,
   673→          orElse: () => DifficultyLevel.medium,
   674→        ) ?? DifficultyLevel.medium;
   675→
   676→    // Determine Bloom's level
   677→    final bloomStr = q['bloom_level'] as String? ??
   678→        request.inputParams['bloom_level'] as String?;
   679→    final bloomLevel = BloomTaxonomy.fromString(bloomStr);
   680→
   681→    // Parse answer options
   682→    List<Map<String, dynamic>> answerOptions = [];
   683→    if (q['answer_options'] != null) {
   684→      answerOptions = (q['answer_options'] as List<dynamic>)
   685→          .whereType<Map<String, dynamic>>()
   686→          .toList();
   687→    } else if (q['options'] != null) {
   688→      answerOptions = (q['options'] as List<dynamic>)
   689→          .whereType<Map<String, dynamic>>()
   690→          .toList();
   691→    }
   692→
   693→    // Parse matching pairs
   694→    List<Map<String, dynamic>> matchingPairs = [];
   695→    if (q['matching_pairs'] != null) {
   696→      matchingPairs = (q['matching_pairs'] as List<dynamic>)
   697→          .whereType<Map<String, dynamic>>()
   698→          .toList();
   699→    }
   700→
   701→    // Parse ordering items
   702→    List<Map<String, dynamic>> orderingItems = [];
   703→    if (q['ordering_items'] != null) {
   704→      orderingItems = (q['ordering_items'] as List<dynamic>)
   705→          .whereType<Map<String, dynamic>>()
   706→          .toList();
   707→    }
   708→
   709→    // Parse fill-in-blank answers
   710→    List<Map<String, dynamic>> fillInBlankAnswers = [];
   711→    if (q['fill_in_blank_answers'] != null) {
   712→      fillInBlankAnswers = (q['fill_in_blank_answers'] as List<dynamic>)
   713→          .whereType<Map<String, dynamic>>()
   714→          .toList();
   715→    }
   716→
   717→    // Parse suggested references
   718→    List<String> references = [];
   719→    if (q['references'] != null) {
   720→      references = (q['references'] as List<dynamic>)
   721→          .whereType<String>()
   722→          .toList();
   723→    } else if (q['suggested_references'] != null) {
   724→      references = (q['suggested_references'] as List<dynamic>)
   725→          .whereType<String>()
   726→          .toList();
   727→    }
   728→
   729→    return GeneratedQuestionEntity(
   730→      id: '',
   731→      generationRequestId: request.id,
   732→      questionBankId: null,
   733→      schoolId: request.schoolId,
   734→      questionType: questionType,
   735→      difficulty: difficulty,
   736→      bloomLevel: bloomLevel,
   737→      content: q['content'] as String? ?? q['question'] as String? ?? '',
   738→      contentJson: q['content_json'] as Map<String, dynamic>?,
   739→      answerOptions: answerOptions,
   740→      matchingPairs: matchingPairs,
   741→      orderingItems: orderingItems,
   742→      fillInBlankAnswers: fillInBlankAnswers,
   743→      explanation: q['explanation'] as String?,
   744→      suggestedReferences: references,
   745→      marks: q['marks'] as int? ?? 1,
   746→      estimatedTimeSeconds: q['estimated_time_seconds'] as int?,
   747→      confidenceScore: (q['confidence_score'] as num?)?.toDouble(),
   748→      curriculumAlignment: q['curriculum_alignment'] as Map<String, dynamic>?,
   749→      reviewStatus: ReviewStatus.pending,
   750→      reviewedBy: null,
   751→      reviewedAt: null,
   752→      reviewNotes: null,
   753→      teacherEdits: null,
   754→      isEdited: false,
   755→      isApproved: false,
   756→      metadata: q['metadata'] as Map<String, dynamic>?,
   757→      createdAt: DateTime.now(),
   758→      updatedAt: DateTime.now(),
   759→    );
   760→  }
