   396→  @override
   397→  Future<Result<TransactionEntity>> verifyPayment(String txRef) async {
   398→    try {
   399→      // ─── STEP 1: Look up our local transaction FIRST ────────────────
   400→      // This is critical for security: we read the EXPECTED amount from
   401→      // our own database before verifying with Flutterwave. This prevents
   402→      // an attacker from paying a different amount and claiming success.
   403→      final localTx = await _remoteDataSource.getTransaction(txRef);
   404→      final expectedAmount = localTx.amount;
   405→      final expectedCurrency = localTx.currency;
   406→
   407→      // ─── STEP 2: Verify with Flutterwave, passing expected amounts ─
   408→      // The FlutterwaveDataSource now validates that the actual charged
   409→      // amount matches what we recorded in our database.
   410→      final flutterwaveData = await _flutterwaveDataSource.verifyTransaction(
   411→        txRef,
   412→        expectedAmount: expectedAmount,
   413→        expectedCurrency: expectedCurrency,
   414→      );
   415→
   416→      // ─── STEP 3: Update our local transaction record ───────────────
   417→      final status = _mapFlutterwaveStatus(
   418→        flutterwaveData['status'] as String? ?? 'pending',
   419→      );
   420→
   421→      // SECURITY: If the transaction is successful, also verify that
   422→      // the Flutterwave transaction ID hasn't been used before (prevents
   423→      // replay attacks where the same Flw ID is claimed for two tx_refs).
   424→      if (status == TransactionStatus.successful) {
   425→        final flwTxId = flutterwaveData['id']?.toString();
   426→        if (flwTxId != null && flwTxId.isNotEmpty) {
   427→          final isReplay = await _isFlutterwaveTransactionReplay(flwTxId, localTx.id);
   428→          if (isReplay) {
   429→            AppLogger.error(
   430→              'PAYMENT REPLAY ATTACK DETECTED: Flutterwave transaction ID '
   431→              '$flwTxId has already been used for a different transaction!',
   432→            );
   433→            return const FailureResult(Failure.server(
   434→              message: 'Payment verification failed: duplicate transaction.',
   435→              statusCode: 409,
   436→            ));
   437→          }
   438→        }
   439→      }
   440→
   441→      final updatedTx = await _remoteDataSource.updateTransaction(
   442→        localTx.id,
   443→        {
   444→          'status': status.value,
   445→          'flutterwave_transaction_id': flutterwaveData['id']?.toString(),
   446→          'flutterwave_flw_ref': flutterwaveData['flw_ref'] as String?,
   447→          'flutterwave_fee': flutterwaveData['app_fee'] as double? ?? 0,
   448→          'net_amount': flutterwaveData['amount_settled'] as double? ?? 0,
   449→          'payment_method_summary': flutterwaveData['payment_type'] as String?,
   450→          'processor_response': flutterwaveData,
   451→          'verified_at': DateTime.now().toIso8601String(),
   452→          if (status == TransactionStatus.successful)
   453→            'completed_at': DateTime.now().toIso8601String(),
   454→        },
   455→      );
   456→      return Success(updatedTx.toEntity());
   457→    } on AuthException catch (e) {
   458→      return FailureResult(Failure.auth(message: e.message, code: e.code));
   459→    } on ServerException catch (e) {
   460→      return FailureResult(Failure.server(
   461→        message: e.message, statusCode: e.statusCode, data: e.data,
   462→      ));
   463→    } on NotFoundException catch (e) {
   464→      return FailureResult(Failure.notFound(message: e.message));
   465→    } catch (e) {
   466→      AppLogger.error('Unexpected verifyPayment error', error: e);
   467→      return const FailureResult(Failure.server(
   468→        message: 'Failed to verify payment.', statusCode: 500,
   469→      ));
   470→    }
   471→  }

   473→  /// Checks whether a Flutterwave transaction ID has already been
   474→  /// associated with a DIFFERENT local transaction. This prevents
   475→  /// replay attacks where the same successful payment is claimed
   476→  /// for multiple transaction references.
   477→  Future<bool> _isFlutterwaveTransactionReplay(
   478→    String flwTxId,
   479→    String currentLocalTxId,
   480→  ) async {
   481→    try {
   482→      // Query our transactions table for any other transaction with
   483→      // the same Flutterwave transaction ID that's already successful.
   484→      final existingTxs = await _remoteDataSource.getTransactions(
   485→        status: 'successful',
   486→        page: 1,
   487→        perPage: 5,
   488→      );
   489→      return existingTxs.any((tx) =>
   490→          tx.flutterwaveTransactionId == flwTxId &&
   491→          tx.id != currentLocalTxId);
   492→    } catch (e) {
   493→      // If we can't check, err on the side of caution and allow
   494→      // the transaction through. The amount verification above is
   495→      // the primary defence.
   496→      AppLogger.warning(
   497→        'Could not verify Flutterwave transaction ID uniqueness',
   498→        error: e,
   499→      );
   500→      return false;
   501→    }
   502→  }

   504→  @override
   505→  Future<Result<bool>> processWebhookEvent(Map<String, dynamic> payload) async {
   506→    try {
   507→      final event = payload['event'] as String? ?? '';
   508→      final data = payload['data'] as Map<String, dynamic>? ?? {};
   509→
   510→      // ─── IDEMPOTENCY CHECK (CRITICAL SECURITY FIX) ──────────────────
   511→      // Build an idempotency key from the event type and the unique
   512→      // Flutterwave event/transaction ID. This prevents the same webhook
   513→      // from being processed twice (Flutterwave retries webhooks on
   514→      // failure, so duplicates are expected).
   515→      final flwId = data['id']?.toString() ?? '';
   516→      final idempotencyKey = '${event}_$flwId';
   517→
   518→      final flutterwaveDs = _flutterwaveDataSource;
   519→      if (flutterwaveDs is FlutterwaveDataSourceImpl) {
   520→        if (flutterwaveDs.isWebhookEventProcessed(idempotencyKey)) {
   521→          AppLogger.info(
   522→            'Webhook event already processed (idempotent): $idempotencyKey. '
   523→            'Returning success without re-processing.',
   524→          );
   525→          return const Success(true);
   526→        }
   527→      }
   528→
   529→      AppLogger.info('Processing Flutterwave webhook event: $event');
   530→
   531→      switch (event) {
   532→        case 'charge.completed':
   533→          final txRef = data['tx_ref'] as String?;
   534→          if (txRef != null) {
   535→            // Verify and update the transaction (includes amount
   536→            // verification via the updated verifyPayment method)
   537→            final result = await verifyPayment(txRef);
   538→            if (result is Success) {
   539→              AppLogger.info(
   540→                'Webhook charge.completed processed successfully for $txRef',
   541→              );
   542→            } else {
   543→              AppLogger.warning(
   544→                'Webhook charge.completed verification failed for $txRef',
   545→              );
   546→            }
   547→          }
   548→          break;
   549→        case 'transfer.completed':
   550→        case 'transfer.failed':
   551→          AppLogger.info('Transfer event received: $event');
   552→          break;
   553→        case 'subscription.cancelled':
   554→          final subId = data['subscription_id'] as String?;
   555→          if (subId != null) {
   556→            await _remoteDataSource.updateSubscription(subId, {
   557→              'status': 'cancelled',
   558→              'cancelled_at': DateTime.now().toIso8601String(),
   559→            });
   560→          }
   561→          break;
   562→        default:
   563→          AppLogger.info('Unhandled webhook event: $event');
   564→      }
   565→
   566→      // Mark event as processed AFTER successful handling
   567→      if (flutterwaveDs is FlutterwaveDataSourceImpl) {
   568→        flutterwaveDs.markWebhookEventProcessed(idempotencyKey);
   569→      }
   570→
   571→      return const Success(true);
   572→    } on AuthException catch (e) {
   573→      return FailureResult(Failure.auth(message: e.message, code: e.code));
   574→    } on ServerException catch (e) {
   575→      return FailureResult(Failure.server(
   576→        message: e.message, statusCode: e.statusCode, data: e.data,
   577→      ));
   578→    } catch (e) {

   649→      if (flwTxId == null || flwTxId.isEmpty) {

   651→          message: 'Cannot refund: no Flutterwave transaction ID found.',
   652→        ));

   655→      // Process refund via Flutterwave
   656→      await _flutterwaveDataSource.processRefund(

   661→      // Update local transaction

   663→        transactionId,

   666→          if (reason != null) 'refund_reason': reason,
   667→          'refunded_at': DateTime.now().toIso8601String(),
   668→          'status': amount >= txModel.amount ? 'refunded' : 'partially_refunded',
   669→        },
   670→      );
   671→      return Success(updatedTx.toEntity());
   672→    } on AuthException catch (e) {
   673→      return FailureResult(Failure.auth(message: e.message, code: e.code));
   674→    } on ServerException catch (e) {
