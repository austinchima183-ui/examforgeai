    21→/// The main offline management page with tabbed navigation.
    22→///
    23→/// Provides a central hub for managing offline content including:
    24→/// - **Overview**: Connectivity status, sync health, storage usage
    25→/// - **Resources**: Offline-available resources by type
    26→/// - **Downloads**: Active, completed, and failed downloads
    27→/// - **Drafts**: Saved draft work with edit/delete actions
    28→/// - **Sync**: Pending/failed sync items with retry controls
    29→class OfflineCenterPage extends ConsumerStatefulWidget {
    30→  const OfflineCenterPage({super.key});
    31→
    32→  @override
    33→  ConsumerState<OfflineCenterPage> createState() => _State();
    34→}
    35→
    36→class _State extends ConsumerState<OfflineCenterPage>
    37→    with SingleTickerProviderStateMixin {
    38→  late final TabController _tabController;
    39→
    40→  static const _tabs = [
    41→    Tab(icon: Icon(Icons.dashboard_outlined), text: 'Overview'),
    42→    Tab(icon: Icon(Icons.folder_outlined), text: 'Resources'),
    43→    Tab(icon: Icon(Icons.download_outlined), text: 'Downloads'),
    44→    Tab(icon: Icon(Icons.edit_note_outlined), text: 'Drafts'),
    45→    Tab(icon: Icon(Icons.sync_outlined), text: 'Sync'),

    47→
    48→  @override
    49→  void initState() {
    50→    super.initState();
    51→    _tabController = TabController(length: _tabs.length, vsync: this);
    52→    WidgetsBinding.instance.addPostFrameCallback((_) {
    53→      _loadData();
    54→    });
    55→  }
    56→
    57→  @override
    58→  void dispose() {
    59→    _tabController.dispose();
    60→    super.dispose();
    61→  }
    62→
    63→  /// Loads all offline data in parallel.
    64→  void _loadData() {
    65→    final userId = ref.read(userIdProvider);
    66→    if (userId == null) return;
    67→
    68→    ref.read(offlineProvider.notifier).loadSyncStatus(userId);
    69→    ref.read(offlineProvider.notifier).loadOfflineResources(userId);
    70→    ref.read(offlineProvider.notifier).loadDrafts(userId);
    71→    ref.read(offlineProvider.notifier).loadConnectivityInfo();
    72→    ref.read(offlineProvider.notifier).loadDownloads(userId);
    73→  }
    74→
    75→  @override
    76→  Widget build(BuildContext context) {
    77→    final offlineState = ref.watch(offlineProvider);
    78→
    79→    return Scaffold(
    80→      appBar: AppAppBar(
    81→        title: 'Offline Center',
    82→        bottom: TabBar(
    83→          controller: _tabController,
    84→          tabs: _tabs,
    85→          isScrollable: context.isMobile,
    86→          labelStyle: context.textTheme.labelLarge,
    87→          unselectedLabelStyle: context.textTheme.labelMedium,
    88→          indicatorSize: TabBarIndicatorSize.label,
    89→        ),
    90→      ),
    91→      body: TabBarView(
    92→        controller: _tabController,
    93→        children: [
    94→          _OverviewTab(state: offlineState, onRefresh: _loadData),
    95→          _ResourcesTab(state: offlineState, onRefresh: _loadData),
    96→          _DownloadsTab(state: offlineState, onRefresh: _loadData),
    97→          _DraftsTab(state: offlineState, onRefresh: _loadData),
    98→          _SyncTab(state: offlineState, onRefresh: _loadData),
    99→        ],
   100→      ),
   101→    );
   102→  }
   103→}

   367→  Color _qualityColor(ConnectionQuality? quality) {
   368→    switch (quality) {
   369→      case ConnectionQuality.excellent:
   370→        return AppColors.success;
   371→      case ConnectionQuality.good:
   372→        return AppColors.info;
   373→      case ConnectionQuality.limited:
   374→        return AppColors.warning;
   375→      case ConnectionQuality.offline:
   376→        return AppColors.error;
   377→      case null:
   378→        return AppColors.error;
   379→    }
   380→  }
   381→
   382→  String _formatDateTime(DateTime dt) {
   383→    final now = DateTime.now();
   384→    final diff = now.difference(dt);
   385→    if (diff.inMinutes < 1) return 'Just now';
   386→    if (diff.inMinutes < 60) return '${diff.inMinutes}m ago';
   387→    if (diff.inHours < 24) return '${diff.inHours}h ago';
   388→    return '${dt.day}/${dt.month}/${dt.year}';
   389→  }
   390→
   391→  String _formatBytes(int bytes) {
   392→    if (bytes < 1024) return '$bytes B';
   393→    if (bytes < 1024 * 1024) return '${(bytes / 1024).toStringAsFixed(1)} KB';
   394→    if (bytes < 1024 * 1024 * 1024) {
   395→      return '${(bytes / (1024 * 1024)).toStringAsFixed(1)} MB';
   396→    }
   397→    return '${(bytes / (1024 * 1024 * 1024)).toStringAsFixed(1)} GB';
   398→  }
   399→}

   405→class _ResourcesTab extends ConsumerWidget {
   406→  const _ResourcesTab({required this.state, required this.onRefresh});
   407→
   408→  final OfflineState state;
   409→  final VoidCallback onRefresh;
   410→
   411→  @override
   412→  Widget build(BuildContext context, WidgetRef ref) {
   413→    final cs = context.colorScheme;
   414→    final tt = context.textTheme;
   415→    final resources = state.offlineResources;
   416→
   417→    if (state.isLoading && resources.isEmpty) {
   418→      return const Center(child: AppLoadingSpinner());
   419→    }
   420→
   421→    if (resources.isEmpty) {
   422→      return AppEmptyState.generic(
   423→        icon: Icons.folder_off_outlined,
   424→        message: 'No offline resources',
   425→        subtitle: 'Download resources to access them without internet',
   426→        onAction: onRefresh,
   427→        actionLabel: 'Refresh',
   428→      );
   429→    }
   430→
   431→    // Group resources by type
   432→    final grouped = <String, List<OfflineResource>>{};
   433→    for (final r in resources) {
   434→      grouped.putIfAbsent(r.resourceType, () => []).add(r);
   435→    }
   436→
   437→    return RefreshIndicator(
   438→      onRefresh: () async => onRefresh(),
   439→      child: ListView.builder(
   440→        physics: const AlwaysScrollableScrollPhysics(),
   441→        padding: const EdgeInsets.symmetric(
   442→          horizontal: Spacings.lg,
   443→          vertical: Spacings.md,
   444→        ),
   445→        itemCount: grouped.length,
   446→        itemBuilder: (context, index) {
   447→          final type = grouped.keys.elementAt(index);
   448→          final items = grouped[type]!;
   449→
   450→          return Column(
   451→            crossAxisAlignment: CrossAxisAlignment.start,
   452→            children: [
   453→              // Section header
   454→              Padding(
   455→                padding: const EdgeInsets.only(
   456→                  bottom: Spacings.sm,
   457→                  top: index == 0 ? 0 : Spacings.lg,
   458→                ),
   459→                child: Text(
   460→                  _resourceTypeLabel(type),
   461→                  style: tt.titleSmall?.copyWith(
   462→                    fontWeight: AppTypography.wSemiBold,
   463→                    color: cs.primary,
   464→                  ),
   465→                ),
   466→              ),
   467→              // Resource cards
   468→              ...items.map((resource) => _buildResourceCard(context, ref, resource)),
   469→            ],
   470→          );
   471→        },
   472→      ),
   473→    );
   474→  }

   528→  String _resourceTypeLabel(String type) {
   529→    const labels = {
   530→      'lesson_note': 'Lesson Notes',
   531→      'worksheet': 'Worksheets',
   532→      'study_guide': 'Study Guides',
   533→      'textbook': 'Textbooks',

   535→      'audio': 'Audio',

   539→      (w) => w.isEmpty ? '' : '${w[0].toUpperCase()}${w.substring(1)}',

   548→class _DownloadsTab extends ConsumerWidget {
   549→  const _DownloadsTab({required this.state, required this.onRefresh});
   550→
   551→  final OfflineState state;
   552→  final VoidCallback onRefresh;
   553→
   554→  @override
   555→  Widget build(BuildContext context, WidgetRef ref) {
   556→    final downloads = state.downloads;
   557→
   558→    if (state.isLoading && downloads.isEmpty) {
   559→      return const Center(child: AppLoadingSpinner());
   560→    }
   561→
   562→    if (downloads.isEmpty) {
   563→      return AppEmptyState.generic(
   564→        icon: Icons.download_outlined,
   565→        message: 'No downloads',
   566→        subtitle: 'Files you download for offline use will appear here',
   567→        onAction: onRefresh,
   568→        actionLabel: 'Refresh',
   569→      );
   570→    }
   571→
   572→    // Separate downloads by status
   573→    final active = downloads.where((d) => d.isDownloading || d.downloadStatus == DownloadStatus.pending).toList();
   574→    final completed = downloads.where((d) => d.isComplete).toList();
   575→    final failed = downloads.where((d) => d.canRetry).toList();
   576→    final expired = downloads.where((d) => d.downloadStatus == DownloadStatus.expired).toList();
   577→
   578→    return RefreshIndicator(
   579→      onRefresh: () async => onRefresh(),
   580→      child: SingleChildScrollView(
   581→        physics: const AlwaysScrollableScrollPhysics(),
   582→        padding: const EdgeInsets.symmetric(
   583→          horizontal: Spacings.lg,
   584→          vertical: Spacings.md,
   585→        ),
   586→        child: Column(
   587→          crossAxisAlignment: CrossAxisAlignment.start,
   588→          children: [
   589→            if (active.isNotEmpty) ...[
   590→              _buildDownloadSection(context, 'Active Downloads', active),
   591→              const SizedBox(height: Spacings.lg),
   592→            ],
   593→            if (completed.isNotEmpty) ...[
   594→              _buildDownloadSection(context, 'Completed', completed),
   595→              const SizedBox(height: Spacings.lg),
   596→            ],
   597→            if (failed.isNotEmpty) ...[
   598→              _buildDownloadSection(context, 'Failed', failed),
   599→              const SizedBox(height: Spacings.lg),
   600→            ],
   601→            if (expired.isNotEmpty) ...[
   602→              _buildDownloadSection(context, 'Expired', expired),
   603→            ],
   604→            const SizedBox(height: Spacings.xxl),
   605→          ],
   606→        ),
   607→      ),
   608→    );
   609→  }

   650→class _DraftsTab extends ConsumerWidget {

   652→
   653→  final OfflineState state;

   655→
   656→  @override
   657→  Widget build(BuildContext context, WidgetRef ref) {
   658→    final drafts = state.drafts;

   660→    if (state.isLoading && drafts.isEmpty) {
   661→      return const Center(child: AppLoadingSpinner());

   666→        icon: Icons.edit_note_outlined,

   669→        onAction: onRefresh,
   670→        actionLabel: 'Refresh',
   671→      );
   672→    }
   673→
   674→    return RefreshIndicator(
   675→      onRefresh: () async => onRefresh(),
   676→      child: ListView.builder(
   677→        physics: const AlwaysScrollableScrollPhysics(),
   678→        padding: const EdgeInsets.symmetric(
   679→          horizontal: Spacings.lg,
   680→          vertical: Spacings.md,
   681→        ),
   682→        itemCount: drafts.length,
   683→        itemBuilder: (context, index) {
   684→          final draft = drafts[index];
   685→          return Padding(
   686→            padding: const EdgeInsets.only(bottom: Spacings.sm),
   687→            child: DraftCard(
   688→              draft: draft,
   689→              onEdit: () {
   690→                // Navigate to draft editor
   691→              },
   692→              onDelete: () {
   693→                ref.read(offlineProvider.notifier).deleteDraft(draft.id);
   694→              },
   695→            ),
   696→          );
   697→        },
   698→      ),
   699→    );
   700→  }
