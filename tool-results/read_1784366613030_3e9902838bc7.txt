   382→  /// Reorders questions according to the given list of question IDs.
   383→  void reorderQuestions(List<String> questionIds) {
   384→    final reordered = <ExamQuestionEntity>[];
   385→    for (var i = 0; i < questionIds.length; i++) {
   386→      final question = state.questions
   387→          .where((q) => q.questionId == questionIds[i])
   388→          .firstOrNull;
   389→      if (question != null) {
   390→        reordered.add(question.copyWith(sortOrder: i));
   391→      }
   392→    }
   393→    state = state.copyWith(questions: reordered);
   394→    AppLogger.info('Questions reordered: ${questionIds.length} items');
   395→  }

   461→  /// Saves the exam — creates if new, updates if editing.
   462→  Future<void> saveExam() async {
   463→    if (state.isSaving) return;
   464→
   465→    state = state.copyWith(isSaving: true, error: null);
   466→

   474→  Future<void> _createExam() async {
   475→    if (state.startTime == null || state.endTime == null) {
   476→      state = state.copyWith(
   477→        isSaving: false,
   478→        error: 'Start time and end time are required',
   479→      );
   480→      return;
   481→    }
   482→
   483→    final input = ExamCreateInput(
   484→      title: state.title,
   485→      description:
   486→          state.description.trim().isEmpty ? null : state.description,
   487→      subjectId: state.subjectId ?? '',
   488→      classId: state.classId ?? '',
   489→      academicSessionId: state.academicSessionId ?? '',
   490→      examType: state.examType,
   491→      startTime: state.startTime!,
   492→      endTime: state.endTime!,
   493→      timeLimitMinutes: state.timeLimitMinutes,
   494→      passMark: state.passMark,
   495→      passMarkType: state.passMarkType,
   496→      instructions:
   497→          state.instructions.trim().isEmpty ? null : state.instructions,
   498→      allowedAttempts: state.allowedAttempts,
   499→      negativeMarkingEnabled: state.negativeMarkingEnabled,
   500→      negativeMarkValue: state.negativeMarkValue,
   501→      gracePeriodMinutes: state.gracePeriodMinutes,
   502→      autoSubmit: state.autoSubmit,
   503→      randomizeQuestions: state.randomizeQuestions,
   504→      randomizeOptions: state.randomizeOptions,
   505→      showResults: state.showResults,
   506→      showCorrectAnswers: state.showCorrectAnswers,
   507→      showExplanations: state.showExplanations,
   508→      maxStudents: state.maxStudents,
   509→      requireFullScreen: state.requireFullScreen,
   510→      allowResume: state.allowResume,
   511→      browserLockdown: state.browserLockdown,
   512→    );
   513→
   514→    final result = await _createExamUseCase(
   515→      CreateExamParams(
   516→        exam: input,
   517→        sections: state.sections,
   518→        questions: state.questions,
   519→        studentIds: state.assignedStudents,
   520→      ),
   521→    );
   522→
   523→    result.fold(
   524→      onSuccess: (exam) {
   525→        state = state.copyWith(
   526→          isSaving: false,
   527→          exam: exam,
   528→          status: exam.status,
   529→          successMessage: 'Exam created successfully',
   530→          error: null,
   531→        );
   532→        AppLogger.info('Exam created: ${exam.id}');
   533→      },
   534→      onFailure: (failure) {
   535→        state = state.copyWith(
   536→          isSaving: false,
   537→          error: _mapFailureToMessage(failure),
   538→        );
   539→        AppLogger.warning('Failed to create exam: $failure');
   540→      },
   541→    );
   542→  }

   554→          state.academicSessionId ?? state.exam!.academicSessionId,

   556→      startTime: state.startTime ?? state.exam!.startTime,
   557→      endTime: state.endTime ?? state.exam!.endTime,
   558→      timeLimitMinutes: state.timeLimitMinutes,
   559→      totalMarks: state.totalMarks,
   560→      passMark: state.passMark,
   561→      passMarkType: state.passMarkType,
   562→      instructions:
   563→          state.instructions.trim().isEmpty ? null : state.instructions,
   564→      allowedAttempts: state.allowedAttempts,
   565→      negativeMarkingEnabled: state.negativeMarkingEnabled,
   566→      negativeMarkValue: state.negativeMarkValue,
   567→      gracePeriodMinutes: state.gracePeriodMinutes,
   568→      autoSubmit: state.autoSubmit,
   569→      randomizeQuestions: state.randomizeQuestions,
   570→      randomizeOptions: state.randomizeOptions,

   573→      showExplanations: state.showExplanations,

   575→      requireFullScreen: state.requireFullScreen,
   576→      allowResume: state.allowResume,
   577→      browserLockdown: state.browserLockdown,
   578→      sections: state.sections,
   579→      questions: state.questions,
   580→      updatedAt: DateTime.now(),
   581→    );
   582→
   583→    final result = await _updateExamUseCase(
   584→      UpdateExamParams(
   585→        exam: updatedExam,
   586→        sections: state.sections,
   587→        questions: state.questions,
   588→        studentIds: state.assignedStudents,
   589→      ),
   590→    );
   591→
   592→    result.fold(
   593→      onSuccess: (exam) {
   594→        state = state.copyWith(
   595→          isSaving: false,
   596→          exam: exam,
   597→          status: exam.status,
   598→          successMessage: 'Exam updated successfully',
   599→          error: null,
   600→        );
   601→        AppLogger.info('Exam updated: ${exam.id}');
   602→      },
   603→      onFailure: (failure) {
   604→        state = state.copyWith(
   605→          isSaving: false,
   606→          error: _mapFailureToMessage(failure),
   607→        );
   608→        AppLogger.warning('Failed to update exam: $failure');
   609→      },
   610→    );
   611→  }
   612→
   613→  // ═══════════════════════════════════════════════════════════════════════
   614→  // PUBLISH EXAM
   615→  // ═══════════════════════════════════════════════════════════════════════
   616→
   617→  /// Publishes the exam, making it visible to assigned students.
   618→  ///
   619→  /// Must be called after saving the exam.
   620→  Future<void> publishExam() async {
   621→    if (state.isPublishing) return;
   622→    if (state.exam == null) {
   623→      state = state.copyWith(error: 'Please save the exam first');
   624→      return;
   625→    }
   626→
   627→    state = state.copyWith(isPublishing: true, error: null);
   628→
   629→    final result = await _manageExamStatusUseCase(
   630→      ManageStatusParams(
   631→        examId: state.exam!.id,
   632→        action: ManageStatusAction.publish,
   633→      ),
   634→    );
   635→
   636→    result.fold(
   637→      onSuccess: (exam) {
   638→        state = state.copyWith(
   639→          isPublishing: false,
   640→          exam: exam,
   641→          status: exam.status,
   642→          successMessage: 'Exam published successfully',
   643→          error: null,
   644→        );
   645→        AppLogger.info('Exam published: ${exam.id}');
   646→      },
   647→      onFailure: (failure) {
   648→        state = state.copyWith(
   649→          isPublishing: false,
   650→          error: _mapFailureToMessage(failure),
   651→        );
   652→        AppLogger.warning('Failed to publish exam: $failure');
   653→      },
   654→    );
   655→  }
   656→

   661→  /// Creates a deep copy of the current exam with a new ID and draft status.
   662→  Future<void> cloneExam() async {

   684→      },
   685→      onFailure: (failure) {

   687→          isSaving: false,
   688→          error: _mapFailureToMessage(failure),

   781→  /// Maps a [Failure] to a user-friendly error message.
   782→  String _mapFailureToMessage(Failure failure) {
   783→    return failure.when(
   784→      server: (message, _, __) => message,
   785→      cache: (message) => message,
   786→      auth: (message, _) => message,
   787→      network: (message) => message,
   788→      validation: (message, _) => message,
   789→      notFound: (message) => message,
   790→      unauthorized: (message) => message,
   791→      forbidden: (message) => message,
   792→    );
   793→  }
