    31→/// Concrete implementation of [OfflineRepository] that coordinates
    32→/// between local and remote data sources with an offline-first strategy.
    33→///
    34→/// When the device is online:
    35→///   - Read operations try the remote source first and cache the result.
    36→///   - Write operations go to the remote source and update the local cache.
    37→///
    38→/// When the device is offline:
    39→///   - Read operations fall back to the local data source.
    40→///   - Write operations are persisted locally and enqueued in the sync
    41→///     queue for automatic replay when connectivity is restored.
    42→class OfflineRepositoryImpl implements OfflineRepository {
    43→  OfflineRepositoryImpl({
    44→    required OfflineLocalDataSource localDataSource,
    45→    required OfflineRemoteDataSource remoteDataSource,
    46→    required NetworkInfo networkInfo,
    47→    required CacheManager cacheManager,
    48→    required SyncEngine syncEngine,
    49→  })  : _localDataSource = localDataSource,
    50→        _remoteDataSource = remoteDataSource,
    51→        _networkInfo = networkInfo,
    52→        _cacheManager = cacheManager,
    53→        _syncEngine = syncEngine;
    54→
    55→  final OfflineLocalDataSource _localDataSource;
    56→  final OfflineRemoteDataSource _remoteDataSource;
    57→  final NetworkInfo _networkInfo;
    58→  final CacheManager _cacheManager;
    59→  final SyncEngine _syncEngine;

    61→  // ═══════════════════════════════════════════════════════════════════════
    62→  // Helper: Convert exceptions to Failures
    63→  // ═══════════════════════════════════════════════════════════════════════
    64→
    65→  Failure _mapExceptionToFailure(Object e) {
    66→    if (e is AuthException) {
    67→      return Failure.auth(message: e.message, code: e.code);
    68→    } else if (e is ServerException) {
    69→      return Failure.server(
    70→        message: e.message,
    71→        statusCode: e.statusCode,
    72→        data: e.data,
    73→      );
    74→    } else if (e is CacheException) {
    75→      return Failure.cache(message: e.message);
    76→    } else if (e is NetworkException) {
    77→      return Failure.network(message: e.message);
    78→    } else if (e is ValidationException) {
    79→      return Failure.validation(
    80→        message: e.message,
    81→        fieldErrors: e.fieldErrors,
    82→      );
    83→    } else if (e is NotFoundException) {
    84→      return Failure.notFound(message: e.message);
    85→    } else if (e is UnauthorizedException) {
    86→      return Failure.unauthorized(message: e.message);
    87→    } else if (e is ForbiddenException) {
    88→      return Failure.forbidden(message: e.message);
    89→    } else {
    90→      AppLogger.error(
    91→        'Unexpected exception in OfflineRepositoryImpl',
    92→        error: e,
    93→      );
    94→      return Failure.server(
    95→        message: 'An unexpected error occurred: $e',
    96→        statusCode: 500,
    97→      );
    98→    }
    99→  }

   118→  @override
   119→  Future<Result<bool>> triggerSync(String userId) async {
   120→    try {
   121→      final isOnline = await _networkInfo.isConnected;
   122→      if (!isOnline) {
   123→        return const FailureResult(
   124→          Failure.network(message: 'No network connection available'),
   125→        );
   126→      }
   127→
   128→      // Process the sync queue via SyncEngine.
   129→      await _syncEngine.processQueue();
   130→      return const Success(true);
   131→    } on NetworkException catch (e) {
   132→      return FailureResult(Failure.network(message: e.message));
   133→    } on ServerException catch (e) {
   134→      return FailureResult(Failure.server(
   135→        message: e.message,
   136→        statusCode: e.statusCode,
   137→        data: e.data,
   138→      ));
   139→    } catch (e) {
   140→      return FailureResult(_mapExceptionToFailure(e));
   141→    }
   142→  }

   167→  @override
   168→  Future<Result<bool>> downloadResourceForOffline(
   169→    String userId,
   170→    String resourceType,
   171→    String resourceId,
   172→  ) async {
   173→    try {
   174→      final isOnline = await _networkInfo.isConnected;
   175→
   176→      if (isOnline) {
   177→        // Get the download URL from the server.
   178→        final downloadUrl = await _remoteDataSource.getFileDownloadUrl(
   179→          resourceType,
   180→          resourceId,
   181→        );
   182→
   183→        // Save the resource metadata locally.
   184→        await _localDataSource.saveOfflineResource({
   185→          'user_id': userId,
   186→          'resource_type': resourceType,
   187→          'resource_id': resourceId,
   188→          'title': '$resourceType - $resourceId',
   189→          'file_path': downloadUrl,
   190→          'file_size_bytes': 0, // Updated when download completes
   191→          'mime_type': 'application/octet-stream',
   192→        });
   193→
   194→        // Enqueue the actual download in the sync queue.
   195→        await _cacheManager.enqueueSync(
   196→          userId: userId,
   197→          tableName: 'resources',
   198→          operation: 'download',
   199→          payload: {
   200→            'resource_type': resourceType,
   201→            'resource_id': resourceId,
   202→            'download_url': downloadUrl,
   203→          },
   204→          priority: 3,
   205→        );
   206→
   207→        return const Success(true);
   208→      } else {
   209→        // Offline: queue the download for later.
   210→        await _cacheManager.enqueueSync(
   211→          userId: userId,
   212→          tableName: 'resources',
   213→          operation: 'download',
   214→          payload: {
   215→            'resource_type': resourceType,
   216→            'resource_id': resourceId,
   217→          },
   218→          priority: 3,
   219→        );
   220→
   221→        return const Success(true);
   222→      }
   223→    } on ServerException catch (e) {
   224→      return FailureResult(Failure.server(
   225→        message: e.message,
   226→        statusCode: e.statusCode,
   227→        data: e.data,
   228→      ));
   229→    } on CacheException catch (e) {
   230→      return FailureResult(Failure.cache(message: e.message));
   231→    } catch (e) {
   232→      return FailureResult(_mapExceptionToFailure(e));
   233→    }
   234→  }

   270→  @override
   271→  Future<Result<List<OfflineExamConfig>>> getOfflineExamConfigs(
   272→    String schoolId,
   273→  ) async {
   274→    try {
   275→      final isOnline = await _networkInfo.isConnected;
   276→
   277→      if (isOnline) {
   278→        // Try remote first.
   279→        try {
   280→          final remoteConfigs =
   281→              await _remoteDataSource.getOfflineExamConfigs(schoolId);
   282→
   283→          final configs = remoteConfigs
   284→              .map((data) => OfflineExamConfig(
   285→                    id: data['id'] as String,
   286→                    schoolId: data['school_id'] as String,
   287→                    examId: data['exam_id'] as String,
   288→                    allowsOffline: data['allows_offline'] as bool? ?? false,
   289→                    offlineType: OfflineType.fromString(
   290→                            data['offline_type'] as String?) ??
   291→                        OfflineType.none,
   292→                    maxOfflineAttempts:
   293→                        data['max_offline_attempts'] as int? ?? 0,
   294→                    requiresOnlineSubmission:
   295→                        data['requires_online_submission'] as bool? ?? true,
   296→                    autoSubmitOnReconnect:
   297→                        data['auto_submit_on_reconnect'] as bool? ?? false,
   298→                    createdAt: data['created_at'] != null
   299→                        ? DateTime.parse(data['created_at'] as String)
   300→                        : DateTime.now(),
   301→                  ))
   302→              .toList();
   303→
   304→          // Cache locally.
   305→          await _cacheManager.cacheData(
   306→            key: 'offline_exam_configs_$schoolId',
   307→            userId: schoolId,
   308→            resourceType: 'offline_exam_config',
   309→            resourceId: schoolId,
   310→            data: {'configs': remoteConfigs},
   311→          );
   312→
   313→          return Success(configs);
   314→        } on ServerException {
   315→          // Fall back to local cache.
   316→          AppLogger.warning(
   317→            '[OfflineRepo] Remote configs fetch failed, falling back to cache',
   318→          );
   319→        }
   320→      }
   321→
   322→      // Read from local cache.
   323→      final cached = await _cacheManager.getCachedData(
   324→        key: 'offline_exam_configs_$schoolId',
   325→        userId: schoolId,
   326→      );
   327→
   328→      if (cached != null && cached['configs'] != null) {
   329→        final configsList =
   330→            List<Map<String, dynamic>>.from(cached['configs'] as List);
   331→        final configs = configsList
   332→            .map((data) => OfflineExamConfig(
   333→                  id: data['id'] as String,
   334→                  schoolId: data['school_id'] as String,
   335→                  examId: data['exam_id'] as String,
   336→                  allowsOffline: data['allows_offline'] as bool? ?? false,
   337→                  offlineType:
   338→                      OfflineType.fromString(data['offline_type'] as String?) ??
   339→                          OfflineType.none,
   340→                  maxOfflineAttempts: data['max_offline_attempts'] as int? ?? 0,
   341→                  requiresOnlineSubmission:
   342→                      data['requires_online_submission'] as bool? ?? true,
   343→                  autoSubmitOnReconnect:
   344→                      data['auto_submit_on_reconnect'] as bool? ?? false,
   345→                  createdAt: data['created_at'] != null
   346→                      ? DateTime.parse(data['created_at'] as String)
   347→                      : DateTime.now(),
   348→                ))
   349→            .toList();
   350→        return Success(configs);
   351→      }
   352→
   353→      return const Success([]);
   354→    } on CacheException catch (e) {
   355→      return FailureResult(Failure.cache(message: e.message));
   356→    } catch (e) {
   357→      return FailureResult(_mapExceptionToFailure(e));
   358→    }
   359→  }

   421→  @override
   422→  Future<Result<OfflineExamAttempt>> saveOfflineExamAttempt(
   423→    OfflineExamAttempt attempt,
   424→  ) async {
   425→    try {
   426→      // Always save locally first.
   427→      final savedAttempt = await _localDataSource.saveExamAttempt({
   428→        'id': attempt.id,
   429→        'exam_id': attempt.examId,
   430→        'student_id': attempt.studentId,
   431→        'school_id': attempt.schoolId,
   432→        'attempt_data': attempt.attemptData,
   433→        'answers': attempt.answers,
   434→        'started_at': attempt.startedAt.toIso8601String(),
   435→        'completed_at': attempt.completedAt?.toIso8601String(),
   436→        'time_taken_seconds': attempt.timeTakenSeconds,
   437→        'integrity_hash': attempt.integrityHash,
   438→      });
   439→
   440→      // Try to sync immediately if online.
   441→      final isOnline = await _networkInfo.isConnected;
   442→      if (isOnline && attempt.completedAt != null) {
   443→        try {
   444→          await _remoteDataSource.syncExamAttempt(
   445→            attempt.id,
   446→            {
   447→              'exam_id': attempt.examId,
   448→              'student_id': attempt.studentId,
   449→              'school_id': attempt.schoolId,
   450→              'attempt_data': attempt.attemptData,
   451→              'answers': attempt.answers,
   452→              'started_at': attempt.startedAt.toIso8601String(),
   453→              'completed_at': attempt.completedAt?.toIso8601String(),
   454→              'time_taken_seconds': attempt.timeTakenSeconds,
   455→              'integrity_hash': attempt.integrityHash,
   456→            },
   457→          );
   458→
   459→          // Mark as synced locally.
   460→          await _localDataSource.markExamAttemptSynced(attempt.id);
   461→
   462→          return Success(savedAttempt.copyWith(
   463→            syncStatus: AttemptSyncStatus.synced,
   464→            syncedAt: DateTime.now(),
   465→          ));
   466→        } on ServerException catch (e) {
   467→          // Sync failed — keep as pending, queue for later.
   468→          AppLogger.warning(
   469→            '[OfflineRepo] Exam attempt sync failed, queued for later: ${e.message}',
   470→          );
   471→
   472→          await _cacheManager.enqueueSync(
   473→            userId: attempt.studentId,
   474→            tableName: 'offline_exam_attempts',
   475→            operation: 'sync',
   476→            payload: {
   477→              'attempt_id': attempt.id,
   478→              'exam_id': attempt.examId,
   479→              'student_id': attempt.studentId,
   480→              'school_id': attempt.schoolId,
   481→              'attempt_data': attempt.attemptData,
   482→              'answers': attempt.answers,
   483→              'integrity_hash': attempt.integrityHash,
   484→            },
   485→            recordId: attempt.id,
   486→            priority: 2, // High priority
   487→          );
   488→        }
   489→      } else if (attempt.completedAt != null) {
   490→        // Offline — queue for sync.
   491→        await _cacheManager.enqueueSync(
   492→          userId: attempt.studentId,
   493→          tableName: 'offline_exam_attempts',
   494→          operation: 'sync',
   495→          payload: {
   496→            'attempt_id': attempt.id,
   497→            'exam_id': attempt.examId,
   498→            'student_id': attempt.studentId,
   499→            'school_id': attempt.schoolId,
   500→            'attempt_data': attempt.attemptData,
   501→            'answers': attempt.answers,
   502→            'integrity_hash': attempt.integrityHash,
   503→          },
   504→          recordId: attempt.id,
   505→          priority: 2,
   506→        );
   507→      }
   508→
   509→      return Success(savedAttempt);
   510→    } on CacheException catch (e) {
   511→      return FailureResult(Failure.cache(message: e.message));
   512→    } catch (e) {
   513→      return FailureResult(_mapExceptionToFailure(e));
   514→    }
   515→  }

   532→  @override
   533→  Future<Result<bool>> syncExamAttempt(String attemptId) async {
   534→    try {
   535→      final isOnline = await _networkInfo.isConnected;
   536→      if (!isOnline) {
   537→        return const FailureResult(
   538→          Failure.network(message: 'Cannot sync while offline'),
   539→        );
   540→      }
   541→
   542→      // Get the attempt from local storage.
   543→      final pendingAttempts =
   544→          await _localDataSource.getPendingExamAttempts('');
   545→      final attempt = pendingAttempts.where((a) => a.id == attemptId).firstOrNull;
   546→
   547→      if (attempt == null) {
   548→        return FailureResult(
   549→          Failure.notFound(message: 'Exam attempt not found: $attemptId'),
   550→        );
   551→      }
   552→
   553→      // Sync with server.
   554→      await _remoteDataSource.syncExamAttempt(
   555→        attemptId,
   556→        {
   557→          'exam_id': attempt.examId,
   558→          'student_id': attempt.studentId,
   559→          'school_id': attempt.schoolId,
   560→          'attempt_data': attempt.attemptData,
   561→          'answers': attempt.answers,
   562→          'started_at': attempt.startedAt.toIso8601String(),
   563→          'completed_at': attempt.completedAt?.toIso8601String(),
   564→          'time_taken_seconds': attempt.timeTakenSeconds,
   565→          'integrity_hash': attempt.integrityHash,
   566→        },
   567→      );
   568→
   569→      // Mark as synced locally.
   570→      await _localDataSource.markExamAttemptSynced(attemptId);
   571→
   572→      return const Success(true);
   573→    } on ServerException catch (e) {
   574→      return FailureResult(Failure.server(
   575→        message: e.message,
   576→        statusCode: e.statusCode,
   577→        data: e.data,
   578→      ));
   579→    } on CacheException catch (e) {
   580→      return FailureResult(Failure.cache(message: e.message));
   581→    } catch (e) {
   582→      return FailureResult(_mapExceptionToFailure(e));
   583→    }
   584→  }

   652→              'content': draft.content,
   653→            },

   657→        }

   978→  Future<Result<bool>> syncAnalyticsEvents(String userId) async {
   979→    try {
   980→      final isOnline = await _networkInfo.isConnected;
   981→      if (!isOnline) {
   982→        return const FailureResult(
   983→          Failure.network(message: 'Cannot sync analytics while offline'),
   984→        );
   985→      }
   986→
   987→      // Get pending analytics sync items.
   988→      final pendingItems = await _cacheManager.getPendingSyncItems(
   989→        userId: userId,
   990→      );
   991→
   992→      final analyticsItems = pendingItems
   993→          .where((item) => item['tableName'] == 'app_usage_events')
   994→          .map((item) => item['payload'] as Map<String, dynamic>)
   995→          .toList();
   996→
   997→      if (analyticsItems.isEmpty) {
   998→        return const Success(true);
   999→      }
  1000→
  1001→      final success =
  1002→          await _remoteDataSource.logUsageEvents(analyticsItems);
  1003→
  1004→      // Mark synced items as completed.
  1005→      for (final item in pendingItems
  1006→          .where((item) => item['tableName'] == 'app_usage_events')) {
  1007→        await _cacheManager.markSyncItemCompleted(id: item['id'] as String);
  1008→      }
  1009→
  1010→      return Success(success);
  1011→    } on ServerException catch (e) {
  1012→      return FailureResult(Failure.server(
  1013→        message: e.message,
  1014→        statusCode: e.statusCode,
  1015→        data: e.data,
  1016→      ));
  1017→    } on CacheException catch (e) {
  1018→      return FailureResult(Failure.cache(message: e.message));
  1019→    } catch (e) {
  1020→      return FailureResult(_mapExceptionToFailure(e));
  1021→    }
  1022→  }
