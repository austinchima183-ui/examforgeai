    14→/// Displays a subscription plan with pricing, features, and selection state.
    15→///

    17→/// a list of features, a "Most Popular" badge when [SubscriptionPlanEntity.isPopular]
    18→/// is true, and either a select or current-plan button at the bottom.
    19→///
    20→/// The card gets a coloured border when [isSelected] is true.
    21→class PlanCard extends StatelessWidget {
    22→  const PlanCard({
    23→    super.key,
    24→    required this.plan,
    25→    this.isSelected = false,
    26→    this.isCurrentPlan = false,
    27→    this.billingCycle = 'monthly',
    28→    this.onSelect,
    29→  });

    31→  final SubscriptionPlanEntity plan;
    32→  final bool isSelected;
    33→  final bool isCurrentPlan;
    34→  final String billingCycle;
    35→  final VoidCallback? onSelect;
    36→

    38→  Widget build(BuildContext context) {
    39→    final cs = context.colorScheme;
    40→    final tt = context.textTheme;
    41→    final isDark = context.isDarkMode;
    42→    final price = plan.priceForCycle(billingCycle);
    43→

    45→      elevation: isSelected ? Spacings.elevationMd : Spacings.elevationSm,
    46→      shadowColor: cs.shadow.withValues(alpha: 0.08),

    48→        borderRadius: Spacings.borderRadiusLg,
    49→        side: isSelected
    50→            ? BorderSide(color: cs.primary, width: 2)

    52→                color: cs.outlineVariant.withValues(alpha: 0.5),
    53→              ),
    54→      ),
    55→      clipBehavior: Clip.antiAlias,
    56→      child: Stack(
    57→        children: [
    58→          // ── Content ──────────────────────────────────────────────────
    59→          Padding(
    60→            padding: const EdgeInsets.all(Spacings.lg),
    61→            child: Column(
    62→              crossAxisAlignment: CrossAxisAlignment.start,
    63→              children: [
    64→                // ── Header Row: Name + Tier Badge ────────────────────
    65→                Row(
    66→                  children: [
    67→                    Expanded(
    68→                      child: Text(
    69→                        plan.name,
    70→                        style: tt.titleLarge?.copyWith(
    71→                          fontWeight: AppTypography.wSemiBold,
    72→                          color: cs.onSurface,
    73→                        ),
    74→                        maxLines: 1,
    75→                        overflow: TextOverflow.ellipsis,
    76→                      ),
    77→                    ),
    78→                    const SizedBox(width: Spacings.sm),
    79→                    PlanTierBadge(tier: plan.tier),

    81→                ),
    82→                const SizedBox(height: Spacings.sm),
    83→
    84→                // ── Description ───────────────────────────────────────
    85→                if (plan.description != null) ...[
    86→                  Text(
    87→                    plan.description!,
    88→                    style: tt.bodySmall?.copyWith(
    89→                      color: cs.onSurfaceVariant,
    90→                    ),
    91→                    maxLines: 2,
    92→                    overflow: TextOverflow.ellipsis,
    93→                  ),
    94→                  const SizedBox(height: Spacings.md),
    95→                ],
    96→
    97→                // ── Price ─────────────────────────────────────────────
    98→                Row(
    99→                  crossAxisAlignment: CrossAxisAlignment.baseline,
   100→                  textBaseline: TextBaseline.alphabetic,
   101→                  children: [
   102→                    Text(
   103→                      plan.isFree
   104→                          ? 'Free'
   105→                          : _formatCurrency(price, plan.currency),
   106→                      style: tt.headlineMedium?.copyWith(
   107→                        fontWeight: AppTypography.wBold,
   108→                        color: cs.onSurface,
   109→                        height: 1.0,
   110→                      ),
   111→                    ),
   112→                    if (!plan.isFree) ...[
   113→                      const SizedBox(width: Spacings.xs),
   114→                      Text(
   115→                        '/${billingCycle == 'annual' ? 'yr' : 'mo'}',
   116→                        style: tt.bodyMedium?.copyWith(
   117→                          color: cs.onSurfaceVariant,
   118→                        ),
   119→                      ),
   120→                    ],
   121→                  ],
   122→                ),
   123→

   126→                    plan.annualSavingsPercent > 0) ...[
   127→                  const SizedBox(height: Spacings.xs),
   128→                  Container(
   129→                    padding: const EdgeInsets.symmetric(
   130→                      horizontal: Spacings.sm,
   131→                      vertical: Spacings.xs,

   133→                    decoration: BoxDecoration(
   134→                      color: AppColors.success

   139→                      'Save ${plan.annualSavingsPercent.toStringAsFixed(0)}%',
   140→                      style: tt.labelSmall?.copyWith(
   141→                        color: isDark ? AppColors.successDark : AppColors.success,
   142→                        fontWeight: AppTypography.wSemiBold,

   144→                    ),
   145→                  ),
   146→                ],
   147→
   148→                const SizedBox(height: Spacings.lg),
   149→
   150→                // ── Features List ─────────────────────────────────────
   151→                ...plan.featuresList.map(
   152→                  (feature) => Padding(
   153→                    padding: const EdgeInsets.only(bottom: Spacings.sm),
   154→                    child: Row(
   155→                      children: [
   156→                        Icon(
   157→                          Icons.check_circle_rounded,
   158→                          size: Spacings.mdIcon - 4,
   159→                          color: AppColors.success,
   160→                        ),
   161→                        const SizedBox(width: Spacings.sm),
   162→                        Expanded(
   163→                          child: Text(
   164→                            feature,
   165→                            style: tt.bodySmall?.copyWith(
   166→                              color: cs.onSurfaceVariant,

   173→                    ),
   174→                  ),
   175→                ),

   177→                const SizedBox(height: Spacings.lg),

   182→                  child: isCurrentPlan

   184→                          onPressed: null,
   185→                          style: OutlinedButton.styleFrom(
   186→                            padding: Spacings.paddingButton,
   187→                            shape: RoundedRectangleBorder(
   188→                              borderRadius: Spacings.borderRadiusMd,
   189→                            ),
   190→                          ),

   194→                          onPressed: onSelect,
   195→                          style: FilledButton.styleFrom(
   196→                            padding: Spacings.paddingButton,
   197→                            shape: RoundedRectangleBorder(
   198→                              borderRadius: Spacings.borderRadiusMd,
   199→                            ),
   200→                            backgroundColor:
   201→                                isSelected ? cs.primary : cs.surface,
   202→                            foregroundColor:
   203→                                isSelected ? cs.onPrimary : cs.primary,
   204→                          ),
   205→                          child: Text(isSelected ? 'Selected' : 'Select Plan'),
   206→                        ),
   207→                ),
   208→              ],
   209→            ),
   210→          ),
   211→
   212→          // ── "Most Popular" Badge ────────────────────────────────────
   213→          if (plan.isPopular)

   215→              top: 0,
   216→              right: 0,
   217→              child: Container(
   218→                padding: const EdgeInsets.symmetric(
   219→                  horizontal: Spacings.md,

   228→                child: Text(
   229→                  'Most Popular',

   234→                ),
   235→              ),
   236→            ),
   237→        ],
   238→      ),
   239→    );
   240→  }

   242→  String _formatCurrency(double amount, String currency) {
   243→    final symbol = currency == 'NGN' ? '\u20A6' : '\$';
   244→    return '$symbol${amount.toStringAsFixed(0)}';
   245→  }

   252→/// Shows AI credit balance with a circular progress ring.
   253→///
   254→/// Displays remaining / total credits, a circular progress indicator
   255→/// representing usage percentage, and a "Purchase More" button when credits
   256→/// are running low.
   257→class CreditBalanceCard extends StatelessWidget {
   258→  const CreditBalanceCard({
   259→    super.key,
   260→    required this.balance,
   261→    this.onPurchaseMore,
   262→  });
   263→
   264→  final AiCreditBalanceEntity balance;
   265→  final VoidCallback? onPurchaseMore;
   266→
   267→  @override
   268→  Widget build(BuildContext context) {
   269→    final cs = context.colorScheme;
   270→    final tt = context.textTheme;
   271→    final isDark = context.isDarkMode;
   272→    final usagePercent = balance.usagePercent;
   273→    final remainingPercent = 100 - usagePercent;
   274→
   275→    // Choose colour based on credit health
   276→    final Color progressColor;
   277→    if (balance.isExhausted) {
   278→      progressColor = AppColors.error;
   279→    } else if (balance.isLow) {
   280→      progressColor = AppColors.warning;
   281→    } else {
   282→      progressColor = AppColors.success;
   283→    }
   284→
   285→    return Card(
   286→      elevation: Spacings.elevationSm,
   287→      shadowColor: cs.shadow.withValues(alpha: 0.06),
   288→      shape: RoundedRectangleBorder(
   289→        borderRadius: Spacings.borderRadiusLg,
   290→      ),
   291→      child: Padding(
   292→        padding: const EdgeInsets.all(Spacings.lg),
   293→        child: Column(
   294→          children: [
   295→            // ── Top Row: Progress Ring + Info ────────────────────────
   296→            Row(
   297→              children: [
   298→                // Circular progress ring
   299→                SizedBox(
   300→                  width: 88,
   301→                  height: 88,
   302→                  child: Stack(
   303→                    alignment: Alignment.center,
   304→                    children: [
   305→                      CircularProgressIndicator(
   306→                        value: balance.totalCredits > 0
   307→                            ? remainingPercent / 100
   308→                            : 0,
   309→                        strokeWidth: 6,
   310→                        backgroundColor:
   311→                            progressColor.withValues(alpha: isDark ? 0.20 : 0.12),
   312→                        valueColor: AlwaysStoppedAnimation(progressColor),
   313→                        strokeCap: StrokeCap.round,
   314→                      ),
   315→                      Column(
   316→                        mainAxisSize: MainAxisSize.min,
   317→                        children: [
   318→                          Text(
   319→                            '${remainingPercent.toStringAsFixed(0)}%',
   320→                            style: tt.titleSmall?.copyWith(
   321→                              fontWeight: AppTypography.wBold,
   322→                              color: cs.onSurface,
   323→                              height: 1.0,
   324→                            ),
   325→                          ),
   326→                          Text(
   327→                            'remaining',
   328→                            style: tt.labelSmall?.copyWith(
   329→                              color: cs.onSurfaceVariant,
   330→                              fontSize: 9,
   331→                            ),
   332→                          ),
   333→                        ],
   334→                      ),
   335→                    ],
   336→                  ),
   337→                ),
   338→                const SizedBox(width: Spacings.lg),
   339→
   340→                // Credit numbers
   341→                Expanded(
   342→                  child: Column(
   343→                    crossAxisAlignment: CrossAxisAlignment.start,
   344→                    children: [
   345→                      Text(
   346→                        'AI Credits',
   347→                        style: tt.titleMedium?.copyWith(
   348→                          fontWeight: AppTypography.wSemiBold,
   349→                          color: cs.onSurface,
   350→                        ),
   351→                      ),
   352→                      const SizedBox(height: Spacings.sm),
   353→                      Text(
   354→                        '${balance.remainingCredits} / ${balance.totalCredits}',
   355→                        style: tt.headlineSmall?.copyWith(
   356→                          fontWeight: AppTypography.wBold,
   357→                          color: cs.onSurface,
   358→                        ),
   359→                      ),
   360→                      const SizedBox(height: Spacings.xs),
   361→                      Text(
   362→                        '${balance.usedCredits} used this cycle',
   363→                        style: tt.bodySmall?.copyWith(
   364→                          color: cs.onSurfaceVariant,
   365→                        ),
   366→                      ),
   367→                    ],
   368→                  ),
   369→                ),
   370→              ],
   371→            ),
   372→
   373→            // ── Warning / Exhausted indicator ─────────────────────────
   374→            if (balance.isExhausted || balance.isLow) ...[
   375→              const SizedBox(height: Spacings.md),
   376→              Container(
   377→                width: double.infinity,
   378→                padding: const EdgeInsets.all(Spacings.md),
   379→                decoration: BoxDecoration(
   380→                  color: (balance.isExhausted ? AppColors.error : AppColors.warning)
   381→                      .withValues(alpha: isDark ? 0.20 : 0.10),
   382→                  borderRadius: Spacings.borderRadiusMd,
   383→                ),
   384→                child: Row(
   385→                  children: [
   386→                    Icon(
   387→                      balance.isExhausted
   388→                          ? Icons.error_outline_rounded
   389→                          : Icons.warning_amber_rounded,
   390→                      size: Spacings.mdIcon,
   391→                      color: balance.isExhausted
   392→                          ? AppColors.error
   393→                          : AppColors.warning,
   394→                    ),
   395→                    const SizedBox(width: Spacings.sm),
   396→                    Expanded(
   397→                      child: Text(
   398→                        balance.isExhausted
   399→                            ? 'Credits exhausted. Purchase more to continue using AI features.'
   400→                            : 'Credits running low. Consider purchasing more.',
   401→                        style: tt.bodySmall?.copyWith(
   402→                          color: balance.isExhausted
   403→                              ? (isDark ? AppColors.errorDark : AppColors.error)
   404→                              : (isDark ? AppColors.warningDark : AppColors.warning),
   405→                          fontWeight: AppTypography.wMedium,
   406→                        ),
   407→                      ),
   408→                    ),
   409→                  ],
   410→                ),
   411→              ),
   412→            ],
   413→
   414→            // ── Purchase More Button ──────────────────────────────────
   415→            if (onPurchaseMore != null) ...[
   416→              const SizedBox(height: Spacings.md),
   417→              SizedBox(
   418→                width: double.infinity,
   419→                child: OutlinedButton.icon(
   420→                  onPressed: onPurchaseMore,
   421→                  icon: const Icon(Icons.add_circle_outline_rounded, size: 18),
   422→                  label: const Text('Purchase More Credits'),
   423→                  style: OutlinedButton.styleFrom(
   424→                    padding: Spacings.paddingButton,
   425→                    shape: RoundedRectangleBorder(
   426→                      borderRadius: Spacings.borderRadiusMd,
   427→                    ),
   428→                  ),
   429→                ),
   430→              ),
   431→            ],
   432→          ],
   433→        ),
   434→      ),
   435→    );
   436→  }
   437→}

   443→/// Displays a transaction in a list.
   444→///
   445→/// Shows the amount, a status badge with colour coding, date,
   446→/// payment method, a channel icon, and an arrow indicator.
   447→class TransactionListTile extends StatelessWidget {
   448→  const TransactionListTile({
   449→    super.key,
   450→    required this.transaction,
   451→    this.onTap,
   452→  });
   453→
   454→  final TransactionEntity transaction;
   455→  final VoidCallback? onTap;
   456→
   457→  @override
   458→  Widget build(BuildContext context) {
   459→    final cs = context.colorScheme;
   460→    final tt = context.textTheme;
   461→    final isDark = context.isDarkMode;
   462→
   463→    final statusColor = _parseHexColor(transaction.status.color);
   464→    final channelIcon = _channelIcon(transaction.channel);
   465→    final isSuccessful = transaction.isSuccessful;
   466→    final isRefund = transaction.status == TransactionStatus.refunded ||
   467→        transaction.status == TransactionStatus.partiallyRefunded;
   468→
   469→    return ListTile(
   470→      onTap: onTap,
   471→      contentPadding: const EdgeInsets.symmetric(
   472→        horizontal: Spacings.lg,
   473→        vertical: Spacings.xs,
   474→      ),
   475→      shape: RoundedRectangleBorder(
   476→        borderRadius: Spacings.borderRadiusMd,
   477→      ),
   478→      leading: Container(
   479→        width: 40,
   480→        height: 40,
   481→        decoration: BoxDecoration(
   482→          color: (isRefund ? AppColors.info : (isSuccessful ? AppColors.success : statusColor))
   483→              .withValues(alpha: isDark ? 0.20 : 0.12),
   484→          borderRadius: Spacings.borderRadiusMd,
   485→        ),
   486→        child: Icon(
   487→          isRefund ? Icons.replay_rounded : channelIcon,
   488→          color: isRefund ? AppColors.info : (isSuccessful ? AppColors.success : statusColor),
   489→          size: Spacings.mdIcon - 4,
   490→        ),
   491→      ),
   492→      title: Row(
   493→        children: [
   494→          Text(
   495→            _formatCurrency(
   496→              isRefund ? transaction.refundAmount : transaction.amount,
   497→              transaction.currency,
   498→            ),
   499→            style: tt.titleSmall?.copyWith(
   500→              fontWeight: AppTypography.wSemiBold,
   501→              color: cs.onSurface,
   502→            ),
   503→          ),
   504→          const SizedBox(width: Spacings.sm),
   505→          _TransactionStatusBadge(status: transaction.status),
   506→        ],
   507→      ),
   508→      subtitle: Text(
   509→        _formatDate(transaction.initiatedAt),
   510→        style: tt.bodySmall?.copyWith(
   511→          color: cs.onSurfaceVariant,
   512→        ),
   513→      ),
   514→      trailing: Row(
   515→        mainAxisSize: MainAxisSize.min,
   516→        children: [
   517→          if (transaction.paymentMethodSummary != null)
   518→            Flexible(
   519→              child: Text(
   520→                transaction.paymentMethodSummary!,
   521→                style: tt.labelSmall?.copyWith(
   522→                  color: cs.onSurfaceVariant,
   523→                ),
   524→                overflow: TextOverflow.ellipsis,
   525→              ),
   526→            ),
   527→          if (onTap != null) ...[
   528→            const SizedBox(width: Spacings.xs),
   529→            Icon(
   530→              Icons.chevron_right_rounded,
   531→              size: Spacings.lgIcon,
   532→              color: cs.onSurfaceVariant,
   533→            ),
   534→          ],
   535→        ],
   536→      ),
   537→    );
   538→  }
   539→
   540→  IconData _channelIcon(PaymentChannel channel) {
   541→    switch (channel) {
   542→      case PaymentChannel.card:
   543→        return Icons.credit_card_rounded;
   544→      case PaymentChannel.bankTransfer:
   545→        return Icons.account_balance_rounded;
   546→      case PaymentChannel.ussd:
   547→        return Icons.phone_android_rounded;
   548→      case PaymentChannel.mobileMoney:
   549→        return Icons.phone_android_rounded;
   550→      case PaymentChannel.qrCode:
   551→        return Icons.qr_code_rounded;
   552→      case PaymentChannel.credit:
   553→        return Icons.auto_awesome_rounded;
   554→      case PaymentChannel.coupon:
   555→        return Icons.confirmation_number_rounded;
   556→      case PaymentChannel.refund:
   557→        return Icons.replay_rounded;
   558→    }
   559→  }
   560→}
