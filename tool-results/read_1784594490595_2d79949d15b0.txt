    18→/// Single product detail page with full product information, reviews,
    19→/// quality checks, related products, and purchase actions.
    20→///
    21→/// ```dart
    22→/// Navigator.push(
    23→///   context,
    24→///   MaterialPageRoute(builder: (_) => ProductDetailPage(productId: '123')),
    25→/// );
    26→/// ```
    27→class ProductDetailPage extends ConsumerStatefulWidget {
    28→  const ProductDetailPage({
    29→    super.key,
    30→    required this.productId,
    31→  });
    32→
    33→  final String productId;
    34→
    35→  @override
    36→  ConsumerState<ProductDetailPage> createState() => _ProductDetailPageState();
    37→}
    38→
    39→class _ProductDetailPageState extends ConsumerState<ProductDetailPage> {
    40→  MarketplaceLicenseType _selectedLicense = MarketplaceLicenseType.personal;
    41→
    42→  @override
    43→  void initState() {
    44→    super.initState();
    45→    WidgetsBinding.instance.addPostFrameCallback((_) => _loadData());
    46→  }
    47→
    48→  Future<void> _loadData() async {
    49→    final notifier = ref.read(productDetailProvider.notifier);
    50→    await notifier.loadProduct(productId: widget.productId);
    51→    // Load supplementary data
    52→    notifier.loadReviews(productId: widget.productId);
    53→    notifier.loadQualityCheck(productId: widget.productId);
    54→    notifier.loadRelatedProducts(productId: widget.productId);
    55→    notifier.recordView(productId: widget.productId);
    56→  }
    57→
    58→  void _showSnackBar(String message, {bool isError = false}) {
    59→    context.scaffoldMessenger.showSnackBar(
    60→      SnackBar(
    61→        content: Text(message),
    62→        backgroundColor: isError ? AppColors.error : null,
    63→        behavior: SnackBarBehavior.floating,
    64→      ),
    65→    );
    66→  }
    67→
    68→  Future<void> _addToCart() async {
    69→    final product = ref.read(productDetailProvider).product;
    70→    if (product == null) return;
    71→
    72→    await ref.read(cartProvider.notifier).addToCart(
    73→      userId: 'current_user',
    74→      item: CartItemEntity(
    75→        id: '',
    76→        cartId: '',
    77→        productId: product.id,
    78→        licenseType: _selectedLicense,
    79→        addedAt: DateTime.now(),
    80→        product: product,
    81→      ),
    82→    );
    83→
    84→    final cartState = ref.read(cartProvider);
    85→    if (cartState.error != null) {
    86→      _showSnackBar(cartState.error!, isError: true);
    87→    } else if (cartState.successMessage != null) {
    88→      _showSnackBar(cartState.successMessage!);
    89→    }
    90→  }
    91→
    92→  void _buyNow() {
    93→    // Navigate to checkout with this product
    94→    _showSnackBar('Proceeding to checkout...');
    95→  }
    96→
    97→  void _toggleWishlist() {
    98→    final product = ref.read(productDetailProvider).product;
    99→    if (product == null) return;
   100→
   101→    ref.read(productDetailProvider.notifier).toggleWishlist(
   102→      userId: 'current_user',
   103→      productId: product.id,
   104→    );
   105→
   106→    final state = ref.read(productDetailProvider);
   107→    _showSnackBar(
   108→      state.isInWishlist
   109→          ? 'Added to wishlist'
   110→          : 'Removed from wishlist',
   111→    );
   112→  }
   113→
   114→  @override
   115→  Widget build(BuildContext context) {
   116→    final state = ref.watch(productDetailProvider);
   117→
   118→    // Listen for errors
   119→    ref.listen<ProductDetailState>(productDetailProvider, (prev, next) {
   120→      if (next.error != null && prev?.error != next.error) {
   121→        _showSnackBar(next.error!, isError: true);
   122→        ref.read(productDetailProvider.notifier).clearError();
   123→      }
   124→    });
   125→
   126→    return Scaffold(
   127→      appBar: AppAppBar(
   128→        title: 'Product Details',
   129→        actions: [
   130→          AppIconButton(
   131→            icon: Icons.share_rounded,
   132→            onPressed: () {
   133→              // Share product
   134→            },
   135→            tooltip: 'Share',
   136→          ),
   137→        ],
   138→      ),
   139→      body: state.isLoading
   140→          ? const Center(child: AppLoadingSpinner())
   141→          : state.error != null && !state.hasProduct
   142→              ? AppErrorState.serverError(onRetry: _loadData)
   143→              : state.hasProduct
   144→                  ? _buildContent(state)
   145→                  : AppErrorState.notFoundError(onRetry: _loadData),
   146→      bottomNavigationBar:
   147→          state.hasProduct ? _buildBottomBar(state) : null,
   148→    );
   149→  }

   170→                _buildTitleSection(product),
   171→
   172→                const SizedBox(height: Spacings.lg),
   173→

   177→                const SizedBox(height: Spacings.lg),

   184→                // ── Rating ───────────────────────────────────────────────
   185→                _buildRatingSection(product),
   186→
   187→                Spacings.sectionGap,
   188→
   189→                // ── Description ──────────────────────────────────────────
   190→                _buildDescription(product),
   191→
   192→                Spacings.sectionGap,
   193→
   194→                // ── Product Details ──────────────────────────────────────
   195→                _buildProductDetails(product),
   196→
   197→                Spacings.sectionGap,
   198→
   199→                // ── Tags ─────────────────────────────────────────────────
   200→                if (product.tags.isNotEmpty) ...[
   201→                  _buildTags(product),
   202→                  Spacings.sectionGap,
   203→                ],
   204→
   205→                // ── AI Summary ───────────────────────────────────────────
   206→                if (product.aiGeneratedSummary != null) ...[
   207→                  _buildAiSummary(product),
   208→                  Spacings.sectionGap,
   209→                ],
   210→
   211→                // ── Quality Check ────────────────────────────────────────
   212→                if (state.hasQualityCheck) ...[
   213→                  QualityScoreCard(qualityCheck: state.qualityCheck!),
   214→                  Spacings.sectionGap,
   215→                ],
   216→
   217→                // ── Preview Section ──────────────────────────────────────
   218→                if (product.previewImages.isNotEmpty ||
   219→                    product.previewDocuments.isNotEmpty) ...[
   220→                  _buildPreviewSection(product),
   221→                  Spacings.sectionGap,
   222→                ],
   223→
   224→                // ── Related Products ─────────────────────────────────────
   225→                if (state.relatedProducts.isNotEmpty) ...[
   226→                  _buildRelatedProducts(state),
   227→                  Spacings.sectionGap,
   228→                ],
   229→
   230→                // ── Reviews ──────────────────────────────────────────────
   231→                _buildReviewsSection(state),
   232→              ],
   233→            ),
   234→          ),
   235→        ],
   236→      ),
   237→    );
   238→  }
