    17→Color _statusColor(MarketplaceStatus status) {
    18→  switch (status) {
    19→    case MarketplaceStatus.pendingReview:
    20→      return AppColors.info;
    21→    case MarketplaceStatus.approved:
    22→      return AppColors.success;
    23→    case MarketplaceStatus.rejected:
    24→      return AppColors.error;
    25→    case MarketplaceStatus.featured:
    26→      return const Color(0xFFD97706); // Gold/Amber
    27→    case MarketplaceStatus.archived:
    28→      return Colors.grey;
    29→    case MarketplaceStatus.flagged:
    30→      return Colors.orange;
    31→    case MarketplaceStatus.suspended:
    32→      return Colors.blueGrey;
    33→  }
    34→}
    35→
    36→Color _contentTypeColor(MarketplaceContentType type) {
    37→  switch (type) {
    38→    case MarketplaceContentType.resource:
    39→      return AppColors.info;
    40→    case MarketplaceContentType.lessonNote:
    41→      return AppColors.success;
    42→    case MarketplaceContentType.worksheet:
    43→      return const Color(0xFF7C3AED);
    44→    case MarketplaceContentType.questionBank:
    45→      return AppColors.warning;
    46→    case MarketplaceContentType.template:
    47→      return const Color(0xFFEA580C);
    48→    case MarketplaceContentType.examFormat:
    49→      return const Color(0xFF06B6D4);
    50→    case MarketplaceContentType.video:
    51→      return AppColors.error;
    52→    case MarketplaceContentType.document:
    53→      return Colors.blueGrey;
    54→  }
    55→}
    56→
    57→String _formatDate(DateTime dt) {
    58→  return '${dt.day.toString().padLeft(2, '0')}/${dt.month.toString().padLeft(2, '0')}/${dt.year}';
    59→}
    60→
    61→String _formatRating(double rating) {
    62→  return rating.toStringAsFixed(1);
    63→}
    64→
    65→String _formatPrice(double price) {
    66→  if (price == 0) return 'Free';
    67→  return '\$${price.toStringAsFixed(2)}';
    68→}

   681→  Widget _buildActionButtons(MarketplaceContent content, ColorScheme cs) {
   682→    return PopupMenuButton<String>(
   683→      icon: const Icon(Icons.more_vert),
   684→      tooltip: 'Actions',
   685→      onSelected: (action) => _handleAction(action, content),
   686→      itemBuilder: (ctx) => [
   687→        const PopupMenuItem(value: 'view', child: Text('View Details')),
   688→        if (content.status == MarketplaceStatus.pendingReview)
   689→          const PopupMenuItem(value: 'approve', child: Text('Approve')),
   690→        if (content.status == MarketplaceStatus.pendingReview)
   691→          const PopupMenuItem(value: 'reject', child: Text('Reject')),
   692→        if (content.status == MarketplaceStatus.approved)
   693→          const PopupMenuItem(value: 'feature', child: Text('Feature')),
   694→        if (!content.isFlagged)
   695→          const PopupMenuItem(value: 'flag', child: Text('Flag')),
   696→        const PopupMenuItem(value: 'remove', child: Text('Remove')),
   697→      ],
   698→    );
   699→  }

   701→  void _handleAction(String action, MarketplaceContent content) {
   702→    switch (action) {
   703→      case 'view':
   704→        _showContentDetailDialog(content);
   705→      case 'approve':
   706→        _approveContent(content.id);
   707→      case 'reject':
   708→        _showRejectDialog(content);
   709→      case 'feature':
   710→        _featureContent(content.id);
   711→      case 'flag':
   712→        _showFlagDialog(content);
   713→      case 'remove':
   714→        _showRemoveConfirmDialog(content);
   715→    }
   716→  }
