   122→/// Provides the Supabase client singleton.

   126→  try {
   127→    return SupabaseConfig.client;

   129→    AppLogger.error('Failed to resolve Supabase client', error: e);
   130→    rethrow;
   131→  }

   133→
   134→/// Provides a configured [Dio] instance with base options pulled from
   135→/// [AppConfig] and [ApiConstants].
   136→final dioProvider = Provider<Dio>((ref) {
   137→  final dio = Dio(
   138→    BaseOptions(
   139→      baseUrl: AppConfig.baseUrl,
   140→      connectTimeout: AppConfig.connectTimeoutDuration,
   141→      receiveTimeout: AppConfig.receiveTimeoutDuration,
   142→      sendTimeout: AppConfig.sendTimeoutDuration,
   143→      headers: ApiConstants.baseHeaders(),
   144→      responseType: ResponseType.json,
   145→      validateStatus: (status) => status != null && status < 500,
   146→    ),
   147→  );
   148→  return dio;
   149→});
   150→
   151→/// Provides the [ApiClient] wrapper around [Dio].

   153→  final dio = ref.watch(dioProvider);

   205→/// Stream provider that emits the current Supabase [sb.AuthState].
   206→///
   207→/// Automatically reconnects on transient failures and emits the latest
   208→/// state to any listener.
   209→final authStateProvider = StreamProvider<sb.AuthState>((ref) {
   210→  final supabaseClient = ref.watch(supabaseClientProvider);
   211→  return supabaseClient.auth.onAuthStateChange;
   212→});
   213→
   214→/// Convenience provider that yields the currently authenticated [sb.User],
   215→/// or `null` if not signed in.
   216→final currentUserProvider = Provider<sb.User?>((ref) {
   217→  final authState = ref.watch(authStateProvider);
   218→  return authState.when(
   219→    data: (state) => state.session?.user,
   220→    loading: () => SupabaseConfig.currentUser,
   221→    error: (_, __) => null,
   222→  );
   223→});
   224→
   225→/// Convenience provider that yields `true` when the user is authenticated.
   226→final isAuthenticatedProvider = Provider<bool>((ref) {
   227→  final user = ref.watch(currentUserProvider);
   228→  return user != null;
   229→});
   230→
   231→/// Provides the current user's role from secure storage, or a default.
   232→///
   233→/// This is a future provider since it reads from secure storage.
   234→final userRoleProvider = FutureProvider<String>((ref) async {
   235→  final storageService = ref.watch(storageServiceProvider);
   236→  return storageService.getUserRole() ?? 'student';

   486→/// Provides the [AiProvidersRegistry] singleton.
   487→///
   488→/// Registers OpenAI and Gemini providers conditionally based on
   489→/// whether their API keys are available in the environment config.
   490→final aiProviderRegistryProvider = Provider<AiProvidersRegistry>((ref) {
   491→  final openaiApiKey = EnvConfig.maybeGet('OPENAI_API_KEY') ?? '';
   492→  final geminiApiKey = EnvConfig.maybeGet('GEMINI_API_KEY') ?? '';
   493→
   494→  if (openaiApiKey.isEmpty && geminiApiKey.isEmpty) {
   495→    AppLogger.warning(
   496→      'No AI API keys configured — AI features will be unavailable',
   497→    );
   498→    return AiProvidersRegistry.createEmpty();
   499→  }
   500→
   501→  return AiProvidersRegistry.createDefault(
   502→    openaiApiKey: openaiApiKey,
   503→    geminiApiKey: geminiApiKey,
   504→  );
   505→});
   506→

   508→final promptEngineProvider = Provider<PromptEngine>((ref) {
   509→  return PromptEngine();
   510→});

   513→final validationEngineProvider = Provider<ValidationEngine>((ref) {
   514→  return ValidationEngine();
   515→});

   517→/// Provides the [AiService] singleton, injecting the provider registry,
   518→/// prompt engine, and validation engine.

   521→  final promptEngine = ref.watch(promptEngineProvider);

   524→  return AiService(

   526→    promptEngine: promptEngine,
