   270→   270→// APP ROUTER PROVIDER
   271→   271→// ═══════════════════════════════════════════════════════════════════════
   272→   272→
   273→   273→/// Riverpod provider for the application's [GoRouter].
   274→   274→///
   275→   275→/// Watches [authStateProvider] so that the router automatically
   276→   276→/// re-evaluates its [redirect] callback whenever the authentication
   277→   277→/// state changes (sign-in, sign-out, token refresh).
   278→   278→///
   279→   279→/// Usage in `main.dart`:
   280→   280→/// ```dart
   281→   281→/// final router = ref.read(appRouterProvider);
   282→   282→/// MaterialApp.router(routerConfig: router);
   283→   283→/// ```
   284→   284→final appRouterProvider = Provider<GoRouter>((ref) {
   285→   285→  // Watch auth state so the router re-evaluates redirects on change.
   286→   286→  final authState = ref.watch(authStateProvider);
   287→   287→
