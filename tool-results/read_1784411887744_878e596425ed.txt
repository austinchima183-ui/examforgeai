   297→   297→-- ═══════════════════════════════════════════════════════════════════════════════
   298→   298→-- PARENT ENGAGEMENT SUMMARY MATERIALIZED VIEW
   299→   299→-- ═══════════════════════════════════════════════════════════════════════════════
   300→   300→
   301→   301→CREATE MATERIALIZED VIEW IF NOT EXISTS parent_engagement_summary AS
   302→   302→SELECT
   303→   303→  pem.school_id,
   304→   304→  pem.parent_id,
   305→   305→  pem.student_id,
   306→   306→  COUNT(DISTINCT CASE WHEN pem.metric_type = 'report_card_viewed' THEN pem.id END) AS report_card_views,
   307→   307→  COUNT(DISTINCT CASE WHEN pem.metric_type = 'announcement_read' THEN pem.id END) AS announcements_read,
   308→   308→  COUNT(DISTINCT CASE WHEN pem.metric_type = 'message_sent' THEN pem.id END) AS messages_sent,
   309→   309→  COUNT(DISTINCT CASE WHEN pem.metric_type = 'meeting_attended' THEN pem.id END) AS meetings_attended,
   310→   310→  COUNT(DISTINCT CASE WHEN pem.metric_type = 'assignment_checked' THEN pem.id END) AS assignment_checks,
   311→   311→  COUNT(DISTINCT CASE WHEN pem.metric_type = 'attendance_checked' THEN pem.id END) AS attendance_checks,
   312→   312→  COUNT(DISTINCT CASE WHEN pem.metric_type = 'ai_assistant_used' THEN pem.id END) AS ai_assistant_uses,
   313→   313→  COUNT(DISTINCT pem.id) AS total_interactions,
   314→   314→  MAX(pem.recorded_at) AS last_active_at,
   315→   315→  CASE
   316→   316→    WHEN MAX(pem.recorded_at) > now() - INTERVAL '7 days' THEN 'active'
   317→   317→    WHEN MAX(pem.recorded_at) > now() - INTERVAL '30 days' THEN 'moderate'
   318→   318→    ELSE 'inactive'
   319→   319→  END AS engagement_level
   320→   320→FROM parent_engagement_metrics pem
   321→   321→GROUP BY pem.school_id, pem.parent_id, pem.student_id;
   322→   322→
   323→   323→CREATE UNIQUE INDEX idx_engagement_summary_pk ON parent_engagement_summary(school_id, parent_id, student_id);

   326→   326→-- PARENT DASHBOARD VIEW
   327→   327→-- ═══════════════════════════════════════════════════════════════════════════════
   328→   328→
   329→   329→CREATE OR REPLACE VIEW parent_dashboard_view AS
   330→   330→SELECT
   331→   331→  p.id AS parent_id,
   332→   332→  p.school_id,
   333→   333→  p.user_id,
   334→   334→  u.full_name AS parent_name,
   335→   335→  u.email AS parent_email,
   336→   336→  u.phone AS parent_phone,
   337→   337→  u.avatar_url AS parent_avatar,
   338→   338→  -- Child count
   339→   339→  (SELECT COUNT(*) FROM parent_students ps WHERE ps.parent_id = p.id AND ps.is_active = TRUE) AS child_count,
   340→   340→  -- Unread notifications
   341→   341→  (SELECT COUNT(*) FROM parent_notifications pn WHERE pn.parent_id = p.user_id AND pn.is_read = FALSE) AS unread_notifications,
   342→   342→  -- Unread messages
   343→   343→  (SELECT COUNT(*) FROM parent_messages pm WHERE pm.recipient_id = p.user_id AND pm.read_at IS NULL AND pm.is_archived = FALSE) AS unread_messages,
   344→   344→  -- Active insights
   345→   345→  (SELECT COUNT(*) FROM parent_ai_insights pai WHERE pai.parent_id = p.user_id AND pai.is_read = FALSE AND pai.is_dismissed = FALSE) AS active_insights,
   346→   346→  -- Last active
   347→   347→  (SELECT MAX(pal.created_at) FROM parent_activity_logs pal WHERE pal.parent_id = p.user_id) AS last_active_at
   348→   348→FROM parent_profiles p
   349→   349→JOIN users u ON u.id = p.user_id;

   547→   547→-- Get child academic performance

   579→   579→        )
   580→   580→      ), '[]'::jsonb)
   581→   581→      FROM class_subjects cs

   584→   584→    ),

   586→   586→      SELECT AVG(er.score) FROM exam_results er
   587→   587→      WHERE er.student_id = p_student_id
   588→   588→    ),
   589→   589→    'class_average', (
   590→   590→      SELECT AVG(er.score) FROM exam_results er
   591→   591→      WHERE er.student_id IN (
   592→   592→        SELECT cs.student_id FROM class_students cs
   593→   593→        WHERE cs.class_id IN (SELECT cs2.class_id FROM class_students cs2 WHERE cs2.student_id = p_student_id)
   594→   594→      )
   595→   595→    ),

   598→   598→        WHEN 0 THEN NULL

   659→   659→      WHERE a.school_id = p_school_id
