   321→   321→  @override
   322→   322→  Future<List<MarketplaceCategoryModel>> getCategories({
   323→   323→    bool activeOnly = true,
   324→   324→  }) async {
   325→   325→    try {
   326→   326→      var query = _supabase.from(_categoriesTable).select();
   327→   327→
   328→   328→      if (activeOnly) {
   329→   329→        query = query.eq('is_active', true);
   330→   330→      }
   331→   331→
   332→   332→      final response = await query.order('sort_order', ascending: true);
   333→   333→
   334→   334→      AppLogger.info('Fetched ${response.length} marketplace categories');
   335→   335→      return response
   336→   336→          .map<MarketplaceCategoryModel>(
   337→   337→            (row) => MarketplaceCategoryModel.fromJson(row),
   338→   338→          )
   339→   339→          .toList();
   340→   340→    } on sb.PostgrestException catch (e) {
   341→   341→      throw _mapPostgrestException(e);
   342→   342→    } catch (e) {
   343→   343→      AppLogger.error('Unexpected getCategories error', error: e);
   344→   344→      throw const ServerException(
   345→   345→        message: 'Failed to fetch marketplace categories.',
   346→   346→        statusCode: 500,
   347→   347→      );
   348→   348→    }
   349→   349→  }
   350→   350→
   351→   351→  @override
   352→   352→  Future<MarketplaceCategoryModel> getCategory(String categoryId) async {
   353→   353→    try {
   354→   354→      final response = await _supabase
   355→   355→          .from(_categoriesTable)
   356→   356→          .select()
   357→   357→          .eq('id', categoryId)
   358→   358→          .limit(1);
   359→   359→
   360→   360→      if (response.isEmpty) {
   361→   361→        throw const NotFoundException('Category not found.');
   362→   362→      }
   363→   363→
   364→   364→      return MarketplaceCategoryModel.fromJson(response.first);
   365→   365→    } on sb.PostgrestException catch (e) {
   366→   366→      throw _mapPostgrestException(e);
   367→   367→    } on NotFoundException {
   368→   368→      rethrow;
   369→   369→    } catch (e) {
   370→   370→      AppLogger.error('Unexpected getCategory error', error: e);
   371→   371→      throw const ServerException(
   372→   372→        message: 'Failed to fetch category.',
   373→   373→        statusCode: 500,
   374→   374→      );
   375→   375→    }
   376→   376→  }

   378→   378→  @override
   379→   379→  Future<MarketplaceCategoryModel> upsertCategory(
   380→   380→    MarketplaceCategoryModel category,
   381→   381→  ) async {
   382→   382→    try {
   383→   383→      final data = category.toJson();
   384→   384→      data['updated_at'] = DateTime.now().toIso8601String();
   385→   385→
   386→   386→      final response = await _supabase
   387→   387→          .from(_categoriesTable)
   388→   388→          .upsert(data)
   389→   389→          .select();
   390→   390→
   391→   391→      if (response.isEmpty) {
   392→   392→        throw const ServerException(
   393→   393→          message: 'Category upsert returned no data.',
   394→   394→          statusCode: 500,
   395→   395→        );
   396→   396→      }
   397→   397→
   398→   398→      AppLogger.info('Category upserted: ${response.first['id']}');
   399→   399→      return MarketplaceCategoryModel.fromJson(response.first);
   400→   400→    } on sb.PostgrestException catch (e) {
   401→   401→      throw _mapPostgrestException(e);
   402→   402→    } on ServerException {
   403→   403→      rethrow;
   404→   404→    } catch (e) {
   405→   405→      AppLogger.error('Unexpected upsertCategory error', error: e);
   406→   406→      throw const ServerException(
   407→   407→        message: 'Failed to upsert category.',
   408→   408→        statusCode: 500,
   409→   409→      );
   410→   410→    }
   411→   411→  }
