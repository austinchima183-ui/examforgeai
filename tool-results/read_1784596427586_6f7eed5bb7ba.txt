   639→   639→  void _checkTimeWarnings(ExamTakerState state) {
   640→   640→    if (state.timeRemaining.inMinutes == 5 &&
   641→   641→        !_hasShownTimeWarning &&

   643→   643→      _hasShownTimeWarning = true;
   644→   644→      Future.microtask(() {
   645→   645→        if (mounted) {
   646→   646→          AppDialog.showInfo(
   647→   647→            context: context,
   648→   648→            title: 'Time Warning',
   649→   649→            message: 'You have less than 5 minutes remaining. Please review your answers.',
   650→   650→            dismissText: 'Continue',
   651→   651→          );
   652→   652→        }
   653→   653→      });
   654→   654→    }
   655→   655→
   656→   656→    if (state.timeRemaining.inMinutes == 0 &&
   657→   657→        state.timeRemaining.inSeconds > 30 &&
   658→   658→        !_hasShownCriticalWarning) {
   659→   659→      _hasShownCriticalWarning = true;

   663→   663→            context: context,

   668→   668→        }
   669→   669→      });
   670→   670→    }
