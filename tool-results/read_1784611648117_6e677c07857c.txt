    15→/// Abstract interface for all remote CBT engine data operations.
    16→///
    17→/// Implementations handle all network communication with the Supabase
    18→/// backend and return plain model instances. Exceptions are allowed to
    19→/// propagate so the repository layer can catch and convert them to
    20→/// domain [Failure] types.

   307→  @override
   308→  Future<List<ExamModel>> getExams(Map<String, dynamic> filters) async {
   309→    try {
   310→      var query = _supabaseClient.from(_examsTable).select();
   311→
   312→      if (filters['school_id'] != null) {
   313→        query = query.eq('school_id', filters['school_id'] as String);
   314→      }
   315→      if (filters['subject_id'] != null) {
   316→        query = query.eq('subject_id', filters['subject_id'] as String);
   317→      }
   318→      if (filters['status'] != null) {
   319→        query = query.eq('status', filters['status'] as String);
   320→      }
   321→      if (filters['class_id'] != null) {
   322→        query = query.eq('class_id', filters['class_id'] as String);
   323→      }
   324→
   325→      final page = filters['page'] as int? ?? 1;
   326→      final perPage = filters['per_page'] as int? ?? filters['perPage'] as int? ?? 20;
   327→      final offset = (page - 1) * perPage;
   328→
   329→      query = query.range(offset, offset + perPage - 1);
   330→      query = query.order('created_at', ascending: false);
   331→
   332→      final response = await query;
   333→
   334→      return response
   335→          .map<ExamModel>((json) => ExamModel.fromJson(json))
   336→          .toList();
   337→    } on sb.PostgrestException catch (e) {
   338→      AppLogger.error('Get exams failed', error: e);
   339→      throw ServerException(
   340→        message: e.message,
   341→        statusCode: int.tryParse(e.code ?? '') ?? 500,
   342→        data: e.details,
   343→      );
   344→    } catch (e) {
   345→      AppLogger.error('Get exams unexpected error', error: e);
   346→      throw ServerException(
   347→        message: 'Failed to get exams: $e',
   348→        statusCode: 500,
   349→      );
   350→    }
   351→  }

   414→  @override
   415→  Future<ExamModel> cloneExam(String examId) async {
   416→    try {
   417→      // Fetch the original exam with details
   418→      final original = await getExamWithDetails(examId);
   419→
   420→      final now = DateTime.now();
   421→      final cloneData = {
   422→        'school_id': original.schoolId,
   423→        'created_by': original.createdBy,
   424→        'title': '${original.title} (Copy)',
   425→        'description': original.description,
   426→        'subject_id': original.subjectId,
   427→        'class_id': original.classId,
   428→        'academic_session_id': original.academicSessionId,
   429→        'exam_type': original.examType,
   430→        'status': 'draft',
   431→        'start_time': now.add(const Duration(days: 7)).toIso8601String(),
   432→        'end_time': now.add(const Duration(days: 7, hours: 2)).toIso8601String(),
   433→        'time_limit_minutes': original.timeLimitMinutes,
   434→        'total_marks': original.totalMarks,
   435→        'pass_mark': original.passMark,
   436→        'pass_mark_type': original.passMarkType,
   437→        'instructions': original.instructions,
   438→        'allowed_attempts': original.allowedAttempts,
   439→        'negative_marking_enabled': original.negativeMarkingEnabled,
   440→        'negative_mark_value': original.negativeMarkValue,
   441→        'grace_period_minutes': original.gracePeriodMinutes,
   442→        'auto_submit': original.autoSubmit,
   443→        'randomize_questions': original.randomizeQuestions,
   444→        'randomize_options': original.randomizeOptions,
   445→        'show_results': original.showResults,
   446→        'show_correct_answers': original.showCorrectAnswers,
   447→        'show_explanations': original.showExplanations,
   448→        'is_template': original.isTemplate,
   449→        'template_id': original.id,
   450→        'max_students': original.maxStudents,
   451→        'ip_restriction': original.ipRestriction,
   452→        'require_full_screen': original.requireFullScreen,
   453→        'allow_resume': original.allowResume,
   454→        'browser_lockdown': original.browserLockdown,
   455→        'metadata': original.metadata,
   456→      };
   457→
   458→      final clonedExam = await createExam(cloneData);
   459→
   460→      // Clone sections
   461→      for (final section in original.sections) {
   462→        await _supabaseClient.from(_examSectionsTable).insert({
   463→          'exam_id': clonedExam.id,
   464→          'title': section.title,
   465→          'description': section.description,
   466→          'instructions': section.instructions,
   467→          'sort_order': section.sortOrder,
   468→          'time_limit_minutes': section.timeLimitMinutes,
   469→          'randomize_questions': section.randomizeQuestions,
   470→        });
   471→      }
   472→
   473→      // Clone questions
   474→      for (final question in original.questions) {
   475→        await _supabaseClient.from(_examQuestionsTable).insert({
   476→          'exam_id': clonedExam.id,
   477→          'section_id': question.sectionId,
   478→          'question_id': question.questionId,
   479→          'sort_order': question.sortOrder,
   480→          'marks': question.marks,
   481→          'negative_marks': question.negativeMarks,
   482→          'is_compulsory': question.isCompulsory,
   483→        });
   484→      }
   485→
   486→      return clonedExam;
   487→    } on ServerException {
   488→      rethrow;
   489→    } on sb.PostgrestException catch (e) {
   490→      AppLogger.error('Clone exam failed', error: e);
   491→      throw ServerException(
   492→        message: e.message,
   493→        statusCode: int.tryParse(e.code ?? '') ?? 500,
   494→        data: e.details,
   495→      );
   496→    } catch (e) {
   497→      AppLogger.error('Clone exam unexpected error', error: e);
   498→      throw ServerException(
   499→        message: 'Failed to clone exam: $e',
   500→        statusCode: 500,
   501→      );
   502→    }
   503→  }

   572→        await _supabaseClient
   573→            .from(_examQuestionsTable)

   579→      AppLogger.error('Reorder questions failed', error: e);
   580→      throw ServerException(
   581→        message: e.message,

   726→  @override
   727→  Future<ExamResultModel> submitAttempt(
   728→    String attemptId, {
   729→    String submissionType = 'manual',
   730→  }) async {
   731→    try {
   732→      final now = DateTime.now();
   733→
   734→      // Fetch the attempt
   735→      final attemptResponse = await _supabaseClient
   736→          .from(_examAttemptsTable)
   737→          .select()
   738→          .eq('id', attemptId)
   739→          .single();
   740→
   741→      final attempt = ExamAttemptModel.fromJson(attemptResponse);
   742→
   743→      // PERF: Select only needed columns instead of all columns
   744→      // Note: Answer fetching for a single attempt is bounded by exam question count
   745→      // (typically 20-100 questions), so no pagination needed.
   746→      final answersResponse = await _supabaseClient
   747→          .from(_studentAnswersTable)
   748→          .select('id, attempt_id, question_id, answer_data, is_correct, marks_awarded, marks_deducted, is_flagged, answered_at, graded_at, graded_by, teacher_comment')
   749→          .eq('attempt_id', attemptId);
   750→
   751→      final answers = answersResponse
   752→          .map<StudentAnswerModel>((a) => StudentAnswerModel.fromJson(a))
   753→          .toList();
   754→
   755→      // Calculate total marks
   756→      double totalMarks = 0;
   757→      for (final answer in answers) {
   758→        totalMarks += answer.marksAwarded - answer.marksDeducted;
   759→      }
   760→
   761→      // Calculate time spent
   762→      final startedAt = attempt.startedAt;
   763→      final timeSpent = now.difference(startedAt).inSeconds;
   764→
   765→      // Determine if all objective questions are graded
   766→      final hasUngradedSubjective = answers.any(
   767→        (a) => a.isCorrect == null && a.marksAwarded == 0,
   768→      );
   769→      final gradingStatus =
   770→          hasUngradedSubjective ? 'partially_graded' : 'auto_graded';
   771→
   772→      // Update the attempt
   773→      await _supabaseClient
   774→          .from(_examAttemptsTable)
   775→          .update({
   776→            'status': submissionType == 'auto_submit' ? 'auto_submitted' : 'submitted',
   777→            'submitted_at': now.toIso8601String(),
   778→            'submission_type': submissionType,
   779→            'time_spent_seconds': timeSpent,
   780→            'total_marks': totalMarks,
   781→            'grading_status': gradingStatus,
   782→            'last_activity_at': now.toIso8601String(),
   783→          })
   784→          .eq('id', attemptId);
   785→
   786→      // Deactivate the session
   787→      await _supabaseClient
   788→          .from(_examSessionsTable)
   789→          .update({
   790→            'is_active': false,
   791→            'connection_status': 'disconnected',
   792→          })
   793→          .eq('attempt_id', attemptId);
   794→
   795→      // Calculate score percentage
   796→      // Fetch exam total marks
   797→      final examResponse = await _supabaseClient
   798→          .from(_examsTable)
   799→          .select('total_marks, pass_mark, pass_mark_type')
   800→          .eq('id', attempt.examId)
   801→          .single();
   802→
   803→      final examTotalMarks =
   804→          (examResponse['total_marks'] as num?)?.toDouble() ?? 0.0;
   805→      final passMark =
   806→          (examResponse['pass_mark'] as num?)?.toDouble() ?? 0.0;
   807→      final passMarkType =
   808→          examResponse['pass_mark_type'] as String? ?? 'percentage';
   809→
   810→      final scorePercentage = examTotalMarks > 0
   811→          ? (totalMarks / examTotalMarks) * 100
   812→          : 0.0;
   813→
   814→      final effectivePassMark = passMarkType == 'percentage'
   815→          ? passMark
   816→          : (examTotalMarks > 0 ? (passMark / examTotalMarks) * 100 : 0.0);
   817→
   818→      final isPassed = scorePercentage >= effectivePassMark;
   819→
   820→      // Create or update result
   821→      final existingResult = await _supabaseClient
   822→          .from(_examResultsTable)
   823→          .select('id')
   824→          .eq('attempt_id', attemptId)
   825→          .maybeSingle();
   826→
   827→      final resultData = {
   828→        'exam_id': attempt.examId,
   829→        'student_id': attempt.studentId,
   830→        'attempt_id': attemptId,
   831→        'total_marks': totalMarks,
   832→        'total_possible': examTotalMarks,
   833→        'score_percentage': scorePercentage,
   834→        'is_passed': isPassed,
   835→        'time_spent_seconds': timeSpent,
   836→        'grading_status': gradingStatus,
   837→        'is_released': false,
   838→      };
   839→
   840→      if (existingResult != null) {
   841→        await _supabaseClient
   842→            .from(_examResultsTable)
   843→            .update(resultData)
   844→            .eq('id', existingResult['id'] as String);
   845→
   846→        final updatedResult = await _supabaseClient
   847→            .from(_examResultsTable)
   848→            .select()
   849→            .eq('id', existingResult['id'] as String)
   850→            .single();
   851→
   852→        return ExamResultModel.fromJson(updatedResult);
   853→      } else {
   854→        final insertResponse = await _supabaseClient
   855→            .from(_examResultsTable)
   856→            .insert(resultData)
   857→            .select()
   858→            .single();
   859→
   860→        return ExamResultModel.fromJson(insertResponse);
   861→      }
   862→    } on sb.PostgrestException catch (e) {
   863→      AppLogger.error('Submit attempt failed', error: e);
   864→      throw ServerException(
   865→        message: e.message,
   866→        statusCode: int.tryParse(e.code ?? '') ?? 500,
   867→        data: e.details,
   868→      );
   869→    } catch (e) {
   870→      if (e is ServerException) rethrow;
   871→      AppLogger.error('Submit attempt unexpected error', error: e);
   872→      throw ServerException(
   873→        message: 'Failed to submit attempt: $e',
   874→        statusCode: 500,
   875→      );
   876→    }
   877→  }

  1367→  @override
  1368→  Future<ExamStatisticsModel> getExamStatistics(String examId) async {
  1369→    try {
  1370→      // PERF: Use DatabasePoolManager.executeMonitored for performance tracking
  1371→      // on the most expensive CBT queries
  1372→      final results = await DatabasePoolManager.executeMonitored(
  1373→        queryType: 'select',
  1374→        table: _examResultsTable,
  1375→        operation: 'get_exam_statistics',
  1376→        query: () => _supabaseClient
  1377→            .from(_examResultsTable)
  1378→            .select('score_percentage, is_passed, grading_status')
  1379→            .eq('exam_id', examId),
  1380→      );
  1381→
  1382→      // PERF: Only count IDs, don't fetch full rows
  1383→      final students = await _supabaseClient
  1384→          .from(_examStudentsTable)
  1385→          .select('id')
  1386→          .eq('exam_id', examId)
  1387→          .limit(PaginatedQueryMixin.maxPageSize); // Safety cap
  1388→
  1389→      // PERF: Select only columns needed for statistics calculation
  1390→      final attempts = await _supabaseClient
  1391→          .from(_examAttemptsTable)
  1392→          .select('status, time_spent_seconds')
  1393→          .eq('exam_id', examId)
  1394→          .limit(PaginatedQueryMixin.maxPageSize); // Safety cap
  1395→
  1396→      final totalStudents = students.length;
  1397→      final startedStudents = attempts
  1398→          .where((a) => a['status'] != 'not_started')
  1399→          .length;
  1400→      final completedStudents = attempts
  1401→          .where((a) =>
  1402→              a['status'] == 'submitted' ||
  1403→              a['status'] == 'auto_submitted' ||
  1404→              a['status'] == 'timed_out')
  1405→          .length;
  1406→      final submittedStudents = completedStudents;
  1407→
  1408→      // Calculate score statistics
  1409→      final scores = results
  1410→          .map<double>((r) => (r['score_percentage'] as num?)?.toDouble() ?? 0.0)
  1411→          .toList();
  1412→
  1413→      final averageScore = scores.isNotEmpty
  1414→          ? scores.reduce((a, b) => a + b) / scores.length
  1415→          : 0.0;
  1416→      final highestScore = scores.isNotEmpty
  1417→          ? scores.reduce((a, b) => a > b ? a : b)
  1418→          : 0.0;
  1419→      final lowestScore = scores.isNotEmpty
  1420→          ? scores.reduce((a, b) => a < b ? a : b)
  1421→          : 0.0;
  1422→
  1423→      // Calculate median
  1424→      double medianScore = 0.0;
  1425→      if (scores.isNotEmpty) {
  1426→        final sorted = List<double>.from(scores)..sort();
  1427→        final mid = sorted.length ~/ 2;
  1428→        medianScore = sorted.length.isEven
  1429→            ? (sorted[mid - 1] + sorted[mid]) / 2
  1430→            : sorted[mid];
  1431→      }
  1432→
  1433→      // Calculate pass rate
  1434→      final passedCount = results
  1435→          .where((r) => r['is_passed'] == true)
  1436→          .length;
  1437→      final passRate =
  1438→          results.isNotEmpty ? (passedCount / results.length) * 100 : 0.0;
  1439→
  1440→      // Average time spent
  1441→      final timeSpentList = attempts
  1442→          .where((a) => a['time_spent_seconds'] != null)
  1443→          .map<int>((a) => a['time_spent_seconds'] as int)
  1444→          .toList();
  1445→      final averageTimeSpent = timeSpentList.isNotEmpty
  1446→          ? timeSpentList.reduce((a, b) => a + b) ~/ timeSpentList.length
  1447→          : 0;
  1448→
  1449→      // Grading completion
  1450→      final gradedCount = results
  1451→          .where((r) =>
  1452→              r['grading_status'] == 'auto_graded' ||
  1453→              r['grading_status'] == 'fully_graded')
  1454→          .length;
  1455→      final gradingCompletion = results.isNotEmpty
  1456→          ? (gradedCount / results.length) * 100
  1457→          : 0.0;
  1458→
  1459→      return ExamStatisticsModel(
  1460→        examId: examId,
  1461→        totalStudents: totalStudents,
  1462→        startedStudents: startedStudents,
  1463→        completedStudents: completedStudents,
  1464→        submittedStudents: submittedStudents,
  1465→        averageScore: averageScore,
  1466→        highestScore: highestScore,
  1467→        lowestScore: lowestScore,
  1468→        medianScore: medianScore,
  1469→        passRate: passRate,
  1470→        questionsByCorrectRate: {}, // Requires per-question analysis
  1471→        averageTimeSpentSeconds: averageTimeSpent,
  1472→        gradingCompletionPercentage: gradingCompletion,
  1473→      );
  1474→    } on sb.PostgrestException catch (e) {
  1475→      AppLogger.error('Get exam statistics failed', error: e);
  1476→      throw ServerException(
  1477→        message: e.message,
  1478→        statusCode: int.tryParse(e.code ?? '') ?? 500,
  1479→        data: e.details,
  1480→      );
  1481→    } catch (e) {
  1482→      AppLogger.error('Get exam statistics unexpected error', error: e);
  1483→      throw ServerException(
  1484→        message: 'Failed to get exam statistics: $e',
  1485→        statusCode: 500,
  1486→      );
  1487→    }
  1488→  }

  1490→  @override
  1491→  Future<LiveExamStatsModel> getLiveExamStats(String examId) async {
  1492→    try {
  1493→      // PERF: Use DatabasePoolManager monitoring for the live stats query
  1494→      // which is polled every 5-10 seconds during active exams
  1495→      final activeSessions = await DatabasePoolManager.executeMonitored(
  1496→        queryType: 'select',
  1497→        table: _examSessionsTable,
  1498→        operation: 'get_live_stats_sessions',
  1499→        query: () => _supabaseClient
  1500→            .from(_examSessionsTable)
  1501→            .select('id, questions_answered, questions_flagged')
  1502→            .eq('exam_id', examId)
  1503→            .eq('is_active', true)
  1504→            .limit(PaginatedQueryMixin.maxPageSize),
  1505→      );
  1506→
  1507→      // Count students (bounded by exam size)
  1508→      final students = await _supabaseClient
  1509→          .from(_examStudentsTable)
  1510→          .select('id')
  1511→          .eq('exam_id', examId)
  1512→          .limit(PaginatedQueryMixin.maxPageSize);
  1513→
  1514→      // PERF: Only select columns needed for counting
  1515→      final submittedAttempts = await _supabaseClient
  1516→          .from(_examAttemptsTable)
  1517→          .select('id, status')
  1518→          .eq('exam_id', examId)
  1519→          .inFilter('status', ['submitted', 'auto_submitted', 'timed_out']);
  1520→
  1521→      // Recent submissions (last 10)
  1522→      final recentSubmissions = await _supabaseClient
  1523→          .from(_examAttemptsTable)
  1524→          .select()
  1525→          .eq('exam_id', examId)
  1526→          .inFilter('status', ['submitted', 'auto_submitted', 'timed_out'])
  1527→          .order('submitted_at', ascending: false)
  1528→          .limit(10);
  1529→
  1530→      // Recent monitoring events (last 20)
  1531→      final recentEvents = await _supabaseClient
  1532→          .from(_monitoringLogsTable)
  1533→          .select()
  1534→          .eq('exam_id', examId)
  1535→          .order('created_at', ascending: false)
  1536→          .limit(20);
  1537→
  1538→      final totalEligible = students.length;
  1539→      final activeNow = activeSessions.length;
  1540→      final completed = submittedAttempts.length;
  1541→      final notStarted = totalEligible - activeNow - completed;
  1542→
  1543→      // Calculate average progress
  1544→      double averageProgress = 0;
  1545→      if (activeSessions.isNotEmpty) {
  1546→        final totalQuestionsAnswered = activeSessions.fold<int>(
  1547→          0,
  1548→          (sum, s) =>
  1549→              sum + ((s['questions_answered'] as int?) ?? 0),
  1550→        );
  1551→        // Estimate total questions per exam
  1552→        averageProgress = totalEligible > 0
  1553→            ? totalQuestionsAnswered / (activeNow * 10) * 100
  1554→            : 0;
  1555→        averageProgress = averageProgress.clamp(0.0, 100.0);
  1556→      }
  1557→
  1558→      return LiveExamStatsModel(
  1559→        examId: examId,
  1560→        totalEligible: totalEligible,
  1561→        activeNow: activeNow,
  1562→        completed: completed,
  1563→        notStarted: notStarted < 0 ? 0 : notStarted,
  1564→        averageProgress: averageProgress,
  1565→        recentSubmissions: recentSubmissions
  1566→            .map<ExamAttemptModel>((a) => ExamAttemptModel.fromJson(a))
  1567→            .toList(),
  1568→        activeSessions: activeSessions
  1569→            .map<ExamSessionModel>((s) => ExamSessionModel.fromJson(s))
  1570→            .toList(),
  1571→        recentMonitoringEvents: recentEvents
  1572→            .map<MonitoringLogModel>((e) => MonitoringLogModel.fromJson(e))
  1573→            .toList(),
  1574→      );
  1575→    } on sb.PostgrestException catch (e) {
  1576→      AppLogger.error('Get live exam stats failed', error: e);
  1577→      throw ServerException(
  1578→        message: e.message,
  1579→        statusCode: int.tryParse(e.code ?? '') ?? 500,
  1580→        data: e.details,
  1581→      );
  1582→    } catch (e) {
  1583→      AppLogger.error('Get live exam stats unexpected error', error: e);
  1584→      throw ServerException(
  1585→        message: 'Failed to get live exam stats: $e',
  1586→        statusCode: 500,
  1587→      );
  1588→    }
  1589→  }
