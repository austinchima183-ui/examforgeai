   113→  // ── Exception mapping helper ───────────────────────────────────────────
   114→  Never _mapPostgrestException(sb.PostgrestException e) {
   115→    AppLogger.error('Postgrest error: ${e.message}', error: e);
   116→    switch (e.code) {
   117→      case 'PGRST116':
   118→        throw NotFoundException(message: e.message);
   119→      case '23505':
   120→        throw ServerException(
   121→          message: 'A record with this data already exists.',
   122→          statusCode: 409,
   123→        );
   124→      case '23503':
   125→        throw ServerException(
   126→          message: 'Referenced record not found.',
   127→          statusCode: 404,
   128→        );
   129→      case '42501':
   130→        throw ForbiddenException(
   131→          message: 'You do not have permission for this action.',
   132→        );
   133→      default:
   134→        throw ServerException(
   135→          message: e.message,
   136→          statusCode: e.statusCode ?? 500,
   137→        );
   138→    }
   139→  }

   240→  @override
   241→  Future<List<LessonPlanModel>> getLessonPlans(
   242→    Map<String, dynamic> filters,
   243→  ) async {
   244→    try {
   245→      AppLogger.info('Fetching lesson plans with filters: $filters');
   246→      var query = _supabase.from(_lessonPlansTable).select();
   247→
   248→      // Search filter
   249→      if (filters['searchQuery'] != null) {
   250→        query = query.ilike('title', '%${filters['searchQuery']}%');
   251→      }
   252→
   253→      // Subject filter
   254→      if (filters['subject_id'] != null) {
   255→        query = query.eq('subject_id', filters['subject_id']);
   256→      }
   257→
   258→      // Class filter
   259→      if (filters['class_id'] != null) {
   260→        query = query.eq('class_id', filters['class_id']);
   261→      }
   262→
   263→      // Published filter
   264→      if (filters['is_published'] != null) {
   265→        query = query.eq('is_published', filters['is_published']);
   266→      }
   267→
   268→      // Archived filter
   269→      if (filters['is_archived'] != null) {
   270→        query = query.eq('is_archived', filters['is_archived']);
   271→      }
   272→
   273→      // Tags overlap filter
   274→      if (filters['tags'] != null) {
   275→        query = query.overlaps('tags', filters['tags']);
   276→      }
   277→
   278→      // Ordering
   279→      query = query.order('created_at', ascending: false);
   280→
   281→      // Pagination
   282→      final page = filters['page'] as int? ?? 1;
   283→      final perPage = filters['perPage'] as int? ?? 20;
   284→      query = query.range((page - 1) * perPage, page * perPage - 1);
   285→
   286→      final response = await query;
   287→      AppLogger.info('Fetched ${response.length} lesson plans');
   288→      return response.map(LessonPlanModel.fromJson).toList();
   289→    } on sb.PostgrestException catch (e) {
   290→      _mapPostgrestException(e);
   291→    } catch (e) {
   292→      AppLogger.error('Failed to fetch lesson plans', error: e);
   293→      throw ServerException(message: e.toString(), statusCode: 500);
   294→    }
   295→  }
